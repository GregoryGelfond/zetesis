1//! The byte dimension of the shared meter, independent of work charges.
2
3use std::mem::size_of;
4
5/// Named simultaneous capacity, with its peak published into the caller's
6/// existing receipt. This dimension owns only the current byte count; it
7/// neither duplicates the peak nor owns the caller's configured ceiling.
8///
9/// Control observations and work charges remain at the caller's established
10/// boundaries. Reservations include actual vector capacity, not allocator
11/// metadata or RSS. Headers and overlapping owners must be observed explicitly.
12#[derive(Debug)]
13#[repr(transparent)]
14pub struct MeterBytes {
15    current: u128,
16}
17
18/// A byte observation or reservation could not complete under its ceiling.
19/// Callers retain their named resource and typed failure when translating it.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum MeterBytesError {
22    /// The complete requested or observed capacity exceeds its inclusive limit.
23    Limit {
24        /// Complete capacity, rather than only the latest increment.
25        observed: u128,
26        /// The caller's unchanged configured ceiling.
27        limit: u128,
28    },
29    /// Capacity addition or release subtraction cannot be represented.
30    Overflow,
31    /// Fallible vector reservation failed.
32    Allocation,
33}
34
35impl MeterBytes {
36    /// Start at zero capacity without polling, charging or allocating.
37    #[must_use]
38    pub const fn new() -> Self {
39        Self { current: 0 }
40    }
41
42    /// Check a caller-supplied byte requirement against its inclusive ceiling.
43    ///
44    /// This pure check records no current or peak capacity. The caller retains
45    /// the distinction between a requested envelope and an actual observation,
46    /// along with its allocation, publication and control boundaries. Zero fits
47    /// every ceiling. No addition, conversion, polling or allocation occurs.
48    ///
49    /// # Errors
50    /// Returns only [`MeterBytesError::Limit`], containing the exact requirement
51    /// and ceiling, when the requirement exceeds that ceiling.
52    pub fn check(total: u128, limit: u128) -> Result<(), MeterBytesError> {
53        if total > limit {
54            Err(MeterBytesError::Limit {
55                observed: total,
56                limit,
57            })
58        } else {
59            Ok(())
60        }
61    }
62
63    /// Record a caller-supplied byte requirement in its existing peak receipt,
64    /// then check that requirement against the inclusive ceiling.
65    ///
66    /// The caller determines whether `total` describes a requested envelope or
67    /// actual capacity. A refused observation remains in the peak. An earlier
68    /// peak above a newly lowered ceiling does not itself refuse a smaller
69    /// current requirement. This operation owns no current counter or other
70    /// state, and performs no allocation, arithmetic addition or control poll.
71    ///
72    /// # Errors
73    /// Returns only [`MeterBytesError::Limit`], containing the exact current
74    /// requirement and ceiling. The peak is updated before returning it.
75    pub fn observe_peak(total: u128, limit: u128, peak: &mut u128) -> Result<(), MeterBytesError> {
76        observe_peak_as(total, limit, peak, std::convert::identity)
77    }
78
79    /// Observe a machine-sized capacity in its existing machine-sized peak.
80    ///
81    /// This has the same publication-before-refusal contract as
82    /// [`Self::observe_peak`], without widening or copying the caller's receipt.
83    /// Current capacity, allocation and control observations remain caller-owned.
84    ///
85    /// # Errors
86    /// Returns only [`MeterBytesError::Limit`], with the exact current requirement
87    /// and ceiling widened to `u128`. The peak is updated before returning it.
88    pub fn observe_peak_usize(
89        total: usize,
90        limit: usize,
91        peak: &mut usize,
92    ) -> Result<(), MeterBytesError> {
93        observe_peak_as(total, limit, peak, |value| value as u128)
94    }
95
96    /// Observe a u64 byte requirement in the caller's existing u64 peak.
97    ///
98    /// Like [`Self::observe_peak`], this publishes the maximum before checking
99    /// the current requirement. It neither widens nor copies the stored peak;
100    /// only the error's exact diagnostic values are widened to u128.
101    ///
102    /// # Errors
103    /// Returns only [`MeterBytesError::Limit`]. A refused observation remains
104    /// recorded; an older peak above the ceiling does not refuse a fitting total.
105    pub fn observe_peak_u64(total: u64, limit: u64, peak: &mut u64) -> Result<(), MeterBytesError> {
106        observe_peak_as(total, limit, peak, u128::from)
107    }
108
109    /// Record an actual total and its peak before checking the ceiling.
110    /// A refused observation is retained. A lower total can represent storage
111    /// already dropped; it never erases an earlier peak.
112    ///
113    /// # Errors
114    /// Returns the complete observed capacity when it exceeds `limit`.
115    pub fn observe(
116        &mut self,
117        total: u128,
118        limit: u128,
119        peak: &mut u128,
120    ) -> Result<(), MeterBytesError> {
121        self.current = total;
122        Self::observe_peak(total, limit, peak)
123    }
124
125    /// Reserve one empty vector after admitting its requested payload.
126    /// Requested capacity that is refused before allocation changes neither
127    /// current bytes nor peak. Actual allocator capacity is observed before
128    /// its ceiling check; on refusal the vector drops but that attempted
129    /// observation remains recorded. No control poll or work charge is added.
130    ///
131    /// # Errors
132    /// Reports unrepresentable addition, requested or observed excess, or a
133    /// failed allocation. No vector is returned on any refusal.
134    pub fn reserve<T>(
135        &mut self,
136        count: usize,
137        limit: u128,
138        peak: &mut u128,
139    ) -> Result<Vec<T>, MeterBytesError> {
140        let requested = self
141            .current
142            .checked_add(count as u128 * size_of::<T>() as u128)
143            .ok_or(MeterBytesError::Overflow)?;
144        Self::check(requested, limit)?;
145        let mut values = Vec::new();
146        values
147            .try_reserve_exact(count)
148            .map_err(|_| MeterBytesError::Allocation)?;
149        let actual = self
150            .current
151            .checked_add(values.capacity() as u128 * size_of::<T>() as u128)
152            .ok_or(MeterBytesError::Overflow)?;
153        self.observe(actual, limit, peak)?;
154        Ok(values)
155    }
156
157    /// Drop a vector, then remove its actual payload from current capacity.
158    /// The vector must belong to this accounted total. No peak, work or control
159    /// observation is changed, and there is no automatic accounting on drop.
160    ///
161    /// # Errors
162    /// Returns an arithmetic refusal after dropping the vector if its payload
163    /// exceeds the currently accounted capacity; current bytes stay unchanged.
164    pub fn release<T>(&mut self, value: Vec<T>) -> Result<(), MeterBytesError> {
165        let bytes = value.capacity() as u128 * size_of::<T>() as u128;
166        drop(value);
167        self.current = self
168            .current
169            .checked_sub(bytes)
170            .ok_or(MeterBytesError::Overflow)?;
171        Ok(())
172    }
173}
174
175impl Default for MeterBytes {
176    /// The same zero-capacity, effect-free construction as [`Self::new`].
177    fn default() -> Self {
178        Self::new()
179    }
180}
181
182// The adapters supply only lossless diagnostic widening. Publication mutates
183// the actual receipt at its own width; no converted account is kept or copied.
184fn observe_peak_as<T: Copy + Ord>(
185    total: T,
186    limit: T,
187    peak: &mut T,
188    widen: impl Fn(T) -> u128,
189) -> Result<(), MeterBytesError> {
190    *peak = (*peak).max(total);
191    MeterBytes::check(widen(total), widen(limit))
192}
