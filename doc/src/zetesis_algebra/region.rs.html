1//! Regions of a candidate space and the outcome of narrowing them.
2//!
3//! A region decides some atoms in, some out, and leaves the rest open. Its
4//! candidates are the sets agreeing with those decisions, the `Region` of
5//! `Search.lean` indexed over a root's atoms. [`crate::walk::Walk`] covers
6//! the region by applying the caller's narrowing and splitting open atoms.
7
8use std::mem::size_of;
9
10use crate::{Stop, words::Words};
11
12/// A region of candidates: every atom held, cut or open. Its decisions are two
13/// atom bitmasks, one for held and one for cut; a narrowing that carries
14/// knowledge from the region's parent applies only the decisions the parent
15/// had not yet seen, by diffing this region's decided mask against a snapshot.
16#[derive(Debug)]
17pub struct Region {
18    /// Bit set iff the atom is held. Disjoint from `cut` word for word, and
19    /// bits at `[atoms, 64*words)` are always zero and are not atoms.
20    held: Words<u64, Box<[u64]>>,
21    /// Bit set iff the atom is cut.
22    cut: Words<u64, Box<[u64]>>,
23    /// The atom universe; each mask is `words(atoms)` words long.
24    atoms: usize,
25}
26
27impl Clone for Region {
28    fn clone(&self) -> Self {
29        Self {
30            held: self.held.clone(),
31            cut: self.cut.clone(),
32            atoms: self.atoms,
33        }
34    }
35
36    /// Reset a region in its existing mask buffers when the word counts
37    /// match. Frozen probes reuse one such region for successive patterns.
38    fn clone_from(&mut self, source: &Self) {
39        self.held.clone_from(&source.held);
40        self.cut.clone_from(&source.cut);
41        self.atoms = source.atoms;
42    }
43}
44
45/// The number of 64-bit words a mask over `atoms` atoms needs.
46const fn words(atoms: usize) -> usize {
47    atoms.div_ceil(64)
48}
49
50/// Two regions are the same when they decide the same atoms the same way over
51/// the same universe. Decision order and a walk's split preference are history
52/// and policy, not the interval's identity.
53impl PartialEq for Region {
54    fn eq(&self, other: &Self) -> bool {
55        self.atoms == other.atoms && self.held == other.held && self.cut == other.cut
56    }
57}
58impl Eq for Region {}

60impl Region {
61    /// Header and owned vector capacities in bytes, including unused capacity.
62    /// Excludes allocator bookkeeping; this is not a process-memory estimate.
63    #[must_use]
64    pub fn retained_bytes(&self) -> u128 {
65        size_of::<Self>() as u128
66            + (self.held.len() + self.cut.len()) as u128 * size_of::<u64>() as u128
67    }
68
69    /// The region in which every atom is open: every candidate lies in it.
70    #[must_use]
71    pub fn all_open(atoms: usize) -> Self {
72        Self::with_masks(
73            atoms,
74            &mut (),
75            |(), length, _| Ok::<_, std::convert::Infallible>(vec![0; length].into_boxed_slice()),
76            |(), _| Ok(()),
77        )
78        .unwrap_or_else(|never| match never {})
79    }
80
81    /// The all-open region, with fallible reservation of its two masks.
82    /// No partially initialized region escapes an allocation refusal.
83    ///
84    /// # Errors
85    /// Returns [`Stop::Allocation`] if either mask cannot be reserved.
86    pub fn try_all_open(atoms: usize) -> Result<Self, Stop> {
87        Self::try_all_open_with(atoms, &mut (), &mut |(), _| Ok(()), &mut |(), _| Ok(()))
88    }
89
90    /// Construct an all-open region with fallible work and storage admission.
91    ///
92    /// `before` admits each logical word operation before it executes. Each
93    /// mask initializes a three-word vector header, two words per appended zero
94    /// (the value and length), and its two-word boxed-slice header. The final
95    /// atom-count field costs one. A complete construction therefore costs
96    /// `11 + 4 * atoms.div_ceil(64)`; a refusal retains only its executed prefix.
97    /// Allocator internals are outside this logical operation count.
98    ///
99    /// `admit` observes this constructor's total named storage, excluding any
100    /// caller baseline: the output record, completed mask payloads, and the
101    /// current vector header and requested or actual capacity. It runs before
102    /// each reservation, after actual capacity is known, and before publishing
103    /// the final region. No partial region escapes either callback's refusal.
104    ///
105    /// # Errors
106    /// Preserves either callback's error. Failed mask reservation returns
107    /// [`Stop::Allocation`] converted into `E`.
108    pub fn try_all_open_with<C, E: From<Stop>>(
109        atoms: usize,
110        context: &mut C,
111        before: &mut impl FnMut(&mut C, u64) -> Result<(), E>,
112        admit: &mut impl FnMut(&mut C, u128) -> Result<(), E>,
113    ) -> Result<Self, E> {
114        Self::with_masks(
115            atoms,
116            &mut (context, before, admit),
117            |(context, before, admit), length, held_bytes| {
118                let headers = size_of::<Self>() as u128 + size_of::<Vec<u64>>() as u128;
119                let requested = length as u128 * size_of::<u64>() as u128;
120                admit(context, headers + held_bytes + requested)?;
121                before(context, 3)?;
122                let mut mask = Vec::new();
123                mask.try_reserve_exact(length)
124                    .map_err(|_| E::from(Stop::Allocation))?;
125                let actual = mask.capacity() as u128 * size_of::<u64>() as u128;
126                admit(context, headers + held_bytes + actual)?;
127                for _ in 0..length {
128                    before(context, 2)?;
129                    mask.push(0);
130                }
131                before(context, 2)?;
132                Ok(mask.into_boxed_slice())
133            },
134            |(context, before, admit), retained| {
135                admit(context, retained)?;
136                before(context, 1)
137            },
138        )
139    }
140
141    fn with_masks<C, E>(
142        atoms: usize,
143        context: &mut C,
144        mut allocate: impl FnMut(&mut C, usize, u128) -> Result<Box<[u64]>, E>,
145        mut publish: impl FnMut(&mut C, u128) -> Result<(), E>,
146    ) -> Result<Self, E> {
147        let held = allocate(context, words(atoms), 0)?;
148        let held_bytes = held.len() as u128 * size_of::<u64>() as u128;
149        let cut = allocate(context, words(atoms), held_bytes)?;
150        let retained =
151            size_of::<Self>() as u128 + held_bytes + cut.len() as u128 * size_of::<u64>() as u128;
152        publish(context, retained)?;
153        Ok(Self {
154            held: held.into(),
155            cut: cut.into(),
156            atoms,
157        })
158    }
159    /// The number of atoms the region decides over.
160    #[must_use]
161    pub fn len(&self) -> usize {
162        self.atoms
163    }
164    /// Whether the region decides over no atom at all.
165    #[must_use]
166    pub fn is_empty(&self) -> bool {
167        self.atoms == 0
168    }
169    /// Hold an atom in every candidate; `false` when the atom is cut or
170    /// outside the region.
171    pub fn hold(&mut self, atom: usize) -> bool {
172        self.decide(atom, true)
173    }
174    /// Cut an atom from every candidate; `false` when the atom is held or
175    /// outside the region.
176    pub fn cut(&mut self, atom: usize) -> bool {
177        self.decide(atom, false)
178    }
179    fn decide(&mut self, atom: usize, value: bool) -> bool {
180        if atom >= self.atoms {
181            return false;
182        }
183        let (word, bit) = Words::<u64>::locate(atom);
184        if self.held[word] & bit != 0 {
185            return value;
186        }
187        if self.cut[word] & bit != 0 {
188            return !value;
189        }
190        if value {
191            self.held[word] |= bit;
192        } else {
193            self.cut[word] |= bit;
194        }
195        true
196    }
197    /// The atom's decision: held, cut, or open; `None` outside the region too.
198    #[must_use]
199    pub fn decision(&self, atom: usize) -> Option<bool> {
200        if atom >= self.atoms {
201            return None;
202        }
203        let (word, bit) = Words::<u64>::locate(atom);
204        if self.held[word] & bit != 0 {
205            Some(true)
206        } else if self.cut[word] & bit != 0 {
207            Some(false)
208        } else {
209            None
210        }
211    }
212    /// Whether every candidate holds the atom.
213    #[must_use]
214    pub fn is_held(&self, atom: usize) -> bool {
215        self.decision(atom) == Some(true)
216    }
217    /// Whether no candidate holds the atom.
218    #[must_use]
219    pub fn is_cut(&self, atom: usize) -> bool {
220        self.decision(atom) == Some(false)
221    }
222    /// Whether the atom is open: some candidates hold it and some do not.
223    #[must_use]
224    pub fn is_open(&self, atom: usize) -> bool {
225        atom < self.atoms && {
226            let (word, bit) = Words::<u64>::locate(atom);
227            (self.held[word] | self.cut[word]) & bit == 0
228        }
229    }
230    /// The highest open atom, the one a split decides when no atom is
231    /// preferred.
232    #[must_use]
233    pub fn highest_open(&self) -> Option<usize> {
234        for word in (0..self.mask_words()).rev() {
235            let open = self.open_word(word, u64::MAX);
236            if open != 0 {
237                return Some(word * 64 + Words::<u64>::last_bit(open));
238            }
239        }
240        None
241    }
242    /// The atoms every candidate holds, ascending.
243    pub fn held(&self) -> impl Iterator<Item = usize> + '_ {
244        self.held.positions().filter(|&atom| atom < self.atoms)
245    }
246    /// The open atoms, ascending.
247    pub fn open(&self) -> impl Iterator<Item = usize> + '_ {
248        self.open_words(std::iter::repeat(u64::MAX))
249    }
250    /// The number of 64-bit words in each of the region's atom masks:
251    /// `self.len().div_ceil(64)`.
252    #[must_use]
253    pub fn mask_words(&self) -> usize {
254        self.held.len()
255    }
256    /// The open atoms of mask word `word` that `selected` selects, as bit
257    /// `atom % 64` for atom `atom` of that word; the bits past the last atom
258    /// are clear. Constant time, for scans that read a word at a time.
259    ///
260    /// # Panics
261    /// Panics if `word` is not below [`Self::mask_words`].
262    #[must_use]
263    pub fn open_word(&self, word: usize, selected: u64) -> u64 {
264        let open = !(self.held[word] | self.cut[word]) & selected;
265        // The unused high bits of the top partial word are not atoms, and
266        // would otherwise read as open.
267        let valid = self.atoms - word * 64;
268        if valid < 64 {
269            open & ((1u64 << valid) - 1)
270        } else {
271            open
272        }
273    }
274    /// Open atoms selected by a borrowed bitmask, ascending. Missing words
275    /// select nothing; excess words and unused tail bits are ignored. Bit
276    /// `atom % 64` in word `atom / 64` selects that atom.
277    ///
278    /// Visits at most `min(selected.len(), self.len().div_ceil(64))` words
279    /// plus the returned atoms, with constant space and no allocation.
280    pub fn open_in<'a>(&'a self, selected: &'a [u64]) -> impl Iterator<Item = usize> + 'a {
281        self.open_words(selected.iter().copied())
282    }
283    fn open_words<'a>(
284        &'a self,
285        selected: impl Iterator<Item = u64> + 'a,
286    ) -> impl Iterator<Item = usize> + 'a {
287        (0..self.mask_words())
288            .zip(selected)
289            .flat_map(move |(word, selected)| {
290                let base = word * 64;
291                Words::<u64>::bits(self.open_word(word, selected)).map(move |bit| base + bit)
292            })
293    }
294    /// The atoms this region decides that are not set in `seen`, each paired
295    /// with whether it is held, ascending. `seen` is a decided-mask snapshot
296    /// over the same atoms; words missing from `seen` count as unseen. Reports
297    /// only real atoms, never a tail bit.
298    pub fn decided_since<'a>(
299        &'a self,
300        seen: &'a [u64],
301    ) -> impl Iterator<Item = (usize, bool)> + 'a {
302        (0..self.held.len()).flat_map(move |word| {
303            let seen_word = seen.get(word).copied().unwrap_or(0);
304            let fresh = (self.held[word] | self.cut[word]) & !seen_word;
305            let held = self.held[word];
306            let base = word * 64;
307            Words::<u64>::bits(fresh)
308                .map(move |bit| (base + bit, held & (1u64 << bit) != 0))
309                .filter(move |&(atom, _)| atom < self.atoms)
310        })
311    }
312    /// Copy this region's decided mask (`held | cut`) into the available prefix
313    /// of `dst`. A complete snapshot needs `self.len().div_ceil(64)` words.
314    /// A shorter destination receives only its prefix; any excess destination
315    /// words remain unchanged. An empty destination is valid. Missing snapshot
316    /// words count as unseen when passed to [`Self::decided_since`].
317    pub fn snapshot_decided(&self, dst: &mut [u64]) {
318        for (word, (&held, &cut)) in dst.iter_mut().zip(self.held.iter().zip(self.cut.iter())) {
319            *word = held | cut;
320        }
321    }
322    /// Every decided atom with whether it is held, ascending.
323    pub fn decided(&self) -> impl Iterator<Item = (usize, bool)> + '_ {
324        self.decided_since(&[])
325    }
326    /// Borrow the held and cut masks, in that order, without copying.
327    /// Both slices contain `self.len().div_ceil(64)` words. Bit `atom % 64`
328    /// in word `atom / 64` represents that atom. The masks are disjoint;
329    /// unused high bits of the last word are zero.
330    ///
331    /// Keeping both polarities permits exact comparison with an earlier region.
332    /// Their union alone cannot distinguish removing or reversing a decision.
333    /// The slices remain borrowed from this region; retaining a snapshot requires
334    /// caller-owned storage and its own resource accounting.
335    #[must_use]
336    pub fn decision_words(&self) -> (&[u64], &[u64]) {
337        (&self.held, &self.cut)
338    }
339    /// The two regions an open atom splits this one into: cut, then held.
340    ///
341    /// # Panics
342    /// Panics unless `atom` is one of the region's open atoms.
343    #[must_use]
344    pub fn split(mut self, atom: usize) -> (Self, Self) {
345        // A region is split on one of its OPEN atoms: an out-of-range or
346        // already-decided atom fails fast, since deciding it would either touch
347        // a non-atom bit or leave a narrowing's knowledge out of step with the
348        // region's decision (a reader's seen-mask already covers a decided atom).
349        assert!(
350            self.is_open(atom),
351            "a region is split on one of its open atoms"
352        );
353        let (word, bit) = Words::<u64>::locate(atom);
354        // Clone once for the held child; reuse this region as the cut child
355        // rather than cloning it a second time.
356        let mut held = self.clone();
357        held.held[word] |= bit;
358        self.cut[word] |= bit;
359        (self, held)
360    }
361}

363/// A region with its walk policy and caller-owned lineage state.
364///
365/// Preference is not part of the region's interval identity. It is accepted
366/// only for an open atom, checked again after narrowing, and reset in both
367/// children. Pending entries retain this policy when handed to another schedule.
368#[derive(Clone, Debug)]
369pub struct Entry<S = ()> {
370    /// The interval this entry still represents.
371    pub region: Region,
372    /// Knowledge or other state carried along this entry's lineage.
373    pub state: S,
374    preferred: Option<usize>,
375}
376
377impl<S> Entry<S> {
378    /// Start an entry with no preference, using the highest open atom by default.
379    #[must_use]
380    pub const fn new(region: Region, state: S) -> Self {
381        Self {
382            region,
383            state,
384            preferred: None,
385        }
386    }
387
388    /// Borrow the semantic interval and lineage state without discarding policy.
389    pub fn parts_mut(&mut self) -> (&mut Region, &mut S) {
390        (&mut self.region, &mut self.state)
391    }
392
393    /// Consume a terminal entry, releasing its no-longer-needed walk policy.
394    /// Pending entries should be transferred as entries instead.
395    #[must_use]
396    pub fn into_parts(self) -> (Region, S) {
397        (self.region, self.state)
398    }
399
400    /// Prefer an open atom for the next split; a decided atom is not retained.
401    pub fn prefer(&mut self, atom: usize) {
402        self.preferred = Some(atom).filter(|&atom| self.region.is_open(atom));
403    }
404
405    /// The preferred atom if still open, otherwise the highest open atom.
406    #[must_use]
407    pub fn split_atom(&self) -> Option<usize> {
408        self.preferred
409            .filter(|&atom| self.region.is_open(atom))
410            .or_else(|| self.region.highest_open())
411    }
412
413    /// Split an admitted open atom, carrying each returned state in its child.
414    ///
415    /// The callback returns the cut child's state followed by the held child's.
416    /// It lets each schedule preserve its existing cloning or allocation-reuse
417    /// policy. To retain the parent on a pending-capacity refusal, the schedule
418    /// admits that capacity before this consuming operation. This method does
419    /// not reserve queue storage. Both children start without a preference.
420    ///
421    /// # Panics
422    /// Panics unless `atom` is one of the region's open atoms.
423    #[must_use]
424    pub fn split_with(self, atom: usize, states: impl FnOnce(S) -> (S, S)) -> (Self, Self) {
425        let (cut, held) = self.region.split(atom);
426        let (cut_state, held_state) = states(self.state);
427        (Self::new(cut, cut_state), Self::new(held, held_state))
428    }
429}

431/// The outcome of narrowing one region.
432#[derive(Clone, Copy, Debug, PartialEq, Eq)]
433pub enum Narrowing {
434    /// No accepted candidate lies in the region.
435    Refuted,
436    /// The fixed point, and whether the narrowing decided an atom.
437    Fixed {
438        /// Whether any atom was decided by the narrowing.
439        changed: bool,
440    },
441}
442
443/// Progress from one application of a region subject, not joint agreement.
444#[derive(Clone, Copy, Debug, PartialEq, Eq)]
445pub enum SubjectStep {
446    /// The applicable inspection completed without a new region decision.
447    Unchanged,
448    /// At least one sound decision was applied; further closure may be needed.
449    Changed,
450    /// The region contains no member of the subject's class.
451    Refuted,
452}
453
454impl SubjectStep {
455    /// Convert a subject whose complete local closure has been established.
456    ///
457    /// The caller must establish that closure separately: `Changed` alone is
458    /// not evidence of a fixed point. Partial late passes must not use this
459    /// conversion; their joint driver first re-establishes eager agreement.
460    #[must_use]
461    pub fn narrowing(self) -> Narrowing {
462        match self {
463            Self::Unchanged => Narrowing::Fixed { changed: false },
464            Self::Changed => Narrowing::Fixed { changed: true },
465            Self::Refuted => Narrowing::Refuted,
466        }
467    }
468}
