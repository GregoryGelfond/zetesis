1//! Shared entry classification and a scalar walk covering a candidate region.
2//!
3//! A region decides some atoms in, some out, and leaves the rest open; its
4//! candidates are the sets that agree with its decisions, the `Region` of
5//! `Search.lean` indexed over the atoms of a root. The traversal walks the
6//! coverage tree of that chapter
7//! with a stack: a region is narrowed by a caller-supplied narrowing, and
8//! is then refuted, a leaf when every atom is decided, split on the open
9//! atom its narrowing preferred or else its highest open atom, or counted
10//! when its narrowing decided nothing beyond the split and the caller
11//! counts such regions. By default the cut branch is visited before the held
12//! one; [`Walk::with_held_first`] reverses that preference. With the default
13//! branch order and no atom preference, leaves follow a binary counter over
14//! the atoms with atom zero as its low bit (`Region.split_partition` and
15//! `Region.split_disjoint` of `Search.lean`). The root is never counted;
16//! it splits only when narrowing leaves it open and unrefuted.
17//!
18//! The narrowing is the caller's: on the formula route it is the theory's
19//! readings. The traversal only requires that a narrowing never
20//! removes an accepted candidate from the region and never refutes a region
21//! holding one; each visit then keeps every accepted candidate of the root
22//! in exactly one region still to visit, visited as a leaf, or counted.
23//!
24//! Termination: a visit pops one region and pushes at most two, each with
25//! one more atom decided, so the number of regions ever pushed is bounded by
26//! twice the number of candidates of the root.

38/// When a region is counted instead of split.
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40pub enum Counting {
41    /// Every region is split until decided: the traversal visits only leaves.
42    Never,
43    /// A region below the root whose narrowing decided nothing is counted:
44    /// its open atoms are offered to a flat counter by the caller.
45    Unchanged,
46}
47
48/// The next action of an inspected, narrowed entry.
49///
50/// A leaf is a fully decided region, not an accepted answer set. A counted
51/// region is handed to the caller's counter; it is not declared semantically
52/// accepted by this classification either.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum Step {
55    /// The narrowing proved the region empty for its class.
56    Refuted,
57    /// Every atom is decided.
58    Leaf,
59    /// An unchanged non-root region is left to the caller's counter.
60    Counted,
61    /// An open atom partitions the region into cut and held children.
62    Split(usize),
63}
64
65/// Inspect an entry after its caller has completed and accounted narrowing.
66///
67/// Refutation takes precedence over every shape of region. Otherwise a decided
68/// region is a leaf, even when unchanged counting is selected. An unchanged
69/// open region is counted only below the root; all other open regions split on
70/// the entry's existing preference or highest-open fallback. `Counting::Never`
71/// cannot return `Step::Counted`.
72///
73/// This is the allocation-free classification shared by the scalar, parallel,
74/// slate and published-query schedules. It reads no theory, invokes no semantic
75/// checker, consumes no entry, and performs no resource admission. The caller
76/// retains its narrowing, receipt-publication and split-admission order. A
77/// returned split atom is open in the inspected entry; mutating that interval
78/// afterwards requires inspection again before consuming the split.
79#[must_use]
80pub fn inspect<S>(entry: &Entry<S>, narrowing: Narrowing, counting: Counting, root: bool) -> Step {
81    let Narrowing::Fixed { changed } = narrowing else {
82        return Step::Refuted;
83    };
84    let Some(atom) = entry.split_atom() else {
85        return Step::Leaf;
86    };
87    if counting == Counting::Unchanged && !root && !changed {
88        Step::Counted
89    } else {
90        Step::Split(atom)
91    }
92}

94/// One region the traversal hands to the caller, with what the caller
95/// carries alongside it.
96#[derive(Clone, Debug, PartialEq, Eq)]
97pub enum Visit<S = ()> {
98    /// Every atom is decided: the region is one candidate.
99    Leaf(Region, S),
100    /// The region's open atoms are the caller's to count.
101    Counted(Region, S),
102}
103
104/// The regions a traversal visited, by outcome.
105#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
106pub struct RegionStatistics {
107    /// Regions popped and narrowed, the root included.
108    pub regions: usize,
109    /// Regions whose narrowing refuted them.
110    pub refuted: usize,
111    /// Regions visited as leaves: every atom decided, one candidate each.
112    pub leaves: usize,
113    /// Regions handed to the caller to count.
114    pub counted: usize,
115}
116
117impl RegionStatistics {
118    /// The regions split since `before`: every region visited that was
119    /// neither refuted nor a leaf nor counted.
120    #[must_use]
121    pub fn splits_since(self, before: Self) -> usize {
122        (self.regions - before.regions)
123            - (self.refuted - before.refuted)
124            - (self.leaves - before.leaves)
125            - (self.counted - before.counted)
126    }
127}

129/// The coverage tree of a root region, walked depth first. Each region
130/// carries the caller's state `S`, what the narrowing knows about the
131/// region; a split clones it into both children, so that a child starts
132/// from its parent's knowledge and the regions share nothing.
133#[derive(Clone, Debug)]
134pub struct Walk<S = ()> {
135    /// The regions still to visit with their state, the next on top.
136    regions: Vec<Entry<S>>,
137    counting: Counting,
138    /// Visit the held child first at future splits; queued entries keep their order.
139    held_first: bool,
140    /// The root has not been visited yet; unchanged counting cannot apply to it.
141    root: bool,
142    /// The caller narrowed the root before the traversal began, so it is
143    /// not narrowed again.
144    narrowed_root: bool,
145    statistics: RegionStatistics,
146}
147
148impl Walk {
149    /// A traversal of every candidate of the root, carrying nothing.
150    #[must_use]
151    pub fn new(root: Region, counting: Counting) -> Self {
152        Self::with_state(root, counting, ())
153    }
154}

156impl<S> Walk<S> {
157    /// A traversal of every candidate of the root, carrying `state` with the
158    /// root. The caller's copy operation supplies each later sibling's state.
159    #[must_use]
160    pub fn with_state(root: Region, counting: Counting, state: S) -> Self {
161        Self::with_entry(Entry::new(root, state), counting)
162    }
163
164    /// Begin with an owned pending entry, retaining its knowledge and preference.
165    #[must_use]
166    pub fn with_entry(root: Entry<S>, counting: Counting) -> Self {
167        Self {
168            regions: vec![root],
169            counting,
170            held_first: false,
171            root: true,
172            narrowed_root: false,
173            statistics: self::RegionStatistics::default(),
174        }
175    }
176
177    /// A traversal of every candidate of a root the caller has narrowed
178    /// already. The root is classified without another narrowing: it is a
179    /// leaf when fully decided and otherwise split. Every region below it
180    /// is narrowed by the caller's narrowing.
181    #[must_use]
182    pub fn with_narrowed_root(root: Entry<S>, counting: Counting) -> Self {
183        Self {
184            narrowed_root: true,
185            ..Self::with_entry(root, counting)
186        }
187    }
188
189    /// Visit the held child before the cut child at subsequent splits.
190    ///
191    /// The default visits cut first. Both children retain the same independent
192    /// copies of their parent's state and cover the same disjoint partition.
193    /// This changes no narrowing, counting, work limit or failure behavior.
194    /// Already queued regions keep their order; choose this before the first
195    /// pull for a consistent held-first traversal. The extra state is one flag.
196    #[must_use]
197    pub const fn with_held_first(mut self) -> Self {
198        self.held_first = true;
199        self
200    }
201
202    /// The regions visited so far, by outcome.
203    #[must_use]
204    pub const fn statistics(&self) -> RegionStatistics {
205        self.statistics
206    }
207
208    /// The regions still to visit.
209    #[must_use]
210    pub fn pending(&self) -> usize {
211        self.regions.len()
212    }
213
214    /// Surrender the regions still to visit with their state, the oldest
215    /// first, so the last is the one the next visit would narrow. They and the
216    /// regions already visited partition the root, so a walker that visits
217    /// every surrendered region, by any narrowing the traversal could have
218    /// used, covers the rest of the tree. A region whose narrowing stopped is
219    /// among them, as is a narrowed parent whose split reservation failed.
220    /// The traversal is left with none, so its next visit ends it, and its
221    /// statistics keep what it visited.
222    pub fn surrender(&mut self) -> Vec<Entry<S>> {
223        std::mem::take(&mut self.regions)
224    }
225
226    /// Narrow regions until one is a leaf or is counted; `None` once every
227    /// region has been visited. The narrowing's failure type is the caller's;
228    /// a failed reservation for the split is reported as `Stop::Allocation`.
229    ///
230    /// # Errors
231    /// A stopped narrowing stops the traversal with its region kept on the
232    /// stack, so a later call narrows the same region again. A failed split
233    /// reservation also keeps the narrowed parent and its state, before any
234    /// child is formed. The completed narrowing remains in the visit receipt;
235    /// retrying or surrendering does not refund it.
236    pub fn next<E: From<Stop>>(
237        &mut self,
238        narrow: impl FnMut(&mut Entry<S>) -> Result<Narrowing, E>,
239    ) -> Result<Option<Visit<S>>, E>
240    where
241        S: Clone,
242    {
243        self.next_with_reservation(narrow, |pending, additional| {
244            pending
245                .try_reserve(additional)
246                .map_err(|_| E::from(Stop::Allocation))
247        })
248    }
249
250    /// Traverse with a fallible independent copy of the inherited state.
251    ///
252    /// Narrowing and copying borrow one context in execution order. Copying
253    /// runs after queue reservation and before consuming or splitting the
254    /// parent. The successful copy belongs to the sibling visited second; the
255    /// first child retains the original state, for either branch preference.
256    /// The copy must preserve its source and return equivalent independent
257    /// lineage state. This method adds no polling or resource policy.
258    ///
259    /// # Errors
260    /// Narrowing, reservation and copy failures keep the unresolved parent on
261    /// the stack. A failed copy retains completed narrowing and the attempted
262    /// split receipt, like a failed reservation. The root and pre-narrowed
263    /// flags are restored, and no child is formed or published. Context-side
264    /// receipts from a refused copy are not rolled back. Retrying an ordinary
265    /// root repeats its narrowing; a pre-narrowed root remains pre-narrowed.
266    pub fn next_with_copy<C, E: From<Stop>>(
267        &mut self,
268        context: &mut C,
269        narrow: impl FnMut(&mut C, &mut Entry<S>) -> Result<Narrowing, E>,
270        mut copy: impl FnMut(&mut C, &S) -> Result<S, E>,
271    ) -> Result<Option<Visit<S>>, E> {
272        self.next_with_split(
273            context,
274            narrow,
275            |_, pending, additional| {
276                pending
277                    .try_reserve(additional)
278                    .map_err(|_| E::from(Stop::Allocation))
279            },
280            |context, entry, atom, held_first| {
281                let copied = match copy(context, &entry.state) {
282                    Ok(copied) => copied,
283                    Err(error) => return Err((error, entry)),
284                };
285                Ok(entry.split_with(atom, |state| {
286                    if held_first {
287                        (copied, state)
288                    } else {
289                        (state, copied)
290                    }
291                }))
292            },
293        )
294    }

296    // The schedule admits pending capacity before consuming a split's parent.
297    // The production callback reserves real Vec capacity; keeping that boundary
298    // explicit also lets the stop contract be tested without a global allocator.
299    fn next_with_reservation<E: From<Stop>>(
300        &mut self,
301        mut narrow: impl FnMut(&mut Entry<S>) -> Result<Narrowing, E>,
302        mut reserve: impl FnMut(&mut Vec<Entry<S>>, usize) -> Result<(), E>,
303    ) -> Result<Option<Visit<S>>, E>
304    where
305        S: Clone,
306    {
307        self.next_with_split(
308            &mut (),
309            |(), entry| narrow(entry),
310            |(), pending, additional| reserve(pending, additional),
311            |(), entry, atom, held_first| {
312                // Preserve the original order: Region::split allocates its
313                // masks before the caller's Clone implementation is invoked.
314                Ok(entry.split_with(atom, |state| {
315                    if held_first {
316                        (state.clone(), state)
317                    } else {
318                        let second = state.clone();
319                        (state, second)
320                    }
321                }))
322            },
323        )
324    }

326    // Both copy contracts share this traversal. The private adapter returns the
327    // exact unconsumed parent on refusal; only successful splits publish children.
328    fn next_with_split<C, E: From<Stop>>(
329        &mut self,
330        context: &mut C,
331        mut narrow: impl FnMut(&mut C, &mut Entry<S>) -> Result<Narrowing, E>,
332        mut reserve: impl FnMut(&mut C, &mut Vec<Entry<S>>, usize) -> Result<(), E>,
333        mut split: impl FnMut(
334            &mut C,
335            Entry<S>,
336            usize,
337            bool,
338        ) -> Result<(Entry<S>, Entry<S>), (E, Entry<S>)>,
339    ) -> Result<Option<Visit<S>>, E> {
340        loop {
341            let Some(mut entry) = self.regions.pop() else {
342                return Ok(None);
343            };
344            let narrowed_root = std::mem::take(&mut self.narrowed_root);
345            let narrowing = if narrowed_root {
346                Narrowing::Fixed { changed: true }
347            } else {
348                match narrow(context, &mut entry) {
349                    Ok(narrowing) => narrowing,
350                    Err(stop) => {
351                        self.regions.push(entry);
352                        return Err(stop);
353                    }
354                }
355            };
356            self.statistics.regions += 1;
357            let root = std::mem::replace(&mut self.root, false);
358            let atom = match inspect(&entry, narrowing, self.counting, root) {
359                Step::Refuted => {
360                    self.statistics.refuted += 1;
361                    continue;
362                }
363                Step::Leaf => {
364                    self.statistics.leaves += 1;
365                    let (region, state) = entry.into_parts();
366                    return Ok(Some(Visit::Leaf(region, state)));
367                }
368                Step::Counted => {
369                    self.statistics.counted += 1;
370                    let (region, state) = entry.into_parts();
371                    return Ok(Some(Visit::Counted(region, state)));
372                }
373                Step::Split(atom) => atom,
374            };
375            if let Err(stop) = reserve(context, &mut self.regions, 2) {
376                // pop vacated one slot, so restoring its parent cannot allocate,
377                // even when the reservation left the capacity unchanged.
378                self.regions.push(entry);
379                self.root = root;
380                self.narrowed_root = narrowed_root;
381                return Err(stop);
382            }
383            let (cut, held) = match split(context, entry, atom, self.held_first) {
384                Ok(children) => children,
385                Err((error, parent)) => {
386                    self.regions.push(parent);
387                    self.root = root;
388                    self.narrowed_root = narrowed_root;
389                    return Err(error);
390                }
391            };
392            let (first, second) = if self.held_first {
393                (held, cut)
394            } else {
395                (cut, held)
396            };
397            self.regions.push(second);
398            self.regions.push(first);
399        }
400    }
