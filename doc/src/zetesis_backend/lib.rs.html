1//! The backend vocabulary every zetesis tool shares: how a program runs.
2//!
3//! A request is [`Backend::Cpu`], the default, or [`Backend::Gpu`] with an
4//! optional [`GpuApi`]. `Gpu(None)` asks for the platform's native API; the one
5//! rule that makes it concrete is [`Backend::resolved_api`], applied where a GPU
6//! is opened, so code past that point never holds an unresolved request.
7//! Choosing one GPU among several is a device choice, not a backend, and is not
8//! part of this vocabulary.
9//!
10//! The vocabulary does not depend on whether GPU support is compiled: a build
11//! without it still reads every value and refuses a GPU request with its reason,
12//! never with "unknown value". Spellings are stable — they appear in arguments,
13//! configuration and retained records — and a spelling zetesis no longer accepts
14//! is refused with what to use instead ([`ParseBackendError`]).
15//!
16//! The CPU backend's default parallelism is here too ([`default_threads`]), so
17//! every tool that runs or measures zetesis means the same count by `auto`.
18#![forbid(unsafe_code)]
19

24/// The CPU backend's default thread count, which `--threads auto` means: the
25/// host's available parallelism, or one when availability is unknown. It reads
26/// the host each time it is called.
27#[must_use]
28pub fn default_threads() -> NonZeroUsize {
29    std::thread::available_parallelism().unwrap_or(NonZeroUsize::MIN)
30}
31
32/// Read a thread count as a command line spells it: `auto` for
33/// [`default_threads`], or a positive count.
34///
35/// # Errors
36/// Refuses any other spelling, naming the accepted forms.
37pub fn parse_threads(value: &str) -> Result<NonZeroUsize, String> {
38    if value == "auto" {
39        Ok(default_threads())
40    } else {
41        value
42            .parse()
43            .map_err(|_| "expected auto or a positive thread count".to_owned())
44    }
45}

47/// Read the common memory allowance: bytes or a whole number with a binary unit.
48///
49/// # Errors
50/// Refuses fractional, negative, unknown-unit or overflowing quantities.
51pub fn parse_memory(value: &str) -> Result<u64, String> {
52    let boundary = value
53        .find(|character: char| !character.is_ascii_digit())
54        .unwrap_or(value.len());
55    let (number, suffix) = value.split_at(boundary);
56    let factor = match suffix {
57        "" | "B" => Some(1),
58        "KiB" => Some(1 << 10),
59        "MiB" => Some(1 << 20),
60        "GiB" => Some(1 << 30),
61        "TiB" => Some(1 << 40),
62        _ => None,
63    };
64    number.parse::<u64>().ok()
65        .zip(factor)
66        .and_then(|(amount, factor)| amount.checked_mul(factor))
67        .ok_or_else(|| {
68            "expected bytes or a whole number with B, KiB, MiB, GiB or TiB; value must fit u64 bytes"
69                .to_owned()
70        })
71}

73/// How a program runs: its execution backend.
74#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
75pub enum Backend {
76    /// CPU execution, the default.
77    #[default]
78    Cpu,
79    /// A GPU through the named API, or through the platform's native API when
80    /// none is named.
81    Gpu(Option<GpuApi>),
82}
83
84/// A GPU compute API zetesis targets.
85#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
86pub enum GpuApi {
87    /// Metal, on macOS.
88    Metal,
89    /// Vulkan, on Linux and on other platforms with a Vulkan driver.
90    Vulkan,
91}
92
93impl GpuApi {
94    /// The platform's native API, defined for every target: Metal on Apple
95    /// targets and Vulkan on all others, the APIs zetesis compiles there.
96    #[must_use]
97    pub const fn native() -> Self {
98        if cfg!(target_vendor = "apple") {
99            Self::Metal
100        } else {
101            Self::Vulkan
102        }
103    }
104
105    /// Stable spelling, as `--backend` takes it.
106    #[must_use]
107    pub const fn label(self) -> &'static str {
108        match self {
109            Self::Metal => "metal",
110            Self::Vulkan => "vulkan",
111        }
112    }
113
114    /// The API's proper name, for messages.
115    #[must_use]
116    pub const fn name(self) -> &'static str {
117        match self {
118            Self::Metal => "Metal",
119            Self::Vulkan => "Vulkan",
120        }
121    }
122}
123
124impl Backend {
125    /// Every value, in the order help and documentation list them.
126    pub const ALL: [Self; 4] = [
127        Self::Cpu,
128        Self::Gpu(None),
129        Self::Gpu(Some(GpuApi::Metal)),
130        Self::Gpu(Some(GpuApi::Vulkan)),
131    ];
132
133    /// Stable spelling for arguments, configuration and records.
134    #[must_use]
135    pub const fn label(self) -> &'static str {
136        match self {
137            Self::Cpu => "cpu",
138            Self::Gpu(None) => "gpu",
139            Self::Gpu(Some(api)) => api.label(),
140        }
141    }
142
143    /// One line of help for the value, as command help shows it.
144    #[must_use]
145    pub const fn help(self) -> &'static str {
146        match self {
147            Self::Cpu => "CPU execution (the default)",
148            Self::Gpu(None) => {
149                "A GPU through the platform's native API: Metal on macOS, Vulkan elsewhere"
150            }
151            Self::Gpu(Some(GpuApi::Metal)) => "A GPU through Metal (macOS)",
152            Self::Gpu(Some(GpuApi::Vulkan)) => {
153                "A GPU through Vulkan (Linux; also other platforms with a Vulkan driver, unqualified there)"
154            }
155        }
156    }
157
158    /// Whether the request is for a GPU.
159    #[must_use]
160    pub const fn is_gpu(self) -> bool {
161        matches!(self, Self::Gpu(_))
162    }
163
164    /// The API a GPU request opens on this platform — the named one, or the
165    /// native one when none is named — and `None` for the CPU. This is the one
166    /// resolution rule; each consumer applies it at the boundary where it opens
167    /// a GPU.
168    #[must_use]
169    pub const fn resolved_api(self) -> Option<GpuApi> {
170        match self {
171            Self::Cpu => None,
172            Self::Gpu(Some(api)) => Some(api),
173            Self::Gpu(None) => Some(GpuApi::native()),
174        }
175    }
176}

190/// Spellings zetesis once accepted, each with what to use instead.
191const RETIRED: [(&str, &str); 4] = [
192    (
193        "auto",
194        "CPU is the default; `auto` returns when zetesis can choose a backend for each problem",
195    ),
196    (
197        "dx12",
198        "zetesis targets Metal on macOS and Vulkan on Linux; use gpu, metal or vulkan",
199    ),
200    (
201        "gl",
202        "zetesis targets Metal on macOS and Vulkan on Linux; use gpu, metal or vulkan",
203    ),
204    (
205        "nvidia",
206        "use gpu or vulkan; choosing one GPU among several will come with NVIDIA qualification",
207    ),

210/// A spelling that is not a backend.
211#[derive(Clone, Debug, PartialEq, Eq)]
212pub struct ParseBackendError {
213    spelling: String,
214    retired: Option<&'static str>,
215}
216
217impl ParseBackendError {
218    /// The spelling that was given.
219    #[must_use]
220    pub fn spelling(&self) -> &str {
221        &self.spelling
222    }
223}
224
225impl fmt::Display for ParseBackendError {
226    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
227        match self.retired {
228            Some(reason) => write!(
229                formatter,
230                "`{}` is no longer a backend: {reason}",
231                self.spelling
232            ),
233            None => write!(
234                formatter,
235                "`{}` is not a backend; use cpu, gpu, metal or vulkan",
236                self.spelling
237            ),
238        }
239    }
240}
241
242impl std::error::Error for ParseBackendError {}
243
244impl FromStr for Backend {
245    type Err = ParseBackendError;
246
247    fn from_str(spelling: &str) -> Result<Self, Self::Err> {
248        Self::ALL
249            .into_iter()
250            .find(|backend| backend.label() == spelling)
251            .ok_or_else(|| ParseBackendError {
252                spelling: spelling.to_owned(),
253                retired: RETIRED
254                    .iter()
255                    .find(|(old, _)| *old == spelling)
256                    .map(|(_, reason)| *reason),
257            })
258    }
259}

261/// Command-line parsing for [`Backend`] with clap's standard behaviour — help
262/// lists the four values with their meaning, an invalid value is refused with
263/// the possible values and a suggestion for a near miss, and `ignore_case` is
264/// honoured — except that a retired spelling is answered with what to use
265/// instead rather than rejected bare.
266#[cfg(feature = "clap")]
267#[derive(Clone, Copy, Debug, Default)]
268pub struct BackendParser;
269
270#[cfg(feature = "clap")]
271impl clap::builder::TypedValueParser for BackendParser {
272    type Value = Backend;
273
274    fn parse_ref(
275        &self,
276        command: &clap::Command,
277        argument: Option<&clap::Arg>,
278        value: &std::ffi::OsStr,
279    ) -> Result<Backend, clap::Error> {
280        let spelling = clap::builder::PossibleValuesParser::new(possible_values())
281            .parse_ref(command, argument, value)
282            .map_err(|refusal| retired(command, argument, value).unwrap_or(refusal))?;
283        // The standard parser accepted a label, in any ASCII case when the
284        // argument ignores case; the labels themselves are lowercase ASCII.
285        spelling
286            .to_ascii_lowercase()
287            .parse()
288            .map_err(|error| invalid(command, argument, &error))
289    }
290
291    fn possible_values(
292        &self,
293    ) -> Option<Box<dyn Iterator<Item = clap::builder::PossibleValue> + '_>> {
294        Some(Box::new(possible_values()))
295    }
296}

298/// Command-line parsing for an argument that names a GPU API alone, such as
299/// the API a physical test stage qualifies: the API spellings, each with the
300/// help of the backend it selects.
301#[cfg(feature = "clap")]
302impl clap::ValueEnum for GpuApi {
303    fn value_variants<'a>() -> &'a [Self] {
304        &[Self::Metal, Self::Vulkan]
305    }
306
307    fn to_possible_value(&self) -> Option<clap::builder::PossibleValue> {
308        let backend = Backend::Gpu(Some(*self));
309        Some(clap::builder::PossibleValue::new(self.label()).help(backend.help()))
310    }
311}

313#[cfg(feature = "clap")]
314fn possible_values() -> impl Iterator<Item = clap::builder::PossibleValue> {
315    Backend::ALL
316        .into_iter()
317        .map(|backend| clap::builder::PossibleValue::new(backend.label()).help(backend.help()))
318}

320/// The explanation of a retired spelling, as a command-line refusal; `None`
321/// for any other value.
322#[cfg(feature = "clap")]
323fn retired(
324    command: &clap::Command,
325    argument: Option<&clap::Arg>,
326    value: &std::ffi::OsStr,
327) -> Option<clap::Error> {
328    let spelling = value.to_str()?;
329    let ignore_case = argument.is_some_and(clap::Arg::is_ignore_case_set);
330    let (_, reason) = RETIRED.iter().find(|(old, _)| {
331        if ignore_case {
332            old.eq_ignore_ascii_case(spelling)
333        } else {
334            *old == spelling
335        }
336    })?;
337    let error = ParseBackendError {
338        spelling: spelling.to_owned(),
339        retired: Some(reason),
340    };
341    Some(invalid(command, argument, &error))
342}
343
344#[cfg(feature = "clap")]
345fn invalid(
346    command: &clap::Command,
347    argument: Option<&clap::Arg>,
348    error: &ParseBackendError,
349) -> clap::Error {
350    let name = argument.map_or_else(|| "the backend".to_owned(), ToString::to_string);
351    clap::Error::raw(
352        clap::error::ErrorKind::InvalidValue,
353        format!(
354            "invalid value '{}' for '{name}': {error}\n",
355            error.spelling()
356        ),
357    )
358    .with_cmd(command)
359}
