7/// `SplitMix64` rejection sampling, version one. Distinct tagged streams never

11impl Draw {

18    /// Draw an ordinal below the exclusive positive bound without modulo bias.
19    ///
20    /// # Panics
21    /// Panics for an empty draw range.
22    pub fn below(&mut self, upper: u64) -> u64 {
23        assert!(upper != 0, "a draw range must be nonempty");
24        let minimum = upper.wrapping_neg() % upper;
25        loop {
26            self.0 = self.0.wrapping_add(0x9e37_79b9_7f4a_7c15);
27            let next = mix(self.0);
28            if next >= minimum {
29                return next % upper;
30            }
31        }
32    }
