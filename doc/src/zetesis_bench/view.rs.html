12pub(super) fn comparison(
13    comparison: &Comparison,
14    json: bool,
15    layout: Layout,
16    output: &mut impl io::Write,
17) -> Result<(), Error> {
18    if json {
19        serde_json::to_writer_pretty(&mut *output, comparison).map_err(Error::Json)?;
20        return writeln!(output).map_err(Error::Io);
21    }
22    let rows = comparison_rows(comparison);
23    table(
24        "Corpus benchmark — timed medians",
25        &[
26            "Report",
27            "Workload",
28            "Profile",
29            "zetesis ms",
30            "clingo ms",
31            "zetesis MiB",
32            "clingo MiB",
33            "Outcome",
34            "Detail",
35        ],
36        rows,
37    )?
38    .write(output, layout)
39    .map_err(Error::Io)?;
40    comparison_failures(comparison, layout, output)?;
41    let rows = comparison
42        .labels
43        .iter()
44        .map(|label| {
45            let provenance = &comparison.provenance[label];
46            Row::new([
47                label.clone(),
48                provenance.passed.to_string(),
49                provenance.accounted.to_string(),
50                provenance.native_sha256.clone(),
51                // A clingo-free campaign sealed no clingo.
52                provenance
53                    .reference_sha256
54                    .clone()
55                    .unwrap_or_else(|| "not run".into()),
56            ])
57            .conclusion()
58        })
59        .collect();
60    table(
61        "Recorded campaign identity",
62        &[
63            "Report",
64            "All passed",
65            "Accounted",
66            "zetesis SHA-256",
67            "clingo SHA-256",
68        ],
69        rows,
70    )?
71    .write(output, layout)
72    .map_err(Error::Io)?;
73    for scoreboard in &comparison.scoreboards {
74        scoreboard_tables(comparison, scoreboard, layout, output)?;
75    }
76    Ok(())
77}

79/// One report's standing against clingo on one profile: every cell where both
80/// passed, fastest ratio first, and their peak memory when it was measured.
81fn scoreboard_tables(
82    comparison: &Comparison,
83    scoreboard: &Scoreboard,
84    layout: Layout,
85    output: &mut impl io::Write,
86) -> Result<(), Error> {
87    let profile = comparison
88        .cells
89        .first()
90        .map(|cell| profile_name(&cell.profiles[scoreboard.profile].profile))
91        .unwrap_or_default();
92    let standing = if scoreboard.compared == 0 {
93        "no cell where both passed".to_owned()
94    } else {
95        format!(
96            "faster on {} of {} cells where both passed",
97            scoreboard.wins, scoreboard.compared
98        )
99    };
100    let rows = scoreboard
101        .verdicts
102        .iter()
103        .map(|verdict| {
104            Row::new([
105                verdict.cell.clone(),
106                milliseconds(verdict.native_ns),
107                milliseconds(verdict.reference_ns),
108                ratio(verdict.native_ns, verdict.reference_ns),
109                optional_milliseconds(verdict.native.grounding),
110                optional_milliseconds(verdict.native.proposal),
111                optional_milliseconds(verdict.native.membership),
112                optional_milliseconds(verdict.reference_grounding_ns),
113                optional_milliseconds(verdict.reference_solving_ns),
114            ])
115        })
116        .collect();
117    table(
118        &format!(
119            "Against clingo — report {}, profile {}: {profile}, search {}; {standing}",
120            scoreboard.report,
121            scoreboard.profile + 1,
122            scoreboard.method
123        ),
124        &[
125            "Cell",
126            "zetesis ms",
127            "clingo ms",
128            "zetesis/clingo",
129            "grounding ms",
130            "proposal ms",
131            "membership ms",
132            "clingo grounding ms",
133            "clingo solving ms",
134        ],
135        rows,
136    )?
137    .write(output, layout)
138    .map_err(Error::Io)?;
139    if !scoreboard.verdicts.iter().any(|verdict| {
140        verdict.native_peak_rss_bytes.is_some()
141            || verdict.reference_peak_rss_bytes.is_some()
142            || verdict.device_bytes.is_some()
143    }) {
144        return Ok(());
145    }
146    let rows = scoreboard
147        .verdicts
148        .iter()
149        .map(|verdict| {
150            Row::new([
151                verdict.cell.clone(),
152                bytes(verdict.native_peak_rss_bytes),
153                bytes(verdict.reference_peak_rss_bytes),
154                bytes(verdict.device_bytes),
155            ])
156        })
157        .collect();
158    table(
159        &format!(
160            "Peak memory against clingo — report {}, profile {}",
161            scoreboard.report,
162            scoreboard.profile + 1
163        ),
164        &["Cell", "zetesis MiB", "clingo MiB", "device MiB"],
165        rows,
166    )?
167    .write(output, layout)
168    .map_err(Error::Io)
169}

171fn comparison_rows(comparison: &Comparison) -> Vec<Row> {
172    comparison
173        .cells
174        .iter()
175        .flat_map(|cell| {
176            cell.profiles
177                .iter()
178                .enumerate()
179                .flat_map(move |(profile_index, profile)| {
180                    comparison.labels.iter().map(move |label| {
181                        let (native, memory, outcome, detail) = match profile.reports.get(label) {
182                            Some(Native::Passed(record)) => (
183                                milliseconds(record.timing.median_ns),
184                                bytes(record.peak_rss_bytes),
185                                "pass".to_owned(),
186                                String::new(),
187                            ),
188                            Some(Native::NotPassed { decisions, reasons }) => (
189                                "—".into(),
190                                "—".into(),
191                                decisions
192                                    .iter()
193                                    .map(|(decision, count)| format!("{decision}: {count}"))
194                                    .collect::<Vec<_>>()
195                                    .join(", "),
196                                reasons
197                                    .iter()
198                                    .map(|(reason, count)| format!("{reason} ×{count}"))
199                                    .collect::<Vec<_>>()
200                                    .join("; "),
201                            ),
202                            None => ("—".into(), "—".into(), "unavailable".into(), String::new()),
203                        };
204                        // Clingo's columns say "not run" where the campaign ran
205                        // without it, and "not timed" and "not measured" where
206                        // it ran clingo only to qualify the cells, apart from a
207                        // missing or failed value's dash.
208                        let provenance = comparison.provenance.get(label);
209                        let (clingo_ms, clingo_memory) = match cell.reference.get(label) {
210                            Some(record) => (
211                                milliseconds(record.timing.median_ns),
212                                bytes(record.peak_rss_bytes),
213                            ),
214                            None if provenance.is_some_and(|seen| seen.reference_timed) => {
215                                ("—".into(), "—".into())
216                            }
217                            None if provenance
218                                .is_some_and(|seen| seen.reference_sha256.is_some()) =>
219                            {
220                                ("not timed".into(), "not measured".into())
221                            }
222                            None => ("not run".into(), "not run".into()),
223                        };
224                        Row::new([
225                            label.clone(),
226                            cell.label.clone(),
227                            format!("{}: {}", profile_index + 1, profile_name(&profile.profile)),
228                            native,
229                            clingo_ms,
230                            memory,
231                            clingo_memory,
232                            outcome,
233                            detail,
234                        ])
235                    })
236                })
237        })
238        .collect()
239}

241fn comparison_failures(
242    comparison: &Comparison,
243    layout: Layout,
244    output: &mut impl io::Write,
245) -> Result<(), Error> {
246    let mut rows = Vec::new();
247    for cell in &comparison.cells {
248        for label in &comparison.labels {
249            if let Some(reasons) = cell.failure_reasons.get(label) {
250                for (reason, count) in reasons {
251                    rows.push(Row::new([
252                        label.clone(),
253                        cell.label.clone(),
254                        reason.clone(),
255                        count.to_string(),
256                    ]));
257                }
258            }
259        }
260    }
261    if rows.is_empty() {
262        return Ok(());
263    }
264    table(
265        "Recorded failure reasons — all scheduled phases",
266        &[
267            "Report",
268            "Workload",
269            "Producer, phase and reason",
270            "Positions",
271        ],
272        rows,
273    )?
274    .write(output, layout)
275    .map_err(Error::Io)
276}

278pub(super) fn table(title: &str, columns: &[&str], rows: Vec<Row>) -> Result<Table, Error> {
279    Table::new(
280        title,
281        columns
282            .iter()
283            .map(|label| {
284                Column::new(
285                    label,
286                    if label.ends_with(" ms")
287                        || label.ends_with(" MiB")
288                        || *label == "Checked"
289                        || *label == "zetesis/clingo"
290                    {
291                        Alignment::Right
292                    } else {
293                        Alignment::Left
294                    },
295                )
296            })
297            .collect(),
298        rows,
299    )
300    .map_err(Error::Table)
301}

303fn profile_name(profile: &serde_json::Value) -> String {
304    format!(
305        "{}/{} ({} threads)",
306        profile["backend"].as_str().unwrap_or("unavailable"),
307        profile["grounder"].as_str().unwrap_or("unavailable"),
308        profile
309            .get("threads")
310            .unwrap_or(&profile["workers"])
311            .as_u64()
312            .map_or_else(|| "?".into(), |workers| workers.to_string())
313    )
314}
315fn milliseconds(nanos: u64) -> String {
316    format!("{}.{:03}", nanos / 1_000_000, nanos % 1_000_000 / 1000)
317}
318fn optional_milliseconds(nanos: Option<u64>) -> String {
319    nanos.map_or_else(|| "—".into(), milliseconds)
320}
321/// `native / reference` to three decimals, rounded half up, by exact integer
322/// arithmetic; a reference of zero reads as one nanosecond.
323fn ratio(native: u64, reference: u64) -> String {
324    let reference = u128::from(reference.max(1));
325    let thousandths = (u128::from(native) * 1000 + reference / 2) / reference;
326    format!("{}.{:03}", thousandths / 1000, thousandths % 1000)
327}
328fn bytes(bytes: Option<u64>) -> String {
329    bytes.map_or_else(
330        || "—".into(),
331        |value| {
332            format!(
333                "{}.{:03}",
334                value / 1_048_576,
335                value % 1_048_576 * 1000 / 1_048_576
336            )
337        },
338    )
339}

341pub(super) fn run(
342    summary: &Summary<'_>,
343    json: bool,
344    layout: Layout,
345    output: &mut impl io::Write,
346) -> Result<(), Error> {
347    if json {
348        serde_json::to_writer_pretty(&mut *output, summary).map_err(Error::Json)?;
349        return writeln!(output).map_err(Error::Io);
350    }
351    let profiles = summary
352        .profiles
353        .iter()
354        .enumerate()
355        .map(|(index, profile)| {
356            Row::new([
357                (index + 1).to_string(),
358                profile.backend.to_string(),
359                profile.grounder.label().to_owned(),
360                profile.oracle.label().to_owned(),
361                profile.threads.to_string(),
362                profile
363                    .memory_bytes
364                    .map_or_else(|| "default".into(), |bytes| bytes.to_string()),
365                profile
366                    .time_limit_seconds
367                    .map_or_else(|| "none".into(), |seconds| seconds.to_string()),
368            ])
369        })
370        .collect();
371    table(
372        "Requested native profiles",
373        &[
374            "Profile",
375            "Backend",
376            "Grounder",
377            "Oracle",
378            "Threads",
379            "Memory bytes",
380            "Deadline seconds",
381        ],
382        profiles,
383    )?
384    .write(output, layout)
385    .map_err(Error::Io)?;
386    let rows = run_rows(summary);
387    table(
388        "Corpus benchmark — successful timed populations",
389        &[
390            "Workload",
391            "Producer",
392            "Qualified by",
393            "Median ms",
394            "Range ms",
395            "RSS MiB",
396            "All positions",
397            "Detail",
398        ],
399        rows,
400    )?
401    .write(output, layout)
402    .map_err(Error::Io)?;
403    table(
404        "Campaign outcome",
405        &["All passed", "Accounted"],
406        vec![Row::new([summary.passed.to_string(), summary.accounted.to_string()]).conclusion()],
407    )?
408    .write(output, layout)
409    .map_err(Error::Io)
410}

412fn run_rows(summary: &Summary<'_>) -> Vec<Row> {
413    summary
414        .cells
415        .iter()
416        .map(|cell| {
417            let producer = match cell.producer {
418                Producer::Reference => "clingo".to_owned(),
419                Producer::Native { profile } => format!("zetesis profile {}", profile + 1),
420            };
421            let decisions = cell
422                .decisions
423                .iter()
424                .map(|count| format!("{}: {}", count.decision.label(), count.positions))
425                .collect::<Vec<_>>()
426                .join(", ");
427            Row::new([
428                format!(
429                    "{}: {}",
430                    cell.case + 1,
431                    summary.workloads.map_or_else(
432                        || summary.cases[cell.case].clone(),
433                        |workloads| workloads[cell.case].label(),
434                    )
435                ),
436                producer,
437                qualification(cell.qualification).to_owned(),
438                cell.timing
439                    .as_ref()
440                    .map_or_else(|| "—".into(), |timing| milliseconds(timing.median_ns)),
441                cell.timing.as_ref().map_or_else(
442                    || "—".into(),
443                    |timing| {
444                        format!(
445                            "{}–{}",
446                            milliseconds(timing.minimum_ns),
447                            milliseconds(timing.maximum_ns)
448                        )
449                    },
450                ),
451                bytes(cell.peak_rss_bytes),
452                decisions,
453                cell.reasons
454                    .iter()
455                    .map(|(reason, count)| format!("{reason} ×{count}"))
456                    .collect::<Vec<_>>()
457                    .join("; "),
458            ])
459        })
460        .collect()
461}

463/// What qualified a cell's answer family, as the tables name it.
464const fn qualification(qualification: Qualification) -> &'static str {
465    match qualification {
466        Qualification::Clingo => "clingo",
467        Qualification::Contract => "recorded contract",
468        Qualification::NeedsClingo => "needs clingo",
469    }
