17enum Input {
18    Text(String),
19    Parsed(ParsedSource),

22enum FormulaInput {
23    Source(zetesis_themelios::AdmittedFormula),
24    Bundle(zetesis_themelios::AdmittedFormulaBundle),
25    Hybrid(zetesis_themelios::HybridFormula),
26    Terminal(zetesis_themelios::TerminalFormula),
27}
28
29impl FormulaInput {
30    fn prepared(&self) -> crate::PreparedInput<'_> {
31        match self {
32            Self::Source(owner) => crate::PreparedInput::formula(owner),
33            Self::Bundle(owner) => crate::PreparedInput::formula_bundle(owner),
34            Self::Hybrid(owner) => crate::PreparedInput::hybrid(owner),
35            Self::Terminal(owner) => crate::PreparedInput::terminal(owner),
36        }
37    }
38
39    fn warnings(&self, diagnostics: &mut Diagnostics<impl Write>) -> std::io::Result<()> {
40        match self {
41            Self::Source(owner) if !owner.warnings().is_empty() => {
42                diagnostics.diagnostic(&owner.warning_view())
43            }
44            Self::Bundle(owner) if !owner.warnings().is_empty() => {
45                diagnostics.diagnostic(&owner.warning_view())
46            }
47            Self::Hybrid(owner) if !owner.warnings().is_empty() => {
48                diagnostics.diagnostic(&owner.warning_view())
49            }
50            Self::Terminal(owner) if !owner.warnings().is_empty() => {
51                diagnostics.diagnostic(&owner.warning_view())
52            }
53            _ => Ok(()),
54        }
55    }
56
57    fn retain_source(&self, failure: PublicationFailure) -> PublicationFailure {
58        match self {
59            Self::Source(owner) => source_failure(failure, "<input>", owner.source()),
60            Self::Bundle(owner) => bundle_failure(failure, owner.bundle()),
61            Self::Hybrid(owner) => {
62                if let Some(bundle) = owner.bundle() {
63                    bundle_failure(failure, bundle)
64                } else if let Some(source) = owner.source() {
65                    source_failure(failure, "<input>", source)
66                } else {
67                    failure
68                }
69            }
70            Self::Terminal(owner) => {
71                if let Some(bundle) = owner.bundle() {
72                    bundle_failure(failure, bundle)
73                } else if let Some(source) = owner.source() {
74                    source_failure(failure, "<input>", source)
75                } else {
76                    failure
77                }
78            }
79        }
80    }
81
82    fn solve(
83        &self,
84        options: &Options,
85        renderer: &mut impl crate::AnswerRenderer,
86        diagnostics: &mut Diagnostics<impl Write>,
87        cancellation: &Cancellation,
88        phases: &Recorder,
89    ) -> Result<Progress, PublicationFailure> {
90        self.warnings(diagnostics)?;
91        crate::publication::solve(
92            self.prepared(),
93            None,
94            &crate::PublicationConfig::from(options),
95            renderer,
96            diagnostics,
97            cancellation,
98            phases,
99        )
100        .map_err(|failure| self.retain_source(failure))
101    }
102}

104fn validate_formula(options: &Options) -> Result<(), RunError> {
105    let config = crate::SolveConfig::from(options);
106    if options.grounder == crate::Grounder::Lazy {
107        config.validate_hybrid().map_err(Into::into)
108    } else {
109        config.validate_formula().map_err(Into::into)
110    }
111}

113pub(crate) fn source(
114    source: String,
115    options: &Options,
116    renderer: &mut impl crate::AnswerRenderer,
117    diagnostics: &mut Diagnostics<impl Write>,
118    cancellation: &Cancellation,
119    phases: &Recorder,
120) -> Result<Progress, PublicationFailure> {
121    crate::SolveConfig::from(options).validate()?;
122    if options.oracle == Oracle::Countermodel
123        && let Some(report) =
124            crate::publication::check_cancellation(renderer, diagnostics, cancellation, phases)?
125    {
126        return Ok(report);
127    }
128    let admission = AdmissionOptions {
129        max_source_bytes: options.max_source_bytes,
130        core_limits: core_limits(options),
131        ..Default::default()
132    };
133    let source = if options.oracle == Oracle::Countermodel {
134        Input::Text(source)
135    } else {
136        let source = phases
137            .measure(SolvePhase::AdmissionMaterialization, || {
138                ParsedSource::new(source, admission)
139            })
140            .map_err(|error| RunError::Expansion(error.into()))?;
141        match phases.measure(SolvePhase::AdmissionMaterialization, || {
142            source.admit_extended(expansion_limits(options))
143        }) {
144            Ok(admitted) => {
145                diagnostics.metadata(Label::Oracle, format_args!("reduct closure"))?;
146                return crate::publication::solve(
147                    crate::PreparedInput::admitted(&admitted),
148                    Some(*admitted.expansion_usage()),
149                    &crate::PublicationConfig::from(options),
150                    renderer,
151                    diagnostics,
152                    cancellation,
153                    phases,
154                )
155                .map_err(|failure| source_failure(failure, "<input>", admitted.source()));
156            }
157            Err(error)
158                if options.oracle == Oracle::Auto && error.error().needs_formula_admission() =>
159            {
160                Input::Parsed(error.into_source())
161            }
162            Err(error) => return Err(RunError::Expansion(error.into_error()).into()),
163        }
164    };
165    validate_formula(options)?;
166    if let Some(report) =
167        crate::publication::check_cancellation(renderer, diagnostics, cancellation, phases)?
168    {
169        return Ok(report);
170    }
171    let observer = phases.grounding_observer();
172    let admitted = phases
173        .measure(SolvePhase::AdmissionMaterialization, || {
174            let parsed = match source {
175                Input::Text(text) => ParsedSource::new(text, admission)?,
176                Input::Parsed(parsed) => parsed,
177            };
178            let prepared = parsed
179                .prepare_formula(expansion_limits(options), formula_limits(options))
180                .map_err(SourceFailure::into_error)?
181                .with_grounding_options(grounding_options(options))
182                .with_domain_analysis(Some(zetesis_themelios::DomainLimits::default()));
183            let observer = observer
184                .as_ref()
185                .map(|observer| observer as &dyn zetesis_themelios::GroundingObserver);
186            if options.grounder == crate::Grounder::Lazy {
187                prepared
188                    .ground_hybrid_with_observer(observer)
189                    .map(FormulaInput::Hybrid)
190            } else if options.grounder == crate::Grounder::Auto {
191                prepared
192                    .ground_adaptive_with_observer(observer)
193                    .map(|admitted| match admitted {
194                        zetesis_themelios::FormulaMaterialization::Complete(owner) => {
195                            FormulaInput::Source(owner)
196                        }
197                        zetesis_themelios::FormulaMaterialization::Terminal(owner) => {
198                            FormulaInput::Terminal(owner)
199                        }
200                    })
201            } else {
202                prepared
203                    .ground_with_observer(observer)
204                    .map(FormulaInput::Source)
205            }
206        })
207        .map_err(RunError::FormulaAdmission)?;
208    admitted.solve(options, renderer, diagnostics, cancellation, phases)
209}

211pub(crate) fn bundle(
212    bundle: SourceBundle,
213    options: &Options,
214    renderer: &mut impl crate::AnswerRenderer,
215    diagnostics: &mut Diagnostics<impl Write>,
216    cancellation: &Cancellation,
217    phases: &Recorder,
218) -> Result<Progress, PublicationFailure> {
219    crate::SolveConfig::from(options).validate()?;
220    if options.oracle == Oracle::Countermodel
221        && let Some(report) =
222            crate::publication::check_cancellation(renderer, diagnostics, cancellation, phases)?
223    {
224        return Ok(report);
225    }
226    diagnostics.metadata(
227        Label::Source,
228        format_args!(
229            "{} original {} ({} bytes)",
230            bundle.sources().len(),
231            if bundle.sources().len() == 1 {
232                "file"
233            } else {
234                "files"
235            },
236            bundle.total_bytes()
237        ),
238    )?;
239    let bundle = if options.oracle == Oracle::Countermodel {
240        bundle
241    } else {
242        match phases.measure(SolvePhase::AdmissionMaterialization, || {
243            admit_bundle_extended(bundle, bundle_options(options), expansion_limits(options))
244        }) {
245            Ok(admitted) => {
246                diagnostics.metadata(Label::Oracle, format_args!("reduct closure"))?;
247                return crate::publication::solve(
248                    crate::PreparedInput::bundle(&admitted),
249                    Some(*admitted.expansion_usage()),
250                    &crate::PublicationConfig::from(options),
251                    renderer,
252                    diagnostics,
253                    cancellation,
254                    phases,
255                )
256                .map_err(|failure| bundle_failure(failure, admitted.bundle()));
257            }
258            Err(error)
259                if options.oracle == Oracle::Auto
260                    && matches!(error.error(), BundleAdmissionError::Expansion(expansion)
261                    if expansion.needs_formula_admission()) =>
262            {
263                error.into_bundle()
264            }
265            Err(error) => return Err(RunError::BundleAdmission(error).into()),
266        }
267    };
268    validate_formula(options)?;
269    if let Some(report) =
270        crate::publication::check_cancellation(renderer, diagnostics, cancellation, phases)?
271    {
272        return Ok(report);
273    }
274    let observer = phases.grounding_observer();
275    let admitted = phases
276        .measure(SolvePhase::AdmissionMaterialization, || {
277            let prepared = zetesis_themelios::prepare_bundle_formula(
278                bundle,
279                bundle_options(options),
280                expansion_limits(options),
281                formula_limits(options),
282            )?
283            .with_grounding_options(grounding_options(options))
284            .with_domain_analysis(Some(zetesis_themelios::DomainLimits::default()));
285            let observer = observer
286                .as_ref()
287                .map(|observer| observer as &dyn zetesis_themelios::GroundingObserver);
288            if options.grounder == crate::Grounder::Lazy {
289                prepared
290                    .ground_hybrid_with_observer(observer)
291                    .map(FormulaInput::Hybrid)
292            } else if options.grounder == crate::Grounder::Auto {
293                prepared
294                    .ground_adaptive_with_observer(observer)
295                    .map(|admitted| match admitted {
296                        zetesis_themelios::FormulaMaterialization::Complete(owner) => {
297                            FormulaInput::Bundle(owner)
298                        }
299                        zetesis_themelios::FormulaMaterialization::Terminal(owner) => {
300                            FormulaInput::Terminal(owner)
301                        }
302                    })
303            } else {
304                prepared
305                    .ground_with_observer(observer)
306                    .map(FormulaInput::Bundle)
307            }
308        })
309        .map_err(RunError::FormulaBundleAdmission)?;
310    admitted.solve(options, renderer, diagnostics, cancellation, phases)
311}

313fn source_failure(
314    mut failure: PublicationFailure,
315    name: &str,
316    source: &zetesis_themelios::base::source::Source,
317) -> PublicationFailure {
318    if let RunError::Observation(error) = failure.cause.as_mut() {
319        error.retain_source(name, source);
320    }
321    failure
322}
323
324fn bundle_failure(mut failure: PublicationFailure, bundle: &SourceBundle) -> PublicationFailure {
325    use zetesis_themelios::base::source::Sources;
326
327    if let RunError::Observation(error) = failure.cause.as_mut()
328        && let Some(location) = error.location()
329        && let Some(source) = bundle.get(location.source)
330        && let Some(name) = bundle.name(location.source)
331    {
332        error.retain_source(name, source.source());
333    }
334    failure
335}

337pub(crate) fn expansion_limits(options: &Options) -> ExpansionLimits {
338    ExpansionLimits {
339        max_term_work: options
340            .max_expansion_work
341            .unwrap_or_else(|| ExpansionLimits::default().max_term_work),
342        max_templates: options.max_expanded_templates,
343        max_values: options.max_expansion_values,
344        max_scalar_bytes: options.max_expansion_bytes,
345        ..Default::default()
346    }
347}

349fn grounding_options(options: &Options) -> zetesis_themelios::GroundingOptions {

355pub(crate) fn formula_limits(options: &Options) -> zetesis_themelios::FormulaLimits {
356    zetesis_themelios::FormulaLimits {
357        max_domain_values: options
358            .max_domain_values
359            .unwrap_or_else(|| zetesis_themelios::FormulaLimits::default().max_domain_values),
360        max_assignment_values: options.max_assignment_values,
361        max_generated_values: options.max_generated_values,
362        max_support_rounds: options.max_support_rounds,
363        max_support_bytes: options.max_support_bytes,
364        max_substitutions: u64::try_from(options.max_substitutions).unwrap_or(u64::MAX),
365        max_work: options.max_expansion_work.map_or_else(
366            || zetesis_themelios::FormulaLimits::default().max_work,
367            |work| u64::try_from(work).unwrap_or(u64::MAX),
368        ),
369        theory: zetesis_ferraris::AdmissionLimits {
370            max_atoms: options.max_atoms,
371            max_roots: options.max_ground_rules,
372            ..Default::default()
373        },
374        ..Default::default()
375    }
376}
