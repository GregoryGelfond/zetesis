16/// Publication counts and search coverage from a completed driver invocation.
17/// Without objectives, stable models are written as found. Optimization retains
18/// incumbent ties and publishes the requested optimal ties only after exhaustion
19/// establishes the optimum.
20/// Interrupted optimization may publish retained incumbents without an optimum claim.
21#[derive(Clone, Debug)]
22pub struct Report {
23    /// Complete model records accepted by the output sink, including objective costs.
24    /// Partial writes are excluded; sink acceptance does not establish durability.
25    /// This count can be smaller than the number of verified stable models.
26    pub models: usize,
27    /// Closure result/control records examined, or classical formula candidates
28    /// proposed (including explicitly retained pending batch work). A whole-batch
29    /// closure interruption is one control record; shared statistics separately
30    /// count every submitted occurrence without a complete check.
31    pub checked: u64,
32    /// Coverage classification independent of satisfiability.
33    pub completion: Completion,
34    /// Interruption reason, present exactly for interrupted reports.
35    pub interruption: Option<Interruption>,
36    /// Gate tuples discovered by the candidate generator.
37    pub discovered_gate_atoms: usize,
38    /// Expansion charges relational admission accepted, each under its
39    /// ceiling; absent for the formula route.
40    pub expansion: Option<zetesis_themelios::ExpansionUsage>,
41    /// Necessary closure-candidate restrictions, including interrupted work.
42    pub candidate_statistics: Option<zetesis_cpu::CandidateStatistics>,
43    /// Cumulative countermodel accounting when available; absent for closure
44    /// and for an initialization failure that returns no statistics.
45    pub countermodel_statistics: Option<zetesis_sat::Statistics>,
46    /// Actual batched formula execution and pending-result accounting; absent
47    /// when the scalar CPU route was used or initialization did not finish.
48    pub formula_execution: Option<crate::FormulaExecutionStatistics>,
49    /// Retained-core answers and their source-constraint decisions. Only accepted
50    /// checks establish membership in the original hybrid program.
51    pub hybrid_execution: Option<zetesis_solve::HybridExecutionStatistics>,
52    /// Base answers and completed original-answer reconstruction.
53    pub terminal_execution: Option<zetesis_solve::TerminalExecutionStatistics>,
54    /// Prepared formula-model ordering and completed selections, including stopped work.
55    pub model_construction: Option<zetesis_solve::ModelConstructionStatistics>,
56    /// Actual lazy device execution, including shared source work and failed
57    /// batch progress. Absent when no lazy device executor was initialized.
58    pub lazy_execution: Option<crate::LazyExecutionStatistics>,
59    /// Shared CPU source and per-world work, including failed-batch prefixes.
60    pub shared_execution: Option<crate::SharedExecutionStatistics>,
61    /// Independent CPU closure counters summed over completed checks; absent
62    /// for the shared, device and formula routes.
63    pub closure_execution: Option<crate::ClosureExecutionStatistics>,
64    /// Prepared independent CPU ownership receipts and any snapshot fault.
65    pub query_execution: Option<crate::QueryExecutionObservation>,
66    /// Whether semantic search established an optimum, independently of delivery.
67    pub optimum_proved: bool,
68    /// Best retained objective score and tied models found so far.
69    /// A retained score is optimal only when `optimum_proved` is true.
70    pub optimization: Option<crate::Optimization>,
71    /// Attempted host timings requested by statistics or the selected renderer.
72    /// The default human view requests coarse stages; detailed phases remain
73    /// absent unless statistics are enabled. A timing snapshot establishes
74    /// neither semantic completion nor successful output.
75    pub phase_timings: Option<crate::PhaseTimings>,
76}

119    MixedStandardInput,

135    BackendUnavailable,

162    },
163    /// An already prepared representation cannot honor the requested strategy.
164    PreparedInput {
165        /// Representation supplied by the caller.

167        /// Explicit membership strategy.

198    /// Concrete shared CPU evaluation violated its round protocol.

200    /// Static closure decoding refused its words or selected-position storage.

210    FormulaBatchShape {

213        /// Number of verdicts returned.

215    },

271                workers,

273                max_closure_batch_bytes,

275                f,

279            Self::SharedCpu(cause) => cause.fmt(f),
280            Self::Formula(error) => error.fmt(f),
281            Self::FormulaAdmission(error) => error.fmt(f),
282            Self::FormulaBundleAdmission(error) => error.fmt(f),
283            Self::Output(error) => write!(f, "output: {error}"),
284            Self::ExecutionObservation(error) => write!(f, "execution observer: {error}"),
285            Self::PublicationStopped(error) => write!(f, "publication stopped: {error}"),
286            Self::Observation(error) => error.fmt(f),
287            Self::JsonRecord(error) => error.fmt(f),
288            Self::ObservationOutputLimit { observed, limit } => write!(f, "human Answer requires at least {observed} bytes; limit {limit}"),
289            Self::Static(error) => error.fmt(f),
290            #[cfg(feature = "gpu")]
291            Self::Gpu(error) => error.fmt(f),
292            #[cfg(feature = "gpu")]
293            Self::LazyGpu(error) => error.fmt(f),
294            Self::LazyStatisticsOverflow => f.write_str("lazy execution statistics overflow"),

297            }
298            Self::ModelStatisticsOverflow => f.write_str("model construction statistics overflow"),

304        }?;
305        self.write_diagnostics(f)
306    }

308impl RunError {
309    fn write_diagnostics(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
310        let (bundle, diagnostics) = match self {
311            // Syntax refusals already render the retained source through the
312            // canonical themelios view. Appending byte-only labels duplicates it.
313            Self::Admission(AdmissionFailure::Syntax(_))
314            | Self::Expansion(ExpansionFailure::Admission(AdmissionFailure::Syntax(_)))
315            | Self::FormulaAdmission(zetesis_themelios::FormulaFailure::Expansion(
316                ExpansionFailure::Admission(AdmissionFailure::Syntax(_)),
317            )) => return Ok(()),
318            Self::Admission(error) => (None, error.diagnostics()),
319            Self::Expansion(error) => (None, error.diagnostics()),
320            Self::FormulaAdmission(error) => (None, error.diagnostics()),
321            Self::BundleAdmission(error) => (Some(error.bundle()), error.diagnostics()),
322            Self::FormulaBundleAdmission(error) => (Some(error.bundle()), error.diagnostics()),
323            _ => return Ok(()),
324        };
325        for diagnostic in diagnostics {
326            // A retained bundle names each file and locates the span by line
327            // and column, with the source excerpt, through the canonical view.
328            let location = diagnostic.primary().location;
329            // A span over the whole file stands for the program, not one
330            // statement: it is named as such, with no line and column.
331            if let Some(bundle) = bundle
332                && let Some(source) = bundle.get(location.source)
333                && location.span == source.source().span()
334            {
335                write!(
336                    f,
337                    "\n  {}: while admitting the program: {}",
338                    source.path().display(),
339                    diagnostic.message()
340                )?;
341                continue;
342            }
343            if let Some(bundle) = bundle
344                && bundle.get(location.source).is_some()
345            {
346                let rendered = zetesis_themelios::base::view::human(&diagnostic, bundle);
347                write!(f, "\n{}", rendered.trim_end_matches('\n'))?;
348                continue;
349            }
350            // Without retained source text only the byte span is known.
351            write!(
352                f,
353                "\n  source {}: bytes {}..{}: {}",
354                location.source.get(),
355                location.span.start().get(),
356                location.span.end().get(),
357                diagnostic.message()
358            )?;
359        }
360        Ok(())
361    }
362}

422/// Admit and stream exact stable models, with separate coverage reporting.
423/// Enumeration checks the empty seed alone before requesting more gate tuples.
424/// The countermodel oracle checks and blocks complete semantic interpretations.
425/// Both routes preserve distinct full models independently of display selection.
426/// With objectives, requested optimal ties are published only after exhaustion
427/// establishes the optimum.
428/// Interrupted optimization may publish retained incumbents without an optimum claim.
429/// A completed exhaustive run with no models is the only UNSAT outcome.
430///
431/// # Errors
432/// Returns [`RunError`] for source, backend, observation/view, or output failures.
433/// Search, objective, and incumbent-retention stops produce an interrupted
434/// [`Report`] unless a later reporting operation fails. Cooperative publication
435/// stops use the legacy `PublicationStopped` adapter; the finalized API exposes
436/// them directly as `PublicationOutcome::Stopped`.
437pub fn run(
438    source: String,
439    options: &Options,
440    output: &mut impl Write,
441    cancellation: &Cancellation,
442) -> Result<Report, RunError> {
443    run_with_diagnostics(source, options, output, &mut io::sink(), cancellation)
444}

447/// This is the process adapter's entry point: model output goes to stdout and
448/// backend selection/fallback reasons go to stderr. [`run`] keeps its original
449/// signature and discards backend diagnostics for callers that do not need them.
450///
451/// # Errors
452/// Returns [`RunError`] for source, backend, observation/view, model-output, or
453/// diagnostics failures. Search, objective, and incumbent-retention stops produce
454/// an interrupted [`Report`] unless a later reporting operation fails.

486/// A failed Answer is excluded from the published count even if its prefix escaped.
487/// Search, objective, and incumbent-retention stops return an interrupted [`Report`]

490/// # Errors

506}

687    bundle: SourceBundle,
688    options: &Options,
689    renderer: &mut impl crate::AnswerRenderer,

708/// Publish an already admitted input without reparsing or collecting its family.
709/// The same session controller and renderer lifecycle serve source and prepared
710/// inputs; semantic membership stays in `zetesis-solve`.
711///
712/// # Errors
713/// Retains execution and publication evidence under the same contract as
714/// [`run_with_renderer`]. No source admission is performed here.
715pub fn publish_prepared(
716    input: crate::PreparedInput<'_>,
717    config: &crate::PublicationConfig,
718    renderer: &mut impl crate::AnswerRenderer,
719    diagnostics: &mut impl Write,
720    cancellation: &Cancellation,
721) -> Result<crate::PublicationOutcome, crate::PublicationFailure> {
722    let mut diagnostics = Diagnostics::new(diagnostics, crate::ColorMode::Never).with_quiet(true);
723    let mut invocation = crate::view::session::Session::start(renderer)?;
724    let renderer = &mut invocation;
725    let phases = crate::phase_timing::recorder(config.solve.stats, renderer);
726    let mut result = crate::publication::solve(
727        input,
728        None,
729        config,
730        renderer,
731        &mut diagnostics,
732        cancellation,
733        &phases,
734    );
735    if let Some(timings) = phases.snapshot() {
736        match &mut result {
737            Ok(progress) => progress.phase_timings = Some(timings),
738            Err(failure) => failure.phase_timings = Some(Box::new(timings)),
739        }
740    }
741    crate::publication::finalize(renderer, result)
742}
743
744fn report_progress_statistics(
745    mut result: Result<Progress, crate::PublicationFailure>,
746    diagnostics: &mut Diagnostics<impl Write>,
747    options: &Options,
748    phases: &Recorder,
749) -> Result<Progress, crate::PublicationFailure> {
750    if let Some(timings) = phases.snapshot() {
751        match &mut result {
752            Ok(progress) => progress.phase_timings = Some(timings),
753            Err(failure) => failure.phase_timings = Some(Box::new(timings)),
754        }
755        if !options.stats {
756            return result;
757        }
758        // The report is rendered whole, then written once at this point: its
759        // fragments would otherwise each be a write on unbuffered stderr. Order
760        // relative to other diagnostics is unchanged, and a write failure
761        // surfaces at this one write.
762        let mut report = Vec::new();
763        let rendered = match options.statistics_view {
764            crate::StatisticsView::Records => crate::statistics::write_progress(
765                &mut report,
766                options,
767                result.as_ref(),
768                timings.driver_elapsed,
769            )
770            .and_then(|()| crate::stage_timing::write(&mut report, &timings.stages))
771            .and_then(|()| crate::phase_timing::write(&mut report, &timings))
772            .and_then(|()| crate::grounding_timing::write(&mut report, &timings.grounding)),
773            crate::StatisticsView::Human => {
774                let view = crate::PublicationView {
775                    result: result.as_ref(),
776                };
777                let config = crate::SolveConfig::from(options);
778                let statistics = crate::statistics_view::Statistics {
779                    requested: &config,
780                    timings: &timings,
781                    semantic: view.semantic(),
782                    publication: view.publication(),
783                    failed: view.failure().is_some(),
784                };
785                let layout = diagnostics.layout();
786                writeln!(report).and_then(|()| statistics.write_human(&mut report, layout))
787            }
788        };
789        let emitted = rendered.and_then(|()| diagnostics.write_all(&report));
790        if let Err(error) = emitted {
791            return Err(match result {
792                Ok(progress) => progress.fail(RunError::Output(error)),
793                Err(mut failure) => {
794                    failure.record_diagnostics(error);
795                    failure
796                }
797            });
798        }
799    }
800    result
801}

803impl From<zetesis_solve::SolveError> for RunError {
804    fn from(error: zetesis_solve::SolveError) -> Self {
805        use zetesis_solve::SolveError;
806        match error {
807            SolveError::Reconstruction(error) => Self::Reconstruction(error),
808            SolveError::TerminalStatisticsOverflow => Self::TerminalStatisticsOverflow,
809            SolveError::Constraint(error) => Self::Constraint(error),
810            SolveError::ConstraintFailureMissing => Self::ConstraintFailureMissing,
811            SolveError::HybridStatisticsOverflow => Self::HybridStatisticsOverflow,
812            SolveError::HybridBackend { backend } => Self::HybridBackend { backend },
813            SolveError::Executor(error) => Self::Executor(error),
814            SolveError::Projection(error) => Self::Projection(error),
815            SolveError::Batch(error) => Self::Batch(error),
816            SolveError::QueryObservation(error) => Self::QueryObservation(error),
817            SolveError::CompletionPool(error) => Self::CompletionPool(error),
818            SolveError::BackendUnavailable => Self::BackendUnavailable,
819            SolveError::UnsupportedSourceBatching => Self::UnsupportedSourceBatching,
820            SolveError::ClosureReservation {
821                workers,
822                max_closure_bytes,
823                max_closure_batch_bytes,
824            } => Self::ClosureReservation {
825                workers,
826                max_closure_bytes,
827                max_closure_batch_bytes,
828            },
829            SolveError::Static(error) => Self::Static(error),
830            SolveError::LazyStatisticsOverflow => Self::LazyStatisticsOverflow,
831            SolveError::ClosureStatisticsOverflow => Self::ClosureStatisticsOverflow,
832            SolveError::ModelStatisticsOverflow => Self::ModelStatisticsOverflow,
833            SolveError::SharedCpu(error) => Self::SharedCpu(error),
834            SolveError::Words(error) => Self::Words(error),
835            SolveError::Model(error) => Self::Model(error),
836            SolveError::CandidateStreamNotExhausted => Self::CandidateStreamNotExhausted,
837            SolveError::UnsupportedOracle { backend, grounder } => {
838                Self::UnsupportedOracle { backend, grounder }
839            }
840            SolveError::PreparedInput {
841                profile,
842                oracle,
843                grounder,
844            } => Self::PreparedInput {
845                profile,
846                oracle,
847                grounder,
848            },
849            SolveError::FormulaBatchShape { expected, actual } => {
850                Self::FormulaBatchShape { expected, actual }
851            }
852            #[cfg(feature = "gpu")]
853            SolveError::Gpu(error) => Self::Gpu(error),
854            #[cfg(feature = "gpu")]
855            SolveError::LazyGpu(error) => Self::LazyGpu(error),
856            SolveError::ExecutionObservation(error) => match error.downcast::<RunError>() {
857                Ok(error) => *error,
858                Err(error) => match error.downcast::<std::io::Error>() {
859                    Ok(error) => Self::Output(*error),
860                    Err(error) => Self::ExecutionObservation(error),
861                },
862            },
863        }
864    }
