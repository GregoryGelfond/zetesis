354/// The host's physical memory in bytes, read once per process the first
355/// time it is asked for, when the default allowance is taken or the
356/// statistics header prints it: the header prints the reading the allowance
357/// came from.
358pub(crate) fn host_memory() -> Option<u64> {
359    static HOST_MEMORY: std::sync::OnceLock<Option<u64>> = std::sync::OnceLock::new();
360    *HOST_MEMORY.get_or_init(read_host_memory)
361}
362
363/// The host's physical memory in bytes, from `/proc/meminfo`.
364#[cfg(target_os = "linux")]
365fn read_host_memory() -> Option<u64> {
366    let text = std::fs::read_to_string("/proc/meminfo").ok()?;
367    let kibibytes: u64 = text
368        .lines()
369        .find_map(|line| line.strip_prefix("MemTotal:"))?
370        .split_whitespace()
371        .next()?
372        .parse()
373        .ok()?;
374    kibibytes.checked_mul(1024)
375}
376
377/// The host's physical memory in bytes, from the system's `sysctl`: the
378/// crate forbids foreign calls, and the system command is the reading
379/// without one.
380#[cfg(target_os = "macos")]
381fn read_host_memory() -> Option<u64> {
382    let output = std::process::Command::new("sysctl")
383        .args(["-n", "hw.memsize"])
384        .output()
385        .ok()?;
386    std::str::from_utf8(&output.stdout)
387        .ok()?
388        .trim()
389        .parse()
390        .ok()
391}
392
393/// The host does not report its physical memory on this platform.
394#[cfg(not(any(target_os = "linux", target_os = "macos")))]
395fn read_host_memory() -> Option<u64> {
396    None
397}

399/// The default allowance: half of the host's memory, at least the reference.
400fn host_memory_allowance() -> u64 {
401    host_memory().map_or(crate::SolveConfig::REFERENCE_MEMORY, |memory| {
402        (memory / 2).max(crate::SolveConfig::REFERENCE_MEMORY)
403    })
404}

406impl Options {
407    /// The library's defaults for the allowance and the workers: what each
408    /// byte ceiling not given on the command line is.
409    fn allowed(&self) -> crate::SolveConfig {
410        crate::SolveConfig::for_allowance(self.memory, self.workers)
411    }
412
413    /// The per-closure allowance: the given value, or each worker's share of
414    /// the collective ceiling.
415    #[must_use]
416    pub fn closure_allowance(&self) -> usize {
417        self.max_closure_bytes.unwrap_or_else(|| {
418            self.max_closure_batch_bytes.map_or_else(
419                || self.allowed().max_closure_bytes,
420                |collective| collective / self.workers.get(),
421            )
422        })
423    }
424
425    /// The collective closure ceiling: the given value, or the library
426    /// default scaled by the allowance.
427    #[must_use]
428    pub fn closure_collective(&self) -> usize {
429        self.max_closure_batch_bytes
430            .unwrap_or_else(|| self.allowed().max_closure_batch_bytes)
431    }
432}

498// The CLI owns each spelling/help mapping, while the parsed values remain the
499// library's policies. Clap validates against the same table before mapping, so
500// every accepted value has a typed entry. Case-insensitive lookup is safe here:
501// PossibleValuesParser already enforces the argument's case-sensitivity policy.
502fn policy_parser<T: Clone + Send + Sync + 'static, const N: usize>(
503    choices: [(T, PossibleValue); N],
504) -> impl TypedValueParser<Value = T> {
505    PossibleValuesParser::new(choices.iter().map(|(_, value)| value.clone())).map(move |value| {
506        choices
507            .iter()
508            .find(|(_, possible)| possible.matches(&value, true))
509            .expect("clap validated this value against the same choice table")
510            .0
511            .clone()
512    })
513}
