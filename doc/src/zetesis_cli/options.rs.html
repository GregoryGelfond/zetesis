151/// The full help's account of `--backend`: every value with its meaning, in the
152/// vocabulary's order, so that each API's scope and limits are stated where the
153/// expert values are offered.
154fn backend_long_help() -> String {
155    let values: Vec<String> = Backend::ALL
156        .into_iter()
157        .map(|backend| format!("  {:<7} {}", backend.label(), backend.help()))
158        .collect();
159    format!(
160        "Execution backend: cpu (the default) or gpu.\n\nValues:\n{}\n\n\
161         An unavailable GPU fails; device failure does not silently retry on CPU.",
162        values.join("\n")
163    )
164}

166/// The memory available to the process in bytes, read once per process on
167/// first use. Building the command uses it, since the default allowance
168/// derives from it, so every invocation reads it, even one with an explicit
169/// `--memory`; a caller that builds `Options` itself first reads it when the
170/// statistics header prints it. The header prints this same reading, the one
171/// any default allowance came from.
172pub(crate) fn host_memory() -> Option<u64> {
173    static HOST_MEMORY: std::sync::OnceLock<Option<u64>> = std::sync::OnceLock::new();
174    *HOST_MEMORY.get_or_init(read_host_memory)
175}

177/// The memory available to the process in bytes, from the files Linux keeps.
178#[cfg(target_os = "linux")]
179fn read_host_memory() -> Option<u64> {
180    available_memory(|path| std::fs::read_to_string(path).ok())
181}

183/// The memory available to the process in bytes, where `read` returns a file's
184/// text: the host's physical memory from `/proc/meminfo`, lowered to the memory
185/// limit of the process's control group when one is set, since inside a
186/// container the host's figure is not the memory the process may use. Without
187/// `/proc/meminfo` there is no reading, whatever a group states: a group's value
188/// alone can be version 1's "unlimited", which only the comparison with
189/// physical memory renders harmless.
190#[cfg(any(target_os = "linux", test))]
191fn available_memory(read: impl Fn(&std::path::Path) -> Option<String>) -> Option<u64> {
192    let physical = memory_total(&read(std::path::Path::new("/proc/meminfo"))?)?;
193    let limit = read(std::path::Path::new("/proc/self/cgroup"))
194        .and_then(|groups| cgroup::memory_limit(&groups, &read));
195    Some(limit.map_or(physical, |limit| limit.min(physical)))
196}

198/// Physical memory in bytes from the text of `/proc/meminfo`.
199#[cfg(any(target_os = "linux", test))]
200fn memory_total(meminfo: &str) -> Option<u64> {
201    let kibibytes: u64 = meminfo
202        .lines()
203        .find_map(|line| line.strip_prefix("MemTotal:"))?
204        .split_whitespace()
205        .next()?
206        .parse()
207        .ok()?;
208    kibibytes.checked_mul(1024)
209}

211/// The host's physical memory in bytes, from the kernel's `hw.memsize`
212/// through one `sysctlbyname` call. macOS reports physical memory only through
213/// system interfaces (`sysctl`, or Mach's `host_info`), each a foreign call;
214/// this is the one in zetesis's own source, and it starts no process. The
215/// kernel writes at most the eight bytes it is given and stores how many it
216/// wrote; anything but a complete eight-byte reading is no reading.
217#[cfg(target_os = "macos")]
218#[expect(
219    unsafe_code,
220    reason = "macOS reports physical memory only through system interfaces: one sysctlbyname call into a fixed eight-byte buffer"
221)]
222fn read_host_memory() -> Option<u64> {
223    use std::ffi::{c_char, c_int, c_void};
224    unsafe extern "C" {
225        fn sysctlbyname(
226            name: *const c_char,
227            oldp: *mut c_void,
228            oldlenp: *mut usize,
229            newp: *mut c_void,
230            newlen: usize,
231        ) -> c_int;
232    }
233    let mut memory = 0_u64;
234    let mut length = size_of::<u64>();
235    // SAFETY: the declaration matches sysctlbyname(3), `int sysctlbyname(const
236    // char *name, void *oldp, size_t *oldlenp, void *newp, size_t newlen)`, with
237    // `usize` for `size_t`. The name is a NUL-terminated string; `oldp` points
238    // to `memory`, whose size `length` states, so the kernel writes at most
239    // those eight bytes and stores the count it wrote in `length`; with a null
240    // `newp` and a zero `newlen` nothing is set.
241    let status = unsafe {
242        sysctlbyname(
243            c"hw.memsize".as_ptr(),
244            (&raw mut memory).cast(),
245            &raw mut length,
246            std::ptr::null_mut(),
247            0,
248        )
249    };
250    (status == 0 && length == size_of::<u64>()).then_some(memory)
251}

259/// The default allowance: half of the memory available to the process, or
260/// the reference when the host does not report it.
261fn host_memory_allowance() -> u64 {
262    allowance(host_memory())
263}
264
265/// Half of `available` memory whenever there is a reading, so that the
266/// allowance stays within what the process may use, a container's limit
267/// included; the reference without one.
268fn allowance(available: Option<u64>) -> u64 {
269    available.map_or(crate::SolveConfig::REFERENCE_MEMORY, |memory| memory / 2)
270}

295// The CLI owns each spelling/help mapping, while the parsed values remain the
296// library's policies. Clap validates against the same table before mapping, so
297// every accepted value has a typed entry. Case-insensitive lookup is safe here:
298// PossibleValuesParser already enforces the argument's case-sensitivity policy.
299fn policy_parser<T: Clone + Send + Sync + 'static, const N: usize>(
300    choices: [(T, PossibleValue); N],
301) -> impl TypedValueParser<Value = T> {
302    PossibleValuesParser::new(choices.iter().map(|(_, value)| value.clone())).map(move |value| {
303        choices
304            .iter()
305            .find(|(_, possible)| possible.matches(&value, true))
306            .expect("clap validated this value against the same choice table")
307            .0
308            .clone()
309    })
310}
