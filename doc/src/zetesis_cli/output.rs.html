9pub(crate) struct Document<W> {
10    sink: W,
11    failed: bool,
12}
13impl<W: Write> Document<W> {
14    pub(crate) const fn new(sink: W) -> Self {
15        Self {
16            sink,
17            failed: false,
18        }
19    }
20
21    pub(crate) const fn failed(&self) -> bool {
22        self.failed
23    }
24
25    pub(crate) fn start(&mut self) {
26        self.failed = false;
27    }
28
29    pub(crate) fn into_inner(self) -> W {
30        self.sink
31    }
32}
33impl<W: Write> Write for Document<W> {
34    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
35        let result = self.sink.write(bytes);
36        if result
37            .as_ref()
38            .is_err_and(|error| error.kind() != io::ErrorKind::Interrupted)
39            || matches!(result, Ok(0)) && !bytes.is_empty()
40        {
41            self.failed = true;
42        }
43        result
44    }
45    fn flush(&mut self) -> io::Result<()> {
46        let result = self.sink.flush();
47        self.failed |= result.is_err();
48        result
49    }
50}

52/// Preflight the bounded UTF-8 record, then write it to the supplied sink.
53/// The record spells the atoms the document has not spelled and refers to
54/// every atom by its index in `atoms`, the document's table.
55pub(crate) fn write_model_record(
56    output: &mut impl Write,
57    number: usize,
58    view: &zetesis_themelios::observation::ModelView<'_>,
59    atoms: &mut zetesis_themelios::observation::json::AtomTable,
60    max_record_bytes: usize,
61    cancellation: &zetesis_cpu::Cancellation,
62) -> Result<(), RunError> {
63    let prefix = format!(
64        "{}{{\"number\":{number},\"model\":",
65        if number == 1 { "" } else { "," }
66    );
67    let overhead = prefix.len() + 2;
68    let maximum = max_record_bytes
69        .checked_sub(overhead)
70        .ok_or(RunError::JsonRecord(ViewError::Bytes))?;
71    let record = view
72        .record(
73            atoms,
74            zetesis_themelios::observation::ViewLimits {
75                max_bytes: maximum,
76                ..Default::default()
77            },
78            cancellation,
79        )
80        .map_err(RunError::JsonRecord)?;
81    cancellation
82        .poll()
83        .map_err(|error| RunError::JsonRecord(ViewError::Stopped(error)))?;
84    // All semantic/encoding failures precede publication. I/O can still fail
85    // after a prefix; the caller increments its count only after all writes.
86    output.write_all(prefix.as_bytes())?;
87    output.write_all(record.as_bytes())?;
88    output.write_all(b"}\n")?;
89    Ok(())
90}

92/// JSON for failures that precede the ordinary source/bundle driver entry.
93pub(crate) fn input_failure(
94    output: &mut impl Write,
95    failure: RunFailure,
96    options: &Options,
97) -> RunFailure {
98    use crate::AnswerRenderer;
99    let mut renderer =
100        crate::JsonRenderer::new(output, options.max_json_record_bytes, options.max_atoms);
101    match renderer.begin() {
102        Ok(()) => crate::publication::finalize(&mut renderer, Err(failure.into()))
103            .expect_err("input failure remains failed")
104            .into_legacy(),
105        Err(output_failure) => {
106            let mut failure = failure;
107            failure.secondary_output = Some(io::Error::other(output_failure));
108            failure
109        }
110    }
111}

121pub(crate) fn summary(
122    publication: crate::PublicationView<'_>,
123    maximum: usize,
124) -> Result<Vec<u8>, RunError> {
125    let result = publication.result;
126    let status = match result {
127        Err(_) => "failed",
128        Ok(progress) if progress.stop.is_some() => "incomplete",
129        Ok(progress) => {
130            let semantic = progress.semantic().ok_or(RunError::CompletionUnavailable)?;
131            match progress.completion()? {
132                Completion::Interrupted => "incomplete",
133                Completion::Exhausted if semantic.unsatisfiable() => "unsatisfiable",
134                _ => "satisfiable",
135            }
136        }
137    };
138    let mut out = Buffer::new(maximum);
139    let view = SummaryView::new(publication);
140    out.text("],\"outcome\":{\"status\":")?;
141    out.string(status)?;
142    out.text(",\"completion\":")?;
143    out.optional_string(view.completion.map(completion))?;
144    out.text(",\"coverage\":")?;
145    out.string(match view.completion {
146        Some(Completion::Exhausted) => "exhausted",
147        Some(_) => "partial",
148        None => "unavailable",
149    })?;
150    out.number_field("published_models", view.published_models)?;
151    out.text(",\"verified_models\":")?;
152    out.optional_number(view.verified_models)?;
153    out.text(",\"checked\":")?;
154    out.optional_number(view.checked)?;
155    out.text(",\"projection\":")?;
156    write_projection(&mut out, view.projection)?;
157    out.text(",\"interruption\":")?;
158    write_interruption(&mut out, view.interruption)?;
159    out.text(",\"publication_stop\":")?;
160    let stop = match result {
161        Ok(progress) => progress.stop.as_ref(),
162        Err(failure) => failure.publication_stop(),
163    };
164    if let Some(stop) = stop {
165        out.text("{\"phase\":")?;
166        out.string(match stop.phase() {
167            crate::PublicationPhase::Observation => "observation",
168            crate::PublicationPhase::Encoding => "encoding",
169            crate::PublicationPhase::RecordPreparation => "record_preparation",
170        })?;
171        out.text(",\"code\":")?;
172        out.string(stop_code(stop.reason()))?;
173        out.text("}")?;
174    } else {
175        out.text("null")?;
176    }
177    out.text(",\"optimization\":")?;
178    write_optimization(&mut out, view.optimization, view.optimum_proved)?;
179    out.text(",\"error\":")?;
180    if let Err(failure) = result {
181        out.text("{\"kind\":")?;
182        out.string(error_kind(&failure.cause))?;
183        out.text(",\"detail\":")?;
184        out.display_string(&failure.cause)?;
185        out.text(",\"secondary_output_failure\":")?;
186        out.text(if failure.secondary_output.is_some() {
187            "true"
188        } else {
189            "false"
190        })?;
191        out.text("}")?;
192    } else {
193        out.text("null")?;
194    }
195    out.text("},\"statistics\":")?;
196    statistics(&mut out, &view)?;
197    out.text("}")?;
198    Ok(out.bytes)
199}

201fn phases(out: &mut Buffer, timings: Option<&PhaseTimings>) -> Result<(), RunError> {
202    let Some(timings) = timings else {
203        return out.text("null");
204    };
205    out.text(&format!(
206        "{{\"schema\":5,\"clock\":\"host_monotonic\",\"driver_elapsed_ns\":{},\"measurements\":{{",
207        timings.driver_elapsed.as_nanos()
208    ))?;
209    for (index, phase) in SolvePhase::ALL.into_iter().enumerate() {
210        if index != 0 {
211            out.text(",")?;
212        }
213        out.string(phase.label())?;
214        out.text(":")?;
215        if let Some(value) = timings.get(phase) {
216            out.text(&format!(
217                "{{\"calls\":{},\"elapsed_ns\":{},\"complete\":{}}}",
218                value.calls,
219                value.elapsed.as_nanos(),
220                !value.overflowed
221            ))?;
222        } else {
223            out.text("null")?;
224        }
225    }
226    out.text("},\"source_loading\":\"excluded\",\"statistics_output\":\"excluded\",\"json_envelope\":\"excluded\",\"kernel_time\":null}")
227}

229fn stages(out: &mut Buffer, timings: Option<&crate::StageTimings>) -> Result<(), RunError> {
230    let Some(timings) = timings else {
231        return out.text("null");
232    };
233    out.text("{\"schema\":1,\"clock\":\"host_monotonic\",\"scope\":\"driver\"")?;
234    out.number_field("driver_elapsed_ns", timings.driver_elapsed.as_nanos())?;
235    out.text(",\"grounding_mode\":")?;
236    out.string(timings.grounding_mode.label())?;
237    out.text(",\"measurements\":{")?;
238    for (index, stage) in crate::SolveStage::ALL.into_iter().enumerate() {
239        if index != 0 {
240            out.text(",")?;
241        }
242        out.string(stage.label())?;
243        out.text(":")?;
244        if let Some(value) = timings.get(stage) {
245            out.text("{\"calls\":")?;
246            out.text(&value.calls.to_string())?;
247            out.number_field("elapsed_ns", value.elapsed.as_nanos())?;
248            out.text(",\"complete\":")?;
249            out.text(if value.overflowed { "false" } else { "true" })?;
250            out.text("}")?;
251        } else {
252            out.text("null")?;
253        }
254    }
255    out.text("},\"unattributed_elapsed_ns\":")?;
256    if let Some(value) = timings.unattributed {
257        out.text(&value.as_nanos().to_string())?;
258    } else {
259        out.text("null")?;
260    }
261    out.text(",\"complete\":")?;
262    out.text(if timings.is_complete() {
263        "true"
264    } else {
265        "false"
266    })?;
267    out.text(",\"timer_overhead\":\"not_separated\",\"source_loading\":\"excluded\",\"statistics_output\":\"excluded\",\"json_envelope\":\"excluded\",\"kernel_time\":null}")
268}

323fn reconstruction_kind(error: &zetesis_themelios::ReconstructionError) -> &'static str {
324    use zetesis_core::{ModelError, ModelFailure};
325    use zetesis_themelios::{FormulaFailure, ReconstructionError};
326    match error {
327        ReconstructionError::Source(error)
328        | ReconstructionError::Model(ModelFailure::Stopped(error))
329            if matches!(error.as_ref(), FormulaFailure::Limit { .. }) =>
330        {
331            "answer_reconstruction_limit"
332        }
333        ReconstructionError::Model(ModelFailure::Model(ModelError::Bytes { .. })) => {
334            "answer_reconstruction_limit"
335        }
336        _ => "answer_reconstruction",
337    }

365                Incomplete::CompletionScratch => "completion_scratch",
366                Incomplete::ReductStorage { .. } => "reduct_storage",
367                Incomplete::BatchCandidateLimit => "batch_candidate_limit",

369                Incomplete::Allocation => "allocation",
370                Incomplete::WorkerPanicked => "worker_panicked",
371                Incomplete::Admission(_) => "admission",

374                Incomplete::ClosedEnumerator => "closed_enumerator",

376                Incomplete::RegionFilter => "region_filter",

386        Interruption::Objective(error) => objective_code(error),
387        Interruption::PreparedObjective(error) => {
388            use zetesis_themelios::objective_bound::{
389                ObjectiveBoundErrorKind, ObjectiveScoreErrorKind,

392                ObjectiveScoreErrorKind::WrongTheory => "wrong_theory",

394                ObjectiveScoreErrorKind::Eligibility(error) => match error.kind() {

397                    ObjectiveBoundErrorKind::Allocation => "allocation",
398                    ObjectiveBoundErrorKind::Overflow => "arithmetic_overflow",
399                    _ => "objective_preparation",
400                },
401            }
402        }
403        Interruption::Incumbent(reason) => match reason {
404            crate::OptimizationStop::Models => "models",
405            crate::OptimizationStop::Atoms => "atoms",
406            crate::OptimizationStop::Bytes => "bytes",
407            crate::OptimizationStop::Overflow => "overflow",
408            crate::OptimizationStop::Allocation => "allocation",
409        },
410    }
411}

450struct Buffer {
451    bytes: Vec<u8>,
452    maximum: usize,
453    refusal: Option<ViewError>,
454}
455impl Buffer {
456    fn new(maximum: usize) -> Self {
457        Self {
458            bytes: Vec::new(),
459            maximum,
460            refusal: None,
461        }
462    }
463    fn text(&mut self, value: &str) -> Result<(), RunError> {
464        self.write_all(value.as_bytes())
465            .map_err(|error| self.error(error))
466    }
467    fn string(&mut self, value: &str) -> Result<(), RunError> {
468        serde_json::to_writer(&mut *self, value)
469            .map_err(|error| self.error(io::Error::other(error)))
470    }
471    /// Escape display fragments directly into the bounded record buffer.
472    /// `collect_str` stops formatting as soon as this writer refuses a fragment.
473    fn display_string(&mut self, value: &impl std::fmt::Display) -> Result<(), RunError> {
474        use serde::Serializer as _;
475        serde_json::Serializer::new(&mut *self)
476            .collect_str(value)
477            .map_err(|error| self.error(io::Error::other(error)))
478    }
479    fn number_field(&mut self, key: &str, value: impl std::fmt::Display) -> Result<(), RunError> {
480        self.text(",")?;
481        self.string(key)?;
482        self.text(":")?;
483        self.text(&value.to_string())
484    }
485    fn optional_string(&mut self, value: Option<&str>) -> Result<(), RunError> {
486        if let Some(value) = value {
487            self.string(value)
488        } else {
489            self.text("null")
490        }
491    }
492    fn optional_number(&mut self, value: Option<u64>) -> Result<(), RunError> {
493        if let Some(value) = value {
494            self.text(&value.to_string())
495        } else {
496            self.text("null")
497        }
498    }
499    fn error(&self, error: io::Error) -> RunError {
500        self.refusal
501            .map_or(RunError::Output(error), RunError::JsonRecord)
502    }
503}
504impl Write for Buffer {
505    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
506        let needed = self
507            .bytes
508            .len()
509            .checked_add(bytes.len())
510            .filter(|needed| *needed <= self.maximum);
511        let Some(needed) = needed else {
512            self.refusal = Some(ViewError::Bytes);
513            return Err(io::Error::other("JSON record byte ceiling"));
514        };
515        if needed > self.bytes.capacity() {
516            let capacity = needed
517                .max(self.bytes.capacity().saturating_mul(2))
518                .min(self.maximum);
519            if self
520                .bytes
521                .try_reserve_exact(capacity - self.bytes.len())
522                .is_err()
523            {
524                self.refusal = Some(ViewError::Allocation);
525                return Err(io::Error::other("JSON record allocation"));
526            }
527        }
528        self.bytes.extend_from_slice(bytes);
529        Ok(bytes.len())
530    }
531    fn flush(&mut self) -> io::Result<()> {
532        Ok(())
533    }
534}

536fn write_interruption(
537    out: &mut Buffer,
538    interruption: Option<Interruption>,
539) -> Result<(), RunError> {
540    if let Some(reason) = interruption {
541        let kind = match reason {
542            Interruption::Preparation(_) => "preparation",
543            Interruption::Oracle(_) => "oracle",
544            Interruption::Countermodel(_) => "countermodel",
545            Interruption::Constraint(_) => "constraint",
546            Interruption::Reconstruction(_) => "answer_reconstruction",
547            Interruption::ModelConstruction(_) => "model_construction",
548            Interruption::Objective(_) => "objective",
549            Interruption::PreparedObjective(_) => "prepared_objective",
550            Interruption::Incumbent(_) => "incumbent",
551        };
552        out.text("{\"kind\":")?;
553        out.string(kind)?;
554        out.text(",\"code\":")?;
555        out.string(reason_code(reason))?;
556        out.text(",\"detail\":")?;
557        out.display_string(&reason)?;
558        if let Interruption::ModelConstruction(stop) = reason {
559            match stop {
560                zetesis_solve::ModelConstructionStop::Work { observed, limit } => {
561                    out.text(",\"resource\":\"work\"")?;
562                    out.number_field("required", observed)?;
563                    out.number_field("limit", limit)?;
564                }
565                zetesis_solve::ModelConstructionStop::Bytes { required, limit } => {
566                    out.text(",\"resource\":\"bytes\"")?;
567                    out.number_field("required", required)?;
568                    out.number_field("limit", limit)?;
569                }
570                zetesis_solve::ModelConstructionStop::Control(_) => {}
571            }
572        }
573        out.text("}")?;
574    } else {
575        out.text("null")?;
576    }
577    Ok(())
578}

580fn statistics(out: &mut Buffer, view: &SummaryView<'_>) -> Result<(), RunError> {
581    // Counts are typed and optional; this is a bounded fixed-shape view, not a
582    // serialization of debug text or the human statistics protocol.
583    if view.timings.is_some() {
584        out.text("{\"search\":")?;
585        search_statistics(
586            out,
587            view.search,
588            if view.terminal_execution.is_some() {
589                "terminal_base"
590            } else if view.hybrid_execution.is_some() {
591                "retained_core"
592            } else {
593                "original_theory"
594            },
595        )?;
596        out.text(",\"hybrid_execution\":")?;
597        hybrid_statistics(out, view.hybrid_execution)?;
598        out.text(",\"terminal_execution\":")?;
599        terminal_statistics(out, view.terminal_execution)?;
600        out.text(",\"objective_work\":")?;
601        out.optional_number(view.objective_work)?;
602        out.text(",\"model_construction\":")?;
603        model_construction_statistics(out, view.model_construction)?;
604        out.text(",\"candidate_restrictions\":")?;
605        candidate_statistics(out, view.candidates)?;
606        out.text(",\"execution\":")?;

646fn grounding(out: &mut Buffer, timings: Option<&crate::GroundingTimings>) -> Result<(), RunError> {
647    let Some(timings) = timings else {
648        return out.text("null");
649    };
650    out.text(
651        "{\"schema\":1,\"clock\":\"host_monotonic\",\"scope\":\"eager_formula\",\"measurements\":{",
652    )?;
653    for (index, phase) in crate::GroundingPhase::ALL.into_iter().enumerate() {
654        if index != 0 {
655            out.text(",")?;
656        }
657        out.string(phase.label())?;
658        out.text(":")?;
659        if let Some(value) = timings.get(phase) {
660            out.text("{\"elapsed_ns\":")?;
661            optional_number(out, value.elapsed.map(|duration| duration.as_nanos()))?;
662            out.text(",\"outcomes\":{")?;
663            for (index, outcome) in crate::GroundingOutcome::ALL.into_iter().enumerate() {
664                if index != 0 {
665                    out.text(",")?;
666                }
667                out.string(outcome.label())?;
668                out.text(":")?;
669                optional_number(out, value.count(outcome).map(u128::from))?;
670            }
671            out.text("},\"work\":{")?;
672            for (index, (name, count)) in crate::grounding_timing::work_fields(&value.work)
673                .into_iter()
674                .enumerate()
675            {
676                if index != 0 {
677                    out.text(",")?;
678                }
679                out.string(name)?;
680                out.text(":")?;
681                optional_number(out, count.map(u128::from))?;
682            }
683            out.text("}}")?;
684        } else {
685            out.text("null")?;
686        }
687    }
688    out.text("},\"overhead\":\"included\",\"relational_grounding\":null,\"kernel_time\":null}")
689}

737// Borrow the retained semantic evidence; publication does not manufacture it.
738struct SummaryView<'a> {
739    projection: Option<&'a zetesis_solve::ProjectionStatistics>,
740    completion: Option<Completion>,
741    optimum_proved: bool,
742    published_models: usize,
743    verified_models: Option<u64>,
744    checked: Option<u64>,
745    interruption: Option<Interruption>,
746    optimization: Option<&'a crate::Optimization>,
747    objective_work: Option<u64>,
748    search: Option<&'a zetesis_sat::Statistics>,
749    candidates: Option<zetesis_cpu::CandidateStatistics>,
750    execution: Option<&'a crate::FormulaExecutionStatistics>,
751    hybrid_execution: Option<&'a zetesis_solve::HybridExecutionStatistics>,
752    terminal_execution: Option<&'a zetesis_solve::TerminalExecutionStatistics>,
753    model_construction: Option<&'a zetesis_solve::ModelConstructionStatistics>,
754    lazy_execution: Option<&'a crate::LazyExecutionStatistics>,
755    shared_execution: Option<&'a crate::SharedExecutionStatistics>,
756    closure_execution: Option<&'a crate::ClosureExecutionStatistics>,
757    query_execution: Option<&'a crate::QueryExecutionObservation>,
758    expansion: Option<zetesis_themelios::ExpansionUsage>,
759    timings: Option<&'a PhaseTimings>,
760}
761impl<'a> SummaryView<'a> {
762    fn new(publication: crate::PublicationView<'a>) -> Self {
763        match publication.result {
764            Ok(progress) => {
765                let semantic = progress.semantic();
766                Self {
767                    projection: semantic.and_then(crate::SemanticOutcome::projection),
768                    completion: semantic.and_then(crate::SemanticOutcome::completion),
769                    optimum_proved: semantic.is_some_and(crate::SemanticOutcome::optimum_proved),
770                    published_models: progress.publication.models,
771                    verified_models: semantic.map(crate::SemanticOutcome::verified_models),
772                    checked: semantic.map(crate::SemanticOutcome::candidate_progress),
773                    interruption: semantic.and_then(crate::SemanticOutcome::interruption),
774                    optimization: semantic.and_then(crate::SemanticOutcome::incumbent),
775                    objective_work: semantic.map(crate::SemanticOutcome::objective_work),
776                    search: semantic.and_then(crate::SemanticOutcome::countermodel_statistics),
777                    candidates: semantic.and_then(crate::SemanticOutcome::candidate_statistics),
778                    execution: semantic.and_then(crate::SemanticOutcome::formula_execution),
779                    hybrid_execution: semantic.and_then(crate::SemanticOutcome::hybrid_execution),
780                    terminal_execution: semantic
781                        .and_then(crate::SemanticOutcome::terminal_execution),
782                    model_construction: semantic
783                        .and_then(crate::SemanticOutcome::model_construction),
784                    lazy_execution: semantic.and_then(crate::SemanticOutcome::lazy_execution),
785                    shared_execution: semantic.and_then(crate::SemanticOutcome::shared_execution),
786                    closure_execution: semantic.and_then(crate::SemanticOutcome::closure_execution),
787                    query_execution: semantic.and_then(crate::SemanticOutcome::query_execution),
788                    expansion: progress.expansion,
789                    timings: progress.phase_timings.as_ref(),
790                }
791            }
792            Err(failure) => {
793                let partial = failure.partial_report.as_deref();
794                Self {
795                    projection: failure
796                        .semantic()
797                        .and_then(crate::SemanticOutcome::projection),
798                    completion: partial.and_then(|p| p.completion),
799                    optimum_proved: failure
800                        .semantic()
801                        .is_some_and(crate::SemanticOutcome::optimum_proved),
802                    published_models: partial.map_or(0, |p| p.published_models),
803                    verified_models: partial.map(|p| p.verified_models),
804                    checked: partial.map(|p| p.checked),
805                    interruption: partial.and_then(|p| p.interruption),
806                    optimization: partial.and_then(|p| p.optimization.as_ref()),
807                    objective_work: failure
808                        .semantic()
809                        .map(crate::SemanticOutcome::objective_work),
810                    search: partial.and_then(|p| p.countermodel_statistics.as_ref()),
811                    candidates: partial.and_then(|p| p.candidate_statistics),
812                    execution: partial.and_then(|p| p.formula_execution.as_ref()),
813                    hybrid_execution: partial.and_then(|p| p.hybrid_execution.as_ref()),
814                    terminal_execution: partial.and_then(|p| p.terminal_execution.as_ref()),
815                    model_construction: partial.and_then(|p| p.model_construction.as_ref()),
816                    lazy_execution: partial.and_then(|p| p.lazy_execution.as_ref()),
817                    shared_execution: partial.and_then(|p| p.shared_execution.as_ref()),
818                    closure_execution: partial.and_then(|p| p.closure_execution.as_ref()),
819                    query_execution: partial.and_then(|p| p.query_execution.as_ref()),
820                    expansion: partial.and_then(|p| p.expansion),
821                    timings: failure.phase_timings.as_deref(),
822                }
823            }
824        }
825    }
826}

1128fn query_statistics(
1129    out: &mut Buffer,
1130    observation: Option<&crate::QueryExecutionObservation>,
1131) -> Result<(), RunError> {
1132    let Some(observation) = observation else {
1133        return out.text("null");
1134    };
1135    out.text("{\"backend\":\"cpu\",\"source_batching\":\"independent\",\"snapshot\":")?;
1136    if let Some(stats) = observation.statistics {

1138        if let Some(preparation) = stats.preparation {
1139            out.text("{\"work\":")?;

1142            out.number_field("predicates", preparation.predicates)?;
1143            out.number_field("dense_predicates", preparation.dense_predicates)?;
1144            out.text("}")?;
1145        } else {
1146            out.text("null")?;
1147        }
1148        out.number_field("preparation_builds", stats.preparation_builds)?;
1149        out.number_field("preparation_adoptions", stats.preparation_adoptions)?;
1150        out.number_field("retained_workspaces", stats.retained_workspaces)?;
1151        out.number_field("active_workspaces", stats.active_workspaces)?;
1152        out.number_field("reused_workspaces", stats.reused_workspaces)?;
1153        out.number_field("retained_bytes", stats.retained_bytes)?;
1154        out.number_field("reserved_bytes", stats.reserved_bytes)?;
1155        out.text("}")?;
1156    } else {
1157        out.text("null")?;
1158    }
1159    out.text(",\"fault\":")?;
1160    if let Some(fault) = &observation.fault {
1161        query_fault(out, fault)?;
1162    } else {
1163        out.text("null")?;
1164    }
1165    out.text("}")
1166}

1168fn query_fault(out: &mut Buffer, fault: &zetesis_cpu::BatchError) -> Result<(), RunError> {
1169    use zetesis_cpu::BatchError;
1170    out.text("{\"kind\":")?;
1171    out.string(match fault {
1172        BatchError::Pool(_) => "pool",
1173        BatchError::Busy => "busy",
1174        BatchError::Poisoned => "poisoned",
1175        BatchError::Preparation(_) => "preparation",
1176        BatchError::ClosureStorage { .. } => "closure_storage",
1177        BatchError::Capacity { .. } => "capacity",
1178    })?;
1179    match fault {
1180        BatchError::Preparation(stop) => {
1181            out.text(",\"reason\":")?;
1182            out.string(stop_code(*stop))?;
1183        }
1184        BatchError::ClosureStorage { required, limit } => {
1185            out.number_field("required", *required)?;
1186            out.number_field("limit", *limit)?;
1187        }
1188        BatchError::Capacity { limit, actual } => {
1189            out.number_field("limit", *limit)?;
1190            out.number_field("actual", *actual)?;
1191        }
1192        _ => {}
1193    }
1194    out.text(",\"detail\":")?;
1195    out.string(&fault.to_string())?;
1196    out.text("}")
1197}
