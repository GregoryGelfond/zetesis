9pub(super) fn reasons(
10    out: &mut Buffer,
11    counts: &FormulaResidualStatistics,
12) -> Result<(), RunError> {
13    out.text("{\"fixed_point\":")?;
14    out.text(&counts.fixed_point.to_string())?;
15    out.number_field("round_limit", counts.round_limit)?;
16    out.number_field("work_limit", counts.work_limit)?;
17    out.text("}")
18}

20pub(super) fn write(
21    out: &mut Buffer,
22    handovers: Option<&ResidualHandoverStatistics>,
23) -> Result<(), RunError> {
24    let Some(stats) = handovers else {
25        return out.text("null");
26    };
27    out.text("{\"attempts\":")?;
28    out.text(&stats.attempts.to_string())?;
29    out.number_field("incomplete_attempts", stats.incomplete_attempts)?;
30    out.number_field("committed_slates", stats.committed_slates)?;
31    out.number_field("zero_residual_slates", stats.zero_residual_slates)?;
32    out.number_field("query_work", stats.query_work)?;
33    out.text(",\"complete_work\":")?;
34    out.text(if stats.complete_work { "true" } else { "false" })?;
35    out.text(",\"counts_exact\":")?;
36    out.text(if stats.counts_exact { "true" } else { "false" })?;
37    out.text(",\"last_attempt\":")?;
38    if let Some(last) = &stats.last_attempt {
39        attempt(out, last)?;
40    } else {
41        out.text("null")?;
42    }
43    if let Some(value) = &stats.samples {
44        out.text(",\"samples\":")?;
45        samples::write(out, value)?;
46    }
47    out.text("}")
48}

50fn attempt(out: &mut Buffer, last: &ResidualHandoverAttempt) -> Result<(), RunError> {
51    out.text("{\"ordinal\":")?;
52    out.text(&last.ordinal.to_string())?;
53    out.number_field("slate_sequence", last.slate_sequence)?;
54    for (name, flag) in [
55        ("retry", last.retry),
56        ("decode_complete", last.decode_complete),
57        ("committed", last.committed),
58    ] {
59        out.text(",")?;
60        out.string(name)?;
61        out.text(if flag { ":true" } else { ":false" })?;
62    }
63    out.text(",\"residuals\":")?;
64    reasons(out, &last.residuals)?;
65    out.number_field("open_atoms", last.open_atoms)?;
66    out.number_field("peak_observation_bytes", last.peak_observation_bytes)?;
67    out.text(",\"search\":")?;
68    if let Some(summary) = &last.search {
69        search(out, summary)?;
70    } else {
71        out.text("null")?;
72    }
73    out.text(",\"by_reason\":")?;
74    if let Some(groups) = &last.by_reason {
75        out.text("{\"fixed_point\":")?;
76        reason_search(out, &groups.fixed_point)?;
77        out.text(",\"round_limit\":")?;
78        reason_search(out, &groups.round_limit)?;
79        out.text(",\"work_limit\":")?;
80        reason_search(out, &groups.work_limit)?;
81        out.text("}")?;
82    } else {
83        out.text("null")?;
84    }
85    out.text("}")
86}

88// Both wire records retain the same CPU totals, with a different qualifier.
89fn start_totals(
90    out: &mut Buffer,
91    row: &zetesis_solve::ResidualReasonSearch,
92) -> Result<(), RunError> {
93    out.text("{\"unentered\":")?;
94    out.text(&row.unentered.to_string())?;
95    out.number_field("certified", row.certified)?;
96    out.number_field("exhausted", row.exhausted)?;
97    out.number_field("witnesses", row.witnesses)?;
98    out.number_field("stopped", row.stopped)?;
99    out.number_field("visited", row.visited)?;
100    out.number_field("attempted_splits", row.attempted_splits)?;
101    out.number_field("published_splits", row.published_splits)?;
102    out.number_field("work", row.work)
103}
104
105fn search(out: &mut Buffer, value: &ResidualSearchStatistics) -> Result<(), RunError> {
106    start_totals(out, &value.totals)?;
107    out.text(",\"deepest\":")?;
108    optional_number(out, value.deepest.map(|depth| depth as u128))?;
109    out.text(",\"width\":")?;
110    out.optional_number(value.width)?;
111    out.text(",\"host_time_ns\":")?;
112    optional_number(out, value.host_time.map(|duration| duration.as_nanos()))?;
113    out.text("}")
114}

116fn reason_search(
117    out: &mut Buffer,
118    row: &zetesis_solve::ResidualReasonSearch,
119) -> Result<(), RunError> {
120    start_totals(out, row)?;
121    out.number_field("open_atoms", row.open_atoms)?;
122    out.text("}")
