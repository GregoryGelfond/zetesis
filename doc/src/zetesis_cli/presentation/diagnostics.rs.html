3use super::ColorMode;
4use std::{
5    fmt,

9pub(crate) enum Label {
10    Source,
11    Oracle,
12    Grounding,
13    Backend,
14}
15impl Label {
16    const fn text(self) -> &'static str {
17        match self {
18            Self::Source => "Source",
19            Self::Oracle => "Oracle",
20            Self::Grounding => "Grounding",
21            Self::Backend => "Backend",
22        }
23    }
24}
25
26// A borrowed or owned writer plus one explicit policy, with no environment
27// lookup or terminal discovery. Metadata adds fixed label/style
28// bytes and delegates value formatting and I/O costs to their implementations.
29// Write calls and their failures remain observable in order.
30pub(crate) struct Diagnostics<W> {
31    writer: W,
32    color: ColorMode,
33    quiet: bool,
34    width: std::num::NonZeroUsize,
35    core: Option<Core>,
36}
37#[derive(Clone, Copy, PartialEq, Eq)]
38enum Core {
39    Constraints,
40    TerminalDefinitions,
41}
42impl<W: Write> Diagnostics<W> {
43    pub(crate) const fn new(writer: W, color: ColorMode) -> Self {
44        Self {
45            writer,
46            color,
47            quiet: false,
48            width: std::num::NonZeroUsize::new(80).unwrap(),
49            core: None,
50        }
51    }
52
53    /// Suppress informational setup and execution records; warnings and errors remain.
54    pub(crate) const fn with_quiet(mut self, quiet: bool) -> Self {
55        self.quiet = quiet;
56        self
57    }
58
59    pub(crate) fn for_solve(writer: W, color: ColorMode, options: &crate::Options) -> Self {
60        let execution_details = options.json
61            || (options.stats && matches!(options.statistics_view, crate::StatisticsView::Records));
62        Self::new(writer, color).with_quiet(!execution_details)
63    }
64
65    pub(crate) const fn with_width(mut self, width: std::num::NonZeroUsize) -> Self {
66        self.width = width;
67        self
68    }
69
70    pub(crate) const fn layout(&self) -> zetesis_presentation::Layout {
71        zetesis_presentation::Layout::new(self.width, self.color)
72    }
73
74    pub(crate) fn metadata(&mut self, label: Label, value: fmt::Arguments<'_>) -> io::Result<()> {
75        if self.quiet {
76            Ok(())
77        } else {
78            self.color.metadata(&mut self.writer, label.text(), value)
79        }
80    }
81
82    pub(crate) fn information(&mut self, value: fmt::Arguments<'_>) -> io::Result<()> {
83        if self.quiet {
84            Ok(())
85        } else {
86            writeln!(self.writer, "{value}")
87        }
88    }
89
90    pub(crate) fn diagnostic(&mut self, diagnostic: &impl fmt::Display) -> io::Result<()> {
91        super::source_error::write(&mut self.writer, self.color, diagnostic)
92    }
93}
94
95impl<W: Write> Write for Diagnostics<W> {
96    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
97        self.writer.write(bytes)
98    }
99    fn write_all(&mut self, bytes: &[u8]) -> io::Result<()> {
100        self.writer.write_all(bytes)
101    }
102    fn write_fmt(&mut self, args: fmt::Arguments<'_>) -> io::Result<()> {
103        self.writer.write_fmt(args)
104    }
105    fn flush(&mut self) -> io::Result<()> {
106        self.writer.flush()
107    }
108}

110impl<W: Write> crate::ExecutionObserver for Diagnostics<W> {
111    type Error = io::Error;
112    fn observe(&mut self, observation: crate::ExecutionObservation<'_>) -> Result<(), Self::Error> {
113        use crate::ExecutionObservation as Event;
114        if self.quiet
115            && !matches!(
116                &observation,
117                Event::KeyAnalysisStopped(_)
118                    | Event::ObjectiveUnavailable(_)
119                    | Event::ObjectiveBoundStopped(_)
120                    | Event::ObjectiveTheoryMismatch
121                    | Event::ObjectiveRestrictionStopped(_)
122            )
123        {
124            return Ok(());
125        }
126        match observation {
127            Event::StaticGrounding { requested, atoms, rules, limits } => self.metadata(
128                Label::Grounding,
129                format_args!("requested={}, effective=eager (static atoms={atoms}, rules={rules}; lowering caps atoms={}, rules={}, substitutions={})", requested.label(), limits.max_atoms, limits.max_ground_rules, limits.max_substitutions),
130            ),
131            Event::LazyGrounding { requested } => self.metadata(Label::Grounding,
132                format_args!("requested={}, effective=lazy (source joins; no complete ground-rule store)", requested.label())),
133            Event::TerminalDefinitions { requested, deferred_templates } => {
134                self.core = Some(Core::TerminalDefinitions);
135                self.metadata(Label::Grounding, format_args!("requested={}, effective=eager_base_terminal_definitions (eager base; {deferred_templates} terminal definitions reconstructed on the host before each original answer)", requested.label()))
136            }
137            Event::HybridGrounding { requested, streamed_templates, streamed_instances } => {
138                self.core = Some(Core::Constraints);
139                self.metadata(Label::Grounding,
140                    format_args!("requested={}, effective=hybrid (eager retained core; streamed constraints: {streamed_templates} templates, {streamed_instances} admitted instances)", requested.label()))
141            }
142            Event::CpuClosure { grounder: crate::Grounder::Eager, workers, .. } => self.metadata(
143                Label::Backend, format_args!("cpu (eager static closure scans, {workers} workers)")),
144            Event::CpuClosure { batching: crate::SourceBatching::Independent, workers, .. } => self.metadata(
145                Label::Backend, format_args!("cpu (lazy source joins, {workers} workers)")),
146            Event::CpuClosure { batching, workers, .. } => self.metadata(Label::Backend,
147                format_args!("cpu (shared {} source rounds, {workers} workers; collective source and per-world evaluation budgets)", batching.label())),
148            Event::CpuFormula { oracle, grounder, search } => {
149                let oracle = if oracle == crate::Oracle::Auto { "Ferraris reduct membership" } else { "Ferraris reduct countermodel" };
150                if self.core == Some(Core::TerminalDefinitions) {
151                    self.metadata(Label::Backend, format_args!("cpu; base oracle: {oracle}; search: {}; base grounding: eager; original answer reconstruction: host", search.label()))
152                } else if self.core == Some(Core::Constraints) {
153                    self.metadata(Label::Backend, format_args!("cpu; retained-core oracle: {oracle}; search: {}; core grounding: eager", search.label()))
154                } else {
155                    self.metadata(Label::Backend, format_args!("cpu; oracle: {oracle}; search: {}; grounder: eager (requested {})", search.label(), grounder.label()))
156                }
157            }
158            Event::ParallelRegions { workers } => writeln!(self,
159                "Parallel regions: {workers} workers; models arrive in the schedule's order"),
160            Event::ParallelProposals { workers } => writeln!(self,
161                "Parallel candidate production: {workers} workers; joined before batch membership checking"),
162            Event::ExactCompletion { workers, max_scratch_bytes } => writeln!(self,
163                "Exact completion: requested workers={workers}; bounded logical scratch bytes={max_scratch_bytes}"),
164            #[cfg(feature = "gpu")]
165            Event::LazyDeviceGrounding { requested } => self.metadata(Label::Grounding,
166                format_args!("requested={}, effective=lazy (host source joins; per-world device consequences; no complete ground-rule store)", requested.label())),
167            #[cfg(feature = "gpu")]
168            Event::DeviceClosure { adapter, static_counts } => match static_counts {
169                None => self.metadata(Label::Backend, format_args!("gpu ({}, {}; vendor=0x{:04x}; lazy immutable reduct rounds)", adapter.name, adapter.backend, adapter.vendor_id)),
170                Some((atoms, rules)) => self.metadata(Label::Backend, format_args!("gpu ({}, {}; vendor=0x{:04x}; static atoms={atoms}, rules={rules})", adapter.name, adapter.backend, adapter.vendor_id)),
171            },
172            #[cfg(feature = "gpu")]
173            Event::DeviceFormula { adapter, grounder, search, batch_size, completion_workers, .. } => self.metadata(Label::Backend,
174                format_args!("hybrid GPU propagation + exact CPU residual search ({}, {}; vendor=0x{:04x}); oracle: Ferraris reduct countermodel; search: {}; grounder: eager (requested {}); batch={batch_size}; CPU completion requested workers={completion_workers}", adapter.name, adapter.backend, adapter.vendor_id, search.label(), grounder.label())),
175            #[cfg(feature = "gpu")]
176            Event::DeviceTight { adapter, grounder, search, batch_size } => self.metadata(Label::Backend,
177                format_args!("GPU tight support ({}, {}; vendor=0x{:04x}); oracle: Ferraris ranked support; search: {}; grounder: eager (requested {}); batch={batch_size}", adapter.name, adapter.backend, adapter.vendor_id, search.label(), grounder.label())),
178            Event::Formula { atoms, nodes, roots, keyed_constraints } if self.core == Some(Core::TerminalDefinitions) => writeln!(self,
179                "Base formula: {atoms} atoms, {nodes} nodes, {roots} roots; {keyed_constraints} constraints asked by key; terminal definitions reconstructed separately"),
180            Event::Formula { atoms, nodes, roots, keyed_constraints } if self.core == Some(Core::Constraints) => writeln!(self,
181                "Retained core formula: {atoms} atoms, {nodes} nodes, {roots} roots; {keyed_constraints} constraints asked by key; streamed constraints checked separately"),
182            Event::Formula { atoms, nodes, roots, keyed_constraints: 0 } => writeln!(self, "Formula: {atoms} atoms, {nodes} nodes, {roots} roots"),
183            Event::Formula { atoms, nodes, roots, keyed_constraints } => writeln!(self,
184                "Formula: {atoms} atoms, {nodes} nodes, {roots} roots; {keyed_constraints} constraints asked by key"),
185            Event::KeyAnalysisStopped(stop) => writeln!(self,
186                "Keyed constraints: the key analysis stopped, {stop}; every constraint not yet asked was grounded as written"),
187            Event::TightMembership if self.core == Some(Core::TerminalDefinitions) => writeln!(self, "Base membership: checked tight support certificate; full reconstruction still pending"),
188            Event::TightMembership if self.core == Some(Core::Constraints) => writeln!(self, "Core membership: checked tight support certificate; original constraints still pending"),
189            Event::TightMembership => writeln!(self, "Membership: checked tight support certificate over the original theory"),
190            Event::PositiveMembership if self.core == Some(Core::TerminalDefinitions) => writeln!(self, "Base membership: positive atomic-head theory; full reconstruction still pending"),
191            Event::PositiveMembership if self.core == Some(Core::Constraints) => writeln!(self, "Core membership: positive atomic-head theory; original streamed constraints still pending"),
192            Event::PositiveMembership => writeln!(self, "Membership: positive atomic-head theory; least consequences with original constraints"),
193            Event::GeneralMembership(error) => writeln!(self, "Membership: general reduct; optional class certificate refused: {error}"),
194            Event::ObjectiveUnavailable(error) => writeln!(self, "Objective preparation unavailable: {error}; detailed scoring is subject to the remaining objective allowance"),
195            Event::ObjectiveBoundStopped(error) => writeln!(self, "Objective pruning stopped: {error}; exact search continues"),
196            Event::ObjectiveTheoryMismatch => writeln!(self, "Objective pruning stopped: original theory mismatch; exact search continues"),
197            Event::ObjectiveRestrictionStopped(error) => writeln!(self, "Objective pruning stopped: {error}; exact search continues"),
198            Event::ObjectiveBound { restrictions, costs, work } => writeln!(self, "Objective pruning: bound {restrictions}; cost <= {costs:?}; construction work {work}"),
199        }
200    }
201}
