1//! The diagnostics writer: whole write units, and slices passed through.
2
3use std::{
4    fmt,
5    io::{self, Write},
6};
7
8/// The bytes a unit's buffer holds before it first grows, sized for the
9/// execution-detail records of an ordinary run. A longer unit, such as the
10/// statistics report or a record carrying an error's text, grows as a `Vec`
11/// does, at least doubling.
12const UNIT_CAPACITY: usize = 256;
13
14/// The diagnostics writer. It admits whole write units, and slices passed
15/// through unchanged, and nothing else.
16///
17/// Each `write_fmt` call and each [`Self::write_whole`] call reaches the
18/// underlying writer as one `write_all` of the unit's bytes, or, if the unit's
19/// rendering fails, not at all. `write` and `write_all` hand their slice to the
20/// writer unchanged, and `flush` delegates; `Write`'s provided methods reach the
21/// writer only through these. Nothing is retained between calls.
22pub(super) struct UnitWriter<W> {
23    writer: W,
24}
25
26impl<W: Write> UnitWriter<W> {
27    pub(super) const fn new(writer: W) -> Self {
28        Self { writer }
29    }
30
31    /// Render one write unit whole, then hand it to the writer in one
32    /// `write_all` call. If the rendering fails, nothing of the unit is written
33    /// and its error is returned.
34    ///
35    /// The unit's buffer belongs to this call: it is created here with
36    /// [`UNIT_CAPACITY`] bytes and dropped on return. So no unit's bytes reach
37    /// the writer after its call has returned, and no capacity outlives the
38    /// unit.
39    pub(super) fn write_whole(
40        &mut self,
41        render: impl FnOnce(&mut UnitBuffer) -> io::Result<()>,
42    ) -> io::Result<()> {
43        let mut unit = UnitBuffer {
44            bytes: Vec::with_capacity(UNIT_CAPACITY),
45        };
46        render(&mut unit)?;
47        self.writer.write_all(&unit.bytes)
48    }
49}
50
51impl<W: Write> Write for UnitWriter<W> {
52    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
53        self.writer.write(bytes)
54    }
55    fn write_all(&mut self, bytes: &[u8]) -> io::Result<()> {
56        self.writer.write_all(bytes)
57    }
58    fn write_fmt(&mut self, args: fmt::Arguments<'_>) -> io::Result<()> {
59        self.write_whole(|unit| unit.write_fmt(args))
60    }
61    fn flush(&mut self) -> io::Result<()> {
62        self.writer.flush()
63    }
64}
65
66/// One write unit's bytes while it is rendered.
67///
68/// `write` and `write_all` append. Its `write_fmt` renders through a sink that
69/// cannot fail, so a formatting error is a `Display` implementation's own; it
70/// becomes the presentation's formatting failure, an `Other` error, where the
71/// standard library would panic.
72pub(crate) struct UnitBuffer {
73    bytes: Vec<u8>,
74}
75
76impl Write for UnitBuffer {
77    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
78        self.bytes.extend_from_slice(bytes);
79        Ok(bytes.len())
80    }
81    fn write_all(&mut self, bytes: &[u8]) -> io::Result<()> {
82        self.bytes.extend_from_slice(bytes);
83        Ok(())
84    }
85    fn write_fmt(&mut self, args: fmt::Arguments<'_>) -> io::Result<()> {
86        fmt::write(&mut Append(&mut self.bytes), args)
87            .map_err(|fmt::Error| super::super::formatting_failure())
88    }
89    fn flush(&mut self) -> io::Result<()> {
90        Ok(())
91    }
92}
93
94/// Appends rendered text to a unit's bytes; it never fails.
95struct Append<'a>(&'a mut Vec<u8>);
96
97impl fmt::Write for Append<'_> {
98    fn write_str(&mut self, text: &str) -> fmt::Result {
99        self.0.extend_from_slice(text.as_bytes());
100        Ok(())
101    }
102}
