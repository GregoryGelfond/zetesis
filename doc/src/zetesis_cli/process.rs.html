9/// Process adapter. Exit 0 means a completed request; test uses 1 for a
10/// completed check with non-passing evidence. Exit 2 is an input/backend/output
11/// error, and 3 interrupted search or publication. Satisfiability and coverage are printed
12/// independently; this is not clingo's numeric exit-code protocol.
13/// Standard output is explicitly flushed before returning. A flush failure is
14/// an output error, independently of any established semantic outcome.
15#[must_use]
16pub fn entry() -> ExitCode {
17    let mut invocation = match Invocation::try_parse_from(std::env::args_os()) {
18        Ok(invocation) => invocation,
19        Err(error) => {
20            let status = if error.use_stderr() {
21                ExitCode::from(2)
22            } else {
23                ExitCode::SUCCESS
24            };
25            return if error.print().is_ok() {
26                status
27            } else {
28                ExitCode::from(2)
29            };
30        }
31    };
32    let output = io::stdout().lock();
33    let output_terminal = output.is_terminal();
34    let no_color = std::env::var_os("NO_COLOR");
35    let term = std::env::var_os("TERM");
36    let disabled = color_disabled(no_color.as_deref(), term.as_deref());
37    let diagnostics = io::stderr().lock();
38    let (mode, json) = match &invocation {
39        Invocation::Solve(options) => (options.color, options.json),
40        Invocation::Devices => (ColorMode::Auto, false),
41        Invocation::Test(command) => (command.color(), command.json()),
42    };
43    let colors = Streams::resolve(
44        mode.human(json),
45        output_terminal,
46        diagnostics.is_terminal(),
47        disabled,
48    );
49    let columns = std::env::var("COLUMNS").ok();
50    let width = terminal_width(columns.as_deref());
51    let layout = zetesis_presentation::Layout::new(width, colors.output);
52    let mut output = buffered_output(output, output_terminal);
53    let mut diagnostics = match &invocation {
54        Invocation::Solve(options) => {
55            Diagnostics::for_solve(diagnostics, colors.diagnostics, options)
56        }
57        Invocation::Devices | Invocation::Test(_) => {
58            Diagnostics::new(diagnostics, colors.diagnostics)
59        }
60    }
61    .with_width(width);
62    let cancelled = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
63    // Only conformance checks own child solvers that an interrupt must settle.
64    #[cfg(any(target_os = "linux", target_os = "macos"))]
65    let _interrupts = if matches!(invocation, Invocation::Test(_)) {
66        match zetesis_validation::process::interrupts::Interrupts::install(&cancelled) {
67            Ok(interrupts) => Some(interrupts),
68            Err(error) => {
69                let _ = diagnostics.diagnostic(&error);
70                return ExitCode::from(2);
71            }
72        }
73    } else {
74        None
75    };
76    let result = match &mut invocation {
77        Invocation::Solve(options) => {
78            options.color = colors.output;
79            run_input(options, &mut output, &mut diagnostics)
80                .map(|outcome| publication_status(&outcome))
81        }
82        Invocation::Devices => crate::devices::devices_with_color(&mut output, colors.output)
83            .map(|()| ExitCode::SUCCESS)
84            .map_err(RunFailure::from),
85        Invocation::Test(command) => {
86            let result = crate::testing::execute_with_cancellation(
87                command,
88                layout,
89                &mut output,
90                &mut diagnostics,
91                cancelled.as_ref(),
92            );
93            Ok(command_status(
94                result.map(|passed| passed == crate::testing::Completion::Passed),
95                &mut diagnostics,
96            ))
97        }
98    };
99    finish_output(output, result, &mut diagnostics)
100}

102fn command_status<E: std::fmt::Display>(
103    result: Result<bool, E>,
104    diagnostics: &mut Diagnostics<impl Write>,
105) -> ExitCode {
106    match result {
107        Ok(true) => ExitCode::SUCCESS,
108        Ok(false) => ExitCode::FAILURE,
109        Err(error) => {
110            let _ = diagnostics.diagnostic(&error);
111            ExitCode::from(2)
112        }
113    }
114}

116fn publication_status(outcome: &PublicationOutcome) -> ExitCode {
117    if matches!(outcome, PublicationOutcome::Stopped(_))
118        || outcome.semantic().completion() == Some(Completion::Interrupted)
119    {
120        ExitCode::from(3)
121    } else {
122        ExitCode::SUCCESS
123    }
124}

126// Fixed process-only staging for redirected output, independent of model count
127// and per-record ceilings. Terminal bytes bypass staging to retain prompt output.
128const OUTPUT_BUFFER_BYTES: usize = 8 * 1024;
129
130fn buffered_output<W: Write>(output: W, terminal: bool) -> BufWriter<W> {
131    BufWriter::with_capacity(if terminal { 0 } else { OUTPUT_BUFFER_BYTES }, output)
132}

134fn finish_output(
135    mut output: BufWriter<impl Write>,
136    result: Result<ExitCode, RunFailure>,
137    diagnostics: &mut Diagnostics<impl Write>,
138) -> ExitCode {
139    let flushed = output.flush();
140    // Do not retry a failed buffered write invisibly from BufWriter::drop.
141    let _ = output.into_parts();
142    let status = match result {
143        Ok(status) => status,
144        Err(error) => {
145            let _ = diagnostics.diagnostic(&error);
146            if let Some(secondary) = error.secondary_output {
147                let _ = diagnostics.diagnostic(&format_args!("secondary output: {secondary}"));
148            }
149            ExitCode::from(2)
150        }
151    };
152    match flushed {
153        Ok(()) => status,
154        Err(error) => {
155            let _ = diagnostics.diagnostic(&format_args!("output flush: {error}"));
156            ExitCode::from(2)
157        }
158    }
159}

161fn run_input(
162    options: &Options,
163    output: &mut impl Write,
164    diagnostics: &mut Diagnostics<impl Write>,
165) -> Result<PublicationOutcome, RunFailure> {
166    let (input, cancellation) = match load_input(options).and_then(|input| {
167        process_cancellation(options.time_limit, Instant::now())
168            .map(|cancellation| (input, cancellation))
169    }) {
170        Ok(input) => input,
171        Err(error) => {
172            let failure = error.into();
173            return Err(if options.json {
174                crate::output::input_failure(output, failure, options)
175            } else {
176                failure
177            });
178        }
179    };
180    let result = match input {
181        Input::Source(source) => crate::driver::run_source_with_writer(
182            source,
183            options,
184            output,
185            diagnostics,
186            &cancellation,
187        ),
188        Input::Bundle(bundle) => crate::driver::run_bundle_with_writer(
189            bundle,
190            options,
191            output,
192            diagnostics,
193            &cancellation,
194        ),
195    };
196    result.map_err(crate::PublicationFailure::into_legacy)
197}

199fn process_cancellation(
200    seconds: Option<u64>,
201    start: Instant,
202) -> Result<zetesis_cpu::Cancellation, RunError> {
203    let Some(seconds) = seconds else {
204        return Ok(zetesis_cpu::Cancellation::default());
205    };
206    let deadline = start
207        .checked_add(Duration::from_secs(seconds))
208        .ok_or(RunError::TimeLimitRange { seconds })?;
209    zetesis_cpu::Cancellation::with_deadline(deadline).map_err(RunError::DeadlineTimer)
210}

212enum Input {
213    Source(String),
214    Bundle(SourceBundle),
215}
216
217fn load_input(options: &Options) -> Result<Input, RunError> {
218    if !options.additional_inputs.is_empty()
219        && std::iter::once(&options.input)
220            .chain(&options.additional_inputs)
221            .any(|path| path.as_os_str() == "-")
222    {
223        return Err(RunError::MixedStandardInput);
224    }
225    crate::SolveConfig::from(options).validate()?;
226    if options.input.as_os_str() == "-" {
227        let source = read_source(options).map_err(RunError::Input)?;
228        Ok(Input::Source(source))
229    } else {
230        let bundle = SourceBundle::load_many(
231            std::iter::once(&options.input).chain(&options.additional_inputs),
232            BundleLimits {
233                max_roots: options.max_source_roots,
234                max_files: options.max_source_files,
235                max_file_bytes: options.max_source_bytes,
236                max_total_bytes: options.max_total_source_bytes,
237                max_include_depth: options.max_include_depth,
238            },
239        )
240        .map_err(RunError::BundleLoad)?;
241        Ok(Input::Bundle(bundle))
242    }
243}

245fn read_source(options: &Options) -> io::Result<String> {
246    read_text(io::stdin().lock(), options.max_source_bytes)
247}
248
249fn read_text(reader: impl Read, limit: usize) -> io::Result<String> {
250    let maximum = u64::try_from(limit).unwrap_or(u64::MAX).saturating_add(1);
251    let mut source = Vec::new();
252    reader.take(maximum).read_to_end(&mut source)?;
253    // Admission counts bytes. Decoding a truncated multibyte character must not
254    // replace the already established byte-limit refusal with a UTF-8 error.
255    if source.len() > limit {
256        return Err(io::Error::new(
257            io::ErrorKind::InvalidInput,
258            "source byte limit exceeded",
259        ));
260    }
261    String::from_utf8(source).map_err(|_| {
262        io::Error::new(
263            io::ErrorKind::InvalidData,
264            "standard input is not valid UTF-8",
265        )
266    })
267}
