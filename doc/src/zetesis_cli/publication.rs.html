28/// Run `input`'s session under `config` and publish its answers. The
29/// session uses `resources`, which the command line fills with an adapter
30/// enumeration begun before reading; every library entry point passes the
31/// default.
32pub(crate) fn solve(
33    input: PreparedInput<'_>,
34    config: &PublicationConfig,
35    resources: RunResources<'_>,
36    renderer: &mut impl AnswerRenderer,
37    diagnostics: &mut Diagnostics<impl Write>,
38    cancellation: &Cancellation,
39    phases: &Recorder,
40) -> Result<Progress, PublicationFailure> {
41    let selection = OutputSelection::default();
42    let observations = ObservationProgram::default();
43    let metadata = input.metadata();
44    let mut display = crate::display::Display::new(
45        metadata.map_or(&selection, zetesis_themelios::SourceMetadata::output),
46        metadata.map_or(
47            &observations,
48            zetesis_themelios::SourceMetadata::observations,
49        ),
50        config.observations,
51        cancellation,
52    );
53    let mut request = Session::builder(input, config.solve, cancellation.clone())
54        .resources(resources.execution)
55        .measurements(phases);
56    if config.mode == PublicationMode::Count {
57        request = request.counting();
58    } else if input
59        .projection()
60        .is_some_and(zetesis_themelios::PreparedProjection::is_explicit)
61    {
62        request = request.projected(zetesis_solve::ProjectionLimits {
63            max_keys: usize::try_from(config.solve.max_projection_entries).unwrap_or(usize::MAX),
64            max_bytes: config.solve.max_projection_bytes,
65            max_work: config.solve.max_search_work,
66        });
67    }
68    let mut configuration = Configuration::new(config.solve.workers);
69    let mut session = request.start_observed(&mut Observer::new(
70        renderer,
71        diagnostics,
72        &mut configuration,
73        phases,
74    ))?;
75    let mut progress = Progress::new();
76    let mut evidence = resources
77        .evidence_request
78        .and_then(|path| residual_evidence::Publication::open(path, &mut session).ok());
79    loop {
80        let next = session.next_observed(&mut Observer::new(
81            renderer,
82            diagnostics,
83            &mut configuration,
84            phases,
85        ));
86        progress.apply(session.progress());
87        if evidence.is_some()
88            && let Some(receipt) = session.finish_residual_evidence()
89            && let Some(publication) = evidence.take()
90            && receipt.is_ok()
91        {
92            // Artifact failure is a separate benchmark outcome. The primary
93            // answer stream and its exit classification remain authoritative.
94            let _ = publication.publish();
95        }
96        match next {
97            Some(Ok(_)) if config.mode == PublicationMode::Count => {}
98            Some(Ok(answer)) => {
99                let result = phases.measure(SolvePhase::ObservationOutput, || {
100                    display.write(
101                        renderer,
102                        progress.publication.models + 1,
103                        answer.interpretation(),
104                        answer.score(),
105                    )
106                });
107                match result {
108                    Ok(std::ops::ControlFlow::Continue(())) => progress.publication.models += 1,
109                    Ok(std::ops::ControlFlow::Break(stop)) => {
110                        progress.stop = Some(stop);
111                        break;
112                    }
113                    Err(error) => return Err(progress.fail(error)),
114                }
115            }
116            Some(Err(error)) => return Err(progress.fail((*error.cause).into())),
117            None => break,
118        }
119    }
120    complete(renderer, diagnostics, progress, phases)
121}

123/// Counting has no partial stdout record; make an unavailable total explicit
124/// on diagnostics even when ordinary informational records are quiet. This
125/// also handles admission's early-control Progress before a session existed.
126pub(crate) fn count_stop(
127    result: Result<Progress, PublicationFailure>,
128    diagnostics: &mut Diagnostics<impl Write>,
129) -> Result<Progress, PublicationFailure> {
130    let progress = result?;
131    let written = if let Some(stop) = &progress.stop {
132        diagnostics.diagnostic(&format_args!("answer count unavailable: {}", stop.reason()))
133    } else if progress
134        .semantic()
135        .and_then(crate::SemanticOutcome::complete_answer_count)
136        .is_none()
137    {
138        if let Some(reason) = progress
139            .semantic()
140            .and_then(crate::SemanticOutcome::interruption)
141        {
142            diagnostics.diagnostic(&format_args!("answer count unavailable: {reason}"))
143        } else {
144            diagnostics
145                .diagnostic(&"answer count unavailable: search did not establish complete coverage")
146        }
147    } else {
148        return Ok(progress);
149    };
150    match written {
151        Ok(()) => Ok(progress),
152        Err(error) => Err(progress.fail(error.into())),
153    }
154}

156fn complete(
157    renderer: &mut impl AnswerRenderer,
158    diagnostics: &mut Diagnostics<impl Write>,
159    mut progress: Progress,
160    phases: &Recorder,
161) -> Result<Progress, PublicationFailure> {
162    if progress.stop.is_none()
163        && let Err(cause) = progress.completion()
164    {
165        return Err(progress.fail(cause));
166    }
167    // Human summaries run before the later statistics snapshot. Supply the
168    // available timing evidence now without claiming that output has finished.
169    progress.phase_timings = phases.snapshot();
170    let _output = phases.enter(SolvePhase::ObservationOutput);
171    let result = (|| {
172        if let Some(projection) = progress
173            .semantic()
174            .and_then(crate::SemanticOutcome::projection)
175        {
176            diagnostics.information(format_args!(
177                "Projection: {} representatives, {} duplicate answers, complete={}",
178                projection.representatives, projection.duplicates, projection.complete
179            ))?;
180        }
181        if let Some(statistics) = progress
182            .semantic()
183            .and_then(crate::SemanticOutcome::countermodel_statistics)
184        {
185            diagnostics.information(format_args!(
186                "Reduct search: {} work, {} decisions, {} classical candidates, {} reduct queries, {} countermodels",
187                statistics.search.work,
188                statistics.search.decisions,
189                statistics.candidates,
190                statistics.countermodel_queries,
191                statistics.countermodels
192            ))?;
193        }
194        if renderer.summary_stage() == SummaryStage::SearchFinished {
195            acknowledge(renderer, &mut progress)?;
196        }
197        Ok(())
198    })();
199    match result {
200        Ok(()) => Ok(progress),
201        Err(error) => Err(progress.fail(error)),
202    }
203}

205pub(crate) fn check_cancellation(
206    renderer: &mut impl AnswerRenderer,
207    diagnostics: &mut Diagnostics<impl Write>,
208    cancellation: &Cancellation,
209    phases: &Recorder,
210) -> Result<Option<Progress>, PublicationFailure> {
211    match cancellation.poll() {
212        Ok(()) => Ok(None),
213        Err(stop) => interrupted(stop, renderer, diagnostics, phases).map(Some),
214    }
215}

217/// Final reporting never replaces an earlier failure with a reporting failure.
218pub(crate) fn finalize(
219    renderer: &mut impl AnswerRenderer,
220    mut result: Result<Progress, PublicationFailure>,
221) -> Result<crate::PublicationOutcome, PublicationFailure> {
222    if renderer.summary_stage() == SummaryStage::Finalized {
223        match renderer.finish(PublicationView {
224            result: result.as_ref(),
225        }) {
226            Ok(SummaryDelivery::Accepted) => match &mut result {
227                Ok(progress) => progress.publication.summary = true,
228                Err(failure) => failure.acknowledge_summary(),
229            },
230            Ok(SummaryDelivery::Omitted) => {}
231            Err(error) => {
232                return Err(match result {
233                    Ok(progress) => progress.fail(error),
234                    Err(mut failure) => {
235                        failure.record_summary(match error {
236                            crate::RunError::Output(error) => error,
237                            other => std::io::Error::other(other),
238                        });
239                        failure
240                    }
241                });
242            }
243        }
244    }
245    result.and_then(Progress::finalize)
246}
247
248fn acknowledge(
249    renderer: &mut impl AnswerRenderer,
250    progress: &mut Progress,
251) -> Result<(), crate::RunError> {
252    if renderer.finish(PublicationView {
253        result: Ok(progress),
254    })? == SummaryDelivery::Accepted
255    {
256        progress.publication.summary = true;
257    }
258    Ok(())
259}

261/// Publish an interruption established by a checked preparation operation.
262/// Callers pass its typed stop, rather than polling after an unrelated failure;
263/// this preserves the operation's diagnostic and failure precedence.
264pub(crate) fn interrupted(
265    reason: zetesis_algebra::Stop,
266    renderer: &mut impl AnswerRenderer,
267    diagnostics: &mut Diagnostics<impl Write>,
268    phases: &Recorder,
269) -> Result<Progress, PublicationFailure> {
270    let mut progress = Progress::new();
271    progress.apply(crate::SemanticOutcome::interrupted_before_start(
272        crate::Interruption::Preparation(reason),
273    ));
274    complete(renderer, diagnostics, progress, phases)
275}
