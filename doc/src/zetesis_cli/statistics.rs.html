8fn header(
9    sink: &mut impl Write,
10    options: &Options,
11    config: &crate::SolveConfig,
12    elapsed: Duration,
13) -> io::Result<()> {
14    writeln!(
15        sink,
16        "Statistics: zetesis {}; GPU compiled={}",
17        env!("CARGO_PKG_VERSION"),
18        cfg!(feature = "gpu")
19    )?;
20    writeln!(
21        sink,
22        "  requested: backend={}; oracle={}; grounder={}; search={}",
23        options.backend.label(),
24        options.oracle.label(),
25        options.grounder.label(),
26        options.search.label()
27    )?;
28    writeln!(
29        sink,
30        "  configured: workers={}; batch={}; displayed models={} (0=all)",
31        options.workers, config.batch_size, options.models
32    )?;
33    writeln!(
34        sink,
35        "  memory allowance: {} bytes (host physical memory {}); shared policy for named input, grounding, solving and output capacities; not resident memory",
36        options.memory,
37        crate::options::host_memory()
38            .map_or_else(|| "unreported".to_owned(), |bytes| bytes.to_string())
39    )?;
40    writeln!(
41        sink,
42        "  formula joins: requested={}; scope=completed_eager_support",
43        match options.formula_joins {
44            zetesis_themelios::JoinStrategy::Indexed => "indexed",
45            zetesis_themelios::JoinStrategy::Table => "table",
46        }
47    )?;
48    limits(sink, options, config)?;
49    writeln!(
50        sink,
51        "  driver wall time: {:.3} ms (admission, search and output; excludes source loading and statistics)",
52        elapsed.as_secs_f64() * 1000.0
53    )?;
54    Ok(())
55}

57pub(crate) fn write_progress(
58    sink: &mut impl Write,
59    options: &Options,
60    result: Result<&crate::failure::Progress, &PublicationFailure>,
61    elapsed: Duration,
62) -> io::Result<()> {
63    let progress = match result {
64        Ok(progress) => progress,
65        Err(failure) => return write_detailed(sink, options, Err(failure), elapsed),
66    };
67    let semantic = progress
68        .semantic()
69        .ok_or_else(|| io::Error::other("statistics require semantic progress"))?;
70    let config = crate::SolveConfig::from(options);
71    header(sink, options, &config, elapsed)?;
72    if let Some(stop) = &progress.stop {
73        writeln!(
74            sink,
75            "  publication: incomplete; {stop}; search completion={:?}",
76            semantic.completion()
77        )?;
78    } else {
79        writeln!(
80            sink,
81            "  completion: {}",
82            match progress.completion().map_err(io::Error::other)? {
83                Completion::Exhausted => "exhausted",
84                Completion::RequestedModels => "requested models reached (partial coverage)",
85                Completion::Interrupted => "interrupted (partial coverage)",
86            }
87        )?;
88    }
89    details(
90        sink,
91        options,
92        &config,
93        &Details {
94            models: progress.publication.models,
95            checked: semantic.candidate_progress(),
96            optimum_proved: semantic.optimum_proved(),
97            interruption: semantic.interruption(),
98            discovered_gate_atoms: semantic.discovered_gate_atoms(),
99            expansion: progress.expansion,
100            candidate_statistics: semantic.candidate_statistics(),
101            countermodel_statistics: semantic.countermodel_statistics(),
102            formula_execution: semantic.formula_execution(),
103            hybrid_execution: semantic.hybrid_execution(),
104            terminal_execution: semantic.terminal_execution(),
105            model_construction: semantic.model_construction(),
106            lazy_execution: semantic.lazy_execution(),
107            shared_execution: semantic.shared_execution(),
108            closure_execution: semantic.closure_execution(),
109            query_execution: semantic.query_execution(),
110            optimization: semantic.incumbent(),
111        },
112    )
113}

115pub(crate) fn write_detailed(
116    sink: &mut impl Write,
117    options: &Options,
118    result: Result<&Report, &PublicationFailure>,
119    elapsed: Duration,
120) -> io::Result<()> {
121    let config = crate::SolveConfig::from(options);
122    header(sink, options, &config, elapsed)?;
123    match result {
124        Ok(report) => completed(sink, options, &config, report),
125        Err(failure) => {
126            writeln!(sink, "  status: failed; completion=unavailable")?;
127            if let Some(partial) = &failure.partial_report {
128                writeln!(
129                    sink,
130                    "  partial execution: published models={}; verified models={}; search completion={:?}; summary published={}",
131                    partial.published_models,
132                    partial.verified_models,
133                    partial.completion,
134                    partial.summary_published,
135                )?;
136                details(sink, options, &config, &Details::from(partial.as_ref()))?;
137            } else {
138                writeln!(
139                    sink,
140                    "  effective execution: unavailable; counters=unavailable"
141                )?;
142            }
143            writeln!(sink, "  failure: {}", failure.cause)
144        }
145    }
146}

148/// The public policy and the actual storage capacities derived from it.
149fn limits(sink: &mut impl Write, o: &Options, c: &crate::SolveConfig) -> io::Result<()> {
150    writeln!(
151        sink,
152        "  work policy: cooperative cancellation; no cumulative operation budget"
153    )?;
154    if let Some(seconds) = o.time_limit {
155        writeln!(
156            sink,
157            "  requested process time limit: {seconds} s (cooperative; begins after input loading)"
158        )?;
159    }
160    let resources = o.resources();
161    let source = resources.bundle_limits();
162    let formula = resources.formula_limits();
163    writeln!(
164        sink,
165        "  configured storage: source bytes/file={}; source bytes/graph={}; support bytes={}; formula nodes={}; formula operands={}",
166        source.max_file_bytes,
167        source.max_total_bytes,
168        formula.max_support_bytes,
169        formula.theory.max_nodes,
170        formula.theory.max_operands
171    )?;
172    writeln!(
173        sink,
174        "  configured execution storage: candidate bytes={}; reduct bytes/owner={}; completion bytes={}; closure bytes/worker={}; closure bytes/collective={}; batch bytes={}",
175        c.max_candidate_bytes,
176        c.max_reduct_bytes,
177        c.max_completion_scratch_bytes,
178        c.max_closure_bytes,
179        c.max_closure_batch_bytes,
180        c.max_batch_bytes
181    )?;
182    writeln!(
183        sink,
184        "  configured publication storage: model bytes={}; incumbent bytes={}; observation bytes={}; JSON record bytes={}",
185        c.max_model_bytes,
186        c.max_optimal_bytes,
187        resources.observation_limits().max_output_bytes,
188        resources.json_record_bytes()
189    )
190}

192fn completed(
193    sink: &mut impl Write,
194    options: &Options,
195    config: &crate::SolveConfig,
196    report: &Report,
197) -> io::Result<()> {
198    let status = match report.completion {
199        Completion::Exhausted => "exhausted",
200        Completion::RequestedModels => "requested models reached (partial coverage)",
201        Completion::Interrupted => "interrupted (partial coverage)",
202    };
203    writeln!(sink, "  completion: {status}")?;
204    details(sink, options, config, &Details::from(report))
205}

277fn details(
278    sink: &mut impl Write,
279    options: &Options,
280    config: &crate::SolveConfig,
281    report: &Details<'_>,
282) -> io::Result<()> {
283    if let Some(usage) = report.expansion {
284        let limits = crate::admission::expansion_limits(options);
285        writeln!(
286            sink,
287            "  expansion used: term work={} of {}; templates={} of {}; values={} of {}; scalar bytes={} of {}; origins={} of {}",
288            usage.term_work,
289            limits.max_term_work,
290            usage.templates,
291            limits.max_templates,
292            usage.values,
293            limits.max_values,
294            usage.scalar_bytes,
295            limits.max_scalar_bytes,
296            usage.origin_locations,
297            limits.max_origin_locations
298        )?;
299    }
300    if let Some(stats) = report.candidate_statistics {
301        candidates(sink, stats)?;
302    }
303    if let Some(observation) = report.query_execution {
304        query(sink, observation)?;
305    }
306    if let Some(stats) = report.shared_execution {
307        shared(sink, stats)?;
308    }
309    if let Some(stats) = report.lazy_execution {
310        lazy(sink, stats)?;
311    }
312    if let Some(stats) = report.hybrid_execution {
313        hybrid(sink, config, stats)?;
314    }
315    if let Some(stats) = report.terminal_execution {
316        terminal(sink, stats)?;
317    }
318    if let Some(stats) = report.model_construction {
319        writeln!(
320            sink,
321            "  model construction: work={}; prepared bytes={}; peak bytes={}; constructed={}",
322            stats.work, stats.prepared_bytes, stats.peak_bytes, stats.constructed
323        )?;
324    }
325    let examined = if report.shared_execution.is_some() {
326        "closure result/control records examined"
327    } else {
328        "candidates examined"
329    };
330    writeln!(
331        sink,
332        "  results: displayed models={}; {examined}={}",
333        report.models, report.checked
334    )?;
335    if let Some(reason) = report.interruption {
336        writeln!(sink, "  interruption: {reason}")?;
337    }
338    if let Some(stats) = report.countermodel_statistics {
339        formula(sink, config, report)?;
340        countermodel(
341            sink,
342            config,
343            stats,
344            if report.terminal_execution.is_some() {
345                "verified base models"
346            } else if report.hybrid_execution.is_some() {
347                "verified core models"
348            } else {
349                "verified stable models"
350            },
351        )?;
352        writeln!(
353            sink,
354            "  discovered gate tuples: inapplicable (complete semantic candidates)"
355        )?;
356    } else if report.checked > 0 {
357        closure(sink, options, report)?;
358    } else {
359        if report
360            .candidate_statistics
361            .is_some_and(|stats| stats.root_refuted)
362        {
363            // The root's narrowing settled the program: no seed was offered
364            // and no execution route ran; the carrier statistics above say
365            // what refuted it.
366            writeln!(
367                sink,
368                "  effective execution: none needed; the root narrowing refuted every seed"
369            )?;
370        } else {
371            writeln!(
372                sink,
373                "  effective execution: unavailable (stopped before execution counters)"
374            )?;
375        }
376        writeln!(
377            sink,
378            "  oracle work: unavailable; discovered gate tuples: {}",
379            report.discovered_gate_atoms
380        )?;
381    }
382    objective(sink, report)
383}

385fn objective(sink: &mut impl Write, report: &Details<'_>) -> io::Result<()> {
386    if let Some(optimum) = report.optimization {
387        let qualification = if report.optimum_proved {
388            "optimal"
389        } else {
390            "incumbent only"
391        };
392        writeln!(
393            sink,
394            "  objective: {qualification}; costs(priority,value)={:?}; full ties={}; scored models={}; evaluation work={}",
395            optimum.score.costs(),
396            optimum.tied_models,
397            optimum.scored_models,
398            optimum.work
399        )?;
400    } else {
401        writeln!(
402            sink,
403            "  objective: no retained score; evaluation counters=unavailable"
404        )?;
405    }
406    Ok(())
407}

409fn candidates(sink: &mut impl Write, stats: zetesis_cpu::CandidateStatistics) -> io::Result<()> {
410    writeln!(
411        sink,
412        "  candidate restrictions: work={}; conjunctions={}; skipped impossible intervals={}; prepared atom occurrences={}; copied payload bytes={}; peak copied payload bytes={}; allocator/index overhead excluded",
413        stats.restriction_work,
414        stats.restriction_conjunctions,
415        stats.conflicts,
416        stats.restriction_atoms,
417        stats.restriction_bytes,
418        stats.restriction_peak_bytes
419    )?;
420    if stats.root_refuted {
421        return writeln!(
422            sink,
423            "  carrier narrowing: refuted by a definite constraint after {} passes; no seed offered",
424            stats.narrowing_passes
425        );
426    }
427    writeln!(
428        sink,
429        "  carrier narrowing: passes={}; cut gate atoms={}; held gate atoms={}",
430        stats.narrowing_passes, stats.cut_gate_atoms, stats.held_gate_atoms
431    )?;
432    if let Some(stop) = stats.narrowing_stop {
433        writeln!(
434            sink,
435            "  carrier narrowing stop: {stop}; the completed passes' bounds were kept"
436        )?;
437    }
438    if stats.regions > 0 {
439        writeln!(
440            sink,
441            "  carrier regions: visited={}; refuted={}; leaves={}; counted={}; narrowing passes={}",
442            stats.regions,
443            stats.regions_refuted,
444            stats.regions_leaves,
445            stats.regions_counted,
446            stats.region_passes
447        )?;
448    }
449    Ok(())
450}

507fn query(sink: &mut impl Write, observation: &crate::QueryExecutionObservation) -> io::Result<()> {
508    if observation.fault.is_some() && observation.statistics.is_some() {
509        writeln!(
510            sink,
511            "  query statistics: retaining last successful snapshot; current snapshot failed"
512        )?;
513    }
514    if let Some(stats) = observation.statistics {
515        writeln!(
516            sink,
517            "  prepared CPU queries: builds={}; adoptions={}; retained workspaces={}; active ranges={}; reused slots={}; retained bytes={}; reserved envelope bytes={}",
518            stats.preparation_builds,
519            stats.preparation_adoptions,
520            stats.retained_workspaces,
521            stats.active_workspaces,
522            stats.reused_workspaces,
523            stats.retained_bytes,
524            stats.reserved_bytes
525        )?;
526        if let Some(preparation) = stats.preparation {
527            writeln!(
528                sink,
529                "  query preparation: work={}; retained bytes={}; dense predicates={} of {}; separate from candidate work; capacities are not RSS",
530                preparation.work,
531                preparation.retained_bytes,
532                preparation.dense_predicates,
533                preparation.predicates
534            )?;
535        }
536    } else {
537        writeln!(
538            sink,
539            "  prepared CPU queries: no successful ownership snapshot"
540        )?;
541    }
542    if let Some(fault) = &observation.fault {
543        writeln!(sink, "  query observation fault: {fault}")?;
544    }
545    Ok(())
546}

637fn lazy_buffer_usage(sink: &mut impl Write, usage: crate::LazyTransportUsage) -> io::Result<()> {
638    write!(sink, "  lazy buffer requests (allocated/reused):")?;
639    for (name, binding) in [
640        ("uniform", usage.uniform),
641        ("offsets", usage.offsets),
642        ("records", usage.records),
643        ("snapshots", usage.snapshots),
644        ("seeds", usage.seeds),
645        ("output", usage.output),
646        ("readback", usage.readback),
647    ] {
648        write!(sink, " {name}={}/{};", binding.allocations, binding.reuses)?;
649    }
650    writeln!(
651        sink,
652        " slack releases: budget={}; accounting overflow={}",
653        usage.budget_releases, usage.accounting_overflow_releases
654    )
655}

657fn countermodel(
658    sink: &mut impl Write,
659    config: &crate::SolveConfig,
660    stats: &zetesis_sat::Statistics,
661    membership: &str,
662) -> io::Result<()> {
663    if let Some(filter) = stats.region_filter {
664        writeln!(
665            sink,
666            "  candidate region checks: preparations={}; checks={}; refuted={}; failed={}; overflowed={}; source work accounted separately; frozen reduct excluded",
667            filter.preparations, filter.checks, filter.refuted, filter.failed, filter.overflowed
668        )?;
669    }
670    if let Some(support) = stats.support {
671        writeln!(
672            sink,
673            "  necessary disjunctive support: status={:?}; construction work={}; encoding work={} (included in search work)",
674            support.status, support.construction_work, support.encoding_work
675        )?;
676    }
677    if let Some(regions) = stats.regions {
678        let counts = regions.counts;
679        writeln!(
680            sink,
681            "  candidate regions: visited={}; refuted={}; leaves={}; propagations={}; held={}; cut={}; support cut={}; reading work={} (included in search work)",
682            counts.regions,
683            counts.refuted,
684            counts.leaves,
685            counts.propagations,
686            counts.held,
687            counts.cut,
688            if regions.producers {
689                "applied"
690            } else {
691                "not applicable"
692            },
693            counts.work,
694        )?;
695    }
696    writeln!(
697        sink,
698        "  countermodel: search work={}; decisions={}; candidates={}; queries={}; witnesses={}; candidate restrictions={}; classical queries={}; {membership}={}",
699        stats.search.work,
700        stats.search.decisions,
701        stats.candidates,
702        stats.countermodel_queries,
703        stats.countermodels,
704        stats.candidate_restrictions,
705        stats.candidate_queries,
706        stats.stable_models
707    )?;
708    writeln!(
709        sink,
710        "  projection history: entries={}; nodes={}; retained bytes={}; peak bytes={}; work={} (included in search work)",
711        stats.projections.entries,
712        stats.projections.nodes,
713        stats.projections.retained_bytes,
714        stats.projections.peak_bytes,
715        stats.projections.work,
716    )?;
717    if let Some(preparation) = stats.reduct.preparation {
718        writeln!(
719            sink,
720            "  prepared reduct: work={}; variables={}; clauses={}; literals={}; retained bytes={}; construction peak bytes={}; work included in search; capacities exclude allocator overhead",
721            preparation.work,
722            preparation.variables,
723            preparation.clauses,
724            preparation.literals,
725            preparation.retained_bytes,
726            preparation.peak_bytes,
727        )?;
728        writeln!(
729            sink,
730            "  prepared reduct queries: original evaluation work={}; parameter work={} (included in search); peak worker workspace bytes={}",
731            stats.reduct.original_work,
732            stats.reduct.parameter_work,
733            stats.reduct.peak_workspace_bytes,
734        )?;
735    }
736    if stats.reduct.regions.regions > 0 || stats.reduct.regions.work > 0 {
737        let regions = stats.reduct.regions;
738        writeln!(
739            sink,
740            "  reduct query regions: visited={}; refuted={}; leaves={}; propagations={}; reading work={} (included in search work)",
741            regions.regions, regions.refuted, regions.leaves, regions.propagations, regions.work,
742        )?;
743    }
744    if let Some(certified) = stats.certified {
745        certificate(sink, &certified, config.max_completion_scratch_bytes)?;
746    }
747    Ok(())
748}

750fn certificate(
751    sink: &mut impl Write,
752    certified: &zetesis_sat::CertifiedStatistics,
753    max_bytes: u64,
754) -> io::Result<()> {
755    writeln!(
756        sink,
757        "  class certificate: eligible={}; refusal={:?}; storage limit={}; construction work={}; checks={}; stable decisions before commit={}; refuted by support={}; failed={}; checking work={}",
758        certified.plan.is_some(),
759        certified.refusal,
760        max_bytes,
761        certified.construction_work,
762        certified.checks,
763        certified.stable,
764        certified.refuted,
765        certified.failed,
766        certified.checking_work
767    )?;
768    if let Some(plan) = certified.plan {
769        match plan {
770            zetesis_sat::CertificatePlanStatistics::Tight(plan) => writeln!(
771                sink,
772                "  tight certificate storage: construction logical bytes={}; resident logical bytes={}; dependencies={}",
773                plan.construction_bytes, plan.resident_bytes, plan.dependencies
774            )?,
775            zetesis_sat::CertificatePlanStatistics::Positive(plan) => writeln!(
776                sink,
777                "  positive certificate: derived atoms={}; activated formula nodes={}; dependencies={}; propagated dependencies={}; construction peak bytes={}; retained bytes={}",
778                plan.derived_atoms,
779                plan.activated_nodes,
780                plan.dependencies,
781                plan.propagated_dependencies,
782                plan.peak_bytes,
783                plan.retained_bytes
784            )?,
785        }
786    }
787    writeln!(
788        sink,
789        "  class attempts: tight refusal={:?}; positive refusal={:?}; restriction refusal={:?}; restriction work={}; committed restriction clauses={}",
790        certified.tight_refusal,
791        certified.positive_refusal,
792        certified.restriction_refusal,
793        certified.restriction_work,
794        certified.restriction_clauses
795    )?;
796    if let Some(peak) = certified.positive_check_peak_bytes {
797        writeln!(
798            sink,
799            "  positive checking: peak plan and evaluation bytes={peak}"
800        )?;
801    }
802    if !matches!(
803        certified.plan,
804        Some(zetesis_sat::CertificatePlanStatistics::Positive(_))
805    ) && let Some(attempt) = certified.positive_attempt
806    {
807        writeln!(
808            sink,
809            "  positive preparation attempt: work={}; observed peak bytes={}; primitive retained bytes={}; positive plan not selected",
810            attempt.work, attempt.peak_bytes, attempt.retained_bytes
811        )?;
812    }
813    Ok(())
814}

816fn formula(
817    sink: &mut impl Write,
818    config: &crate::SolveConfig,
819    report: &Details<'_>,
820) -> io::Result<()> {
821    let grounder = if let Some(terminal) = report.terminal_execution {
822        match terminal.base {
823            zetesis_themelios::BaseKind::Eager => "eager_base_terminal_definitions",
824            zetesis_themelios::BaseKind::Hybrid => "hybrid_base_terminal_definitions",
825        }
826    } else if report.hybrid_execution.is_some() {
827        "hybrid"
828    } else {
829        "eager"
830    };
831    let oracle = match report
832        .countermodel_statistics
833        .and_then(|s| s.certified)
834        .and_then(|s| s.plan)
835    {
836        Some(zetesis_sat::CertificatePlanStatistics::Tight(_)) => "tight-support",
837        Some(zetesis_sat::CertificatePlanStatistics::Positive(_)) => "positive-consequences",
838        None => "countermodel",
839    };
840    if let Some(execution) = report.formula_execution {
841        let backend = if execution.adapter.is_empty() {
842            "cpu batched exact completion"
843        } else if execution.tight_work_per_candidate.is_some() {
844            "GPU tight support"
845        } else {
846            "hybrid GPU propagation + exact CPU residual search"
847        };
848        writeln!(
849            sink,
850            "  effective execution: backend={backend}; oracle={oracle}; grounder={grounder}; CPU completion requested workers={}; peak preflight workers={}; adapter={}",
851            config.completion_workers, execution.completion.effective_workers, execution.adapter
852        )?;
853        writeln!(
854            sink,
855            "  formula completion: entered={}; residuals entered={}; completed before commit={}; failed={}; peak requested scratch bytes={}; peak shared owner, query and transient/result scratch bytes={}; scratch limit={}; counters overflowed={}",
856            execution.completion.entered,
857            execution.completion.residuals,
858            execution.completion.completed,
859            execution.completion.failed,
860            execution.completion.requested_scratch_bytes,
861            execution.completion.peak_scratch_bytes,
862            config.max_completion_scratch_bytes,
863            execution.completion.overflowed
864        )?;
865        writeln!(
866            sink,
867            "  formula residual completion: completed locally={}; failed={}",
868            execution.completion.residual_completed, execution.completion.residual_failed
869        )?;
870        for (label, measurement) in [
871            ("completion coordinator wall", execution.completion.wall),
872            (
873                "summed worker original validation",
874                execution.completion.worker_original,
875            ),
876            ("summed worker reduct", execution.completion.worker_reduct),
877        ] {
878            if let Some(measurement) = measurement {
879                writeln!(
880                    sink,
881                    "  {label}: calls={}; elapsed ns={}; overflowed={}",
882                    measurement.calls,
883                    measurement.elapsed.as_nanos(),
884                    measurement.overflowed
885                )?;
886            }
887        }
888        writeln!(
889            sink,
890            "  formula accounting: pending candidates={}; queued verified models={}",
891            execution.pending_candidates, execution.queued_models
892        )?;
893        writeln!(
894            sink,
895            "  formula batch limits: candidates={}; pending bytes={}",
896            config.batch_size, config.max_batch_bytes
897        )?;
898        if !execution.adapter.is_empty() {
899            formula_gpu(sink, execution)?;
900        }
901        Ok(())
902    } else {
903        writeln!(
904            sink,
905            "  effective execution: backend=cpu; oracle={oracle}; grounder={grounder}; native CPU search; batch-completion scratch limit=inapplicable"
906        )
907    }
908}

911    sink: &mut impl Write,
912    execution: &crate::FormulaExecutionStatistics,
913) -> io::Result<()> {
914    writeln!(
915        sink,
916        "  formula GPU decoded: batches={}; candidates={}; primitive work={}; completed sweeps={}; GPU-decided committed={}; CPU residuals completed={}",
917        execution.gpu_batches,
918        execution.gpu_candidates,
919        execution.gpu_work,
920        execution.gpu_rounds,
921        execution.gpu_decided,
922        execution.cpu_residuals
923    )?;
924    writeln!(
925        sink,
926        "  formula GPU submitted: batches={}; candidates={}; completed work/sweeps for unreturned results=unavailable",
927        execution.gpu_submitted_batches, execution.gpu_submitted_candidates
928    )?;
929    if let Some(residuals) = execution.gpu_residuals {
930        writeln!(
931            sink,
932            "  formula GPU decoded residuals: fixed point={}; round limit={}; work limit={}; precede CPU completion and commit",
933            residuals.fixed_point, residuals.round_limit, residuals.work_limit,
934        )?;
935    }
936    if let Some(work) = execution.tight_work_per_candidate {
937        writeln!(
938            sink,
939            "  tight GPU limits: complete support scan work/candidate={work}; propagation sweeps=not applicable"
940        )?;
941    }
942    if let Some(work) = execution.gpu_scheduled_work {
943        writeln!(
944            sink,
945            "  tight GPU scheduled full-scan work={work}; includes submitted unreturned batches"
946        )?;
947    }
948    if let Some(limits) = execution.gpu_limits {
949        writeln!(
950            sink,
951            "  formula GPU limits: propagation sweeps/candidate={}; propagation work/candidate={}",
952            limits.rounds_per_candidate, limits.work_per_candidate
953        )?;
954    }
955    writeln!(
956        sink,
957        "  formula GPU resources: peak authored GPU bytes={}; GPU kernel timing=unavailable",
958        execution.peak_accounted_bytes
959    )

962fn closure(sink: &mut impl Write, options: &Options, report: &Details<'_>) -> io::Result<()> {
963    // The statistics say which route ran; without one, the requested policy
964    // says which would have.
965    let grounder = report.closure_execution.map_or(
966        if options.grounder == Grounder::Eager {
967            "eager"
968        } else {
969            "lazy"
970        },
971        |closure| closure.route.label(),
972    );
973    let cpu = report.shared_execution.is_some() || options.backend == Backend::Cpu;
974    if cpu {
975        writeln!(
976            sink,
977            "  effective execution: backend=cpu; oracle=closure; grounder={grounder}; workers={}",
978            options.workers
979        )?;
980    } else {
981        writeln!(
982            sink,
983            "  effective execution: oracle=closure; backend=requested GPU policy; grounder={grounder}; see backend diagnostics for actual adapter"
984        )?;
985    }
986    if report.shared_execution.is_some() {
987        writeln!(
988            sink,
989            "  discovered gate tuples: {}; source and world work reported separately above",
990            report.discovered_gate_atoms
991        )
992    } else if let Some(closure) = report.closure_execution {
993        writeln!(
994            sink,
995            "  discovered gate tuples: {}",
996            report.discovered_gate_atoms
997        )?;
998        independent_closure(sink, closure)
999    } else {
1000        writeln!(
1001            sink,
1002            "  discovered gate tuples: {}; oracle work=unavailable (not accumulated by this driver)",
1003            report.discovered_gate_atoms
1004        )
1005    }
1006}

1008fn independent_closure(
1009    sink: &mut impl Write,
1010    closure: &crate::ClosureExecutionStatistics,
1011) -> io::Result<()> {
1012    let (rounds, units) = match closure.route {
1013        crate::ClosureRoute::Eager => ("rule passes", "eager scan units"),
1014        crate::ClosureRoute::Lazy(_) => ("source rounds", "join/copy units"),
1015    };
1016    writeln!(
1017        sink,
1018        "  independent closure: checks completed={}; stopped={}; {rounds}={}; work={} ({units}); derived atoms={}; stopped checks return no counters",
1019        closure.completed_checks,
1020        closure.stopped_checks,
1021        closure.rounds,
1022        closure.work,
1023        closure.derived_atoms
1024    )?;
1025    if let crate::ClosureRoute::Lazy(joins) = closure.route {
1026        writeln!(
1027            sink,
1028            "  closure joins: catalog work={} (within work); bindings={}; tuple probes={}; dense heads={} (recorded as bits); block steps={} (blocks joined by words); peak named closure bytes={} (admitted or reserved capacity, not RSS)",
1029            joins.catalog_work,
1030            joins.bindings,
1031            joins.tuple_probes,
1032            joins.dense_heads,
1033            joins.block_steps,
1034            joins.peak_closure_bytes
1035        )?;
1036    }
1037    Ok(())
1038}
