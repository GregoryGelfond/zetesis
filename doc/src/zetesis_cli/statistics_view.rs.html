1//! Human statistics views over solver-owned measurements and outcomes.
2//!
3//! Rendering does not start timers, rerun work or derive semantic completion.
4//! Library consumers can inspect the borrowed values or supply their own view.
5
6use crate::{Completion, PhaseTimings, Publication, SemanticOutcome, SolveConfig, SolveStage};
7use std::io::{self, Write};
8use zetesis_presentation::{Alignment, Column, Layout, Row, Table};
9
10mod work;
11
12/// Selection of the human table or retained telemetry record protocol.
13#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
14pub enum StatisticsView {
15    /// Line records consumed by existing measurement tools; no terminal styling.
16    #[default]
17    Records,
18    /// Human tables with explicit units and measurement scope.
19    Human,
20}
21
22/// Borrowed inputs to a statistics view, independent of command-line parsing.
23pub struct Statistics<'a> {
24    /// Requested settings; these do not establish the route actually executed.
25    pub requested: &'a SolveConfig,
26    /// Attempted host intervals, including failed work.
27    pub timings: &'a PhaseTimings,
28    /// Established semantic evidence, if execution reached a session.
29    pub semantic: Option<&'a SemanticOutcome>,
30    /// Publication acknowledgements preceding this statistics snapshot.
31    pub publication: Option<Publication>,
32    /// Whether the controller had encountered a failure before this snapshot.
33    pub failed: bool,
34}
35
36impl Statistics<'_> {
37    /// Write bounded-size human tables. Lazy grounding is reported as interleaved,
38    /// never as a zero-duration materialization. Requested policy is named as
39    /// such. A compact work view retains each operation's units and scope;
40    /// absent receipts are not inferred from requested settings. The complete
41    /// counters remain available on `semantic` and `timings`.
42    ///
43    /// # Errors
44    /// Propagates the first writer failure without retrying or flushing.
45    pub fn write_human(&self, output: &mut impl Write, layout: Layout) -> io::Result<()> {
46        self.timing_table()?.write(output, layout)?;
47        writeln!(output)?;
48        self.execution_table()?.write(output, layout)?;
49        writeln!(output)?;
50        work::table(self.semantic, &self.timings.grounding)?.write(output, layout)?;
51        writeln!(
52            output,
53            "Time includes attempted work and output; excludes source loading and this statistics view."
54        )?;
55        writeln!(
56            output,
57            "These are host intervals, not GPU kernel times or memory measurements."
58        )?;
59        writeln!(
60            output,
61            "Work units differ by operation; subtotals must not be added. Unavailable is not zero."
62        )
63    }
64
65    fn timing_table(&self) -> io::Result<Table> {
66        let mut rows = Vec::new();
67        for stage in SolveStage::ALL {
68            let value = self.timings.stages.get(stage);
69            let note = stage_scope(stage, self.timings.stages.grounding_mode, value);
70            rows.push(Row::new([
71                stage_name(stage).to_owned(),
72                duration(value.map(|value| value.elapsed)),
73                note,
74            ]));
75        }
76        rows.push(Row::new([
77            "Unattributed".to_owned(),
78            duration(self.timings.stages.unattributed),
79            "host elapsed".to_owned(),
80        ]));
81        rows.push(
82            Row::new([
83                "Total".to_owned(),
84                duration(Some(self.timings.driver_elapsed)),
85                "driver interval".to_owned(),
86            ])
87            .conclusion(),
88        );
89        Table::new(
90            "Statistics",
91            vec![
92                Column::new("Stage", Alignment::Left),
93                Column::new("Time (ms)", Alignment::Right),
94                Column::new("Scope", Alignment::Left),
95            ],
96            rows,
97        )
98        .map_err(io::Error::other)
99    }
100
101    fn execution_table(&self) -> io::Result<Table> {
102        let completion = self.semantic.and_then(SemanticOutcome::completion).map_or(
103            "unavailable",
104            |completion| match completion {
105                Completion::Exhausted => "exhausted",
106                Completion::RequestedModels => "requested answers reached; partial coverage",
107                Completion::Interrupted => "interrupted; partial coverage",
108            },
109        );
110        let rows = vec![
111            Row::new(["Requested backend", self.requested.backend.label()]),
112            Row::new(["Requested grounder", self.requested.grounder.label()]),
113            Row::new([
114                "Requested threads".to_owned(),
115                self.requested.workers.to_string(),
116            ]),
117            Row::new([
118                "Requested detailed measurements",
119                if self.requested.stats {
120                    "enabled"
121                } else {
122                    "disabled"
123                },
124            ]),
125            Row::new([
126                "Recorded execution".to_owned(),
127                work::execution(self.semantic),
128            ]),
129            Row::new([
130                "Published answers".to_owned(),
131                self.publication.map_or_else(
132                    || "unavailable".to_owned(),
133                    |value| value.models.to_string(),
134                ),
135            ]),
136            Row::new([
137                "Candidate progress".to_owned(),
138                self.semantic.map_or_else(
139                    || "unavailable".to_owned(),
140                    |value| value.candidate_progress().to_string(),
141                ),
142            ]),
143            Row::new(["Search coverage", completion]),
144            Row::new([
145                "Failure before snapshot",
146                if self.failed { "yes" } else { "no" },
147            ]),
148        ];
149        Table::new(
150            "Execution",
151            vec![
152                Column::new("Measure", Alignment::Left),
153                Column::new("Value", Alignment::Left),
154            ],
155            rows,
156        )
157        .map_err(io::Error::other)
158    }
159}

161fn stage_name(stage: SolveStage) -> &'static str {
162    match stage {
163        SolveStage::SourcePreparation => "Source preparation",
164        SolveStage::Grounding => "Grounding",
165        SolveStage::Solving => "Solving",
166        SolveStage::ObservationOutput => "Observation and output",
167    }
168}
169
170fn duration(value: Option<std::time::Duration>) -> String {
171    value.map_or_else(
172        || "—".to_owned(),
173        |value| format!("{:.3}", value.as_secs_f64() * 1000.0),
174    )
175}

177fn stage_scope(
178    stage: SolveStage,
179    mode: crate::GroundingMode,
180    value: Option<crate::StageMeasurement>,
181) -> String {
182    let scope = match (stage, mode, value) {
183        (SolveStage::Grounding, crate::GroundingMode::LazyInterleaved, _) => {
184            "interleaved with solving"
185        }
186        (SolveStage::Grounding, crate::GroundingMode::Mixed, _) => {
187            "eager attempts; lazy work interleaved"
188        }
189        (SolveStage::Grounding, crate::GroundingMode::EagerBaseTerminalDefinitions, _) => {
190            "eager base; terminal definitions reconstructed during solving"
191        }
192        (SolveStage::Solving, crate::GroundingMode::EagerBaseTerminalDefinitions, _) => {
193            "base search and full-answer reconstruction"
194        }
195        (_, _, Some(_)) => "host elapsed",
196        (_, _, None) => "not measured",
197    };
198    if value.is_some_and(|value| value.overflowed) {
199        format!("{scope}; incomplete measurement")
200    } else {
201        scope.to_owned()
202    }
203}

205#[cfg(test)]
206mod tests {
207    use super::stage_scope;
208    use crate::{GroundingMode, SolveStage, StageMeasurement};
209
210    #[test]
211    fn interleaving_does_not_hide_measurement_overflow() {
212        for mode in [GroundingMode::Mixed, GroundingMode::LazyInterleaved] {
213            let note = stage_scope(
214                SolveStage::Grounding,
215                mode,
216                Some(StageMeasurement {
217                    overflowed: true,
218                    ..StageMeasurement::default()
219                }),
220            );
221            assert!(note.contains("incomplete measurement"), "{note}");
222            assert!(note.contains("interleaved"), "{note}");
223        }
224    }
