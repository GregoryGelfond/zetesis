12/// Conformance tasks, separate from benchmarking and developer qualification.
13#[derive(Debug, Subcommand)]
14pub enum TestCommand {
15    /// Compare all 94 local corpus cases with external clingo.
16    Corpus(CorpusOptions),
17    /// Check three fixed full answer families and actual CPU or Metal execution.
18    Backend(BackendOptions),
19    /// Qualify maintained workloads across CPU thread counts; requires --report.
20    Scalability(ScalabilityOptions),

94/// Completed test disposition; an explicit nonpass is still a published report.
95#[derive(Clone, Copy, Debug, PartialEq, Eq)]
96pub enum Completion {
97    /// Every requested check passed.
98    Passed,
99    /// At least one check failed or was not completed.
100    NonPass,
101}
102
103/// Preparation or presentation failure, separate from reported check nonpasses.
104#[derive(Debug)]
105pub enum Error {
106    /// Source/executable resolution or a supplied writer failed.
107    Io(io::Error),
108    /// Corpus preparation failed before per-case results existed.
109    Corpus(corpus_comparison::Error),
110    /// Backend preparation failed before per-case results existed.
111    Backend(backend_check::Error),
112    /// Scalability workload admission or qualification preparation failed.
113    Scalability(zetesis_validation::performance::Error),
114    /// Complete scalability evidence could not be published after the campaign.
115    ScalabilityPublication(zetesis_validation::performance::Error),
116    /// Structured publication failed.
117    Json(serde_json::Error),
118    /// Human table shape was invalid.
119    Table(zetesis_presentation::TableError),
120    /// The requested seconds cannot be represented as corpus milliseconds.
121    DurationOverflow,
122    /// The primary command failure and a later failure-document write error.
123    FailurePublication {
124        /// Original command error.
125        primary: Box<Error>,
126        /// Failure while publishing its structured diagnostic.
127        publication: Box<Error>,
128    },
129}
130impl fmt::Display for Error {
131    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
132        match self {
133            Self::Io(error) => error.fmt(formatter),
134            Self::Corpus(error) => error.fmt(formatter),
135            Self::Backend(error) => error.fmt(formatter),
136            Self::Scalability(error) | Self::ScalabilityPublication(error) => error.fmt(formatter),
137            Self::Json(error) => error.fmt(formatter),
138            Self::Table(error) => error.fmt(formatter),
139            Self::DurationOverflow => {
140                formatter.write_str("test timeout exceeds the supported millisecond range")
141            }
142            Self::FailurePublication {
143                primary,
144                publication,
145            } => write!(
146                formatter,
147                "{primary}; failure report publication also failed: {publication}"
148            ),
149        }
150    }
151}
152impl std::error::Error for Error {
153    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
154        match self {
155            Self::Io(error) => Some(error),
156            Self::Corpus(error) => Some(error),
157            Self::Backend(error) => Some(error),
158            Self::Scalability(error) | Self::ScalabilityPublication(error) => Some(error),
159            Self::Json(error) => Some(error),
160            Self::Table(error) => Some(error),
161            Self::DurationOverflow => None,
162            Self::FailurePublication { primary, .. } => Some(primary.as_ref()),
163        }
164    }
165}

187/// Run the requested typed workflow and publish through injected writers.
188///
189/// Failures before any stdout write attempt receive a complete JSON failure
190/// document when requested. A publication failure can leave a partial document;
191/// no second document is appended. Child nonpasses return `NonPass` with a report.
192/// No global stdout/stderr or compatibility executable forwarding is used.
193///
194/// # Errors
195/// Returns input/setup or publication failure; completed failed checks are values.
196pub fn execute(
197    command: &TestCommand,
198    layout: Layout,
199    output: &mut impl io::Write,
200    diagnostics: &mut impl io::Write,
201) -> Result<Completion, Error> {
202    execute_with_cancellation(
203        command,
204        layout,
205        output,
206        diagnostics,
207        &AtomicBool::new(false),
208    )
209}

211/// Execute tests with an explicit caller-owned cancellation flag.
212///
213/// Active process capture performs bounded cleanup before returning. Later
214/// children are not launched once cancellation is observed. Partial checks are
215/// published as nonpassing evidence; this adapter installs no signal handlers.
216///
217/// # Errors
218/// Returns the setup/publication errors of [`execute`], preserving a primary
219/// failure if publishing its structured failure document also fails.
220pub fn execute_with_cancellation(
221    command: &TestCommand,
222    layout: Layout,
223    output: &mut impl io::Write,
224    diagnostics: &mut impl io::Write,
225    cancelled: &AtomicBool,
226) -> Result<Completion, Error> {
227    let mut tracked = zetesis_presentation::TrackedWriter::new(output);
228    let result = execute_inner(command, layout, &mut tracked, diagnostics, cancelled);
229    let error = match result {
230        Ok(completion) => return Ok(completion),
231        Err(error) => error,
232    };
233    if command.json() && !tracked.write_attempted() {
234        let task = match command {
235            TestCommand::Corpus(_) => "corpus",
236            TestCommand::Backend(_) => "backend",
237            TestCommand::Scalability(_) => "scalability",
238        };
239        let status = match error {
240            Error::Corpus(_)
241            | Error::Backend(_)
242            | Error::Scalability(_)
243            | Error::DurationOverflow => "preparation_failed",
244            _ => "failed",
245        };
246        let failure = serde_json::json!({"schema":1,"format":"zetesis-test-failure","task":task,"status":status,"kind":error.code(),"detail":error.to_string()});
247        if let Err(publication) = view::json(&failure, &mut tracked) {
248            return Err(Error::FailurePublication {
249                primary: Box::new(error),
250                publication: Box::new(publication),
251            });
252        }
253    }
254    Err(error)
255}

257fn execute_inner(
258    command: &TestCommand,
259    layout: Layout,
260    output: &mut impl io::Write,
261    diagnostics: &mut impl io::Write,
262    cancelled: &AtomicBool,
263) -> Result<Completion, Error> {
264    match command {
265        TestCommand::Scalability(options) => {
266            scalability::execute(options, layout, output, diagnostics, cancelled)
267        }
268        TestCommand::Corpus(options) => {
269            let report = corpus(options, cancelled)?;
270            retain(&options.view, |file| {
271                view::json(&report.to_json().map_err(Error::Json)?, file)
272            })?;
273            view::corpus(&report, &options.view, layout, output)?;
274            if !report.passed() {
275                writeln!(
276                    diagnostics,
277                    "Corpus comparison did not pass; inspect the reported case outcomes."
278                )
279                .map_err(Error::Io)?;
280            }
281            Ok(completion(report.passed()))
282        }
283        TestCommand::Backend(options) => {
284            let report = backend(options, cancelled)?;
285            retain(&options.view, |file| view::backend_json(&report, file))?;
286            view::backend(&report, &options.view, layout, output)?;
287            if !report.passed() {
288                writeln!(
289                    diagnostics,
290                    "Backend checks did not pass; this does not establish backend qualification."
291                )
292                .map_err(Error::Io)?;
293            }
294            Ok(completion(report.passed()))
295        }
296    }
297}

307fn executable(path: Option<&PathBuf>) -> Result<PathBuf, Error> {
308    path.map_or_else(std::env::current_exe, std::path::absolute)
309        .map_err(Error::Io)

312fn corpus(
313    options: &CorpusOptions,
314    cancelled: &AtomicBool,
315) -> Result<corpus_comparison::Report, Error> {
316    let request = corpus_comparison::Request {
317        repo: options.repo.clone(),
318        clingo: options.clingo.clone(),
319        zetesis: executable(options.zetesis.as_ref())?,
320        native_backend: options.backend,
321        native_oracle: if options.backend.is_gpu() {
322            corpus_comparison::NativeOracle::Countermodel
323        } else {
324            corpus_comparison::NativeOracle::Auto
325        },
326        native_stats: options.view.stats,
327        timeout_ms: options
328            .process
329            .timeout_seconds
330            .checked_mul(1000)
331            .ok_or(Error::DurationOverflow)?,
332        max_output_bytes: options.process.capture_bytes,
333        ..corpus_comparison::Request::default()
334    };
335    corpus_comparison::run_with_cancellation(
336        &request,
337        corpus_comparison::NativeInvocation::Solve,
338        cancelled,
339        |_| {},
340    )
341    .map_err(Error::Corpus)
342}

344fn backend(
345    options: &BackendOptions,
346    cancelled: &AtomicBool,
347) -> Result<backend_check::Report, Error> {
348    backend_check::run_with_cancellation(
349        &backend_check::Request {
350            executable: executable(options.zetesis.as_ref())?,
351            backend: options.backend,
352            limits: zetesis_validation::process::Limits {
353                timeout: Duration::from_secs(options.process.timeout_seconds),
354                max_output_bytes: options.process.capture_bytes,
355                ..zetesis_validation::process::Limits::default()
356            },
357        },
358        cancelled,
359    )
360    .map_err(Error::Backend)
361}

363fn retain(
364    view: &ViewOptions,
365    write: impl FnOnce(&mut std::fs::File) -> Result<(), Error>,
366) -> Result<(), Error> {
367    use io::Write as _;
368
369    if let Some(path) = &view.report {
370        let mut file = std::fs::OpenOptions::new()
371            .write(true)
372            .create_new(true)
373            .open(path)
374            .map_err(Error::Io)?;
375        write(&mut file)?;
376        file.flush().map_err(Error::Io)?;
377    }
378    Ok(())
379}
