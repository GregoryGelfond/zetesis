1//! One bounded JSON record at a time, with a bounded document atom table.
2
3use std::io::Write;
4
5use zetesis_cpu::Cancellation;
6use zetesis_themelios::observation::json::AtomTable;
7
8use crate::{AnswerRenderer, AnswerView, PublicationView, RunError, SummaryDelivery};
9
10/// Streaming JSON renderer over the same typed views as the human renderer.
11/// Retains one encoded record and a bounded table of previously spelled atoms,
12/// not a complete `WorldView`. It does not poll control while emitting the final
13/// failure/stop receipt, so cancellation cannot suppress that retained evidence.
14pub struct JsonRenderer<W> {
15    output: crate::output::Document<W>,
16    atoms: AtomTable,
17    max_record_bytes: usize,
18    max_atoms: usize,
19}
20
21impl<W: Write> JsonRenderer<W> {
22    /// Configure inclusive record and distinct document-atom ceilings.
23    /// Construction performs no I/O; the controller starts the document.
24    #[must_use]
25    pub fn new(output: W, max_record_bytes: usize, max_atoms: usize) -> Self {
26        Self::from_document(
27            crate::output::Document::new(output),
28            max_record_bytes,
29            max_atoms,
30        )
31    }
32
33    pub(crate) fn from_document(
34        output: crate::output::Document<W>,
35        max_record_bytes: usize,
36        max_atoms: usize,
37    ) -> Self {
38        Self {
39            output,
40            atoms: AtomTable::new(max_atoms),
41            max_record_bytes,
42            max_atoms,
43        }
44    }
45
46    /// Recover the sink after publication, including any accepted prefix.
47    #[must_use]
48    pub fn into_inner(self) -> W {
49        self.output.into_inner()
50    }
51}
52
53impl<W: Write> AnswerRenderer for JsonRenderer<W> {
54    fn begin(&mut self) -> Result<(), RunError> {
55        // A reused renderer begins a new document, not a continuation whose
56        // atom indices could refer to a previous invocation's table.
57        self.atoms = AtomTable::new(self.max_atoms);
58        self.output.start();
59        write!(
60            self.output,
61            "{{\"schema\":{},\"format\":\"zetesis\",\"models\":[",
62            zetesis_themelios::observation::json::RECORD_SCHEMA_VERSION
63        )?;
64        Ok(())
65    }
66
67    fn answer(
68        &mut self,
69        view: AnswerView<'_>,
70        cancellation: &Cancellation,
71    ) -> Result<(), RunError> {
72        crate::output::write_model_record(
73            &mut self.output,
74            view.number,
75            view.model,
76            &mut self.atoms,
77            self.max_record_bytes,
78            cancellation,
79        )
80    }
81
82    fn finish(&mut self, view: PublicationView<'_>) -> Result<SummaryDelivery, RunError> {
83        if self.output.failed() {
84            return Ok(SummaryDelivery::Omitted);
85        }
86        let record = crate::output::summary(view, self.max_record_bytes)?;
87        self.output.write_all(&record)?;
88        Ok(SummaryDelivery::Accepted)
89    }
90}
