1//! Clingo, the external comparison oracle, for the zetesis workspace's tests.
2//!
3//! A test that compares zetesis with clingo finds clingo ([`executable`]),
4//! runs it within bounds on a source ([`run`], [`run_accepting`]) or on
5//! program files ([`run_in`]), and decodes its JSON report ([`json`]): as
6//! complete records of atoms and costs ([`model_records`]), as full models
7//! ([`full_models`], [`objective_free_models`]), from a witness's parts
8//! ([`atoms`], [`costs`]), or as `zetesis-validation` reads it ([`answers()`]).
9//! [`records`] runs and decodes a complete enumeration in one call. Each test
10//! keeps the clingo arguments its comparison needs.
11//!
12//! The crate depends on `zetesis-validation`, whose bounded process capture
13//! runs clingo, and on nothing that solves, so a test reaches clingo without
14//! compiling a solver. It is not published or installed.
15

25/// The clingo release the comparisons are recorded against.
26pub const VERSION: &str = "clingo version 5.8.2";

28/// The arguments of a complete enumeration: every answer set, optimal or
29/// not, as JSON, without warnings.
30pub const ENUMERATION: [&str; 4] = ["0", "--outf=2", "--opt-mode=enum", "--warn=none"];

32/// The clingo the comparisons run: `CLINGO` when it is set, which must name
33/// an absolute path, as the oracle gate requires; otherwise the first eligible
34/// `clingo` on `PATH`.
35///
36/// # Panics
37/// Panics if `CLINGO` names a relative path, or, naming both sources, if
38/// neither yields clingo.
39#[must_use]
40pub fn executable() -> PathBuf {
41    discover(
42        std::env::var_os("CLINGO"),
43        std::env::var_os("PATH").as_deref(),
44    )
45}
46
47/// [`executable`]'s rule, over the values of `CLINGO` and `PATH`.
48fn discover(clingo: Option<OsString>, search: Option<&OsStr>) -> PathBuf {
49    if let Some(path) = clingo {
50        let path = PathBuf::from(path);
51        assert!(
52            path.is_absolute(),
53            "CLINGO must name an absolute path: {}",
54            path.display()
55        );
56        return path;
57    }
58    process::resolve_executable("clingo".as_ref(), search)
59        .unwrap_or_else(|error| panic!("CLINGO is unset and PATH holds no clingo: {error}"))
60}

62/// The byte ceiling of a clingo report the comparisons read: 64 KiB.
63pub const REPORT_BYTES: usize = 64 * 1024;

65/// The bounds of one clingo run.
66#[derive(Clone, Copy, Debug)]
67pub struct Limits {
68    /// The run's deadline.
69    pub timeout: Duration,
70    /// An inclusive ceiling on the bytes of standard output and error together.
71    pub max_output_bytes: usize,
72}
73
74impl Default for Limits {
75    /// Five seconds and [`REPORT_BYTES`].
76    fn default() -> Self {
77        Self {
78            timeout: Duration::from_secs(5),
79            max_output_bytes: REPORT_BYTES,
80        }
81    }
82}

112/// The exit codes of a run that decided its program: satisfiable (10),
113/// unsatisfiable (20), or satisfiable with its search exhausted (30).
114pub const DECIDED: [i32; 3] = zetesis_validation::answers::CLINGO_DECIDED_EXITS;

116/// Run clingo on `source` with `arguments`, within `limits`, expecting it to
117/// decide the program ([`DECIDED`]).
118///
119/// # Panics
120/// As [`run_accepting`].
121#[must_use]
122pub fn run(source: &str, arguments: &[&str], limits: Limits) -> Run {
123    run_accepting(source, arguments, &DECIDED, limits)
124}

126/// Run clingo on `source` with `arguments`, within `limits`, accepting the
127/// exit codes in `exits`: a comparison that expects clingo to refuse a
128/// program accepts its error code, 65.
129///
130/// The source is written to a file in a fresh temporary directory, where
131/// clingo runs; the file is its last argument.
132///
133/// # Panics
134/// As [`run_in`]; the message quotes the source.
135#[must_use]
136pub fn run_accepting(source: &str, arguments: &[&str], exits: &[i32], limits: Limits) -> Run {
137    run_source(&executable(), source, arguments, exits, limits)
138}
139
140/// Run clingo in `directory` with `arguments`, which name the program files
141/// it reads, within `limits`, accepting the exit codes in `exits`. A relative
142/// file name resolves from `directory`.
143///
144/// # Panics
145/// Panics if clingo cannot start, does not complete within the limits, is
146/// signalled, exits with a code `exits` does not hold, or leaves a child that
147/// cannot be reaped cleanly. The message names the arguments and carries
148/// clingo's standard error.
149#[must_use]
150pub fn run_in<I>(directory: &Path, arguments: I, exits: &[i32], limits: Limits) -> Run
151where
152    I: IntoIterator,
153    I::Item: AsRef<OsStr>,
154{
155    let arguments: Vec<OsString> = arguments
156        .into_iter()
157        .map(|argument| argument.as_ref().to_owned())
158        .collect();
159    let subject = format!("{arguments:?}");
160    invoke(
161        &executable(),
162        directory,
163        &arguments,
164        exits,
165        limits,
166        &subject,
167    )
168}

170/// [`run_accepting`], with `executable` in clingo's place.
171fn run_source(
172    executable: &Path,
173    source: &str,
174    arguments: &[&str],
175    exits: &[i32],
176    limits: Limits,
177) -> Run {
178    let directory = tempfile::tempdir().expect("temporary directory for clingo");
179    let input = directory.path().join("case.lp");
180    std::fs::write(&input, source).expect("clingo input");
181    let mut arguments: Vec<OsString> = arguments.iter().map(OsString::from).collect();
182    arguments.push(input.into_os_string());
183    invoke(
184        executable,
185        directory.path(),
186        &arguments,
187        exits,
188        limits,
189        source,
190    )
191}

193/// One bounded run of `executable` in `directory`; `subject` names its input
194/// in a failure's message. The executable is clingo, except in this crate's
195/// tests of a run's contract.
196fn invoke(
197    executable: &Path,
198    directory: &Path,
199    arguments: &[OsString],
200    exits: &[i32],
201    limits: Limits,
202    subject: &str,
203) -> Run {
204    let (capture, pending) = process::invoke(
205        process::Invocation {
206            executable,
207            arguments,
208            directory,
209        },
210        process::Limits {
211            timeout: limits.timeout,
212            max_output_bytes: limits.max_output_bytes,
213            cleanup_timeout: Duration::from_secs(1),
214        },
215    )
216    .expect("clingo starts")
217    .into_parts();
218    if let Some(pending) = pending {
219        let cleanup = pending.retry(Duration::from_secs(1));
220        if let Some(pending) = cleanup.pending {
221            panic!(
222                "clingo child {} could not be reaped: {:?}",
223                pending.abandon(),
224                cleanup.failure
225            );
226        }
227        assert!(
228            cleanup.failure.is_none(),
229            "clingo's cleanup failed: {:?}",
230            cleanup.failure
231        );
232    }
233    let context = format!(
234        "clingo on {subject}\nstandard error:\n{}",
235        String::from_utf8_lossy(capture.stderr())
236    );
237    assert_eq!(capture.stop(), process::Stop::Completed, "{context}");
238    assert!(
239        capture.failure().is_none(),
240        "{:?}: {context}",
241        capture.failure()
242    );
243    assert!(
244        capture.cleanup_failure().is_none(),
245        "{:?}: {context}",
246        capture.cleanup_failure()
247    );
248    let exit = capture.exit().expect("a completed run has an exit");
249    assert!(exit.signal.is_none(), "signalled: {context}");
250    let code = exit.code.expect("an unsignalled exit has a code");
251    assert!(exits.contains(&code), "exited {code}: {context}");
252    Run {
253        code,
254        stdout: capture.stdout().to_vec(),
255        stderr: capture.stderr().to_vec(),
256    }
257}

268/// Every answer set of `source` with its costs, as clingo enumerates them
269/// ([`ENUMERATION`]).
270///
271/// # Panics
272/// As [`run`] and [`model_records`], and if the report is not [`VERSION`]'s.
273#[must_use]
274pub fn records(source: &str) -> Records {
275    let report = json(&run(source, &ENUMERATION, Limits::default()));
276    assert_eq!(report["Solver"].as_str(), Some(VERSION));
277    model_records(&report)
278}

280/// Every witness of a complete JSON report, with its costs.
281///
282/// # Panics
283/// Panics unless the enumeration is complete and satisfiable, unsatisfiable
284/// or optimal, if a witness's atoms are not an array of distinct strings or
285/// its present costs are not an array of integers, if a witness repeats, or if
286/// the report's model count differs from its witnesses.
287#[must_use]
288pub fn model_records(report: &Value) -> Records {
289    assert_eq!(
290        report["Models"]["More"].as_str(),
291        Some("no"),
292        "incomplete enumeration"
293    );
294    assert!(
295        matches!(
296            report["Result"].as_str(),
297            Some("SATISFIABLE" | "UNSATISFIABLE" | "OPTIMUM FOUND")
298        ),
299        "undecided result: {}",
300        report["Result"]
301    );
302    let mut records = Records::new();
303    let mut count = 0;
304    for call in report["Call"].as_array().expect("clingo calls") {
305        if let Some(witnesses) = call["Witnesses"].as_array() {
306            for witness in witnesses {
307                count += 1;
308                assert!(
309                    records.insert((atoms(&witness["Value"]), costs(&witness["Costs"]))),
310                    "duplicate full model"
311                );
312            }
313        }
314    }
315    assert_eq!(
316        report["Models"]["Number"].as_u64(),
317        Some(count),
318        "the model count contradicts the witnesses"
319    );
320    records
321}

323/// The full models of a complete JSON report: its witnesses' atoms.
324///
325/// # Panics
326/// As [`model_records`].
327#[must_use]
328pub fn full_models(report: &Value) -> Models {
329    model_records(report)
330        .into_iter()
331        .map(|(atoms, _)| atoms)
332        .collect()
333}

335/// The full models of a complete JSON report of a program without an
336/// objective.
337///
338/// # Panics
339/// As [`model_records`], and if a witness has costs.
340#[must_use]
341pub fn objective_free_models(report: &Value) -> Models {
342    model_records(report)
343        .into_iter()
344        .map(|(atoms, costs)| {
345            assert_eq!(
346                costs, None,
347                "a witness of a program without an objective has costs"
348            );
349            atoms
350        })
351        .collect()
352}

354/// A witness's atoms, as their spellings.
355///
356/// # Panics
357/// Panics if `values` is not an array of strings, or names an atom twice.
358#[must_use]
359pub fn atoms(values: &Value) -> BTreeSet<String> {
360    let values = values
361        .as_array()
362        .expect("a witness's atoms are an array of strings");
363    let atoms: BTreeSet<String> = values
364        .iter()
365        .map(|atom| {
366            atom.as_str()
367                .expect("a witness's atoms are an array of strings")
368                .into()
369        })
370        .collect();
371    assert_eq!(atoms.len(), values.len(), "full atom identities are unique");
372    atoms
373}

375/// A witness's costs: absent when the report omits them or gives `null`, as
376/// for a program without an objective.
377///
378/// # Panics
379/// Panics if present costs are not an array of integers.
380#[must_use]
381pub fn costs(values: &Value) -> Option<Vec<i64>> {
382    if values.is_null() {
383        return None;
384    }
385    let values = values.as_array().expect("costs are an array of integers");
386    Some(
387        values
388            .iter()
389            .map(|value| value.as_i64().expect("costs are an array of integers"))
390            .collect(),
391    )
392}

394/// The answers a run reports, as `zetesis-validation` reads a clingo JSON
395/// report.
396///
397/// # Panics
398/// Panics if the report does not decode, or is not [`VERSION`]'s.
399#[must_use]
400pub fn answers(run: &Run) -> answers::ReportedAnswers {
401    let report = answers::clingo_json(run.stdout(), answers::Limits::default()).unwrap();
402    assert_eq!(report.solver(), VERSION);
403    report
404}
