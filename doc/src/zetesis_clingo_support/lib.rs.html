1//! Clingo, the external comparison oracle, for the zetesis workspace's tests.
2//!
3//! A test that compares zetesis with clingo finds clingo ([`executable`]),
4//! runs it within bounds on a source ([`run`], [`run_accepting`]) or on
5//! program files ([`run_in`]), and decodes its JSON report ([`json`],
6//! [`model_records`], [`answers()`]); [`records`] does all three for a complete
7//! enumeration. Each test keeps the clingo arguments its comparison needs.
8//!
9//! The crate depends on `zetesis-validation`, whose bounded process capture
10//! runs clingo, and on nothing that solves, so a test reaches clingo without
11//! compiling a solver. It is not published or installed.
12

22/// The clingo release the comparisons are recorded against.
23pub const VERSION: &str = "clingo version 5.8.2";

25/// The arguments of a complete enumeration: every answer set, optimal or
26/// not, as JSON, without warnings.
27pub const ENUMERATION: [&str; 4] = ["0", "--outf=2", "--opt-mode=enum", "--warn=none"];

29/// The clingo the comparisons run: `CLINGO` when it is set, which must name
30/// an absolute path, as the oracle gate requires; otherwise the first eligible
31/// `clingo` on `PATH`.
32///
33/// # Panics
34/// Panics if `CLINGO` names a relative path, or, naming both sources, if
35/// neither yields clingo.
36#[must_use]
37pub fn executable() -> PathBuf {
38    discover(
39        std::env::var_os("CLINGO"),
40        std::env::var_os("PATH").as_deref(),
41    )
42}
43
44/// [`executable`]'s rule, over the values of `CLINGO` and `PATH`.
45fn discover(clingo: Option<OsString>, search: Option<&OsStr>) -> PathBuf {
46    if let Some(path) = clingo {
47        let path = PathBuf::from(path);
48        assert!(
49            path.is_absolute(),
50            "CLINGO must name an absolute path: {}",
51            path.display()
52        );
53        return path;
54    }
55    process::resolve_executable("clingo".as_ref(), search)
56        .unwrap_or_else(|error| panic!("CLINGO is unset and PATH holds no clingo: {error}"))
57}

59/// The byte ceiling of a clingo report the comparisons read: 64 KiB.
60pub const REPORT_BYTES: usize = 64 * 1024;
61
62/// The bounds of one clingo run.
63#[derive(Clone, Copy, Debug)]
64pub struct Limits {
65    /// The run's deadline.
66    pub timeout: Duration,
67    /// An inclusive ceiling on the bytes of standard output and error together.
68    pub max_output_bytes: usize,
69}
70
71impl Default for Limits {
72    /// Five seconds and [`REPORT_BYTES`].
73    fn default() -> Self {
74        Self {
75            timeout: Duration::from_secs(5),
76            max_output_bytes: REPORT_BYTES,
77        }
78    }
79}

81/// A completed clingo run: its exit code and both output streams.
82#[derive(Debug)]
83pub struct Run {
84    code: i32,
85    stdout: Vec<u8>,
86    stderr: Vec<u8>,
87}

89impl Run {
90    /// The exit code, one of those the run accepted.
91    #[must_use]
92    pub fn code(&self) -> i32 {
93        self.code
94    }
95
96    /// The complete standard output.
97    #[must_use]
98    pub fn stdout(&self) -> &[u8] {
99        &self.stdout
100    }
101
102    /// The complete standard error.
103    #[must_use]
104    pub fn stderr(&self) -> &[u8] {
105        &self.stderr
106    }
107}

109/// The exit codes of a run that decided its program: satisfiable (10),
110/// unsatisfiable (20), or satisfiable with its search exhausted (30).
111pub const DECIDED: [i32; 3] = [10, 20, 30];
112
113/// Run clingo on `source` with `arguments`, within `limits`, expecting it to
114/// decide the program ([`DECIDED`]).
115///
116/// # Panics
117/// As [`run_accepting`].
118#[must_use]
119pub fn run(source: &str, arguments: &[&str], limits: Limits) -> Run {
120    run_accepting(source, arguments, &DECIDED, limits)
121}
122
123/// Run clingo on `source` with `arguments`, within `limits`, accepting the
124/// exit codes in `exits`: a comparison that expects clingo to refuse a
125/// program accepts its error code, 65.
126///
127/// The source is written to a file in a fresh temporary directory, where
128/// clingo runs; the file is its last argument.
129///
130/// # Panics
131/// As [`run_in`]; the message quotes the source.
132#[must_use]
133pub fn run_accepting(source: &str, arguments: &[&str], exits: &[i32], limits: Limits) -> Run {
134    run_source(&executable(), source, arguments, exits, limits)
135}
136
137/// Run clingo in `directory` with `arguments`, which name the program files
138/// it reads, within `limits`, accepting the exit codes in `exits`. A relative
139/// file name resolves from `directory`.
140///
141/// # Panics
142/// Panics if clingo cannot start, does not complete within the limits, is
143/// signalled, exits with a code `exits` does not hold, or leaves a child that
144/// cannot be reaped cleanly. The message names the arguments and carries
145/// clingo's standard error.
146#[must_use]
147pub fn run_in<I>(directory: &Path, arguments: I, exits: &[i32], limits: Limits) -> Run
148where
149    I: IntoIterator,
150    I::Item: AsRef<OsStr>,
151{
152    let arguments: Vec<OsString> = arguments
153        .into_iter()
154        .map(|argument| argument.as_ref().to_owned())
155        .collect();
156    let subject = format!("{arguments:?}");
157    invoke(
158        &executable(),
159        directory,
160        &arguments,
161        exits,
162        limits,
163        &subject,
164    )
165}

167/// [`run_accepting`], with `executable` in clingo's place.
168fn run_source(
169    executable: &Path,
170    source: &str,
171    arguments: &[&str],
172    exits: &[i32],
173    limits: Limits,
174) -> Run {
175    let directory = tempfile::tempdir().expect("temporary directory for clingo");
176    let input = directory.path().join("case.lp");
177    std::fs::write(&input, source).expect("clingo input");
178    let mut arguments: Vec<OsString> = arguments.iter().map(OsString::from).collect();
179    arguments.push(input.into_os_string());
180    invoke(
181        executable,
182        directory.path(),
183        &arguments,
184        exits,
185        limits,
186        source,
187    )
188}

190/// One bounded run of `executable` in `directory`; `subject` names its input
191/// in a failure's message. The executable is clingo, except in this crate's
192/// tests of a run's contract.
193fn invoke(
194    executable: &Path,
195    directory: &Path,
196    arguments: &[OsString],
197    exits: &[i32],
198    limits: Limits,
199    subject: &str,
200) -> Run {
201    let (capture, pending) = process::invoke(
202        process::Invocation {
203            executable,
204            arguments,
205            directory,
206        },
207        process::Limits {
208            timeout: limits.timeout,
209            max_output_bytes: limits.max_output_bytes,
210            cleanup_timeout: Duration::from_secs(1),
211        },
212    )
213    .expect("clingo starts")
214    .into_parts();
215    if let Some(pending) = pending {
216        let cleanup = pending.retry(Duration::from_secs(1));
217        if let Some(pending) = cleanup.pending {
218            panic!(
219                "clingo child {} could not be reaped: {:?}",
220                pending.abandon(),
221                cleanup.failure
222            );
223        }
224        assert!(
225            cleanup.failure.is_none(),
226            "clingo's cleanup failed: {:?}",
227            cleanup.failure
228        );
229    }
230    let context = format!(
231        "clingo on {subject}\nstandard error:\n{}",
232        String::from_utf8_lossy(capture.stderr())
233    );
234    assert_eq!(capture.stop(), process::Stop::Completed, "{context}");
235    assert!(
236        capture.failure().is_none(),
237        "{:?}: {context}",
238        capture.failure()
239    );
240    assert!(
241        capture.cleanup_failure().is_none(),
242        "{:?}: {context}",
243        capture.cleanup_failure()
244    );
245    let exit = capture.exit().expect("a completed run has an exit");
246    assert!(exit.signal.is_none(), "signalled: {context}");
247    let code = exit.code.expect("an unsignalled exit has a code");
248    assert!(exits.contains(&code), "exited {code}: {context}");
249    Run {
250        code,
251        stdout: capture.stdout().to_vec(),
252        stderr: capture.stderr().to_vec(),
253    }
254}

256/// The JSON report of a run made with `--outf=2`.
257///
258/// # Panics
259/// Panics if the standard output is not one JSON document.
260#[must_use]
261pub fn json(run: &Run) -> Value {
262    serde_json::from_slice(run.stdout()).expect("clingo's JSON report")
263}

265/// Every answer set of `source` with its costs, as clingo enumerates them
266/// ([`ENUMERATION`]).
267///
268/// # Panics
269/// As [`run`] and [`model_records`], and if the report is not [`VERSION`]'s.
270#[must_use]
271pub fn records(source: &str) -> Records {
272    let report = json(&run(source, &ENUMERATION, Limits::default()));
273    assert_eq!(report["Solver"].as_str(), Some(VERSION));
274    model_records(&report)
275}

277/// Every witness of a complete JSON report, with its costs.
278///
279/// # Panics
280/// Panics unless the enumeration is complete and satisfiable, unsatisfiable
281/// or optimal, if a witness repeats, or if the report's model count differs
282/// from its witnesses.
283#[must_use]
284pub fn model_records(report: &Value) -> Records {
285    assert_eq!(
286        report["Models"]["More"].as_str(),
287        Some("no"),
288        "incomplete enumeration"
289    );
290    assert!(
291        matches!(
292            report["Result"].as_str(),
293            Some("SATISFIABLE" | "UNSATISFIABLE" | "OPTIMUM FOUND")
294        ),
295        "undecided result: {}",
296        report["Result"]
297    );
298    let mut records = Records::new();
299    let mut count = 0;
300    for call in report["Call"].as_array().expect("clingo calls") {
301        if let Some(witnesses) = call["Witnesses"].as_array() {
302            for witness in witnesses {
303                count += 1;
304                assert!(
305                    records.insert((atoms(&witness["Value"]), costs(&witness["Costs"]))),
306                    "duplicate full model"
307                );
308            }
309        }
310    }
311    assert_eq!(
312        report["Models"]["Number"].as_u64(),
313        Some(count),
314        "the model count contradicts the witnesses"
315    );
316    records
317}

319/// A witness's atoms, as their spellings.
320///
321/// # Panics
322/// Panics if `values` is not an array of strings.
323#[must_use]
324pub fn atoms(values: &Value) -> BTreeSet<String> {
325    values
326        .as_array()
327        .unwrap()
328        .iter()
329        .map(|atom| atom.as_str().unwrap().into())
330        .collect()
331}
332
333/// A witness's costs, absent when the program has no objective.
334///
335/// # Panics
336/// Panics if a cost is not an integer.
337#[must_use]
338pub fn costs(values: &Value) -> Option<Vec<i64>> {
339    values
340        .as_array()
341        .map(|values| values.iter().map(|value| value.as_i64().unwrap()).collect())
342}

344/// The answers a run reports, as `zetesis-validation` reads a clingo JSON
345/// report.
346///
347/// # Panics
348/// Panics if the report does not decode, or is not [`VERSION`]'s.
349#[must_use]
350pub fn answers(run: &Run) -> answers::ReportedAnswers {
351    let report = answers::clingo_json(run.stdout(), answers::Limits::default()).unwrap();
352    assert_eq!(report.solver(), VERSION);
353    report
354}
