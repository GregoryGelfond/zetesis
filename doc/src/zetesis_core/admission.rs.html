1//! Shared limits interpreted by semantic admission owners.
2
3const DEFAULT_MAX_TEMPLATES: usize = 100_000;
4const DEFAULT_MAX_PREDICATE_ARITY: usize = 32;
5const DEFAULT_MAX_VARIABLES: usize = 64;
6const DEFAULT_MAX_POSITIVE_BODY: usize = 1_024;
7const DEFAULT_MAX_DOMAIN_VALUES: usize = 1_000_000;
8// The same independent named-storage default as core relations and CPU closure.
9const DEFAULT_MAX_BYTES: usize = 128 * 1024 * 1024;
10
11/// Template-level admission budgets. Zero is a real limit, never unlimited.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct AdmissionLimits {
14    /// Maximum number of source-normalized rules: templates and fact occurrences.
15    pub max_templates: usize,
16    /// Maximum predicate arity.
17    pub max_predicate_arity: usize,
18    /// Maximum distinct variables in any one template.
19    pub max_variables_per_template: usize,
20    /// Maximum ordinary positive antecedents per template.
21    pub max_positive_body: usize,
22    /// Maximum distinct values appearing anywhere, including filters/gates.
23    pub max_domain_values: usize,
24    /// Inclusive canonical storage and admitted metadata allowance, including
25    /// indexes, vector capacity, admission scratch and publication overlap.
26    /// Caller-supplied construction descriptions, allocator bookkeeping and Arc
27    /// reference counters are excluded. The default is 128 MiB, independently
28    /// of the template/domain count limits; zero is a real allowance.
29    pub max_bytes: usize,
30}
31impl Default for AdmissionLimits {
32    fn default() -> Self {
33        Self {
34            max_templates: DEFAULT_MAX_TEMPLATES,
35            max_predicate_arity: DEFAULT_MAX_PREDICATE_ARITY,
36            max_variables_per_template: DEFAULT_MAX_VARIABLES,
37            max_positive_body: DEFAULT_MAX_POSITIVE_BODY,
38            max_domain_values: DEFAULT_MAX_DOMAIN_VALUES,
39            max_bytes: DEFAULT_MAX_BYTES,
40        }
41    }
42}
