1//! Whole-argument unification over borrowed terms and caller-owned scratch.
2//!
3//! Predicate selection and join traversal remain with the caller. This shared
4//! operation owns no payload and uses the canonical checked term comparator.
5
6use std::fmt;
7
8use crate::{PatternTerms, TemplateTerm, catalog::TermRef};

10/// Malformed input to an argument unification, rather than a logical mismatch.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum UnificationError {
13    /// The row ended before the requested argument could be read.
14    RowTooShort {
15        /// Complete pattern arity.
16        expected: usize,
17        /// Number of row arguments consumed before its end.
18        observed: usize,
19    },
20    /// A row argument remained after the complete pattern matched.
21    RowTooLong {
22        /// Complete pattern arity; at least one more row argument was present.
23        expected: usize,
24    },
25    /// A pattern variable is outside the supplied assignment.
26    Variable {
27        /// Requested source-variable coordinate.
28        variable: usize,
29        /// Number of assignment slots.
30        slots: usize,
31    },
32    /// A fresh capture would exceed the caller's already reserved trail.
33    TrailCapacity {
34        /// Existing trail capacity, in variable coordinates.
35        capacity: usize,
36    },
37}
38
39impl fmt::Display for UnificationError {
40    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
41        match self {
42            Self::RowTooShort { expected, observed } => write!(
43                formatter,
44                "argument row ended after {observed} values; expected {expected}"
45            ),
46            Self::RowTooLong { expected } => {
47                write!(formatter, "argument row exceeds arity {expected}")
48            }
49            Self::Variable { variable, slots } => {
50                write!(
51                    formatter,
52                    "variable {variable} is outside {slots} assignment slots"
53                )
54            }
55            Self::TrailCapacity { capacity } => {
56                write!(
57                    formatter,
58                    "capture trail exceeds its reserved {capacity} cells"
59                )
60            }
61        }
62    }
63}
64
65impl std::error::Error for UnificationError {}
66
67/// An invalid argument frame or the original caller's refused permit.
68#[derive(Clone, Debug, PartialEq, Eq)]
69pub enum UnificationFailure<E> {
70    /// Row shape, variable extent or prepared trail capacity is invalid.
71    Input(UnificationError),
72    /// Work or cancellation stopped before the next operation.
73    Stopped(E),
74}
75
76impl<E: fmt::Display> fmt::Display for UnificationFailure<E> {
77    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
78        match self {
79            Self::Input(error) => error.fmt(formatter),
80            Self::Stopped(error) => error.fmt(formatter),
81        }
82    }
83}
84
85impl<E: std::error::Error + 'static> std::error::Error for UnificationFailure<E> {
86    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
87        match self {
88            Self::Input(error) => Some(error),
89            Self::Stopped(error) => Some(error),
90        }
91    }
92}

94impl PatternTerms<'_> {
95    /// Unify a row's complete arguments with these constants and variables.
96    /// Predicates are not inspected; the caller selects a compatible signature.
97    /// Captures borrow `row`'s terms, and constants use checked structural
98    /// equality even across independent canonical owners.
99    ///
100    /// One permit precedes each argument's metadata/row read and possible
101    /// capture. Typed equality requests its own navigation and text permits.
102    /// After every argument matches, one final permit precedes checking the
103    /// row's end, including for a nullary pattern. At most arity + 1 row values
104    /// are requested. A mismatch returns immediately, so its unused suffix is
105    /// not inspected or shape-validated. Iterator-internal work remains the
106    /// caller's responsibility.
107    ///
108    /// No allocation occurs. A fresh variable is appended to `changes` only
109    /// when its existing capacity suffices, then its assignment slot is set.
110    /// Repeated variables see earlier captures from the same row. The caller
111    /// reserves/accountably owns the assignment and trail before this call.
112    /// Existing trail entries are preserved, and only previously absent slots
113    /// are appended. Work is linear in the inspected argument prefix plus
114    /// checked equality work; additional storage is constant.
115    ///
116    /// On mismatch, error or unwind, successful prefix captures remain in the
117    /// assignment and trail. The caller must undo that trail suffix before
118    /// trying another row, or discard the private frame on failure. `true`
119    /// alone establishes a complete argument match, never semantic membership.
120    ///
121    /// ```
122    /// use zetesis_core::{AtomPattern, PatternRef, Predicate, Term, Value};
123    /// use zetesis_core::catalog::TermRef;
124    /// let pattern = AtomPattern::new(Predicate::new("edge", 2)?,
125    ///     vec![Term::Variable(0), Term::Variable(0)])?;
126    /// let values = [Value::Number(7), Value::Number(7)];
127    /// let mut slots = [None];
128    /// let mut changes = Vec::with_capacity(1);
129    /// assert!(PatternRef::from(&pattern).terms().unify_with(
130    ///     values.iter().map(TermRef::from), &mut slots, &mut changes,
131    ///     || Ok::<_, std::convert::Infallible>(()),
132    /// )?);
133    /// assert_eq!(changes, [0]);
134    /// # Ok::<(), Box<dyn std::error::Error>>(())
135    /// ```
136    ///
137    /// # Errors
138    /// Returns the first observed row/slot/trail defect or caller refusal. A
139    /// refused final end probe is an error, not a successful match. There is no
140    /// implicit trail growth or fallback allocation.
141    ///
142    /// # Panics
143    /// Panics if the immutable pattern view violates its own reported length.
144    pub fn unify_with<'row, E>(
145        self,
146        row: impl IntoIterator<Item = TermRef<'row>>,
147        assignment: &mut [Option<TermRef<'row>>],
148        changes: &mut Vec<usize>,
149        mut before: impl FnMut() -> Result<(), E>,
150    ) -> Result<bool, UnificationFailure<E>> {
151        let mut row = row.into_iter();
152        let arity = self.len();
153        for column in 0..arity {
154            before().map_err(UnificationFailure::Stopped)?;
155            let term = self
156                .at(column)
157                .expect("column is within immutable pattern arity");
158            let value =
159                row.next()
160                    .ok_or(UnificationFailure::Input(UnificationError::RowTooShort {
161                        expected: arity,
162                        observed: column,
163                    }))?;
164            let expected = match term {
165                TemplateTerm::Constant(expected) => Some(expected),
166                TemplateTerm::Variable(variable) => {
167                    let slots = assignment.len();
168                    let target = assignment
169                        .get_mut(variable)
170                        .ok_or(UnificationFailure::Input(UnificationError::Variable {
171                            variable,
172                            slots,
173                        }))?;
174                    if let Some(expected) = *target {
175                        Some(expected)
176                    } else {
177                        if changes.len() == changes.capacity() {
178                            return Err(UnificationFailure::Input(
179                                UnificationError::TrailCapacity {
180                                    capacity: changes.capacity(),
181                                },
182                            ));
183                        }
184                        changes.push(variable);
185                        *target = Some(value);
186                        None
187                    }
188                }
189            };
190            if let Some(expected) = expected
191                && !expected
192                    .equals_ref_with(value, &mut before)
193                    .map_err(UnificationFailure::Stopped)?
194            {
195                return Ok(false);
196            }
197        }
198        before().map_err(UnificationFailure::Stopped)?;
199        if row.next().is_some() {
200            return Err(UnificationFailure::Input(UnificationError::RowTooLong {
201                expected: arity,
202            }));
203        }
204        Ok(true)
205    }
206}
