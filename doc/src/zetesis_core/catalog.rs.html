5//! remain distinct from canonical identities: equal atoms may occur at several
6//! positions in a supplied program.

40const LENGTH_BYTES: usize = size_of::<u64>();
41const ATOM_HEADER_BYTES: usize = 2 * LENGTH_BYTES + 1;

43/// Per-value limits for canonical import, construction and indexed lookup.
44/// The writer's total named-storage ceiling is separate. Unlike
45/// [`crate::ValueLimits`], bytes here do not count an owned flat node vector.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub struct Limits {
48    /// Maximum nodes in the expanded logical tree, counting repeated children.
49    pub max_nodes: usize,
50    /// Maximum root-inclusive logical depth.
51    pub max_depth: usize,
52    /// Maximum encoded identity and rendered spelling lengths plus temporary
53    /// operation ID capacity. Spelling is measured, not retained as another value.
54    pub max_bytes: usize,
55}
56
57impl Default for Limits {
58    fn default() -> Self {
59        Self {
60            max_nodes: 262_144,
61            max_depth: 128,
62            max_bytes: 16_777_216,
63        }
64    }
65}

67/// One append authority for canonical terms and complete atom tuples.
68///
69/// Read snapshots retain sealed allocations, never this writer or its indexes.
70/// Appending cannot mutate a previously published model. The storage ceiling
71/// measures the named capacities described by [`Error`], not process memory.
72pub struct Catalog {
73    store: storage::Store,
74}
75
76impl Catalog {
77    /// Create an empty authority with an inclusive named-storage allowance,
78    /// enforced before import and publication. The empty authority and its Arc
79    /// envelope are constructed here. Zero is a real allowance, not unlimited.
80    #[must_use]
81    pub fn new(max_bytes: usize) -> Self {
82        Self {
83            store: storage::Store::new(max_bytes),
84        }
85    }
86
87    /// Open a scoped append capability. Keys cannot escape this scope except
88    /// inside a published catalog or model. This prevents mixing owner-local
89    /// integers from unrelated catalogs without storing an owner in every cell.
90    ///
91    /// Successfully interned components may remain after a later refusal.
92    /// Existing snapshots remain valid; no partial atom row is published.
93    ///
94    /// An atom key from another authority cannot be supplied to this builder:
95    ///
96    /// ```compile_fail
97    /// use zetesis_core::{Atom, Predicate};
98    /// use zetesis_core::catalog::{Catalog, Limits};
99    /// let atom = Atom::new(Predicate::new("p", 0).unwrap(), vec![]).unwrap();
100    /// let mut first = Catalog::new(1_000_000);
101    /// let mut second = Catalog::new(1_000_000);
102    /// first.edit(Limits::default(), |left| {
103    ///     let foreign = left.intern_atom(&atom)?;
104    ///     second.edit(Limits::default(), |right| right.publish(&[foreign]))
105    /// });
106    /// ```
107    ///
108    /// # Errors
109    /// Returns construction, storage or caller failure from the operation.
110    pub fn edit<R>(
111        &mut self,
112        limits: Limits,
113        operation: impl for<'id> FnOnce(&mut Builder<'_, 'id>) -> Result<R, Error>,
114    ) -> Result<R, Error> {
115        operation(&mut Builder {
116            store: &mut self.store,
117            limits,
118            brand: PhantomData,
119        })
120    }
121}

123/// Scoped construction in one canonical owner. Cloning an immutable result
124/// never creates another append capability.
125pub struct Builder<'store, 'id> {
126    store: &'store mut storage::Store,
127    limits: Limits,
128    brand: PhantomData<fn(&'id ()) -> &'id ()>,
129}

131/// A complete canonical atom identity valid only in one edit scope.
132/// The key denotes neither an occurrence position nor a true atom.
133#[derive(Clone, Copy)]
134pub struct AtomKey<'id> {
135    id: storage::AtomId,
136    brand: PhantomData<fn(&'id ()) -> &'id ()>,
137}
138
139impl<'id> Builder<'_, 'id> {
140    /// Intern a typed ingress description. Equal complete atoms share one row;
141    /// strings and symbols remain distinct and argument order is preserved.
142    ///
143    /// # Errors
144    /// Refuses exceeded logical or storage bounds before publishing the atom.
145    pub fn intern_atom(&mut self, atom: &Atom) -> Result<AtomKey<'id>, Error> {
146        self.store.import_atom(atom, self.limits).map(|id| AtomKey {
147            id,
148            brand: PhantomData,
149        })
150    }
151
152    /// Publish original occurrence positions over the current immutable prefix.
153    /// Repeated keys retain repeated positions. Only integer mappings are copied;
154    /// authoritative term and atom payloads remain in sealed shared storage.
155    ///
156    /// # Errors
157    /// Refuses allocation or publication bounds without exposing a partial catalog.
158    pub fn publish(&mut self, atoms: &[AtomKey<'id>]) -> Result<AtomCatalog, Error> {
159        self.store
160            .check_publication(occurrence_bytes(atoms.len()))?;
161        let mut occurrences = Vec::new();
162        occurrences
163            .try_reserve_exact(atoms.len())
164            .map_err(|_| Error::Allocation)?;
165        occurrences.extend(atoms.iter().map(|atom| atom.id));
166        let snapshot = self
167            .store
168            .snapshot(occurrence_bytes(occurrences.capacity()))?;
169        Ok(AtomCatalog::published(snapshot, occurrences))
170    }
171}

173/// Immutable original atom occurrences over one canonical typed store.
174///
175/// Equal logical atoms may occupy different dense positions. Clone shares the
176/// snapshot and occurrence mapping in constant time. Models retain the declared
177/// prefix, including unselected occurrences, but never later append segments.
178#[derive(Clone)]
179pub struct AtomCatalog(Arc<CatalogData>);
180
181struct CatalogData {
182    snapshot: storage::Snapshot,
183    occurrences: Vec<storage::AtomId>,
184    canonical_bytes: Option<usize>,
185}
186
187/// Retained canonical-prefix population and named storage.
188/// These counts include the entire retained prefix, not only selected atoms or
189/// the occurrences in one model. They establish neither truth nor support.
190#[derive(Clone, Copy, Debug, PartialEq, Eq)]
191pub struct Storage {
192    /// Distinct typed terms, including interned subterms.
193    pub terms: usize,
194    /// Distinct complete atoms in the retained canonical prefix.
195    pub atoms: usize,
196    /// Named retained headers and capacities, including the occurrence mapping.
197    /// Arc reference counters and allocator bookkeeping are excluded. Adding
198    /// this value across shared catalogs overcounts shared allocations; it is
199    /// not the portable result-admission measure or process RSS.
200    pub bytes: u128,
201}
202
203impl AtomCatalog {
204    /// Import owned atom descriptions into canonical column storage, preserving
205    /// their original occurrence order. Input addresses and capacities are not
206    /// retained. This replaces the former vector-adoption contract.
207    ///
208    /// Values have already been admitted by their constructors. This operation
209    /// does not impose the default source depth limit a second time. Checked ID
210    /// widths, arithmetic and fallible buffer reservations still apply. Arc
211    /// envelope allocation remains infallible on stable Rust.
212    ///
213    /// # Errors
214    /// Returns a canonical storage or representation refusal; no catalog escapes.
215    pub fn new(atoms: Vec<Atom>) -> Result<Self, Error> {
216        let mut store = storage::Store::new(usize::MAX);
217        let mut occurrences = Vec::new();
218        occurrences
219            .try_reserve_exact(atoms.len())
220            .map_err(|_| Error::Allocation)?;
221        let limits = Limits {
222            max_nodes: usize::MAX,
223            max_depth: usize::MAX,
224            max_bytes: usize::MAX,
225        };
226        for atom in atoms {
227            occurrences.push(store.import_atom(&atom, limits)?);
228        }
229        Ok(Self::published(
230            store.snapshot(occurrence_bytes(occurrences.capacity()))?,
231            occurrences,
232        ))
233    }
234
235    fn published(snapshot: storage::Snapshot, occurrences: Vec<storage::AtomId>) -> Self {
236        let canonical_bytes = occurrences.iter().try_fold(LENGTH_BYTES, |bytes, &id| {
237            let atom = AtomRef::new(&snapshot, id).expect("published atom belongs to snapshot");
238            let bytes = bytes
239                .checked_add(ATOM_HEADER_BYTES)?
240                .checked_add(atom.predicate().name().len())?;
241            atom.values().iter().try_fold(bytes, |bytes, term| {
242                bytes.checked_add(term.canonical_bytes())
243            })
244        });
245        Self(Arc::new(CatalogData {
246            snapshot,
247            occurrences,
248            canonical_bytes,
249        }))
250    }
251
252    /// Borrow original dense-order occurrences without allocating or copying
253    /// typed payload. Resolving a reference searches immutable segment ranges.
254    #[must_use]
255    pub fn atoms(&self) -> Atoms<'_> {
256        Atoms {
257            snapshot: storage::Read::from(&self.0.snapshot),
258            positions: &self.0.occurrences,
259        }
260    }
261
262    /// Borrow this catalog's complete immutable canonical prefix. Occurrence
263    /// selection does not restrict which admitted identities can be resolved;
264    /// resolution alone establishes no support or truth.
265    #[must_use]
266    pub fn read(&self) -> CatalogRead<'_> {
267        CatalogRead(storage::Read::from(&self.0.snapshot))
268    }
269
270    /// This publication's private header and occurrence-map capacity.
271    /// The immutable snapshot and canonical segments are shared with its writer
272    /// or other readers and are excluded here. Add this amount to an owner
273    /// ledger that already counts that exact prefix, rather than adding the
274    /// catalog's complete retained storage a second time. Arc bookkeeping and
275    /// allocator metadata remain outside named capacity accounting.
276    #[must_use]
277    pub fn publication_bytes(&self) -> u128 {
278        occurrence_bytes(self.0.occurrences.capacity())
279    }
280
281    /// Retained capacity of the occurrence-to-identity mapping, in cells.
282    /// Canonical payload storage is separate; input vector capacity is not kept.
283    #[must_use]
284    pub fn capacity(&self) -> usize {
285        self.0.occurrences.capacity()
286    }
287
288    /// Whether both handles share the exact occurrence catalog allocation.
289    /// This is not logical equality or merely a shared term-store lineage.
290    #[must_use]
291    pub fn same_owner(&self, other: &Self) -> bool {
292        Arc::ptr_eq(&self.0, &other.0)
293    }
294
295    /// Whether both catalogs retain the exact same immutable prefix allocation.
296    /// Their occurrence maps may differ. Equal contents, a common vocabulary or
297    /// partially shared segments do not establish this stronger relation.
298    #[must_use]
299    pub fn shares_snapshot(&self, other: &Self) -> bool {
300        self.0.snapshot.shares_snapshot(&other.0.snapshot)
301    }
302
303    /// Shared immutable prefix allocations, excluding this catalog's header and
304    /// occurrence map. Subtract this subtotal once only when `shares_snapshot`
305    /// establishes that two catalogs retain exactly these same allocations.
306    /// Measuring visits sealed segment metadata; this is not process memory.
307    #[must_use]
308    pub fn shared_snapshot_bytes(&self) -> u128 {
309        self.0.snapshot.retained_bytes() - size_of::<storage::Snapshot>() as u128
310    }
311
312    /// Whether two catalogs share their exact canonical term authority.
313    /// Atom-row authorities, prefixes and occurrence positions may still differ.
314    #[must_use]
315    pub fn shares_terms(&self, other: &Self) -> bool {
316        storage::Read::from(&self.0.snapshot)
317            .same_vocabulary(storage::Read::from(&other.0.snapshot))
318    }
319
320    /// Portable encoded payload of all declared occurrences, including repeats
321    /// and unselected atoms. Canonical identities outside that occurrence map
322    /// are excluded, even though the shared prefix keeps them alive. See
323    /// [`Self::storage`] for named retained storage; neither measure is RSS.
324    #[must_use]
325    pub fn retained_payload_bytes(&self) -> Option<usize> {
326        self.0.canonical_bytes
327    }
328
329    /// Inspect the immutable prefix's population and named retained storage.
330    /// Work is proportional to the number of sealed storage segments. Later
331    /// append operations cannot change the returned charge for this catalog.
332    #[must_use]
333    pub fn storage(&self) -> Storage {
334        Storage {
335            terms: self.0.snapshot.term_count(),
336            atoms: self.0.snapshot.atom_count(),
337            bytes: self.0.snapshot.retained_bytes()
338                + (size_of::<CatalogData>() - size_of::<storage::Snapshot>()) as u128
339                + (self.0.occurrences.capacity() as u128) * size_of::<storage::AtomId>() as u128,
340        }
341    }
342
343    /// Checked version of [`Self::storage`]. Admission precedes each retained
344    /// vocabulary/row segment, predicate-column group and argument capacity read.
345    /// It visits metadata only, not individual atoms or term payload bytes.
346    ///
347    /// # Errors
348    /// Returns the first caller refusal without a completed storage measure.
349    pub fn storage_with<E>(&self, mut before: impl FnMut() -> Result<(), E>) -> Result<Storage, E> {
350        let bytes = self.0.snapshot.retained_bytes_with(&mut before)?;
351        before()?;
352        Ok(Storage {
353            terms: self.0.snapshot.term_count(),
354            atoms: self.0.snapshot.atom_count(),
355            bytes: bytes + self.publication_bytes() - size_of::<storage::Snapshot>() as u128,
356        })
357    }
358
359    pub(crate) fn owner_key(&self) -> usize {
360        Arc::as_ptr(&self.0).addr()
361    }
362}
363
364fn occurrence_bytes(capacity: usize) -> u128 {
365    size_of::<CatalogData>() as u128 + capacity as u128 * size_of::<storage::AtomId>() as u128
366}

383/// Allocation-free view of original dense atom occurrences.
384#[derive(Clone, Copy)]
385pub struct Atoms<'a> {
386    snapshot: storage::Read<'a>,
387    positions: &'a [storage::AtomId],
388}
389
390impl<'a> Atoms<'a> {
391    /// Borrow the already-bound canonical prefix, without another scope handle.
392    pub(crate) fn read(self) -> CatalogRead<'a> {
393        CatalogRead(self.snapshot)
394    }
395
396    fn new(snapshot: impl Into<storage::Read<'a>>, positions: &'a [storage::AtomId]) -> Self {
397        Self {
398            snapshot: snapshot.into(),
399            positions,
400        }
401    }
402    /// Number of original occurrences, including repeated logical atoms.
403    #[must_use]
404    pub fn len(self) -> usize {
405        self.positions.len()
406    }
407
408    /// Whether this occurrence range is empty.
409    #[must_use]
410    pub fn is_empty(self) -> bool {
411        self.positions.is_empty()
412    }
413
414    /// Borrow the occurrence at a checked dense position.
415    #[must_use]
416    pub fn at(self, position: usize) -> Option<AtomRef<'a>> {
417        self.positions.get(position).map(|&id| self.resolve(id))
418    }
419
420    /// Resolve an admitted identity in this immutable occurrence map. The map's
421    /// construction establishes prefix membership once, before any view exists.
422    fn resolve(self, id: storage::AtomId) -> AtomRef<'a> {
423        AtomRef::new(self.snapshot, id).expect("catalog occurrence is admitted")
424    }
425
426    /// Whether two live views borrow the exact same occurrence mapping and atom
427    /// authority. Logical equality alone does not establish position identity.
428    /// Constant time; neither view can outlive its mapping.
429    #[must_use]
430    pub fn same_occurrences(self, other: Self) -> bool {
431        self.snapshot.same_atoms(other.snapshot) && std::ptr::eq(self.positions, other.positions)
432    }
433
434    /// Compare positions already checked against this exact occurrence view.
435    pub(crate) fn compare_positions(self, left: usize, right: usize) -> Ordering {
436        self.resolve(self.positions[left])
437            .cmp(&self.resolve(self.positions[right]))
438    }
439
440    /// Search occurrences already sorted in semantic atom order. The returned
441    /// position is a dense occurrence, never a canonical identity. As for a
442    /// sorted slice, an unsorted input has no specified search result.
443    /// No allocation or atom materialization occurs.
444    ///
445    /// # Errors
446    /// Returns the insertion position when no occurrence denotes the atom.
447    pub fn binary_search(self, atom: AtomRef<'_>) -> Result<usize, usize> {
448        self.positions
449            .binary_search_by(|&id| self.resolve(id).cmp(&atom))
450    }
451
452    pub(crate) fn binary_search_key(self, key: &crate::AtomKey<'_>) -> Result<usize, usize> {
453        self.positions
454            .binary_search_by(|&id| key.compare_ref(self.resolve(id)).reverse())
455    }
456
457    /// Iterate in original dense order; cloning the cursor copies only borrows.
458    #[must_use]
459    pub fn iter(self) -> AtomIter<'a> {
460        AtomIter {
461            snapshot: self.snapshot,
462            positions: self.positions.iter(),
463        }
464    }
465}
466
467impl fmt::Debug for Atoms<'_> {
468    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
469        formatter.debug_list().entries(self.iter()).finish()
470    }
471}

473impl PartialEq for Atoms<'_> {
474    fn eq(&self, other: &Self) -> bool {
475        self.len() == other.len() && self.iter().eq(other.iter())
476    }

479impl PartialOrd for Atoms<'_> {

498/// Exact, double-ended traversal of original atom occurrences.
499#[derive(Clone)]
500pub struct AtomIter<'a> {
501    snapshot: storage::Read<'a>,
502    positions: slice::Iter<'a, storage::AtomId>,
503}
504
505impl<'a> Iterator for AtomIter<'a> {
506    type Item = AtomRef<'a>;
507    fn next(&mut self) -> Option<Self::Item> {
508        let id = *self.positions.next()?;
509        Some(AtomRef::new(self.snapshot, id).expect("catalog occurrence is admitted"))
510    }
511    fn size_hint(&self) -> (usize, Option<usize>) {
512        self.positions.size_hint()
513    }
514}
515
516impl DoubleEndedIterator for AtomIter<'_> {
517    fn next_back(&mut self) -> Option<Self::Item> {
518        let id = *self.positions.next_back()?;
519        Some(AtomRef::new(self.snapshot, id).expect("catalog occurrence is admitted"))
520    }
521}
