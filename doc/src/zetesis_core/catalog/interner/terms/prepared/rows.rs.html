11/// One variable argument projected from an original source-row occurrence.
12/// Input order is the caller's declared relation order, not a posting position,
13/// dictionary coordinate, source atom identity or assertion of truth.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub struct RowColumn {
16    /// Index of the immutable relation supplied during preparation.
17    pub input: usize,
18    /// Original argument column of that relation.
19    pub column: usize,
20}
21
22/// One pattern and its exact immutable row sources, bound to a single writer.
23/// This owns only projection metadata; all canonical tuples remain in their
24/// existing authority. It establishes no join, injectivity, freshness, support
25/// membership or complete producer family. The caller establishes those separate
26/// source properties and supplies only the actually selected rows.
27///
28/// Preparation allocates nothing: it takes ownership of the supplied vectors.
29/// Their capacities and this header belong to the caller's retained accounting,
30/// exposed by [`Self::storage_bytes`], and are excluded from interner `Limits`
31/// during both preparation and insertion. Borrowed source owners are separate.
32pub struct PreparedRows<'rows, 'source> {
33    pattern: PreparedPattern<'source>,
34    sources: Vec<&'rows Relation<'source>>,
35    columns: Vec<Option<RowColumn>>,
36    check_terms: bool,
37}
38
39impl PreparedRows<'_, '_> {
40    /// This owned header and metadata capacity, excluding borrowed sources and
41    /// the writer. Moving the input vectors creates no second allocation.
42    #[must_use]
43    pub fn storage_bytes(&self) -> u128 {
44        size_of::<Self>() as u128
45            + self.sources.capacity() as u128 * size_of::<&Relation<'_>>() as u128
46            + self.columns.capacity() as u128 * size_of::<Option<RowColumn>>() as u128
47    }
48}
49
50impl AtomAppender<'_> {
51    /// Prepare a pattern projected from exact immutable canonical relations.
52    /// `columns` follows head argument order: constants require `None`, variables
53    /// require a source column, and repeated occurrences of the same variable
54    /// require the same column. Different variables may share a column.
55    ///
56    /// Only metadata is inspected: no relation row is visited, selected or
57    /// published. Construction patterns or noncanonical relations return `None`
58    /// for the general route. Fixed constant limits are checked here; finite
59    /// variable limits apply only to rows later supplied for insertion. For `a`
60    /// head arguments and `b` sources, mapping validation costs O(a² + b)
61    /// metadata visits; repeated-slot comparisons are charged individually.
62    ///
63    /// # Errors
64    /// Refuses incompatible canonical owners, invalid column mappings, constant
65    /// logical limits, interner population/storage bounds or caller work.
66    /// Supplied vectors are consumed on every outcome; no writer state changes.
67    pub fn prepare_row_pattern_with<'rows, 'source, E>(
68        &self,
69        pattern: PatternRef<'source>,
70        sources: Vec<&'rows Relation<'source>>,
71        columns: Vec<Option<RowColumn>>,
72        term_limits: TermLimits,
73        limits: Limits,
74        mut before: impl FnMut() -> Result<(), E>,
75    ) -> Result<Option<PreparedRows<'rows, 'source>>, AssignedFailure<E>> {
76        let Some(pattern) = self.prepare_pattern_with(pattern, term_limits, limits, &mut before)?
77        else {
78            return Ok(None);
79        };
80        let read = storage::Read::from(&*self.store);
81        for source in &sources {
82            before().map_err(Failure::Stopped)?;
83            let Some(source) = source.canonical_read() else {
84                return Ok(None);
85            };
86            if !read.same_vocabulary(source.storage()) {
87                return Err(AssignedFailure::Assignment(AssignmentError::Read(
88                    ReadError::ForeignCatalog,
89                )));
90            }
91        }
92        if columns.len() != pattern.pattern.terms.len() {
93            return Err(shape());
94        }
95        for (index, term) in pattern.pattern.terms.iter().enumerate() {
96            before().map_err(Failure::Stopped)?;
97            match (*term, columns[index]) {
98                (TermData::Constant(_), None) => {}
99                (TermData::Variable(slot), Some(column)) => {
100                    let Some(source) = sources.get(column.input) else {
101                        return Err(shape());
102                    };
103                    if column.column >= source.predicate().arity() {
104                        return Err(shape());
105                    }
106                    for (prior, term) in pattern.pattern.terms[..index].iter().enumerate() {
107                        before().map_err(Failure::Stopped)?;
108                        if matches!(*term, TermData::Variable(known) if known == slot)
109                            && columns[prior] != Some(column)
110                        {
111                            return Err(shape());
112                        }
113                    }
114                }
115                _ => return Err(shape()),
116            }
117        }
118        // This uses admitted Store terms, not arbitrary Measures. Every new
119        // term passes Measures::check before publication, including the u128
120        // canonical+rendered byte sum against a usize limit. Hence that sum,
121        // nodes and depth all fit usize::MAX. Snapshots retain those same terms.
122        let check_terms = term_limits
123            != TermLimits {
124                max_nodes: usize::MAX,
125                max_depth: usize::MAX,
126                max_bytes: usize::MAX,
127            };
128        Ok(Some(PreparedRows {
129            pattern,
130            sources,
131            columns,
132            check_terms,
133        }))
134    }
135
136    /// Instantiate the prepared pattern from its exact selected relation rows.
137    /// The original signed predicate, constants, order and repeated arguments
138    /// are retained. No raw term IDs, assignment frame or tuple copy is accepted.
139    /// Identity, collision checks and both discovery indexes use the ordinary
140    /// canonical publication path; old and duplicate identities remain exact.
141    ///
142    /// # Errors
143    /// Refuses a different writer/relation, missing rows, selected logical term
144    /// limits, interner bounds or caller work. Failed insertion publishes no
145    /// discovery position; complete canonical components/capacities may remain.
146    ///
147    /// # Panics
148    /// Panics if an admitted canonical relation row loses its canonical term
149    /// coordinates. Safe relation construction and the exact-owner checks keep
150    /// those coordinates immutable for this operation.
151    pub fn insert_prepared_rows_with<E>(
152        &mut self,
153        prepared: &PreparedRows<'_, '_>,
154        rows: &[Row<'_, '_>],
155        limits: Limits,
156        mut before: impl FnMut() -> Result<(), E>,
157    ) -> Result<usize, AssignedFailure<E>> {
158        before().map_err(Failure::Stopped)?;
159        if !storage::Read::from(&*self.store).accepts_atom_scope(&prepared.pattern.owner) {
160            return Err(AssignedFailure::Assignment(AssignmentError::Read(
161                ReadError::ForeignCatalog,
162            )));
163        }
164        self.admit_entry(
165            Projected::HEADER_BYTES + size_of::<RowArguments<'_>>() as u128,
166            limits,
167        )?;
168        if rows.len() != prepared.sources.len() {
169            return Err(shape());
170        }
171        for (row, source) in rows.iter().zip(&prepared.sources) {
172            before().map_err(Failure::Stopped)?;
173            if !row.belongs_to(source) {
174                return Err(AssignedFailure::Assignment(AssignmentError::Read(
175                    ReadError::ForeignCatalog,
176                )));
177            }
178        }
179        let arguments = RowArguments {
180            source: prepared.pattern.source,
181            terms: &prepared.pattern.pattern.terms,
182            columns: &prepared.columns,
183            rows,
184        };
185        if prepared.check_terms {
186            for (index, term) in arguments.terms.iter().enumerate() {
187                if matches!(term, TermData::Variable(_)) {
188                    before().map_err(Failure::Stopped)?;
189                    let id = arguments
190                        .argument(index)
191                        .canonical()
192                        .expect("authenticated canonical row")
193                        .1;
194                    self.store
195                        .check_projected_term_with(id, prepared.pattern.term_limits, &mut before)
196                        .map_err(assigned_storage)?;
197                }
198            }
199        }
200        let projected = Projected {
201            predicate: prepared.pattern.pattern.predicate,
202            values: &[],
203            arguments: ProjectionSource::Rows(&arguments),
204        };
205        self.insert_projection_at(&projected, Some(prepared.pattern.locations), limits, before)
206    }
207}
208
209fn shape<E>() -> AssignedFailure<E> {
210    Failure::Catalog(storage::Fault::Shape).into()
211}
