1//! One append authority and immutable, fixed-prefix canonical storage.
2//!
3//! Imports retain no `Value`/`Atom` payload. Children precede parents; text and
4//! argument tuples occur once. Failed imports may retain complete interned
5//! components and reserved capacity, but never expose a partial term or row.
6//! Fixed owner, snapshot and segment envelopes use stable Rust's infallible
7//! Box/Arc allocation; named buffers and indexes use checked fallible reserves.

66/// A typed refusal; previously published snapshots remain valid.
67#[derive(Clone, Debug, PartialEq, Eq)]
68pub enum Fault {
69    /// The supplied logical value exceeds its admission contract.
70    Value(ValueError),
71    /// A named buffer or index could not reserve its required capacity.
72    Allocation,
73    /// Named storage capacities exceed the owner ceiling.
74    Storage {
75        /// Requested or observed named bytes, not allocator RSS.
76        required: u128,
77        /// Inclusive byte allowance.
78        limit: usize,
79    },
80    /// Another stable identifier cannot be represented without truncation.
81    IdExhausted,
82    /// A checked size or expanded-tree measure cannot be represented.
83    Overflow,
84    /// The input does not describe a complete valid value or atom.
85    Shape,
86    /// A closed vocabulary cannot admit a new text, term or predicate.
87    FrozenVocabulary,
88    /// Vocabulary freezing requires an authority with no materialized atoms.
89    VocabularyHasAtoms,
90    /// A published read snapshot does not retain indexed vocabulary admission.
91    UnindexedVocabulary,
92    /// A closed-catalog descendant cannot itself become another shared base.
93    CatalogHasBase,
94}
95
96impl From<ValueError> for Fault {
97    fn from(value: ValueError) -> Self {
98        Self::Value(value)
99    }
100}
101
102impl fmt::Display for Fault {
103    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
104        match self {
105            Self::Value(error) => error.fmt(f),
106            Self::Allocation => f.write_str("canonical storage reservation failed"),
107            Self::Storage { required, limit } => {
108                write!(f, "canonical storage {required} exceeds {limit} bytes")
109            }
110            Self::IdExhausted => f.write_str("canonical identifier space exhausted"),
111            Self::Overflow => f.write_str("canonical storage measure overflow"),
112            Self::Shape => f.write_str("canonical input shape is invalid"),
113            Self::FrozenVocabulary => {
114                f.write_str("canonical frozen vocabulary does not contain this identity")
115            }
116            Self::VocabularyHasAtoms => {
117                f.write_str("canonical vocabulary freeze contains atom rows")
118            }
119            Self::UnindexedVocabulary => {
120                f.write_str("canonical snapshot has no vocabulary lookup indexes")
121            }
122            Self::CatalogHasBase => f.write_str("canonical catalog already extends a closed base"),
123        }
124    }
125}

128/// Canonical import failed either in storage or at the caller's work boundary.
129/// The caller's stop value is preserved without conversion to a storage fault.
130#[derive(Clone, Debug, PartialEq, Eq)]
131pub enum Failure<E> {
132    /// A logical, capacity, shape or allocation refusal.
133    Storage(Fault),
134    /// The caller stopped before the next admitted operation.
135    Stopped(E),
136}
137
138impl<E> From<Fault> for Failure<E> {
139    fn from(error: Fault) -> Self {
140        Self::Storage(error)
141    }
142}
143
144impl<E: fmt::Display> fmt::Display for Failure<E> {
145    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
146        match self {
147            Self::Storage(error) => error.fmt(f),
148            Self::Stopped(error) => error.fmt(f),
149        }
150    }
151}
152
153impl<E: std::error::Error + 'static> std::error::Error for Failure<E> {
154    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
155        Some(match self {
156            Self::Storage(error) => error,
157            Self::Stopped(error) => error,
158        })
159    }
160}

166#[derive(Debug)]
167pub(crate) struct Store {
168    atom_owner: Arc<Owner>,
169    vocabulary: Vocabulary,
170    base: Option<Arc<IndexedRows>>,
171    sealed: Vec<Arc<RowSegment>>,
172    tail: RowSegment,
173    atoms: Index,
174    budget: Budget,
175}
176
177impl Store {
178    pub(crate) const fn current_bytes(&self) -> u128 {
179        self.budget.used
180    }
181    /// Named capacity and successful old/replacement overlap, not allocator RSS.
182    pub(crate) const fn peak_bytes(&self) -> u128 {
183        self.budget.peak
184    }
185    pub(crate) fn restart_peak(&mut self) {
186        self.budget.peak = self.budget.used;
187    }
188    pub(crate) fn ceiling(&mut self, max_bytes: usize) -> Result<(), Fault> {
189        self.budget.limit = max_bytes;
190        self.budget.check()
191    }
192
193    pub(crate) fn new(max_bytes: usize) -> Self {
194        Self {
195            atom_owner: Arc::new(Owner),
196            vocabulary: Vocabulary::new(),
197            base: None,
198            sealed: Vec::new(),
199            tail: RowSegment::new(0),
200            atoms: Index::default(),
201            budget: Budget::new(
202                max_bytes,
203                size_of::<Self>() + size_of::<vocabulary::Growing>(),
204            ),
205        }
206    }
207
208    /// A closed execution keeps one exact vocabulary and a fresh atom scope.
209    /// The shared base is included in this standalone writer's ceiling once.
210    pub(crate) fn with_vocabulary(base: FrozenVocabulary, max_bytes: usize) -> Result<Self, Fault> {
211        let mut budget = Budget::new(max_bytes, size_of::<Self>());
212        budget.add(base.retained_bytes())?;
213        Ok(Self {
214            atom_owner: Arc::new(Owner),
215            vocabulary: Vocabulary::Frozen(base),
216            base: None,
217            sealed: Vec::new(),
218            tail: RowSegment::new(0),
219            atoms: Index::default(),
220            budget,
221        })
222    }
223
224    /// Shared frozen allocations, already included in `current_bytes`. A combined
225    /// owner ledger may union this base instead of summing it per writer.
226    pub(crate) fn shared_vocabulary_bytes(&self) -> u128 {
227        match &self.vocabulary {
228            Vocabulary::Growing(_) => 0,
229            Vocabulary::Frozen(base) => base.retained_bytes(),
230        }
231    }
232
233    pub(crate) fn has_unpublished(&self) -> bool {
234        !self.tail.is_empty() || self.vocabulary.has_unpublished()
235    }
236
237    /// Initial read for empty discovery. Growing vocabulary starts empty; a
238    /// closed vocabulary and shared row base are available in full. Base rows
239    /// acquire this writer's fresh atom scope, without asserting discovery.
240    pub(crate) fn empty_snapshot(&self) -> Snapshot {
241        Snapshot {
242            atom_owner: Arc::clone(&self.atom_owner),
243            data: Arc::new(SnapshotData {
244                vocabulary: self.vocabulary.empty_prefix(),
245                base: self.base.as_ref().map(|base| Arc::clone(&base.payload)),
246                segments: Vec::new(),
247                atoms: self.base_count(),
248            }),
249        }
250    }
251
252    pub(crate) fn check_publication(&self, extra: u128) -> Result<(), Fault> {
253        self.budget.check_extra(extra)
254    }
255
256    fn counts(&self) -> Counts {
257        let mut counts = self.vocabulary.counts();
258        counts.atoms = self.tail.count();
259        counts
260    }
261
262    fn vocabulary_segment(
263        &self,
264        id: usize,
265        start: impl Fn(Counts) -> usize,
266    ) -> Option<&VocabularySegment> {
267        self.vocabulary.segment(id, start)
268    }
269
270    fn row_segment(&self, id: usize) -> Option<&RowSegment> {
271        if id >= self.tail.start {
272            Some(&self.tail)
273        } else {
274            self.rows().segment(id)
275        }
276    }
277
278    fn rows(&self) -> RowView<'_> {
279        RowView {
280            base: self.base.as_ref().map(|base| base.payload.as_ref()),
281            segments: &self.sealed,
282        }
283    }
284
285    fn base_count(&self) -> usize {
286        self.base.as_ref().map_or(0, |base| base.payload.atoms)
287    }
288}

290#[derive(Debug)]
291struct SnapshotData {
292    vocabulary: Arc<VocabularyData>,
293    base: Option<Arc<RowBase>>,
294    segments: Vec<Arc<RowSegment>>,
295    atoms: usize,
296}

298#[derive(Clone, Debug)]
299pub(crate) struct Snapshot {
300    atom_owner: Arc<Owner>,
301    data: Arc<SnapshotData>,
302}
303
304impl Snapshot {
305    pub(super) fn shares_snapshot(&self, other: &Self) -> bool {
306        Arc::ptr_eq(&self.atom_owner, &other.atom_owner) && Arc::ptr_eq(&self.data, &other.data)
307    }
308
309    /// Allocation-free buffers; fixed Arc envelopes retain the standard Rust
310    /// infallible allocation boundary.
311    pub(super) fn empty() -> Self {
312        Self {
313            atom_owner: Arc::new(Owner),
314            data: Arc::new(SnapshotData {
315                vocabulary: Arc::new(VocabularyData::empty(Arc::new(Owner))),
316                base: None,
317                segments: Vec::new(),
318                atoms: 0,
319            }),
320        }
321    }
322    pub(super) fn atom_count(&self) -> usize {
323        self.data.atoms
324    }
325    pub(super) fn term_count(&self) -> usize {
326        self.data.vocabulary.counts.terms
327    }
328    /// Independent prefix envelopes/directories; excludes a shared frozen base,
329    /// segments, inline Snapshot, Arc counters and allocator metadata.
330    pub(super) fn metadata_bytes(&self) -> u128 {
331        size_of::<SnapshotData>() as u128
332            + budget::capacity(&self.data.segments)
333            + if self.data.vocabulary.frozen {
334                0
335            } else {
336                self.data.vocabulary.metadata_bytes()
337            }
338    }
339    pub(super) fn retained_bytes(&self) -> u128 {
340        size_of::<Self>() as u128
341            + size_of::<SnapshotData>() as u128
342            + self.data.vocabulary.retained_bytes()
343            + self.rows().bytes()
344    }
345
346    pub(super) fn retained_bytes_with<E>(
347        &self,
348        mut before: impl FnMut() -> Result<(), E>,
349    ) -> Result<u128, E> {
350        before()?;
351        let mut bytes = size_of::<Self>() as u128
352            + size_of::<SnapshotData>() as u128
353            + self.data.vocabulary.metadata_bytes();
354        for segment in &self.data.vocabulary.segments {
355            before()?;
356            bytes += segment.bytes();
357        }
358        bytes += self.rows().bytes_with(&mut before)?;
359        Ok(bytes)
360    }
361
362    fn rows(&self) -> RowView<'_> {
363        RowView {
364            base: self.data.base.as_deref(),
365            segments: &self.data.segments,
366        }
367    }
368}

370fn locate<T>(segments: &[Arc<T>], id: usize, start: impl Fn(&T) -> usize) -> Option<&T> {
371    let after = segments.partition_point(|segment| start(segment) <= id);
372    after.checked_sub(1).map(|index| segments[index].as_ref())
373}
374
375fn next_id(count: usize) -> Result<u32, Fault> {
376    count.checked_add(1).ok_or(Fault::IdExhausted)?;
377    u32::try_from(count).map_err(|_| Fault::IdExhausted)
378}
