1//! Named-capacity accounting, with requested replacement and observed capacity
2//! checks. A refused reservation can leave larger retained capacity; it is
3//! charged before returning. These measures are not allocator/RSS guarantees.
4
5use std::mem::size_of;
6
7use super::Fault;
8
9#[derive(Clone, Debug)]
10pub(super) struct Budget {
11    pub(super) used: u128,
12    pub(super) peak: u128,
13    pub(super) limit: usize,
14}
15
16impl Budget {
17    pub(super) fn new(limit: usize, initial: usize) -> Self {
18        Self {
19            used: initial as u128,
20            peak: initial as u128,
21            limit,
22        }
23    }
24
25    pub(super) fn check(&self) -> Result<(), Fault> {
26        self.check_extra(0)
27    }
28
29    pub(super) fn check_extra(&self, extra: u128) -> Result<(), Fault> {
30        let required = self.used.checked_add(extra).ok_or(Fault::Overflow)?;
31        if required > self.limit as u128 {
32            return Err(Fault::Storage {
33                required,
34                limit: self.limit,
35            });
36        }
37        Ok(())
38    }
39
40    pub(super) fn add(&mut self, bytes: u128) -> Result<(), Fault> {
41        self.check_extra(bytes)?;
42        self.used += bytes;
43        self.peak = self.peak.max(self.used);
44        Ok(())
45    }
46
47    pub(super) fn observed(&mut self, before: u128, after: u128) -> Result<(), Fault> {
48        // Reservation succeeded. Count both old and replacement capacities as
49        // a conservative overlap envelope; rejected preflight checks never
50        // contribute to this peak. This is not an allocator trace.
51        let overlap = self.used.checked_add(after).ok_or(Fault::Overflow)?;
52        self.peak = self.peak.max(overlap);
53        self.used = self
54            .used
55            .checked_sub(before)
56            .and_then(|n| n.checked_add(after))
57            .ok_or(Fault::Overflow)?;
58        if overlap > self.limit as u128 {
59            return Err(Fault::Storage {
60                required: overlap,
61                limit: self.limit,
62            });
63        }
64        self.check()
65    }
66}
67
68pub(super) fn capacity<T>(values: &Vec<T>) -> u128 {
69    values.capacity() as u128 * size_of::<T>() as u128
70}
71
72fn target_capacity(len: usize, capacity: usize, additional: usize) -> Result<usize, Fault> {
73    let count = len.checked_add(additional).ok_or(Fault::Overflow)?;
74    Ok(if count <= capacity {
75        capacity
76    } else {
77        count.max(capacity.checked_mul(2).unwrap_or(count)).max(4)
78    })
79}
80
81/// Known requested capacity, before any allocation. An allocator may still
82/// return more, so callers must also check the observed capacity afterwards.
83pub(super) fn reservation_bytes<T>(values: &Vec<T>, additional: usize) -> Result<u128, Fault> {
84    Ok(
85        target_capacity(values.len(), values.capacity(), additional)? as u128
86            * size_of::<T>() as u128,
87    )
88}
89
90pub(super) fn reserve<T>(
91    values: &mut Vec<T>,
92    additional: usize,
93    budget: &mut Budget,
94) -> Result<(), Fault> {
95    let target = target_capacity(values.len(), values.capacity(), additional)?;
96    if target <= values.capacity() {
97        return budget.check();
98    }
99    let before = capacity(values);
100    budget.check_extra(target as u128 * size_of::<T>() as u128)?;
101    values
102        .try_reserve_exact(target - values.len())
103        .map_err(|_| Fault::Allocation)?;
104    budget.observed(before, capacity(values))
105}
106
107pub(super) fn reserve_text(
108    text: &mut String,
109    additional: usize,
110    budget: &mut Budget,
111) -> Result<(), Fault> {
112    let target = target_capacity(text.len(), text.capacity(), additional)?;
113    if target <= text.capacity() {
114        return budget.check();
115    }
116    let before = text.capacity() as u128;
117    budget.check_extra(target as u128)?;
118    text.try_reserve_exact(target - text.len())
119        .map_err(|_| Fault::Allocation)?;
120    budget.observed(before, text.capacity() as u128)
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126
127    #[test]
128    fn observed_overlap_refuses_excess_allocator_capacity() {
129        let mut budget = Budget::new(24, 12);
130        assert_eq!(
131            budget.observed(8, 16),
132            Err(Fault::Storage {
133                required: 28,
134                limit: 24
135            })
136        );
137        // A refusal cannot undo the completed allocation. Its retained capacity
138        // remains charged even though the transient envelope exceeded the limit.
139        assert_eq!(budget.used, 20);
140        assert_eq!(budget.peak, 28);
141    }
142
143    #[test]
144    fn observed_overlap_accepts_the_exact_allowance() {
145        let mut budget = Budget::new(28, 12);
146        assert_eq!(budget.observed(8, 16), Ok(()));
147        assert_eq!(budget.used, 20);
148        assert_eq!(budget.peak, 28);
149    }
