11/// Exact row result for one unchanged input under exclusive writer access.
12/// It owns no tuple, bucket address or permission to use another authority.
13pub(crate) struct PreparedAtom(RowPreparation);
14
15enum RowPreparation {
16    Present(AtomId),
17    Vacant {
18        predicate: PredicateId,
19        arity: usize,
20        hash: u64,
21        column: Option<usize>,
22    },
23}
24
25impl PreparedAtom {
26    /// A location is only a hint: publication checks its signed predicate in
27    /// the current tail before reuse. It carries no tuple-absence authority.
28    pub(crate) fn with_column(mut self, column: usize) -> Self {
29        if let RowPreparation::Vacant {
30            column: location, ..
31        } = &mut self.0
32        {
33            *location = Some(column);
34        }
35        self
36    }
37
38    /// Transfer the opaque single-use result into its publication state.
39    fn into_preparation(self) -> RowPreparation {
40        self.0
41    }
42
43    /// The caller has authenticated this identity in the same atom authority.
44    pub(crate) const fn existing(id: AtomId) -> Self {
45        Self(RowPreparation::Present(id))
46    }
47
48    pub(crate) const fn present(&self) -> Option<AtomId> {
49        match self.0 {
50            RowPreparation::Present(id) => Some(id),
51            RowPreparation::Vacant { .. } => None,
52        }
53    }
54}

70    pub(crate) fn declare_constructor_with<E>(
71        &mut self,
72        descriptor: ValueNodeRef<'_>,
73        mut before: impl FnMut() -> Result<(), E>,
74    ) -> Result<ConstructorData, Failure<E>> {
75        let mut work = Work::new(&mut before);
76        work.step()?;
77        self.budget.check()?;
78        let data = match descriptor {
79            ValueNodeRef::Function { name, sign, arity } if !name.is_empty() => {
80                ConstructorData::Function {
81                    name: self.intern_text_with(name, &mut work)?,
82                    sign,
83                    arity,
84                }
85            }
86            ValueNodeRef::Tuple { arity } => ConstructorData::Tuple { arity },
87            _ => return Err(Fault::Shape.into()),
88        };
89        work.step()?;
90        Ok(data)
91    }

93    fn intern_text_with<E>(
94        &mut self,
95        text: &str,
96        work: &mut Work<'_, E>,
97    ) -> Result<TextId, Failure<E>> {
98        work.step()?;
99        self.budget.check()?;
100        let hash = work.text_hash(text)?;
101        if let Some(id) = self.find_text_hash_with(text, hash, work)? {
102            return Ok(id);
103        }
104        work.step()?;
105        self.vocabulary.growing()?;
106        let id = TextId(next_id(self.counts().texts)?);
107        let growing = self.vocabulary.growing()?;
108        let quoted_bytes = work.quoted_bytes(text)?;
109        let start = growing.tail.text.len();
110        let end = start.checked_add(text.len()).ok_or(Fault::Overflow)?;
111        work.reserve_text(&mut growing.tail.text, text.len(), &mut self.budget)?;
112        work.reserve(&mut growing.tail.texts, 1, &mut self.budget)?;
113        growing
114            .indexes
115            .texts
116            .reserve_with(hash, &mut self.budget, work)?;
117        work.steps(text.len())?;
118        work.steps(3)?;
119        growing.tail.text.push_str(text);
120        growing.tail.texts.push(Text {
121            bytes: start..end,
122            quoted_bytes,
123        });
124        growing.indexes.texts.insert(hash, id.0);
125        Ok(id)
126    }

138    pub(super) fn intern_node_scratch_with<E>(
139        &mut self,
140        descriptor: ValueNodeRef<'_>,
141        children: &[TermId],
142        scratch: u128,
143        limits: Limits,
144        work: &mut Work<'_, E>,
145    ) -> Result<TermId, Failure<E>> {
146        work.step()?;
147        self.budget.check()?;
148        if arity(descriptor) != children.len() {
149            return Err(Fault::Shape.into());
150        }
151        let descriptor = normalized(descriptor)?;
152        let measures = node_measures(
153            descriptor,
154            children,
155            |child| self.term_measures(child),
156            work,
157        )?;
158        measures.check(limits)?;
159        ceiling(
160            ValueResource::Bytes,
161            measures.canonical as u128 + measures.rendered as u128 + scratch,
162            limits.max_bytes,
163        )?;
164        let key = node_key(descriptor, work, |text, work| {
165            self.intern_text_with(text, work).map(Some)
166        })?
167        .ok_or(Fault::Shape)?;
168        let hash = work.key_hash(key, children)?;
169        if let Some(id) = self
170            .vocabulary
171            .indexes()
172            .terms
173            .find_with(hash, work, |id, work| {
174                self.equal_node(TermId(id), key, children, work)
175            })?
176        {
177            return Ok(TermId(id));
178        }
179        work.step()?;
180        self.vocabulary.growing()?;
181        let id = TermId(next_id(self.counts().terms)?);
182        let growing = self.vocabulary.growing()?;
183        let payload = if key.compound() {
184            next_id(growing.tail.terms.compounds.len())?
185        } else {
186            key.payload
187        };
188        let start = growing.tail.terms.children.len();
189        let end = start.checked_add(children.len()).ok_or(Fault::Overflow)?;
190        self.reserve_node(key, children.len(), hash, work)?;
191        // Permit all child measure reads and writes before the first commit.
192        for _ in children {
193            work.steps(3)?;
194        }
195        work.steps(5)?;
196        if key.compound() {
197            let mut expanded = 0;
198            for child in children {
199                expanded += self.term_measures(*child).nodes;
200                let tail = &mut self.vocabulary.admitted_growing().tail;
201                tail.terms.children.push(*child);
202                tail.terms.ends.push(expanded);
203            }
204            self.vocabulary
205                .admitted_growing()
206                .tail
207                .terms
208                .compounds
209                .push(Compound {
210                    name: key.name,
211                    sign: key.sign,
212                    children: start..end,
213                    measures,
214                });
215        }
216        let growing = self.vocabulary.admitted_growing();
217        growing.tail.terms.kinds.push(key.kind);
218        growing.tail.terms.payloads.push(payload);
219        growing.indexes.terms.insert(hash, id.0);
220        Ok(id)
221    }

248    pub(super) fn find_node_with<E>(
249        &self,
250        descriptor: ValueNodeRef<'_>,
251        children: &[TermId],
252        scratch: u128,
253        limits: Limits,
254        work: &mut Work<'_, E>,
255    ) -> Result<Option<TermId>, Failure<E>> {
256        work.step()?;
257        if arity(descriptor) != children.len() {
258            return Err(Fault::Shape.into());
259        }
260        let descriptor = normalized(descriptor)?;
261        let measures = node_measures(
262            descriptor,
263            children,
264            |child| self.term_measures(child),
265            work,
266        )?;
267        measures.check(limits)?;
268        ceiling(
269            ValueResource::Bytes,
270            measures.canonical as u128 + measures.rendered as u128 + scratch,
271            limits.max_bytes,
272        )?;
273        let Some(key) = node_key(descriptor, work, |text, work| {
274            self.find_text_with(text, work)
275        })?
276        else {
277            return Ok(None);
278        };
279        let hash = work.key_hash(key, children)?;
280        self.vocabulary
281            .indexes()
282            .terms
283            .find_with(hash, work, |id, work| {
284                self.equal_node(TermId(id), key, children, work)
285            })
286            .map(|id| id.map(TermId))
287    }

289    fn equal_node<E>(
290        &self,
291        id: TermId,
292        key: NodeKey,
293        children: &[TermId],
294        work: &mut Work<'_, E>,
295    ) -> Result<bool, Failure<E>> {
296        work.step()?;
297        let (segment, local) = self.term_segment(id);
298        work.step()?;
299        if segment.terms.kinds[local] != key.kind {
300            return Ok(false);
301        }
302        let payload = segment.terms.payloads[local];
303        if !key.compound() {
304            return Ok(payload == key.payload);
305        }
306        work.step()?;
307        let compound = &segment.terms.compounds[payload as usize];
308        if compound.name != key.name
309            || compound.sign != key.sign
310            || compound.children.len() != children.len()
311        {
312            return Ok(false);
313        }
314        for (index, child) in children.iter().enumerate() {
315            work.step()?;
316            if segment.terms.children[compound.children.start + index] != *child {
317                return Ok(false);
318            }
319        }
320        Ok(true)
321    }

323    fn reserve_node<E>(
324        &mut self,
325        key: NodeKey,
326        children: usize,
327        hash: u64,
328        work: &mut Work<'_, E>,
329    ) -> Result<(), Failure<E>> {
330        let growing = self.vocabulary.growing()?;
331        work.reserve(&mut growing.tail.terms.kinds, 1, &mut self.budget)?;
332        work.reserve(&mut growing.tail.terms.payloads, 1, &mut self.budget)?;
333        if key.compound() {
334            work.reserve(&mut growing.tail.terms.compounds, 1, &mut self.budget)?;
335            work.reserve(&mut growing.tail.terms.children, children, &mut self.budget)?;
336            work.reserve(&mut growing.tail.terms.ends, children, &mut self.budget)?;
337        }
338        growing
339            .indexes
340            .terms
341            .reserve_with(hash, &mut self.budget, work)
342    }

344    pub(super) fn intern_predicate_with<E>(
345        &mut self,
346        predicate: PredicateRef<'_>,
347        work: &mut Work<'_, E>,
348    ) -> Result<PredicateId, Failure<E>> {
349        work.step()?;
350        self.budget.check()?;
351        work.step()?;
352        if let Some((read, id)) = predicate.canonical() {
353            work.step()?;
354            if read.vocabulary_belongs_to(self) {
355                return Ok(id);
356            }
357        }
358        work.step()?;
359        let name = predicate.name();
360        work.step()?;
361        let sign = predicate.sign();
362        work.step()?;
363        let arity = predicate.arity();
364        let signature = Signature {
365            name: self.intern_text_with(name, work)?,
366            sign,
367            arity,
368        };
369        let hash = work.key_hash(signature, &[])?;
370        if let Some(id) =
371            self.vocabulary
372                .indexes()
373                .predicates
374                .find_with(hash, work, |id, work| {
375                    work.step()?;
376                    Ok(self.signature(PredicateId(id)) == signature)
377                })?
378        {
379            return Ok(PredicateId(id));
380        }
381        work.step()?;
382        self.vocabulary.growing()?;
383        let id = PredicateId(next_id(self.counts().predicates)?);
384        let growing = self.vocabulary.growing()?;
385        work.reserve(&mut growing.tail.predicates, 1, &mut self.budget)?;
386        growing
387            .indexes
388            .predicates
389            .reserve_with(hash, &mut self.budget, work)?;
390        work.steps(3)?;
391        growing.tail.predicates.push(signature);
392        growing.indexes.predicates.insert(hash, id.0);
393        Ok(id)
394    }

409    /// Consume an exact result for the same authenticated tuple under one
410    /// exclusive writer. No logical atom insertion may intervene. Reservations
411    /// may rehash: only the hash survives preparation, never a bucket position.
412    /// `argument` is the same already authenticated immutable internal source:
413    /// it performs only infallible coordinate reads, with no caller callback or
414    /// deferred validation. All permits precede row writes. An arbitrary stateful
415    /// or panicking accessor would violate this private publication contract.
416    pub(crate) fn publish_prepared_atom_with<E>(
417        &mut self,
418        prepared: PreparedAtom,
419        argument: impl FnMut(usize) -> TermId,
420        mut before: impl FnMut() -> Result<(), E>,
421    ) -> Result<AtomId, Failure<E>> {
422        let mut work = Work::new(&mut before);
423        work.step()?;
424        self.budget.check()?;
425        self.publish_atom(prepared, argument, &mut work)
426    }

428    fn publish_atom<E>(
429        &mut self,
430        prepared: PreparedAtom,
431        mut argument: impl FnMut(usize) -> TermId,
432        work: &mut Work<'_, E>,
433    ) -> Result<AtomId, Failure<E>> {
434        let (predicate, arity, hash, column) = match prepared.into_preparation() {
435            RowPreparation::Present(id) => return Ok(id),
436            RowPreparation::Vacant {
437                predicate,
438                arity,
439                hash,
440                column,
441            } => (predicate, arity, hash, column),
442        };
443        work.step()?;
444        let id = AtomId(next_id(self.counts().atoms)?);
445        let local = u32::try_from(
446            (id.0 as usize)
447                .checked_sub(self.base_count())
448                .ok_or(Fault::Shape)?,
449        )
450        .map_err(|_| Fault::IdExhausted)?;
451        let block = self.column_block(predicate, arity, column, work)?;
452        for column in &mut self.tail.columns[block].arguments {
453            work.reserve(column, 1, &mut self.budget)?;
454        }
455        work.reserve(&mut self.tail.atoms, 1, &mut self.budget)?;
456        self.atoms.reserve_with(hash, &mut self.budget, work)?;
457        let columns = &mut self.tail.columns[block];
458        let row = columns.rows;
459        let rows = row.checked_add(1).ok_or(Fault::Overflow)?;
460        // As with relocation, one coordinate-copy permit covers its immutable
461        // source read and destination write. Admit the complete copy first.
462        work.steps(arity.checked_add(4).ok_or(Fault::Overflow)?)?;
463        // The private accessor only reads the validated, unchanged coordinate
464        // source; no caller code, validation or allocation occurs among writes.
465        for (index, column) in columns.arguments.iter_mut().enumerate() {
466            column.push(argument(index));
467        }
468        columns.rows = rows;
469        self.tail.atoms.push(Locator {
470            predicate,
471            block,
472            row,
473        });
474        self.atoms.insert(hash, local);
475        Ok(id)
476    }

492    fn prepare_atom<E>(
493        &self,
494        predicate: PredicateId,
495        arity: usize,
496        mut argument: impl FnMut(usize) -> TermId,
497        work: &mut Work<'_, E>,
498    ) -> Result<PreparedAtom, Failure<E>> {
499        work.step()?;
500        if self.signature(predicate).arity != arity {
501            return Err(Fault::Shape.into());
502        }
503        let hash = work.key_hash_by(predicate, arity, &mut argument)?;
504        self.prepare_atom_hash(predicate, arity, argument, hash, work)
505    }

507    fn prepare_atom_hash<E>(
508        &self,
509        predicate: PredicateId,
510        arity: usize,
511        argument: impl FnMut(usize) -> TermId,
512        hash: u64,
513        work: &mut Work<'_, E>,
514    ) -> Result<PreparedAtom, Failure<E>> {
515        let found = self.find_atom_hash_with(predicate, arity, argument, hash, work)?;
516        Ok(PreparedAtom(found.map_or(
517            RowPreparation::Vacant {
518                predicate,
519                arity,
520                hash,
521                column: None,
522            },
523            RowPreparation::Present,
524        )))
525    }

527    fn find_atom_hash_with<E>(
528        &self,
529        predicate: PredicateId,
530        arity: usize,
531        mut argument: impl FnMut(usize) -> TermId,
532        hash: u64,
533        work: &mut Work<'_, E>,
534    ) -> Result<Option<AtomId>, Failure<E>> {
535        if let Some(base) = &self.base
536            && let Some(id) = base.index.find_with(hash, work, |id, work| {
537                self.atom_matches(AtomId(id), predicate, arity, &mut argument, work)
538            })?
539        {
540            return Ok(Some(AtomId(id)));
541        }
542        let offset = self.base_count();
543        self.atoms
544            .find_with(hash, work, |local, work| {
545                let id = atom_offset(offset, local)?;
546                self.atom_matches(id, predicate, arity, &mut argument, work)
547            })?
548            .map(|local| atom_offset(offset, local))
549            .transpose()
550            .map_err(Failure::Storage)
551    }

553    fn atom_matches<E>(
554        &self,
555        id: AtomId,
556        predicate: PredicateId,
557        arity: usize,
558        argument: &mut impl FnMut(usize) -> TermId,
559        work: &mut Work<'_, E>,
560    ) -> Result<bool, Failure<E>> {
561        work.step()?;
562        let (columns, locator) = self.atom_columns(id);
563        if locator.predicate != predicate {
564            return Ok(false);
565        }
566        for index in 0..arity {
567            work.step()?;
568            if columns.arguments[index][locator.row] != argument(index) {
569                return Ok(false);
570            }
571        }
572        Ok(true)
573    }

575    /// Candidate location in the current tail, or its next insertion position.
576    /// No absence result survives this immutable operation. The consuming
577    /// writer must validate the actual block predicate before reusing it.
578    pub(crate) fn atom_column_location_with<E>(
579        &self,
580        predicate: PredicateId,
581        mut before: impl FnMut() -> Result<(), E>,
582    ) -> Result<usize, E> {
583        for (index, column) in self.tail.columns.iter().enumerate() {
584            before()?;
585            if column.predicate == predicate {
586                return Ok(index);
587            }
588        }
589        Ok(self.tail.columns.len())
590    }

592    fn column_block<E>(
593        &mut self,
594        predicate: PredicateId,
595        arity: usize,
596        location: Option<usize>,
597        work: &mut Work<'_, E>,
598    ) -> Result<usize, Failure<E>> {
599        if let Some(index) = location {
600            work.step()?;
601            if self
602                .tail
603                .columns
604                .get(index)
605                .is_some_and(|column| column.predicate == predicate)
606            {
607                return Ok(index);
608            }
609        }
610        let index = self.atom_column_location_with(predicate, || work.step())?;
611        if index < self.tail.columns.len() {
612            return Ok(index);
613        }
614        work.reserve(&mut self.tail.columns, 1, &mut self.budget)?;
615        let mut arguments = Vec::new();
616        let reserved = (|| {
617            work.reserve(&mut arguments, arity, &mut self.budget)?;
618            work.steps(arity.checked_add(1).ok_or(Fault::Overflow)?)
619        })();
620        if let Err(error) = reserved {
621            self.budget.used -= budget::capacity(&arguments);
622            return Err(error);
623        }
624        arguments.resize_with(arity, Vec::new);
625        let block = self.tail.columns.len();
626        self.tail.columns.push(Columns {
627            predicate,
628            arguments,
629            rows: 0,
630        });
631        Ok(block)
632    }

665fn atom_offset(offset: usize, local: u32) -> Result<AtomId, Fault> {
666    let global = offset.checked_add(local as usize).ok_or(Fault::Overflow)?;
667    u32::try_from(global)
668        .map(AtomId)
669        .map_err(|_| Fault::IdExhausted)
670}

672/// Writer and immutable lookup build exactly the same typed hash key. Only the
673/// authority to add a missing spelling differs at their text boundary.
674fn node_key<E>(
675    descriptor: ValueNodeRef<'_>,
676    work: &mut Work<'_, E>,
677    mut text: impl FnMut(&str, &mut Work<'_, E>) -> Result<Option<TextId>, Failure<E>>,
678) -> Result<Option<NodeKey>, Failure<E>> {
679    work.step()?;
680    let mut key = NodeKey {
681        kind: Kind::Infimum,
682        payload: 0,
683        name: None,
684        sign: Sign::Positive,
685    };
686    match descriptor {
687        ValueNodeRef::Infimum => {}
688        ValueNodeRef::Supremum => key.kind = Kind::Supremum,
689        ValueNodeRef::Number(number) => {
690            key.kind = Kind::Number;
691            key.payload = u32::from_ne_bytes(number.to_ne_bytes());
692        }
693        ValueNodeRef::String(value) | ValueNodeRef::Symbol(value) => {
694            key.kind = if matches!(descriptor, ValueNodeRef::String(_)) {
695                Kind::String
696            } else {
697                Kind::Symbol
698            };
699            let Some(id) = text(value, work)? else {
700                return Ok(None);
701            };
702            key.payload = id.0;
703        }
704        ValueNodeRef::Function { name, sign, .. } => {
705            key.kind = Kind::Function;
706            let Some(id) = text(name, work)? else {
707                return Ok(None);
708            };
709            key.name = Some(id);
710            key.sign = sign;
711        }
712        ValueNodeRef::Tuple { .. } => key.kind = Kind::Tuple,
713    }
714    Ok(Some(key))
715}
