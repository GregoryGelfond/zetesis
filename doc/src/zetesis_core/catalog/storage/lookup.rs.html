1//! Immutable indexed lookup. Local ID scratch and its peak belong to this
2//! capability, never to the shared writer or snapshot payload.
3
4use super::control::Work;
5use super::nodes::{Measures, arity, ceiling, descriptor_measures};
6use super::{Failure, Fault, Read, Store, TermId, budget};
7use crate::catalog::{Limits, TermRef};
8use crate::{ValueNodeRef, ValueResource};
9
10pub(crate) struct TermLookup<'a> {
11    store: &'a Store,
12    budget: budget::Budget,
13}
14
15impl<'a> TermLookup<'a> {
16    pub(crate) fn new(store: &'a Store) -> Self {
17        Self {
18            store,
19            budget: budget::Budget {
20                used: store.current_bytes(),
21                peak: store.current_bytes(),
22                limit: usize::MAX,
23            },
24        }
25    }
26    pub(crate) fn read(&self) -> Read<'a> {
27        Read::from(self.store)
28    }
29    pub(crate) fn peak_bytes(&self) -> u128 {
30        self.budget.peak
31    }
32    pub(crate) fn restart_peak(&mut self) {
33        self.budget.peak = self.budget.used;
34    }
35    pub(crate) fn ceiling(&mut self, limit: usize) -> Result<(), Fault> {
36        self.budget.limit = limit;
37        self.budget.check()
38    }
39
40    pub(crate) fn find_term_with<E>(
41        &mut self,
42        term: TermRef<'_>,
43        limits: Limits,
44        mut before: impl FnMut() -> Result<(), E>,
45    ) -> Result<Option<TermId>, Failure<E>> {
46        let mut work = Work::new(&mut before);
47        work.step()?;
48        self.budget.check()?;
49        work.step()?;
50        let nodes = term.expanded_nodes();
51        ceiling(ValueResource::Nodes, nodes as u128, limits.max_nodes)?;
52        work.step()?;
53        let canonical = term.canonical();
54        if let Some((read, id)) = canonical {
55            work.step()?;
56            let source = read.term(id).ok_or(Fault::Shape)?;
57            work.steps(4)?;
58            Measures {
59                nodes,
60                depth: source.depth(),
61                canonical: source.canonical_bytes(),
62                rendered: source.rendered_bytes(),
63            }
64            .check(limits)?;
65            work.step()?;
66            if read.vocabulary_belongs_to(self.store) {
67                return Ok(Some(id));
68            }
69        }
70        if nodes == 1 {
71            work.step()?;
72            return self
73                .store
74                .find_node_with(term.descriptor(), &[], 0, limits, &mut work);
75        }
76        let mut ids = Vec::new();
77        let result = (|| {
78            ceiling(
79                ValueResource::Bytes,
80                budget::reservation_bytes(&ids, nodes)?,
81                limits.max_bytes,
82            )?;
83            work.reserve(&mut ids, nodes, &mut self.budget)?;
84            let mut logical = 0u128;
85            // Reverse preorder leaves each node's children at the top of this
86            // stack in reverse argument order. Collapse them to one existing ID.
87            // Canonical rank selection may revisit ancestors; no linear-time
88            // claim is made for this constant-navigation-memory fallback.
89            for position in (0..nodes).rev() {
90                work.step()?;
91                let descriptor = if term.is_canonical() {
92                    let subtree = term
93                        .subterm_with(position, || work.step())?
94                        .ok_or(Fault::Shape)?;
95                    work.step()?;
96                    subtree.descriptor()
97                } else {
98                    term.flat_node(position).ok_or(Fault::Shape)?
99                };
100                ceiling(
101                    ValueResource::Bytes,
102                    logical
103                        .checked_add(descriptor.canonical_bytes())
104                        .and_then(|bytes| bytes.checked_add(budget::capacity(&ids)))
105                        .ok_or(Fault::Overflow)?,
106                    limits.max_bytes,
107                )?;
108                let head = descriptor_measures(descriptor, &mut work)?;
109                logical = logical
110                    .checked_add(head.canonical as u128 + head.rendered as u128)
111                    .ok_or(Fault::Overflow)?;
112                ceiling(
113                    ValueResource::Bytes,
114                    logical
115                        .checked_add(budget::capacity(&ids))
116                        .ok_or(Fault::Overflow)?,
117                    limits.max_bytes,
118                )?;
119                let children = arity(descriptor);
120                let start = ids.len().checked_sub(children).ok_or(Fault::Shape)?;
121                work.steps(children)?;
122                ids[start..].reverse();
123                let Some(id) = self.store.find_node_with(
124                    descriptor,
125                    &ids[start..],
126                    budget::capacity(&ids),
127                    limits,
128                    &mut work,
129                )?
130                else {
131                    return Ok(None);
132                };
133                work.step()?;
134                ids.truncate(start);
135                ids.push(id);
136            }
137            if ids.len() != 1 {
138                return Err(Fault::Shape.into());
139            }
140            Ok(Some(ids[0]))
141        })();
142        self.budget.used -= budget::capacity(&ids);
143        result
144    }

146    pub(crate) fn find_constructed_with<E>(
147        &mut self,
148        descriptor: ValueNodeRef<'_>,
149        values: &[Option<TermId>],
150        slots: &[usize],
151        limits: Limits,
152        mut before: impl FnMut() -> Result<(), E>,
153    ) -> Result<Option<TermId>, Failure<E>> {
154        let mut work = Work::new(&mut before);
155        let mut ids = Vec::new();
156        let result = (|| {
157            work.step()?;
158            self.budget.check()?;
159            ceiling(
160                ValueResource::Bytes,
161                budget::reservation_bytes(&ids, slots.len())?,
162                limits.max_bytes,
163            )?;
164            work.reserve(&mut ids, slots.len(), &mut self.budget)?;
165            for &slot in slots {
166                work.step()?;
167                let id = values.get(slot).copied().flatten().ok_or(Fault::Shape)?;
168                work.step()?;
169                ids.push(id);
170            }
171            self.store
172                .find_node_with(descriptor, &ids, budget::capacity(&ids), limits, &mut work)
173        })();
174        self.budget.used -= budget::capacity(&ids);
175        result
176    }
177}
