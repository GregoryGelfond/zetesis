1//! Extend sole-owned prefix directories after admitting every reservation.
2//!
3//! Segment payload remains immutable. Old readable prefixes retained by another
4//! consumer take the fresh-publication path instead. Repeated exclusive growth
5//! moves directory entries only at geometric capacity boundaries.
6
7use super::super::vocabulary::Vocabulary;
8use super::super::{Budget, Fault, RowSegment, Snapshot, Store, VocabularySegment, budget};
9use std::{mem::size_of, sync::Arc};
10
11impl Store {
12    /// Owner identity and extents authenticate the current sealed prefix. The
13    /// exclusive mutable borrow prevents new sharing after this decision.
14    pub(super) fn can_renew(&self, prior: &mut Snapshot) -> bool {
15        let Some(data) = Arc::get_mut(&mut prior.data) else {
16            return false;
17        };
18        if !Arc::ptr_eq(&self.atom_owner, &prior.atom_owner)
19            || data.atoms != self.tail.start
20            || data.segments.len() != self.sealed.len()
21        {
22            return false;
23        }
24        match &self.vocabulary {
25            Vocabulary::Frozen(base) => Arc::ptr_eq(&base.data, &data.vocabulary),
26            Vocabulary::Growing(growing) => {
27                let Some(vocabulary) = Arc::get_mut(&mut data.vocabulary) else {
28                    return false;
29                };
30                Arc::ptr_eq(&growing.owner, &vocabulary.owner)
31                    && growing.tail.start == vocabulary.counts
32                    && growing.sealed.len() == vocabulary.segments.len()
33            }
34        }
35    }
36
37    pub(super) fn renewal_steps(&self, prior: &Snapshot) -> Result<usize, Fault> {
38        let rows_added = !self.tail.is_empty();
39        let rows = growth_steps(&self.sealed, rows_added)
40            .checked_add(growth_steps(&prior.data.segments, rows_added))
41            .ok_or(Fault::Overflow)?;
42        let vocabulary = match &self.vocabulary {
43            Vocabulary::Frozen(_) => 0,
44            Vocabulary::Growing(growing) => {
45                let added = !growing.tail.is_empty();
46                growth_steps(&growing.sealed, added)
47                    .checked_add(growth_steps(&prior.data.vocabulary.segments, added))
48                    .ok_or(Fault::Overflow)?
49            }
50        };
51        // The same fixed metadata operations as fresh publication. Directory
52        // relocation is additional; existing references are otherwise untouched.
53        rows.checked_add(vocabulary)
54            .and_then(|steps| steps.checked_add(16))
55            .ok_or(Fault::Overflow)
56    }
57
58    pub(super) fn renew(&mut self, prior: &mut Snapshot, extra: u128) -> Result<(), Fault> {
59        let data = Arc::get_mut(&mut prior.data).expect("exclusive current snapshot");
60        let rows_added = !self.tail.is_empty();
61        let vocabulary_added = self.vocabulary.has_unpublished();
62        let headers = u128::from(rows_added) * size_of::<RowSegment>() as u128
63            + u128::from(vocabulary_added) * size_of::<VocabularySegment>() as u128;
64        let mut reservation = Reservation::new(&mut self.budget, extra, headers)?;
65        if rows_added {
66            reservation.writer(&mut self.sealed)?;
67        }
68        if let Vocabulary::Growing(growing) = &mut self.vocabulary
69            && vocabulary_added
70        {
71            reservation.writer(&mut growing.sealed)?;
72        }
73        if rows_added {
74            reservation.snapshot(&mut data.segments)?;
75        }
76        if vocabulary_added {
77            let vocabulary = Arc::get_mut(&mut data.vocabulary).expect("exclusive growing prefix");
78            reservation.snapshot(&mut vocabulary.segments)?;
79        }
80        // Every fallible reservation is complete. Counts and visible lengths
81        // stay unchanged until this point, even if an earlier reserve failed.
82        reservation.finish();
83        if let Vocabulary::Growing(growing) = &mut self.vocabulary
84            && vocabulary_added
85        {
86            let vocabulary = Arc::get_mut(&mut data.vocabulary).expect("exclusive growing prefix");
87            let counts = growing.tail.counts();
88            let tail = std::mem::replace(&mut growing.tail, VocabularySegment::new(counts));
89            let segment = Arc::new(tail);
90            vocabulary.segments.push(Arc::clone(&segment));
91            growing.sealed.push(segment);
92            vocabulary.counts = counts;
93        }
94        let atoms = self.tail.count();
95        if rows_added {
96            let tail = std::mem::replace(&mut self.tail, RowSegment::new(atoms));
97            let segment = Arc::new(tail);
98            data.segments.push(Arc::clone(&segment));
99            self.sealed.push(segment);
100        }
101        data.atoms = atoms;
102        Ok(())
103    }
104}

106fn growth_steps<T>(directory: &Vec<T>, append: bool) -> usize {
107    if append && directory.len() == directory.capacity() {
108        directory.len()
109    } else {
110        0
111    }
112}

114/// One live ledger includes the writer and caller-owned snapshot capacities.
115/// Only writer reservations update the writer's retained-byte ledger. A failed
116/// reservation still records actual capacity growth and replacement overlap.
117struct Reservation<'a> {
118    owner: &'a mut Budget,
119    live: Budget,
120    headers: u128,
121}
122
123impl<'a> Reservation<'a> {
124    fn new(owner: &'a mut Budget, extra: u128, headers: u128) -> Result<Self, Fault> {
125        owner.check_extra(extra)?;
126        owner.peak = owner.peak.max(owner.used + extra);
127        let mut live = owner.clone();
128        live.used += extra;
129        live.add(headers)?;
130        live.peak = live.used;
131        Ok(Self {
132            owner,
133            live,
134            headers,
135        })
136    }
137
138    fn writer<T>(&mut self, directory: &mut Vec<T>) -> Result<(), Fault> {
139        let before = budget::capacity(directory);
140        let result = self.snapshot(directory);
141        self.owner.used = self.owner.used - before + budget::capacity(directory);
142        result
143    }
144
145    fn snapshot<T>(&mut self, directory: &mut Vec<T>) -> Result<(), Fault> {
146        let result = budget::reserve(directory, 1, &mut self.live);
147        // The fixed headers are admitted but not allocated until finish.
148        self.owner.peak = self.owner.peak.max(self.live.peak - self.headers);
149        result
150    }
151
152    fn finish(self) {
153        self.owner.used += self.headers;
154        self.owner.peak = self.owner.peak.max(self.live.peak);
155    }
156}
