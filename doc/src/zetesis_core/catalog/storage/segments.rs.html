46#[derive(Debug)]
47pub(super) struct VocabularySegment {
48    pub(super) start: Counts,
49    pub(super) text: String,
50    pub(super) texts: Vec<Text>,
51    pub(super) terms: Nodes<TermId, TextId>,
52    pub(super) predicates: Vec<Signature>,
53}
54
55impl VocabularySegment {
56    pub(super) fn new(start: Counts) -> Self {
57        Self {
58            start,
59            text: String::new(),
60            texts: Vec::new(),
61            terms: Nodes::new(),
62            predicates: Vec::new(),
63        }
64    }
65
66    pub(super) fn counts(&self) -> Counts {
67        Counts {
68            texts: self.start.texts + self.texts.len(),
69            terms: self.start.terms + self.terms.kinds.len(),
70            predicates: self.start.predicates + self.predicates.len(),
71            atoms: 0,
72        }
73    }
74
75    pub(super) fn is_empty(&self) -> bool {
76        self.texts.is_empty() && self.terms.kinds.is_empty() && self.predicates.is_empty()
77    }
78
79    pub(super) fn bytes(&self) -> u128 {
80        size_of::<Self>() as u128
81            + self.text.capacity() as u128
82            + budget::capacity(&self.texts)
83            + self.terms.buffer_bytes()
84            + budget::capacity(&self.predicates)
85    }
86}

88#[derive(Debug)]
89pub(super) struct RowSegment {
90    pub(super) start: usize,
91    pub(super) atoms: Vec<Locator>,
92    pub(super) columns: Vec<Columns>,
93}
94
95impl RowSegment {
96    pub(super) fn new(start: usize) -> Self {
97        Self {
98            start,
99            atoms: Vec::new(),
100            columns: Vec::new(),
101        }
102    }
103
104    pub(super) fn count(&self) -> usize {
105        self.start + self.atoms.len()
106    }
107    pub(super) fn is_empty(&self) -> bool {
108        self.atoms.is_empty()
109    }
110    pub(super) fn bytes_with<E>(
111        &self,
112        mut before: impl FnMut() -> Result<(), E>,
113    ) -> Result<u128, E> {
114        before()?;
115        let mut bytes = size_of::<Self>() as u128
116            + budget::capacity(&self.atoms)
117            + budget::capacity(&self.columns);
118        for columns in &self.columns {
119            before()?;
120            bytes += budget::capacity(&columns.arguments);
121            for column in &columns.arguments {
122                before()?;
123                bytes += budget::capacity(column);
124            }
125        }
126        Ok(bytes)
127    }
128}

130impl Store {
131    fn text_measures(&self, id: TextId) -> (usize, usize) {
132        let segment = self
133            .vocabulary_segment(id.0 as usize, |counts| counts.texts)
134            .expect("admitted text ID");
135        text_measures(segment, id)
136    }
137
138    pub(super) fn text(&self, id: TextId) -> &str {
139        let segment = self
140            .vocabulary_segment(id.0 as usize, |counts| counts.texts)
141            .expect("admitted text ID");
142        text(segment, id)
143    }
144
145    pub(super) fn term_segment(&self, id: TermId) -> (&VocabularySegment, usize) {
146        let segment = self
147            .vocabulary_segment(id.0 as usize, |counts| counts.terms)
148            .expect("admitted term ID");
149        (segment, id.0 as usize - segment.start.terms)
150    }
151
152    pub(super) fn term_measures(&self, id: TermId) -> Measures {
153        let (segment, local) = self.term_segment(id);
154        segment
155            .terms
156            .measures(local, |id| self.text_measures(TextId(id)))
157    }
158
159    pub(super) fn signature(&self, id: PredicateId) -> Signature {
160        let segment = self
161            .vocabulary_segment(id.0 as usize, |counts| counts.predicates)
162            .expect("admitted predicate ID");
163        segment.predicates[id.0 as usize - segment.start.predicates]
164    }
165
166    pub(super) fn atom_columns(&self, id: AtomId) -> (&Columns, Locator) {
167        let segment = self.row_segment(id.0 as usize).expect("admitted atom ID");
168        let locator = segment.atoms[id.0 as usize - segment.start];
169        (&segment.columns[locator.block], locator)
170    }
171}
172
173pub(super) fn text(segment: &VocabularySegment, id: TextId) -> &str {
174    let range = segment.texts[id.0 as usize - segment.start.texts]
175        .bytes
176        .clone();
177    // Append accepts str, and each recorded endpoint is an append boundary.
178    // String slicing checks endpoints without rescanning the complete payload.
179    segment.text.get(range).expect("admitted text boundaries")
180}
181
182pub(super) fn text_measures(segment: &VocabularySegment, id: TextId) -> (usize, usize) {
183    let text = &segment.texts[id.0 as usize - segment.start.texts];
184    (text.bytes.len(), text.quoted_bytes)
185}

187#[derive(Clone, Copy, Debug)]
188pub(crate) struct Term<'a> {
189    pub(super) read: Read<'a>,
190    pub(super) segment: &'a VocabularySegment,
191    pub(super) local: usize,
192}
193
194impl<'a> Term<'a> {
195    pub(crate) fn descriptor(self) -> ValueNodeRef<'a> {
196        self.segment.terms.descriptor(
197            self.local,
198            |id| self.read.text(TextId(id)),
199            |id| self.read.text(id),
200        )
201    }
202
203    fn compound(self) -> Option<&'a Compound<TextId>> {
204        self.segment.terms.compound(self.local)
205    }
206
207    pub(crate) fn child(self, index: usize) -> Option<TermId> {
208        self.segment.terms.child(self.local, index)
209    }
210
211    pub(crate) fn child_end(self, index: usize) -> Option<usize> {
212        self.segment.terms.child_end(self.local, index)
213    }
214
215    fn measures(self) -> Measures {
216        self.segment
217            .terms
218            .measures(self.local, |id| self.read.text_measures(TextId(id)))
219    }
220    pub(crate) fn expanded_nodes(self) -> usize {
221        self.compound()
222            .map_or(1, |compound| compound.measures.nodes)
223    }
224    pub(crate) fn depth(self) -> usize {
225        self.compound()
226            .map_or(1, |compound| compound.measures.depth)
227    }
228    pub(crate) fn canonical_bytes(self) -> usize {
229        self.measures().canonical
230    }
231    pub(crate) fn rendered_bytes(self) -> usize {
232        self.measures().rendered
233    }
234}

236// The name is vocabulary text, resolved only by an operation that reads it;
237// arity and sign come from the signature alone.
238#[derive(Clone, Copy, Debug)]
239pub(crate) struct Predicate<'a> {
240    pub(super) read: Read<'a>,
241    pub(super) name: TextId,
242    pub(super) arity: usize,
243    pub(super) sign: Sign,
244}
245
246impl<'a> Predicate<'a> {
247    pub(crate) fn name(self) -> &'a str {
248        self.read.text(self.name)
249    }
250    pub(crate) fn sign(self) -> Sign {
251        self.sign
252    }
253    pub(crate) fn arity(self) -> usize {
254        self.arity
255    }
256}

258#[derive(Clone, Copy, Debug)]
259pub(crate) struct Atom<'a> {
260    pub(super) columns: &'a Columns,
261    pub(super) locator: Locator,
262}
263
264impl Atom<'_> {
265    pub(crate) fn predicate(self) -> PredicateId {
266        self.locator.predicate
267    }
268    // Publication checks each row against its signature's arity, and a block
269    // holds one column per argument, so no signature read is needed.
270    pub(crate) fn arity(self) -> usize {
271        self.columns.arguments.len()
272    }
273    pub(crate) fn argument(self, index: usize) -> Option<TermId> {
274        self.columns
275            .arguments
276            .get(index)
277            .map(|column| column[self.locator.row])
278    }
279}

281#[cfg(test)]
282mod layout_tests {
283    use super::*;
284
285    #[test]
286    fn grouped_lanes_preserve_the_vocabulary_header() {
287        // Before grouping, the segment held Counts, String and seven Vec
288        // headers directly. No extra owner, enum or box belongs in that layout.
289        let prior = size_of::<Counts>() + size_of::<String>() + 7 * size_of::<Vec<u8>>();
290        assert_eq!(size_of::<VocabularySegment>(), prior);
291        assert_eq!(
292            VocabularySegment::new(Counts::default()).bytes(),
293            prior as u128
294        );
295    }
