1//! Stable iterative index sorting; the caller supplies the comparison contract.
2
3use std::cmp::Ordering;
4
5/// Both buffers have the same initialized length. The comparator owns permits
6/// for reads/comparisons; this primitive permits each output write. On refusal
7/// neither buffer promises a complete order. Successful output is in `values`.
8/// Each pass merges adjacent sorted runs of `width` cells and doubles that
9/// width; the final pass covers the whole finite occurrence extent. Equal keys
10/// retain their input order because the left run wins an equal comparison.
11pub(crate) fn sort<E, F: FnMut() -> Result<(), E>>(
12    values: &mut Vec<usize>,
13    scratch: &mut Vec<usize>,
14    mut compare: impl FnMut(&[usize], usize, usize, &mut F) -> Result<Ordering, E>,
15    before: &mut F,
16) -> Result<(), E> {
17    let count = values.len();
18    let mut width = 1;
19    while width < count {
20        let mut start = 0;
21        while start < count {
22            let middle = start.saturating_add(width).min(count);
23            let end = middle.saturating_add(width).min(count);
24            let (mut left, mut right) = (start, middle);
25            for output in &mut scratch[start..end] {
26                let take_left = if right == end {
27                    true
28                } else if left == middle {
29                    false
30                } else {
31                    !compare(values, left, right, before)?.is_gt()
32                };
33                before()?;
34                *output = if take_left {
35                    let value = values[left];
36                    left += 1;
37                    value
38                } else {
39                    let value = values[right];
40                    right += 1;
41                    value
42                };
43            }
44            start = end;
45        }
46        std::mem::swap(values, scratch);
47        width = width.saturating_mul(2);
48    }
49    Ok(())
50}
