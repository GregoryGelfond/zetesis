1//! Stable iterative index sorting; the caller supplies the comparison contract.
2
3use std::cmp::Ordering;
4
5/// Both buffers have the same initialized length. The comparator owns permits
6/// for reads/comparisons; this primitive permits each output write. On refusal
7/// neither buffer promises a complete order. Successful output is in `values`.
8/// Each pass merges adjacent sorted runs of `width` cells and doubles that
9/// width; the final pass covers the whole finite occurrence extent. Equal keys
10/// retain their input order because the left run wins an equal comparison.
11///
12/// Two runs of more than one cell whose last and first cells are already in
13/// order are copied without another comparison: every cell of the left run
14/// is at most its last, and every cell of the right run at least its first,
15/// so their concatenation is the stable merge. An input that is already in
16/// order therefore costs one comparison for each pair of runs, fewer than
17/// its cells in all, where merging them compared half the cells in every
18/// pass. An input in no order pays one comparison more for each merge of
19/// longer runs, fewer than half its cells in all. Runs of one cell are merged
20/// directly: the merge's one comparison is the test.
21pub(crate) fn sort<E, F: FnMut() -> Result<(), E>>(
22    values: &mut Vec<usize>,
23    scratch: &mut Vec<usize>,
24    mut compare: impl FnMut(&[usize], usize, usize, &mut F) -> Result<Ordering, E>,
25    before: &mut F,
26) -> Result<(), E> {
27    let count = values.len();
28    let mut width = 1;
29    while width < count {
30        let mut start = 0;
31        while start < count {
32            let middle = start.saturating_add(width).min(count);
33            let end = middle.saturating_add(width).min(count);
34            let ordered =
35                width > 1 && middle < end && !compare(values, middle - 1, middle, before)?.is_gt();
36            let (mut left, mut right) = (start, middle);
37            for output in &mut scratch[start..end] {
38                let take_left = if right == end {
39                    true
40                } else if left == middle {
41                    false
42                } else {
43                    ordered || !compare(values, left, right, before)?.is_gt()
44                };
45                before()?;
46                *output = if take_left {
47                    let value = values[left];
48                    left += 1;
49                    value
50                } else {
51                    let value = values[right];
52                    right += 1;
53                    value
54                };
55            }
56            start = end;
57        }
58        std::mem::swap(values, scratch);
59        width = width.saturating_mul(2);
60    }
61    Ok(())
62}

71    /// The positions `0..keys.len()` sorted by their keys, and the number of
72    /// comparisons the sort asked for.
73    fn sorted(keys: &[usize]) -> (Vec<usize>, usize) {
74        let mut values: Vec<usize> = (0..keys.len()).collect();
75        let mut scratch = vec![0; keys.len()];
76        let mut comparisons = 0;
77        let mut permit = || Ok::<(), Infallible>(());
78        let result: Result<(), Infallible> = sort(
79            &mut values,
80            &mut scratch,
81            |cells, left, right, _| {
82                comparisons += 1;
83                Ok::<Ordering, Infallible>(keys[cells[left]].cmp(&keys[cells[right]]))
84            },
85            &mut permit,
86        );
87        result.unwrap();
88        (values, comparisons)
89    }

91    /// Whether `order` lists every position once, by ascending key, and equal
92    /// keys by ascending position.
93    fn stable(keys: &[usize], order: &[usize]) -> bool {
94        let mut seen = vec![false; keys.len()];
95        order
96            .iter()
97            .all(|&position| !std::mem::replace(&mut seen[position], true))
98            && order.len() == keys.len()
99            && order
100                .windows(2)
101                .all(|pair| (keys[pair[0]], pair[0]) < (keys[pair[1]], pair[1]))
102    }

104    /// A fixed permutation-like sequence of `count` keys with repetitions.
105    fn scattered(count: usize) -> Vec<usize> {
106        (0..count).map(|index| (index * 7919 + 13) % 257).collect()
107    }

109    #[test]
110    fn an_input_in_no_order_is_sorted_stably() {
111        for count in [0, 1, 2, 3, 5, 8, 100, 1000] {
112            let keys = scattered(count);
113            let (order, _) = sorted(&keys);
114            assert!(stable(&keys, &order), "{count}");
115        }
116    }

118    #[test]
119    fn an_ordered_input_costs_fewer_comparisons_than_cells() {
120        // One comparison for each pair of runs: 512 pairs of single cells,
121        // then 256, 128 and so on down to the last pair.
122        let keys: Vec<usize> = (0..1024).collect();
123        let (order, comparisons) = sorted(&keys);
124        assert_eq!(order, keys);
125        assert_eq!(comparisons, 1023);
126    }

131        let (order, comparisons) = sorted(&keys);
132        assert_eq!(order, (0..1024).collect::<Vec<_>>());
133        assert_eq!(comparisons, 1023);

136    #[test]
137    fn runs_of_one_cell_are_merged_by_one_comparison() {
138        // Reversed: no two runs are in order. Four merges of single cells
139        // take one comparison each; a test before them would double that.
140        // The two merges of pairs and the last of fours each pay their test
141        // and then merge: 4 + 2 * (1 + 2) + (1 + 4) = 15.
142        let keys: Vec<usize> = (0..8).rev().collect();
143        let (order, comparisons) = sorted(&keys);
144        assert!(stable(&keys, &order));
145        assert_eq!(comparisons, 15);
146    }

148    #[test]
149    fn runs_are_tested_by_the_last_cell_of_the_left_one() {
150        // The left run begins below the right one and ends above its start:
151        // only its last cell shows that the runs must be merged.
152        let keys = [1, 5, 2, 6];
153        let (order, _) = sorted(&keys);
154        assert_eq!(order, [0, 2, 1, 3]);
