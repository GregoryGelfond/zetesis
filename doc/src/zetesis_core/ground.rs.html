13/// Dense atom index within one [`GroundProgram`]. It is never a symbolic identity.
14pub type AtomId = u32;
15const WORD_BITS: usize = u32::BITS as usize;
16const DEFAULT_MAX_ATOMS: usize = 1_000_000;
17const DEFAULT_MAX_GROUND_RULES: usize = 1_000_000;
18const DEFAULT_MAX_SUBSTITUTIONS: usize = 10_000_000;

20/// Explicit limits for the eager static profile; zero is not unlimited.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub struct StaticLimits {
23    /// Maximum number of atom tuples materialized across every predicate.
24    pub max_atoms: usize,
25    /// Maximum filter-valid rules retained, including constraints.
26    pub max_ground_rules: usize,
27    /// Maximum complete variable assignments inspected across every template.
28    pub max_substitutions: usize,
29}
30impl Default for StaticLimits {
31    fn default() -> Self {
32        Self {
33            max_atoms: DEFAULT_MAX_ATOMS,
34            max_ground_rules: DEFAULT_MAX_GROUND_RULES,
35            max_substitutions: DEFAULT_MAX_SUBSTITUTIONS,
36        }
37    }
38}

72/// An explicitly eager graph with canonical dense IDs and instance identity.
73#[derive(Clone, Debug)]
74pub struct GroundProgram {
75    program: Program,
76    atoms: AtomCatalog,
77    rules: Vec<GroundRule>,
78    gate_atom_ids: Vec<AtomId>,
79}
80impl GroundProgram {
81    /// Materialize the complete carrier and all substitutions under hard bounds.
82    /// Filters use exact equality and ground antecedent duplicates are removed.
83    ///
84    /// # Errors
85    /// Returns [`StaticError`] for count overflow, budget exhaustion, allocation
86    /// refusal, or an internal inconsistency. None denotes logical UNSAT.
87    pub fn compile(program: &Program, limits: StaticLimits) -> Result<Self, StaticError> {
88        match Self::compile_with(program, limits, || Ok::<_, std::convert::Infallible>(())) {
89            Ok(graph) => Ok(graph),
90            Err(StaticFailure::Static(error)) => Err(error),
91            Err(StaticFailure::Stopped(never)) => match never {},
92        }
93    }
94
95    /// Materialize the same ordered carrier and rule family with caller control.
96    /// The callback runs before each dimension, carrier row, gate inspection,
97    /// substitution, assignment field, filter and instantiated pattern, and at
98    /// the canonical writer's existing checked operation boundaries. It does not
99    /// change count ceilings or establish a wall-clock latency bound for allocation,
100    /// one term comparison or sorting a rule's admitted antecedents.
101    ///
102    /// # Errors
103    /// Preserves static admission failures and returns the callback's exact error
104    /// as [`StaticFailure::Stopped`]. A refused private prefix is dropped; no
105    /// partial graph is published and the source program remains unchanged.
106    pub fn compile_with<E>(
107        program: &Program,
108        limits: StaticLimits,
109        mut before: impl FnMut() -> Result<(), E>,
110    ) -> Result<Self, StaticFailure<E>> {
111        before().map_err(StaticFailure::Stopped)?;
112        let mut atom_count = 0usize;
113        for predicate in program.predicates() {
114            before().map_err(StaticFailure::Stopped)?;
115            atom_count = atom_count
116                .checked_add(power(program.domain().len(), predicate.arity())?)
117                .ok_or(StaticError::CountOverflow)?;
118            bound("atoms", atom_count, limits.max_atoms)?;
119        }
120        if u32::try_from(atom_count).is_err() {
121            return Err(StaticError::DenseIdOverflow.into());
122        }
123        let mut substitution_count = 0usize;
124        for template in program.templates() {
125            before().map_err(StaticFailure::Stopped)?;
126            substitution_count = substitution_count
127                .checked_add(power(program.domain().len(), template.variable_count())?)
128                .ok_or(StaticError::CountOverflow)?;
129            bound(
130                "substitutions",
131                substitution_count,
132                limits.max_substitutions,
133            )?;
134        }
135        // Static admission already bounds the complete carrier. Its tuple
136        // writer shares the exact Program vocabulary; only rows and discovery
137        // positions are constructed here, never a second term store.
138        let mut atoms =
139            AtomInterner::for_program(program, usize::MAX).map_err(StaticError::Catalog)?;
140        let atom_limits = InternLimits {
141            max_atoms: atom_count,
142            max_bytes: u128::MAX,
143        };
144        for atom in program.carrier_atoms() {
145            before().map_err(StaticFailure::Stopped)?;
146            let atom = atom.map_err(StaticError::Carrier)?;
147            atoms
148                .entry_atom_with(atom.atom(), atom_limits, &mut before)
149                .map_err(intern_failure)?
150                .insert_with(atom_limits, &mut before)
151                .map_err(intern_failure)?;
152        }
153        let mut graph = Self {
154            program: program.clone(),
155            atoms: atoms
156                .into_catalog_with(atom_limits, &mut before)
157                .map_err(intern_failure)?,
158            rules: Vec::new(),
159            gate_atom_ids: Vec::new(),
160        };
161        for (index, atom) in graph.atoms.atoms().iter().enumerate() {
162            before().map_err(StaticFailure::Stopped)?;
163            if program.contains_gate_atom(atom) {
164                graph
165                    .gate_atom_ids
166                    .try_reserve(1)
167                    .map_err(|_| StaticError::Allocation)?;
168                graph
169                    .gate_atom_ids
170                    .push(u32::try_from(index).map_err(|_| StaticError::DenseIdOverflow)?);
171            }
172        }
173        for template in program.templates() {
174            graph.compile_template(template, limits.max_ground_rules, &mut before)?;
175        }
176        Ok(graph)
177    }

179    fn compile_template<E>(
180        &mut self,
181        template: TemplateRef<'_>,
182        max_rules: usize,
183        before: &mut impl FnMut() -> Result<(), E>,
184    ) -> Result<(), StaticFailure<E>> {
185        if self.program.domain().is_empty() && template.variable_count() != 0 {
186            return Ok(());
187        }
188        let mut coordinates = Vec::new();
189        coordinates
190            .try_reserve_exact(template.variable_count())
191            .map_err(|_| StaticError::Allocation)?;
192        coordinates.resize(template.variable_count(), 0);
193        loop {
194            before().map_err(StaticFailure::Stopped)?;
195            let mut assignment = Vec::new();
196            assignment
197                .try_reserve_exact(coordinates.len())
198                .map_err(|_| StaticError::Allocation)?;
199            for index in &coordinates {
200                before().map_err(StaticFailure::Stopped)?;
201                assignment.push(
202                    self.program
203                        .domain()
204                        .get(*index)
205                        .ok_or(StaticError::InvalidAdmittedProgram)?,
206                );
207            }
208            let mut enabled = true;
209            for filter in template.filters() {
210                before().map_err(StaticFailure::Stopped)?;
211                if !filter
212                    .evaluate(assignment.as_slice())
213                    .map_err(|_| StaticError::InvalidAdmittedProgram)?
214                {
215                    enabled = false;
216                    break;
217                }
218            }
219            if enabled {
220                bound(
221                    "ground rules",
222                    self.rules
223                        .len()
224                        .checked_add(1)
225                        .ok_or(StaticError::CountOverflow)?,
226                    max_rules,
227                )?;
228                let head = template
229                    .head()
230                    .map(|pattern| self.instantiate_id(pattern, &assignment, before))
231                    .transpose()?;
232                let rule = GroundRule {
233                    head,
234                    positive: self.instantiate_ids(template.positive(), &assignment, before)?,
235                    gate_true: self.instantiate_ids(template.gate_true(), &assignment, before)?,
236                    gate_false: self.instantiate_ids(template.gate_false(), &assignment, before)?,
237                };
238                self.rules
239                    .try_reserve(1)
240                    .map_err(|_| StaticError::Allocation)?;
241                self.rules.push(rule);
242            }
243            if !advance(&mut coordinates, self.program.domain().len()) {
244                break;
245            }
246        }
247        Ok(())
248    }

250    fn instantiate_id<E>(
251        &self,
252        pattern: PatternRef<'_>,
253        assignment: &[TermRef<'_>],
254        before: &mut impl FnMut() -> Result<(), E>,
255    ) -> Result<AtomId, StaticFailure<E>> {
256        before().map_err(StaticFailure::Stopped)?;
257        let key = pattern
258            .key(assignment)
259            .map_err(|_| StaticError::InvalidAdmittedProgram)?;
260        let position = self
261            .atoms
262            .atoms()
263            .binary_search_key(&key)
264            .map_err(|_| StaticError::InvalidAdmittedProgram)?;
265        u32::try_from(position).map_err(|_| StaticError::DenseIdOverflow.into())
266    }
267    fn instantiate_ids<E>(
268        &self,
269        patterns: Patterns<'_>,
270        assignment: &[TermRef<'_>],
271        before: &mut impl FnMut() -> Result<(), E>,
272    ) -> Result<Vec<AtomId>, StaticFailure<E>> {
273        let mut ids = Vec::new();
274        ids.try_reserve_exact(patterns.len())
275            .map_err(|_| StaticError::Allocation)?;
276        for pattern in patterns {
277            ids.push(self.instantiate_id(pattern, assignment, before)?);
278        }
279        ids.sort_unstable();
280        ids.dedup();
281        Ok(ids)
282    }

315    #[must_use]
316    pub fn atom_id<'a>(&self, atom: impl Into<crate::catalog::AtomRef<'a>>) -> Option<AtomId> {

320            .binary_search(atom)

330    /// Pack an instance-matched sparse seed, leaving all complement/tail bits zero.
331    ///
332    /// # Errors
333    /// Returns [`SeedError::WrongProgram`] for a foreign seed, or a capacity
334    /// refusal. Equal syntax admitted separately does not satisfy identity.
335    pub fn seed_words(&self, seed: &Seed) -> Result<Vec<u32>, SeedError> {
336        if !self.program.same_instance(seed.program()) {
337            return Err(SeedError::WrongProgram);
338        }
339        let mut words = Vec::new();
340        words
341            .try_reserve_exact(self.word_count())
342            .map_err(|_| SeedError::Allocation)?;
343        words.resize(self.word_count(), 0);
344        self.seed_words_into(seed.view(), &mut words)?;
345        Ok(words)
346    }

348    /// Pack a borrowed seed into caller-owned, exact-width storage. On success
349    /// every output word is replaced, including zero complement and tail bits.
350    /// No input tree, payload copy or temporary word vector is constructed.
351    /// Core-minted entries resolve by gate position; manual entries use canonical
352    /// binary lookup. Both initialize all output words and set each selected bit.
353    ///
354    /// # Errors
355    /// Refuses foreign program identity, then a wrong output word count, before
356    /// changing output. Equal source syntax admitted separately is foreign.
357    /// An atom absent from the graph reports [`SeedError::OutsideCarrier`]; a
358    /// missing indexed position reports [`SeedError::InvalidGatePosition`]. Both
359    /// violate the admitted complete-carrier invariant and may leave partial
360    /// output, which must not be used as a packed candidate.
361    pub fn seed_words_into(&self, seed: SeedView<'_>, words: &mut [u32]) -> Result<(), SeedError> {
362        if !self.program.same_instance(seed.program()) {
363            return Err(SeedError::WrongProgram);
364        }
365        if words.len() != self.word_count() {
366            return Err(SeedError::WordCount {
367                expected: self.word_count(),
368                actual: words.len(),
369            });
370        }
371        words.fill(0);
372        for atom in seed.entries() {
373            let id = atom.resolve_in(self)?;
374            let index = id as usize;
375            words[index / WORD_BITS] |= 1 << (index % WORD_BITS);
376        }
377        Ok(())
378    }

380    /// Decode exact words after validating length and zero unused tail bits.
381    /// The empty graph accepts the empty slice. The result alone claims no stability.
382    /// The interpretation shares this graph's whole atom catalog and retains
383    /// only a canonical selected-position vector of its own. No atom is cloned.
384    ///
385    /// # Errors
386    /// Returns [`WordError`] for an invalid word count, nonzero padding or
387    /// unavailable selected-position storage.
388    pub fn model_from_words(&self, words: &[u32]) -> Result<Model, WordError> {
389        if words.len() != self.word_count() {
390            return Err(WordError::Length {
391                expected: self.word_count(),
392                actual: words.len(),
393            });
394        }
395        let remainder = self.atom_count() % WORD_BITS;
396        if remainder != 0 && words.last().is_some_and(|word| word >> remainder != 0) {
397            return Err(WordError::TailBits);
398        }
399        Model::from_positions(
400            &self.atoms,
401            (0..self.atom_count())
402                .filter(|index| words[index / WORD_BITS] & (1 << (index % WORD_BITS)) != 0),
403        )
404        .map_err(WordError::Model)
405    }
406
407    /// Decode an arbitrary interpretation with this graph's atom-index meanings.
408    /// This is the accurately named counterpart of [`Self::model_from_words`];
409    /// no satisfaction or stability check occurs. Decoding scans the whole carrier
410    /// and selects positions in canonical order without cloning atom payloads.
411    /// Index storage is fallible; the shared Arc envelope allocation is infallible.
412    /// The result retains the entire atom catalog after the graph is dropped.
413    ///
414    /// # Errors
415    /// Refuses an incorrect word count, nonzero tail padding or unavailable
416    /// selection storage. Raw words carry
417    /// no instance identity; callers must establish their correspondence to this graph.
418    pub fn interpretation_from_words(
419        &self,
420        words: &[u32],
421    ) -> Result<crate::Interpretation, WordError> {
422        self.model_from_words(words)
423    }

426fn intern_failure<E>(error: InternFailure<E>) -> StaticFailure<E> {
427    let error = match error {
428        InternFailure::Catalog(error) => StaticError::Catalog(error),
429        InternFailure::Atoms { required, limit } => StaticError::LimitExceeded {
430            resource: "atoms",
431            actual: required,
432            limit,
433        },
434        InternFailure::Allocation(_) => StaticError::Allocation,
435        InternFailure::Overflow | InternFailure::Bytes { .. } => StaticError::CountOverflow,
436        InternFailure::Stopped(error) => return StaticFailure::Stopped(error),
437    };
438    StaticFailure::Static(error)
439}

441fn power(base: usize, exponent: usize) -> Result<usize, StaticError> {
442    let mut result = 1usize;
443    for _ in 0..exponent {
444        result = result.checked_mul(base).ok_or(StaticError::CountOverflow)?;
445    }
446    Ok(result)
447}
448fn bound(resource: &'static str, actual: usize, limit: usize) -> Result<(), StaticError> {
449    if actual > limit {
450        Err(StaticError::LimitExceeded {
451            resource,
452            actual,
453            limit,
454        })
455    } else {
456        Ok(())
457    }
458}

460/// Controlled static compilation refused before publishing a graph.
461#[derive(Clone, Debug, PartialEq, Eq)]
462pub enum StaticFailure<E> {
463    /// Existing count, allocation, catalog or admitted-program failure.
464    Static(StaticError),
465    /// Exact caller cancellation, deadline or injected refusal.
466    Stopped(E),
467}
468impl<E> From<StaticError> for StaticFailure<E> {
469    fn from(error: StaticError) -> Self {
470        Self::Static(error)
471    }
472}
473impl<E: fmt::Display> fmt::Display for StaticFailure<E> {
474    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
475        match self {
476            Self::Static(error) => error.fmt(f),
477            Self::Stopped(error) => error.fmt(f),
478        }
479    }
480}
481impl<E: std::error::Error + 'static> std::error::Error for StaticFailure<E> {
482    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
483        Some(match self {
484            Self::Static(error) => error,
485            Self::Stopped(error) => error,
486        })
487    }
488}

490/// Static compilation failed before any graph could be accepted.
491#[derive(Clone, Debug, PartialEq, Eq)]
492pub enum StaticError {
493    /// A finite Cartesian count exceeded the host integer representation.
494    CountOverflow,
495    /// The dense representation cannot identify every atom with a u32 ID.
496    DenseIdOverflow,
497    /// A configured materialization bound was exceeded.
498    LimitExceeded {
499        /// The counted representation or work category.
500        resource: &'static str,
501        /// Actual or proposed count.
502        actual: usize,
503        /// Configured maximum.
504        limit: usize,
505    },
506    /// A vector's storage could not be reserved.
507    Allocation,
508    /// The streaming atom carrier refused tuple storage.
509    Carrier(CarrierError),
510    /// Canonical term or atom storage refused publication.
511    Catalog(crate::catalog::Error),
512    /// An admitted pattern could not be instantiated inside its carrier.
513    InvalidAdmittedProgram,
514}
515impl fmt::Display for StaticError {
516    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
517        match self {
518            Self::CountOverflow => f.write_str("static Cartesian count overflow"),
519            Self::DenseIdOverflow => f.write_str("static atom carrier exceeds dense u32 IDs"),
520            Self::LimitExceeded {
521                resource,
522                actual,
523                limit,
524            } => write!(f, "static {resource} count {actual} exceeds {limit}"),
525            Self::Allocation => f.write_str("static storage could not be reserved"),
526            Self::Carrier(error) => error.fmt(f),
527            Self::Catalog(error) => error.fmt(f),
528            Self::InvalidAdmittedProgram => {
529                f.write_str("admitted template escaped its validated carrier")
530            }
531        }
532    }
533}
534impl std::error::Error for StaticError {}

536/// A dense result has an invalid representation, independently of its semantics.
537#[derive(Clone, Debug, PartialEq, Eq)]
538pub enum WordError {
539    /// The buffer has the wrong number of words.
540    Length {
541        /// Required words.
542        expected: usize,
543        /// Supplied words.
544        actual: usize,
545    },
546    /// Bits outside the finite atom carrier were set.
547    TailBits,
548    /// A decoded interpretation could not retain its checked selection.
549    Model(ModelError),
550}
551impl fmt::Display for WordError {
552    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
553        match self {
554            Self::Length { expected, actual } => {
555                write!(f, "expected {expected} result words, received {actual}")
556            }
557            Self::TailBits => f.write_str("result contains nonzero bits outside the atom carrier"),
558            Self::Model(error) => error.fmt(f),
559        }
560    }
561}
562impl std::error::Error for WordError {
563    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
564        match self {
565            Self::Model(error) => Some(error),
566            Self::Length { .. } | Self::TailBits => None,
567        }
568    }
