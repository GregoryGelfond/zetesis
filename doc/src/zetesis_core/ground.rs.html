13/// Dense atom index within one [`GroundProgram`]. It is never a symbolic identity.
14pub type AtomId = u32;
15const WORD_BITS: usize = u32::BITS as usize;
16const DEFAULT_MAX_ATOMS: usize = 1_000_000;
17const DEFAULT_MAX_GROUND_RULES: usize = 1_000_000;
18const DEFAULT_MAX_SUBSTITUTIONS: usize = 10_000_000;
19
20/// Explicit limits for the eager static profile; zero is not unlimited.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub struct StaticLimits {
23    /// Maximum number of atom tuples materialized across every predicate.
24    pub max_atoms: usize,
25    /// Maximum filter-valid rules retained, including constraints.
26    pub max_ground_rules: usize,
27    /// Maximum complete variable assignments inspected across every template.
28    pub max_substitutions: usize,
29}
30impl Default for StaticLimits {
31    fn default() -> Self {
32        Self {
33            max_atoms: DEFAULT_MAX_ATOMS,
34            max_ground_rules: DEFAULT_MAX_GROUND_RULES,
35            max_substitutions: DEFAULT_MAX_SUBSTITUTIONS,
36        }
37    }
38}

40/// A ground support rule. Every list is sorted and duplicate-free after
41/// substitution. Filters have already been checked; constraints have no head.
42#[derive(Clone, Debug, PartialEq, Eq)]
43pub struct GroundRule {
44    head: Option<AtomId>,
45    positive: Vec<AtomId>,
46    gate_true: Vec<AtomId>,
47    gate_false: Vec<AtomId>,
48}
49impl GroundRule {
50    /// Dense consequence ID, or `None` for a constraint.
51    #[must_use]
52    pub fn head(&self) -> Option<AtomId> {
53        self.head
54    }
55    /// Distinct ordinary positive antecedent IDs.
56    #[must_use]
57    pub fn positive(&self) -> &[AtomId] {
58        &self.positive
59    }
60    /// Distinct IDs required true in the frozen candidate.
61    #[must_use]
62    pub fn gate_true(&self) -> &[AtomId] {
63        &self.gate_true
64    }
65    /// Distinct IDs required false in the frozen candidate.
66    #[must_use]
67    pub fn gate_false(&self) -> &[AtomId] {
68        &self.gate_false
69    }
70}

72/// An explicitly eager graph with canonical dense IDs and instance identity.
73#[derive(Clone, Debug)]
74pub struct GroundProgram {
75    program: Program,
76    atoms: AtomCatalog,
77    rules: Vec<GroundRule>,
78    gate_atom_ids: Vec<AtomId>,
79}
80impl GroundProgram {
81    /// Materialize the complete carrier and all substitutions under hard bounds.
82    /// Filters use exact equality and ground antecedent duplicates are removed.
83    ///
84    /// # Errors
85    /// Returns [`StaticError`] for count overflow, budget exhaustion, allocation
86    /// refusal, or an internal inconsistency. None denotes logical UNSAT.
87    pub fn compile(program: &Program, limits: StaticLimits) -> Result<Self, StaticError> {
88        let mut atom_count = 0usize;
89        for predicate in program.predicates() {
90            atom_count = atom_count
91                .checked_add(power(program.domain().len(), predicate.arity())?)
92                .ok_or(StaticError::CountOverflow)?;
93            bound("atoms", atom_count, limits.max_atoms)?;
94        }
95        if u32::try_from(atom_count).is_err() {
96            return Err(StaticError::DenseIdOverflow);
97        }
98        let mut substitution_count = 0usize;
99        for template in program.templates() {
100            substitution_count = substitution_count
101                .checked_add(power(program.domain().len(), template.variable_count())?)
102                .ok_or(StaticError::CountOverflow)?;
103            bound(
104                "substitutions",
105                substitution_count,
106                limits.max_substitutions,
107            )?;
108        }
109        // Static admission already bounds the complete carrier. Its tuple
110        // writer shares the exact Program vocabulary; only rows and discovery
111        // positions are constructed here, never a second term store.
112        let mut atoms =
113            AtomInterner::for_program(program, usize::MAX).map_err(StaticError::Catalog)?;
114        let atom_limits = InternLimits {
115            max_atoms: atom_count,
116            max_bytes: u128::MAX,
117        };
118        for atom in program.carrier_atoms() {
119            let atom = atom.map_err(StaticError::Carrier)?;
120            atoms
121                .entry_atom_with(atom.atom(), atom_limits, || {
122                    Ok::<_, std::convert::Infallible>(())
123                })
124                .map_err(intern_failure)?
125                .insert_with(atom_limits, || Ok::<_, std::convert::Infallible>(()))
126                .map_err(intern_failure)?;
127        }
128        let mut graph = Self {
129            program: program.clone(),
130            atoms: atoms
131                .into_catalog_with(atom_limits, || Ok::<_, std::convert::Infallible>(()))
132                .map_err(intern_failure)?,
133            rules: Vec::new(),
134            gate_atom_ids: Vec::new(),
135        };
136        for (index, atom) in graph.atoms.atoms().iter().enumerate() {
137            if program.contains_gate_atom(atom) {
138                graph
139                    .gate_atom_ids
140                    .try_reserve(1)
141                    .map_err(|_| StaticError::Allocation)?;
142                graph
143                    .gate_atom_ids
144                    .push(u32::try_from(index).map_err(|_| StaticError::DenseIdOverflow)?);
145            }
146        }
147        for template in program.templates() {
148            graph.compile_template(template, limits.max_ground_rules)?;
149        }
150        Ok(graph)
151    }

153    fn compile_template(
154        &mut self,
155        template: TemplateRef<'_>,
156        max_rules: usize,
157    ) -> Result<(), StaticError> {
158        if self.program.domain().is_empty() && template.variable_count() != 0 {
159            return Ok(());
160        }
161        let mut coordinates = Vec::new();
162        coordinates
163            .try_reserve_exact(template.variable_count())
164            .map_err(|_| StaticError::Allocation)?;
165        coordinates.resize(template.variable_count(), 0);
166        loop {
167            let mut assignment = Vec::new();
168            assignment
169                .try_reserve_exact(coordinates.len())
170                .map_err(|_| StaticError::Allocation)?;
171            for index in &coordinates {
172                assignment.push(
173                    self.program
174                        .domain()
175                        .get(*index)
176                        .ok_or(StaticError::InvalidAdmittedProgram)?,
177                );
178            }
179            let mut enabled = true;
180            for filter in template.filters() {
181                if !filter
182                    .evaluate(assignment.as_slice())
183                    .map_err(|_| StaticError::InvalidAdmittedProgram)?
184                {
185                    enabled = false;
186                    break;
187                }
188            }
189            if enabled {
190                bound(
191                    "ground rules",
192                    self.rules
193                        .len()
194                        .checked_add(1)
195                        .ok_or(StaticError::CountOverflow)?,
196                    max_rules,
197                )?;
198                let head = template
199                    .head()
200                    .map(|pattern| self.instantiate_id(pattern, &assignment))
201                    .transpose()?;
202                let rule = GroundRule {
203                    head,
204                    positive: self.instantiate_ids(template.positive(), &assignment)?,
205                    gate_true: self.instantiate_ids(template.gate_true(), &assignment)?,
206                    gate_false: self.instantiate_ids(template.gate_false(), &assignment)?,
207                };
208                self.rules
209                    .try_reserve(1)
210                    .map_err(|_| StaticError::Allocation)?;
211                self.rules.push(rule);
212            }
213            if !advance(&mut coordinates, self.program.domain().len()) {
214                break;
215            }
216        }
217        Ok(())
218    }

220    fn instantiate_id(
221        &self,
222        pattern: PatternRef<'_>,
223        assignment: &[TermRef<'_>],
224    ) -> Result<AtomId, StaticError> {
225        let key = pattern
226            .key(assignment)
227            .map_err(|_| StaticError::InvalidAdmittedProgram)?;
228        let position = self
229            .atoms
230            .atoms()
231            .binary_search_key(&key)
232            .map_err(|_| StaticError::InvalidAdmittedProgram)?;
233        u32::try_from(position).map_err(|_| StaticError::DenseIdOverflow)
234    }
235    fn instantiate_ids(
236        &self,
237        patterns: Patterns<'_>,
238        assignment: &[TermRef<'_>],
239    ) -> Result<Vec<AtomId>, StaticError> {
240        let mut ids = Vec::new();
241        ids.try_reserve_exact(patterns.len())
242            .map_err(|_| StaticError::Allocation)?;
243        for pattern in patterns {
244            ids.push(self.instantiate_id(pattern, assignment)?);
245        }
246        ids.sort_unstable();
247        ids.dedup();
248        Ok(ids)
249    }

281    /// Find a symbolic atom's dense ID without changing the graph.
282    #[must_use]
283    pub fn atom_id<'a>(&self, atom: impl Into<crate::catalog::AtomRef<'a>>) -> Option<AtomId> {
284        let atom = atom.into();
285        self.atoms
286            .atoms()
287            .binary_search(atom)
288            .ok()
289            .and_then(|index| u32::try_from(index).ok())

297    /// Pack an instance-matched sparse seed, leaving all complement/tail bits zero.
298    ///
299    /// # Errors
300    /// Returns [`SeedError::WrongProgram`] for a foreign seed, or a capacity
301    /// refusal. Equal syntax admitted separately does not satisfy identity.
302    pub fn seed_words(&self, seed: &Seed) -> Result<Vec<u32>, SeedError> {
303        if !self.program.same_instance(seed.program()) {
304            return Err(SeedError::WrongProgram);
305        }
306        let mut words = Vec::new();
307        words
308            .try_reserve_exact(self.word_count())
309            .map_err(|_| SeedError::Allocation)?;
310        words.resize(self.word_count(), 0);
311        self.seed_words_into(seed.view(), &mut words)?;
312        Ok(words)
313    }

315    /// Pack a borrowed seed into caller-owned, exact-width storage. On success
316    /// every output word is replaced, including zero complement and tail bits.
317    /// No input tree, payload copy or temporary word vector is constructed.
318    /// Core-minted entries resolve by gate position; manual entries use canonical
319    /// binary lookup. Both initialize all output words and set each selected bit.
320    ///
321    /// # Errors
322    /// Refuses foreign program identity, then a wrong output word count, before
323    /// changing output. Equal source syntax admitted separately is foreign.
324    /// An atom absent from the graph reports [`SeedError::OutsideCarrier`]; a
325    /// missing indexed position reports [`SeedError::InvalidGatePosition`]. Both
326    /// violate the admitted complete-carrier invariant and may leave partial
327    /// output, which must not be used as a packed candidate.
328    pub fn seed_words_into(&self, seed: SeedView<'_>, words: &mut [u32]) -> Result<(), SeedError> {
329        if !self.program.same_instance(seed.program()) {
330            return Err(SeedError::WrongProgram);
331        }
332        if words.len() != self.word_count() {
333            return Err(SeedError::WordCount {
334                expected: self.word_count(),
335                actual: words.len(),
336            });
337        }
338        words.fill(0);
339        for atom in seed.entries() {
340            let id = atom.resolve_in(self)?;
341            let index = id as usize;
342            words[index / WORD_BITS] |= 1 << (index % WORD_BITS);
343        }
344        Ok(())
345    }

347    /// Decode exact words after validating length and zero unused tail bits.
348    /// The empty graph accepts the empty slice. The result alone claims no stability.
349    /// The interpretation shares this graph's whole atom catalog and retains
350    /// only a canonical selected-position vector of its own. No atom is cloned.
351    ///
352    /// # Errors
353    /// Returns [`WordError`] for an invalid word count, nonzero padding or
354    /// unavailable selected-position storage.
355    pub fn model_from_words(&self, words: &[u32]) -> Result<Model, WordError> {
356        if words.len() != self.word_count() {
357            return Err(WordError::Length {
358                expected: self.word_count(),
359                actual: words.len(),
360            });
361        }
362        let remainder = self.atom_count() % WORD_BITS;
363        if remainder != 0 && words.last().is_some_and(|word| word >> remainder != 0) {
364            return Err(WordError::TailBits);
365        }
366        Model::from_positions(
367            &self.atoms,
368            (0..self.atom_count())
369                .filter(|index| words[index / WORD_BITS] & (1 << (index % WORD_BITS)) != 0),
370        )
371        .map_err(WordError::Model)
372    }
373
374    /// Decode an arbitrary interpretation with this graph's atom-index meanings.
375    /// This is the accurately named counterpart of [`Self::model_from_words`];
376    /// no satisfaction or stability check occurs. Decoding scans the whole carrier
377    /// and selects positions in canonical order without cloning atom payloads.
378    /// Index storage is fallible; the shared Arc envelope allocation is infallible.
379    /// The result retains the entire atom catalog after the graph is dropped.
380    ///
381    /// # Errors
382    /// Refuses an incorrect word count, nonzero tail padding or unavailable
383    /// selection storage. Raw words carry
384    /// no instance identity; callers must establish their correspondence to this graph.
385    pub fn interpretation_from_words(
386        &self,
387        words: &[u32],
388    ) -> Result<crate::Interpretation, WordError> {
389        self.model_from_words(words)
390    }

393fn intern_failure(error: InternFailure<std::convert::Infallible>) -> StaticError {
394    match error {
395        InternFailure::Catalog(error) => StaticError::Catalog(error),
396        InternFailure::Atoms { required, limit } => StaticError::LimitExceeded {
397            resource: "atoms",
398            actual: required,
399            limit,
400        },
401        InternFailure::Allocation(_) => StaticError::Allocation,
402        InternFailure::Overflow | InternFailure::Bytes { .. } => StaticError::CountOverflow,
403        InternFailure::Stopped(never) => match never {},
404    }
405}

407fn power(base: usize, exponent: usize) -> Result<usize, StaticError> {
408    let mut result = 1usize;
409    for _ in 0..exponent {
410        result = result.checked_mul(base).ok_or(StaticError::CountOverflow)?;
411    }
412    Ok(result)
413}
414fn bound(resource: &'static str, actual: usize, limit: usize) -> Result<(), StaticError> {
415    if actual > limit {
416        Err(StaticError::LimitExceeded {
417            resource,
418            actual,
419            limit,
420        })
421    } else {
422        Ok(())
423    }
424}

426/// Static compilation failed before any graph could be accepted.
427#[derive(Clone, Debug, PartialEq, Eq)]
428pub enum StaticError {
429    /// A finite Cartesian count exceeded the host integer representation.
430    CountOverflow,
431    /// The dense representation cannot identify every atom with a u32 ID.
432    DenseIdOverflow,
433    /// A configured materialization bound was exceeded.
434    LimitExceeded {
435        /// The counted representation or work category.
436        resource: &'static str,
437        /// Actual or proposed count.
438        actual: usize,
439        /// Configured maximum.
440        limit: usize,
441    },
442    /// A vector's storage could not be reserved.
443    Allocation,
444    /// The streaming atom carrier refused tuple storage.
445    Carrier(CarrierError),
446    /// Canonical term or atom storage refused publication.
447    Catalog(crate::catalog::Error),
448    /// An admitted pattern could not be instantiated inside its carrier.
449    InvalidAdmittedProgram,
450}
451impl fmt::Display for StaticError {
452    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
453        match self {
454            Self::CountOverflow => f.write_str("static Cartesian count overflow"),
455            Self::DenseIdOverflow => f.write_str("static atom carrier exceeds dense u32 IDs"),
456            Self::LimitExceeded {
457                resource,
458                actual,
459                limit,
460            } => write!(f, "static {resource} count {actual} exceeds {limit}"),
461            Self::Allocation => f.write_str("static storage could not be reserved"),
462            Self::Carrier(error) => error.fmt(f),
463            Self::Catalog(error) => error.fmt(f),
464            Self::InvalidAdmittedProgram => {
465                f.write_str("admitted template escaped its validated carrier")
466            }
467        }
468    }
469}
470impl std::error::Error for StaticError {}

472/// A dense result has an invalid representation, independently of its semantics.
473#[derive(Clone, Debug, PartialEq, Eq)]
474pub enum WordError {
475    /// The buffer has the wrong number of words.
476    Length {
477        /// Required words.
478        expected: usize,
479        /// Supplied words.
480        actual: usize,
481    },
482    /// Bits outside the finite atom carrier were set.
483    TailBits,
484    /// A decoded interpretation could not retain its checked selection.
485    Model(ModelError),
486}
487impl fmt::Display for WordError {
488    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
489        match self {
490            Self::Length { expected, actual } => {
491                write!(f, "expected {expected} result words, received {actual}")
492            }
493            Self::TailBits => f.write_str("result contains nonzero bits outside the atom carrier"),
494            Self::Model(error) => error.fmt(f),
495        }
496    }
497}
498impl std::error::Error for WordError {
499    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
500        match self {
501            Self::Model(error) => Some(error),
502            Self::Length { .. } | Self::TailBits => None,
503        }
504    }
