1//! Device-independent relational vocabulary for zetesis.
2//!
3//! Programs are immutable admitted templates. A [`Seed`] is an exact sparse true
4//! set bound to one program instance; absent tuples are its false complement.
5//! [`GroundProgram::compile`] is explicitly eager. Admission, seed construction,
6//! and carrier-iterator creation never invoke it. See the
7//! [library map](https://github.com/GregoryGelfond/zetesis/blob/main/docs/book/rust/libraries.md)
8//! and [grounding architecture](https://github.com/GregoryGelfond/zetesis/blob/main/docs/book/architecture/grounding.md).
9//!
10//! ```
11//! use zetesis_core::{AdmissionLimits, Atom, AtomPattern, GroundProgram,
12//!     Predicate, Program, Seed, StaticLimits, Template};
13//!
14//! let signature = Predicate::new("selected", 0)?;
15//! let selected = AtomPattern::new(signature.clone(), vec![])?;
16//! // The normalized singleton choice `{selected}.` uses a true candidate gate.
17//! let choice = Template::new(Some(selected.clone()), vec![], vec![selected], vec![], vec![]);
18//! let program = Program::new(vec![choice], AdmissionLimits::default())?;
19//! let seed = Seed::new(&program, [Atom::new(signature, vec![])?])?;
20//! let graph = GroundProgram::compile(&program, StaticLimits::default())?;
21//! assert_eq!(graph.seed_words(&seed)?, vec![1]);
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```
