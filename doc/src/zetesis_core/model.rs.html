17/// A canonical true-atom set, also named [`Interpretation`]. Construction
18/// establishes neither derivation, satisfaction nor answer-set membership.
19///
20/// One shared immutable catalog owns the atoms; one ordered index selection
21/// denotes the true set. Clone shares both in constant time without allocation
22/// or payload copying. Equality and ordering compare logical atoms, independently
23/// of catalog identity or dense index order. No program or oracle is retained.
24/// A sparse model can retain a much larger catalog than its selected true set.
25#[derive(Clone)]
26pub struct Model(Arc<Selected>);
27
28struct Selected {
29    catalog: AtomCatalog,
30    positions: Vec<usize>,
31}

33impl Model {
34    /// Named private selection header and position-buffer capacity, excluding
35    /// the retained catalog, allocator metadata and Arc reference counters.
36    /// Clones share this allocation; a family ledger counts it once per owner.
37    #[must_use]
38    pub fn selection_bytes(&self) -> u128 {
39        size_of::<Selected>() as u128
40            + self.0.positions.capacity() as u128 * size_of::<usize>() as u128
41    }
42
43    /// Accept an already ordered unique catalog as one interpretation.
44    /// Every adjacent pair is checked in semantic atom order; no sorting or
45    /// payload import is repeated. The selection allowance covers this model's
46    /// header and dense position buffer, separately from its retained catalog.
47    /// Callback admission precedes reservations, occurrence resolution,
48    /// comparisons and publication.
49    ///
50    /// # Errors
51    /// Refuses unordered/duplicate atoms, selection capacity or callback work.
52    /// No interpretation is returned after a partial check.
53    ///
54    /// # Panics
55    /// Panics if the immutable catalog iterator violates its exact length.
56    pub fn from_ordered_catalog_with<E>(
57        catalog: AtomCatalog,
58        max_selection_bytes: usize,
59        before: impl FnMut() -> Result<(), E>,
60    ) -> Result<Self, ModelFailure<E>> {
61        Self::publish_ordered_catalog_with(catalog, max_selection_bytes, before)
62            .map_err(|failure| failure.into_parts().0)
63    }
64
65    /// Publish an already ordered unique catalog, retaining an actual selection
66    /// allocation receipt when the attempt is refused. The checks, allowance
67    /// and callback order are those of [`Self::from_ordered_catalog_with`].
68    ///
69    /// The successful attempt's peak is exactly [`Self::selection_bytes`]: one
70    /// position buffer is moved into the final selection without copying. On
71    /// refusal, the receipt counts only the buffer capacity actually allocated;
72    /// the final selection header has not been allocated. Rejected reservation
73    /// proposals are not observations. The retained catalog, caller frames, Arc
74    /// counters and allocator bookkeeping are excluded; this is not process RSS.
75    ///
76    /// Vector reservation is fallible. The final Arc envelope allocation uses
77    /// stable Rust's infallible boundary and follows the last callback permit.
78    ///
79    /// # Errors
80    /// Returns the original typed refusal and the actual named allocation peak.
81    /// No partial interpretation is returned; earlier catalog clones stay valid.
82    ///
83    /// # Panics
84    /// Panics if the immutable catalog iterator violates its exact length.
85    /// A callback panic returns no receipt; earlier catalog clones stay valid.
86    pub fn publish_ordered_catalog_with<E>(
87        catalog: AtomCatalog,
88        max_selection_bytes: usize,
89        before: impl FnMut() -> Result<(), E>,
90    ) -> Result<Self, ModelPublicationFailure<E>> {
91        let mut positions = Vec::new();
92        prepare_ordered_selection(&catalog, &mut positions, max_selection_bytes, before)?;
93        Ok(Self(Arc::new(Selected { catalog, positions })))
94    }
95
96    /// Import supplied atoms and coalesce them into canonical set order.
97    /// The iterator must terminate. Construction consumes the descriptions,
98    /// sorts and deduplicates them, and imports their typed payload into one
99    /// canonical catalog. It performs `O(n log n)` typed atom comparisons;
100    /// import work includes admitted term structure and exact interning probes.
101    /// Input and selection vector growth is fallible. Arc envelope allocation
102    /// remains infallible on stable Rust. Input addresses are not retained.
103    ///
104    /// # Errors
105    /// Returns a vector reservation or canonical catalog admission refusal.
106    /// No partial interpretation is returned.
107    pub fn new(atoms: impl IntoIterator<Item = Atom>) -> Result<Self, ModelError> {
108        let mut admitted = Vec::new();
109        for atom in atoms {
110            admitted
111                .try_reserve(1)
112                .map_err(|_| ModelError::Allocation)?;
113            admitted.push(atom);
114        }
115        admitted.sort_unstable();
116        admitted.dedup();
117        Self::from_ordered(admitted)
118    }
119
120    /// Import atoms already in canonical order without sorting or deduplication.
121    /// Canonical payload is admitted once; only occurrence and selection indices
122    /// are retained beside it. Input addresses and capacities are not adopted.
123    ///
124    /// The atoms must be strictly increasing in canonical order; a debug build
125    /// checks this, a release build trusts the producer, and a violated
126    /// precondition would make membership queries wrong, not merely slow.
127    ///
128    /// # Errors
129    /// Returns a selection reservation or canonical catalog admission refusal.
130    /// No partial interpretation is returned.
131    pub fn from_ordered(atoms: Vec<Atom>) -> Result<Self, ModelError> {
132        debug_assert!(
133            atoms.windows(2).all(|pair| pair[0] < pair[1]),
134            "atoms imported as a model are strictly increasing"
135        );
136        let mut positions = Vec::new();
137        positions
138            .try_reserve_exact(atoms.len())
139            .map_err(|_| ModelError::Allocation)?;
140        positions.extend(0..atoms.len());
141        Ok(Self(Arc::new(Selected {
142            catalog: AtomCatalog::new(atoms).map_err(ModelError::Catalog)?,
143            positions,
144        })))
145    }
146
147    /// Select original catalog positions as a canonical interpretation.
148    ///
149    /// Every index is checked before access. Duplicate indices and equal logical
150    /// atoms coalesce. For `m` input positions, construction uses `O(m log m)`
151    /// typed atom comparisons and `O(m)` index storage; no atom is cloned.
152    /// Comparisons include immutable segment resolution and term traversal. The
153    /// iterator must terminate. Index-vector growth is fallible, while the final
154    /// Arc envelope allocation is infallible. The whole catalog remains live.
155    ///
156    /// # Errors
157    /// Returns the first outside-catalog position or an index-vector reservation
158    /// failure. No partial model is returned and the catalog remains unchanged.
159    pub fn from_positions(
160        catalog: &AtomCatalog,
161        positions: impl IntoIterator<Item = usize>,
162    ) -> Result<Self, ModelError> {
163        let mut selected = Vec::new();
164        for position in positions {
165            if position >= catalog.atoms().len() {
166                return Err(ModelError::Position {
167                    position,
168                    atoms: catalog.atoms().len(),
169                });
170            }
171            selected
172                .try_reserve(1)
173                .map_err(|_| ModelError::Allocation)?;
174            selected.push(position);
175        }
176        let atoms = catalog.atoms();
177        selected.sort_unstable_by(|&left, &right| atoms.compare_positions(left, right));
178        selected.dedup_by(|left, right| atoms.compare_positions(*left, *right).is_eq());
179        Ok(Self(Arc::new(Selected {
180            catalog: catalog.clone(),
181            positions: selected,
182        })))
183    }

225    /// Retained selection-vector capacity in cells, excluding shared catalog
226    /// storage and the Arc envelope. Model clones share this same vector.
227    #[must_use]
228    pub fn selection_capacity(&self) -> usize {
229        self.0.positions.capacity()
230    }

232    /// Portable encoded payload of every catalog occurrence, including repeats
233    /// and unselected atoms, plus a u64 selection length and u64 positions.
234    /// Constant time after catalog construction. Returns `None` on size overflow.
235    ///
236    /// Counting this per retained model conservatively recounts shared catalogs.
237    /// [`crate::retention::ModelRetention`] accounts shared catalogs once instead.
238    /// This is a portable admission measure, not allocated bytes or RSS: spare
239    /// vector capacity, Arc envelopes and allocator bookkeeping are excluded,
240    /// as are canonical identities retained outside the occurrence map.
241    #[must_use]
242    pub fn retained_payload_bytes(&self) -> Option<usize> {
243        self.catalog()
244            .retained_payload_bytes()?
245            .checked_add(self.selection_payload_bytes()?)
246    }
247
248    /// Canonical selected-position record: a u64 length and u64 per position.
249    /// Constant time; excludes the catalog, spare capacity and Arc overhead.
250    /// Returns `None` on arithmetic overflow. Retention accounts this per model
251    /// entry, even when a cloned model physically shares its selection vector.
252    #[must_use]
253    pub fn selection_payload_bytes(&self) -> Option<usize> {
254        LENGTH_BYTES.checked_add(self.0.positions.len().checked_mul(LENGTH_BYTES)?)
255    }

298/// Read-only canonical atom-set view. All returned references retain the model's
299/// lifetime. Order is signed predicate and typed value storage order, not ASP
300/// arithmetic term order. No allocation or atom copying occurs during iteration.
301#[derive(Clone, Copy)]
302pub struct ModelAtoms<'a> {
303    atoms: catalog::Atoms<'a>,
304    positions: &'a [usize],
305}
306
307impl<'a> ModelAtoms<'a> {
308    /// Borrow the checked lookup over this selection without allocating.
309    /// Only selected occurrences participate; returned row positions address
310    /// the original catalog. The existing semantic order serves both identity
311    /// and predicate searches.
312    #[must_use]
313    pub fn lookup(self) -> crate::AtomLookup<'a, 'a> {
314        crate::AtomLookup {
315            atoms: crate::atom_lookup::Source::Canonical(self.atoms),
316            keys: self.positions,
317            rows: self.positions,
318        }
319    }
320
321    /// Number of distinct true atoms. Constant time.
322    #[must_use]
323    pub const fn len(self) -> usize {
324        self.positions.len()
325    }
326
327    /// Whether the true set is empty. Constant time.
328    #[must_use]
329    pub const fn is_empty(self) -> bool {
330        self.positions.is_empty()
331    }
332
333    /// The atom at a position of the model's canonical order. The view checks
334    /// prefix membership only; reading its payload resolves the row through the
335    /// retained immutable segment ranges.
336    #[must_use]
337    pub fn at(self, position: usize) -> Option<AtomRef<'a>> {
338        self.positions
339            .get(position)
340            .and_then(|&index| self.atoms.at(index))
341    }
342
343    /// Canonical double-ended iteration with an exact remaining length.
344    #[must_use]
345    pub fn iter(self) -> ModelIter<'a> {
346        ModelIter {
347            atoms: self.atoms,
348            positions: self.positions.iter(),
349        }
350    }
351
352    /// Least true atom in canonical storage order, borrowed without resolving it.
353    #[must_use]
354    pub fn first(self) -> Option<AtomRef<'a>> {
355        self.positions
356            .first()
357            .and_then(|&position| self.atoms.at(position))
358    }
359
360    /// Greatest true atom in canonical storage order, borrowed without resolving it.
361    #[must_use]
362    pub fn last(self) -> Option<AtomRef<'a>> {
363        self.positions
364            .last()
365            .and_then(|&position| self.atoms.at(position))
366    }
367
368    /// The true atoms of one signed predicate, a contiguous sub-view found by
369    /// two logarithmic searches; empty when the predicate has none.
370    #[must_use]
371    pub fn of_predicate<'query>(
372        self,
373        predicate: impl Into<PredicateRef<'query>>,
374    ) -> ModelAtoms<'a> {
375        let predicate = predicate.into();
376        let start = self.positions.partition_point(|&position| {
377            self.selected(position).predicate().cmp(&predicate).is_lt()
378        });
379        let end = start
380            + self.positions[start..].partition_point(|&position| {
381                self.selected(position).predicate().cmp(&predicate).is_eq()
382            });
383        ModelAtoms {
384            atoms: self.atoms,
385            positions: &self.positions[start..end],
386        }
387    }
388
389    /// Borrow the original matching atom using logarithmic typed comparisons.
390    #[must_use]
391    pub fn get<'query>(self, atom: impl Into<AtomRef<'query>>) -> Option<AtomRef<'a>> {
392        let atom = atom.into();
393        self.positions
394            .binary_search_by(|&position| self.selected(position).cmp(&atom))
395            .ok()
396            .map(|index| self.selected(self.positions[index]))
397    }
398
399    fn selected(self, position: usize) -> AtomRef<'a> {
400        self.atoms
401            .at(position)
402            .expect("model occurrence is checked")
403    }
404
405    /// Exact membership using logarithmic typed comparisons.
406    #[must_use]
407    pub fn contains<'query>(self, atom: impl Into<AtomRef<'query>>) -> bool {
408        self.get(atom).is_some()
409    }
410}

472/// Interpretation construction failed, without a partial true set.
473#[derive(Clone, Debug, PartialEq, Eq)]
474pub enum ModelError {
475    /// A purportedly ordered unique catalog repeats or reverses an atom.
476    Order {
477        /// The second position of the first non-increasing pair.
478        position: usize,
479    },
480    /// Named model-construction metadata exceed the operation's allowance.
481    /// Ordered-catalog publication counts selection metadata; prepared-order
482    /// operations additionally count their retained order and live scratch.
483    Bytes {
484        /// Requested or actual named capacity.
485        required: u128,
486        /// Inclusive metadata allowance for the called operation.
487        limit: usize,
488    },
489    /// Canonical payload admission or representation failed.
490    Catalog(catalog::Error),
491    /// A supplied index has no atom in the retained catalog.
492    Position {
493        /// Refused dense position.
494        position: usize,
495        /// Number of atoms in this catalog.
496        atoms: usize,
497    },
498    /// Input, selection or prepared-order metadata reservation failed.
499    Allocation,
500}
501impl fmt::Display for ModelError {
502    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
503        match self {
504            Self::Order { position } => write!(
505                f,
506                "model catalog is not strictly ordered at position {position}"
507            ),
508            Self::Bytes { required, limit } => write!(
509                f,
510                "model construction requires {required} bytes, allowance is {limit}"
511            ),
512            Self::Catalog(error) => error.fmt(f),
513            Self::Position { position, atoms } => {
514                write!(
515                    f,
516                    "model position {position} is outside a catalog of {atoms} atoms"
517                )
518            }
519            Self::Allocation => f.write_str("model construction storage could not be reserved"),
520        }
521    }
522}
523impl std::error::Error for ModelError {
524    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
525        match self {
526            Self::Catalog(error) => Some(error),
527            Self::Position { .. } | Self::Order { .. } | Self::Bytes { .. } | Self::Allocation => {
528                None
529            }
530        }
531    }
532}

534/// Prepare a fresh, unallocated position buffer through the final publication
535/// permit. The caller retains it on every return and moves it into Selected only
536/// on success. One reservation is the only capacity change, so its retained
537/// capacity is also the refused attempt's peak; no Selected envelope exists yet.
538fn prepare_ordered_selection<E>(
539    catalog: &AtomCatalog,
540    positions: &mut Vec<usize>,
541    max_selection_bytes: usize,
542    mut before: impl FnMut() -> Result<(), E>,
543) -> Result<(), ModelPublicationFailure<E>> {
544    let result = (|| {
545        let mut checked = || before().map_err(ModelFailure::Stopped);
546        let count = catalog.atoms().len();
547        selection_allowance(count, max_selection_bytes).map_err(ModelFailure::Model)?;
548        checked()?;
549        positions
550            .try_reserve_exact(count)
551            .map_err(|_| ModelFailure::Model(ModelError::Allocation))?;
552        selection_allowance(positions.capacity(), max_selection_bytes)
553            .map_err(ModelFailure::Model)?;
554        let mut previous: Option<AtomRef<'_>> = None;
555        let mut atoms = catalog.atoms().iter();
556        for index in 0..count {
557            checked()?;
558            let atom = atoms
559                .next()
560                .expect("immutable catalog iterator has exact length");
561            if let Some(left) = previous
562                && !left.compare_ref_with(atom, &mut checked)?.is_lt()
563            {
564                return Err(ModelFailure::Model(ModelError::Order { position: index }));
565            }
566            previous = Some(atom);
567            positions.push(index);
568        }
569        checked()
570    })();
571    // Observe the same still-live buffer after every normal refusal, including
572    // reservation or actual-capacity refusal. Requested bytes are not a receipt.
573    result.map_err(|failure| ModelPublicationFailure {
574        failure,
575        peak_bytes: positions.capacity() as u128 * size_of::<usize>() as u128,
576    })
577}
578
579fn selection_allowance(capacity: usize, limit: usize) -> Result<(), ModelError> {
580    let required = size_of::<Selected>() as u128 + capacity as u128 * size_of::<usize>() as u128;
581    if required > limit as u128 {
582        Err(ModelError::Bytes { required, limit })
583    } else {
584        Ok(())
585    }
586}

648/// Model construction refused after observing its named metadata allocations.
649/// The called operation defines its receipt scope. Existing catalog clones and
650/// previously published models remain valid; no partial interpretation escapes.
651#[derive(Clone, Debug, PartialEq, Eq)]
652pub struct ModelPublicationFailure<E> {
653    failure: ModelFailure<E>,
654    peak_bytes: u128,
655}
656
657impl<E> ModelPublicationFailure<E> {
658    /// Original ordering, construction-storage, allocation or caller refusal.
659    #[must_use]
660    pub const fn failure(&self) -> &ModelFailure<E> {
661        &self.failure
662    }
663
664    /// Actual named metadata peak under the called operation's scope.
665    /// `publish_ordered_catalog_with` counts its allocated position capacity;
666    /// `ModelOrder` also counts its prepared order and temporary vector headers.
667    /// Unallocated final headers, catalog payload, unnamed caller frames, Arc
668    /// counters, allocator bookkeeping and rejected proposals are excluded.
669    #[must_use]
670    pub const fn peak_bytes(&self) -> u128 {
671        self.peak_bytes
672    }
673
674    /// Recover the original typed refusal together with its attempt receipt.
675    #[must_use]
676    pub fn into_parts(self) -> (ModelFailure<E>, u128) {
677        (self.failure, self.peak_bytes)
678    }
679}
