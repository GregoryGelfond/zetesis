10const VECTOR_BYTES: u128 = size_of::<Vec<usize>>() as u128;
11const MARK_BITS: usize = usize::BITS as usize;
12
13/// Immutable semantic equivalence ranks for one exact borrowed occurrence map.
14///
15/// Equal logical atoms have equal ranks; rank ordering agrees with the existing
16/// typed atom storage ordering. Original positions and canonical IDs are never
17/// reinterpreted as semantic order. This value owns only integer metadata and
18/// borrows its catalog, not a writer or another atom authority. A selected [`Model`]
19/// shares that catalog and can outlive this preparation.
20///
21/// Preparation uses O(n log n) checked semantic comparisons and retains two
22/// n-cell integer buffers: each occurrence's rank, and the occurrences in
23/// semantic order (equal ranks by ascending position). Selection marks the m
24/// supplied positions in an n-bit mask, then either walks the semantic order
25/// once (O(n) integer steps) or sorts the marked positions (O(m log m) integer
26/// comparisons), whichever the counts make cheaper; both publish the same model
27/// and neither compares atom payloads again. Structural comparison includes
28/// segment resolution and inspected term/text contents.
29#[derive(Debug)]
30pub struct ModelOrder<'catalog> {
31    catalog: &'catalog AtomCatalog,
32    ranks: Vec<usize>,
33    semantic: Vec<usize>,
34    preparation_peak_bytes: u128,
35}
36
37impl<'catalog> ModelOrder<'catalog> {
38    /// Prepare semantic ranks without changing or copying the catalog payload.
39    /// Duplicate logical occurrences are admitted and share an equivalence rank.
40    ///
41    /// `max_bytes` covers this order's header and actual integer capacities plus
42    /// live temporary vector headers and old/new reservation overlap. Catalog
43    /// payload, caller inputs, scalar/callback/receipt frames, allocator metadata
44    /// and Arc counters are excluded. Actual peaks survive refusal; rejected
45    /// proposals are not observed allocations. Initialized named headers count
46    /// even on initial refusal; no integer capacity exists then. Zero is a real
47    /// allowance.
48    ///
49    /// `before` precedes reservations, occurrence resolution, checked semantic
50    /// comparisons, integer writes and publication. It supplies cumulative work
51    /// and cancellation policy. The first callback precedes any allocation.
52    ///
53    /// # Errors
54    /// Returns a typed metadata-capacity, allocation or caller refusal and the
55    /// actual named peak. No partial order is returned; the catalog is unchanged.
56    ///
57    /// # Panics
58    /// Panics if an immutable catalog violates its admitted occurrence extent.
59    pub fn prepare_with<E>(
60        catalog: &'catalog AtomCatalog,
61        max_bytes: usize,
62        mut before: impl FnMut() -> Result<(), E>,
63    ) -> Result<Self, ModelPublicationFailure<E>> {
64        let mut order = Self {
65            catalog,
66            ranks: Vec::new(),
67            semantic: Vec::new(),
68            preparation_peak_bytes: 0,
69        };
70        let mut scratch = Vec::new();
71        let base = size_of::<Self>() as u128 + VECTOR_BYTES;
72        let mut account = Account::new(max_bytes, base);
73        let result = (|| {
74            let mut checked = || before().map_err(ModelFailure::Stopped);
75            checked()?;
76            account.admit(base)?;
77            let count = catalog.atoms().len();
78            reserve(&mut order.ranks, count, base, &mut account, &mut checked)?;
79            reserve(
80                &mut scratch,
81                count,
82                base + cells(order.ranks.capacity()),
83                &mut account,
84                &mut checked,
85            )?;
86            for position in 0..count {
87                checked()?;
88                order.ranks.push(position);
89                checked()?;
90                scratch.push(0);
91            }
92            let atoms = catalog.atoms();
93            crate::checked_sort::sort(
94                &mut order.ranks,
95                &mut scratch,
96                |values, left, right, permit| {
97                    permit()?;
98                    let left = atoms
99                        .at(values[left])
100                        .expect("prepared occurrence is in the catalog");
101                    permit()?;
102                    let right = atoms
103                        .at(values[right])
104                        .expect("prepared occurrence is in the catalog");
105                    left.compare_ref_with(right, permit)
106                },
107                &mut checked,
108            )?;
109            let mut previous = None;
110            let mut rank = 0;
111            for &position in &order.ranks {
112                checked()?;
113                let atom = atoms
114                    .at(position)
115                    .expect("sorted occurrence remains in the catalog");
116                if let Some(previous) = previous {
117                    checked()?;
118                    let previous = atoms
119                        .at(previous)
120                        .expect("previous occurrence remains in the catalog");
121                    if !atom.equals_ref_with(previous, &mut checked)? {
122                        // There are at most count groups, so the last rank is < count.
123                        rank += 1;
124                    }
125                }
126                checked()?;
127                scratch[position] = rank;
128                previous = Some(position);
129            }
130            checked()?;
131            // The sorted positions are the semantic order; the scratch holds ranks.
132            std::mem::swap(&mut order.ranks, &mut scratch);
133            order.semantic = scratch;
134            Ok(())
135        })();
136        result.map_err(|failure| ModelPublicationFailure {
137            failure,
138            peak_bytes: account.peak,
139        })?;
140        order.preparation_peak_bytes = account.peak;
141        Ok(order)
142    }

149    }

157    /// Publish supplied positions using this exact catalog's prepared order.
158    ///
159    /// Every position is checked. Repeats and equal logical atoms coalesce,
160    /// retaining the least supplied original position of each equivalence class.
161    /// An unselected occurrence is never introduced. The existing [`Model`] shape is
162    /// published directly; its catalog and selection remain independent of this
163    /// order's lifetime. No semantic comparison or payload import is repeated.
164    ///
165    /// Supplied positions are marked in an n-bit mask. When the m marked
166    /// positions are dense enough that m·⌈log₂(m + 1)⌉ reaches n, the prepared
167    /// semantic order is walked once, keeping the first marked occurrence of
168    /// each rank (the least marked position, since equal ranks ascend by
169    /// position), and the walk ends at the last marked occurrence. Otherwise the
170    /// marked positions are sorted by rank and position and each rank's first is
171    /// kept. The choice depends only on m and n and is made before either runs;
172    /// both publish the same model.
173    ///
174    /// The allowance and success/failure peak include the retained order plus
175    /// this attempt's live mask/position/scratch headers, actual capacities,
176    /// growth overlap and final selection header when allocated. Previously
177    /// published models, catalog payload, caller inputs, callback/receipt frames,
178    /// allocator metadata and Arc counters are excluded. The mask and scratch are
179    /// not retained.
180    ///
181    /// Permits precede iterator pulls, mask and rank access, sort comparisons,
182    /// writes, reservations and final publication. The walk's loop control and
183    /// its rank comparison follow the permit taken for the same occurrence. A finite caller work
184    /// budget bounds successful iterator pulls; it cannot preempt a blocking
185    /// `next`. The first callback precedes iterator conversion and allocation.
186    /// Stable Rust's final Arc allocation remains infallible after the last
187    /// permit.
188    ///
189    /// # Errors
190    /// Returns the first invalid position, metadata-capacity, allocation or
191    /// caller refusal with the actual named peak. No partial model escapes;
192    /// the order, catalog and earlier models remain valid for subsequent calls.
193    pub fn select_with<E>(
194        &self,
195        positions: impl IntoIterator<Item = usize>,
196        max_bytes: usize,
197        mut before: impl FnMut() -> Result<(), E>,
198    ) -> Result<ModelPublication, ModelPublicationFailure<E>> {
199        let mut marks = Vec::new();
200        let mut selected = Vec::new();
201        let mut scratch = Vec::new();
202        let base = self.retained_bytes() + 3 * VECTOR_BYTES;
203        let mut account = Account::new(max_bytes, base);
204        let result = (|| {
205            let mut checked = || before().map_err(ModelFailure::Stopped);
206            checked()?;
207            account.admit(base)?;
208            let count = self.ranks.len();
209            let words = count.div_ceil(MARK_BITS);
210            reserve(&mut marks, words, base, &mut account, &mut checked)?;
211            for _ in 0..words {
212                checked()?;
213                marks.push(0);
214            }
215            let mut marked = 0;
216            let mut positions = positions.into_iter();
217            loop {
218                checked()?;
219                let Some(position) = positions.next() else {
220                    break;
221                };
222                checked()?;
223                if position >= count {
224                    return Err(ModelFailure::Model(ModelError::Position {
225                        position,
226                        atoms: count,
227                    }));
228                }
229                let bit = 1 << (position % MARK_BITS);
230                let word = &mut marks[position / MARK_BITS];
231                if *word & bit == 0 {
232                    *word |= bit;
233                    marked += 1;
234                }
235            }
236            let other = base + cells(marks.capacity());
237            reserve(&mut selected, marked, other, &mut account, &mut checked)?;
238            if walks(marked, count) {
239                self.walk(&marks, marked, &mut selected, &mut checked)?;
240            } else {
241                for (index, &word) in marks.iter().enumerate() {
242                    checked()?;
243                    let mut bits = word;
244                    while bits != 0 {
245                        checked()?;
246                        selected.push(index * MARK_BITS + bits.trailing_zeros() as usize);
247                        bits &= bits - 1;
248                    }
249                }
250                reserve(
251                    &mut scratch,
252                    selected.len(),
253                    other + cells(selected.capacity()),
254                    &mut account,
255                    &mut checked,
256                )?;
257                for _ in 0..selected.len() {
258                    checked()?;
259                    scratch.push(0);
260                }
261                self.sort_marked(&mut selected, &mut scratch, &mut checked)?;
262            }
263            drop(scratch);
264            drop(marks);
265            let published =
266                self.retained_bytes() + size_of::<Selected>() as u128 + cells(selected.capacity());
267            account.admit(published)?;
268            checked()?;
269            let model = Model(Arc::new(Selected {
270                catalog: self.catalog.clone(),
271                positions: selected,
272            }));
273            account.record(published);
274            Ok(ModelPublication {
275                model,
276                peak_bytes: account.peak,
277            })
278        })();
279        result.map_err(|failure| ModelPublicationFailure {
280            failure,
281            peak_bytes: account.peak,
282        })
283    }
284
285    /// Keep the first marked occurrence of each rank along the semantic order,
286    /// stopping after the last marked occurrence. `selected` has room for
287    /// `marked` cells.
288    ///
289    /// The first marked occurrence of a rank is its least marked position, as
290    /// the sorting route keeps: preparation's merge sort is stable over
291    /// ascending positions, and occurrences of one rank compare equal, so equal
292    /// ranks appear in the semantic order by ascending position.
293    fn walk<E>(
294        &self,
295        marks: &[usize],
296        marked: usize,
297        selected: &mut Vec<usize>,
298        checked: &mut impl FnMut() -> Result<(), ModelFailure<E>>,
299    ) -> Result<(), ModelFailure<E>> {
300        let mut previous = None;
301        let mut seen = 0;
302        for &position in &self.semantic {
303            if seen == marked {
304                break;
305            }
306            checked()?;
307            if marks[position / MARK_BITS] & (1 << (position % MARK_BITS)) == 0 {
308                continue;
309            }
310            seen += 1;
311            checked()?;
312            let rank = self.ranks[position];
313            if previous != Some(rank) {
314                checked()?;
315                selected.push(position);
316                previous = Some(rank);
317            }
318        }
319        Ok(())
320    }
321
322    /// Sort marked positions by rank, then position, and keep each rank's first.
323    fn sort_marked<E>(
324        &self,
325        selected: &mut Vec<usize>,
326        scratch: &mut Vec<usize>,
327        checked: &mut impl FnMut() -> Result<(), ModelFailure<E>>,
328    ) -> Result<(), ModelFailure<E>> {
329        crate::checked_sort::sort(
330            selected,
331            scratch,
332            |values, left, right, permit| {
333                permit()?;
334                let left = values[left];
335                permit()?;
336                let right = values[right];
337                permit()?;
338                let left_rank = self.ranks[left];
339                permit()?;
340                let right_rank = self.ranks[right];
341                permit()?;
342                let order = left_rank.cmp(&right_rank);
343                if order.is_eq() {
344                    permit()?;
345                    Ok(left.cmp(&right))
346                } else {
347                    Ok(order)
348                }
349            },
350            checked,
351        )?;
352        let mut previous = None;
353        let mut kept = 0;
354        for index in 0..selected.len() {
355            checked()?;
356            let position = selected[index];
357            checked()?;
358            let rank = self.ranks[position];
359            checked()?;
360            if previous != Some(rank) {
361                checked()?;
362                selected[kept] = position;
363                kept += 1;
364                previous = Some(rank);
365            }
366        }
367        checked()?;
368        selected.truncate(kept);
369        Ok(())
370    }
371}

373/// Whether walking the n-occurrence semantic order is expected to cost no
374/// more than sorting `marked` positions: m·⌈log₂(m + 1)⌉ ≥ n.
375const fn walks(marked: usize, count: usize) -> bool {
376    let depth = (usize::BITS - marked.leading_zeros()) as usize;
377    marked.saturating_mul(depth) >= count
378}

380fn cells(capacity: usize) -> u128 {
381    capacity as u128 * size_of::<usize>() as u128
382}

384struct Account {
385    limit: usize,
386    peak: u128,
387}
388
389impl Account {
390    const fn new(limit: usize, initialized_headers: u128) -> Self {
391        Self {
392            limit,
393            peak: initialized_headers,
394        }
395    }
396
397    fn admit<E>(&self, required: u128) -> Result<(), ModelFailure<E>> {
398        if required > self.limit as u128 {
399            Err(ModelFailure::Model(ModelError::Bytes {
400                required,
401                limit: self.limit,
402            }))
403        } else {
404            Ok(())
405        }
406    }
407
408    fn record(&mut self, observed: u128) {
409        self.peak = self.peak.max(observed);
410    }
411
412    fn observe<E>(&mut self, observed: u128) -> Result<(), ModelFailure<E>> {
413        self.record(observed);
414        self.admit(observed)
415    }
416}

418/// `other` includes named owner/vector headers and all other live capacities.
419/// Allocate a separate replacement to make old/new overlap explicit. Refusal
420/// during copying preserves the original buffer; actual replacement capacity is
421/// recorded before any later callback can refuse. All byte arithmetic uses u128
422/// over a fixed number of usize capacities, so it cannot overflow.
423fn reserve<E>(
424    buffer: &mut Vec<usize>,
425    needed: usize,
426    other: u128,
427    account: &mut Account,
428    before: &mut impl FnMut() -> Result<(), ModelFailure<E>>,
429) -> Result<(), ModelFailure<E>> {
430    if needed <= buffer.capacity() {
431        return Ok(());
432    }
433    let requested = needed.max(buffer.capacity().saturating_mul(2));
434    let overlap = other + cells(buffer.capacity()) + VECTOR_BYTES;
435    account.admit(overlap + cells(requested))?;
436    before()?;
437    let mut replacement = Vec::new();
438    account.observe(overlap)?;
439    let result = replacement.try_reserve_exact(requested);
440    account.observe(overlap + cells(replacement.capacity()))?;
441    result.map_err(|_| ModelFailure::Model(ModelError::Allocation))?;
442    for &position in buffer.iter() {
443        before()?;
444        replacement.push(position);
445    }
446    before()?;
447    *buffer = replacement;
448    Ok(())
