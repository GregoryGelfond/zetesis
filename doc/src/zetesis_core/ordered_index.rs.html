1//! An append-ID AVL tree. Paths own only tentative metadata; publication follows
2//! every fallible comparison, reservation and work admission. Balance is right
3//! height minus left height. Inserting one leaf causes at most one rotation;
4//! only that search path and its parent link change. No payload is stored here.
5
6use std::{cmp::Ordering, num::NonZeroUsize};

10pub(crate) type Link = Option<NonZeroUsize>;

12pub(crate) fn encoded(position: usize) -> NonZeroUsize {
13    NonZeroUsize::new(position.checked_add(1).expect("admitted position")).expect("positive ID")
14}

16pub(crate) fn position(value: NonZeroUsize) -> usize {
17    value.get() - 1

42/// Metadata already established by one successful insertion plan. The changed
43/// suffix includes the new leaf; earlier path cells remain published unchanged.
44#[derive(Clone, Copy)]
45pub(crate) struct Planned {
46    pub root: Link,
47    pub changed_from: usize,
48    pub rotation: Option<Rotation>,
49}
50
51#[derive(Clone, Copy)]
52pub(crate) enum Rotation {
53    Single(usize),
54    Double,
55}

57impl Index {
58    /// Plan against an unpublished root whose nodes were supplied in `path`.
59    /// This supports several leaves in one bounded metadata transaction.
60    pub fn plan_from<E>(
61        &mut self,
62        root: Link,
63        id: usize,
64        before: &mut impl FnMut() -> Result<(), E>,
65    ) -> Result<Link, E> {
66        self.plan_changes_from(root, id, before)
67            .map(|plan| plan.root)
68    }
69
70    /// Compute a new root and its actual changed suffix without changing any
71    /// published link. The path ends at the absent child's new leaf. Propagation
72    /// stops once subtree height is unchanged. A consumer reusing an all-right
73    /// path must establish that property separately before invoking the planner.
74    pub fn plan_changes_from<E>(
75        &mut self,
76        root: Link,
77        id: usize,
78        before: &mut impl FnMut() -> Result<(), E>,
79    ) -> Result<Planned, E> {
80        let mut child = Some(encoded(id));
81        let mut changed_from = self.path.len() - 1;
82        for level in (0..self.path.len() - 1).rev() {
83            before()?;
84            let step = &mut self.path[level];
85            let side = usize::from(step.right);
86            step.node.children[side] = child;
87            step.node.balance += if step.right { 1 } else { -1 };
88            step.changed = true;
89            changed_from = level;
90            child = Some(encoded(step.id));
91            if step.node.balance == 0 {
92                return Ok(Planned {
93                    root,
94                    changed_from,
95                    rotation: None,
96                });
97            }
98            if step.node.balance.abs() == 2 {
99                let (rotated, rotation) = self.rotate(level, before)?;
100                child = rotated;
101                if level == 0 {
102                    return Ok(Planned {
103                        root: child,
104                        changed_from,
105                        rotation: Some(rotation),
106                    });
107                }
108                before()?;
109                let parent = &mut self.path[level - 1];
110                parent.node.children[usize::from(parent.right)] = child;
111                parent.changed = true;
112                return Ok(Planned {
113                    root,
114                    changed_from: level - 1,
115                    rotation: Some(rotation),
116                });
117            }
118        }
119        Ok(Planned {
120            root: child,
121            changed_from,
122            rotation: None,
123        })
124    }
125
126    fn rotate<E>(
127        &mut self,
128        level: usize,
129        before: &mut impl FnMut() -> Result<(), E>,
130    ) -> Result<(Link, Rotation), E> {
131        // Before the first imbalance, the heavy child and (for a double
132        // rotation) grandchild are the next two nodes in the insertion path.
133        before()?;
134        let mut root = self.path[level];
135        let mut child = self.path[level + 1];
136        let side = usize::from(root.right);
137        let opposite = 1 - side;
138        let sign = if root.right { 1 } else { -1 };
139        if child.node.balance == sign {
140            before()?;
141            root.node.children[side] = child.node.children[opposite];
142            child.node.children[opposite] = Some(encoded(root.id));
143            root.node.balance = 0;
144            child.node.balance = 0;
145            self.path[level] = root;
146            self.path[level + 1] = child;
147            return Ok((Some(encoded(child.id)), Rotation::Single(level)));
148        }
149        before()?;
150        let mut pivot = self.path[level + 2];
151        root.node.children[side] = pivot.node.children[opposite];
152        child.node.children[opposite] = pivot.node.children[side];
153        pivot.node.children[opposite] = Some(encoded(root.id));
154        pivot.node.children[side] = Some(encoded(child.id));
155        root.node.balance = if pivot.node.balance == sign { -sign } else { 0 };
156        child.node.balance = if pivot.node.balance == -sign { sign } else { 0 };
157        pivot.node.balance = 0;
158        pivot.changed = true;
159        self.path[level] = root;
160        self.path[level + 1] = child;
161        self.path[level + 2] = pivot;
162        Ok((Some(encoded(pivot.id)), Rotation::Double))
163    }
164
165    /// Publish a suffix established by `plan_changes_from`; nodes before this
166    /// boundary were not mutated. The owner pre-admits every write and publishes
167    /// its root in the same indivisible transaction, without callbacks.
168    pub fn publish_nodes_from(&mut self, changed_from: usize) {
169        let leaf = self.path.last().expect("planned leaf");
170        self.nodes.push(leaf.node);
171        for step in &self.path[changed_from..self.path.len() - 1] {
172            if step.changed {
173                self.nodes[step.id] = step.node;
174            }
175        }
176    }
177}

179/// Where a query stands against the last node in order, the maximum.
180pub(crate) enum Last {
181    /// The query equals the maximum, at this position.
182    Found(usize),
183    /// The query follows every node; its place is after the maximum.
184    Beyond,
185    /// The tree is empty, or the query precedes the maximum: the last node
186    /// cannot place it, and the full search decides.
187    Search,
188}
189
190/// Classify a query against the owner's last node before searching. A query
191/// beyond that node is absent and its insertion path is the right spine. The
192/// owner may prepare that path directly without first recording its directions.
193/// This comparison reads no links and changes neither published nodes nor scratch.
194pub(crate) fn compare_last<C, E>(
195    root: Link,
196    last: Option<usize>,
197    context: &mut C,
198    compare: impl FnOnce(usize, &mut C) -> Result<Ordering, E>,
199) -> Result<Last, E> {
200    let (Some(_), Some(last)) = (root, last) else {
201        return Ok(Last::Search);
202    };
203    Ok(match compare(last, context)? {
204        Ordering::Equal => Last::Found(last),
205        Ordering::Less => Last::Search,
206        Ordering::Greater => Last::Beyond,
207    })
208}

210/// Search the owner's ordered identities without owning or copying any payload.
211/// `compare` admits one node visit before comparing the query with the ID's
212/// authoritative value. The callback's ordering must be the tree's ordering.
213/// `descend` records only successful comparisons; an error publishes no result
214/// and changes no node. A miss visits at most the AVL height in nodes.
215pub(crate) fn search<E>(
216    nodes: &[Node],
217    mut cursor: Link,
218    mut compare: impl FnMut(usize) -> Result<Ordering, E>,
219    mut descend: impl FnMut(bool),
220) -> Result<Option<usize>, E> {
221    while let Some(next) = cursor {
222        let id = position(next);
223        let order = compare(id)?;
224        if order.is_eq() {
225            return Ok(Some(id));
226        }
227        let right = order == Ordering::Greater;
228        descend(right);
229        cursor = nodes[id].children[usize::from(right)];
230    }
231    Ok(None)
232}

234/// Fixed traversal state, never another index or an atom owner.
235///
236/// Let `N(h)` be the minimum node count of an AVL of height h, with empty height
237/// zero. Its recurrence gives `N(h) >= 2*N(h-2)+1` and hence
238/// `N(h) >= 2^ceil(h/2)-1`. For a nonempty tree with n nodes and bit width b,
239/// `n < 2^b` implies `h <= 2*b`. Since `b <= usize::BITS`, two target-sized words
240/// hold every descent. The planned insertion leaf is not a descent bit.
241/// The empty tree records no directions. This argument uses mathematical `n+1`;
242/// no potentially overflowing machine addition is needed to compute the bound.
243///
244/// Checked packing protects the representation independently of that AVL
245/// invariant. A failed push from an actual search means the internal AVL height
246/// invariant was broken, not a user resource refusal or missing atom.
247/// This local stack record lives only during entry lookup/path preparation;
248/// its two words and checked length are excluded from named vector capacities.
249#[derive(Default)]
250pub(crate) struct Directions {
251    words: [usize; 2],
252    length: usize,
253}
254
255impl Directions {
256    pub(crate) fn push(&mut self, right: bool) -> Option<()> {
257        let next = self.length.checked_add(1)?;
258        if next > self.words.len() * usize::BITS as usize {
259            return None;
260        }
261        // The newest direction occupies bit zero. Transfer the low word's
262        // oldest bit before shifting; the admitted length ensures that no
263        // recorded bit can leave the high word. Decode positions only on replay.
264        let carry = self.words[0] >> (usize::BITS - 1);
265        self.words[0] = (self.words[0] << 1) | usize::from(right);
266        self.words[1] = (self.words[1] << 1) | carry;
267        self.length = next;
268        Some(())
269    }
270
271    pub(crate) fn len(&self) -> usize {
272        self.length
273    }
274
275    pub(crate) fn get(&self, position: usize) -> Option<bool> {
276        if position >= self.length {
277            return None;
278        }
279        let (word, mask) = Self::position(self.length - 1 - position)?;
280        Some(*self.words.get(word)? & mask != 0)
281    }
282
283    fn position(position: usize) -> Option<(usize, usize)> {
284        let width = usize::BITS as usize;
285        let offset = u32::try_from(position % width).ok()?;
286        Some((position / width, 1_usize.checked_shl(offset)?))
287    }
288}
