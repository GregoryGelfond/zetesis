1//! An append-ID AVL tree. Paths own only tentative metadata; publication follows
2//! every fallible comparison, reservation and work admission. Balance is right
3//! height minus left height. Inserting one leaf causes at most one rotation;
4//! only that search path and its parent link change. No payload is stored here.
5
6use std::{cmp::Ordering, num::NonZeroUsize};
7
8pub(crate) type Link = Option<NonZeroUsize>;
9
10pub(crate) fn encoded(position: usize) -> NonZeroUsize {
11    NonZeroUsize::new(position.checked_add(1).expect("admitted position")).expect("positive ID")
12}
13
14pub(crate) fn position(value: NonZeroUsize) -> usize {
15    value.get() - 1
16}
17
18#[derive(Clone, Copy, Default)]
19pub(crate) struct Node {
20    pub children: [Link; 2],
21    pub balance: i8,
22}
23
24#[derive(Clone, Copy)]
25pub(crate) struct Step {
26    pub id: usize,
27    pub node: Node,
28    pub right: bool,
29    pub changed: bool,
30}
31
32#[derive(Default)]
33pub(crate) struct Index {
34    pub nodes: Vec<Node>,
35    pub root: Link,
36    pub path: Vec<Step>,
37    pub peak: u128,
38}
39
40impl Index {
41    /// Compute a new root and path metadata without changing any published link.
42    /// The recorded search path ends at an absent child. The inserted leaf has
43    /// height one; propagation stops once subtree height is unchanged.
44    pub fn plan<E>(
45        &mut self,
46        id: usize,
47        before: &mut impl FnMut() -> Result<(), E>,
48    ) -> Result<Link, E> {
49        self.plan_from(self.root, id, before)
50    }
51
52    /// Plan against an unpublished root whose nodes were supplied in `path`.
53    /// This supports several leaves in one bounded metadata transaction.
54    pub fn plan_from<E>(
55        &mut self,
56        root: Link,
57        id: usize,
58        before: &mut impl FnMut() -> Result<(), E>,
59    ) -> Result<Link, E> {
60        let mut child = Some(encoded(id));
61        for level in (0..self.path.len() - 1).rev() {
62            before()?;
63            let step = &mut self.path[level];
64            let side = usize::from(step.right);
65            step.node.children[side] = child;
66            step.node.balance += if step.right { 1 } else { -1 };
67            step.changed = true;
68            child = Some(encoded(step.id));
69            if step.node.balance == 0 {
70                return Ok(root);
71            }
72            if step.node.balance.abs() == 2 {
73                child = self.rotate(level, before)?;
74                if level == 0 {
75                    return Ok(child);
76                }
77                before()?;
78                let parent = &mut self.path[level - 1];
79                parent.node.children[usize::from(parent.right)] = child;
80                parent.changed = true;
81                return Ok(root);
82            }
83        }
84        Ok(child)
85    }
86
87    fn rotate<E>(
88        &mut self,
89        level: usize,
90        before: &mut impl FnMut() -> Result<(), E>,
91    ) -> Result<Link, E> {
92        // Before the first imbalance, the heavy child and (for a double
93        // rotation) grandchild are the next two nodes in the insertion path.
94        before()?;
95        let mut root = self.path[level];
96        let mut child = self.path[level + 1];
97        let side = usize::from(root.right);
98        let opposite = 1 - side;
99        let sign = if root.right { 1 } else { -1 };
100        if child.node.balance == sign {
101            before()?;
102            root.node.children[side] = child.node.children[opposite];
103            child.node.children[opposite] = Some(encoded(root.id));
104            root.node.balance = 0;
105            child.node.balance = 0;
106            self.path[level] = root;
107            self.path[level + 1] = child;
108            return Ok(Some(encoded(child.id)));
109        }
110        before()?;
111        let mut pivot = self.path[level + 2];
112        root.node.children[side] = pivot.node.children[opposite];
113        child.node.children[opposite] = pivot.node.children[side];
114        pivot.node.children[opposite] = Some(encoded(root.id));
115        pivot.node.children[side] = Some(encoded(child.id));
116        root.node.balance = if pivot.node.balance == sign { -sign } else { 0 };
117        child.node.balance = if pivot.node.balance == -sign { sign } else { 0 };
118        pivot.node.balance = 0;
119        pivot.changed = true;
120        self.path[level] = root;
121        self.path[level + 1] = child;
122        self.path[level + 2] = pivot;
123        Ok(Some(encoded(pivot.id)))
124    }
125
126    /// All writes below are pre-admitted as one indivisible publication. A
127    /// callback cannot observe a half-rotated tree or an unindexed payload.
128    pub fn publish(&mut self, root: Link) {
129        self.publish_nodes();
130        self.root = root;
131    }
132
133    /// Publish the planned leaf and the changed path nodes, leaving the root
134    /// to an owner that keeps one root per subtree.
135    pub fn publish_nodes(&mut self) {
136        let leaf = self.path.last().expect("planned leaf");
137        self.nodes.push(leaf.node);
138        for step in &self.path[..self.path.len() - 1] {
139            if step.changed {
140                self.nodes[step.id] = step.node;
141            }
142        }
143    }
144}

146/// Search the owner's ordered identities without owning or copying any payload.
147/// `compare` admits one node visit before comparing the query with the ID's
148/// authoritative value. The callback's ordering must be the tree's ordering.
149/// `descend` records only successful comparisons; an error publishes no result
150/// and changes no node. A miss visits at most the AVL height in nodes.
151pub(crate) fn search<E>(
152    nodes: &[Node],
153    mut cursor: Link,
154    mut compare: impl FnMut(usize) -> Result<Ordering, E>,
155    mut descend: impl FnMut(bool),
156) -> Result<Option<usize>, E> {
157    while let Some(next) = cursor {
158        let id = position(next);
159        let order = compare(id)?;
160        if order.is_eq() {
161            return Ok(Some(id));
162        }
163        let right = order == Ordering::Greater;
164        descend(right);
165        cursor = nodes[id].children[usize::from(right)];
166    }
167    Ok(None)
168}

170/// Fixed traversal state, never another index or an atom owner.
171///
172/// Let `N(h)` be the minimum node count of an AVL of height h, with empty height
173/// zero. Its recurrence gives `N(h) >= 2*N(h-2)+1` and hence
174/// `N(h) >= 2^ceil(h/2)-1`. For a nonempty tree with n nodes and bit width b,
175/// `n < 2^b` implies `h <= 2*b`. Since `b <= usize::BITS`, two target-sized words
176/// hold every descent. The planned insertion leaf is not a descent bit.
177/// The empty tree records no directions. This argument uses mathematical `n+1`;
178/// no potentially overflowing machine addition is needed to compute the bound.
179///
180/// Checked packing protects the representation independently of that AVL
181/// invariant. A failed push from an actual search means the internal AVL height
182/// invariant was broken, not a user resource refusal or missing atom.
183/// This local stack record lives only during entry lookup/path preparation;
184/// its two words and checked length are excluded from named vector capacities.
185#[derive(Default)]
186pub(crate) struct Directions {
187    words: [usize; 2],
188    length: usize,
189}
190
191impl Directions {
192    pub(crate) fn push(&mut self, right: bool) -> Option<()> {
193        let next = self.length.checked_add(1)?;
194        if next > self.words.len() * usize::BITS as usize {
195            return None;
196        }
197        // The newest direction occupies bit zero. Transfer the low word's
198        // oldest bit before shifting; the admitted length ensures that no
199        // recorded bit can leave the high word. Decode positions only on replay.
200        let carry = self.words[0] >> (usize::BITS - 1);
201        self.words[0] = (self.words[0] << 1) | usize::from(right);
202        self.words[1] = (self.words[1] << 1) | carry;
203        self.length = next;
204        Some(())
205    }
206
207    pub(crate) fn len(&self) -> usize {
208        self.length
209    }
210
211    pub(crate) fn get(&self, position: usize) -> Option<bool> {
212        if position >= self.length {
213            return None;
214        }
215        let (word, mask) = Self::position(self.length - 1 - position)?;
216        Some(*self.words.get(word)? & mask != 0)
217    }
218
219    fn position(position: usize) -> Option<(usize, usize)> {
220        let width = usize::BITS as usize;
221        let offset = u32::try_from(position % width).ok()?;
222        Some((position / width, 1_usize.checked_shl(offset)?))
223    }
224}
