1//! A published right spine in the owner's existing mutation buffer.
2//!
3//! Every scratch writer first takes its certificate. Only complete atomic
4//! publication restores it; refusal or unwind leaves tentative cells untrusted.
5//! The owner separately proves that an insertion follows its semantic maximum.
6
7use super::{Index, Link, Planned, Rotation, Step};
8
9#[derive(Clone, Copy)]
10pub(crate) struct RightSpine {
11    root: Link,
12    last: usize,
13    length: usize,
14}
15
16impl RightSpine {
17    /// Root and endpoint come from this same published tree. The owner revokes
18    /// this certificate before every mutation/repurposing of its path buffer.
19    pub(crate) fn matches(self, index: &Index, root: Link, last: usize) -> bool {
20        self.root == root
21            && self.last == last
22            && self.length == index.path.len()
23            && index.path.last().is_some_and(|step| step.id == last)
24    }
25
26    pub(crate) fn last(self) -> usize {
27        self.last
28    }
29}
30
31/// All-right insertion permits only a single left rotation. Its old root leaves
32/// the right spine, while the promoted child and unchanged prefix remain. Admit
33/// the removal's actual shifted suffix and every reset cell before publication.
34pub(crate) fn admit<E>(
35    index: &Index,
36    plan: Planned,
37    before: &mut impl FnMut() -> Result<(), E>,
38) -> Result<(), E> {
39    let removed = match plan.rotation {
40        Some(Rotation::Single(at)) => {
41            for _ in at + 1..index.path.len() {
42                before()?;
43            }
44            1
45        }
46        Some(Rotation::Double) => unreachable!("all-right insertion has no double rotation"),
47        None => 0,
48    };
49    for _ in plan.changed_from..index.path.len() - removed {
50        before()?;
51    }
52    before()?; // Replacement certificate publication.
53    Ok(())
54}
55
56/// Only call after successful publication and `admit`, with an all-right input
57/// path. Existing prefix cells already equal published nodes; reset just the
58/// planned suffix. No callback, allocation or recoverable failure occurs here.
59pub(crate) fn retain(index: &mut Index, plan: Planned, last: usize) -> RightSpine {
60    if let Some(rotation) = plan.rotation {
61        let Rotation::Single(at) = rotation else {
62            unreachable!("admitted right-spine rotation")
63        };
64        index.path.remove(at);
65    }
66    for Step { right, changed, .. } in &mut index.path[plan.changed_from..] {
67        *right = true;
68        *changed = false;
69    }
70    RightSpine {
71        root: plan.root,
72        last,
73        length: index.path.len(),
74    }
75}
