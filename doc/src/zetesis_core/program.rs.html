25const DEFAULT_MAX_TEMPLATES: usize = 100_000;
26const DEFAULT_MAX_PREDICATE_ARITY: usize = 32;
27const DEFAULT_MAX_VARIABLES: usize = 64;
28const DEFAULT_MAX_POSITIVE_BODY: usize = 1_024;
29const DEFAULT_MAX_DOMAIN_VALUES: usize = 1_000_000;

33/// Template-level admission budgets. Zero is a real limit, never unlimited.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub struct AdmissionLimits {
36    /// Maximum number of source-normalized rules: templates and fact occurrences.
37    pub max_templates: usize,
38    /// Maximum predicate arity.
39    pub max_predicate_arity: usize,
40    /// Maximum distinct variables in any one template.
41    pub max_variables_per_template: usize,
42    /// Maximum ordinary positive antecedents per template.
43    pub max_positive_body: usize,
44    /// Maximum distinct values appearing anywhere, including filters/gates.
45    pub max_domain_values: usize,
46    /// Inclusive canonical storage and admitted metadata allowance, including
47    /// indexes, vector capacity, admission scratch and publication overlap.
48    /// Caller-supplied construction descriptions, allocator bookkeeping and Arc
49    /// reference counters are excluded. The default is 128 MiB, independently
50    /// of the template/domain count limits; zero is a real allowance.
51    pub max_bytes: usize,
52}
53impl Default for AdmissionLimits {
54    fn default() -> Self {
55        Self {
56            max_templates: DEFAULT_MAX_TEMPLATES,
57            max_predicate_arity: DEFAULT_MAX_PREDICATE_ARITY,
58            max_variables_per_template: DEFAULT_MAX_VARIABLES,
59            max_positive_body: DEFAULT_MAX_POSITIVE_BODY,
60            max_domain_values: DEFAULT_MAX_DOMAIN_VALUES,
61            max_bytes: DEFAULT_MAX_BYTES,
62        }
63    }
64}

66/// A resource counted by source admission, before any carrier expansion.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68pub enum AdmissionResource {
69    /// Total template and table-fact occurrences.
70    Templates,
71    /// Predicate argument count.
72    PredicateArity,
73    /// Distinct variables within one template.
74    Variables,
75    /// Ordinary positive antecedent count.
76    PositiveBody,
77    /// Distinct closed values.
78    DomainValues,
79}

81/// The native input associated with a canonical admission failure.
82/// Template and fact indices refer to their separate original input sequences;
83/// this is an input identity, not a parsed source coordinate.
84#[derive(Clone, Copy, Debug, PartialEq, Eq)]
85pub enum AdmissionOrigin {
86    /// A whole-program operation with no individual occurrence.
87    Program,
88    /// An original template occurrence.
89    Template(usize),
90    /// An original table-fact occurrence.
91    Fact(usize),
92}

94/// A typed admission refusal. Template and fact indices retain original input identity.
95#[derive(Clone, Debug, PartialEq, Eq)]
96pub enum AdmissionError {
97    /// A configured count bound failed at an original fact occurrence.
98    FactLimitExceeded {
99        /// Count whose inclusive bound was exceeded.
100        resource: AdmissionResource,
101        /// Configured maximum.
102        limit: usize,
103        /// Requested count.
104        actual: usize,
105        /// Original table-fact occurrence index.
106        fact: usize,
107    },
108    /// Canonical storage or admitted metadata could not be constructed.
109    Canonical {
110        /// Typed storage, allocation or representation cause. Storage byte
111        /// counts include the simultaneously live admitted metadata.
112        error: crate::catalog::Error,
113        /// Original input occurrence, or a whole-program operation. One shared
114        /// cause representation reports allocation failures without allocating.
115        origin: AdmissionOrigin,
116    },
117    /// A configured bound was exceeded.
118    LimitExceeded {
119        /// Which count exceeded its bound.
120        resource: AdmissionResource,
121        /// Configured maximum.
122        limit: usize,
123        /// Observed count.
124        actual: usize,
125        /// Original template index, or `None` for a program-wide count.
126        template: Option<usize>,
127    },
128    /// A variable is used without an ordinary positive binding antecedent.
129    UnsafeVariable {
130        /// Original template index.
131        template: usize,
132        /// Unbound variable ID.
133        variable: usize,
134    },
135    /// Variable IDs do not form the dense range starting at zero.
136    NonDenseVariable {
137        /// Original template index.
138        template: usize,
139        /// Next required ID.
140        expected: usize,
141        /// Actual ID at that sorted position.
142        actual: usize,
143    },
144}
145impl AdmissionError {
146    /// Original fact-occurrence position for parallel source evidence. Template
147    /// positions and fact positions are distinct; neither is the whole rule order.
148    #[must_use]
149    pub const fn fact_index(&self) -> Option<usize> {
150        match self {
151            Self::Canonical {
152                origin: AdmissionOrigin::Fact(fact),
153                ..
154            }
155            | Self::FactLimitExceeded { fact, .. } => Some(*fact),
156            _ => None,
157        }
158    }
159    /// Locate a template-level refusal without parsing a message.
160    #[must_use]
161    pub fn template_index(&self) -> Option<usize> {
162        match self {
163            Self::Canonical {
164                origin: AdmissionOrigin::Template(template),
165                ..
166            } => Some(*template),
167            Self::Canonical {
168                origin: AdmissionOrigin::Program | AdmissionOrigin::Fact(_),
169                ..
170            }
171            | Self::FactLimitExceeded { .. } => None,
172            Self::LimitExceeded { template, .. } => *template,
173            Self::UnsafeVariable { template, .. } | Self::NonDenseVariable { template, .. } => {
174                Some(*template)
175            }
176        }
177    }
178}
179impl fmt::Display for AdmissionError {
180    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
181        match self {
182            Self::FactLimitExceeded {
183                resource,
184                limit,
185                actual,
186                fact,
187            } => write!(
188                f,
189                "{resource:?} count {actual} exceeds {limit} (fact {fact})"
190            ),
191            Self::Canonical { error, origin } => match origin {
192                AdmissionOrigin::Fact(fact) => {
193                    write!(f, "program admission failed (fact {fact}): {error}")
194                }
195                AdmissionOrigin::Template(template) => write!(
196                    f,
197                    "program admission failed (template Some({template})): {error}"
198                ),
199                AdmissionOrigin::Program => {
200                    write!(f, "program admission failed (template None): {error}")
201                }
202            },
203            Self::LimitExceeded {
204                resource,
205                limit,
206                actual,
207                template,
208            } => write!(
209                f,
210                "{resource:?} count {actual} exceeds {limit} (template {template:?})"
211            ),
212            Self::UnsafeVariable { template, variable } => write!(
213                f,
214                "template {template} variable {variable} has no positive binding"
215            ),
216            Self::NonDenseVariable {
217                template,
218                expected,
219                actual,
220            } => write!(
221                f,
222                "template {template} expected variable ID {expected}, found {actual}"
223            ),
224        }
225    }
226}
227impl std::error::Error for AdmissionError {
228    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
229        match self {
230            Self::Canonical { error, .. } => Some(error),
231            _ => None,
232        }
233    }
234}

257impl ProgramData {
258    fn term(&self, id: TermId) -> TermRef<'_> {
259        crate::template::catalog::term(self.catalog.read(), id)

262        crate::template::catalog::predicate(self.catalog.read(), id)

265        self.catalog

270        Patterns::admitted(self.catalog.read(), &self.row(template).patterns[range])
271    }

274/// An immutable admitted program with one canonical term/predicate authority.
275/// Clones share identity and payload; equal independently admitted syntax is a
276/// different instance and cannot silently reuse a seed. Stored templates contain
277/// only variable slots and references into the frozen authority. Canonical
278/// subterms do not become domain members unless explicitly present as constants.
279#[derive(Clone, Debug)]
280pub struct Program(Arc<ProgramData>);
281
282impl Program {
283    /// Consume templates and explicit fact tables without enumerating variable
284    /// substitutions. Fact admission interns their distinct atoms once and keeps
285    /// original occurrences beside predicate-local relation columns.
286    /// Variables must be dense and occur in at least one positive pattern;
287    /// comparisons and candidate gates do not establish safety. Closed values
288    /// already passed their constructors, so no implicit depth-128 limit is
289    /// imposed again. The named storage allowance still bounds canonical import.
290    ///
291    /// Original template/body and fact occurrences survive as metadata. Facts
292    /// supplied as templates remain templates; this constructor never reclassifies.
293    /// Owned construction payloads are discarded; borrowed execution views use
294    /// the frozen base.
295    /// Buffer reservations are fallible. Arc envelopes retain stable Rust's
296    /// infallible allocation boundary, not universal allocation recovery.
297    ///
298    /// # Cost and storage
299    /// Imports each explicit scalar alternative into the shared vocabulary,
300    /// then visits every fact occurrence and its argument columns once. Exact
301    /// atom interning and relation indexing additionally perform their checked
302    /// key lookups and reservations. Intervals avoid an expanded owned input
303    /// vector, but this admission still enumerates their scalar values and rows.
304    /// For `f` original occurrences and `u` distinct atoms, retained occurrence
305    /// metadata is O(f); canonical atom payload and relation columns are O(u)
306    /// at fixed arity. Template topology, explicit-domain entries and canonical
307    /// vocabulary have their own retained capacities. Temporary imported-column
308    /// IDs and old/replacement/publication overlap share the byte allowance.
309    ///
310    /// # Errors
311    /// Refuses unsafe/non-dense variables, exceeded counts, unavailable named
312    /// storage or canonical representation limits. No partial `Program` escapes.
313    pub fn new(
314        templates: Vec<Template>,
315        facts: FactTables,
316        limits: AdmissionLimits,
317    ) -> Result<Self, AdmissionError> {
318        admission::admit(templates, facts, limits).map(|data| Self(Arc::new(data)))
319    }
320
321    /// All template and fact occurrences in original admitted order. Facts retain
322    /// duplicate occurrences while sharing each distinct atom's canonical payload.
323    #[must_use]
324    pub fn rules(&self) -> Rules<'_> {
325        Rules::new(&self.0)
326    }
327
328    /// Whether neither templates nor facts occur in this program.
329    #[must_use]
330    pub fn is_empty(&self) -> bool {
331        self.0.templates.is_empty() && self.facts().is_empty()
332    }
333
334    /// Original fact occurrences, for traversal and parallel source evidence.
335    #[must_use]
336    pub fn facts(&self) -> FactOccurrences<'_> {
337        FactOccurrences {
338            data: self.0.facts.as_ref(),
339        }
340    }
341
342    /// Distinct fact relations in signed predicate order, borrowing their closed
343    /// authority, column dictionaries and prepared canonical runs.
344    pub fn fact_relations(&self) -> impl Iterator<Item = FactRelation<'_>> {
345        facts::relations(self.0.facts.as_ref())
346    }
347
348    /// One immutable fact authority for oracle descendants. Its storage presence
349    /// is not discovery or truth; each descendant starts with empty discovery.
350    /// A program with no table facts has no closed fact authority.
351    #[must_use]
352    pub fn fact_catalog(&self) -> Option<&crate::atom_interner::ClosedCatalog> {
353        self.0.facts.as_ref().map(|facts| &facts.store)
354    }
355
356    /// Non-fact templates in original admitted order, including duplicates.
357    /// Body-free templates explicitly supplied to `new` remain templates; only
358    /// the separate `FactTables` input becomes table facts.
359    #[must_use]
360    pub fn templates(&self) -> Templates<'_> {
361        Templates { program: &self.0 }
362    }
363
364    /// Distinct explicitly occurring closed constants in semantic storage order.
365    /// The view excludes merely interned subterms and later execution identities.
366    #[must_use]
367    pub fn domain(&self) -> Domain<'_> {
368        Domain { program: &self.0 }
369    }
370
371    /// Distinct signed signatures in name, arity and sign order.
372    #[must_use]
373    pub fn predicates(&self) -> Predicates<'_> {
374        Predicates {
375            program: &self.0,
376            ids: &self.0.predicates,
377        }
378    }
379
380    /// Signatures consulted by any gate, including constraint-only gates.
381    #[must_use]
382    pub fn gate_predicates(&self) -> Predicates<'_> {
383        Predicates {
384            program: &self.0,
385            ids: &self.0.gate_predicates,
386        }
387    }
388
389    pub(crate) fn vocabulary(&self) -> &FrozenVocabulary {
390        self.0
391            .catalog
392            .vocabulary()
393            .expect("Program admission constructs an indexed closed vocabulary")
394    }
395
396    /// Named canonical-base and admitted-metadata retention. Shared clones name
397    /// the same allocations; adding their byte counts would count them twice.
398    /// This excludes caller input, allocator overhead and future tuple storage.
399    #[must_use]
400    pub fn storage_bytes(&self) -> u128 {
401        self.0.catalog.storage_bytes()
402            + self.0.metadata_bytes
403            + self
404                .0
405                .facts
406                .as_ref()
407                .map_or(0, |facts| facts.bytes(self.shared_vocabulary_bytes()))
408    }
409
410    /// Immutable vocabulary payload and lookup indexes shared by tuple owners
411    /// created with [`crate::atom_interner::AtomInterner::for_program`]. This is
412    /// included in [`Self::storage_bytes`]; it excludes Program metadata and the
413    /// tuple owner's separate rows, indexes, scratch and ownership envelopes.
414    /// A caller whose allowance excludes the input Program may subtract this
415    /// exact shared subtotal once from that tuple owner's complete receipt.
416    #[must_use]
417    pub fn shared_vocabulary_bytes(&self) -> u128 {
418        self.0.catalog.shared_vocabulary_bytes()
419    }
420
421    /// Whether two handles share precisely the same admitted instance.
422    #[must_use]
423    pub fn same_instance(&self, other: &Self) -> bool {
424        Arc::ptr_eq(&self.0, &other.0)
425    }
426
427    /// Test symbolic atom-carrier membership without enumerating its tuples.
428    #[must_use]
429    pub fn contains_atom<'a>(&self, atom: impl Into<AtomRef<'a>>) -> bool {
430        let atom = atom.into();
431        self.predicates().binary_search(atom.predicate()).is_ok()
432            && atom
433                .values()
434                .iter()
435                .all(|value| self.domain().binary_search(value).is_ok())
436    }
437
438    /// Test the conservative gate carrier: gate predicates over the whole domain.
439    #[must_use]
440    pub fn contains_gate_atom<'a>(&self, atom: impl Into<AtomRef<'a>>) -> bool {
441        let atom = atom.into();
442        self.gate_predicates()
443            .binary_search(atom.predicate())
444            .is_ok()
445            && self.contains_atom(atom)
446    }
447
448    /// Iterate the gate carrier lazily, without pre-expansion.
449    #[must_use]
450    pub fn gate_atoms(&self) -> AtomIter<'_> {
451        AtomIter::new(self, self.gate_predicates())
452    }
453
454    /// Iterate the full conceptual atom carrier, without pre-expansion.
455    #[must_use]
456    pub fn carrier_atoms(&self) -> AtomIter<'_> {
457        AtomIter::new(self, self.predicates())
458    }

461fn check_limit(
462    resource: AdmissionResource,
463    actual: usize,
464    limit: usize,
465    template: Option<usize>,
466) -> Result<(), AdmissionError> {
467    if actual > limit {
468        Err(AdmissionError::LimitExceeded {
469            resource,
470            limit,
471            actual,
472            template,
473        })
474    } else {
475        Ok(())
476    }
