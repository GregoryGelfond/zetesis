35impl std::fmt::Debug for Data {

37        f.debug_struct("Facts")
38            .field("predicates", &self.relations.len())

41            .finish_non_exhaustive()

45impl Data {
46    /// Extra retained payload beyond the containing `ProgramData` header and its
47    /// already-counted vocabulary. Relation headers reside in their vector.
48    pub(super) fn bytes(&self, vocabulary: u128) -> u128 {
49        self.store.storage_bytes() - size_of::<ClosedCatalog>() as u128 - vocabulary
50            + self.relations.capacity() as u128 * size_of::<Catalog>() as u128
51            + self
52                .relations
53                .iter()
54                .map(|relation| relation.retained_bytes() as u128 - size_of::<Catalog>() as u128)
55                .sum::<u128>()
56            + self.relation_order.capacity() as u128 * size_of::<usize>() as u128
57            + self.occurrences.capacity() as u128 * size_of::<Occurrence>() as u128
58            + self.blocks.capacity() as u128 * size_of::<Block>() as u128
59    }
60
61    fn atom(&self, occurrence: Occurrence) -> AtomRef<'_> {
62        self.relations[occurrence.relation]
63            .atoms(self.store.read())
64            .expect("fact relation belongs to its closed authority")
65            .at(occurrence.row)
66            .expect("fact occurrence names an admitted relation row")
67    }
68}

70/// One original fact occurrence, preserving source order and duplicates.
71/// Its atom borrows the shared fact store. No value or tuple is reconstructed.
72#[derive(Clone, Copy)]
73pub struct FactRef<'a> {
74    data: &'a Data,
75    index: usize,
76}
77
78impl<'a> FactRef<'a> {
79    /// Position in the original fact-occurrence sequence, for parallel evidence.
80    #[must_use]
81    pub const fn index(self) -> usize {
82        self.index
83    }
84
85    /// Canonical atom of this occurrence. Equal occurrences share its payload.
86    #[must_use]
87    pub fn atom(self) -> AtomRef<'a> {
88        self.data.atom(self.data.occurrences[self.index])
89    }
90
91    /// A constant head pattern borrowing the same canonical atom.
92    #[must_use]
93    pub fn pattern(self) -> PatternRef<'a> {
94        PatternRef::fact(self.atom())
95    }
96
97    /// A body-free rule view for readers that already consume normalized rules.
98    /// This allocates no template, term array or owned value.
99    #[must_use]
100    pub fn as_template(self) -> TemplateRef<'a> {
101        TemplateRef::fact(self.atom())
102    }
103}

105impl std::fmt::Debug for FactRef<'_> {
106    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
107        f.debug_struct("Fact")
108            .field("index", &self.index)
109            .field("atom", &self.atom())
110            .finish()
111    }

114/// Borrowed random access and source-order traversal of original fact occurrences.
115#[derive(Clone, Copy)]
116pub struct FactOccurrences<'a> {
117    pub(super) data: Option<&'a Data>,
118}
119impl<'a> FactOccurrences<'a> {
120    #[must_use]
121    /// Number of original occurrences, including duplicates.
122    pub fn len(self) -> usize {
123        self.data.map_or(0, |data| data.occurrences.len())
124    }
125    #[must_use]
126    /// Whether this view contains no occurrence.
127    pub fn is_empty(self) -> bool {
128        self.len() == 0
129    }
130    #[must_use]
131    /// Borrow an occurrence by its original table-fact index.
132    pub fn at(self, index: usize) -> Option<FactRef<'a>> {
133        let data = self.data?;
134        (index < data.occurrences.len()).then_some(FactRef { data, index })
135    }
136    /// Traverse every original occurrence in order.
137    #[must_use]
138    pub fn iter(self) -> impl ExactSizeIterator<Item = FactRef<'a>> {
139        self.into_iter()
140    }
141}
142impl<'a> IntoIterator for FactOccurrences<'a> {
143    type Item = FactRef<'a>;
144    type IntoIter = FactIter<'a>;
145    fn into_iter(self) -> Self::IntoIter {
146        FactIter {
147            facts: self,
148            next: 0,
149        }
150    }
151}
152/// Iterator over original table-fact occurrences.
153pub struct FactIter<'a> {
154    facts: FactOccurrences<'a>,
155    next: usize,
156}
157impl<'a> Iterator for FactIter<'a> {
158    type Item = FactRef<'a>;
159    fn next(&mut self) -> Option<Self::Item> {
160        let fact = self.facts.at(self.next)?;
161        self.next += 1;
162        Some(fact)
163    }
164    fn size_hint(&self) -> (usize, Option<usize>) {
165        let count = self.facts.len() - self.next;
166        (count, Some(count))
167    }
168}
169impl ExactSizeIterator for FactIter<'_> {}
170impl std::iter::FusedIterator for FactIter<'_> {}

172/// One predicate's distinct fact rows, columns and canonical order, all borrowed
173/// from the program's single closed authority and relation catalog.
174#[derive(Clone, Copy)]
175pub struct FactRelation<'a> {
176    relation: &'a Catalog,
177    read: CatalogRead<'a>,
178}
179impl<'a> FactRelation<'a> {
180    /// Columnar values and row IDs; equality IDs have no numeric ordering meaning.
181    ///
182    /// # Panics
183    /// Panics if admission's internal shared-authority invariant is violated.
184    /// Callers cannot construct an invalid relation view.
185    #[must_use]
186    pub fn relation(self) -> Relation<'a> {
187        self.relation.view(self.read).expect("closed fact relation")
188    }
189    /// Distinct atoms in catalog insertion order.
190    ///
191    /// # Panics
192    /// Panics if admission's internal shared-authority invariant is violated.
193    /// Callers cannot construct an invalid relation view.
194    #[must_use]
195    pub fn atoms(self) -> Atoms<'a> {
196        self.relation
197            .atoms(self.read)
198            .expect("closed fact relation")
199    }
200    /// Distinct atoms in canonical semantic storage order. Admission prepares the
201    /// catalog once after its final append, so its sole sorted run covers all rows.
202    ///
203    /// # Panics
204    /// Panics if admission's internal shared-authority or ordered-row invariant is
205    /// violated. Callers cannot construct an invalid relation view.
206    #[must_use]
207    pub fn canonical_atoms(self) -> impl ExactSizeIterator<Item = AtomRef<'a>> {
208        let atoms = self.atoms();
209        self.relation
210            .ordered()
211            .expect("sealed ordered facts")
212            .tail()
213            .iter()
214            .map(move |&row| atoms.at(row).expect("ordered fact row"))
215    }
216}

218pub(super) fn relations(data: Option<&Data>) -> impl Iterator<Item = FactRelation<'_>> {
219    data.into_iter().flat_map(|data| {
220        data.relation_order.iter().map(|&index| FactRelation {
221            relation: &data.relations[index],
222            read: data.store.read(),
223        })
224    })
225}

227/// An original program occurrence. Template indices and fact occurrence indices
228/// belong to separate sequences and are never inferred from one another.
229#[derive(Clone, Copy, Debug)]
230pub enum RuleRef<'a> {
231    /// One caller-supplied template.
232    Template {
233        /// Position in the template-only sequence.
234        index: usize,
235        /// Borrowed canonical template.
236        template: TemplateRef<'a>,
237    },
238    /// One explicit table-fact occurrence.
239    Fact(FactRef<'a>),
240}

242/// Whole-program traversal, merging anchored fact blocks with template order.
243pub struct Rules<'a> {
244    pub(super) program: &'a ProgramData,
245    template: usize,
246    block: usize,
247    fact: usize,
248    remaining: usize,
249}
250impl<'a> Rules<'a> {
251    pub(super) fn new(program: &'a ProgramData) -> Self {
252        Self {
253            program,
254            template: 0,
255            block: 0,
256            fact: 0,
257            remaining: program.templates.len()
258                + program
259                    .facts
260                    .as_ref()
261                    .map_or(0, |data| data.occurrences.len()),
262        }
263    }
264}
265impl<'a> Iterator for Rules<'a> {
266    type Item = RuleRef<'a>;
267    fn next(&mut self) -> Option<Self::Item> {
268        if self.remaining == 0 {
269            return None;
270        }
271        if let Some(data) = &self.program.facts
272            && let Some(block) = data.blocks.get(self.block)
273            && block.before == self.template
274        {
275            let index = block.occurrences.start + self.fact;
276            self.fact += 1;
277            if index + 1 == block.occurrences.end {
278                self.block += 1;
279                self.fact = 0;
280            }
281            self.remaining -= 1;
282            return Some(RuleRef::Fact(FactRef { data, index }));
283        }
284        let index = self.template;
285        let template = Templates {
286            program: self.program,
287        }
288        .at(index)
289        .expect("valid fact anchors preserve template traversal");
290        self.template += 1;
291        self.remaining -= 1;
292        Some(RuleRef::Template { index, template })
293    }
294    fn size_hint(&self) -> (usize, Option<usize>) {
295        (self.remaining, Some(self.remaining))
296    }
297}
298impl ExactSizeIterator for Rules<'_> {}
299impl std::iter::FusedIterator for Rules<'_> {}
