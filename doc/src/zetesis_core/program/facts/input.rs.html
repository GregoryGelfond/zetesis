1//! Bounded construction descriptions, consumed at the program admission door.
2
3use std::ops::Range;
4
5use crate::catalog::Error;
6use crate::{Predicate, Value};

8/// One closed argument alternative. An integer interval is inclusive; a
9/// descending interval contributes no values. Segment order is source order.
10#[derive(Clone, Debug)]
11pub enum FactSegment {
12    /// One already constructed closed logical value.
13    Value(Value),
14    /// Consecutive signed integers without an expanded input vector.
15    Interval {
16        /// Inclusive first integer.
17        first: i32,
18        /// Inclusive last integer.
19        last: i32,
20    },
21}
22
23impl FactSegment {
24    pub(super) fn len(&self) -> Result<usize, Error> {
25        match self {
26            Self::Value(_) => Ok(1),
27            Self::Interval { first, last } if first <= last => {
28                usize::try_from(i64::from(*last) - i64::from(*first) + 1)
29                    .map_err(|_| Error::Overflow)
30            }
31            Self::Interval { .. } => Ok(0),
32        }
33    }
34}

36/// Source-ordered fact construction descriptions, separate from rule templates.
37/// Each block is anchored immediately before a template position. Products keep
38/// compact intervals; adjacent rows of one predicate and anchor share column
39/// vectors. The program consumes these descriptions and retains one canonical
40/// atom store plus occurrence positions, not these owned values.
41///
42/// Caller input is outside `AdmissionLimits::max_bytes`, as with `Template`.
43/// Builder vector reservations are fallible. A failed push poisons this value;
44/// admission then refuses it rather than publishing an accepted prefix.
45#[derive(Default, Debug)]
46pub struct FactTables {
47    pub(in crate::program) blocks: Vec<Block>,
48    len: usize,
49    pub(in crate::program) failed: Option<Error>,
50}
51
52#[derive(Debug)]
53pub(in crate::program) struct Block {
54    pub(in crate::program) before: usize,
55    pub(in crate::program) predicate: Predicate,
56    pub(in crate::program) first: usize,
57    pub(in crate::program) len: usize,
58    pub(in crate::program) columns: Columns,
59}
60
61#[derive(Debug)]
62pub(in crate::program) enum Columns {
63    Product(Vec<Vec<FactSegment>>),
64    Run(Vec<Vec<Value>>),
65}
66
67impl FactTables {
68    /// Original fact occurrences, including duplicate atoms.
69    #[must_use]
70    pub const fn len(&self) -> usize {
71        self.len
72    }
73
74    /// Whether no fact occurrence has been submitted.
75    #[must_use]
76    pub const fn is_empty(&self) -> bool {
77        self.len == 0
78    }
79
80    /// Append one closed row at `before`, the count of preceding templates.
81    /// Returns its original fact-occurrence position. Equal atoms remain distinct
82    /// occurrences; compatible adjacent rows share one column-major input run.
83    /// Moves O(arity) owned value handles; growing columns can move their prior
84    /// handles. Amortized work is O(arity) per row, excluding predicate comparison
85    /// and allocator behavior. Values are neither canonicalized nor expanded here.
86    ///
87    /// # Errors
88    /// Refuses an arity mismatch, decreasing anchor, index overflow, allocation,
89    /// or an earlier failed push. A refusal poisons this construction owner.
90    pub fn push_row(
91        &mut self,
92        before: usize,
93        predicate: Predicate,
94        values: Vec<Value>,
95    ) -> Result<usize, Error> {
96        let result = self.row(before, predicate, values);
97        if let Err(error) = &result {
98            self.failed = Some(error.clone());
99        }
100        result
101    }
102
103    fn row(
104        &mut self,
105        before: usize,
106        predicate: Predicate,
107        values: Vec<Value>,
108    ) -> Result<usize, Error> {
109        self.check(before, &predicate, values.len())?;
110        let next = self.len.checked_add(1).ok_or(Error::Overflow)?;
111        if let Some(Block {
112            before: previous,
113            predicate: prior,
114            len,
115            columns: Columns::Run(columns),
116            ..
117        }) = self.blocks.last_mut()
118            && *previous == before
119            && prior == &predicate
120        {
121            // Every nested reservation precedes any row write.
122            for column in columns.iter_mut() {
123                column.try_reserve(1).map_err(|_| Error::Allocation)?;
124            }
125            for (column, value) in columns.iter_mut().zip(values) {
126                column.push(value);
127            }
128            *len += 1;
129        } else {
130            self.blocks.try_reserve(1).map_err(|_| Error::Allocation)?;
131            let mut columns = Vec::new();
132            columns
133                .try_reserve_exact(values.len())
134                .map_err(|_| Error::Allocation)?;
135            for value in values {
136                let mut column = Vec::new();
137                column.try_reserve_exact(1).map_err(|_| Error::Allocation)?;
138                column.push(value);
139                columns.push(column);
140            }
141            self.blocks.push(Block {
142                before,
143                predicate,
144                first: self.len,
145                len: 1,
146                columns: Columns::Run(columns),
147            });
148        }
149        let first = self.len;
150        self.len = next;
151        Ok(first)
152    }
153
154    /// Append a Cartesian product in last-argument-fastest order. Columns contain
155    /// their alternatives in source order; one empty column makes an empty
156    /// product, while a zero-arity product contains the one nullary fact.
157    /// Returns the original occurrence range for parallel source evidence.
158    /// Checks O(segments + arity) metadata and retains the supplied columns;
159    /// interval width and Cartesian products are checked without enumerating
160    /// scalar values or rows. `Program::new` performs that admitted enumeration.
161    ///
162    /// # Errors
163    /// Same structural and allocation refusals as [`Self::push_row`], plus an
164    /// overflowing alternative or product count. A refusal poisons this owner.
165    pub fn push_product(
166        &mut self,
167        before: usize,
168        predicate: Predicate,
169        columns: Vec<Vec<FactSegment>>,
170    ) -> Result<Range<usize>, Error> {
171        let result = self.product(before, predicate, columns);
172        if let Err(error) = &result {
173            self.failed = Some(error.clone());
174        }
175        result
176    }
177
178    fn product(
179        &mut self,
180        before: usize,
181        predicate: Predicate,
182        columns: Vec<Vec<FactSegment>>,
183    ) -> Result<Range<usize>, Error> {
184        self.check(before, &predicate, columns.len())?;
185        let width = |column: &[FactSegment]| {
186            column.iter().try_fold(0_usize, |sum, segment| {
187                sum.checked_add(segment.len()?).ok_or(Error::Overflow)
188            })
189        };
190        let empty = columns.iter().try_fold(false, |empty, column| {
191            let count = width(column)?;
192            Ok::<_, Error>(empty || count == 0)
193        })?;
194        if empty {
195            return Ok(self.len..self.len);
196        }
197        let count = columns.iter().try_fold(1_usize, |rows, column| {
198            rows.checked_mul(width(column)?).ok_or(Error::Overflow)
199        })?;
200        let next = self.len.checked_add(count).ok_or(Error::Overflow)?;
201        self.blocks.try_reserve(1).map_err(|_| Error::Allocation)?;
202        self.blocks.push(Block {
203            before,
204            predicate,
205            first: self.len,
206            len: count,
207            columns: Columns::Product(columns),
208        });
209        let range = self.len..next;
210        self.len = next;
211        Ok(range)
212    }
213
214    fn check(&self, before: usize, predicate: &Predicate, arity: usize) -> Result<(), Error> {
215        if let Some(error) = &self.failed {
216            return Err(error.clone());
217        }
218        if predicate.arity() != arity
219            || self
220                .blocks
221                .last()
222                .is_some_and(|block| block.before > before)
223        {
224            return Err(Error::Shape);
225        }
226        Ok(())
227    }
228
229    /// Signed signatures and their first original occurrence, in input block
230    /// order. Coherence scans each block once without expanding its rows.
231    pub fn predicates(&self) -> impl Iterator<Item = (&Predicate, usize)> {
232        self.blocks
233            .iter()
234            .map(|block| (&block.predicate, block.first))
235    }
236}
