1//! Bounded immutable columns over typed relation rows.
2//!
3//! This execution view preserves the source's signed predicate,
4//! row order and duplicate row occurrences. It is a finite indexed tuple view:
5//! two equal row occurrences do not denote distinct ASP atoms or additional
6//! truth. Extensional atom uniqueness and source-instance provenance remain
7//! separate caller contracts. It neither admits a program nor
8//! establishes candidate truth. Equality identifiers have no numeric or ASP
9//! ordering meaning. Typed row access decodes the same dictionary used by CPU
10//! selection and device views.
11//!
12//! The source remains borrowed. Columns and dictionary representatives are
13//! either owned by this view or borrowed from an appendable [`Catalog`]; no
14//! complete atom or logical payload is cloned. A catalog mediates membership
15//! append and binds rows from a separately charged canonical authority. Other
16//! borrowed-source constructors remain explicit ingress views.
17//! Eager formula support uses columns for typed lookup and
18//! row access; device consumers use the same representation for equality masks.
19//!
20//! Ordered access is through the explicit [`Catalog::prepare_ordered`] and
21//! [`Catalog::ordered`] views over sorted runs.
22//! A missing prepared view denotes required preparation, never an empty
23//! relation; preparation after appends sorts them into a new run and keeps the
24//! older runs as geometrically shrinking levels, so no preparation copies the
25//! whole extent. [`Catalog::canonical`] merges the runs into one sequence when
26//! rank access is needed. Relation and
27//! catalog work charge actual typed descriptor/text-prefix comparisons.
28//!
29//! Limits cover one operation's relation, supplied query/selection and newly
30//! allocated buffers. Other live caller frames and borrowed source allocations
31//! remain the caller's responsibility. Reported capacity is not process RSS.
32

49/// Inclusive construction and operation ceilings.
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub struct Limits {
52    /// Maximum source row occurrences, including duplicates.
53    pub max_rows: usize,
54    /// Maximum predicate arity.
55    pub max_columns: usize,
56    /// Maximum distinct typed dictionary values.
57    pub max_values: usize,
58    /// Maximum operation-scoped named capacity, including input views and the
59    /// conservative old/replacement-buffer overlap of a growing owner. Hash
60    /// storage measures addressable entry capacity, not opaque bucket/control
61    /// allocation or allocator bookkeeping.
62    pub max_bytes: usize,
63    /// Maximum charged inspections, comparisons, copied reference/ID cells,
64    /// payload bytes and fixed-key hash operations. Internal container probes
65    /// are not individually charged or cancellable.
66    pub max_work: u64,
67}
68
69impl Default for Limits {
70    fn default() -> Self {
71        Self {
72            max_rows: 65_536,
73            max_columns: 64,
74            max_values: 1_048_576,
75            max_bytes: 134_217_728,
76            max_work: 100_000_000,
77        }
78    }
79}

81/// The finite resource whose inclusive ceiling was exceeded.
82#[derive(Clone, Copy, Debug, PartialEq, Eq)]
83pub enum Resource {
84    /// Source row occurrences.
85    Rows,
86    /// Predicate arity.
87    Columns,
88    /// Distinct typed dictionary entries.
89    Values,
90    /// Operation-scoped named capacity, including addressable hash entries but
91    /// excluding opaque bucket/control allocation and allocator bookkeeping.
92    Bytes,
93    /// Charged logical work.
94    Work,
95}

97/// A relation operation failed without publishing a partial result.
98#[derive(Clone, Debug, PartialEq, Eq)]
99pub enum Failure {
100    /// Canonical membership could not bind to the supplied authority/prefix.
101    Read(crate::catalog::ReadError),
102    /// A source atom has a different signed predicate or arity.
103    Predicate,
104    /// An input catalog index is outside the borrowed atom catalog.
105    CatalogIndex,
106    /// A requested equality column is outside the predicate's arity.
107    Column,
108    /// Selection positions are not increasing, unique and in range.
109    Selection,
110    /// A packed selection has the wrong word count or nonzero unused tail bits.
111    Mask,
112    /// The supplied query or selection belongs to a different relation object.
113    Owner,
114    /// A validated source value was unexpectedly absent from its dictionary.
115    Dictionary,
116    /// A shape or capacity cannot be represented.
117    Overflow,
118    /// Storage could not be reserved.
119    Allocation,
120    /// Ordered access was requested for an extent whose appends are not yet
121    /// prepared.
122    Order,
123    /// An inclusive ceiling was exceeded.
124    Limit {
125        /// Exhausted resource.
126        resource: Resource,
127        /// Requested amount.
128        observed: u128,
129        /// Inclusive allowance.
130        limit: u128,
131    },
132}
133
134impl fmt::Display for Failure {
135    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
136        match self {
137            Self::Read(error) => error.fmt(f),
138            Self::Predicate => f.write_str("relation row has a foreign predicate"),
139            Self::CatalogIndex => f.write_str("relation catalog index is out of range"),
140            Self::Column => f.write_str("relation equality column is out of range"),
141            Self::Selection => {
142                f.write_str("relation selection is not ordered, unique and in range")
143            }
144            Self::Mask => f.write_str("relation mask has the wrong shape or nonzero tail bits"),
145            Self::Owner => f.write_str("relation query or selection has a foreign owner"),
146            Self::Dictionary => f.write_str("relation dictionary does not contain a source value"),
147            Self::Overflow => f.write_str("relation shape or capacity is not representable"),
148            Self::Allocation => f.write_str("relation storage could not be reserved"),
149            Self::Order => f.write_str("relation ordered access requires preparation after append"),
150            Self::Limit {
151                resource,
152                observed,
153                limit,
154            } => {
155                write!(f, "relation {resource:?} {observed} exceeds {limit}")
156            }
157        }
158    }
159}

170/// Capacity and work observed during one relation or catalog operation.
171#[derive(Clone, Copy, Debug, PartialEq, Eq)]
172pub struct Storage {
173    /// Operation owner's object, retained vector capacities and addressable
174    /// dictionary hash-entry capacity; opaque bucket/control allocation and
175    /// allocator bookkeeping are excluded. A borrowed
176    /// catalog view excludes the catalog's separately reported owner capacity.
177    pub retained_bytes: usize,
178    /// Largest operation-scoped capacity envelope, including temporary buffers
179    /// and old/replacement capacity overlap when a reservation occurs.
180    pub peak_construction_bytes: usize,
181    /// Sum of canonical typed encoding bytes per argument occurrence.
182    ///
183    /// Repeated terms are counted repeatedly. This is neither unique storage
184    /// nor physical memory and is excluded from `retained_bytes`. The former
185    /// `referenced_payload_bytes` field measured owned source heap payload;
186    /// historical observations under that name are not this encoding measure.
187    pub referenced_encoding_bytes: u128,
188    /// Borrowed source catalog-index slice bytes; zero for contiguous rows.
189    pub borrowed_mapping_bytes: usize,
190    /// Charged construction work; payload comparisons are not unit-cost integers.
191    pub construction_work: u128,
192}

194/// One immutable execution view of a signed predicate relation.
195///
196/// The dictionary refers to whole typed source terms; columns hold equality IDs.
197/// Borrowed-source construction owns the layout. A catalog view borrows it.
198/// Local row positions preserve input occurrence order. Catalog indices remain
199/// separately accessible through [`Row::source_index`]. Identity is this live
200/// owner, not its address retained after destruction, contents or dimensions.
201pub struct Relation<'source> {
202    predicate: PredicateRef<'source>,
203    source: Source<'source>,
204    layout: LayoutOwner<'source>,
205    storage: Storage,
206}

208/// One equality layout shared by borrowed-source construction and owned catalogs.
209/// Dictionary representatives are source positions, never duplicated values.

221#[derive(Clone, Copy)]
222struct Cell {
223    row: usize,
224    column: usize,
225}

227impl Cell {
228    fn value<'source>(self, source: &Source<'source>) -> Result<TermRef<'source>, Failure> {
229        source
230            .atom(self.row)
231            .and_then(|atom| atom.values().get(self.column))
232            .ok_or(Failure::CatalogIndex)
233    }
234}

236enum LayoutOwner<'source> {
237    Owned(Layout),
238    Borrowed(&'source Layout),
239}
240
241impl std::ops::Deref for LayoutOwner<'_> {
242    type Target = Layout;
243    fn deref(&self) -> &Layout {
244        match self {
245            Self::Owned(layout) => layout,
246            Self::Borrowed(layout) => layout,
247        }
248    }

251enum Source<'source> {
252    Atoms(&'source [Atom]),
253    Catalog {
254        atoms: &'source [Atom],
255        indices: &'source [usize],
256    },
257    Canonical(Atoms<'source>),
258    CanonicalCatalog {
259        atoms: Atoms<'source>,
260        indices: &'source [usize],
261    },
262}
263
264impl<'source> Source<'source> {
265    fn read(&self) -> Option<crate::catalog::CatalogRead<'source>> {
266        match self {
267            Self::Canonical(atoms) | Self::CanonicalCatalog { atoms, .. } => Some(atoms.read()),
268            Self::Atoms(_) | Self::Catalog { .. } => None,
269        }
270    }
271
272    fn len(&self) -> usize {
273        match self {
274            Self::Atoms(atoms) => atoms.len(),
275            Self::Canonical(atoms) => atoms.len(),
276            Self::Catalog { indices, .. } | Self::CanonicalCatalog { indices, .. } => indices.len(),
277        }
278    }
279
280    fn index(&self, row: usize) -> Option<usize> {
281        match self {
282            Self::Atoms(atoms) => (row < atoms.len()).then_some(row),
283            Self::Canonical(atoms) => (row < atoms.len()).then_some(row),
284            Self::Catalog { indices, .. } | Self::CanonicalCatalog { indices, .. } => {
285                indices.get(row).copied()
286            }
287        }
288    }
289
290    fn atom(&self, row: usize) -> Option<AtomRef<'source>> {
291        let index = self.index(row)?;
292        match self {
293            Self::Atoms(atoms) | Self::Catalog { atoms, .. } => atoms.get(index).map(AtomRef::from),
294            Self::Canonical(atoms) | Self::CanonicalCatalog { atoms, .. } => atoms.at(index),
295        }
296    }
297}

299impl<'source> Relation<'source> {
300    /// Build columns without cloning atoms or their logical payload.
301    ///
302    /// The explicit predicate also identifies empty/nullary relations. Duplicate
303    /// atom occurrences are retained. Construction uses O(c log c + c log d)
304    /// typed comparisons for c argument cells and d distinct values, plus their
305    /// payload costs. Temporary sort/reference capacity is charged.
306    ///
307    /// # Errors
308    /// Refuses foreign predicates, exceeded limits, unrepresentable shapes or
309    /// allocation failure. No partial relation is returned.
310    pub fn from_atoms(
311        predicate: &'source Predicate,
312        atoms: &'source [Atom],
313        limits: Limits,
314    ) -> Result<Self, Failure> {
315        storage::build(PredicateRef::from(predicate), Source::Atoms(atoms), limits)
316    }
317
318    /// Build an ordered predicate view over explicit original catalog indices.
319    ///
320    /// Indices are borrowed and checked; arbitrary input order and repeated
321    /// indices remain distinct row occurrences. This supplies a legitimate
322    /// original-ID view without renumbering the catalog. Costs match
323    /// [`Self::from_atoms`], with an additional borrowed index per row.
324    ///
325    /// # Errors
326    /// Also refuses out-of-range catalog indices. All selected atoms must have
327    /// the supplied signed predicate.
328    pub fn from_catalog(
329        predicate: &'source Predicate,
330        atoms: &'source [Atom],
331        indices: &'source [usize],
332        limits: Limits,
333    ) -> Result<Self, Failure> {
334        storage::build(
335            PredicateRef::from(predicate),
336            Source::Catalog { atoms, indices },
337            limits,
338        )
339    }
340
341    /// Build columns over canonical atom occurrences without copying terms.
342    ///
343    /// The explicit predicate identifies empty/nullary relations. Construction
344    /// and occurrence semantics match [`Self::from_atoms`]; comparisons also
345    /// charge canonical descriptor/navigation probes before they occur.
346    ///
347    /// # Errors
348    /// Refuses foreign predicates, exceeded limits, unrepresentable shapes or
349    /// allocation failure without publishing a partial relation.
350    pub fn from_refs(
351        predicate: PredicateRef<'source>,
352        atoms: Atoms<'source>,
353        limits: Limits,
354    ) -> Result<Self, Failure> {
355        storage::build(predicate, Source::Canonical(atoms), limits)
356    }
357
358    /// Build columns over checked canonical occurrence indices.
359    ///
360    /// Arbitrary input order and repeated indices remain distinct occurrences.
361    /// Construction shares [`Self::from_catalog`]'s checked layout operation.
362    ///
363    /// # Errors
364    /// Also refuses out-of-range indices before publishing a relation.
365    pub fn from_catalog_refs(
366        predicate: PredicateRef<'source>,
367        atoms: Atoms<'source>,
368        indices: &'source [usize],
369        limits: Limits,
370    ) -> Result<Self, Failure> {
371        storage::build(
372            predicate,
373            Source::CanonicalCatalog { atoms, indices },
374            limits,
375        )
376    }
377
378    /// The full signed predicate. Constant-time borrowed view.
379    #[must_use]
380    pub const fn predicate(&self) -> PredicateRef<'source> {
381        self.predicate
382    }
383
384    /// Number of row occurrences, independent of arity. Constant time.
385    #[must_use]
386    pub fn row_count(&self) -> usize {
387        self.source.len()
388    }
389
390    /// Charged construction and retained-storage observations. Constant time.
391    #[must_use]
392    pub const fn storage(&self) -> Storage {
393        self.storage
394    }
395
396    /// Borrow one contiguous equality-ID column in original row order.
397    ///
398    /// IDs are meaningful only with this relation's dictionary. They must not
399    /// be interpreted as numbers, term-order ranks or candidate truth.
400    #[must_use]
401    pub fn column(&self, column: usize) -> Option<&[u32]> {
402        if column >= self.predicate.arity() {
403            return None;
404        }
405        self.layout.columns.get(column).map(Vec::as_slice)
406    }
407
408    /// Borrow columns in original argument order without packing or allocation.
409    ///
410    /// Exactly arity slices are returned, each containing `row_count` IDs.
411    /// Nullary relations yield no slices. A device consumer may copy these
412    /// slices consecutively into its admitted column-major upload buffer.
413    #[must_use]
414    pub fn columns(&self) -> impl ExactSizeIterator<Item = &[u32]> {
415        self.layout.columns.iter().map(Vec::as_slice)
416    }
417
418    /// Access a typed row occurrence without allocating or cloning an atom.
419    #[must_use]
420    pub fn row(&self, position: usize) -> Option<Row<'_, 'source>> {
421        self.source.index(position).map(|source_index| Row {
422            relation: self,
423            position,
424            source_index,
425        })
426    }
427
428    /// Whether two live borrows identify exactly the same immutable owner.
429    ///
430    /// Equal rows, dimensions, predicates or overlapping lifetimes are not
431    /// sufficient. No token can outlive the owner.
432    #[must_use]
433    pub fn same_owner(&self, other: &Self) -> bool {
434        std::ptr::eq(self, other)
435    }
436
437    fn work(&self, limits: Limits, additional: usize) -> Result<Work, Failure> {
438        ceiling(
439            Resource::Rows,
440            self.row_count() as u128,
441            limits.max_rows as u128,
442        )?;
443        ceiling(
444            Resource::Columns,
445            self.predicate.arity() as u128,
446            limits.max_columns as u128,
447        )?;
448        ceiling(
449            Resource::Values,
450            self.layout.dictionary.len() as u128,
451            limits.max_values as u128,
452        )?;
453        Work::new(
454            limits,
455            self.storage.retained_bytes as u128 + additional as u128,
456        )
457    }
458}

460/// Borrowed typed row access, separate from an owned `Atom` representation.
461#[derive(Clone, Copy)]
462pub struct Row<'owner, 'source> {
463    relation: &'owner Relation<'source>,
464    position: usize,
465    source_index: usize,
466}
467
468impl<'source> Row<'_, 'source> {
469    /// Borrow this complete source occurrence without copying its arguments.
470    /// The row's checked position resolves through the relation's original
471    /// occurrence map; dictionary order is not atom identity.
472    ///
473    /// # Panics
474    /// Panics if the relation's admitted occurrence map refers outside its source.
475    /// Relation construction checks this map, which cannot change through a row.
476    #[must_use]
477    pub fn atom(&self) -> AtomRef<'source> {
478        self.relation
479            .source
480            .atom(self.position)
481            .expect("a relation row refers to its admitted source occurrence")
482    }
483
484    /// The full signed predicate. Constant time.
485    #[must_use]
486    pub fn predicate(&self) -> PredicateRef<'source> {
487        self.relation.predicate()
488    }
489
490    /// Original occurrence position within this relation. Constant time.
491    #[must_use]
492    pub const fn position(&self) -> usize {
493        self.position
494    }
495
496    /// Index in the borrowed source atom slice/catalog. Constant time.
497    ///
498    /// This is distinct from local position for catalog views. Its meaning
499    /// depends on the source catalog used to construct the relation.
500    #[must_use]
501    pub const fn source_index(&self) -> usize {
502        self.source_index
503    }
504
505    /// Locate this occurrence in its exact canonical source mapping. A mapped
506    /// relation returns the original source position, not its local row number.
507    /// An independent equal catalog or owned ingress is not that source.
508    /// Constant time; no atom or predicate comparison is needed.
509    #[must_use]
510    pub fn occurrence_in(&self, atoms: Atoms<'_>) -> Option<usize> {
511        match &self.relation.source {
512            Source::Canonical(source) | Source::CanonicalCatalog { atoms: source, .. }
513                if source.same_occurrences(atoms) =>
514            {
515                Some(self.source_index)
516            }
517            _ => None,
518        }
519    }
520
521    /// Decode one argument to an equal borrowed dictionary representative.
522    ///
523    /// Interning preserves typed value equality, not the address of the original
524    /// occurrence's ingress cell. [`Self::source_index`] retains its catalog identity.
525    /// The value borrows the source and can outlive this row and relation view.
526    #[must_use]
527    pub fn value(&self, column: usize) -> Option<TermRef<'source>> {
528        let id = *self.relation.column(column)?.get(self.position)?;
529        self.relation
530            .layout
531            .dictionary
532            .get(id as usize)?
533            .value(&self.relation.source)
534            .ok()
535    }
536}

538fn ceiling(resource: Resource, observed: u128, limit: u128) -> Result<(), Failure> {
539    if observed > limit {
540        Err(Failure::Limit {
541            resource,
542            observed,
543            limit,
544        })
545    } else {
546        Ok(())
547    }
548}

550struct Work {
551    limits: Limits,
552    used: u128,
553    live: usize,
554    peak: usize,
555}
556
557impl Work {
558    fn new(limits: Limits, bytes: u128) -> Result<Self, Failure> {
559        ceiling(Resource::Bytes, bytes, limits.max_bytes as u128)?;
560        let live = usize::try_from(bytes).map_err(|_| Failure::Overflow)?;
561        Ok(Self {
562            limits,
563            used: 0,
564            live,
565            peak: live,
566        })
567    }
568
569    fn tick(&mut self, amount: u128) -> Result<(), Failure> {
570        let next = self.used.checked_add(amount).ok_or(Failure::Overflow)?;
571        ceiling(Resource::Work, next, u128::from(self.limits.max_work))?;
572        self.used = next;
573        Ok(())
574    }
575
576    fn compare(
577        &mut self,
578        left: TermRef<'_>,
579        right: TermRef<'_>,
580    ) -> Result<std::cmp::Ordering, Failure> {
581        left.compare_ref_with(right, || self.tick(1))
582    }
583
584    fn include(&mut self, bytes: usize) -> Result<(), Failure> {
585        let next = self.live.checked_add(bytes).ok_or(Failure::Overflow)?;
586        ceiling(Resource::Bytes, next as u128, self.limits.max_bytes as u128)?;
587        self.live = next;
588        self.peak = self.peak.max(next);
589        Ok(())
590    }
591
592    fn reserve<T>(&mut self, count: usize) -> Result<Vec<T>, Failure> {
593        let requested = self.live as u128 + count as u128 * size_of::<T>() as u128;
594        ceiling(Resource::Bytes, requested, self.limits.max_bytes as u128)?;
595        let mut values = Vec::new();
596        values
597            .try_reserve_exact(count)
598            .map_err(|_| Failure::Allocation)?;
599        let actual = self.live as u128 + values.capacity() as u128 * size_of::<T>() as u128;
600        self.live = usize::try_from(actual).map_err(|_| Failure::Overflow)?;
601        self.peak = self.peak.max(self.live);
602        ceiling(Resource::Bytes, actual, self.limits.max_bytes as u128)?;
603        Ok(values)
604    }
605
606    fn grow<T>(&mut self, values: &mut Vec<T>, additional: usize) -> Result<(), Failure> {
607        let needed = values
608            .len()
609            .checked_add(additional)
610            .ok_or(Failure::Overflow)?;
611        if needed <= values.capacity() {
612            return Ok(());
613        }
614        let previous = values.capacity();
615        let proposed = needed.max(previous.saturating_mul(2));
616        ceiling(
617            Resource::Bytes,
618            self.live as u128 + proposed as u128 * size_of::<T>() as u128,
619            self.limits.max_bytes as u128,
620        )?;
621        self.tick(values.len() as u128)?;
622        values
623            .try_reserve_exact(proposed - values.len())
624            .map_err(|_| Failure::Allocation)?;
625        // Reservation may transiently retain both old and replacement buffers.
626        // Record actual slack even when that post-allocation envelope is refused.
627        self.replacement(
628            previous * size_of::<T>(),
629            values.capacity() * size_of::<T>(),
630        )
631    }
632
633    /// Observe actual replacement capacity, including old/new overlap before
634    /// any post-allocation refusal. The owner's retained capacity already changed.
635    fn replacement(&mut self, previous: usize, current: usize) -> Result<(), Failure> {
636        let overlap = self.live as u128 + current as u128;
637        let actual = self.live as u128 + (current - previous) as u128;
638        self.live = usize::try_from(actual).map_err(|_| Failure::Overflow)?;
639        self.peak = self
640            .peak
641            .max(usize::try_from(overlap).map_err(|_| Failure::Overflow)?);
642        ceiling(Resource::Bytes, overlap, self.limits.max_bytes as u128)?;
643        Ok(())
644    }
645
646    fn release<T>(&mut self, values: Vec<T>) {
647        self.live -= values.capacity() * size_of::<T>();
648        drop(values);
649    }
650}
