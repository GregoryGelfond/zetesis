1//! An exclusive append session retains one tuple transaction's metadata.
2//!
3//! Every call commits at most one row. Earlier successful calls remain published
4//! after a later refusal; dropping the session neither commits nor rolls back.
5
6use std::mem::size_of;
7
8use crate::catalog::AtomRef;
9
10use super::{Catalog, CatalogFailure, Insertion, Limits, plan};
11
12/// Reusable insertion preparation over one existing relation catalog.
13///
14/// This owns only transaction metadata and exclusively borrows the catalog.
15/// It retains no canonical payload or second identity index. Tuple lookup and
16/// publication use the same engine as [`Catalog::insert`]. Work remains charged
17/// per insertion; reusing allocations does not erase comparisons or writes.
18/// Dropping the appender frees scratch but preserves every completed insertion.
19pub struct Appender<'catalog> {
20    catalog: &'catalog mut Catalog,
21    plan: plan::Plan,
22}
23
24/// One complete insertion and its temporarily borrowed equality coordinates.
25///
26/// The borrow prevents another session insertion until these coordinates have
27/// been consumed. Equality IDs belong to this catalog, never to another owner.
28#[derive(Debug)]
29pub struct Appended<'step> {
30    /// Same stable row identity and insertion flag as scalar insertion.
31    pub insertion: Insertion,
32    /// The newly published row's columns, in argument order. A duplicate has
33    /// no new columns; a newly inserted nullary tuple has `Some(&[])`.
34    pub equality_ids: Option<&'step [u32]>,
35    /// Session header and actual scratch capacity, excluding the catalog.
36    pub scratch_bytes: usize,
37}
38
39impl Catalog {
40    /// Borrow this catalog for repeated checked, individually atomic insertion.
41    ///
42    /// The session retains its tuple-plan vectors between calls. `limits`
43    /// include both the catalog and the session header. Each subsequent call
44    /// admits its current complete envelope against its supplied limits, so a
45    /// caller can account for other owners growing between insertions.
46    ///
47    /// # Errors
48    /// Refuses current population or byte limits before creating the session.
49    /// No row, equality ID or retained catalog capacity changes.
50    pub fn appender(&mut self, limits: Limits) -> Result<Appender<'_>, CatalogFailure> {
51        self.work_with(limits, size_of::<Appender<'_>>())
52            .map_err(|mut error| {
53                // The proposed session has no retained header on refusal.
54                error.peak_construction_bytes = error.retained_bytes;
55                error
56            })?;
57        Ok(Appender {
58            catalog: self,
59            plan: plan::Plan::default(),
60        })
61    }
62}
63
64impl Appender<'_> {
65    /// Current session header and scratch capacities, excluding catalog storage.
66    /// Empty logical scratch retains its allocation, including after refusal.
67    #[must_use]
68    pub fn scratch_bytes(&self) -> usize {
69        size_of::<Self>() + self.plan.capacity_bytes()
70    }
71
72    /// Insert one atom using the existing canonical owner and transaction engine.
73    ///
74    /// Limits and operation peak include catalog plus session scratch and actual
75    /// growth overlap. The result's retained catalog bytes exclude this session;
76    /// `scratch_bytes` reports its separate retained component. Work is for this
77    /// call only. Repeated calls must receive the caller's remaining work budget.
78    ///
79    /// # Errors
80    /// Returns the same typed owner, shape, allocation, overflow and limit errors
81    /// as scalar insertion. Only this call's logical insertion is atomic: earlier
82    /// successful rows remain present. Catalog and scratch reservations may remain
83    /// after refusal. Inspect [`Self::scratch_bytes`] on an error; the error's
84    /// retained bytes name only the catalog and its peak includes both owners.
85    pub fn insert(
86        &mut self,
87        atom: AtomRef<'_>,
88        limits: Limits,
89    ) -> Result<Appended<'_>, CatalogFailure> {
90        let mut work = match self.catalog.work_with(limits, self.scratch_bytes()) {
91            Ok(work) => work,
92            Err(error) => {
93                self.plan.clear();
94                return Err(error);
95            }
96        };
97        let insertion = match self.catalog.insert_inner(
98            atom,
99            &mut self.plan,
100            plan::Admission::Retained,
101            &mut work,
102        ) {
103            Ok(insertion) => insertion,
104            Err(error) => {
105                self.plan.clear();
106                return Err(self.catalog.failed(error, &work));
107            }
108        };
109        Ok(Appended {
110            insertion,
111            equality_ids: insertion.inserted.then_some(self.plan.ids.as_slice()),
112            scratch_bytes: self.scratch_bytes(),
113        })
114    }
115}
