12/// The complete canonical order of a prepared extent, with the merge's receipt.
13pub struct Canonical {
14    /// Every row ID once, in canonical order.
15    pub ids: Vec<usize>,
16    /// Capacity and charged work of the merge that produced it.
17    pub storage: Storage,
18}

20/// The prepared canonical order of one extent as sorted runs of row IDs.
21///
22/// The runs are the levels, each in canonical order and each less than half
23/// the length of the level before it, followed by the tail: the run merged in
24/// by the last preparation that appended anything. A row belongs to exactly
25/// one run. Every run is a sorted sequence, so a bound-prefix window is one
26/// binary search per run; the runs together are not one sequence, and rank
27/// access across them needs [`Catalog::canonical`]. The borrow prevents
28/// append. Row IDs are stable catalog-local insertion IDs.
29#[derive(Clone, Copy)]
30pub struct Runs<'a> {
31    levels: &'a [Vec<usize>],
32    tail: &'a [usize],
33}
34
35/// The runs a preparation published, with that preparation's receipt.
36pub struct Preparation<'a> {
37    /// The prepared runs.
38    pub runs: Runs<'a>,
39    /// Current owner capacity and the work of this preparation or reuse.
40    pub storage: Storage,
41}
42
43impl<'a> Runs<'a> {
44    /// Number of rows across every run. Linear in the number of runs.
45    #[must_use]
46    pub fn len(self) -> usize {
47        self.levels.iter().map(Vec::len).sum::<usize>() + self.tail.len()
48    }
49
50    /// Whether no run holds a row.
51    #[must_use]
52    pub fn is_empty(self) -> bool {
53        self.len() == 0
54    }
55
56    /// The levels, oldest first: the rows present before the last appending
57    /// preparation, in geometrically shrinking sorted runs.
58    #[must_use]
59    pub fn levels(self) -> &'a [Vec<usize>] {
60        self.levels
61    }
62
63    /// The rows the last appending preparation merged in, in canonical order.
64    /// After the first preparation of a nonempty extent this is every row.
65    #[must_use]
66    pub fn tail(self) -> &'a [usize] {
67        self.tail
68    }
69
70    /// Every run, levels first and then the tail; empty runs are omitted.
71    pub fn runs(self) -> impl Iterator<Item = &'a [usize]> {
72        self.levels
73            .iter()
74            .map(Vec::as_slice)
75            .chain(std::iter::once(self.tail))
76            .filter(|run| !run.is_empty())
77    }
78}

80impl Catalog {
81    /// Prepare ordered row access without cloning tuple payload.
82    ///
83    /// A prepared extent is reused with zero traversal work. The first
84    /// preparation of a nonempty extent traverses the row index once, O(n),
85    /// into the tail. A later preparation first promotes the previous tail to
86    /// a level and merges levels until each is less than half its
87    /// predecessor, then sorts the `d` rows appended since into the new tail.
88    /// Merging two runs of `a` and `b` rows costs `a + b` charged comparisons
89    /// and copies; over a whole derivation each row is merged O(log n) times,
90    /// so the views cost O(n log n) charged work in all, and no preparation
91    /// copies the whole extent. Sorting the appended rows costs O(d log d)
92    /// comparisons. The levels and tail are this owner's retained capacity; a
93    /// merge holds one scratch run of the merged length until it replaces its
94    /// two inputs. Only successful insertion invalidates the runs; a refused
95    /// insertion or a refused preparation leaves every published run whole.
96    ///
97    /// # Errors
98    /// Current shape, bytes and work must fit `limits`. Refused preparation does
99    /// not publish a partial run or change atom/equality identity.
100    pub fn prepare_ordered(
101        &mut self,
102        read: CatalogRead<'_>,
103        limits: Limits,
104    ) -> Result<Preparation<'_>, CatalogFailure> {
105        let mut work = self.work(limits)?;
106        let atoms = self
107            .membership
108            .bind(read)
109            .map_err(|error| self.failed(Failure::Read(error), &work))?;
110        self.ordered
111            .prepare(atoms, &self.rows, &mut work)
112            .map_err(|error| self.failed(error, &work))?;
113        Ok(Preparation {
114            storage: self.receipt(&work),
115            runs: self
116                .ordered
117                .runs(self.len())
118                .ok_or_else(|| self.failed(Failure::Order, &work))?,
119        })
120    }
121
122    /// Borrow already prepared runs without allocating, traversing rows or
123    /// measuring capacity: constant time, fit for every join probe.
124    /// `None` means a successful append requires a new preparation. It does not
125    /// mean this relation has no tuples. A new empty catalog is already prepared.
126    #[must_use]
127    pub fn ordered(&self) -> Option<Runs<'_>> {
128        self.ordered.runs(self.len())
129    }
130
131    /// The complete canonical order of a prepared extent as one sequence of
132    /// row IDs, merged from the runs, into a new vector the caller owns: each
133    /// of the `n` rows is copied after a scan of the `k` run heads for the
134    /// least, `O(n·k)` charged comparisons, the runs being few (one per level
135    /// and the tail). The run slices and cursors are reserved through the
136    /// work beside the merged ids. This is the rank access the runs
137    /// themselves do not offer.
138    ///
139    /// # Errors
140    /// Refuses an unprepared extent as [`Failure::Order`], or work and bytes
141    /// above `limits`.
142    pub fn canonical(
143        &self,
144        read: CatalogRead<'_>,
145        limits: Limits,
146    ) -> Result<Canonical, CatalogFailure> {
147        let mut work = self.work(limits)?;
148        let atoms = self
149            .atoms(read)
150            .map_err(|error| self.failed(error, &work))?;
151        let ids = self
152            .ordered
153            .canonical(atoms, &mut work)
154            .map_err(|error| self.failed(error, &work))?;
155        Ok(Canonical {
156            ids,
157            storage: self.receipt(&work),
158        })
159    }
160}

162/// Geometric runs are derived local-row order, never another payload owner.
163#[derive(Default)]
164pub(super) struct Ordered {
165    levels: Vec<Vec<usize>>,
166    tail: Vec<usize>,
167    prepared: usize,
168}
169
170impl Ordered {
171    pub(super) fn capacity_bytes(&self) -> usize {
172        self.levels.capacity() * size_of::<Vec<usize>>()
173            + (self.levels.iter().map(Vec::capacity).sum::<usize>() + self.tail.capacity())
174                * size_of::<usize>()
175    }
176
177    pub(super) fn level_count(&self) -> usize {
178        self.levels.len()
179    }
180
181    pub(super) fn clear(&mut self) {
182        self.levels.clear();
183        self.tail.clear();
184        self.prepared = 0;
185    }
186
187    fn runs(&self, length: usize) -> Option<Runs<'_>> {
188        (self.prepared == length).then_some(Runs {
189            levels: &self.levels,
190            tail: &self.tail,
191        })
192    }
193
194    fn prepare(&mut self, atoms: Atoms<'_>, rows: &Index, work: &mut Work) -> Result<(), Failure> {
195        if self.prepared != atoms.len() {
196            if self.levels.is_empty() && self.tail.is_empty() {
197                self.traverse(rows, atoms.len(), work)
198                    .inspect_err(|_| self.tail.clear())?;
199            } else {
200                self.append_run(atoms, work)?;
201            }
202            self.prepared = atoms.len();
203        }
204        Ok(())
205    }
206
207    fn canonical(&self, atoms: Atoms<'_>, work: &mut Work) -> Result<Vec<usize>, Failure> {
208        let Some(prepared) = self.runs(atoms.len()) else {
209            return Err(Failure::Order);
210        };
211        let mut merged = work.reserve::<usize>(atoms.len())?;
212        let mut runs = work.reserve::<&[usize]>(prepared.runs().count())?;
213        runs.extend(prepared.runs());
214        let mut cursors = work.reserve::<usize>(runs.len())?;
215        cursors.resize(runs.len(), 0);
216        // The least head among the runs, until every run is exhausted:
217        // the runs hold every prepared row once, so the merge then holds
218        // them all.
219        loop {
220            let mut least: Option<usize> = None;
221            for (run, &cursor) in cursors.iter().enumerate() {
222                if cursor == runs[run].len() {
223                    continue;
224                }
225                least = Some(match least {
226                    Some(best)
227                        if order(atoms, runs[best][cursors[best]], runs[run][cursor], work)?
228                            .is_le() =>
229                    {
230                        best
231                    }
232                    _ => run,
233                });
234            }
235            let Some(run) = least else {
236                break;
237            };
238            work.tick(1)?;
239            merged.push(runs[run][cursors[run]]);
240            cursors[run] += 1;
241        }
242        Ok(merged)
243    }
244
245    /// Build the first run by an in-order traversal of the row index.
246    fn traverse(&mut self, rows: &Index, length: usize, work: &mut Work) -> Result<(), Failure> {
247        work.grow(&mut self.tail, length)?;
248        work.include(size_of::<Vec<usize>>())?;
249        let mut path = work.reserve::<usize>(0)?;
250        self.tail.clear();
251        let mut cursor = rows.root;
252        loop {
253            while let Some(link) = cursor {
254                work.tick(1)?;
255                let id = position(link);
256                work.grow(&mut path, 1)?;
257                work.tick(1)?;
258                path.push(id);
259                cursor = rows.nodes[id].children[0];
260            }
261            let Some(id) = path.pop() else {
262                break;
263            };
264            work.tick(1)?;
265            self.tail.push(id);
266            cursor = rows.nodes[id].children[1];
267        }
268        work.release(path);
269        work.live -= size_of::<Vec<usize>>();
270        Ok(())
271    }
272
273    /// Promote the previous tail to a level, compact the levels, and sort the
274    /// rows appended since the last preparation into the new tail. Each step
275    /// publishes only complete runs, so a refusal at any point leaves the
276    /// levels whole and the extent unprepared.
277    fn append_run(&mut self, atoms: Atoms<'_>, work: &mut Work) -> Result<(), Failure> {
278        if !self.tail.is_empty() {
279            // Admit the level slot before the move, so a refusal leaves the
280            // run where it was.
281            work.grow(&mut self.levels, 1)?;
282            work.include(size_of::<Vec<usize>>())?;
283            let promoted = std::mem::take(&mut self.tail);
284            self.levels.push(promoted);
285        }
286        // A refused compaction may already have promoted the tail or merged
287        // some levels. Resume that phase even when the tail is now empty;
288        // only fully compacted levels may accompany a newly published tail.
289        self.compact(atoms, work)?;
290        let appended = self.prepared..atoms.len();
291        let mut run = work.reserve::<usize>(appended.len())?;
292        run.extend(appended);
293        work.tick(run.len() as u128)?;
294        let mut scratch = work.reserve::<usize>(run.len())?;
295        Self::sort(atoms, &mut run, &mut scratch, work)?;
296        work.release(scratch);
297        work.release(std::mem::replace(&mut self.tail, run));
298        Ok(())
299    }
300
301    /// Merge the top two levels while the newer is at least half the older,
302    /// so the levels shrink geometrically and their count stays O(log n).
303    fn compact(&mut self, atoms: Atoms<'_>, work: &mut Work) -> Result<(), Failure> {
304        while let [.., below, top] = &self.levels[..] {
305            if top.len() * 2 < below.len() {
306                return Ok(());
307            }
308            let mut merged = work.reserve::<usize>(below.len() + top.len())?;
309            let (mut left, mut right) = (0, 0);
310            while left < below.len() && right < top.len() {
311                if order(atoms, top[right], below[left], work)?.is_lt() {
312                    merged.push(top[right]);
313                    right += 1;
314                } else {
315                    merged.push(below[left]);
316                    left += 1;
317                }
318            }
319            merged.extend_from_slice(&below[left..]);
320            merged.extend_from_slice(&top[right..]);
321            work.tick(merged.len() as u128)?;
322            let top = self.levels.pop().expect("two levels");
323            let below = self.levels.pop().expect("two levels");
324            work.release(top);
325            work.release(below);
326            work.live -= size_of::<Vec<usize>>();
327            self.levels.push(merged);
328        }
329        Ok(())
330    }
331
332    /// Bottom-up merge sort of a run by typed row comparison; `scratch` has
333    /// capacity for the whole run. O(d log d) comparisons, each charged.
334    fn sort(
335        atoms: Atoms<'_>,
336        run: &mut Vec<usize>,
337        scratch: &mut Vec<usize>,
338        work: &mut Work,
339    ) -> Result<(), Failure> {
340        let length = run.len();
341        let mut width = 1;
342        while width < length {
343            scratch.clear();
344            let mut start = 0;
345            while start < length {
346                let middle = (start + width).min(length);
347                let end = (start + 2 * width).min(length);
348                let (mut left, mut right) = (start, middle);
349                while left < middle && right < end {
350                    if order(atoms, run[right], run[left], work)?.is_lt() {
351                        scratch.push(run[right]);
352                        right += 1;
353                    } else {
354                        scratch.push(run[left]);
355                        left += 1;
356                    }
357                }
358                scratch.extend_from_slice(&run[left..middle]);
359                scratch.extend_from_slice(&run[right..end]);
360                start = end;
361            }
362            work.tick(length as u128)?;
363            std::mem::swap(run, scratch);
364            width *= 2;
365        }
366        Ok(())
367    }
368}

370/// Typed identity order of two rows, charged per compared value.
371fn order(
372    atoms: Atoms<'_>,
373    left: usize,
374    right: usize,
375    work: &mut Work,
376) -> Result<std::cmp::Ordering, Failure> {
377    work.tick(1)?;
378    let left = atoms.at(left).ok_or(Failure::CatalogIndex)?;
379    let right = atoms.at(right).ok_or(Failure::CatalogIndex)?;
380    for (left, right) in left.values().iter().zip(right.values()) {
381        let order = work.compare(left, right)?;
382        if !order.is_eq() {
383            return Ok(order);
384        }
385    }
386    Ok(std::cmp::Ordering::Equal)
