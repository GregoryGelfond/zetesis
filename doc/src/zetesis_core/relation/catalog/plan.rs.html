1//! A one-tuple transaction containing only equality IDs and tentative AVL nodes.
2//!
3//! Later values inspect the latest tentative patch before the published tree.
4//! No row, dictionary representative or tree link changes before final admission.
5//! With a new values and height h, retained patches use O(a h) cells. Their linear
6//! reverse lookup can cost O(a² h²) metadata probes; each probe is charged. This
7//! depends on this tuple's width, never on shifting the historical dictionary.
8

34/// Temporary scalar scratch is admitted only after a novel row is established.
35/// A session includes its retained header and capacities at operation entry.
36#[derive(Clone, Copy)]
37pub(super) enum Admission {
38    Temporary,
39    Retained,
40}

42impl Plan {
43    pub(super) fn capacity_bytes(&self) -> usize {
44        super::vector_bytes(&self.ids)
45            + super::vector_bytes(&self.added)
46            + super::vector_bytes(&self.patches)
47    }
48
49    pub(super) fn clear(&mut self) {
50        self.ids.clear();
51        self.added.clear();
52        self.patches.clear();
53        self.root = None;
54        self.encoding_bytes = 0;
55    }
56
57    /// Only this tuple's unpublished distinct values are scanned. Historical
58    /// dictionary hits use the complete canonical inverse instead.
59    fn staged(
60        &self,
61        term: TermId,
62        start: usize,
63        work: &mut Work,
64    ) -> Result<Option<usize>, Failure> {
65        for (offset, added) in self.added.iter().enumerate() {
66            work.tick(1)?;
67            if added.term == term {
68                return start.checked_add(offset).map(Some).ok_or(Failure::Overflow);
69            }
70        }
71        Ok(None)
72    }
73
74    fn node(&self, nodes: &[Node], id: usize, work: &mut Work) -> Result<Node, Failure> {
75        for patch in self.patches.iter().rev() {
76            work.tick(1)?;
77            if patch.id == id {
78                return Ok(patch.node);
79            }
80        }
81        work.tick(1)?;
82        nodes.get(id).copied().ok_or(Failure::Dictionary)
83    }
84
85    fn value<'a>(
86        &self,
87        dictionary: &[Cell],
88        atoms: Atoms<'a>,
89        atom: AtomRef<'a>,
90        id: usize,
91    ) -> Result<TermRef<'a>, Failure> {
92        let cell = if let Some(cell) = dictionary.get(id) {
93            return atoms
94                .at(cell.row)
95                .and_then(|atom| atom.values().get(cell.column))
96                .ok_or(Failure::Dictionary);
97        } else {
98            self.added
99                .get(id - dictionary.len())
100                .map(|added| &added.cell)
101                .ok_or(Failure::Dictionary)?
102        };
103        atom.values().get(cell.column).ok_or(Failure::Dictionary)
104    }
105
106    fn locate(
107        &self,
108        index: &Index,
109        dictionary: &[Cell],
110        atoms: Atoms<'_>,
111        atom: AtomRef<'_>,
112        value: TermRef<'_>,
113        work: &mut Work,
114    ) -> Result<(Option<usize>, Directions), Failure> {
115        let mut route = Directions::default();
116        let mut cursor = self.root;
117        while let Some(link) = cursor {
118            work.tick(1)?;
119            let id = position(link);
120            let order = work.compare(value, self.value(dictionary, atoms, atom, id)?)?;
121            if order.is_eq() {
122                return Ok((Some(id), route));
123            }
124            let right = order == Ordering::Greater;
125            route.push(right).ok_or(Failure::Dictionary)?;
126            cursor = self.node(&index.nodes, id, work)?.children[usize::from(right)];
127        }
128        Ok((None, route))
129    }
130
131    fn extend(
132        &mut self,
133        index: &mut Index,
134        id: usize,
135        route: &Directions,
136        work: &mut Work,
137    ) -> Result<(), Failure> {
138        index.path.clear();
139        let mut cursor = self.root;
140        for offset in 0..route.len() {
141            work.tick(1)?;
142            let right = route.get(offset).ok_or(Failure::Dictionary)?;
143            let previous = position(cursor.ok_or(Failure::Dictionary)?);
144            let node = self.node(&index.nodes, previous, work)?;
145            push(
146                index,
147                Step {
148                    id: previous,
149                    node,
150                    right,
151                    changed: false,
152                },
153                work,
154            )?;
155            cursor = node.children[usize::from(right)];
156        }
157        if cursor.is_some() {
158            return Err(Failure::Dictionary);
159        }
160        push(
161            index,
162            Step {
163                id,
164                node: Node::default(),
165                right: false,
166                changed: true,
167            },
168            work,
169        )?;
170        let root = index.plan_from(self.root, id, &mut || work.tick(1))?;
171        work.tick(index.path.len() as u128)?;
172        let changed = index.path.iter().filter(|step| step.changed).count();
173        work.grow(&mut self.patches, changed)?;
174        for step in &index.path {
175            work.tick(1)?;
176            if step.changed {
177                self.patches.push(*step);
178            }
179        }
180        self.root = root;
181        Ok(())
182    }
183}

185pub(super) fn values(
186    layout: &mut Layout,
187    atoms: Atoms<'_>,
188    atom: AtomRef<'_>,
189    encoding_bytes: u128,
190    plan: &mut Plan,
191    work: &mut Work,
192) -> Result<(), Failure> {
193    let Layout {
194        dictionary, index, ..
195    } = layout;
196    let DictionaryIndex::Append(index) = index else {
197        return Err(Failure::Dictionary);
198    };
199    plan.root = index.order.root;
200    plan.encoding_bytes = encoding_bytes;
201    work.grow(&mut plan.ids, atom.values().len())?;
202    work.grow(&mut plan.added, atom.values().len())?;
203    for (column, value) in atom.values().iter().enumerate() {
204        work.tick(1)?;
205        plan.encoding_bytes = plan
206            .encoding_bytes
207            .checked_add(value.canonical_bytes_with(|| work.tick(1))? as u128)
208            .ok_or(Failure::Overflow)?;
209        let (term, found) = match index
210            .identities
211            .probe_with(atoms.read(), value, &mut || work.tick(1))?
212        {
213            Probe::Local { term, equality } => (term, equality.map(|id| id as usize)),
214            Probe::Unavailable(error) => return Err(Failure::Read(error)),
215        };
216        let found = match found {
217            Some(id) => Some(id),
218            None => plan.staged(term, dictionary.len(), work)?,
219        };
220        let id = if let Some(id) = found {
221            id
222        } else {
223            let (found, route) = plan.locate(&index.order, dictionary, atoms, atom, value, work)?;
224            if found.is_some() {
225                return Err(Failure::Dictionary);
226            }
227            let id = dictionary
228                .len()
229                .checked_add(plan.added.len())
230                .ok_or(Failure::Overflow)?;
231            ceiling(
232                Resource::Values,
233                id as u128 + 1,
234                (work.limits.max_values as u128).min(u128::from(u32::MAX) + 1),
235            )?;
236            plan.extend(&mut index.order, id, &route, work)?;
237            work.tick(1)?;
238            plan.added.push(Added {
239                cell: Cell {
240                    row: atoms.len(),
241                    column,
242                },
243                term,
244            });
245            id
246        };
247        work.tick(1)?;
248        plan.ids
249            .push(u32::try_from(id).map_err(|_| Failure::Overflow)?);
250    }
251    Ok(())
252}

254/// Both routes come from this owner's actual checked tuple comparisons.
255#[derive(Clone, Copy)]
256pub(super) enum RowPath<'a> {
257    Recorded(&'a Directions),
258    BeyondLast(usize),
259}
260
261pub(super) fn row(
262    index: &mut Index,
263    id: usize,
264    route: RowPath<'_>,
265    retained: Option<RightSpine>,
266    work: &mut Work,
267) -> Result<Planned, Failure> {
268    let reuse = match (&route, retained) {
269        (RowPath::BeyondLast(last), Some(spine)) => {
270            work.tick(1)?;
271            spine.matches(index, index.root, *last)
272        }
273        _ => false,
274    };
275    if !reuse {
276        index.path.clear();
277    }
278    let mut cursor = if reuse { None } else { index.root };
279    let mut offset = 0;
280    while let Some(link) = cursor {
281        work.tick(1)?;
282        let right = match &route {
283            RowPath::Recorded(directions) => directions.get(offset).ok_or(Failure::Dictionary)?,
284            RowPath::BeyondLast(_) => true,
285        };
286        let previous = position(link);
287        let node = index.nodes[previous];
288        push(
289            index,
290            Step {
291                id: previous,
292                node,
293                right,
294                changed: false,
295            },
296            work,
297        )?;
298        cursor = node.children[usize::from(right)];
299        offset += 1;
300    }
301    match route {
302        RowPath::Recorded(directions) if offset != directions.len() => {
303            return Err(Failure::Dictionary);
304        }
305        RowPath::BeyondLast(last) => {
306            debug_assert_eq!(index.path.last().map(|step| step.id), Some(last));
307        }
308        RowPath::Recorded(_) => {}
309    }
310    push(
311        index,
312        Step {
313            id,
314            node: Node::default(),
315            right: false,
316            changed: true,
317        },
318        work,
319    )?;
320    index.plan_changes_from(index.root, id, &mut || work.tick(1))
321}
322
323fn push(index: &mut Index, step: Step, work: &mut Work) -> Result<(), Failure> {
324    work.grow(&mut index.path, 1)?;
325    work.tick(1)?;
326    index.path.push(step);
327    Ok(())
