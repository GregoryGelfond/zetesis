1//! A one-tuple transaction containing only equality IDs and tentative AVL nodes.
2//!
3//! Later values inspect the latest tentative patch before the published tree.
4//! No row, dictionary representative or tree link changes before final admission.
5//! With a new values and height h, retained patches use O(a h) cells. Their linear
6//! reverse lookup can cost O(a² h²) metadata probes; each probe is charged. This
7//! depends on this tuple's width, never on shifting the historical dictionary.
8

33impl Plan {
34    /// Only this tuple's unpublished distinct values are scanned. Historical
35    /// dictionary hits use the complete canonical inverse instead.
36    fn staged(
37        &self,
38        term: TermId,
39        start: usize,
40        work: &mut Work,
41    ) -> Result<Option<usize>, Failure> {
42        for (offset, added) in self.added.iter().enumerate() {
43            work.tick(1)?;
44            if added.term == term {
45                return start.checked_add(offset).map(Some).ok_or(Failure::Overflow);
46            }
47        }
48        Ok(None)
49    }
50
51    fn node(&self, nodes: &[Node], id: usize, work: &mut Work) -> Result<Node, Failure> {
52        for patch in self.patches.iter().rev() {
53            work.tick(1)?;
54            if patch.id == id {
55                return Ok(patch.node);
56            }
57        }
58        work.tick(1)?;
59        nodes.get(id).copied().ok_or(Failure::Dictionary)
60    }
61
62    fn value<'a>(
63        &self,
64        dictionary: &[Cell],
65        atoms: Atoms<'a>,
66        atom: AtomRef<'a>,
67        id: usize,
68    ) -> Result<TermRef<'a>, Failure> {
69        let cell = if let Some(cell) = dictionary.get(id) {
70            return atoms
71                .at(cell.row)
72                .and_then(|atom| atom.values().get(cell.column))
73                .ok_or(Failure::Dictionary);
74        } else {
75            self.added
76                .get(id - dictionary.len())
77                .map(|added| &added.cell)
78                .ok_or(Failure::Dictionary)?
79        };
80        atom.values().get(cell.column).ok_or(Failure::Dictionary)
81    }
82
83    fn locate(
84        &self,
85        index: &Index,
86        dictionary: &[Cell],
87        atoms: Atoms<'_>,
88        atom: AtomRef<'_>,
89        value: TermRef<'_>,
90        work: &mut Work,
91    ) -> Result<(Option<usize>, Directions), Failure> {
92        let mut route = Directions::default();
93        let mut cursor = self.root;
94        while let Some(link) = cursor {
95            work.tick(1)?;
96            let id = position(link);
97            let order = work.compare(value, self.value(dictionary, atoms, atom, id)?)?;
98            if order.is_eq() {
99                return Ok((Some(id), route));
100            }
101            let right = order == Ordering::Greater;
102            route.push(right).ok_or(Failure::Dictionary)?;
103            cursor = self.node(&index.nodes, id, work)?.children[usize::from(right)];
104        }
105        Ok((None, route))
106    }
107
108    fn extend(
109        &mut self,
110        index: &mut Index,
111        id: usize,
112        route: &Directions,
113        work: &mut Work,
114    ) -> Result<(), Failure> {
115        index.path.clear();
116        let mut cursor = self.root;
117        for offset in 0..route.len() {
118            work.tick(1)?;
119            let right = route.get(offset).ok_or(Failure::Dictionary)?;
120            let previous = position(cursor.ok_or(Failure::Dictionary)?);
121            let node = self.node(&index.nodes, previous, work)?;
122            push(
123                index,
124                Step {
125                    id: previous,
126                    node,
127                    right,
128                    changed: false,
129                },
130                work,
131            )?;
132            cursor = node.children[usize::from(right)];
133        }
134        if cursor.is_some() {
135            return Err(Failure::Dictionary);
136        }
137        push(
138            index,
139            Step {
140                id,
141                node: Node::default(),
142                right: false,
143                changed: true,
144            },
145            work,
146        )?;
147        let root = index.plan_from(self.root, id, &mut || work.tick(1))?;
148        work.tick(index.path.len() as u128)?;
149        let changed = index.path.iter().filter(|step| step.changed).count();
150        work.grow(&mut self.patches, changed)?;
151        for step in &index.path {
152            work.tick(1)?;
153            if step.changed {
154                self.patches.push(*step);
155            }
156        }
157        self.root = root;
158        Ok(())
159    }
160
161    pub fn release(self, work: &mut Work) {
162        work.release(self.ids);
163        work.release(self.added);
164        work.release(self.patches);
165        work.live -= size_of::<Self>();
166    }
167}

169pub(super) fn values(
170    layout: &mut Layout,
171    atoms: Atoms<'_>,
172    atom: AtomRef<'_>,
173    encoding_bytes: u128,
174    work: &mut Work,
175) -> Result<Plan, Failure> {
176    let Layout {
177        dictionary, index, ..
178    } = layout;
179    let DictionaryIndex::Append(index) = index else {
180        return Err(Failure::Dictionary);
181    };
182    work.include(size_of::<Plan>())?;
183    let mut plan = Plan {
184        ids: work.reserve(atom.values().len())?,
185        added: work.reserve(atom.values().len())?,
186        patches: Vec::new(),
187        root: index.order.root,
188        encoding_bytes,
189    };
190    for (column, value) in atom.values().iter().enumerate() {
191        work.tick(1)?;
192        plan.encoding_bytes = plan
193            .encoding_bytes
194            .checked_add(value.canonical_bytes_with(|| work.tick(1))? as u128)
195            .ok_or(Failure::Overflow)?;
196        let (term, found) = match index
197            .identities
198            .probe_with(atoms.read(), value, &mut || work.tick(1))?
199        {
200            Probe::Local { term, equality } => (term, equality.map(|id| id as usize)),
201            Probe::Unavailable(error) => return Err(Failure::Read(error)),
202        };
203        let found = match found {
204            Some(id) => Some(id),
205            None => plan.staged(term, dictionary.len(), work)?,
206        };
207        let id = if let Some(id) = found {
208            id
209        } else {
210            let (found, route) = plan.locate(&index.order, dictionary, atoms, atom, value, work)?;
211            if found.is_some() {
212                return Err(Failure::Dictionary);
213            }
214            let id = dictionary
215                .len()
216                .checked_add(plan.added.len())
217                .ok_or(Failure::Overflow)?;
218            ceiling(
219                Resource::Values,
220                id as u128 + 1,
221                (work.limits.max_values as u128).min(u128::from(u32::MAX) + 1),
222            )?;
223            plan.extend(&mut index.order, id, &route, work)?;
224            work.tick(1)?;
225            plan.added.push(Added {
226                cell: Cell {
227                    row: atoms.len(),
228                    column,
229                },
230                term,
231            });
232            id
233        };
234        work.tick(1)?;
235        plan.ids
236            .push(u32::try_from(id).map_err(|_| Failure::Overflow)?);
237    }
238    Ok(plan)
239}

241pub(super) fn row(
242    index: &mut Index,
243    id: usize,
244    route: &Directions,
245    work: &mut Work,
246) -> Result<Link, Failure> {
247    index.path.clear();
248    let mut cursor = index.root;
249    for offset in 0..route.len() {
250        work.tick(1)?;
251        let right = route.get(offset).ok_or(Failure::Dictionary)?;
252        let previous = position(cursor.ok_or(Failure::Dictionary)?);
253        let node = index.nodes[previous];
254        push(
255            index,
256            Step {
257                id: previous,
258                node,
259                right,
260                changed: false,
261            },
262            work,
263        )?;
264        cursor = node.children[usize::from(right)];
265    }
266    if cursor.is_some() {
267        return Err(Failure::Dictionary);
268    }
269    push(
270        index,
271        Step {
272            id,
273            node: Node::default(),
274            right: false,
275            changed: true,
276        },
277        work,
278    )?;
279    index.plan(id, &mut || work.tick(1))
280}

282fn push(index: &mut Index, step: Step, work: &mut Work) -> Result<(), Failure> {
283    work.grow(&mut index.path, 1)?;
284    work.tick(1)?;
285    index.path.push(step);
286    Ok(())
