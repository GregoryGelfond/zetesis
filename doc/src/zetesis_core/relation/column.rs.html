7/// A borrowed equality column in original row order.
8///
9/// Physical widths change storage only. Every decoded value is the original
10/// `u32` identifier in the enclosing relation's dictionary, never a logical
11/// number or term-order rank. Variants permit dispatch once per bulk operation;
12/// borrowing this view allocates no unpacked copy.
13#[derive(Clone, Copy, Debug)]
14pub enum Column<'a> {
15    /// Identifiers at most 255.
16    U8(&'a [u8]),
17    /// Identifiers at most 65,535.
18    U16(&'a [u16]),
19    /// Full-width identifiers.
20    U32(&'a [u32]),
21}
22
23impl Column<'_> {
24    /// Number of row occurrences.
25    #[must_use]
26    pub const fn len(self) -> usize {
27        match self {
28            Self::U8(values) => values.len(),
29            Self::U16(values) => values.len(),
30            Self::U32(values) => values.len(),
31        }
32    }
33
34    /// Whether there are no row occurrences.
35    #[must_use]
36    pub const fn is_empty(self) -> bool {
37        self.len() == 0
38    }
39
40    /// Physical bits per cell, independent of the dictionary's semantic order.
41    #[must_use]
42    pub const fn bits(self) -> u32 {
43        match self {
44            Self::U8(_) => 8,
45            Self::U16(_) => 16,
46            Self::U32(_) => 32,
47        }
48    }
49
50    /// The unchanged logical identifier at one original row position.
51    #[must_use]
52    pub fn get(self, position: usize) -> Option<u32> {
53        match self {
54            Self::U8(values) => values.get(position).copied().map(u32::from),
55            Self::U16(values) => values.get(position).copied().map(u32::from),
56            Self::U32(values) => values.get(position).copied(),
57        }
58    }
59}

66            Self::U8(values) => Values(Iter::U8(values.iter())),
67            Self::U16(values) => Values(Iter::U16(values.iter())),
68            Self::U32(values) => Values(Iter::U32(values.iter())),
69        }
70    }
71}

147/// Only this owner stores cells. Growth can widen their physical type, but
148/// never changes a published logical cell or publishes a partial appended row.
149pub(super) enum OwnedColumn {
150    U8(Vec<u8>),
151    U16(Vec<u16>),
152    U32(Vec<u32>),
153}
154
155impl OwnedColumn {
156    pub(super) const fn new() -> Self {
157        Self::U8(Vec::new())
158    }
159
160    pub(super) fn zeroed(rows: usize, largest_id: u32, work: &mut Work) -> Result<Self, Failure> {
161        if u8::try_from(largest_id).is_ok() {
162            Ok(Self::U8(zeroed(rows, work)?))
163        } else if u16::try_from(largest_id).is_ok() {
164            Ok(Self::U16(zeroed(rows, work)?))
165        } else {
166            Ok(Self::U32(zeroed(rows, work)?))
167        }
168    }
169
170    pub(super) fn view(&self) -> Column<'_> {
171        match self {
172            Self::U8(values) => Column::U8(values),
173            Self::U16(values) => Column::U16(values),
174            Self::U32(values) => Column::U32(values),
175        }
176    }
177
178    /// Admit capacity and any exact widening before an indivisible append.
179    /// A refused copy leaves all old cells and their owner intact; an admitted
180    /// widening may remain when a later reservation for another column fails.
181    pub(super) fn reserve(&mut self, id: u32, work: &mut Work) -> Result<(), Failure> {
182        let replacement = match self {
183            Self::U8(values) if u8::try_from(id).is_ok() => return work.grow(values, 1),
184            Self::U16(values) if u16::try_from(id).is_ok() => return work.grow(values, 1),
185            Self::U32(values) => return work.grow(values, 1),
186            Self::U8(values) if u16::try_from(id).is_ok() => Self::U16(widen(values, work)?),
187            Self::U8(values) => Self::U32(widen(values, work)?),
188            Self::U16(values) => Self::U32(widen(values, work)?),
189        };
190        match std::mem::replace(self, replacement) {
191            Self::U8(values) => work.release(values),
192            Self::U16(values) => work.release(values),
193            Self::U32(values) => work.release(values),
194        }
195        Ok(())
196    }
197
198    /// `reserve(id)` establishes width and spare capacity before publication.
199    pub(super) fn push(&mut self, id: u32) {
200        match self {
201            Self::U8(values) => values.push(u8::try_from(id).expect("reserved column width")),
202            Self::U16(values) => values.push(u16::try_from(id).expect("reserved column width")),
203            Self::U32(values) => values.push(id),
204        }
205    }
206
207    /// Immutable construction admitted the dictionary's largest possible ID.
208    pub(super) fn set(&mut self, row: usize, id: u32) {
209        match self {
210            Self::U8(values) => values[row] = u8::try_from(id).expect("admitted dictionary width"),
211            Self::U16(values) => {
212                values[row] = u16::try_from(id).expect("admitted dictionary width");
213            }
214            Self::U32(values) => values[row] = id,
215        }
216    }
217
218    pub(super) fn clear(&mut self) {
219        match self {
220            Self::U8(values) => values.clear(),
221            Self::U16(values) => values.clear(),
222            Self::U32(values) => values.clear(),
223        }
224    }
225
226    /// Heap capacity only; the enclosing vector owns this enum's header.
227    pub(super) fn retained_bytes(&self) -> usize {
228        match self {
229            Self::U8(values) => values.capacity(),
230            Self::U16(values) => values.capacity() * size_of::<u16>(),
231            Self::U32(values) => values.capacity() * size_of::<u32>(),
232        }
233    }
234}

236fn zeroed<T: Copy + Default>(rows: usize, work: &mut Work) -> Result<Vec<T>, Failure> {
237    let mut values = reserve(rows, work)?;
238    if let Err(error) = work.tick(rows as u128) {
239        work.release(values);
240        return Err(error);
241    }
242    values.resize(rows, T::default());
243    Ok(values)
244}
245
246fn widen<T: Copy, U: From<T>>(values: &[T], work: &mut Work) -> Result<Vec<U>, Failure> {
247    let needed = values.len().checked_add(1).ok_or(Failure::Overflow)?;
248    let mut widened = reserve(needed, work)?;
249    for &value in values {
250        if let Err(error) = work.tick(1) {
251            work.release(widened);
252            return Err(error);
253        }
254        widened.push(U::from(value));
255    }
256    Ok(widened)
257}
258
259/// A refused local reservation is dropped, unlike an owner-retained growth.
260/// Keep the observed peak but remove that temporary buffer from live bytes.
261fn reserve<T>(count: usize, work: &mut Work) -> Result<Vec<T>, Failure> {
262    let previous = work.live;
263    let result = work.reserve(count);
264    if result.is_err() {
265        work.live = previous;
266    }
267    result
268}
