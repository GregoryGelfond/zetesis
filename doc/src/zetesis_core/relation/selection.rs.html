30/// A checked conjunction of whole-value equalities for one immutable owner.
31///
32/// Empty equalities impose no restriction. An absent dictionary value makes
33/// `is_possible()` false even when the remaining equalities are empty. This is
34/// row-filter feasibility only, not an ASP consistency or membership result.
35pub struct Query<'owner, 'source> {
36    relation: &'owner Relation<'source>,
37    equalities: Vec<Equality>,
38    possible: bool,
39    bytes: usize,
40    work: u128,
41}
42
43/// One complete equality query or typed refusal with its actual work prefix.
44///
45/// No partially resolved equalities escape a refusal. Capacity includes the
46/// borrowed relation and the query's named frame/buffer, not caller scratch,
47/// this by-value receipt wrapper or source payload. These observations neither grant additional budget nor
48/// establish row feasibility or semantic truth.
49pub struct QueryAttempt<'owner, 'source, E = Failure> {
50    /// Complete owner-bound query, or the original typed failure.
51    pub result: Result<Query<'owner, 'source>, E>,
52    /// Admitted inspection/comparison work, including a failed attempt's prefix.
53    pub work: u128,
54    /// Highest admitted or actually allocated named capacity in this attempt.
55    /// Zero means initial relation/frame admission failed before a meter existed.
56    /// Refused proposed allocations are excluded; actual allocator slack can
57    /// exceed the byte limit and is retained even when the attempt then fails.
58    pub peak_bytes: usize,
59}
60
61/// A relation refusal or an enclosing caller's refused work permit.
62#[derive(Debug, PartialEq, Eq)]
63pub enum QueryFailure<E> {
64    /// Relation shape, allocation or operation-scoped resource failure.
65    Relation(Failure),
66    /// The caller refused before the next inspection or comparison.
67    Stopped(E),
68}
69
70impl<E> From<Failure> for QueryFailure<E> {
71    fn from(error: Failure) -> Self {
72        Self::Relation(error)
73    }
74}
75
76impl<E: fmt::Display> fmt::Display for QueryFailure<E> {
77    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
78        match self {
79            Self::Relation(error) => error.fmt(formatter),
80            Self::Stopped(error) => error.fmt(formatter),
81        }
82    }
83}

85impl<E: std::error::Error + 'static> std::error::Error for QueryFailure<E> {
86    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
87        match self {
88            Self::Relation(error) => Some(error),
89            Self::Stopped(error) => Some(error),
90        }
91    }

214impl<'source> Relation<'source> {
215    /// Resolve whole-value equalities without retaining caller value borrows.
216    ///
217    /// Terms may come from this catalog, another catalog or [`crate::Value`]
218    /// ingress via [`TermRef::from`]. Exact typed identity resolves each through
219    /// the same dictionary; raw IDs from distinct catalogs are never compared.
220    ///
221    /// Append dictionaries use expected-constant canonical-ID lookup for a
222    /// compatible prefix; other input uses logarithmic semantic lookup,
223    /// including typed payload costs. Hash operations admit one fixed-key
224    /// container operation, not each internal probe. Input order and repeated columns remain
225    /// explicit. A missing value cannot make a later invalid column valid.
226    ///
227    /// # Errors
228    /// Refuses invalid columns, resource excess and allocation failure. Limits
229    /// include the relation and resulting query; other caller frames are external.
230    /// [`Self::query_attempt`] additionally retains accounting on failure.
231    pub fn query(
232        &self,
233        equalities: &[(usize, TermRef<'_>)],
234        limits: Limits,
235    ) -> Result<Query<'_, 'source>, Failure> {
236        self.query_attempt(equalities, limits).result
237    }
238
239    /// Resolve equalities with the same implementation as [`Self::query`],
240    /// retaining actual admitted work and named capacity on every outcome.
241    ///
242    /// A sequential caller can supply its local remaining work allowance and
243    /// charge the returned prefix, including failure. Concurrent shared quotas
244    /// require [`Self::query_attempt_with`] to admit each step before execution;
245    /// a remaining-quota snapshot or later charge cannot reserve shared work.
246    /// A failed lookup cannot supply a partially resolved query or refund work.
247    #[must_use]
248    pub fn query_attempt(
249        &self,
250        equalities: &[(usize, TermRef<'_>)],
251        limits: Limits,
252    ) -> QueryAttempt<'_, 'source> {
253        let attempt =
254            self.query_attempt_with(
255                equalities,
256                limits,
257                || Ok::<(), std::convert::Infallible>(()),
258            );
259        QueryAttempt {
260            result: attempt.result.map_err(|error| match error {
261                QueryFailure::Relation(error) => error,
262                QueryFailure::Stopped(never) => match never {},
263            }),
264            work: attempt.work,
265            peak_bytes: attempt.peak_bytes,
266        }
267    }
268
269    /// Resolve equalities while admitting every charged step through `before`.
270    ///
271    /// The operation's own work ceiling is checked first. The callback then
272    /// admits one inspection, identity descriptor/text comparison or equality
273    /// write before it occurs. Its refusal does not increment the work receipt
274    /// and no later step runs. Caller charges already accepted are not refunded.
275    /// Shape/storage admission remains governed by `limits`; the caller's
276    /// callback can enforce a shared cumulative quota and cancellation.
277    ///
278    /// The same resolver implements [`Self::query_attempt`]. Every outcome
279    /// retains actual accepted work/capacity, and neither failure kind publishes
280    /// a partially resolved query.
281    #[must_use]
282    pub fn query_attempt_with<E>(
283        &self,
284        equalities: &[(usize, TermRef<'_>)],
285        limits: Limits,
286        mut before: impl FnMut() -> Result<(), E>,
287    ) -> QueryAttempt<'_, 'source, QueryFailure<E>> {
288        let mut work = match self.work(limits, size_of::<Query<'_, '_>>()) {
289            Ok(work) => work,
290            Err(error) => {
291                return QueryAttempt {
292                    result: Err(QueryFailure::Relation(error)),
293                    work: 0,
294                    peak_bytes: 0,
295                };
296            }
297        };
298        let result = self.resolve_query(equalities, &mut work, &mut before);
299        QueryAttempt {
300            result,
301            work: work.used,
302            peak_bytes: work.peak,
303        }
304    }

306    fn resolve_query<E>(
307        &self,
308        equalities: &[(usize, TermRef<'_>)],
309        work: &mut Work,
310        before: &mut impl FnMut() -> Result<(), E>,
311    ) -> Result<Query<'_, 'source>, QueryFailure<E>> {
312        let mut resolved = work.reserve(equalities.len())?;
313        let mut possible = true;
314        let mut tick = || -> Result<(), QueryFailure<E>> {
315            let next = work.used.checked_add(1).ok_or(Failure::Overflow)?;
316            super::ceiling(
317                super::Resource::Work,
318                next,
319                u128::from(work.limits.max_work),
320            )?;
321            before().map_err(QueryFailure::Stopped)?;
322            work.used = next;
323            Ok(())
324        };
325        for &(column, value) in equalities {
326            tick()?;
327            if column >= self.predicate.arity() {
328                return Err(Failure::Column.into());
329            }
330            if let Some(index) = storage::lookup_with(&self.layout, &self.source, value, &mut tick)?
331            {
332                let value_id = index;
333                tick()?;
334                resolved.push(Equality { column, value_id });
335            } else {
336                possible = false;
337            }
338        }
339        Ok(Query {
340            relation: self,
341            equalities: resolved,
342            possible,
343            bytes: work.live - self.storage.retained_bytes,
344            work: work.used,
345        })
346    }

348    /// Copy a structurally valid ordered subset into this owner's selection.
349    ///
350    /// This is also the checked boundary for reconstructed device row positions.
351    /// It does not assert that a device or query selected the correct rows.
352    /// Work and retained output capacity are linear in supplied positions.
353    ///
354    /// # Errors
355    /// Refuses repeated, descending or out-of-range positions, resource excess
356    /// and allocation failure. Limits include the relation and returned selection.
357    pub fn selection(
358        &self,
359        positions: &[usize],
360        limits: Limits,
361    ) -> Result<Selection<'_, 'source>, Failure> {
362        let mut work = self.work(limits, size_of::<Selection<'_, '_>>())?;
363        let mut previous = None;
364        for &position in positions {
365            work.tick(1)?;
366            if position >= self.row_count() || previous.is_some_and(|last| last >= position) {
367                return Err(Failure::Selection);
368            }
369            previous = Some(position);
370        }
371        let mut copied = work.reserve(positions.len())?;
372        for &position in positions {
373            work.tick(1)?;
374            copied.push(position);
375        }
376        Ok(Selection {
377            relation: self,
378            positions: copied,
379            bytes: work.live - self.storage.retained_bytes,
380            work: work.used,
381        })
382    }

384    /// Decode one packed row mask into an increasing structural selection.
385    ///
386    /// Bit zero of the first word denotes original row position zero. The
387    /// mask has exactly `row_count().div_ceil(32)` words, with unused high bits
388    /// zero. Ordered bit traversal supplies unique in-range positions directly;
389    /// no temporary row vector or second copy is retained.
390    ///
391    /// Limits include this relation, the borrowed mask's slice bytes and the
392    /// returned selection. Allocation capacity outside that slice, an enclosing
393    /// batch object and other live selections remain caller-accounted. Work
394    /// charges two word scans and two operations per selected row (decode and
395    /// copy). The returned work counter covers that complete reconstruction.
396    ///
397    /// Raw mask bits carry no owner or semantic certificate. The caller chooses
398    /// this owner; shape validity does not prove query satisfaction, complete
399    /// filtering, pattern matching or answer-set membership.
400    ///
401    /// # Errors
402    /// Refuses wrong word count, nonzero unused tail bits, resource excess and
403    /// allocation failure without publishing a partial selection.
404    pub fn selection_from_mask(
405        &self,
406        words: &[u32],
407        limits: Limits,
408    ) -> Result<Selection<'_, 'source>, Failure> {
409        let bits = u32::BITS as usize;
410        if words.len() != self.row_count().div_ceil(bits) {
411            return Err(Failure::Mask);
412        }
413        let mask_bytes = std::mem::size_of_val(words);
414        let extra = mask_bytes
415            .checked_add(size_of::<Selection<'_, '_>>())
416            .ok_or(Failure::Overflow)?;
417        let mut work = self.work(limits, extra)?;
418        let tail = self.row_count() % bits;
419        let mut count = 0_usize;
420        for (index, &word) in words.iter().enumerate() {
421            work.tick(1)?;
422            if index + 1 == words.len() && tail != 0 && word >> tail != 0 {
423                return Err(Failure::Mask);
424            }
425            count = count
426                .checked_add(word.count_ones() as usize)
427                .ok_or(Failure::Overflow)?;
428        }
429        let mut positions = work.reserve(count)?;
430        for (index, &word) in words.iter().enumerate() {
431            work.tick(1)?;
432            let mut remaining = word;
433            while remaining != 0 {
434                work.tick(2)?;
435                positions.push(index * bits + remaining.trailing_zeros() as usize);
436                remaining &= remaining - 1;
437            }
438        }
439        Ok(Selection {
440            relation: self,
441            positions,
442            bytes: work.live - self.storage.retained_bytes - mask_bytes,
443            work: work.used,
444        })
445    }

447    /// Construct the complete input row sequence in original occurrence order.
448    ///
449    /// This asserts only structural row coverage, not satisfaction or membership.
450    ///
451    /// # Errors
452    /// Refuses resource excess or allocation failure before publishing a result.
453    /// Limits include the relation and resulting position-vector capacity.
454    pub fn all(&self, limits: Limits) -> Result<Selection<'_, 'source>, Failure> {
455        let mut work = self.work(limits, size_of::<Selection<'_, '_>>())?;
456        let mut positions = work.reserve(self.row_count())?;
457        for position in 0..self.row_count() {
458            work.tick(1)?;
459            positions.push(position);
460        }
461        Ok(Selection {
462            relation: self,
463            positions,
464            bytes: work.live - self.storage.retained_bytes,
465            work: work.used,
466        })
467    }

469    /// Return exactly the input rows satisfying every supplied equality.
470    ///
471    /// The result preserves the complete input order. It covers the relation
472    /// only when the caller supplied all rows. No equality discharges structural
473    /// matching, repeated unbound variables, guards, support or minimality.
474    /// Work is O(input rows times (1 + equalities)), with constant-width IDs.
475    ///
476    /// # Errors
477    /// Refuses foreign owners before filtering, resource excess or allocation
478    /// failure. No partial selection is published. Limits include this relation,
479    /// both supplied objects and output capacity; unrelated live frames and
480    /// borrowed source payload are the caller's responsibility.
481    pub fn select(
482        &self,
483        query: &Query<'_, 'source>,
484        input: &Selection<'_, 'source>,
485        limits: Limits,
486    ) -> Result<Selection<'_, 'source>, Failure> {
487        let inputs = self.selection_inputs(query, input)?;
488        let extra = inputs
489            .checked_add(size_of::<Selection<'_, '_>>())
490            .ok_or(Failure::Overflow)?;
491        let mut work = self.work(limits, extra)?;
492        let count = if query.possible {
493            input.positions.len()
494        } else {
495            0
496        };
497        let mut positions = work.reserve(count)?;
498        self.visit_matches(query, input, &mut work, |position| positions.push(position))?;
499        Ok(Selection {
500            relation: self,
501            positions,
502            bytes: work.live - self.storage.retained_bytes - inputs,
503            work: work.used,
504        })
505    }

507    /// Return exactly the matching input rows as a packed original-row mask.
508    ///
509    /// Uses the same typed equality predicate and ordered input traversal as
510    /// [`Self::select`], writing bits directly without a position vector. Work
511    /// additionally charges one zero write per mask word. Even an impossible
512    /// query retains `row_count().div_ceil(32)` zero words. This bounded dense
513    /// output can cost more storage than positions for a sparse selection.
514    ///
515    /// # Errors
516    /// Refuses foreign owners before allocation, resource excess or allocation
517    /// failure. No partial mask is returned. Limits include this relation, both
518    /// supplied objects, the mask object and its actual word capacity. Borrowed
519    /// source payload and unrelated live frames remain caller-accounted.
520    pub fn select_mask(
521        &self,
522        query: &Query<'_, 'source>,
523        input: &Selection<'_, 'source>,
524        limits: Limits,
525    ) -> Result<Mask<'_, 'source>, Failure> {
526        let inputs = self.selection_inputs(query, input)?;
527        let extra = inputs
528            .checked_add(size_of::<Mask<'_, '_>>())
529            .ok_or(Failure::Overflow)?;
530        let mut work = self.work(limits, extra)?;
531        let bits = u32::BITS as usize;
532        let count = self.row_count().div_ceil(bits);
533        let mut words = work.reserve(count)?;
534        work.tick(count as u128)?;
535        words.resize(count, 0_u32);
536        self.visit_matches(query, input, &mut work, |position| {
537            words[position / bits] |= 1 << (position % bits);
538        })?;
539        Ok(Mask {
540            relation: self,
541            words,
542            bytes: work.live - self.storage.retained_bytes - inputs,
543            work: work.used,
544        })
545    }
546
547    fn selection_inputs(
548        &self,
549        query: &Query<'_, 'source>,
550        input: &Selection<'_, 'source>,
551    ) -> Result<usize, Failure> {
552        if !self.same_owner(query.relation) || !self.same_owner(input.relation) {
553            return Err(Failure::Owner);
554        }
555        query
556            .bytes
557            .checked_add(input.bytes)
558            .ok_or(Failure::Overflow)
559    }
560
561    fn visit_matches(
562        &self,
563        query: &Query<'_, 'source>,
564        input: &Selection<'_, 'source>,
565        work: &mut Work,
566        mut emit: impl FnMut(usize),
567    ) -> Result<(), Failure> {
568        if query.possible {
569            for &position in &input.positions {
570                work.tick(1)?;
571                if self.matches_row(query, position, work)? {
572                    work.tick(1)?;
573                    emit(position);
574                }
575            }
576        }
577        Ok(())
578    }
579
580    fn matches_row(
581        &self,
582        query: &Query<'_, 'source>,
583        position: usize,
584        work: &mut Work,
585    ) -> Result<bool, Failure> {
586        for equality in &query.equalities {
587            work.tick(1)?;
588            let id = self.layout.columns[equality.column][position];
589            if id != equality.value_id {
590                return Ok(false);
591            }
592        }
593        Ok(true)
594    }
