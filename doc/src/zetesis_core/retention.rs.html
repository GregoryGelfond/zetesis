1//! Transactional canonical payload accounting for retained interpretations.
2//!
3//! Each occurrence catalog's portable encoding is charged once per allocation,
4//! selections once per retained entry, and caller-described payload separately.
5//! Canonical identities outside an occurrence map remain uncharged even when
6//! its shared prefix keeps them alive. This is not an allocator or RSS bound:
7//! vector/hash capacity, index entries, Arc envelopes, subjects and transient
8//! candidates also remain outside this measure.
9
10use std::{collections::HashMap, fmt};
11
12use crate::{AtomCatalog, Model};
13
14/// Canonical payload admitted by a [`ModelRetention`] ledger.
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
16pub struct RetainedPayload {
17    /// Distinct catalog allocations, including an empty catalog if retained.
18    pub catalogs: usize,
19    /// Encoded occurrence catalogs, including repeated and unselected entries,
20    /// counted once per catalog allocation; excludes other prefix identities.
21    pub catalog_bytes: usize,
22    /// Canonical position records, counted separately for every retained entry.
23    pub selection_bytes: usize,
24    /// Caller-supplied canonical payload, such as objective score records.
25    pub associated_bytes: usize,
26    /// Checked sum of the three byte subtotals.
27    pub bytes: usize,
28}
29
30/// Owner-aware accounting for an append-only retained family or its replacement.
31///
32/// The private hash index holds the actual catalog handle beside its allocation
33/// identity. Equal-content distinct owners remain distinct; a live handle keeps
34/// its address from being reused. No atom payload is cloned. Expected amortized
35/// lookup is constant time, with hash-table worst-case linear lookup; metadata
36/// space is linear in distinct retained owners. Replacement/clear releases each
37/// old handle and costs linear time in retained index capacity.
38///
39/// This ledger does not retain model selections or establish answer-set
40/// membership. The consumer must retain the corresponding model entries and
41/// describe associated payload honestly. Prepare an admission, reserve the
42/// consumer's entry slot, then commit and publish without another fallible step.
43/// A refused or abandoned admission leaves the old payload and owner set intact;
44/// successful reservation can still grow uncharged index capacity.
45#[derive(Debug, Default)]
46pub struct ModelRetention {
47    owners: HashMap<usize, AtomCatalog>,
48    payload: RetainedPayload,
49}
50
51impl ModelRetention {
52    /// Current committed accounting. Pending admissions are not included.
53    #[must_use]
54    pub const fn payload(&self) -> RetainedPayload {
55        self.payload
56    }
57
58    /// Prepare one additional retained entry without publishing its charge.
59    /// `associated_bytes` is the additional caller-owned canonical payload.
60    /// The ceiling applies to the complete resulting payload, including when
61    /// the caller lowers it below a previously admitted total.
62    ///
63    /// # Errors
64    /// Returns checked overflow, the complete prospective byte-limit refusal,
65    /// or owner-index allocation failure. No partial admission is committed.
66    pub fn admit(
67        &mut self,
68        model: &Model,
69        associated_bytes: usize,
70        max_bytes: usize,
71    ) -> Result<RetentionAdmission<'_>, RetentionError> {
72        self.prepare(model, associated_bytes, max_bytes, false)
73    }
74
75    /// Prepare replacement of the complete retained family by one model entry.
76    /// The new owner's full occurrence encoding and supplied associated payload
77    /// are charged against an empty ledger; old owners remain live until commit.
78    /// The byte limit describes the new payload, not old/new transient overlap.
79    ///
80    /// # Errors
81    /// Returns the same failures as [`Self::admit`], preserving the old family
82    /// accounting on every refusal and when the returned admission is dropped.
83    pub fn replace(
84        &mut self,
85        model: &Model,
86        associated_bytes: usize,
87        max_bytes: usize,
88    ) -> Result<RetentionAdmission<'_>, RetentionError> {
89        self.prepare(model, associated_bytes, max_bytes, true)
90    }
91
92    /// Release all catalog handles and reset accounting, retaining index capacity.
93    /// The caller must also release or transfer the associated retained entries.
94    pub fn clear(&mut self) {
95        self.owners.clear();
96        self.payload = RetainedPayload::default();
97    }
98
99    fn prepare(
100        &mut self,
101        model: &Model,
102        associated_bytes: usize,
103        max_bytes: usize,
104        replacement: bool,
105    ) -> Result<RetentionAdmission<'_>, RetentionError> {
106        let key = model.catalog().owner_key();
107        let new_owner = replacement || !self.owners.contains_key(&key);
108        let mut payload = if replacement {
109            RetainedPayload::default()
110        } else {
111            self.payload
112        };
113        if new_owner {
114            payload.catalogs = add(payload.catalogs, 1)?;
115            payload.catalog_bytes = add(
116                payload.catalog_bytes,
117                model
118                    .catalog()
119                    .retained_payload_bytes()
120                    .ok_or(RetentionError::Overflow)?,
121            )?;
122        }
123        payload.selection_bytes = add(
124            payload.selection_bytes,
125            model
126                .selection_payload_bytes()
127                .ok_or(RetentionError::Overflow)?,
128        )?;
129        payload.associated_bytes = add(payload.associated_bytes, associated_bytes)?;
130        payload.bytes = add(
131            add(payload.catalog_bytes, payload.selection_bytes)?,
132            payload.associated_bytes,
133        )?;
134        if payload.bytes > max_bytes {
135            return Err(RetentionError::Bytes {
136                required: payload.bytes,
137                limit: max_bytes,
138            });
139        }
140        // Replacement can reuse any existing slot, but an empty index still
141        // needs one. No reserve or fallible operation remains after commit starts.
142        self.owners
143            .try_reserve(payload.catalogs.saturating_sub(self.owners.len()))
144            .map_err(|_| RetentionError::Allocation)?;
145        Ok(RetentionAdmission {
146            ledger: self,
147            owner: new_owner.then(|| model.catalog().clone()),
148            payload,
149            replacement,
150        })
151    }
152}
153
154fn add(left: usize, right: usize) -> Result<usize, RetentionError> {
155    left.checked_add(right).ok_or(RetentionError::Overflow)
156}

158/// Exclusive pending admission; dropping it abandons publication.
159/// The borrow prevents intervening ledger changes from making its charge stale.
160#[derive(Debug)]
161#[must_use = "commit only after all consumer admission steps succeed"]
162pub struct RetentionAdmission<'a> {
163    ledger: &'a mut ModelRetention,
164    owner: Option<AtomCatalog>,
165    payload: RetainedPayload,
166    replacement: bool,
167}
168
169impl RetentionAdmission<'_> {
170    /// Complete prospective accounting, available before publication.
171    #[must_use]
172    pub const fn payload(&self) -> RetainedPayload {
173        self.payload
174    }
175
176    /// Publish the pre-admitted owner set and payload. Allocates no storage and
177    /// invokes no caller callback; the consumer can now publish its reserved entry.
178    pub fn commit(self) {
179        if self.replacement {
180            self.ledger.owners.clear();
181        }
182        if let Some(owner) = self.owner {
183            self.ledger.owners.insert(owner.owner_key(), owner);
184        }
185        self.ledger.payload = self.payload;
186    }
187}

189/// A retention request failed before publishing any owner or payload change.
190#[derive(Clone, Copy, Debug, PartialEq, Eq)]
191pub enum RetentionError {
192    /// The complete proposed canonical payload exceeds the requested ceiling.
193    Bytes {
194        /// Complete prospective payload, not only the new entry's increment.
195        required: usize,
196        /// Requested ceiling; zero is a real limit.
197        limit: usize,
198    },
199    /// Canonical payload/count arithmetic cannot be represented by `usize`.
200    Overflow,
201    /// The distinct-owner index could not reserve storage.
202    Allocation,
203}
204
205impl fmt::Display for RetentionError {
206    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
207        match self {
208            Self::Bytes { required, limit } => write!(
209                f,
210                "retained payload needs {required} bytes; limit is {limit}"
211            ),
212            Self::Overflow => f.write_str("retained payload arithmetic overflowed"),
213            Self::Allocation => f.write_str("retained catalog index could not reserve storage"),
214        }
215    }
216}
