12/// One preorder node of a closed value. Child nodes immediately follow a head.
13#[derive(Clone, Debug, PartialEq, Eq)]
14pub enum ValueNode {
15    /// Least ASP value.
16    Infimum,
17    /// Checked-width integer leaf.
18    Number(i32),
19    /// Decoded string leaf.
20    String(String),
21    /// Positive named constant leaf.
22    Symbol(String),
23    /// Named function with exactly `arity` following subtrees.
24    Function {
25        /// Exact constructor name; source spelling is the adapter's contract.
26        name: String,
27        /// Constructor sign, independent of the outer predicate sign.
28        sign: Sign,
29        /// Number of ordered immediate children.
30        arity: usize,
31    },
32    /// Anonymous constructor, including `()` and `(x,)`.
33    Tuple {
34        /// Number of ordered immediate children.
35        arity: usize,
36    },
37    /// Greatest ASP value.
38    Supremum,
39}
40
41impl PartialOrd for ValueNode {
42    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
43        Some(self.cmp(other))
44    }
45}
46impl Ord for ValueNode {
47    fn cmp(&self, other: &Self) -> Ordering {
48        crate::term_order::storage(self.view(), other.view())

50}

58/// Construction ceilings for one owned value and bounded construction scratch.
59#[derive(Clone, Copy, Debug, PartialEq, Eq)]
60pub struct ValueLimits {
61    /// Maximum number of preorder nodes.
62    pub max_nodes: usize,
63    /// Maximum root-inclusive depth.
64    pub max_depth: usize,
65    /// Maximum retained node capacity, text and canonical spelling bytes, plus
66    /// reserved validation/render frame bytes. This is accounting, not allocator RSS.
67    pub max_bytes: usize,
68}
69impl Default for ValueLimits {
70    fn default() -> Self {
71        Self {
72            max_nodes: 262_144,
73            max_depth: 128,
74            max_bytes: 16_777_216,
75        }
76    }
77}

79/// A construction resource, measured before retaining the value.
80#[derive(Clone, Copy, Debug, PartialEq, Eq)]
81pub enum ValueResource {
82    /// Preorder nodes.
83    Nodes,
84    /// Root-inclusive logical depth.
85    Depth,
86    /// Retained payload and construction frame reservations.
87    Bytes,
88}

90/// Closed-value construction failed without returning a partial value.
91#[derive(Clone, Debug, PartialEq, Eq)]
92pub enum ValueError {
93    /// The supplied sequence is not exactly one finite tree, or a function name is empty.
94    Shape,
95    /// An inclusive construction ceiling was exceeded.
96    Limit {
97        /// The exhausted resource.
98        resource: ValueResource,
99        /// Requested amount.
100        observed: u128,
101        /// Inclusive allowance.
102        limit: usize,
103    },
104    /// Temporary validation storage could not be reserved.
105    Allocation,
106}
107impl fmt::Display for ValueError {
108    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
109        match self {
110            Self::Shape => f.write_str("closed value is not one valid constructor tree"),
111            Self::Limit {
112                resource,
113                observed,
114                limit,
115            } => write!(f, "closed value {resource:?} {observed} exceeds {limit}"),
116            Self::Allocation => {
117                f.write_str("closed value validation storage could not be reserved")
118            }
119        }
120    }
121}
122impl std::error::Error for ValueError {}

124/// Private validated preorder storage; integer positions are never semantic identity.
125#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
126pub struct StructuralValue {
127    nodes: std::sync::Arc<Vec<ValueNode>>,
128    depth: usize,
129    rendered: std::sync::Arc<String>,
130    // Depth and spelling are deterministic derivatives of the canonical nodes.
131    // Derived traits never compare allocation capacities or pointer addresses.
132}
133impl std::hash::Hash for StructuralValue {
134    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
135        crate::term_hash::nodes(self.nodes(), state);
136    }
137}
138impl StructuralValue {
139    /// Exact canonical preorder nodes. No caller can mutate the validated shape.
140    #[must_use]
141    pub fn nodes(&self) -> &[ValueNode] {
142        &self.nodes
143    }
144
145    /// Root-inclusive logical depth.
146    #[must_use]
147    pub fn depth(&self) -> usize {
148        self.depth
149    }
150
151    /// Retained node-buffer capacity, text lengths and spelling capacity; not RSS.
152    #[must_use]
153    pub fn payload_bytes(&self) -> usize {
154        self.nodes.capacity() * std::mem::size_of::<ValueNode>()
155            + self.nodes.iter().map(ValueNode::text_bytes).sum::<usize>()
156            + self.rendered.capacity()
157    }
158
159    /// Exact bytes of the cached ASP spelling, without allocating.
160    #[must_use]
161    pub fn rendered_bytes(&self) -> usize {
162        self.rendered.len()
163    }
164
165    /// Canonical identity encoding size, independent of allocator capacity.
166    #[must_use]
167    pub fn canonical_bytes(&self) -> usize {
168        self.nodes.iter().map(ValueNode::canonical_bytes).sum()
169    }
170}

172fn ceiling(resource: ValueResource, observed: u128, limit: usize) -> Result<(), ValueError> {
173    if observed > limit as u128 {
174        Err(ValueError::Limit {
175            resource,
176            observed,
177            limit,
178        })
179    } else {
180        Ok(())
181    }
182}

184impl Value {
185    /// Validate one flat closed tree. Positive nullary functions normalize to symbols.
186    /// Scalars retain their existing public variants; tuples never collapse to a scalar.
187    ///
188    /// # Errors
189    /// Refuses malformed trees, empty function names, resource excess or validation allocation.
190    pub fn from_nodes(mut nodes: Vec<ValueNode>, limits: ValueLimits) -> Result<Self, ValueError> {
191        ceiling(ValueResource::Nodes, nodes.len() as u128, limits.max_nodes)?;
192        let mut bytes = nodes.capacity() as u128 * std::mem::size_of::<ValueNode>() as u128;
193        ceiling(ValueResource::Bytes, bytes, limits.max_bytes)?;
194        let frames = nodes.len().min(limits.max_depth);
195        let mut scratch = frames as u128
196            * (std::mem::size_of::<usize>() + std::mem::size_of::<(usize, bool, bool)>()) as u128;
197        ceiling(ValueResource::Bytes, bytes + scratch, limits.max_bytes)?;
198        let mut remaining = Vec::new();
199        let mut maximum_depth = 0;
200        remaining
201            .try_reserve_exact(frames)
202            .map_err(|_| ValueError::Allocation)?;
203        scratch = remaining.capacity() as u128 * std::mem::size_of::<usize>() as u128
204            + frames as u128 * std::mem::size_of::<(usize, bool, bool)>() as u128;
205        ceiling(ValueResource::Bytes, bytes + scratch, limits.max_bytes)?;
206        for (index, node) in nodes.iter_mut().enumerate() {
207            while remaining.last() == Some(&0) {
208                remaining.pop();
209            }
210            if let Some(parent) = remaining.last_mut() {
211                *parent -= 1;
212            } else if index != 0 {
213                return Err(ValueError::Shape);
214            }
215            maximum_depth = maximum_depth.max(remaining.len() + 1);
216            ceiling(
217                ValueResource::Depth,
218                remaining.len() as u128 + 1,
219                limits.max_depth,
220            )?;
221            bytes += node.text_bytes() as u128;
222            ceiling(ValueResource::Bytes, bytes + scratch, limits.max_bytes)?;
223            if let ValueNode::Function { name, sign, arity } = node {
224                if name.is_empty() {
225                    return Err(ValueError::Shape);
226                }
227                if *arity == 0 && *sign == Sign::Positive {
228                    *node = ValueNode::Symbol(std::mem::take(name));
229                }
230            }
231            let arity = node.arity();
232            if arity != 0 {
233                remaining.push(arity);
234            }
235        }
236        if nodes.is_empty() || remaining.iter().any(|left| *left != 0) {
237            return Err(ValueError::Shape);
238        }
239        if nodes.len() == 1 {
240            match nodes.pop().ok_or(ValueError::Shape)? {
241                ValueNode::Infimum => return Ok(Self::Infimum),
242                ValueNode::Supremum => return Ok(Self::Supremum),
243                ValueNode::Number(n) => return Ok(Self::Number(n)),
244                ValueNode::String(s) => return Ok(Self::String(s)),
245                ValueNode::Symbol(s) => return Ok(Self::Symbol(s)),
246                node => nodes.push(node),
247            }
248        }
249        let spelling_bytes = nodes.iter().map(ValueNode::rendered_bytes).sum::<u128>();
250        ceiling(
251            ValueResource::Bytes,
252            bytes + scratch + spelling_bytes,
253            limits.max_bytes,
254        )?;
255        // The preceding ceiling proves representability because max_bytes is usize.
256        let spelling_bytes = usize::try_from(spelling_bytes).map_err(|_| ValueError::Allocation)?;
257        let mut rendered = String::new();
258        rendered
259            .try_reserve_exact(spelling_bytes)
260            .map_err(|_| ValueError::Allocation)?;
261        ceiling(
262            ValueResource::Bytes,
263            bytes + scratch + rendered.capacity() as u128,
264            limits.max_bytes,
265        )?;
266        let mut render_frames = Vec::new();
267        render_frames
268            .try_reserve_exact(frames)
269            .map_err(|_| ValueError::Allocation)?;
270        scratch = remaining.capacity() as u128 * std::mem::size_of::<usize>() as u128
271            + render_frames.capacity() as u128 * std::mem::size_of::<(usize, bool, bool)>() as u128;
272        ceiling(
273            ValueResource::Bytes,
274            bytes + scratch + rendered.capacity() as u128,
275            limits.max_bytes,
276        )?;
277        render_nodes(&nodes, &mut render_frames, &mut rendered)
278            .map_err(|_| ValueError::Allocation)?;
279        Ok(Self::Structured(StructuralValue {
280            nodes: std::sync::Arc::new(nodes),
281            depth: maximum_depth,
282            rendered: std::sync::Arc::new(rendered),
283        }))
284    }

286    /// Referenced value payload, excluding its outer `Value` slot. Structural
287    /// clones share this payload; consumers may charge it conservatively per use.
288    #[must_use]
289    pub fn payload_bytes(&self) -> usize {
290        match self {
291            Self::String(text) | Self::Symbol(text) => text.len(),
292            Self::Structured(value) => value.payload_bytes(),
293            _ => 0,
294        }
295    }
296
297    /// Named nested buffer capacity, excluding this inline value.
298    ///
299    /// Includes actual string/node-vector/spelling capacities and the structural
300    /// vector/string owner headers. Shared buffers are counted per occurrence;
301    /// Arc counters and allocator metadata are excluded. Visits structural node
302    /// descriptors without comparing or copying their text. Returns `None` if
303    /// the wide accounting sum cannot be represented.
304    #[must_use]
305    pub fn checked_payload_capacity_bytes(&self) -> Option<u128> {
306        match self {
307            Self::String(text) | Self::Symbol(text) => Some(text.capacity() as u128),
308            Self::Structured(value) => {
309                let fixed = (value.nodes.capacity() as u128)
310                    .checked_mul(std::mem::size_of::<ValueNode>() as u128)?
311                    .checked_add(value.rendered.capacity() as u128)?
312                    .checked_add(std::mem::size_of::<Vec<ValueNode>>() as u128)?
313                    .checked_add(std::mem::size_of::<String>() as u128)?;
314                value.nodes.iter().try_fold(fixed, |bytes, node| {
315                    let capacity = match node {
316                        ValueNode::String(text)
317                        | ValueNode::Symbol(text)
318                        | ValueNode::Function { name: text, .. } => text.capacity() as u128,
319                        _ => 0,
320                    };
321                    bytes.checked_add(capacity)
322                })
323            }
324            _ => Some(0),
325        }
326    }

328    /// Logical identity encoding bytes: tags, lengths, numbers and text.
329    #[must_use]
330    pub fn canonical_bytes(&self) -> usize {
331        match self {
332            Self::Number(_) => 5,
333            Self::String(text) | Self::Symbol(text) => 9 + text.len(),
334            Self::Structured(value) => value.canonical_bytes(),
335            Self::Infimum | Self::Supremum => 1,
336        }
337    }

340impl ValueNode {
341    /// Borrow the complete node description in constant time, without copying
342    /// text. The view remains valid only while this node is borrowed; it does
343    /// not validate any surrounding tree.
344    #[must_use]
345    pub fn view(&self) -> ValueNodeRef<'_> {
346        match self {
347            Self::Infimum => ValueNodeRef::Infimum,
348            Self::Number(number) => ValueNodeRef::Number(*number),
349            Self::String(text) => ValueNodeRef::String(text),
350            Self::Symbol(name) => ValueNodeRef::Symbol(name),
351            Self::Function { name, sign, arity } => ValueNodeRef::Function {
352                name,
353                sign: *sign,
354                arity: *arity,
355            },
356            Self::Tuple { arity } => ValueNodeRef::Tuple { arity: *arity },
357            Self::Supremum => ValueNodeRef::Supremum,
358        }
359    }
360
361    fn text_bytes(&self) -> usize {
362        self.view().text_bytes()
363    }
364    fn rendered_bytes(&self) -> u128 {
365        self.view().rendered_bytes()
366    }
367    fn canonical_bytes(&self) -> usize {
368        usize::try_from(self.view().canonical_bytes()).expect("admitted node encoding length")
369    }
370    fn arity(&self) -> usize {
371        match self {
372            Self::Function { arity, .. } | Self::Tuple { arity } => *arity,
373            _ => 0,
374        }
375    }
376}

384fn render_nodes(
385    nodes: &[ValueNode],
386    frames: &mut Vec<spelling::Frame>,
387    output: &mut String,
388) -> fmt::Result {
389    for node in nodes {
390        spelling::node(node.view(), frames, output, &mut |_| {
391            Ok::<_, std::convert::Infallible>(())
392        })
393        .map_err(|_| fmt::Error)?;
394    }
395    Ok(())
