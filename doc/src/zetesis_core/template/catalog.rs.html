14#[derive(Debug)]
15struct Data {
16    source: Source,
17    rows: Vec<RowData>,
18    metadata_bytes: u128,
19}

21#[derive(Debug)]
22enum Source {
23    Frozen(FrozenVocabulary),
24    Selected(AtomCatalog),
25}
26impl Source {
27    fn read(&self) -> Read<'_> {
28        match self {
29            Self::Frozen(base) => Read::from(base),
30            Self::Selected(catalog) => catalog.read().storage(),
31        }
32    }
33    fn bytes(&self) -> u128 {
34        match self {
35            Self::Frozen(base) => base.retained_bytes(),
36            Self::Selected(catalog) => catalog.storage().bytes,
37        }
38    }
39}

41/// Ordered template components over one canonical term/predicate vocabulary.
42///
43/// Rows and repeated occurrences retain source order. This storage establishes
44/// no variable safety, rule support, truth, objective priority or carrier domain.
45/// Semantic owners validate those policies and keep their own row topology.
46/// Clones share the immutable source and component metadata without copying it.
47/// The source is either an indexed closed vocabulary or an exact published atom
48/// catalog. A selected catalog retains that entire prefix, including unselected atoms.
49#[derive(Clone, Debug)]
50pub struct TemplateCatalog(Arc<Data>);
51impl TemplateCatalog {
52    /// Number of admitted rows, including equal repeated descriptions.
53    #[must_use]
54    pub fn len(&self) -> usize {
55        self.0.rows.len()
56    }
57    /// Whether there are no component rows.
58    #[must_use]
59    pub fn is_empty(&self) -> bool {
60        self.0.rows.is_empty()
61    }
62    /// Borrow one complete component row by its local occurrence position.
63    #[must_use]
64    pub fn at(&self, row: usize) -> Option<TemplateRow<'_>> {
65        self.0.rows.get(row).map(|data| TemplateRow {
66            read: self.0.source.read(),
67            data,
68        })
69    }
70    /// Named retained base and metadata capacity, excluding external metadata
71    /// registered by the semantic owner during admission and Arc counters.
72    #[must_use]
73    pub fn storage_bytes(&self) -> u128 {
74        self.0.source.bytes() + self.0.metadata_bytes
75    }
76    /// Component header and actual row/argument/filter capacities, excluding
77    /// every allocation retained by the source vocabulary or atom catalog.
78    /// Constant time; this amount is included in `storage_bytes`.
79    #[must_use]
80    pub fn metadata_bytes(&self) -> u128 {
81        self.0.metadata_bytes
82    }
83
84    /// Indexed closed vocabulary included in `storage_bytes`. Selected catalogs
85    /// return zero because they retain a read snapshot without lookup indexes.
86    /// A tuple writer created from a closed catalog shares this exact allocation.
87    #[must_use]
88    pub fn shared_vocabulary_bytes(&self) -> u128 {
89        match &self.0.source {
90            Source::Frozen(base) => base.retained_bytes(),
91            Source::Selected(_) => 0,
92        }
93    }
94    /// Immutable prefix allocations, excluding lookup indexes and any selected
95    /// source occurrence map. Closed catalogs share these with their tuple writers;
96    /// selected catalogs share these with their exact `source_catalog` snapshot.
97    /// Subtract only after establishing that exact sharing. Visits segment metadata.
98    #[must_use]
99    pub fn shared_payload_bytes(&self) -> u128 {
100        match &self.0.source {
101            Source::Frozen(base) => base.payload_bytes(),
102            Source::Selected(catalog) => catalog.shared_snapshot_bytes(),
103        }
104    }
105    /// The exact atom catalog supplied when publishing selected template metadata.
106    /// Closed standalone catalogs have no source atom occurrences and return None.
107    #[must_use]
108    pub fn source_catalog(&self) -> Option<&AtomCatalog> {
109        match &self.0.source {
110            Source::Frozen(_) => None,
111            Source::Selected(catalog) => Some(catalog),
112        }
113    }
114    pub(crate) fn read(&self) -> Read<'_> {
115        self.0.source.read()
116    }
117    pub(crate) fn vocabulary(&self) -> Option<&FrozenVocabulary> {
118        match &self.0.source {
119            Source::Frozen(base) => Some(base),
120            Source::Selected(_) => None,
121        }
122    }
123    pub(crate) fn data(&self, row: usize) -> Option<&RowData> {
124        self.0.rows.get(row)
125    }
126}

128/// Borrowed original-order terms, patterns and filters in one admitted row.
129#[derive(Clone, Copy, Debug)]
130pub struct TemplateRow<'a> {
131    read: Read<'a>,
132    data: &'a RowData,
133}
134impl<'a> TemplateRow<'a> {
135    /// Scalar fields, including variable slots and repeated constants.
136    #[must_use]
137    pub fn terms(self) -> PatternTerms<'a> {
138        PatternTerms::admitted(self.read, &self.data.terms)
139    }
140    /// Relational pattern occurrences, without a support or gate interpretation.
141    #[must_use]
142    pub fn patterns(self) -> Patterns<'a> {
143        Patterns::admitted(self.read, &self.data.patterns)
144    }
145    /// Exact equality/inequality filter occurrences.
146    #[must_use]
147    pub fn filters(self) -> Filters<'a> {
148        Filters::admitted(self.read, &self.data.filters)
149    }
150}

152/// Fallible construction of one shared canonical template authority.
153///
154/// Inputs are borrowed only while importing and are never retained. No implicit
155/// depth limit is imposed on already validated values. The explicit byte limit
156/// covers canonical storage/indexes, component capacities, import scratch and
157/// publication overlap together. Caller ingress and allocator/Arc bookkeeping
158/// are excluded. An interrupted append poisons publication; no partial row can
159/// escape through `finish`. Admission has no execution work-quota contract.
160#[derive(Debug)]
161pub struct TemplateCatalogBuilder {
162    store: Store,
163    rows: Vec<RowData>,
164    metadata: u128,
165    external: u128,
166    limit: usize,
167    failed: Option<Error>,
168}
169impl TemplateCatalogBuilder {
170    /// Start a catalog under an inclusive named storage allowance.
171    ///
172    /// # Errors
173    /// Refuses even the empty authority when its named envelope cannot fit.
174    pub fn new(max_bytes: usize) -> Result<Self, Error> {
175        let mut builder = Self {
176            store: Store::new(max_bytes),
177            rows: Vec::new(),
178            metadata: size_of::<Data>() as u128,
179            external: 0,
180            limit: max_bytes,
181            failed: None,
182        };
183        builder.limit_store()?;
184        Ok(builder)
185    }
186    /// Admit simultaneously live metadata belonging to the semantic owner.
187    /// Before an external reserve, supply old capacity plus proposed replacement;
188    /// after reservation, replace it with actual retained capacity. This does not
189    /// take ownership or infer how the caller allocated those buffers.
190    ///
191    /// # Errors
192    /// Refuses combined storage before updating the admitted external envelope.
193    pub fn set_external_bytes(&mut self, bytes: u128) -> Result<(), Error> {
194        self.check(self.store.current_bytes() + self.metadata + bytes)?;
195        self.external = bytes;
196        self.limit_store()
197    }
198    /// Current named authority, component and admitted external storage.
199    #[must_use]
200    pub fn storage_bytes(&self) -> u128 {
201        self.store.current_bytes() + self.metadata + self.external
202    }
203
204    /// Import a complete row without interpreting its fields as a logical rule.
205    /// Repeated occurrences and variable numbers remain unchanged. A returned
206    /// position identifies this catalog's occurrence, not a global identity.
207    ///
208    /// # Errors
209    /// Refuses canonical import, arithmetic, allocation or the combined byte
210    /// allowance. After a failed append, this builder cannot publish a catalog.
211    pub fn append<'a>(
212        &mut self,
213        terms: impl IntoIterator<Item = TemplateTerm<'a>>,
214        patterns: impl IntoIterator<Item = PatternRef<'a>>,
215        filters: impl IntoIterator<Item = FilterRef<'a>>,
216    ) -> Result<usize, Error> {
217        if let Some(error) = &self.failed {
218            return Err(error.clone());
219        }
220        let result = self.append_row(terms, patterns, filters);
221        if let Err(error) = &result {
222            self.failed = Some(error.clone());
223        }
224        result
225    }
226    fn append_row<'a>(
227        &mut self,
228        terms: impl IntoIterator<Item = TemplateTerm<'a>>,
229        patterns: impl IntoIterator<Item = PatternRef<'a>>,
230        filters: impl IntoIterator<Item = FilterRef<'a>>,
231    ) -> Result<usize, Error> {
232        let mut rows = std::mem::take(&mut self.rows);
233        let reserved = self.reserve(&mut rows, 1);
234        self.rows = rows;
235        reserved?;
236        let mut row = RowData::default();
237        for term in terms {
238            self.reserve(&mut row.terms, 1)?;
239            row.terms.push(self.term(term)?);
240        }
241        for pattern in patterns {
242            self.reserve(&mut row.patterns, 1)?;
243            let predicate = self.predicate(pattern.predicate())?;
244            let mut terms = Vec::new();
245            self.reserve(&mut terms, pattern.terms().len())?;
246            for term in pattern.terms() {
247                terms.push(self.term(term)?);
248            }
249            row.patterns.push(PatternData { predicate, terms });
250        }
251        for filter in filters {
252            self.reserve(&mut row.filters, 1)?;
253            let (left, right) = filter.terms();
254            row.filters.push(FilterData {
255                equal: filter.is_equality(),
256                left: self.term(left)?,
257                right: self.term(right)?,
258            });
259        }
260        let position = self.rows.len();
261        self.rows.push(row);
262        Ok(position)
263    }
264    /// Include a closed input atom's vocabulary without retaining an atom row,
265    /// pattern occurrence, fact or domain-membership assertion. A tuple writer
266    /// can later reuse this frozen base for the atom's actual query metadata.
267    ///
268    /// # Errors
269    /// Same canonical and byte refusals as append; failure prevents publication.
270    pub fn include_atom_vocabulary(&mut self, atom: AtomRef<'_>) -> Result<(), Error> {
271        if let Some(error) = &self.failed {
272            return Err(error.clone());
273        }
274        let result: Result<(), Error> = (|| {
275            self.predicate(atom.predicate())?;
276            for value in atom.values() {
277                self.term(TemplateTerm::Constant(value))?;
278            }
279            Ok(())
280        })();
281        if let Err(error) = &result {
282            self.failed = Some(error.clone());
283        }
284        result
285    }
286    /// Publish immutable rows and their single vocabulary; consumes the writer.
287    ///
288    /// # Errors
289    /// Refuses a prior incomplete append or publication storage. Arc envelope
290    /// allocation retains Rust's existing infallible boundary.
291    pub fn finish(mut self) -> Result<TemplateCatalog, Error> {
292        if let Some(error) = self.failed {
293            return Err(error);
294        }
295        self.store.ceiling(self.limit)?;
296        let vocabulary = self
297            .store
298            .freeze_vocabulary_with(self.metadata + self.external, || Ok::<(), Infallible>(()))
299            .map_err(|failure| failure_error(failure, 0, self.limit))?;
300        Ok(TemplateCatalog(Arc::new(Data {
301            source: Source::Frozen(vocabulary),
302            rows: self.rows,
303            metadata_bytes: self.metadata,
304        })))
305    }
306
307    pub(crate) fn data(&self, row: usize) -> Option<&RowData> {
308        self.rows.get(row)
309    }
310    pub(crate) fn read(&self) -> Read<'_> {
311        Read::from(&self.store)
312    }
313    fn term(&mut self, term: TemplateTerm<'_>) -> Result<TermData, Error> {
314        match term {
315            TemplateTerm::Variable(variable) => Ok(TermData::Variable(variable)),
316            TemplateTerm::Constant(value) => {
317                let limits = Limits {
318                    max_nodes: usize::MAX,
319                    max_depth: usize::MAX,
320                    max_bytes: usize::MAX,
321                };
322                self.store
323                    .import_term_with(value, limits, || Ok::<(), Infallible>(()))
324                    .map(TermData::Constant)
325                    .map_err(|failure| {
326                        failure_error(failure, self.metadata + self.external, self.limit)
327                    })
328            }
329        }
330    }
331    fn predicate(&mut self, predicate: PredicateRef<'_>) -> Result<PredicateId, Error> {
332        self.store
333            .import_predicate_with(predicate, || Ok::<(), Infallible>(()))
334            .map_err(|failure| failure_error(failure, self.metadata + self.external, self.limit))
335    }
336    fn reserve<T>(&mut self, values: &mut Vec<T>, additional: usize) -> Result<(), Error> {
337        let required = values
338            .len()
339            .checked_add(additional)
340            .ok_or(Error::Overflow)?;
341        if required <= values.capacity() {
342            return Ok(());
343        }
344        let target = required
345            .max(values.capacity().checked_mul(2).ok_or(Error::Overflow)?)
346            .max(4);
347        self.check(self.storage_bytes() + target as u128 * size_of::<T>() as u128)?;
348        let old = values.capacity() as u128 * size_of::<T>() as u128;
349        values
350            .try_reserve_exact(target - values.len())
351            .map_err(|_| Error::Allocation)?;
352        let actual = values.capacity() as u128 * size_of::<T>() as u128;
353        self.check(self.storage_bytes() + actual)?;
354        self.metadata += actual - old;
355        self.limit_store()
356    }
357    fn check(&self, required: u128) -> Result<(), Error> {
358        if required > self.limit as u128 {
359            Err(Error::Storage {
360                required,
361                limit: self.limit,
362            })
363        } else {
364            Ok(())
365        }
366    }
367    fn limit_store(&mut self) -> Result<(), Error> {
368        let metadata = self.metadata + self.external;
369        let available = (self.limit as u128)
370            .checked_sub(metadata)
371            .and_then(|bytes| usize::try_from(bytes).ok())
372            .ok_or(Error::Storage {
373                required: self.store.current_bytes() + metadata,
374                limit: self.limit,
375            })?;
376        self.store
377            .ceiling(available)
378            .map_err(|error| storage_error(error, metadata, self.limit))
379    }
380}
381fn failure_error(failure: Failure<Infallible>, metadata: u128, limit: usize) -> Error {
382    match failure {
383        Failure::Storage(error) => storage_error(error, metadata, limit),
384        Failure::Stopped(never) => match never {},
385    }
386}
387fn storage_error(error: Error, metadata: u128, limit: usize) -> Error {
388    match error {
389        Error::Storage { required, .. } => Error::Storage {
390            required: required + metadata,
391            limit,
392        },
393        other => other,
394    }
395}
