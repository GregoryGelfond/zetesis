11/// Canonical component metadata could not be completed or bound.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub enum TemplateCatalogFailure<E = Infallible> {
14    /// A constant, signature, constructor or binding uses an incompatible prefix.
15    Read(ReadError),
16    /// Named metadata capacity or allocation could not be admitted.
17    Storage(Error),
18    /// The caller refused before the next metadata operation.
19    Stopped(E),
20    /// An earlier mutation failed; incomplete components cannot be exposed.
21    Incomplete,
22}
23impl<E: fmt::Display> fmt::Display for TemplateCatalogFailure<E> {
24    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
25        match self {
26            Self::Read(error) => error.fmt(formatter),
27            Self::Storage(error) => error.fmt(formatter),
28            Self::Stopped(error) => error.fmt(formatter),
29            Self::Incomplete => formatter.write_str("canonical component metadata is incomplete"),
30        }
31    }
32}
33impl<E: std::error::Error + 'static> std::error::Error for TemplateCatalogFailure<E> {}
34impl<E> From<Error> for TemplateCatalogFailure<E> {
35    fn from(error: Error) -> Self {
36        Self::Storage(error)
37    }
38}
39impl<E> From<ReadError> for TemplateCatalogFailure<E> {
40    fn from(error: ReadError) -> Self {
41        Self::Read(error)
42    }
43}

45/// Ordered template metadata naming one existing canonical vocabulary.
46///
47/// This owner imports no values or signatures. Constants and predicates must
48/// already belong to each supplied reader. It retains one identity witness and
49/// local metadata, never a writer, snapshot, or per-cell reference-counted key.
50/// Publication binds all rows to one exact supplied immutable atom catalog.
51/// Neither rows nor that catalog establish logical support or variable safety.
52///
53/// Metadata allowances cover this header, actual row/argument capacities and
54/// old/replacement allocation overlap. Shared source storage, caller input and
55/// allocator/Arc bookkeeping are excluded; source and caller storage must be
56/// charged separately by the caller.
57/// Every append failure prevents publication, while retained capacity and the
58/// largest actual overlap remain inspectable for the caller's storage ledger.
59/// A caller panic also poisons publication. After a caught panic, receipts may
60/// conservatively retain charges for abandoned private buffers.
61#[derive(Debug)]
62pub struct TemplateCatalogSelection {
63    metadata: Metadata,
64    rows: Vec<RowData>,
65    complete: usize,
66}
67
68impl TemplateCatalogSelection {
69    /// Start an empty selection carrying this vocabulary's identity witness.
70    /// Even an empty selection must later bind to that same vocabulary.
71    ///
72    /// # Errors
73    /// Refuses a metadata allowance smaller than the empty owner header.
74    pub fn new(
75        read: CatalogRead<'_>,
76        max_metadata_bytes: usize,
77    ) -> Result<Self, TemplateCatalogFailure> {
78        Ok(Self {
79            metadata: Metadata::new(read, size_of::<Self>(), max_metadata_bytes)?,
80            rows: Vec::new(),
81            complete: 0,
82        })
83    }
84
85    /// Number of completed appends. An interrupted final row is never included;
86    /// after a caller panic the poisoned owner cannot expose its earlier rows.
87    #[must_use]
88    pub fn len(&self) -> usize {
89        self.complete
90    }
91    /// Whether no complete row has been admitted.
92    #[must_use]
93    pub fn is_empty(&self) -> bool {
94        self.complete == 0
95    }
96    /// Current header and metadata capacity, including a retained partial row.
97    /// Canonical source payload and indexes are not owned or included here.
98    #[must_use]
99    pub fn storage_bytes(&self) -> u128 {
100        self.metadata.bytes
101    }
102    /// Greatest actual retained or replacement-overlap metadata envelope since
103    /// construction or the latest `restart_storage_peak`. Refused proposals that
104    /// allocated nothing do not increase this receipt.
105    #[must_use]
106    pub fn storage_peak_bytes(&self) -> u128 {
107        self.metadata.peak
108    }
109
110    /// Start a new measurement interval at current retained metadata. Call before
111    /// composing an operation's peak with other storage whose lifetime may have
112    /// changed since the previous operation. No allocation or payload read occurs.
113    pub fn restart_storage_peak(&mut self) {
114        self.metadata.peak = self.metadata.bytes;
115    }
116
117    /// Named overlap required while replacing this selection's header with the
118    /// published component header. Shared metadata buffers are counted once.
119    /// This is a proposed envelope, not an observed allocation receipt: record
120    /// it as an actual peak only after `finish_with` successfully publishes.
121    #[must_use]
122    pub fn publication_peak_bytes(&self) -> u128 {
123        self.metadata.bytes + size_of::<Data>() as u128
124    }
125
126    /// Append an original-order row of already canonical fields. Variables and
127    /// repeated occurrences retain their exact supplied positions. Charges precede
128    /// input iterator advances, canonical field reads, metadata copies, existing
129    /// cell relocation and allocation. Caller iterator work is not inspected.
130    /// Successful append returns this selection's local row coordinate.
131    ///
132    /// # Errors
133    /// Refuses ingress constants/signatures, foreign owners, inaccessible input
134    /// prefixes, metadata limits, allocation or caller work. Every failure poisons
135    /// publication; no incomplete logical row can subsequently escape.
136    pub fn append_with<'a, E>(
137        &mut self,
138        read: CatalogRead<'_>,
139        terms: impl IntoIterator<Item = TemplateTerm<'a>>,
140        patterns: impl IntoIterator<Item = PatternRef<'a>>,
141        filters: impl IntoIterator<Item = FilterRef<'a>>,
142        max_metadata_bytes: usize,
143        mut before: impl FnMut() -> Result<(), E>,
144    ) -> Result<usize, TemplateCatalogFailure<E>> {
145        self.metadata.begin(read, max_metadata_bytes, &mut before)?;
146        let position = self.append(
147            read,
148            terms,
149            patterns,
150            filters,
151            max_metadata_bytes,
152            &mut before,
153        )?;
154        self.metadata.complete();
155        Ok(position)
156    }
157
158    fn append<'a, E>(
159        &mut self,
160        read: CatalogRead<'_>,
161        terms: impl IntoIterator<Item = TemplateTerm<'a>>,
162        patterns: impl IntoIterator<Item = PatternRef<'a>>,
163        filters: impl IntoIterator<Item = FilterRef<'a>>,
164        limit: usize,
165        before: &mut impl FnMut() -> Result<(), E>,
166    ) -> Result<usize, TemplateCatalogFailure<E>> {
167        self.metadata.reserve(&mut self.rows, 1, limit, before)?;
168        // Attach the empty row before growing nested buffers, so actual capacities
169        // survive ordinary refusal and remain represented by the shared ledger.
170        before().map_err(TemplateCatalogFailure::Stopped)?;
171        self.rows.push(RowData::default());
172        let row = self.rows.last_mut().expect("just appended row");
173        self.metadata
174            .terms(&mut row.terms, read, terms, limit, before)?;
175        let mut patterns = patterns.into_iter();
176        while let Some(pattern) = next(&mut patterns, before)? {
177            self.metadata
178                .pattern(&mut row.patterns, read, pattern, limit, before)?;
179        }
180        let mut filters = filters.into_iter();
181        while let Some(filter) = next(&mut filters, before)? {
182            self.metadata
183                .filter(&mut row.filters, read, filter, limit, before)?;
184        }
185        before().map_err(TemplateCatalogFailure::Stopped)?;
186        let position = self.complete;
187        self.complete += 1;
188        Ok(position)
189    }
190
191    /// Bind all complete metadata to this exact final source catalog. The source
192    /// may be a later compatible prefix; its occurrences are not template rows.
193    /// No snapshot is rebuilt and no payload or lookup index is copied.
194    /// The metadata limit admits the old header and new publication envelope
195    /// together. Arc envelope allocation keeps Rust's infallible boundary.
196    ///
197    /// # Errors
198    /// Refuses a prior failed append, foreign vocabulary, missing referenced
199    /// identity, publication allowance or caller stop. No partial catalog escapes.
200    pub fn finish_with<E>(
201        self,
202        source: AtomCatalog,
203        max_metadata_bytes: usize,
204        mut before: impl FnMut() -> Result<(), E>,
205    ) -> Result<TemplateCatalog, TemplateCatalogFailure<E>> {
206        self.metadata.bind(source.read(), &mut before)?;
207        check(self.publication_peak_bytes(), max_metadata_bytes)?;
208        before().map_err(TemplateCatalogFailure::Stopped)?;
209        let metadata_bytes =
210            self.metadata.bytes - size_of::<Self>() as u128 + size_of::<Data>() as u128;
211        Ok(TemplateCatalog(Arc::new(Data {
212            source: Source::Selected(source),
213            rows: self.rows,
214            metadata_bytes,
215        })))
216    }
217}
