3use std::fmt;
4use std::io;
5use std::sync::atomic::{AtomicBool, Ordering};
6use std::sync::{Arc, Condvar, Mutex, PoisonError};
7use std::thread;
8use std::time::Instant;

10/// Shared cancellation and an optional immutable deadline. Clones observe the
11/// same flags. A deadline is observed the way cancellation is: a timer thread
12/// sets an expiry flag when the deadline passes, and a poll reads flags only,
13/// never the clock. The timer retires at the deadline or when the last clone
14/// drops, whichever comes first.
15#[derive(Clone, Debug, Default)]
16pub struct Cancellation {
17    cancelled: Arc<AtomicBool>,
18    deadline: Option<Arc<DeadlineOwner>>,
19}
20
21impl Cancellation {
22    /// A fresh cancellation handle with the specified absolute deadline. A deadline that
23    /// has already passed is expired at once and starts no thread.
24    ///
25    /// # Errors
26    /// Returns the operating system's refusal to start the timer thread.
27    pub fn with_deadline(deadline: Instant) -> io::Result<Self> {
28        Ok(Self {
29            deadline: Some(Arc::new(DeadlineOwner::arm(deadline)?)),
30            ..Self::default()
31        })
32    }
33
34    /// Request cancellation through this handle and every clone of it.
35    pub fn cancel(&self) {
36        self.cancelled.store(true, Ordering::Relaxed);
37    }
38
39    /// Observe cancellation and the deadline at a bounded work boundary.
40    /// At most two relaxed loads; no clock is read.
41    ///
42    /// # Errors
43    /// Returns cancellation first, otherwise an expired deadline.
44    #[inline]
45    pub fn poll(&self) -> Result<(), Stop> {
46        if self.cancelled.load(Ordering::Relaxed) {
47            Err(Stop::Cancelled)
48        } else if self
49            .deadline
50            .as_ref()
51            .is_some_and(|owner| owner.deadline.expired.load(Ordering::Relaxed))
52        {
53            Err(Stop::Deadline)
54        } else {
55            Ok(())
56        }
57    }
58}
59
60/// The flags a timer sets and the wake-up it waits on.
61#[derive(Debug)]
62struct Deadline {
63    at: Instant,
64    expired: AtomicBool,
65    /// True once every cancellation handle has dropped; the timer then exits early.
66    retired: Mutex<bool>,
67    wake: Condvar,
68}
69
70/// Held by the cancellation handles, not by the timer, so that dropping the last
71/// clone is observable as the owner's drop.
72#[derive(Debug)]
73struct DeadlineOwner {
74    deadline: Arc<Deadline>,
75}
76
77impl DeadlineOwner {
78    fn arm(at: Instant) -> io::Result<Self> {
79        let deadline = Arc::new(Deadline {
80            at,
81            expired: AtomicBool::new(false),
82            retired: Mutex::new(false),
83            wake: Condvar::new(),
84        });
85        if Instant::now() >= at {
86            deadline.expired.store(true, Ordering::Relaxed);
87        } else {
88            let timer = Arc::clone(&deadline);
89            thread::Builder::new()
90                .name("zetesis-deadline".into())
91                .spawn(move || timer.wait())?;
92        }
93        Ok(Self { deadline })
94    }
95}
96
97impl Drop for DeadlineOwner {
98    fn drop(&mut self) {
99        *self.deadline.lock() = true;
100        self.deadline.wake.notify_all();
101    }
102}
103
104impl Deadline {
105    fn lock(&self) -> std::sync::MutexGuard<'_, bool> {
106        self.retired.lock().unwrap_or_else(PoisonError::into_inner)
107    }
108
109    /// Sleep until the deadline or retirement. Timed waits bound the sleep to
110    /// the remaining time, so a spurious wake-up costs one clock read.
111    fn wait(&self) {
112        let mut retired = self.lock();
113        loop {
114            if *retired {
115                return;
116            }
117            let now = Instant::now();
118            if now >= self.at {
119                self.expired.store(true, Ordering::Relaxed);
120                return;
121            }
122            retired = self
123                .wake
124                .wait_timeout(retired, self.at - now)
125                .unwrap_or_else(PoisonError::into_inner)
126                .0;
127        }
128    }
129}

131/// An incomplete operation or invalid invocation. None denotes a rejected
132/// candidate, exhaustive search, or logical UNSAT.
133#[derive(Clone, Copy, Debug, PartialEq, Eq)]
134pub enum Stop {
135    /// Shared cancellation was observed.
136    Cancelled,
137    /// The absolute deadline expired.
138    Deadline,
139    /// The per-oracle charged work ceiling was reached.
140    WorkLimit,
141    /// Another complete source round would exceed the round ceiling.
142    RoundLimit,
143    /// Named live closure storage would exceed its byte ceiling.
144    StorageLimit,
145    /// Adding another distinct consequence would exceed the atom ceiling.
146    DerivedAtomLimit,
147    /// Producing another candidate would exceed the candidate ceiling.
148    CandidateLimit,
149    /// Discovering another carrier atom would exceed its ceiling.
150    CarrierLimit,
151    /// A fallible vector or carrier allocation was refused.
152    Allocation,
153    /// Candidate and program have different admitted instance identities.
154    WrongProgram,
155    /// An admitted shape failed an internal instantiation invariant.
156    InvalidProgram,
157}
158
159impl fmt::Display for Stop {
160    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
161        f.write_str(match self {
162            Self::Cancelled => "operation cancelled",
163            Self::Deadline => "operation deadline expired",
164            Self::WorkLimit => "oracle work limit reached",
165            Self::RoundLimit => "source round limit reached",
166            Self::StorageLimit => "closure storage limit reached",
167            Self::DerivedAtomLimit => "derived atom limit reached",
168            Self::CandidateLimit => "candidate limit reached",
169            Self::CarrierLimit => "candidate carrier atom limit reached",
170            Self::Allocation => "operation storage could not be reserved",
171            Self::WrongProgram => "candidate belongs to a different admitted program",
172            Self::InvalidProgram => "admitted program violated an instantiation invariant",
173        })
174    }
175}

177impl std::error::Error for Stop {}
178
179impl Stop {
180    /// Canonical construction is an execution refusal, never model absence.
181    /// Identity-space limits remain distinct from invalid admitted shapes.
182    pub(crate) fn catalog(error: &zetesis_core::catalog::Error) -> Self {
183        use zetesis_core::{ValueError, catalog::Error};
184        match error {
185            Error::Allocation
186            | Error::Storage { .. }
187            | Error::Overflow
188            | Error::Value(ValueError::Allocation) => Self::Allocation,
189            Error::IdExhausted | Error::Value(ValueError::Limit { .. }) => Self::CarrierLimit,
190            Error::Shape
191            | Error::FrozenVocabulary
192            | Error::UnindexedVocabulary
193            | Error::VocabularyHasAtoms
194            | Error::CatalogHasBase
195            | Error::Value(ValueError::Shape) => Self::InvalidProgram,
196        }
197    }
198
199    pub(crate) fn model(error: &zetesis_core::ModelError) -> Self {
200        match error {
201            zetesis_core::ModelError::Allocation => Self::Allocation,
202            zetesis_core::ModelError::Bytes { .. } => Self::StorageLimit,
203            zetesis_core::ModelError::Catalog(error) => Self::catalog(error),
204            zetesis_core::ModelError::Position { .. } | zetesis_core::ModelError::Order { .. } => {
205                Self::InvalidProgram
206            }
207        }
208    }
209}

211#[cfg(test)]
212mod tests {
213    use super::*;
214    use std::sync::mpsc;
215    use std::time::Duration;
216
217    fn wait_for_expiry(cancellation: &Cancellation) {
218        let deadline = &cancellation.deadline.as_ref().unwrap().deadline;
219        let timeout = Instant::now() + Duration::from_secs(5);
220        // Observe the flag directly: a boundary poll must not cause the expiry.
221        // The bound detects a stalled timer, not a promised scheduling latency.
222        while !deadline.expired.load(Ordering::Relaxed) {
223            assert!(Instant::now() < timeout, "deadline timer did not expire");
224            thread::sleep(Duration::from_millis(1));
225        }
226    }
227
228    #[test]
229    fn a_deadline_expires_without_polling() {
230        let cancellation =
231            Cancellation::with_deadline(Instant::now() + Duration::from_millis(30)).unwrap();
232        wait_for_expiry(&cancellation);
233        assert_eq!(cancellation.poll(), Err(Stop::Deadline));
234    }
235
236    #[test]
237    fn clones_share_the_deadline() {
238        let cancellation =
239            Cancellation::with_deadline(Instant::now() + Duration::from_millis(30)).unwrap();
240        let clone = cancellation.clone();
241        assert!(Arc::ptr_eq(
242            cancellation.deadline.as_ref().unwrap(),
243            clone.deadline.as_ref().unwrap(),
244        ));
245        wait_for_expiry(&cancellation);
246        assert_eq!(clone.poll(), Err(Stop::Deadline));
247        assert_eq!(cancellation.poll(), Err(Stop::Deadline));
248    }
249
250    #[test]
251    fn dropping_the_last_clone_retires_the_timer() {
252        // A second waiter on the same deadline stands in for the timer thread,
253        // whose handle is not retained: retirement wakes every waiter.
254        let owner = DeadlineOwner::arm(Instant::now() + Duration::from_hours(1)).unwrap();
255        let deadline = Arc::clone(&owner.deadline);
256        let (retired, observer) = mpsc::channel();
257        thread::spawn(move || {
258            deadline.wait();
259            retired.send(()).unwrap();
260        });
261        assert!(observer.recv_timeout(Duration::from_millis(100)).is_err());
262        drop(owner);
263        assert!(observer.recv_timeout(Duration::from_secs(5)).is_ok());
264    }
265
266    #[test]
267    fn retirement_does_not_expire_the_deadline() {
268        let owner = DeadlineOwner::arm(Instant::now() + Duration::from_hours(1)).unwrap();
269        let deadline = Arc::clone(&owner.deadline);
270        drop(owner);
271        deadline.wait();
272        assert!(!deadline.expired.load(Ordering::Relaxed));
273    }
