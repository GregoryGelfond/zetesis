14/// Shared cancellation and an optional immutable deadline. Clones observe the
15/// same flags. A deadline is observed the way cancellation is: a timer thread
16/// sets an expiry flag when the deadline passes, and a poll reads flags only,
17/// never the clock. The timer retires at the deadline or when the last clone
18/// drops, whichever comes first.
19#[derive(Clone, Debug, Default)]
20pub struct Cancellation {
21    cancelled: Arc<AtomicBool>,
22    deadline: Option<Arc<DeadlineOwner>>,
23    slot: Option<slot::Membership>,
24}
25
26impl Cancellation {
27    /// A fresh cancellation handle with the specified absolute deadline. A deadline that
28    /// has already passed is expired at once and starts no thread.
29    ///
30    /// # Errors
31    /// Returns the operating system's refusal to start the timer thread.
32    pub fn with_deadline(deadline: Instant) -> io::Result<Self> {
33        Ok(Self {
34            deadline: Some(Arc::new(DeadlineOwner::arm(deadline)?)),
35            ..Self::default()
36        })
37    }
38
39    /// Request cancellation through this handle and every clone of it.
40    pub fn cancel(&self) {
41        self.cancelled.store(true, Ordering::Relaxed);
42    }
43
44    /// Observe cancellation and the deadline at a bounded work boundary.
45    /// Ordinary handles read at most two relaxed flags; a handle from a
46    /// [`CancellationSlot`] additionally reads its run's membership. No clock
47    /// is read. A retired or superseded run reports cancellation.
48    ///
49    /// # Errors
50    /// Returns cancellation first, otherwise an expired deadline.
51    #[inline]
52    pub fn poll(&self) -> Result<(), Stop> {
53        self.polling().poll()
54    }
55
56    /// Borrow the immutable handle shape once for repeated boundary polls.
57    /// Preparing this view takes constant work and space, without allocation,
58    /// reference-count updates or flag reads. It borrows the underlying flags,
59    /// so cancellation, deadline expiry and slot retirement remain live.
60    /// The borrow keeps the token's storage alive; it does not keep a slot's
61    /// run active or extend its deadline.
62    #[must_use]
63    #[inline]
64    pub fn polling(&self) -> CancellationPoll<'_> {
65        CancellationPoll {
66            cancelled: &self.cancelled,
67            expired: DeadlineOwner::expiry(self.deadline.as_ref()),
68            membership: slot::Membership::polling(self.slot.as_ref()),
69        }
70    }
71}
72
73/// A borrowed view of the flags read by [`Cancellation::poll`].
74///
75/// Tight loops can retain this view to avoid traversing the owning handle at
76/// each boundary. It caches references and the immutable run identity, never
77/// an observed flag value. Copying it neither owns a token nor prolongs a run.
78#[derive(Clone, Copy, Debug)]
79pub struct CancellationPoll<'a> {
80    cancelled: &'a AtomicBool,
81    expired: Option<&'a AtomicBool>,
82    membership: Option<slot::MembershipPoll<'a>>,
83}
84
85impl CancellationPoll<'_> {
86    /// Observe exactly the same flags, in the same order, as the owning token.
87    /// Reads only relaxed atomics; no clock, allocation or lock is involved.
88    ///
89    /// # Errors
90    /// Returns cancellation first, otherwise an expired deadline.
91    #[inline]
92    pub fn poll(&self) -> Result<(), Stop> {
93        if self.cancelled.load(Ordering::Relaxed)
94            || slot::MembershipPoll::is_cancelled(self.membership)
95        {
96            Err(Stop::Cancelled)
97        } else if self
98            .expired
99            .is_some_and(|expired| expired.load(Ordering::Relaxed))
100        {
101            Err(Stop::Deadline)
102        } else {
103            Ok(())
104        }
105    }
106}

108/// The flags a timer sets and the wake-up it waits on.
109#[derive(Debug)]
110struct Deadline {
111    at: Instant,
112    expired: AtomicBool,
113    /// True once every cancellation handle has dropped; the timer then exits early.
114    retired: Mutex<bool>,
115    wake: Condvar,
116}
117
118/// Held by the cancellation handles, not by the timer, so that dropping the last
119/// clone is observable as the owner's drop.
120#[derive(Debug)]
121struct DeadlineOwner {
122    deadline: Arc<Deadline>,
123}
124
125impl DeadlineOwner {
126    /// Borrow an optional timer flag without observing its current value.
127    #[inline]
128    fn expiry(deadline: Option<&Arc<Self>>) -> Option<&AtomicBool> {
129        let owner = deadline?;
130        Some(&owner.deadline.expired)
131    }
132
133    fn arm(at: Instant) -> io::Result<Self> {
134        let deadline = Arc::new(Deadline {
135            at,
136            expired: AtomicBool::new(false),
137            retired: Mutex::new(false),
138            wake: Condvar::new(),
139        });
140        if Instant::now() >= at {
141            deadline.expired.store(true, Ordering::Relaxed);
142        } else {
143            let timer = Arc::clone(&deadline);
144            thread::Builder::new()
145                .name("zetesis-deadline".into())
146                .spawn(move || timer.wait())?;
147        }
148        Ok(Self { deadline })
149    }
150}
151
152impl Drop for DeadlineOwner {
153    fn drop(&mut self) {
154        *self.deadline.lock() = true;
155        self.deadline.wake.notify_all();
156    }
157}
158
159impl Deadline {
160    fn lock(&self) -> std::sync::MutexGuard<'_, bool> {
161        self.retired.lock().unwrap_or_else(PoisonError::into_inner)
162    }
163
164    /// Sleep until the deadline or retirement. Timed waits bound the sleep to
165    /// the remaining time, so a spurious wake-up costs one clock read.
166    fn wait(&self) {
167        let mut retired = self.lock();
168        loop {
169            if *retired {
170                return;
171            }
172            let now = Instant::now();
173            if now >= self.at {
174                self.expired.store(true, Ordering::Relaxed);
175                return;
176            }
177            retired = self
178                .wake
179                .wait_timeout(retired, self.at - now)
180                .unwrap_or_else(PoisonError::into_inner)
181                .0;
182        }
183    }
184}

186/// An incomplete operation or invalid invocation. None denotes a rejected
187/// candidate, exhaustive search, or logical UNSAT.
188#[derive(Clone, Copy, Debug, PartialEq, Eq)]
189pub enum Stop {
190    /// Shared cancellation was observed.
191    Cancelled,
192    /// The absolute deadline expired.
193    Deadline,
194    /// The per-oracle charged work ceiling was reached.
195    WorkLimit,
196    /// Another complete source round would exceed the round ceiling.
197    RoundLimit,
198    /// Named live closure storage would exceed its byte ceiling.
199    StorageLimit,
200    /// Adding another distinct consequence would exceed the atom ceiling.
201    DerivedAtomLimit,
202    /// Producing another candidate would exceed the candidate ceiling.
203    CandidateLimit,
204    /// Discovering another carrier atom would exceed its ceiling.
205    CarrierLimit,
206    /// A fallible vector or carrier allocation was refused.
207    Allocation,
208    /// Candidate and program have different admitted instance identities.
209    WrongProgram,
210    /// An admitted shape failed an internal instantiation invariant.
211    InvalidProgram,
212}
213
214impl fmt::Display for Stop {
215    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
216        f.write_str(match self {
217            Self::Cancelled => "operation cancelled",
218            Self::Deadline => "operation deadline expired",
219            Self::WorkLimit => "oracle work limit reached",
220            Self::RoundLimit => "source round limit reached",
221            Self::StorageLimit => "closure storage limit reached",
222            Self::DerivedAtomLimit => "derived atom limit reached",
223            Self::CandidateLimit => "candidate limit reached",
224            Self::CarrierLimit => "candidate carrier atom limit reached",
225            Self::Allocation => "operation storage could not be reserved",
226            Self::WrongProgram => "candidate belongs to a different admitted program",
227            Self::InvalidProgram => "admitted program violated an instantiation invariant",
228        })
229    }
230}

232impl std::error::Error for Stop {}
233
234impl Stop {
235    /// Canonical construction is an execution refusal, never model absence.
236    /// Identity-space limits remain distinct from invalid admitted shapes.
237    pub(crate) fn catalog(error: &zetesis_core::catalog::Error) -> Self {
238        use zetesis_core::{ValueError, catalog::Error};
239        match error {
240            Error::Allocation
241            | Error::Storage { .. }
242            | Error::Overflow
243            | Error::Value(ValueError::Allocation) => Self::Allocation,
244            Error::IdExhausted | Error::Value(ValueError::Limit { .. }) => Self::CarrierLimit,
245            Error::Shape
246            | Error::FrozenVocabulary
247            | Error::UnindexedVocabulary
248            | Error::VocabularyHasAtoms
249            | Error::CatalogHasBase
250            | Error::Value(ValueError::Shape) => Self::InvalidProgram,
251        }
252    }
253
254    pub(crate) fn model(error: &zetesis_core::ModelError) -> Self {
255        match error {
256            zetesis_core::ModelError::Allocation => Self::Allocation,
257            zetesis_core::ModelError::Bytes { .. } => Self::StorageLimit,
258            zetesis_core::ModelError::Catalog(error) => Self::catalog(error),
259            zetesis_core::ModelError::Position { .. } | zetesis_core::ModelError::Order { .. } => {
260                Self::InvalidProgram
261            }
262        }
263    }
264}

267mod tests {
268    use super::*;
269    use std::sync::mpsc;

272    fn wait_for_expiry(cancellation: &Cancellation) {
273        let deadline = &cancellation.deadline.as_ref().unwrap().deadline;
274        let timeout = Instant::now() + Duration::from_secs(5);
275        // Observe the flag directly: a boundary poll must not cause the expiry.
276        // The bound detects a stalled timer, not a promised scheduling latency.
277        while !deadline.expired.load(Ordering::Relaxed) {
278            assert!(Instant::now() < timeout, "deadline timer did not expire");
279            thread::sleep(Duration::from_millis(1));
280        }
281    }
282
283    #[test]
284    fn a_deadline_expires_without_polling() {
285        let cancellation =
286            Cancellation::with_deadline(Instant::now() + Duration::from_millis(30)).unwrap();
287        let polling = cancellation.polling();
288        wait_for_expiry(&cancellation);
289        assert_eq!(polling.poll(), Err(Stop::Deadline));
290        assert_eq!(cancellation.poll(), Err(Stop::Deadline));
291    }
292
293    #[test]
294    fn clones_share_the_deadline() {
295        let cancellation =
296            Cancellation::with_deadline(Instant::now() + Duration::from_millis(30)).unwrap();
297        let clone = cancellation.clone();
298        assert!(Arc::ptr_eq(
299            cancellation.deadline.as_ref().unwrap(),
300            clone.deadline.as_ref().unwrap(),
301        ));
302        wait_for_expiry(&cancellation);
303        assert_eq!(clone.poll(), Err(Stop::Deadline));
304        assert_eq!(cancellation.poll(), Err(Stop::Deadline));
305    }
306
307    #[test]
308    fn dropping_the_last_clone_retires_the_timer() {
309        // A second waiter on the same deadline stands in for the timer thread,
310        // whose handle is not retained: retirement wakes every waiter.
311        let owner = DeadlineOwner::arm(Instant::now() + Duration::from_hours(1)).unwrap();
312        let deadline = Arc::clone(&owner.deadline);
313        let (retired, observer) = mpsc::channel();
314        thread::spawn(move || {
315            deadline.wait();
316            retired.send(()).unwrap();
317        });
318        assert!(observer.recv_timeout(Duration::from_millis(100)).is_err());
319        drop(owner);
320        assert!(observer.recv_timeout(Duration::from_secs(5)).is_ok());
321    }
322
323    #[test]
324    fn retirement_does_not_expire_the_deadline() {
325        let owner = DeadlineOwner::arm(Instant::now() + Duration::from_hours(1)).unwrap();
326        let deadline = Arc::clone(&owner.deadline);
327        drop(owner);
328        deadline.wait();
329        assert!(!deadline.expired.load(Ordering::Relaxed));
330    }
