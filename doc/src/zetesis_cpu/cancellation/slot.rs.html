1//! A reusable cancellation handle with one generation per run.
2
3use std::fmt;
4use std::io;
5use std::sync::Arc;
6use std::sync::atomic::{AtomicU64, Ordering};
7use std::time::Instant;
8
9use super::{Cancellation, DeadlineOwner};

11// The upper bits identify a generation; the low bits are its explicit status.
12// Generation zero starts inactive. No generation is reused, including after a
13// failed deadline setup, so a delayed compare-exchange cannot encounter ABA.
14const STATUS_MASK: u64 = 3;
15const GENERATION_STEP: u64 = STATUS_MASK + 1;
16const ACTIVE: u64 = 1;
17const CANCELLED: u64 = 2;
18
19/// A reusable handle that cancels only its current run.
20///
21/// Clones address the same slot. [`open`](Self::open) gives each run a fresh
22/// generation and returns its lifetime guard; cloned [`Cancellation`] tokens
23/// observe that run but do not keep its window active. Opening a new run
24/// supersedes the previous one, whose tokens then observe cancellation even
25/// if its guard was leaked. Dropping a guard retires only its own generation.
26///
27/// A pull observed while idle does nothing. A pull racing with a replacement
28/// either cancels the generation it observed or does nothing; it never retries
29/// against the replacement. This is one atomic word, not a pointer to mutable
30/// run state: no lock, relay thread or allocation participates in a pull.
31/// Slot clones share one allocation; opening also allocates the ordinary
32/// cancellation flag, with a deadline's existing timer resources if requested.
33#[derive(Clone, Debug, Default)]
34pub struct CancellationSlot {
35    state: Arc<AtomicU64>,
36}
37
38impl CancellationSlot {
39    /// Open a fresh run, optionally bounded by an absolute deadline.
40    ///
41    /// The slot becomes active before its flag is allocated or the deadline
42    /// timer is armed, so a pull during setup is retained. A failed timer setup
43    /// retires this run. Superseded
44    /// tokens report cancellation; no older guard can retire the new run.
45    /// The guard must stay alive until completion or abandonment of the run.
46    ///
47    /// Opening uses constant space and retries its atomic transition only
48    /// when another operation changed the slot; concurrent openings can
49    /// therefore increase its work. Deadline setup has the same costs as
50    /// [`Cancellation::with_deadline`].
51    ///
52    /// # Errors
53    /// Returns [`CancellationSlotError::GenerationExhausted`] after all
54    /// `2^62 - 1` generations have been used, without altering the current
55    /// run, or [`CancellationSlotError::Deadline`] if its timer cannot start.
56    pub fn open(
57        &self,
58        deadline: Option<Instant>,
59    ) -> Result<CancellationRun, CancellationSlotError> {
60        self.open_with(deadline, DeadlineOwner::arm)
61    }
62
63    fn open_with(
64        &self,
65        deadline: Option<Instant>,
66        arm: impl FnOnce(Instant) -> io::Result<DeadlineOwner>,
67    ) -> Result<CancellationRun, CancellationSlotError> {
68        // Retain pulls before allocating the ordinary flag. The guard then
69        // owns retirement, including a failure while arming the deadline.
70        let active = self.open_generation()?;
71        let cancellation = Cancellation {
72            slot: Some(Membership {
73                state: Arc::clone(&self.state),
74                active,
75            }),
76            ..Cancellation::default()
77        };
78        let mut run = CancellationRun { cancellation };
79        if let Some(at) = deadline {
80            run.cancellation.deadline =
81                Some(Arc::new(arm(at).map_err(CancellationSlotError::Deadline)?));
82        }
83        Ok(run)
84    }
85
86    fn open_generation(&self) -> Result<u64, CancellationSlotError> {
87        let mut observed = self.state.load(Ordering::Relaxed);
88        loop {
89            let generation = (observed & !STATUS_MASK)
90                .checked_add(GENERATION_STEP)
91                .ok_or(CancellationSlotError::GenerationExhausted)?;
92            let active = generation | ACTIVE;
93            match self.state.compare_exchange(
94                observed,
95                active,
96                Ordering::Relaxed,
97                Ordering::Relaxed,
98            ) {
99                Ok(_) => return Ok(active),
100                Err(current) => observed = current,
101            }
102        }
103    }
104
105    /// Request cancellation of the active run, if any. Repeated pulls are
106    /// idempotent; a pull while idle has no effect on a later run.
107    ///
108    /// Constant work: one relaxed load and at most one strong compare-exchange,
109    /// with no retry, allocation, lock or wait. If the observed run changes
110    /// before the exchange, the pull does nothing to its replacement.
111    pub fn cancel(&self) {
112        self.cancel_observed(self.state.load(Ordering::Relaxed));
113    }
114
115    fn cancel_observed(&self, observed: u64) {
116        if observed & STATUS_MASK == ACTIVE {
117            let _ = self.state.compare_exchange(
118                observed,
119                (observed & !STATUS_MASK) | CANCELLED,
120                Ordering::Relaxed,
121                Ordering::Relaxed,
122            );
123        }
124    }
125
126    /// Retire the currently observed run, for example when its backend drops.
127    /// Its tokens then report cancellation and idle pulls affect no later run.
128    /// A concurrent replacement is not retired. Constant work: one load and
129    /// at most two strong compare-exchanges, with no allocation, lock or wait.
130    pub fn clear(&self) {
131        retire(&self.state, self.state.load(Ordering::Relaxed));
132    }
133}

135/// The owner of one slot's active window.
136///
137/// Dropping this guard retires only its run, whose tokens subsequently report
138/// cancellation. The guard is not cloned: token clones do not own the window.
139/// Superseding this guard by opening another run stops its tokens as well.
140/// Retirement makes at most two atomic attempts and cannot retire a newer
141/// generation. Dropping the last deadline owner also wakes its timer through
142/// the existing deadline retirement mechanism.
143#[derive(Debug)]
144#[must_use = "dropping the guard retires the run's cancellation window"]
145pub struct CancellationRun {
146    cancellation: Cancellation,
147}
148
149impl CancellationRun {
150    /// The ordinary cancellation token for this run. Clone it for workers or
151    /// owned sessions while retaining the guard until the whole run closes.
152    /// Borrowing is constant work; cloning shares its flags and membership.
153    #[must_use]
154    pub fn cancellation(&self) -> &Cancellation {
155        &self.cancellation
156    }
157}
158
159impl Drop for CancellationRun {
160    fn drop(&mut self) {
161        let membership = self
162            .cancellation
163            .slot
164            .as_ref()
165            .expect("a slot run retains its membership");
166        retire(&membership.state, membership.active);
167    }
168}

170/// Why a run's cancellation window could not be opened.
171#[derive(Debug)]
172pub enum CancellationSlotError {
173    /// Every representable generation was used; identities are never reused.
174    GenerationExhausted,
175    /// The operating system refused to start the deadline timer.
176    Deadline(io::Error),
177}
178
179impl fmt::Display for CancellationSlotError {
180    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
181        match self {
182            Self::GenerationExhausted => f.write_str("cancellation slot generations exhausted"),
183            Self::Deadline(error) => write!(f, "cancellation deadline timer: {error}"),
184        }
185    }
186}
187
188impl std::error::Error for CancellationSlotError {
189    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
190        match self {
191            Self::GenerationExhausted => None,
192            Self::Deadline(error) => Some(error),
193        }
194    }
195}

197/// Immutable run identity alongside the shared slot, held by ordinary tokens.
198/// These atomic reads signal stopping only; they publish no other mutable data,
199/// so relaxed ordering suffices throughout the slot's state machine.
200#[derive(Clone, Debug)]
201pub(super) struct Membership {
202    state: Arc<AtomicU64>,
203    active: u64,
204}
205
206impl Membership {
207    pub(super) fn is_cancelled(&self) -> bool {
208        self.state.load(Ordering::Relaxed) != self.active
209    }
210}

212/// Retire this generation, whether a pull already cancelled it or races the
213/// first exchange. Within one generation only active -> cancelled can race
214/// retirement, so a second attempt suffices. Neither attempt names a later
215/// generation; monotonic identities rule out ABA.
216fn retire(state: &AtomicU64, observed: u64) {
217    let generation = observed & !STATUS_MASK;
218    if state
219        .compare_exchange(
220            generation | ACTIVE,
221            generation,
222            Ordering::Relaxed,
223            Ordering::Relaxed,
224        )
225        .is_err_and(|current| current == (generation | CANCELLED))
226    {
227        let _ = state.compare_exchange(
228            generation | CANCELLED,
229            generation,
230            Ordering::Relaxed,
231            Ordering::Relaxed,
232        );
233    }
234}
