1//! Bounded, world-isolated reduct rounds with an injected chunk evaluator.
2//!
3//! Every chunk reads the same immutable round snapshots. Only complete per-world
4//! source coverage and successful evaluation of its final chunk permit a commit.
5//! A new round rebuilds source relations from the union of derived atoms; seed
6//! atoms remain separate. The union offers instances and never establishes truth
7//! in an individual world.
8//! One appendable catalog owns demanded atom rows over the Program's shared
9//! immutable term/predicate vocabulary. Each source round borrows its
10//! committed prefix through canonical local IDs while callbacks append only to
11//! a disjoint tail. Identity publication does not establish consequence truth;
12//! every offered identity is committed even when the last round adds no truth.
13

40/// Which positive source bindings may be offered within each immutable round.
41/// Every evaluator still checks per-world positives and all frozen gates.

43pub enum SourceSelection {

52/// Explicit batch, source, catalog and transport limits for lazy rounds.
53#[derive(Clone, Copy, Debug)]
54pub struct Limits {
55    /// Maximum submitted seed occurrences, including duplicates.
56    pub max_candidates: usize,
57    /// Maximum demanded catalog atoms; no complete carrier is enumerated.
58    pub max_atoms: usize,
59    /// Maximum completed or pending round scans.
60    pub max_rounds: u64,
61    /// Shared source work across the entire batch, never reset per chunk/round.
62    /// Includes seed interning, checked identity comparisons/metadata, canonical
63    /// row selection and committing newly demanded identities. Transport packing
64    /// and backend-private evaluation retain their separate accounting contracts.
65    pub max_source_work: u64,
66    /// Maximum instances per injected evaluation.
67    pub max_chunk_rules: usize,
68    /// Maximum encoded instance words, excluding the offset table.
69    pub max_chunk_words: usize,
70    /// Logical key metadata and referenced encoding allowance for one source
71    /// instance, reserved separately from catalog growth throughout every scan.
72    /// Keys borrow terms; this conservative reservation includes their metadata.
73    pub max_instance_bytes: usize,
74    /// Independent named source-scan workspace allowance: borrowed row handles,
75    /// assignment/cursor/undo buffers and one offered instance's key capacity.
76    /// Program/catalog payload, world masks and callback transport remain under
77    /// their existing owners and bounds; this is separate from `max_host_bytes`.
78    pub max_scan_bytes: usize,
79    /// Maximum admitted host storage envelope: actual committed/pending catalog,
80    /// tuple metadata, AVL/path and ordered-ID capacities, requested
81    /// packed truth/seed/delta/chunk/result storage, source membership/workspace
82    /// and requested final model-selection slots. Catalog and mask growth
83    /// include their named old/new overlap. Source rows borrow catalog atoms.
84    /// The immutable Program vocabulary and lookup indexes are shared and counted
85    /// with the input Program, not charged again per batch. This includes unused
86    /// input terms; no full carrier is materialized. Input program/seeds, Arc
87    /// reference counters, allocator overhead, final selection
88    /// Vec capacity beyond its requested slots, rounding outside the catalog and
89    /// ordered-ID capacities, and backend-private transport are excluded. This
90    /// bound is not RSS; a backend bounds its transport separately.
91    pub max_host_bytes: usize,
92}

110/// Exact progress, also retained after source or evaluator failure.
111#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
112pub struct Progress {
113    /// Rounds with complete per-world source coverage and evaluation.
114    pub rounds: u64,
115    /// Source and demanded-identity operation charges, including interrupted

117    pub source_work: u64,
118    /// Filter-valid instances offered by source scans.
119    pub instances: u64,
120    /// Successfully evaluated nonempty chunks.
121    pub chunks: u64,
122    /// Distinct demanded atom identities, including seed and underived heads.
123    pub catalog_atoms: usize,

125    /// This is a subset of `source_work`, not additional work to add to it.

128    /// Unvisited extensions are not claimed as enumerated instances.
129    pub pruned_prefixes: u64,
130    /// Peak retained ordered source-ID capacity plus requested membership/frame
131    /// payload, including join storage retained between source rounds. Interner

134    pub peak_mask_bytes: usize,
135}

261/// Immutable dense input to one consequence step. IDs remain stable throughout
262/// the batch. Every round starts from complete snapshots and separate seeds.
263///
264/// Records contain `head + 1` (zero denotes a constraint), three antecedent
265/// lengths, then positive, true-gate and false-gate IDs. The offset table indexes
266/// record starts. All indices and lengths are validated before construction.
267/// A backend evaluates each record against one world's snapshot and frozen seed,
268/// emits only newly true heads, and sets its constraint word to zero or one.
269pub struct Chunk<'a> {
270    round_index: u64,
271    words: usize,
272    worlds: usize,
273    catalog_atoms: usize,
274    snapshots: &'a [u32],
275    seeds: &'a [u32],
276    offsets: &'a [u32],
277    records: &'a [u32],
278}
279
280impl Chunk<'_> {
281    /// Zero-based immutable-round index within this invoking batch only. It is
282    /// not a globally unique identity: a new batch starts at zero. Snapshots
283    /// remain unchanged within this round except for zero-filled width growth;
284    /// seeds remain frozen throughout the batch. A cache must own its batch
285    /// boundary and include the current layout when reusing either input.
286    #[must_use]
287    pub const fn round_index(&self) -> u64 {
288        self.round_index
289    }
290    /// Current bitset width per world, grown only as catalog IDs are demanded.
291    #[must_use]
292    pub const fn words(&self) -> usize {
293        self.words
294    }
295    /// Exact seed occurrences, including duplicates.
296    #[must_use]
297    pub const fn worlds(&self) -> usize {
298        self.worlds
299    }
300    /// Words per returned world record: derived-head delta and violation flag.
301    #[must_use]
302    pub const fn result_words(&self) -> usize {
303        self.words + 1
304    }
305    /// Constraint-violation offset within one returned world record.
306    #[must_use]
307    pub const fn violation_offset(&self) -> usize {
308        self.words
309    }
310    /// Number of currently assigned stable atom IDs.
311    #[must_use]
312    pub const fn catalog_atoms(&self) -> usize {
313        self.catalog_atoms
314    }
315    /// Immutable positive truth, world-major with [`Self::words`] stride.
316    #[must_use]
317    pub const fn snapshots(&self) -> &[u32] {
318        self.snapshots
319    }
320    /// Immutable gate truth, with the same stride as snapshots.
321    #[must_use]
322    pub const fn seeds(&self) -> &[u32] {
323        self.seeds
324    }
325    /// Start offsets in [`Self::records`], one for each offered instance.
326    #[must_use]
327    pub const fn offsets(&self) -> &[u32] {
328        self.offsets
329    }
330    /// Validated linear rule encoding described by [`Chunk`].
331    #[must_use]
332    pub const fn records(&self) -> &[u32] {
333        self.records
334    }
335}

337/// Evaluate a chunk with exact scalar integer operations. The returned layout
338/// has `words + 1` entries per world: head delta followed by constraint violation.
339/// This is the portable reference for device transport tests; it never mutates
340/// snapshots or seeds. Work is O(worlds * encoded antecedents), and returned
341/// storage is O(worlds * words), bounded when the chunk was constructed.
342///
343/// # Errors
344/// Returns allocation failure before output construction.
345pub fn evaluate(chunk: &Chunk<'_>) -> Result<Vec<u32>, Stop> {
346    let mut output = zeros(chunk.worlds * chunk.result_words())?;
347    for world in 0..chunk.worlds {
348        let result = &mut output[world * chunk.result_words()..][..chunk.result_words()];
349        evaluate_world(chunk, world, result, &mut |_| Ok(()))?;
350    }
351    Ok(output)
352}

354#[derive(Clone, Copy)]
355enum EvaluationStep {
356    Instance,
357    Antecedent,
358}
359
360/// One world's pure consequence relation, with an injected work boundary.
361/// Output is private until every world and the source round have completed.
362fn evaluate_world(
363    chunk: &Chunk<'_>,
364    world: usize,
365    result: &mut [u32],
366    step: &mut impl FnMut(EvaluationStep) -> Result<(), Stop>,
367) -> Result<(), Stop> {
368    let snapshot = &chunk.snapshots[world * chunk.words..][..chunk.words];
369    let seed = &chunk.seeds[world * chunk.words..][..chunk.words];
370    for offset in chunk.offsets {
371        step(EvaluationStep::Instance)?;
372        let record = &chunk.records[*offset as usize..];
373        let counts = [record[1] as usize, record[2] as usize, record[3] as usize];
374        let mut start = RECORD_HEADER_WORDS;
375        let mut enabled = true;
376        for (count, truth, required) in [
377            (counts[0], snapshot, true),
378            (counts[1], seed, true),
379            (counts[2], seed, false),
380        ] {
381            for atom in &record[start..start + count] {
382                step(EvaluationStep::Antecedent)?;
383                enabled &= contains(truth, *atom as usize) == required;
384            }
385            start += count;
386        }
387        if enabled {
388            if record[0] == CONSTRAINT_HEAD {
389                result[chunk.violation_offset()] = 1;
390            } else {
391                let head = (record[0] - 1) as usize;
392                if !contains(snapshot, head) {
393                    insert(result, head);
394                }
395            }
396        }
397    }
398    Ok(())
399}

401/// Compute complete per-world reduct closures by bounded immutable rounds.
402/// `execute` must implement [`evaluate`]'s consequence relation exactly; shape
403/// validation cannot establish gate truth or semantic correctness of an injected
404/// backend. Rust/WGSL refinement is an explicit proof and qualification boundary.
405///
406/// No completed checks are returned on any incomplete scan or execution failure.
407/// Atom IDs grow only when seeds or source instances demand them. All seed atoms
408/// participate in the final projection check, even if no rule derives them.
409/// Empty batches perform no source or evaluator work. Catalog/index, ordered-ID
410/// and transport vector reservations return typed allocation failures. Borrowed
411/// source-relation grouping and shared ownership envelopes retain their
412/// infallible allocation contracts. Source instances borrow canonical terms;
413/// their checked key-vector reservations are fallible.
414/// Completed worlds share one final atom catalog and retain their own selected
415/// positions. This is a logical payload bound, not a process-RSS guarantee.
416///
417/// # Errors
418/// Returns source, evaluator or malformed-output failure with retained progress.
419pub fn check_with<E>(
420    program: &Program,
421    seeds: &[Seed],
422    limits: Limits,
423    cancellation: &Cancellation,
424    execute: impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
425) -> Result<Batch, Failure<E>> {
426    check_with_views(
427        program,
428        seeds.iter().map(Seed::view),
429        limits,
430        cancellation,
431        execute,
432    )
433}

435/// Check borrowed candidate views with the same immutable-round protocol as
436/// [`check_with`]. Cloning the iterator must preserve its length, order and
437/// candidate identities. The views borrow their true atoms; this boundary
438/// creates no owned seed or temporary view vector. Demanded atom rows use the
439/// Program's canonical vocabulary; round selections own only IDs. The
440/// final catalog transfers to shared result models without copying atom payloads.
441///
442/// # Errors
443/// Returns the same failures and retained progress as [`check_with`]. Actual
444/// occurrence counts inconsistent with the claimed size refuse as
445/// [`Stop::InvalidProgram`] before an out-of-range world write or publication.
446pub fn check_with_views<'seed, E>(
447    program: &Program,
448    seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
449    limits: Limits,
450    cancellation: &Cancellation,
451    execute: impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
452) -> Result<Batch, Failure<E>> {
453    check_with_source_views(
454        program,
455        seeds,
456        limits,
457        SourceSelection::Union,
458        cancellation,
459        execute,
460    )
461}

463/// Check the same per-world reduct closures using an explicit source selection.
464/// [`SourceSelection::Worlds`] retains complete coverage of each current world,
465/// but deliberately does not exhaust every cross-world union combination.
466///
467/// Identity preparation, mask construction/intersections and committing newly
468/// demanded identities consume the batch's cumulative source work.
469/// Join frames/indices are reused within the batch; membership is rebuilt from
470/// each immutable round. Live storage is charged against `max_host_bytes` during
471/// catalog growth and between rounds.
472/// This optional optimization can change resource stopping points. The injected
473/// evaluator's exactness contract is identical to [`check_with`].
474///
475/// # Errors
476/// Returns charged progress and no completed checks after any source, resource,
477/// evaluator or protocol failure, including failed mask construction.
478pub fn check_with_source<E>(
479    program: &Program,
480    seeds: &[Seed],
481    limits: Limits,
482    selection: SourceSelection,
483    cancellation: &Cancellation,
484    execute: impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
485) -> Result<Batch, Failure<E>> {
486    check_with_source_views(
487        program,
488        seeds.iter().map(Seed::view),
489        limits,
490        selection,
491        cancellation,
492        execute,
493    )
494}

496/// Check borrowed candidates under the selected source traversal. The iterator
497/// contract and allocation boundary are those of [`check_with_views`]; source
498/// work, complete-world coverage and failure progress are unchanged from
499/// [`check_with_source`].
500///
501/// # Errors
502/// Returns no complete checks after any source, resource or evaluator failure.
503pub fn check_with_source_views<'seed, E>(
504    program: &Program,
505    seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
506    limits: Limits,
507    selection: SourceSelection,
508    cancellation: &Cancellation,
509    mut execute: impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
510) -> Result<Batch, Failure<E>> {
511    let mut progress = Progress::default();
512    run(
513        program,
514        seeds,
515        limits,
516        selection,
517        cancellation,
518        &mut progress,
519        &mut execute,
520    )
521    .map(|checks| Batch { checks, progress })
522    .map_err(|cause| Failure { cause, progress })
523}

555enum SourceSnapshot<'a> {
556    Union(worlds::Rows<'a>),
557    Worlds(worlds::Snapshot<'a>),
558}
559
560impl SourceSnapshot<'_> {
561    fn bytes(&self) -> usize {
562        match self {
563            Self::Union(rows) => rows.bytes(),
564            Self::Worlds(snapshot) => snapshot.bytes(),
565        }
566    }
567
568    fn into_workspace(self) -> Option<worlds::Workspace> {
569        match self {
570            Self::Union(_) => None,
571            Self::Worlds(snapshot) => Some(snapshot.into_workspace()),
572        }
573    }
574
575    fn scan<E>(
576        &mut self,
577        program: &Program,
578        limits: source::ScanLimits,
579        work: &mut Work<'_>,
580        consume: impl FnMut(source::Instance<'_>, &mut Work<'_>) -> Result<(), E>,
581    ) -> Result<source::ScanStatistics, source::ScanFailure<E>> {
582        match self {
583            Self::Union(rows) => source::scan_rows(program, rows.iter(), limits, work, consume),
584            Self::Worlds(snapshot) => source::scan_worlds(program, snapshot, limits, work, consume),
585        }
586    }
587}

589fn prepare_snapshot<'a>(
590    catalog: &'a mut AtomInterner,
591    transport: &Transport,
592    workspace: &mut Option<worlds::Workspace>,
593    program: &Program,
594    selection: SourceSelection,
595    limits: Limits,
596    work: &mut Work<'_>,
597) -> Result<(SourceSnapshot<'a>, AtomAppender<'a>), Stop> {
598    if let Some(retained) = workspace.as_ref() {
599        retained.record_retained(work);
600    }
601    let bounds = transport.catalog_limits(limits, catalog.len())?;
602    let ordered = catalog
603        .ordered_ids_with(bounds, || work.tick())
604        .map_err(|error| intern_stop(&error))?;
605    let available = transport.source_bytes(limits, catalog.storage_bytes(), catalog.len())?;
606    let (committed, appender) = catalog.split();
607    let rows = worlds::Rows::select(
608        committed.atoms(),
609        ordered,
610        &transport.snapshots,
611        transport.words,
612        transport.violated.len(),
613        available,
614        work,
615    )?;
616    let snapshot = match selection {
617        SourceSelection::Union => SourceSnapshot::Union(rows),
618        SourceSelection::Worlds => {
619            let retained = workspace.take().map_or_else(
620                || {
621                    worlds::Workspace::new(
622                        program,
623                        transport.violated.len(),
624                        available.saturating_sub(rows.bytes()),
625                        work,
626                    )
627                },
628                Ok,
629            )?;
630            SourceSnapshot::Worlds(retained.snapshot(
631                rows,
632                &transport.snapshots,
633                transport.words,
634                available,
635                work,
636            )?)
637        }
638    };
639    Ok((snapshot, appender))
640}

642// Clone equivalence remains the caller's iterator contract. Check every actual
643// length before indexed world writes so a bad ExactSizeIterator cannot overrun
644// or silently omit an admitted occurrence.
645fn validate_seeds<'seed>(
646    program: &Program,
647    seeds: impl Iterator<Item = SeedView<'seed>>,
648    expected: usize,
649) -> Result<(), Stop> {
650    let mut count = 0;
651    for seed in seeds {
652        if !seed.program().same_instance(program) {
653            return Err(Stop::WrongProgram);
654        }
655        if count >= expected {
656            return Err(Stop::InvalidProgram);
657        }
658        count += 1;
659    }
660    if count != expected {
661        return Err(Stop::InvalidProgram);
662    }
663    Ok(())
664}

672    progress: &mut Progress,
673    execute: &mut impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
674) -> Result<Vec<Check>, Cause<E>> {
675    cancellation.poll()?;
676    let candidates = seeds.len();
677    validate_seeds(program, seeds.clone(), candidates)?;
678    if candidates == 0 {
679        return Ok(Vec::new());
680    }
681    let mut state = State::new(program, candidates, limits)?;
682    state.initialize(seeds.clone(), candidates, limits, cancellation, progress)?;
683    loop {
684        progress.catalog_atoms = state.catalog.len();
685        if progress.rounds >= limits.max_rounds {
686            return Err(Stop::RoundLimit.into());
687        }
688        cancellation.poll()?;
689        state.transport.pending.fill(0);
690        let scanned = state.scan(program, selection, limits, cancellation, progress, execute);
691        let source_statistics = match scanned {
692            Ok(statistics) => statistics,
693            Err(failure) => {
694                progress.record_source(failure.statistics);
695                progress.catalog_atoms = state.catalog.len();
696                return Err(match failure.cause {
697                    source::ScanCause::Source(stop) => Cause::Source(stop),
698                    source::ScanCause::Consumer(cause) => cause,
699                });
700            }
701        };
702        progress.record_source(source_statistics);
703        progress.catalog_atoms = state.catalog.len();
704        state.transport.flush(
705            state.catalog.len(),
706            &mut Evaluation {
707                cancellation,
708                progress,
709                execute,
710            },
711        )?;
712        // Identity may grow even when no consequence does. Freeze every offered
713        // identity before testing convergence, including a final zero-delta round.
714        state.commit(limits, cancellation, progress)?;
715        cancellation.poll()?;
716        progress.rounds += 1;
717        let mut grew = false;
718        for (old, delta) in state
719            .transport
720            .snapshots
721            .iter_mut()
722            .zip(&state.transport.pending)
723        {
724            grew |= delta & !*old != 0;
725            *old |= delta;
726        }
727        if !grew {
728            break;
729        }
730    }
731    progress.catalog_atoms = state.catalog.len();
732    state
733        .conclusions(program, seeds, limits, cancellation, progress)
734        .map_err(Cause::Source)
735}

737impl State {
738    fn new(program: &Program, candidates: usize, limits: Limits) -> Result<Self, Stop> {
739        let transport = Transport::new(candidates, limits, program.shared_vocabulary_bytes())?;
740        let allowance = transport.catalog_limits(limits, 0)?.max_bytes;
741        // Core storage ceilings use usize. The full allowance may include a
742        // shared input credit larger than the remaining addressable capacity;
743        // clipping to usize::MAX then remains at most that admitted allowance.
744        let catalog =
745            AtomInterner::for_program(program, usize::try_from(allowance).unwrap_or(usize::MAX))
746                .map_err(|error| Stop::catalog(&error))?;
747        let state = Self {
748            catalog,
749            transport,
750            world_workspace: None,
751        };
752        if state.catalog.storage_bytes() > state.transport.catalog_limits(limits, 0)?.max_bytes {
753            return Err(Stop::Allocation);
754        }
755        Ok(state)
756    }
757
758    fn intern<'a>(
759        &mut self,
760        atom: impl Into<AtomRef<'a>>,
761        limits: Limits,
762        work: &mut Work<'_>,
763    ) -> Result<u32, Stop> {
764        let (_, mut appender) = self.catalog.split();
765        self.transport
766            .intern_atom(&mut appender, atom.into(), limits, work)
767    }
768
769    fn initialize<'seed>(
770        &mut self,
771        seeds: impl Iterator<Item = SeedView<'seed>>,
772        candidates: usize,
773        limits: Limits,
774        cancellation: &Cancellation,
775        progress: &mut Progress,
776    ) -> Result<(), Stop> {
777        let mut work = Work::source(
778            cancellation,
779            limits.max_source_work.saturating_sub(progress.source_work),
780        );
781        let result = (|| {
782            let mut packed = 0;
783            for (world, seed) in seeds.enumerate() {
784                if world >= candidates {
785                    return Err(Stop::InvalidProgram);
786                }
787                packed += 1;
788                for atom in seed.atoms() {
789                    let id = self.intern(atom, limits, &mut work)?;
790                    let words = self.transport.words;
791                    insert(
792                        &mut self.transport.seeds[world * words..][..words],
793                        id as usize,
794                    );
795                }
796            }
797            if packed != candidates {
798                return Err(Stop::InvalidProgram);
799            }
800            Ok(())
801        })();
802        progress.record_source(work.source_statistics(0));
803        progress.catalog_atoms = self.catalog.len();
804        result?;
805        self.commit(limits, cancellation, progress)
806    }
807
808    fn commit(
809        &mut self,
810        limits: Limits,
811        cancellation: &Cancellation,
812        progress: &mut Progress,
813    ) -> Result<(), Stop> {
814        let mut work = Work::source(
815            cancellation,
816            limits.max_source_work.saturating_sub(progress.source_work),
817        );
818        let result = self
819            .transport
820            .catalog_limits(limits, self.catalog.len())
821            .and_then(|bounds| {
822                self.catalog
823                    .commit_with(bounds, || work.tick())
824                    .map_err(|error| intern_stop(&error))
825            });
826        progress.record_source(work.source_statistics(0));
827        // Pending published identities count even if vector-transfer admission
828        // fails; a failed commit returns no completed checks or round receipt.
829        progress.catalog_atoms = self.catalog.len();
830        result
831    }
832
833    fn scan<E>(
834        &mut self,
835        program: &Program,
836        selection: SourceSelection,
837        limits: Limits,
838        cancellation: &Cancellation,
839        progress: &mut Progress,
840        execute: &mut impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>,
841    ) -> Result<source::ScanStatistics, source::ScanFailure<Cause<E>>> {
842        let scan_limits = source::ScanLimits {
843            max_work: limits.max_source_work.saturating_sub(progress.source_work),
844            max_instance_atoms: limits
845                .max_chunk_words
846                .saturating_sub(RECORD_HEADER_WORDS - 1),
847            max_instance_bytes: limits.max_instance_bytes,
848            max_scan_bytes: limits.max_scan_bytes,
849        };
850        let mut work = Work::source(cancellation, scan_limits.max_work);
851        let prepared = prepare_snapshot(
852            &mut self.catalog,
853            &self.transport,
854            &mut self.world_workspace,
855            program,
856            selection,
857            limits,
858            &mut work,
859        );
860        let transport = &mut self.transport;
861        let workspace = &mut self.world_workspace;
862        let mut evaluation = Evaluation {
863            cancellation,
864            progress,
865            execute,
866        };
867        match prepared {
868            Ok((mut snapshot, mut appender)) => {
869                transport.source_mask_bytes = snapshot.bytes();
870                let result = snapshot.scan(program, scan_limits, &mut work, |instance, work| {
871                    transport.offer(&mut appender, &instance, limits, work, &mut evaluation)
872                });
873                // Round truth has dropped, but fixed join storage remains live
874                // through flushing, catalog growth and the next snapshot.
875                *workspace = snapshot.into_workspace();
876                transport.source_mask_bytes =
877                    workspace.as_ref().map_or(0, worlds::Workspace::bytes);
878                result
879            }
880            Err(stop) => {
881                // Preparation may stop before taking the retained workspace.
882                transport.source_mask_bytes =
883                    workspace.as_ref().map_or(0, worlds::Workspace::bytes);
884                Err(source::ScanFailure {
885                    cause: source::ScanCause::Source(stop),
886                    statistics: work.source_statistics(0),
887                })
888            }
889        }
890    }
891
892    fn conclusions<'seed>(
893        self,
894        program: &Program,
895        seeds: impl ExactSizeIterator<Item = SeedView<'seed>>,
896        limits: Limits,
897        cancellation: &Cancellation,
898        progress: &mut Progress,
899    ) -> Result<Vec<Check>, Stop> {
900        // Source rounds are complete. Transfer the canonical snapshot and
901        // discovery mapping; lookup metadata drops without copying terms.
902        let bounds = self.transport.catalog_limits(limits, self.catalog.len())?;
903        let mut work = Work::source(
904            cancellation,
905            limits.max_source_work.saturating_sub(progress.source_work),
906        );
907        let catalog = self
908            .catalog
909            .into_catalog_with(bounds, || work.tick())
910            .map_err(|error| intern_stop(&error));
911        progress.record_source(work.source_statistics(0));
912        let catalog = catalog?;
913        let transport = self.transport;
914        let mut checks = Vec::new();
915        checks
916            .try_reserve_exact(transport.violated.len())
917            .map_err(|_| Stop::Allocation)?;
918        for (world, seed) in seeds.enumerate() {
919            cancellation.poll()?;
920            if world >= transport.violated.len() {
921                return Err(Stop::InvalidProgram);
922            }
923            let words = &transport.snapshots[world * transport.words..][..transport.words];
924            let closure = Model::from_positions(
925                &catalog,
926                (0..catalog.atoms().len()).filter(|id| contains(words, *id)),
927            )
928            .map_err(|error| Stop::model(&error))?;
929            let mismatch = closure
930                .atoms()
931                .iter()
932                .any(|atom| program.contains_gate_atom(atom) && !seed.contains(atom))
933                || seed.atoms().any(|atom| !closure.contains(atom));
934            checks.push(Check {
935                program: program.clone(),
936                closure,
937                constraint_violated: transport.violated[world],
938                seed_mismatch: mismatch,
939            });
940        }
941        if checks.len() != transport.violated.len() {
942            return Err(Stop::InvalidProgram);
943        }
944        cancellation.poll()?;
945        Ok(checks)
946    }
947}

949impl Transport {
950    fn new(candidates: usize, limits: Limits, shared_vocabulary_bytes: u128) -> Result<Self, Stop> {
951        if candidates > limits.max_candidates
952            || limits.max_atoms == 0
953            || limits.max_atoms >= u32::MAX as usize
954            || limits.max_chunk_rules == 0
955            || limits.max_chunk_words < RECORD_HEADER_WORDS
956        {
957            return Err(Stop::CarrierLimit);
958        }
959        let words = 1usize;
960        let bits = words.checked_mul(candidates).ok_or(Stop::Allocation)?;
961        let fixed_words = bits
962            .checked_mul(ROUND_MASK_VECTORS)
963            .and_then(|n| candidates.checked_mul(2).and_then(|m| n.checked_add(m)))
964            .and_then(|n| n.checked_add(limits.max_chunk_words))
965            .and_then(|n| n.checked_add(limits.max_chunk_rules))
966            .ok_or(Stop::Allocation)?;
967        let fixed_bytes = fixed_words
968            .checked_mul(size_of::<u32>())
969            .and_then(|bytes| {
970                candidates
971                    .checked_mul(size_of::<Check>())
972                    .and_then(|checks| bytes.checked_add(checks))
973            })
974            .and_then(|bytes| bytes.checked_add(limits.max_instance_bytes))
975            .ok_or(Stop::Allocation)?;
976        if fixed_bytes > limits.max_host_bytes {
977            return Err(Stop::Allocation);
978        }
979        let mut offsets = Vec::new();
980        offsets
981            .try_reserve_exact(limits.max_chunk_rules)
982            .map_err(|_| Stop::Allocation)?;
983        let mut records = Vec::new();
984        records
985            .try_reserve_exact(limits.max_chunk_words)
986            .map_err(|_| Stop::Allocation)?;
987        Ok(Self {
988            fixed_bytes,
989            source_mask_bytes: 0,
990            shared_vocabulary_bytes,
991            words,
992            snapshots: zeros(bits)?,
993            seeds: zeros(bits)?,
994            pending: zeros(bits)?,
995            violated: vec![false; candidates],
996            offsets,
997            records,
998        })
999    }

1001    fn catalog_limits(&self, limits: Limits, atoms: usize) -> Result<InternLimits, Stop> {
1002        let external = self
1003            .external_bytes(atoms)?
1004            .checked_add(self.source_mask_bytes as u128)
1005            .ok_or(Stop::Allocation)?;
1006        let max_bytes = (limits.max_host_bytes as u128)
1007            .checked_sub(external)
1008            .and_then(|bytes| bytes.checked_add(self.shared_vocabulary_bytes))
1009            .ok_or(Stop::Allocation)?;
1010        Ok(InternLimits {
1011            max_atoms: limits.max_atoms,
1012            max_bytes,
1013        })
1014    }
1015
1016    fn external_bytes(&self, atoms: usize) -> Result<u128, Stop> {
1017        // Completed worlds share the catalog. Bound requested selected positions
1018        // by one per catalog atom per world; Model Vec capacity slack is excluded
1019        // from this requested-payload allowance. No symbolic snapshot copy exists.
1020        let selected = atoms
1021            .checked_mul(self.violated.len())
1022            .and_then(|cells| cells.checked_mul(size_of::<usize>()))
1023            .ok_or(Stop::Allocation)?;
1024        Ok(self.fixed_bytes as u128 + selected as u128)
1025    }
1026
1027    fn catalog_bytes(&self, full: u128) -> Result<u128, Stop> {
1028        full.checked_sub(self.shared_vocabulary_bytes)
1029            .ok_or(Stop::InvalidProgram)
1030    }
1031
1032    fn source_bytes(
1033        &self,
1034        limits: Limits,
1035        catalog_bytes: u128,
1036        atoms: usize,
1037    ) -> Result<usize, Stop> {
1038        let used = self
1039            .external_bytes(atoms)?
1040            .checked_add(self.catalog_bytes(catalog_bytes)?)
1041            .ok_or(Stop::Allocation)?;
1042        let remaining = (limits.max_host_bytes as u128)
1043            .checked_sub(used)
1044            .ok_or(Stop::Allocation)?;
1045        usize::try_from(remaining).map_err(|_| Stop::Allocation)
1046    }

1048    fn intern_atom(
1049        &mut self,
1050        appender: &mut AtomAppender<'_>,
1051        atom: AtomRef<'_>,
1052        limits: Limits,
1053        work: &mut Work<'_>,
1054    ) -> Result<u32, Stop> {
1055        let count = appender.len();
1056        let bounds = self.catalog_limits(limits, count)?;
1057        let entry = appender
1058            .entry_atom_with(atom, bounds, || work.tick())
1059            .map_err(|error| intern_stop(&error))?;
1060        self.publish(entry, count, limits, work)
1061    }

1063    fn intern_key(
1064        &mut self,
1065        appender: &mut AtomAppender<'_>,
1066        key: AtomKey<'_>,
1067        limits: Limits,
1068        work: &mut Work<'_>,
1069    ) -> Result<u32, Stop> {
1070        let count = appender.len();
1071        let bounds = self.catalog_limits(limits, count)?;
1072        let entry = appender
1073            .entry_key_with(key, bounds, || work.tick())
1074            .map_err(|error| intern_stop(&error))?;
1075        self.publish(entry, count, limits, work)
1076    }

1078    /// Both ingress seed atoms and borrowed instance keys use one discovery
1079    /// publication. Tuple growth belongs to the interner allowance; transport
1080    /// admits its masks and requested model positions. The input vocabulary is
1081    /// shared, and every accepted seed already belongs to its Program carrier.
1082    fn publish(
1083        &mut self,
1084        entry: AtomEntry<'_, '_>,
1085        count: usize,
1086        limits: Limits,
1087        work: &mut Work<'_>,
1088    ) -> Result<u32, Stop> {
1089        if let Some(id) = entry.position() {
1090            return u32::try_from(id).map_err(|_| Stop::CarrierLimit);
1091        }
1092        if count >= limits.max_atoms {
1093            return Err(Stop::CarrierLimit);
1094        }
1095        self.grow(count, entry.storage_bytes(), limits)?;
1096        let bounds = self.catalog_limits(limits, count + 1)?;
1097        let id = entry
1098            .insert_with(bounds, || work.tick())
1099            .map_err(|error| intern_stop(&error))?;
1100        u32::try_from(id).map_err(|_| Stop::CarrierLimit)
1101    }

1103    fn grow(&mut self, atom: usize, catalog_bytes: u128, limits: Limits) -> Result<(), Stop> {
1104        let needed = atom / 32 + 1;
1105        if needed <= self.words {
1106            return Ok(());
1107        }
1108        let width = needed
1109            .checked_next_power_of_two()
1110            .ok_or(Stop::Allocation)?
1111            .min(limits.max_atoms.div_ceil(32));
1112        let candidates = self.violated.len();
1113        let current_bits = self.words.checked_mul(candidates).ok_or(Stop::Allocation)?;
1114        let next_bits = width.checked_mul(candidates).ok_or(Stop::Allocation)?;
1115        let current_bytes = current_bits
1116            .checked_mul(ROUND_MASK_VECTORS * size_of::<u32>())
1117            .ok_or(Stop::Allocation)?;
1118        let base = self
1119            .fixed_bytes
1120            .checked_sub(current_bytes)
1121            .ok_or(Stop::InvalidProgram)?;
1122        // All three new vectors coexist with their old counterparts. Since the
1123        // width grows, six new-width vectors bound that temporary ownership.
1124        let peak = next_bits
1125            .checked_mul(GROWTH_MASK_VECTORS * size_of::<u32>())
1126            .and_then(|n| n.checked_add(base))
1127            .ok_or(Stop::Allocation)?;
1128        let selected = (atom + 1)
1129            .checked_mul(candidates)
1130            .and_then(|cells| cells.checked_mul(size_of::<usize>()))
1131            .ok_or(Stop::Allocation)?;
1132        let peak = peak as u128
1133            + selected as u128
1134            + self.source_mask_bytes as u128
1135            + self.catalog_bytes(catalog_bytes)?;
1136        if peak > limits.max_host_bytes as u128 {
1137            return Err(Stop::Allocation);
1138        }
1139        let mut snapshots = zeros(next_bits)?;
1140        let mut seeds = zeros(next_bits)?;
1141        let mut pending = zeros(next_bits)?;
1142        for world in 0..candidates {
1143            let old = world * self.words;
1144            let next = world * width;
1145            snapshots[next..next + self.words]
1146                .copy_from_slice(&self.snapshots[old..old + self.words]);
1147            seeds[next..next + self.words].copy_from_slice(&self.seeds[old..old + self.words]);
1148            pending[next..next + self.words].copy_from_slice(&self.pending[old..old + self.words]);
1149        }
1150        self.snapshots = snapshots;
1151        self.seeds = seeds;
1152        self.pending = pending;
1153        self.words = width;
1154        self.fixed_bytes = base + next_bits * ROUND_MASK_VECTORS * size_of::<u32>();
1155        Ok(())
1156    }

1158    fn offer<E>(
1159        &mut self,
1160        appender: &mut AtomAppender<'_>,
1161        instance: &source::Instance<'_>,
1162        limits: Limits,
1163        work: &mut Work<'_>,
1164        evaluation: &mut Evaluation<'_, impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>>,
1165    ) -> Result<(), Cause<E>> {
1166        let count = [
1167            instance.positive().len(),
1168            instance.gate_true().len(),
1169            instance.gate_false().len(),
1170        ]
1171        .into_iter()
1172        .try_fold(RECORD_HEADER_WORDS, usize::checked_add)
1173        .ok_or(Stop::Allocation)?;
1174        if count > limits.max_chunk_words {
1175            return Err(Stop::CarrierLimit.into());
1176        }
1177        if self.offsets.len() == limits.max_chunk_rules
1178            || self.records.len() > limits.max_chunk_words - count
1179        {
1180            self.flush(appender.len(), evaluation)?;
1181        }
1182        let offset = u32::try_from(self.records.len()).map_err(|_| Stop::CarrierLimit)?;
1183        let head = instance
1184            .head()
1185            .map(|atom| self.intern_key(appender, atom, limits, work))
1186            .transpose()?
1187            .map_or(CONSTRAINT_HEAD, |id| id + 1);
1188        self.records.push(head);
1189        for atoms in [
1190            instance.positive(),
1191            instance.gate_true(),
1192            instance.gate_false(),
1193        ] {
1194            self.records
1195                .push(u32::try_from(atoms.len()).map_err(|_| Stop::CarrierLimit)?);
1196        }
1197        for atom in instance
1198            .positive()
1199            .iter()
1200            .chain(instance.gate_true())
1201            .chain(instance.gate_false())
1202        {
1203            let id = self.intern_key(appender, atom, limits, work)?;
1204            self.records.push(id);
1205        }
1206        self.offsets.push(offset);
1207        Ok(())
1208    }
1209
1210    fn flush<E>(
1211        &mut self,
1212        catalog_atoms: usize,
1213        evaluation: &mut Evaluation<'_, impl FnMut(&Chunk<'_>) -> Result<Vec<u32>, E>>,
1214    ) -> Result<(), Cause<E>> {
1215        if self.offsets.is_empty() {
1216            return Ok(());
1217        }
1218        evaluation.cancellation.poll()?;
1219        let worlds = self.violated.len();
1220        let chunk = Chunk {
1221            round_index: evaluation.progress.rounds,
1222            words: self.words,
1223            worlds,
1224            catalog_atoms,
1225            snapshots: &self.snapshots,
1226            seeds: &self.seeds,
1227            offsets: &self.offsets,
1228            records: &self.records,
1229        };
1230        let result = (evaluation.execute)(&chunk).map_err(Cause::Execution)?;
1231        evaluation.cancellation.poll()?;
1232        if result.len() != worlds * (self.words + 1) {
1233            return Err(Cause::InvalidOutput);
1234        }
1235        let mut offered = zeros(self.words)?;
1236        for offset in &self.offsets {
1237            let head = self.records[*offset as usize];
1238            if head != CONSTRAINT_HEAD {
1239                insert(&mut offered, (head - 1) as usize);
1240            }
1241        }
1242        for (world, output) in result.chunks_exact(self.words + 1).enumerate() {
1243            if output[self.words] > 1 {
1244                return Err(Cause::InvalidOutput);
1245            }
1246            for (word, delta) in output[..self.words].iter().enumerate() {
1247                if delta & !offered[word] != 0
1248                    || delta & self.snapshots[world * self.words + word] != 0
1249                {
1250                    return Err(Cause::InvalidOutput);
1251                }
1252                self.pending[world * self.words + word] |= delta;
1253            }
1254            self.violated[world] |= output[self.words] == 1;
1255        }
1256        evaluation.progress.chunks += 1;
1257        self.offsets.clear();
1258        self.records.clear();
1259        Ok(())
1260    }
1261}

1263fn zeros(length: usize) -> Result<Vec<u32>, Stop> {
1264    let mut words = Vec::new();
1265    words
1266        .try_reserve_exact(length)
1267        .map_err(|_| Stop::Allocation)?;
1268    words.resize(length, 0);
1269    Ok(words)
1270}

1272fn intern_stop(error: &InternFailure<Stop>) -> Stop {
1273    match error {
1274        InternFailure::Stopped(stop) => *stop,
1275        InternFailure::Catalog(error) => Stop::catalog(error),
1276        InternFailure::Atoms { .. } => Stop::CarrierLimit,
1277        InternFailure::Bytes { .. } | InternFailure::Allocation(_) | InternFailure::Overflow => {
1278            Stop::Allocation
1279        }
1280    }
1281}

1283fn contains(words: &[u32], atom: usize) -> bool {
1284    words[atom / 32] & (1 << (atom % 32)) != 0
1285}
1286fn insert(words: &mut [u32], atom: usize) {
1287    words[atom / 32] |= 1 << (atom % 32);
1288}
