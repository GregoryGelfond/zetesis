35/// Exact checking budgets, applied before the next charged operation/insertion.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub struct Limits {
38    /// Maximum charged template, tuple-probe, atom-key argument span, gate/filter,
39    /// delta-view preparation and output operations.
40    /// Lazy joins also charge bound-prefix inspections and ordered comparisons,
41    /// including visited typed descriptors and text prefixes. Work counts can
42    /// change with the execution algorithm; they are not ground-instance counts.
43    /// A key charges its complete argument span even when an absent slot defers
44    /// a gate. The bounded construction itself polls only at that boundary.
45    pub max_work: u64,
46    /// Maximum distinct derived atoms, including pending round outputs.
47    pub max_derived_atoms: usize,
48    /// Named capacity of the canonical authority, relation indexes/columns and
49    /// prepared-order runs, dense words, pending discovery IDs/marks, operation
50    /// scratch and growth overlap. Canonical payload is charged once by the
51    /// authority; result catalog/selection metadata is charged during assembly.
52    /// Prepared checks also include preparation and assignment/cursor/undo
53    /// capacities. Prior results retained by the caller, allocator metadata and
54    /// Arc reference counters remain outside this per-check ledger.
55    /// Actual allocator slack can exceed the proposed reservation before refusal;
56    /// only completed checks publish their observed peak statistics.
57    /// This is an independent finite allowance, not a process RSS ceiling.
58    pub max_closure_bytes: usize,
59}
60
61impl Default for Limits {
62    fn default() -> Self {
63        Self {
64            max_work: 10_000_000,
65            max_derived_atoms: 1_000_000,
66            max_closure_bytes: 134_217_728,
67        }
68    }
69}

71/// Counters for a completed oracle invocation; these are execution work counts,
72/// not a device performance estimate or a count of conceptual ground instances.
73#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
74pub struct Statistics {
75    /// Completed source rounds, including the final no-delta round. The empty
76    /// program needs no rounds because its source coverage is vacuous.
77    pub rounds: u64,
78    /// Charged operations, as described by [`Limits::max_work`].
79    pub work: u64,
80    /// Subset of work spent constructing, extending and probing retained typed
81    /// catalogs and their prepared-order runs. Source joins and final
82    /// interpretation assembly are separate.
83    pub catalog_work: u64,
84    /// Largest admitted or actually reserved named scalar closure envelope,
85    /// under [`Limits::max_closure_bytes`]. Refused unallocated proposals do not
86    /// increase this maximum. It includes result assembly and is not RSS.
87    pub peak_closure_bytes: usize,
88    /// Fully matched enabled/filter-valid bindings visited by the selected round
89    /// schedule. Old bindings are not revisited by later scalar delta rounds.
90    pub bindings: u64,
91    /// Source rows offered to the whole-row matcher, including rejected rows.
92    /// Excludes prefix-search comparisons and catalog membership lookups. Each
93    /// probe belongs to an already charged join-loop step, so it is bounded by work.
94    pub tuple_probes: u64,
95    /// Distinct atoms in the final least consequence closure.
96    pub derived_atoms: usize,
97    /// Derived heads recorded as a bit of a dense relation's pending marks, so
98    /// that no atom was built for them before the model was assembled. The
99    /// closure's other heads were built as atoms when first derived.
100    pub dense_heads: u64,
101    /// Blocks of body rows joined into a head's pending marks a word at a
102    /// time, each in place of binding its rows one by one. The rows of such a
103    /// block are counted in `bindings` and not in `tuple_probes`: none is
104    /// offered to the row matcher.
105    pub block_steps: u64,
106}

111pub struct Check {
112    program: Program,
113    closure: Model,
114    constraint_violated: bool,
115    seed_mismatch: bool,
116    statistics: Statistics,
117}

119impl Check {
120    /// The immutable instance whose reduct closure was checked. Constant time.
121    #[must_use]
122    pub const fn program(&self) -> &Program {
123        &self.program
124    }
125
126    /// Exact least reduct closure, including rejected candidates. Constant-time
127    /// borrow; the raw interpretation alone carries no stable-membership claim.
128    #[must_use]
129    pub const fn interpretation(&self) -> &zetesis_core::Interpretation {
130        &self.closure
131    }
132
133    /// Transfer an accepted closure into an instance-bound stable receipt without
134    /// rechecking membership, cloning atoms, or allocating.
135    ///
136    /// # Errors
137    /// A rejected check is returned intact, retaining both rejection conditions.
138    pub fn into_stable_interpretation(self) -> Result<crate::StableInterpretation, Self> {
139        if self.accepted() {
140            Ok(crate::StableInterpretation::new(self.program, self.closure))
141        } else {
142            Err(self)
143        }
144    }
145
146    /// Whether the reconstructed closure is stable for the supplied seed.
147    #[must_use]
148    pub fn accepted(&self) -> bool {
149        !self.constraint_violated && !self.seed_mismatch
150    }
151    /// The exact least closure of the frozen reduct, even for rejected seeds.
152    #[must_use]
153    pub fn closure(&self) -> &Model {
154        &self.closure
155    }
156    /// An enabled constraint has a true positive body in the final closure.
157    #[must_use]
158    pub fn constraint_violated(&self) -> bool {
159        self.constraint_violated
160    }
161    /// The closure's gate-carrier projection differs from the frozen seed.
162    #[must_use]
163    pub fn seed_mismatch(&self) -> bool {
164        self.seed_mismatch
165    }
166    /// Completed invocation's counters.
167    #[must_use]
168    pub fn statistics(&self) -> Statistics {
169        self.statistics
170    }

173pub(crate) struct Work<'a> {
174    cancellation: &'a Cancellation,
175    limits: Limits,
176    statistics: Statistics,
177    mask_words: u64,
178    pruned_prefixes: u64,
179    mask_bytes: usize,
180}
181
182impl Work<'_> {
183    pub(crate) fn tick(&mut self) -> Result<(), Stop> {
184        self.charge(1)
185    }
186
187    /// Charge bookkeeping units before one indivisible operation, without
188    /// iterating over its payload. With unchanged control, a refusal records
189    /// the same admitted prefix as repeated ticks. Cancellation is polled before any
190    /// nonzero charge; zero preserves the old empty-loop behavior and neither
191    /// polls nor changes work. No real operation happens between these units.
192    fn charge(&mut self, amount: usize) -> Result<(), Stop> {
193        if amount == 0 {
194            return Ok(());
195        }
196        self.cancellation.poll()?;
197        let remaining = self.limits.max_work.saturating_sub(self.statistics.work);
198        if let Ok(amount) = u64::try_from(amount)
199            && amount <= remaining
200        {
201            self.statistics.work += amount;
202            Ok(())
203        } else {
204            self.statistics.work += remaining;
205            Err(Stop::WorkLimit)
206        }
207    }
208
209    pub(crate) fn source(cancellation: &Cancellation, max_work: u64) -> Work<'_> {
210        Work {
211            cancellation,
212            limits: Limits {
213                max_work,
214                max_derived_atoms: 0,
215                max_closure_bytes: Limits::default().max_closure_bytes,
216            },
217            statistics: Statistics::default(),
218            mask_words: 0,
219            pruned_prefixes: 0,
220            mask_bytes: 0,
221        }
222    }
223
224    pub(crate) fn source_statistics(&self, bindings: u64) -> source::ScanStatistics {
225        source::ScanStatistics {
226            work: self.statistics.work,
227            bindings,
228            mask_words: self.mask_words,
229            pruned_prefixes: self.pruned_prefixes,
230            mask_bytes: self.mask_bytes,
231        }
232    }
233
234    fn mask_word(&mut self) -> Result<(), Stop> {
235        self.tick()?;
236        self.mask_words += 1;
237        Ok(())
238    }
239}

241/// Compute the exact least positive closure selected by a sparse frozen seed.
242/// Bootstrap checks every template against empty truth, including zero-positive
243/// rules and constraints. Later rounds visit each binding at its first newly
244/// derived source occurrence. Completed earlier scans account for old bindings;
245/// the final no-change round completes source coverage before the full
246/// gate-carrier seed comparison. Shared source traversal has its own full schedule.
247///
248/// # Errors
249/// Returns [`Stop`] for cancellation, deadlines, budgets, foreign seed identity,
250/// or a violated admitted-program invariant; no partial result is accepted.
251pub fn check(
252    program: &Program,
253    seed: &Seed,
254    limits: Limits,
255    cancellation: &Cancellation,
256) -> Result<Check, Stop> {
257    check_view(program, seed.view(), limits, cancellation)
258}
259
260/// Check a borrowed owned seed or shared selection without copying its true
261/// atoms. This is the same reduct computation, identity validation and operation
262/// accounting as [`check`]; derived consequences still own their output payload.
263///
264/// # Errors
265/// Returns the same typed stops as [`check`], without accepting partial results.
266pub fn check_view(
267    program: &Program,
268    seed: SeedView<'_>,
269    limits: Limits,
270    cancellation: &Cancellation,
271) -> Result<Check, Stop> {
272    cancellation.poll()?;
273    if !program.same_instance(seed.program()) {
274        return Err(Stop::WrongProgram);
275    }
276    if program.templates().is_empty() {
277        return Ok(Check {
278            program: program.clone(),
279            closure: Model::default(),
280            constraint_violated: false,
281            seed_mismatch: false,
282            statistics: Statistics::default(),
283        });
284    }
285    let mut work = Work {
286        cancellation,
287        limits,
288        statistics: Statistics::default(),
289        mask_words: 0,
290        pruned_prefixes: 0,
291        mask_bytes: 0,
292    };
293    let prepared = PreparedQueries::prepare(
294        program,
295        PreparationLimits::default().max_dense_atoms,
296        &mut work,
297    )?;
298    prepared.check_with(seed, &mut ClosureWorkspace::default(), &mut work)
299}

303pub(crate) struct CompletedClosure {
304    pub(crate) atoms: Model,
305    /// A constraint fired in the completed closure: under a frozen seed, the
306    /// candidate is rejected; under the definite reading of a cube, no seed
307    /// of the cube is accepted (`Bounds.lower_constraint_refutes`).
308    pub(crate) constraint_violated: bool,
309}

311/// A region of seeds: the gate atoms every seed of it holds and the gate
312/// atoms some seed of it may hold. `None` for `may` is the whole symbolic
313/// carrier, which is never materialized. This is the `Cube` of
314/// `Search.lean`, with `Bounds.undecided` as its origin; a `Region` of the
315/// traversal is the same cube indexed over the narrowed root's atoms. Its
316/// sides are the region's: `must` is the held set, the complement of
317/// `may` the cut set, and the rest is open.
318pub(crate) struct Cube {
319    pub(crate) must: bounds::CarrierSet,
320    pub(crate) may: Option<bounds::CarrierSet>,
321}
322impl Cube {
323    /// Every atom open: every seed lies in this cube.
324    pub(crate) fn all_open() -> Self {
325        Self {
326            must: bounds::CarrierSet::new(),
327            may: None,
328        }
329    }
330    fn must_hold(
331        &self,
332        key: &zetesis_core::AtomKey<'_>,
333        work: &mut Work<'_>,
334    ) -> Result<bool, Stop> {
335        self.must.contains_key(key, work)
336    }
337    fn may_hold(&self, key: &zetesis_core::AtomKey<'_>, work: &mut Work<'_>) -> Result<bool, Stop> {
338        match &self.may {
339            None => Ok(true),
340            Some(may) => may.contains_key(key, work),
341        }
342    }
343}

345// Positive bodies, pure equality filters and the closure's gate reading make old enabled
346// bindings persist. Complete bootstrap plus disjoint first-new scans therefore
347// cover the same inflationary step as full rescanning. Constraints latch only
348// after a complete selected family; they never truncate another occurrence.
349// Every growing round adds an atom and has admitted its pending publication.
350// No resource stop can produce CompletedClosure or publish a reusable frontier.
351#[cfg(test)]
352fn least_closure(
353    program: &Program,
354    seed: SeedView<'_>,
355    work: &mut Work<'_>,
356) -> Result<CompletedClosure, Stop> {
357    let prepared =
358        PreparedQueries::prepare(program, PreparationLimits::default().max_dense_atoms, work)?;
359    prepared.closure_with(
360        Gates::Frozen(seed),
361        &mut ClosureWorkspace::default(),
362        Schedule::Delta,
363        work,
364    )
365}

367/// How a rule's gates are read while a closure is computed. The two cube
368/// readings are `MustGate` and `MayGate` of `Bounds.lean`.
369#[derive(Clone, Copy)]
370enum Gates<'a> {
371    /// The frozen seed decides every gate: a candidate's closure.
372    Frozen(SeedView<'a>),
373    /// A gate holds only if it holds under every seed of the cube, so the
374    /// closure lies inside every answer set the cube contains: its lower
375    /// closure.
376    Definite(Bounds<'a>),
377    /// Each gate is tested for possibility separately. This overapproximation
378    /// contains every answer set of the cube: its upper closure.
379    Possible(Bounds<'a>),
380    /// Every gate passes without being read: the join is asked for its
381    /// bindings alone, as the source scan and the restriction plan ask, and
382    /// no gate key is built.
383    Unjudged,
384}
385impl Gates<'_> {
386    /// Whether a rule can fire under this reading before its gates are
387    /// bound: a definite rule needs each `not not` gate in `must` and each
388    /// `not` gate outside `may`, which an empty `must` or an unbounded `may`
389    /// rules out for the whole template.
390    fn admits(self, template: TemplateRef<'_>) -> bool {
391        match self {
392            Self::Frozen(_) | Self::Possible(_) | Self::Unjudged => true,
393            Self::Definite(cube) => {
394                (template.gate_true().is_empty() || !cube.lower_is_empty())
395                    && (template.gate_false().is_empty() || cube.upper_is_bounded())
396            }
397        }
398    }
399    /// Whether the gate `pattern` bound to `key` holds under this reading;
400    /// `required` is true for a `not not` gate and false for a `not` gate.
401    fn holds(
402        self,
403        key: &zetesis_core::AtomKey<'_>,
404        required: bool,
405        work: &mut Work<'_>,
406    ) -> Result<bool, Stop> {
407        Ok(match (self, required) {
408            (Self::Unjudged, _) => true,
409            (Self::Frozen(seed), _) => seed.contains_key_with(key, || work.tick())? == required,
410            (Self::Definite(cube), true) => cube.must_hold(key, work)?,
411            (Self::Definite(cube), false) => !cube.may_hold(key, work)?,
412            (Self::Possible(cube), true) => cube.may_hold(key, work)?,
413            (Self::Possible(cube), false) => !cube.must_hold(key, work)?,
414        })
415    }
416}

418/// The lower closure of `cube` over `prepared`, computed in `workspace`:
419/// rules fire only under gates every seed of the cube satisfies, so the
420/// closure lies inside every answer set of the cube and its gate atoms
421/// inside every accepted seed (`Bounds.undecided_bounds_accepted` from the
422/// undecided cube). A fired constraint refutes the whole cube.
423///
424/// # Errors
425/// Returns [`Stop`] for cancellation, deadlines and the closure budgets of
426/// `limits`, charged as one candidate check would be; the workspace is then
427/// retired and no partial closure is returned.
428pub(crate) fn definite_closure(
429    prepared: &PreparedQueries,
430    workspace: &mut ClosureWorkspace,
431    cube: Bounds<'_>,
432    limits: Limits,
433    cancellation: &Cancellation,
434) -> Result<CompletedClosure, Stop> {
435    prepared.closure_of(Gates::Definite(cube), workspace, limits, cancellation)
436}

438/// The upper closure of `cube` over `prepared`, computed in `workspace`:
439/// rules fire when each gate remains possible under the bounds. There need
440/// not be one seed satisfying all those gates. Every answer set of the cube
441/// lies inside this overapproximation, and every accepted seed inside its gate
442/// atoms; a gate atom outside it belongs to no answer set of the cube. A fired
443/// upper constraint does not refute a seed.
444///
445/// # Errors

448    prepared: &PreparedQueries,

450    cube: Bounds<'_>,
451    limits: Limits,
452    cancellation: &Cancellation,
453) -> Result<CompletedClosure, Stop> {
454    prepared.closure_of(Gates::Possible(cube), workspace, limits, cancellation)
455}

464#[derive(Clone, Copy)]
465enum Selection {
466    All,
467    FirstNew(usize),
468}
469
470impl Selection {
471    fn rows(self, occurrence: usize) -> RowSet {
472        match self {
473            Self::FirstNew(pivot) if occurrence < pivot => RowSet::Old,
474            Self::FirstNew(pivot) if occurrence == pivot => RowSet::New,
475            Self::All | Self::FirstNew(_) => RowSet::Current,
476        }
477    }
478
479    /// The source occurrence visited at a join depth. An incremental round
480    /// visits its new rows first: they are the fewest, and the variables they
481    /// bind turn every other occurrence into a bound-prefix window instead of
482    /// a scan. The remaining occurrences keep their body order.
483    fn occurrence(self, depth: usize) -> usize {
484        match self {
485            Self::All => depth,
486            Self::FirstNew(pivot) => match depth {
487                0 => pivot,
488                depth if depth <= pivot => depth - 1,
489                depth => depth,
490            },
491        }
492    }
493}

495struct RoundWorkspace<'a> {
496    buffers: &'a mut prepared::Buffers,
497    dimensions: &'a prepared::Dimensions,
498    rules: &'a prepared::Rules,
499    layouts: &'a Layouts,
500    block_steps: &'a BlockSteps,
501    heads: &'a prepared::Heads,
502    pending: &'a mut PendingMarks,
503    overhead: u128,
504}
505
506fn least_closure_with(
507    program: &Program,
508    gates: Gates<'_>,
509    closure: &mut Catalogs,
510    workspace: RoundWorkspace<'_>,
511    schedule: Schedule,
512    work: &mut Work<'_>,
513) -> Result<CompletedClosure, Stop> {
514    let RoundWorkspace {
515        buffers,
516        dimensions,
517        rules,
518        layouts,
519        block_steps,
520        heads,
521        pending,
522        overhead,
523    } = workspace;
524    closure.bind_program(program, work)?;
525    let mut constraint_violated = false;
526    closure.create_dense_relations(layouts, work)?;
527    loop {
528        work.tick()?;
529        let incremental = schedule == Schedule::Delta && work.statistics.rounds != 0;
530        if incremental {
531            closure.prepare_delta(work)?;
532        } else {
533            closure.prepare(work)?;
534        }
535        // Bindings borrow this round's immutable catalog extent only. The
536        // reference-free cursor/undo buffers survive after these bindings drop.
537        let consequences = {
538            let (mut assignment, bytes) = prepared::assignment(
539                dimensions,
540                closure
541                    .owned_bytes()
542                    .checked_add(overhead)
543                    .ok_or(Stop::StorageLimit)?,
544                work,
545            )?;
546            closure.set_overhead(overhead.checked_add(bytes).ok_or(Stop::StorageLimit)?, work)?;
547            let (round, mut appender) = closure.split()?;
548            let result = visit_round(
549                program,
550                gates,
551                &round,
552                &mut appender,
553                RoundPlan {
554                    incremental,
555                    rules,
556                    layouts,
557                    block_steps,
558                    heads,
559                },
560                RoundScratch {
561                    assignment: &mut assignment,
562                    buffers: &mut *buffers,
563                    pending: &mut *pending,
564                },
565                work,
566            );
567            drop(assignment);
568            result?
569        };
570        let RoundConsequences {
571            atoms: delta,
572            constraint_violated: triggered,
573        } = consequences;
574        constraint_violated |= triggered;
575        closure.set_overhead(overhead, work)?;
576        work.statistics.rounds += 1;
577        if delta.is_empty() && pending.is_empty() {
578            break;
579        }
580        if schedule == Schedule::Delta {
581            closure.advance(work)?;
582        }
583        let pending_bytes = delta.bytes();
584        closure.commit(pending_bytes, work)?;
585        for &identity in delta.ids() {
586            closure.insert(identity, pending_bytes, work)?;
587        }
588        drop(delta);
589        closure.absorb(pending, layouts, work)?;
590    }
591    Ok(CompletedClosure {
592        atoms: closure.take_model(work)?,
593        constraint_violated,
594    })
595}

597/// What a round derived for the tree relations, and whether a constraint
598/// fired. Its dense heads are the marks left in the pending marks.
599struct RoundConsequences {
600    atoms: Pending,
601    constraint_violated: bool,
602}
603
604/// Record a derived head unless the closure or the round already holds it: a
605/// head of a dense relation as a pending bit, any other as a pending atom. The
606/// atoms the closure holds, the round's pending atoms and its pending bits are
607/// disjoint, so their sum is the count the derived-atom limit bounds.
608/// The atoms a round holds so far: the closure's, its pending atoms and its
609/// pending marks, which are disjoint, so their sum is what the derived-atom
610/// limit bounds.
611fn atoms_held(
612    closure: &RoundRead<'_>,
613    result: &RoundConsequences,
614    pending: &PendingMarks,
615) -> Result<usize, Stop> {
616    closure
617        .len()
618        .checked_add(result.atoms.len())
619        .and_then(|atoms| atoms.checked_add(pending.len()))
620        .ok_or(Stop::DerivedAtomLimit)
621}
622
623/// Whether `more` atoms beyond those held stay within the derived-atom limit.
624fn admits_more(held: usize, more: usize, work: &Work<'_>) -> Result<(), Stop> {
625    if held
626        .checked_add(more)
627        .is_none_or(|atoms| atoms > work.limits.max_derived_atoms)
628    {
629        return Err(Stop::DerivedAtomLimit);
630    }
631    Ok(())
632}

634fn record_head(
635    key: AtomKey<'_>,
636    dense_head: Option<DenseHead<'_>>,
637    closure: &RoundRead<'_>,
638    appender: &mut AtomAppender<'_>,
639    result: &mut RoundConsequences,
640    pending: &mut PendingMarks,
641    work: &mut Work<'_>,
642) -> Result<(), Stop> {
643    let held = atoms_held(closure, result, pending)?;
644    if let Some(DenseHead {
645        slot,
646        relation,
647        position,
648    }) = dense_head
649    {
650        // The bounds cover every derivable head: a key without a position
651        // violates the admitted program's invariant.
652        let position = match position {
653            Some(position) => position,
654            None => relation.position(&key, work)?.ok_or(Stop::InvalidProgram)?,
655        };
656        if !relation.contains(position) && pending.mark(slot, position) {
657            admits_more(held, 1, work)?;
658            work.statistics.dense_heads += 1;
659        }
660    } else {
661        let pending_bytes = result.atoms.bytes();
662        let transient = appender
663            .storage_bytes()
664            .checked_add(pending_bytes)
665            .ok_or(Stop::StorageLimit)?;
666        if closure.contains(&key, transient, work)? {
667            return Ok(());
668        }
669        let other = closure
670            .base_bytes()
671            .checked_add(pending_bytes)
672            .ok_or(Stop::StorageLimit)?;
673        let limits = relations::storage::atom_limits(work, other)?;
674        appender.restart_storage_peak();
675        let inserted = appender
676            .entry_key_with(key, limits, || relations::charge(work, 1))
677            .and_then(|entry| entry.insert_with(limits, || relations::charge(work, 1)));
678        relations::storage::record(work, other + appender.storage_peak_bytes())?;
679        let identity = inserted.map_err(relations::storage::atom_failure)?;
680        relations::charge(work, 1)?;
681        if !result.atoms.contains(identity) {
682            admits_more(held, 1, work)?;
683            result.atoms.insert(
684                identity,
685                closure.base_bytes() + appender.storage_bytes(),
686                work,
687            )?;
688        }
689    }
690    Ok(())
691}

693/// A round's immutable rule-selection mode and prepared indexes. Dense layouts
694/// name each head's slot; the block-step plan names eligible join occurrences.
695#[derive(Clone, Copy)]
696struct RoundPlan<'a> {
697    incremental: bool,
698    rules: &'a prepared::Rules,
699    layouts: &'a Layouts,
700    block_steps: &'a BlockSteps,
701    heads: &'a prepared::Heads,
702}
703
704impl RoundPlan<'_> {
705    /// Prepare the exact template sequence before visiting bindings. Bootstrap
706    /// visits source order; incremental rounds sort and coalesce rule indices
707    /// named by new predicates, retaining the existing checked work sequence.
708    fn select_rules(
709        self,
710        program: &Program,
711        closure: &RoundRead<'_>,
712        selected: &mut Vec<usize>,
713        work: &mut Work<'_>,
714    ) -> Result<usize, Stop> {
715        if !self.incremental {
716            return Ok(program.templates().len());
717        }
718        selected.clear();
719        for predicate in closure.predicates_with_new() {
720            work.tick()?;
721            for &index in self.rules.naming(predicate, work)? {
722                work.tick()?;
723                selected.push(index);
724            }
725        }
726        work.charge(selected.len())?;
727        selected.sort_unstable();
728        selected.dedup();
729        Ok(selected.len())
730    }
731}

734/// pending marks, which take a dense head's position.
735struct RoundScratch<'a, 'source> {

737    buffers: &'a mut prepared::Buffers,

741/// What a join reports to: each complete binding, and, where the consumer
742/// can take them whole, the block of rows its innermost depth would bind one
743/// by one. A plain function, an `FnMut`, is a consumer of bindings alone.
744trait Sink<'source, E> {
745    fn binding(
746        &mut self,
747        assignment: &[Option<TermRef<'source>>],
748        work: &mut Work<'_>,
749    ) -> Result<(), E>;
750
751    /// Whether the rows of this occurrence, visited innermost, are taken by
752    /// [`Self::rows`] instead of bound singly.
753    fn steps_by_blocks(&self, _occurrence: usize) -> bool {
754        false
755    }
756
757    /// Take the block of rows matching the assignment, which binds every
758    /// variable of the rule but the occurrence's last, and report how many
759    /// rows were taken as bindings, which the join counts as it counts the
760    /// bindings it makes singly.
761    fn rows(
762        &mut self,
763        _rows: Block<'source>,
764        _assignment: &[Option<TermRef<'source>>],
765        _work: &mut Work<'_>,
766    ) -> Result<u64, E> {
767        Ok(0)
768    }
769}

773    F: FnMut(&[Option<TermRef<'source>>], &mut Work<'_>) -> Result<(), E>,

775    fn binding(
776        &mut self,
777        assignment: &[Option<TermRef<'source>>],
778        work: &mut Work<'_>,
779    ) -> Result<(), E> {
780        self(assignment, work)
781    }

784/// A template's head relation with an optional prepared ground coordinate.
785#[derive(Clone, Copy)]
786struct DenseHead<'a> {
787    slot: usize,
788    relation: &'a Dense,
789    /// The head's whole ground tuple in the exact prepared layout. A
790    /// nonground head is ranked under the current binding instead.
791    position: Option<usize>,
792}

794/// The consumer of one template's joins in a closure round.
795struct RoundSink<'a, 'source, 'writer> {
796    template: TemplateRef<'a>,
797    closure: &'source RoundRead<'source>,
798    appender: &'a mut AtomAppender<'writer>,
799    /// The head's pending-marks slot and relation, when the head is laid out.
800    dense_head: Option<DenseHead<'source>>,
801    /// The template's block-step plan, by positive occurrence.
802    block_steps: &'a [bool],
803    result: &'a mut RoundConsequences,
804    pending: &'a mut PendingMarks,
805}
806
807impl<'source> Sink<'source, Stop> for RoundSink<'_, 'source, '_> {
808    fn binding(
809        &mut self,
810        assignment: &[Option<TermRef<'source>>],
811        work: &mut Work<'_>,
812    ) -> Result<(), Stop> {
813        work.tick()?;
814        if let Some(head) = self.template.head() {
815            work.charge(head.terms().len())?;
816            let key = head.key(assignment).map_err(|_| Stop::InvalidProgram)?;
817            record_head(
818                key,
819                self.dense_head,
820                self.closure,
821                self.appender,
822                self.result,
823                self.pending,
824                work,
825            )?;
826        } else {
827            self.result.constraint_violated = true;
828        }
829        Ok(())
830    }
831
832    fn steps_by_blocks(&self, occurrence: usize) -> bool {
833        self.block_steps.get(occurrence).copied().unwrap_or(false)
834    }
835
836    // The plan admitted the occurrence, so the head is laid out, ends in the
837    // occurrence's last variable and lists its values as the occurrence does:
838    // the heads of the block are the block of the head's bound prefix, place
839    // for place, and each row of the block is one binding.
840    fn rows(
841        &mut self,
842        rows: Block<'source>,
843        assignment: &[Option<TermRef<'source>>],
844        work: &mut Work<'_>,
845    ) -> Result<u64, Stop> {
846        let (
847            Some(head),
848            Some(DenseHead {
849                slot,
850                relation: dense,
851                ..
852            }),
853        ) = (self.template.head(), self.dense_head)
854        else {
855            return Err(Stop::InvalidProgram);
856        };
857        let bound = head.terms().len().saturating_sub(1);
858        work.charge(bound)?;
859        let heads = dense.layout().prefix_range(
860            head.terms()
861                .iter()
862                .take(bound)
863                .map_while(|term| resolve(term, assignment)),
864            work,
865        )?;
866        if heads.len() != rows.len {
867            // The head's bounds cover every derivable head: a head prefix
868            // outside them derives nothing, so the block holds no row.
869            return if rows.is_empty(work)? {
870                Ok(0)
871            } else {
872                Err(Stop::InvalidProgram)
873            };
874        }
875        let held_atoms = atoms_held(self.closure, self.result, self.pending)?;
876        let joined = self
877            .pending
878            .join_row(slot, dense, heads.start, rows, work)?;
879        admits_more(held_atoms, joined.marked, work)?;
880        let count = |n: usize| u64::try_from(n).map_err(|_| Stop::InvalidProgram);
881        work.statistics.dense_heads += count(joined.marked)?;
882        work.statistics.block_steps += 1;
883        count(joined.offered)
884    }
885}

887// Complete the disjoint source family before publishing either its history or
888// its consequences. A constraint latch never skips another source occurrence.
889/// Visit every template in the bootstrap round, or, given the rule index of
890/// an incremental round, only the templates whose body names a predicate
891/// with new rows: no other template can bind anew.
892fn visit_round<'source>(
893    program: &Program,
894    gates: Gates<'_>,
895    closure: &'source RoundRead<'source>,
896    appender: &mut AtomAppender<'_>,
897    plan: RoundPlan<'_>,
898    scratch: RoundScratch<'_, 'source>,
899    work: &mut Work<'_>,
900) -> Result<RoundConsequences, Stop> {
901    let RoundScratch {
902        assignment,
903        buffers,
904        pending,
905    } = scratch;
906    let mut result = RoundConsequences {
907        atoms: Pending::default(),
908        constraint_violated: false,
909    };
910    let live = closure.base_bytes() + appender.storage_bytes() + result.atoms.bytes();
911    relations::storage::admit(work, live)?;
912    relations::storage::record(work, live)?;
913    let visited = plan.select_rules(program, closure, &mut buffers.rules, work)?;
914    for position in 0..visited {
915        let index = if plan.incremental {
916            buffers.rules[position]
917        } else {
918            position
919        };
920        let template = program.templates().at(index).ok_or(Stop::InvalidProgram)?;
921        work.tick()?;
922        if !gates.admits(template) {
923            continue;
924        }
925        // A laid-out head has its relation from the start of the closure, so
926        // both are resolved once for the template, not once per binding.
927        let dense_head = match template.head().and_then(|head| {
928            plan.layouts
929                .slot(head.predicate())
930                .zip(Some(head.predicate()))
931        }) {
932            Some((slot, predicate)) => Some(DenseHead {
933                slot,
934                relation: closure
935                    .dense(predicate, work)?
936                    .ok_or(Stop::InvalidProgram)?,
937                position: plan.heads.position(index, slot)?,
938            }),
939            None => None,
940        };
941        let mut emit = RoundSink {
942            template,
943            closure,
944            appender: &mut *appender,
945            dense_head,
946            block_steps: plan.block_steps.of(index),
947            result: &mut result,
948            pending: &mut *pending,
949        };
950        if plan.incremental {
951            // Repeated predicates retain distinct occurrences. Earlier Old,
952            // this New and later Current rows select the unique first new row.
953            for (pivot, pattern) in template.positive().iter().enumerate() {
954                if closure.has_new(pattern.predicate(), work)? {
955                    visit_with(
956                        template,
957                        closure,
958                        gates,
959                        None,
960                        work,
961                        &mut emit,
962                        Frame {
963                            assignment: &mut *assignment,
964                            buffers: &mut *buffers,
965                            selection: Selection::FirstNew(pivot),
966                        },
967                    )?;
968                }
969            }
970        } else {
971            // Bootstrap includes every zero-positive head and constraint under
972            // the closure's gate reading. The test reference repeats this complete scan.
973            visit_with(
974                template,
975                closure,
976                gates,
977                None,
978                work,
979                &mut emit,
980                Frame {
981                    assignment: &mut *assignment,
982                    buffers: &mut *buffers,
983                    selection: Selection::All,
984                },
985            )?;
986        }
987    }
988    result.atoms.order(
989        appender,
990        closure.base_bytes() + appender.storage_bytes(),
991        work,
992    )?;
993    Ok(result)
994}

996// Establish closure ∩ gate_carrier = seed in both directions by one merge:
997// the closure's rows of each gate predicate, a contiguous range of the model,
998// walked against the seed's atoms, both in canonical order. Predicate windows
999// and atom comparisons use the shared checked lookup and typed comparator.
1000// The scan continues after a mismatch; rejection does not bypass work limits.
1001// Positive-only atoms do not belong to this comparison's projected carrier.
1002fn gate_agreement(
1003    program: &Program,
1004    seed: SeedView<'_>,
1005    closure: ModelAtoms<'_>,
1006    work: &mut Work<'_>,
1007) -> Result<bool, Stop> {
1008    let mut agreement = true;
1009    let mut seed = seed.atoms().peekable();
1010    for predicate in program.gate_predicates() {
1011        work.tick()?;
1012        let rows = closure.lookup().predicate_with(predicate, || work.tick())?;
1013        for row in rows {
1014            work.tick()?;
1015            let atom = row.atom();
1016            let mut matched = false;
1017            while let Some(pending) = seed.peek() {
1018                match pending.compare_ref_with(atom, || work.tick())? {
1019                    std::cmp::Ordering::Less => {
1020                        // A seed atom the closure never derived.
1021                        work.tick()?;
1022                        seed.next();
1023                        agreement = false;
1024                    }
1025                    std::cmp::Ordering::Equal => {
1026                        seed.next();
1027                        matched = true;
1028                        break;
1029                    }
1030                    std::cmp::Ordering::Greater => break,
1031                }
1032            }
1033            agreement &= matched;
1034        }
1035    }
1036    for _ in seed {
1037        work.tick()?;
1038        agreement = false;
1039    }
1040    Ok(agreement)
1041}

1045    buffers: &'frame mut prepared::Buffers,
1046    selection: Selection,

1049/// The caller's named retained metadata shares this allowance with one query's
1050/// local assignment, cursor and undo buffers. Callback-owned storage has its
1051/// own account; the callback receives the live scratch subtotal to compose it.
1052#[derive(Clone, Copy)]
1053struct QueryStorage {
1054    retained_bytes: u128,
1055    max_bytes: usize,
1056}
1057
1058fn visit<'source, E: From<Stop>>(
1059    template: TemplateRef<'_>,
1060    relations: &'source impl Relational,
1061    gates: Gates<'_>,
1062    membership: Option<&mut worlds::Join<'_>>,
1063    storage: QueryStorage,
1064    work: &mut Work<'_>,
1065    mut emit: impl FnMut(&[Option<TermRef<'source>>], u128, &mut Work<'_>) -> Result<(), E>,
1066) -> Result<u128, E> {
1067    let dimensions = prepared::Dimensions::for_template(template);
1068    let previous = work.limits.max_closure_bytes;
1069    work.limits.max_closure_bytes = storage.max_bytes;
1070    let prepared = (|| {
1071        let (assignment, bytes) = prepared::assignment(&dimensions, storage.retained_bytes, work)?;
1072        let buffers = prepared::Buffers::local(template, storage.retained_bytes + bytes, work)?;
1073        let scratch = bytes + size_of::<prepared::Buffers>() as u128 + buffers.bytes()?;
1074        Ok::<_, Stop>((assignment, buffers, scratch))
1075    })();
1076    // Restore even after a failed reserve. Source consumers can have independent
1077    // catalog or chunk admission and must not inherit this scratch allowance.
1078    work.limits.max_closure_bytes = previous;
1079    let (mut assignment, mut buffers, scratch) = prepared?;
1080    let mut borrowed_emit = |assignment: &[Option<TermRef<'source>>], work: &mut Work<'_>| {
1081        emit(assignment, scratch, work)
1082    };
1083    visit_with(
1084        template,
1085        relations,
1086        gates,
1087        membership,
1088        work,
1089        &mut borrowed_emit,
1090        Frame {
1091            assignment: &mut assignment,
1092            buffers: &mut buffers,
1093            selection: Selection::All,
1094        },
1095    )?;
1096    Ok(scratch)
1097}

1099fn visit_with<'source, E: From<Stop>>(
1100    template: TemplateRef<'_>,
1101    relations: &'source impl Relational,
1102    gates: Gates<'_>,
1103    mut membership: Option<&mut worlds::Join<'_>>,
1104    work: &mut Work<'_>,
1105    emit: &mut impl Sink<'source, E>,
1106    frame: Frame<'_, 'source>,
1107) -> Result<(), E> {
1108    // Reset all query state before this source occurrence. No binding survives
1109    // template reuse; positive matching still returns borrowed term references.
1110    let Frame {
1111        assignment,
1112        buffers,
1113        selection,
1114    } = frame;
1115    let assignment = &mut assignment[..template.variable_count()];
1116    work.charge(assignment.len())?;
1117    assignment.fill(None);
1118    if !guards(template, assignment, gates, work)? {
1119        return Ok(());
1120    }
1121    let count = template.positive().len();
1122    if count == 0 {
1123        if !assignment.is_empty() {
1124            return Err(Stop::InvalidProgram.into());
1125        }
1126        work.statistics.bindings += 1;
1127        return emit.binding(assignment, work);
1128    }
1129    if let Some(membership) = membership.as_mut() {
1130        membership.reset(template, work)?;
1131    }
1132    // None means this depth has not yet been opened for the current parent
1133    // assignment. A retained range advances in the original relation order.
1134    let cursors = &mut buffers.cursors[..count];
1135    let resolutions = &mut buffers.resolutions[..count];
1136    let undo = &mut buffers.undo[..count];
1137    work.charge(count)?;
1138    cursors.fill(None);
1139    resolutions.fill(Resolution::Unresolved);
1140    for row in undo.iter_mut() {
1141        work.tick()?;
1142        row.clear();
1143    }
1144    let mut depth = 0;
1145    loop {
1146        work.tick()?;
1147        if depth == count {
1148            if assignment.iter().any(Option::is_none) {
1149                return Err(Stop::InvalidProgram.into());
1150            }
1151            work.statistics.bindings += 1;
1152            emit.binding(assignment, work)?;
1153            depth -= 1;
1154            clear(assignment, &mut undo[depth]);
1155            continue;
1156        }
1157        let occurrence = selection.occurrence(depth);
1158        let pattern = template
1159            .positive()
1160            .get(occurrence)
1161            .ok_or(Stop::InvalidProgram)?;
1162        // The relation is fixed for the depth: resolve it on first entry and
1163        // index it on every later probe.
1164        if resolutions[depth] == Resolution::Unresolved {
1165            work.tick()?;
1166            resolutions[depth] = relations
1167                .resolve(pattern.predicate(), work)?
1168                .map_or(Resolution::Absent, Resolution::At);
1169        }
1170        let tuples = match resolutions[depth] {
1171            Resolution::At(handle) => relations.rows_at(handle, selection.rows(occurrence))?,
1172            Resolution::Absent | Resolution::Unresolved => Rows::Borrowed(&[]),
1173        };
1174        let cursor = &mut cursors[depth];
1175        // The innermost depth, entered for this parent assignment. A
1176        // membership join follows single rows, so it keeps them.
1177        if depth + 1 == count
1178            && cursor.is_none()
1179            && membership.is_none()
1180            && step_by_blocks(pattern, tuples, occurrence, assignment, emit, work)?
1181        {
1182            if depth == 0 {
1183                return Ok(());
1184            }
1185            depth -= 1;
1186            clear(assignment, &mut undo[depth]);
1187            continue;
1188        }
1189        let window = match cursor {
1190            Some(window) => window,
1191            None => cursor.insert(window::Window::open(pattern, tuples, assignment, work)?),
1192        };
1193        let Some((run, index)) = window.next(pattern, tuples, assignment, work)? else {
1194            *cursor = None;
1195            if depth == 0 {
1196                return Ok(());
1197            }
1198            depth -= 1;
1199            clear(assignment, &mut undo[depth]);
1200            continue;
1201        };
1202        let row = tuples.get(run, index).ok_or(Stop::InvalidProgram)?;
1203        // This iteration already charged one join step and offers at most one
1204        // row. The cumulative probe count therefore cannot exceed charged work.
1205        work.statistics.tuple_probes += 1;
1206        if bind(pattern, row, assignment, &mut undo[depth], work)?
1207            && guards(template, assignment, gates, work)?
1208            && match membership.as_mut() {
1209                Some(membership) => membership.extend(occurrence, index, work)?,
1210                None => true,
1211            }
1212        {
1213            depth += 1;
1214        } else {
1215            clear(assignment, &mut undo[depth]);
1216        }
1217    }
1218}

1220/// Give a consumer that steps by blocks the whole block of an occurrence's rows
1221/// matching the assignment, in place of binding them one by one; whether it
1222/// was given. The block is that of the bound leading terms, which for such an
1223/// occurrence are all but its last.
1224fn step_by_blocks<'source, E: From<Stop>>(
1225    pattern: PatternRef<'_>,
1226    tuples: Rows<'source>,
1227    occurrence: usize,
1228    assignment: &[Option<TermRef<'source>>],
1229    emit: &mut impl Sink<'source, E>,
1230    work: &mut Work<'_>,
1231) -> Result<bool, E> {
1232    let Rows::Dense { relation, set } = tuples else {
1233        return Ok(false);
1234    };
1235    if !emit.steps_by_blocks(occurrence) {
1236        return Ok(false);
1237    }
1238    // A bound value outside its argument's bound matches no position: the
1239    // block is empty and there is nothing to take.
1240    let block = window::matching_prefix(pattern, tuples, 0, assignment, work)?;
1241    if !block.is_empty() {
1242        let bound = emit.rows(
1243            Block {
1244                relation,
1245                set,
1246                start: block.start,
1247                len: block.len(),
1248            },
1249            assignment,
1250            work,
1251        )?;
1252        work.statistics.bindings += bound;
1253    }
1254    Ok(true)
1255}
