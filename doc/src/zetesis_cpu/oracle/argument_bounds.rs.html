1//! Upper bounds on the values each argument of a predicate can take in any
2//! closure of a program.
3//!
4//! Every atom of a closure is a table fact or an instance of a template's head, so an
5//! argument's values are bounded by what the heads can put there: a constant
6//! contributes itself, and a variable ranges within the values of every
7//! positive body position that binds it, the intersection of their bounds.
8//! Gates and filters only restrict, so they are ignored and the bound stays an
9//! upper bound. Closed under the templates to a least fixed point, the bounds
10//! are finite because every value is a constant of the program; an argument
11//! whose set would exceed the width ceiling is unknown instead, and unknown
12//! absorbs: a variable bound only at unknown positions is unknown.
13//!
14//! This is the finite-set lattice of the domain analysis in `zetesis-domain`,
15//! computed over the admitted templates after expansion, where the closure
16//! runs and where a range such as `node(1..16)` is already its facts. A
17//! bounded predicate whose product of widths fits a ceiling is read as a
18//! dense relation, a bit array over the mixed-radix index of its arguments'
19//! ranks; every other predicate keeps its tree. The inference serves
20//! preparation alone and takes its width ceiling and its work from it.
21

32/// An upper bound on one argument's values.
33#[derive(Clone, Debug, PartialEq, Eq)]
34pub(crate) enum Bound {
35    /// Only these values occur, in canonical order and without repetition.
36    Finite(Vec<usize>),
37    /// No finite bound: the argument is unbounded or too wide to keep.
38    Unknown,
39}
40
41/// The bounds of every argument of every predicate of one program.
42#[derive(Clone, Debug)]
43pub(crate) struct ArgumentBounds {
44    program: Program,
45    bounds: Vec<Vec<Bound>>,
46}
47
48/// One argument's growing bound during the fixed point.
49#[derive(Clone, Debug)]
50enum Growing {
51    Finite(BTreeSet<usize>),
52    Unknown,
53}
54
55impl ArgumentBounds {
56    /// The bounds of every argument of a predicate, in argument order.
57    pub(crate) fn bounds<'a>(&self, predicate: impl Into<PredicateRef<'a>>) -> Option<&[Bound]> {
58        self.program
59            .predicates()
60            .binary_search(predicate)
61            .ok()
62            .map(|table| self.bounds[table].as_slice())
63    }
64}

66/// Infer the bounds of the program's predicates, charging the given work;
67/// the width ceiling is `max_values`, an argument wider than it unknown. A
68/// constant head term's position is read from `positions` when it names it,
69/// and found by the domain's content search otherwise.
70///
71/// # Errors
72/// Returns the stop when the inference exceeds the work's ceiling or its
73/// control stops it.
74pub(super) fn infer(
75    program: &Program,
76    positions: Option<&DomainPositions<'_>>,
77    max_values: usize,
78    work: &mut Work<'_>,
79) -> Result<ArgumentBounds, Stop> {
80    let mut growing: Vec<Vec<Growing>> = program
81        .predicates()
82        .iter()
83        .map(|predicate| {
84            (0..predicate.arity())
85                .map(|_| Growing::Finite(BTreeSet::new()))
86                .collect()
87        })
88        .collect();
89    // A constant head term contributes the same position on every pass, so
90    // it is added once, before the fixed point. Every contribution only adds
91    // values or turns an argument unknown, so the order in which they are
92    // added does not change the least fixed point the passes reach.
93    for template in program.templates() {
94        let Some(head) = template.head() else {
95            continue;
96        };
97        if !head
98            .terms()
99            .iter()
100            .any(|term| matches!(term, TemplateTerm::Constant(_)))
101        {
102            continue;
103        }
104        let target = table(program, head.predicate(), work)?;
105        for (index, term) in head.terms().iter().enumerate() {
106            let TemplateTerm::Constant(value) = term else {
107                continue;
108            };
109            work.tick()?;
110            let position = domain_position(program, positions, value, work)?;
111            widen(
112                &mut growing[target][index],
113                Growing::Finite(BTreeSet::from([position])),
114                max_values,
115                work,
116            )?;
117        }
118    }
119    seed_fact_columns(program, positions, max_values, &mut growing, work)?;
120    // Each pass adds values or turns an argument unknown, never the
121    // reverse, and the values come from the program's constants, so the
122    // passes end when one changes nothing.
123    loop {
124        let mut changed = false;
125        for template in program.templates() {
126            let Some(head) = template.head() else {
127                continue;
128            };
129            if !head
130                .terms()
131                .iter()
132                .any(|term| matches!(term, TemplateTerm::Variable(_)))
133            {
134                continue;
135            }
136            let target = table(program, head.predicate(), work)?;
137            for (index, term) in head.terms().iter().enumerate() {
138                let TemplateTerm::Variable(variable) = term else {
139                    continue;
140                };
141                work.tick()?;
142                let contribution = variable_bound(program, template, variable, &growing, work)?;
143                changed |= widen(&mut growing[target][index], contribution, max_values, work)?;
144            }
145        }
146        if !changed {
147            break;
148        }
149    }
150    let bounds = growing
151        .into_iter()
152        .map(|arguments| {
153            arguments
154                .into_iter()
155                .map(|argument| match argument {
156                    Growing::Finite(values) => Bound::Finite(values.into_iter().collect()),
157                    Growing::Unknown => Bound::Unknown,
158                })
159                .collect()
160        })
161        .collect();
162    Ok(ArgumentBounds {
163        program: program.clone(),
164        bounds,
165    })
166}

168/// Seed each admitted fact column once, outside variable propagation. The
169/// relation view borrows canonical values; neither equality IDs nor occurrence
170/// positions are interpreted as numeric values or domain ranks.
171fn seed_fact_columns(
172    program: &Program,
173    positions: Option<&DomainPositions<'_>>,
174    max_values: usize,
175    growing: &mut [Vec<Growing>],
176    work: &mut Work<'_>,
177) -> Result<(), Stop> {
178    for facts in program.fact_relations() {
179        let relation = facts.relation();
180        let target = table(program, relation.predicate(), work)?;
181        for (column, argument) in growing[target].iter_mut().enumerate() {
182            for row in 0..relation.row_count() {
183                work.tick()?;
184                let value = relation
185                    .row(row)
186                    .and_then(|row| row.value(column))
187                    .ok_or(Stop::InvalidProgram)?;
188                let position = domain_position(program, positions, value, work)?;
189                widen(
190                    argument,
191                    Growing::Finite(BTreeSet::from([position])),
192                    max_values,
193                    work,
194                )?;
195            }
196        }
197    }
198    Ok(())
199}

201/// The values a head variable can take: the intersection of the bounds of
202/// the positive body positions binding it, unknown positions binding nothing.
203/// A variable no positive position binds is unknown: the bound stays an
204/// upper domain whatever else binds it.
205fn variable_bound(
206    program: &Program,
207    template: TemplateRef<'_>,
208    variable: usize,
209    growing: &[Vec<Growing>],
210    work: &mut Work<'_>,
211) -> Result<Growing, Stop> {
212    let mut bound: Option<BTreeSet<usize>> = None;
213    for pattern in template.positive() {
214        let source = table(program, pattern.predicate(), work)?;
215        for (index, term) in pattern.terms().iter().enumerate() {
216            if term != TemplateTerm::Variable(variable) {
217                continue;
218            }
219            work.tick()?;
220            match &growing[source][index] {
221                Growing::Unknown => {}
222                Growing::Finite(values) => {
223                    work.charge(values.len())?;
224                    bound = Some(match bound {
225                        None => values.clone(),
226                        Some(known) => known.intersection(values).copied().collect(),
227                    });
228                }
229            }
230        }
231    }
232    Ok(bound.map_or(Growing::Unknown, Growing::Finite))
233}

235/// The domain position of a program constant: two charged steps when the
236/// table names it, the domain's content search otherwise.
237pub(super) fn domain_position(
238    program: &Program,
239    positions: Option<&DomainPositions<'_>>,
240    value: TermRef<'_>,
241    work: &mut Work<'_>,
242) -> Result<usize, Stop> {
243    if let Some(positions) = positions {
244        work.charge(2)?;
245        if let Some(position) = positions.position(value) {
246            return Ok(position);
247        }
248    }
249    program
250        .domain()
251        .binary_search_with(value, || work.tick())?
252        .map_err(|_| Stop::InvalidProgram)
253}
254
255fn table(
256    program: &Program,
257    predicate: PredicateRef<'_>,
258    work: &mut Work<'_>,
259) -> Result<usize, Stop> {
260    program
261        .predicates()
262        .binary_search_with(predicate, || work.tick())?
263        .map_err(|_| Stop::InvalidProgram)
264}

266/// Widen an argument's bound by a contribution; whether anything changed.
267fn widen(
268    argument: &mut Growing,
269    contribution: Growing,
270    max_values: usize,
271    work: &mut Work<'_>,
272) -> Result<bool, Stop> {
273    match (&mut *argument, contribution) {
274        (Growing::Unknown, _) => Ok(false),
275        (Growing::Finite(_), Growing::Unknown) => {
276            *argument = Growing::Unknown;
277            Ok(true)
278        }
279        (Growing::Finite(values), Growing::Finite(added)) => {
280            let before = values.len();
281            work.charge(added.len())?;
282            values.extend(added);
283            if values.len() > max_values {
284                *argument = Growing::Unknown;
285                return Ok(true);
286            }
287            Ok(values.len() != before)
288        }
289    }
290}
