1//! Upper bounds on the values each argument of a predicate can take in any
2//! closure of a program.
3//!
4//! Every atom of a closure is an instance of a template's head, so an
5//! argument's values are bounded by what the heads can put there: a constant
6//! contributes itself, and a variable ranges within the values of every
7//! positive body position that binds it, the intersection of their bounds.
8//! Gates and filters only restrict, so they are ignored and the bound stays an
9//! upper bound. Closed under the templates to a least fixed point, the bounds
10//! are finite because every value is a constant of the program; an argument
11//! whose set would exceed the width ceiling is unknown instead, and unknown
12//! absorbs: a variable bound only at unknown positions is unknown.
13//!
14//! This is the finite-set lattice of the domain analysis in `zetesis-domain`,
15//! computed over the admitted templates after expansion, where the closure
16//! runs and where a range such as `node(1..16)` is already its facts. A
17//! bounded predicate whose product of widths fits a ceiling is read as a
18//! dense relation, a bit array over the mixed-radix index of its arguments'
19//! ranks; every other predicate keeps its tree. The inference serves
20//! preparation alone and takes its width ceiling and its work from it.
21

29/// An upper bound on one argument's values.
30#[derive(Clone, Debug, PartialEq, Eq)]
31pub(crate) enum Bound {
32    /// Only these values occur, in canonical order and without repetition.
33    Finite(Vec<usize>),
34    /// No finite bound: the argument is unbounded or too wide to keep.
35    Unknown,
36}
37
38/// The bounds of every argument of every predicate of one program.
39#[derive(Clone, Debug)]
40pub(crate) struct ArgumentBounds {
41    program: Program,
42    bounds: Vec<Vec<Bound>>,
43}
44
45/// One argument's growing bound during the fixed point.
46#[derive(Clone, Debug)]
47enum Growing {
48    Finite(BTreeSet<usize>),
49    Unknown,
50}
51
52impl ArgumentBounds {
53    /// The bounds of every argument of a predicate, in argument order.
54    pub(crate) fn bounds<'a>(&self, predicate: impl Into<PredicateRef<'a>>) -> Option<&[Bound]> {
55        self.program
56            .predicates()
57            .binary_search(predicate)
58            .ok()
59            .map(|table| self.bounds[table].as_slice())
60    }
61}
62
63/// Infer the bounds of the program's predicates, charging the given work;
64/// the width ceiling is `max_values`, an argument wider than it unknown.
65///
66/// # Errors
67/// Returns the stop when the inference exceeds the work's ceiling or its
68/// control stops it.
69pub(super) fn infer(
70    program: &Program,
71    max_values: usize,
72    work: &mut Work<'_>,
73) -> Result<ArgumentBounds, Stop> {
74    let mut growing: Vec<Vec<Growing>> = program
75        .predicates()
76        .iter()
77        .map(|predicate| {
78            (0..predicate.arity())
79                .map(|_| Growing::Finite(BTreeSet::new()))
80                .collect()
81        })
82        .collect();
83    // Each pass adds values or turns an argument unknown, never the
84    // reverse, and the values come from the program's constants, so the
85    // passes end when one changes nothing.
86    loop {
87        let mut changed = false;
88        for template in program.templates() {
89            let Some(head) = template.head() else {
90                continue;
91            };
92            let target = table(program, head.predicate(), work)?;
93            for (index, term) in head.terms().iter().enumerate() {
94                work.tick()?;
95                let contribution = match term {
96                    TemplateTerm::Constant(value) => Growing::Finite(BTreeSet::from([program
97                        .domain()
98                        .binary_search_with(value, || work.tick())?
99                        .map_err(|_| Stop::InvalidProgram)?])),
100                    TemplateTerm::Variable(variable) => {
101                        variable_bound(program, template, variable, &growing, work)?
102                    }
103                };
104                changed |= widen(&mut growing[target][index], contribution, max_values, work)?;
105            }
106        }
107        if !changed {
108            break;
109        }
110    }
111    let bounds = growing
112        .into_iter()
113        .map(|arguments| {
114            arguments
115                .into_iter()
116                .map(|argument| match argument {
117                    Growing::Finite(values) => Bound::Finite(values.into_iter().collect()),
118                    Growing::Unknown => Bound::Unknown,
119                })
120                .collect()
121        })
122        .collect();
123    Ok(ArgumentBounds {
124        program: program.clone(),
125        bounds,
126    })
127}

129/// The values a head variable can take: the intersection of the bounds of
130/// the positive body positions binding it, unknown positions binding nothing.
131/// A variable no positive position binds is unknown: the bound stays an
132/// upper domain whatever else binds it.
133fn variable_bound(
134    program: &Program,
135    template: TemplateRef<'_>,
136    variable: usize,
137    growing: &[Vec<Growing>],
138    work: &mut Work<'_>,
139) -> Result<Growing, Stop> {
140    let mut bound: Option<BTreeSet<usize>> = None;
141    for pattern in template.positive() {
142        let source = table(program, pattern.predicate(), work)?;
143        for (index, term) in pattern.terms().iter().enumerate() {
144            if term != TemplateTerm::Variable(variable) {
145                continue;
146            }
147            work.tick()?;
148            match &growing[source][index] {
149                Growing::Unknown => {}
150                Growing::Finite(values) => {
151                    work.charge(values.len())?;
152                    bound = Some(match bound {
153                        None => values.clone(),
154                        Some(known) => known.intersection(values).copied().collect(),
155                    });
156                }
157            }
158        }
159    }
160    Ok(bound.map_or(Growing::Unknown, Growing::Finite))
161}
162
163fn table(
164    program: &Program,
165    predicate: PredicateRef<'_>,
166    work: &mut Work<'_>,
167) -> Result<usize, Stop> {
168    program
169        .predicates()
170        .binary_search_with(predicate, || work.tick())?
171        .map_err(|_| Stop::InvalidProgram)
172}

174/// Widen an argument's bound by a contribution; whether anything changed.
175fn widen(
176    argument: &mut Growing,
177    contribution: Growing,
178    max_values: usize,
179    work: &mut Work<'_>,
180) -> Result<bool, Stop> {
181    match (&mut *argument, contribution) {
182        (Growing::Unknown, _) => Ok(false),
183        (Growing::Finite(_), Growing::Unknown) => {
184            *argument = Growing::Unknown;
185            Ok(true)
186        }
187        (Growing::Finite(values), Growing::Finite(added)) => {
188            let before = values.len();
189            work.charge(added.len())?;
190            values.extend(added);
191            if values.len() > max_values {
192                *argument = Growing::Unknown;
193                return Ok(true);
194            }
195            Ok(values.len() != before)
196        }
197    }
198}
