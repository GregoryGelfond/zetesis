1//! Where a rule's innermost join may be taken a block at a time.
2//!
3//! At the innermost depth of a join every other positive occurrence has bound
4//! its variables. Suppose the occurrence visited there is over a dense
5//! relation, its last argument is a variable nothing else in the rule
6//! mentions but the head, in its own last argument, and every other argument
7//! of the two patterns is a constant or bound. Then the occurrence's rows
8//! are one block of its relation, the heads they derive are one block of the
9//! head's relation, and a tuple's place in the first block is its head's
10//! place in the second when both relations list that argument's values
11//! alike. The join may then mark the block of heads from the block of rows
12//! a word at a time instead of binding each row: no gate or filter reads the
13//! variable, so every guard was judged before the depth was reached, and each
14//! row of the block is one binding of the rule.
15//!
16//! The plan is a property of the templates and the layouts, fixed at
17//! preparation. Which occurrence a round visits innermost depends on the
18//! occurrence it pivots on, so the plan answers for every occurrence.
19

28/// For each template and positive occurrence, whether the occurrence may be
29/// taken a block at a time when the join visits it innermost.
30///
31/// Retains one flag for each positive occurrence of the program. Planning
32/// reads every term of a template once for each of its occurrences, so its
33/// work is at most the square of the largest template's terms for each
34/// template; a round reads a flag in constant time.
35#[derive(Default)]
36pub(super) struct BlockSteps(Vec<Vec<bool>>);
37
38impl BlockSteps {
39    /// One unit for each term of each pattern inspected for each occurrence.
40    pub(super) fn plan(
41        program: &Program,
42        layouts: &Layouts,
43        work: &mut Work<'_>,
44    ) -> Result<Self, Stop> {
45        let mut steps = Vec::new();
46        steps
47            .try_reserve_exact(program.templates().len())
48            .map_err(|_| Stop::Allocation)?;
49        for template in program.templates() {
50            let mut occurrences = Vec::new();
51            occurrences
52                .try_reserve_exact(template.positive().len())
53                .map_err(|_| Stop::Allocation)?;
54            for occurrence in 0..template.positive().len() {
55                work.charge(1 + terms(template))?;
56                occurrences.push(admits(template, occurrence, layouts));
57            }
58            steps.push(occurrences);
59        }
60        Ok(Self(steps))
61    }
62
63    pub(super) fn of(&self, template: usize) -> &[bool] {
64        self.0.get(template).map_or(&[], Vec::as_slice)
65    }
66
67    /// Retained bytes; `None` when the sum does not fit.
68    pub(super) fn bytes(&self) -> Option<u128> {
69        let flags = self
70            .0
71            .iter()
72            .try_fold(0u128, |sum, plan| sum.checked_add(plan.capacity() as u128))?;
73        (self.0.capacity() as u128)
74            .checked_mul(size_of::<Vec<bool>>() as u128)?
75            .checked_add(flags)
76    }
77}

79fn terms(template: TemplateRef<'_>) -> usize {
80    template
81        .head()
82        .into_iter()
83        .chain(template.positive())
84        .chain(template.gate_true())
85        .chain(template.gate_false())
86        .map(|pattern| pattern.terms().len())
87        .sum::<usize>()
88        + 2 * template.filters().len()
89}
90
91fn mentions(pattern: PatternRef<'_>, variable: usize) -> usize {
92    pattern
93        .terms()
94        .iter()
95        .filter(|term| *term == TemplateTerm::Variable(variable))
96        .count()
97}

99/// The condition of the module documentation, clause by clause.
100fn admits(template: TemplateRef<'_>, occurrence: usize, layouts: &Layouts) -> bool {
101    let (Some(head), Some(body)) = (template.head(), template.positive().get(occurrence)) else {
102        return false;
103    };
104    let (Some(TemplateTerm::Variable(free)), Some(TemplateTerm::Variable(last))) = (
105        body.terms().iter().next_back(),
106        head.terms().iter().next_back(),
107    ) else {
108        return false;
109    };
110    if free != last || mentions(body, free) != 1 || mentions(head, free) != 1 {
111        return false;
112    }
113    let others = || {
114        template
115            .positive()
116            .iter()
117            .enumerate()
118            .filter(move |(index, _)| *index != occurrence)
119            .map(|(_, pattern)| pattern)
120    };
121    let read_elsewhere = others()
122        .chain(template.gate_true())
123        .chain(template.gate_false())
124        .any(|pattern| mentions(pattern, free) != 0)
125        || template.filters().iter().any(|filter| {
126            let (left, right) = filter.terms();
127            [left, right].contains(&TemplateTerm::Variable(free))
128        });
129    // Every other variable of the occurrence is bound by an earlier depth.
130    let bound = body.terms().iter().all(|term| match term {
131        TemplateTerm::Constant(_) => true,
132        TemplateTerm::Variable(variable) => {
133            variable == free || others().any(|pattern| mentions(pattern, variable) != 0)
134        }
135    });
136    if read_elsewhere || !bound {
137        return false;
138    }
139    match (layouts.get(body.predicate()), layouts.get(head.predicate())) {
140        (Some(body), Some(head)) => body.last_coordinates() == head.last_coordinates(),
141        _ => false,
142    }
143}

154    fn pattern(name: &str, terms: &[Term]) -> AtomPattern {
155        AtomPattern::new(Predicate::new(name, terms.len()).unwrap(), terms.to_vec()).unwrap()
156    }

158    fn numbers(program: &Program, values: std::ops::RangeInclusive<i32>) -> Bound {
159        Bound::Finite(
160            values
161                .map(|value| {
162                    program
163                        .domain()
164                        .binary_search(&Value::Number(value))
165                        .unwrap()
166                })
167                .collect(),
168        )
169    }

171    /// `e` and `reach` over 1..=4 by 2..=5, `r` over 0..=5, `q` over 2..=5.
172    fn layouts() -> Layouts {
173        let source: Vec<_> = [("e", 2), ("q", 1), ("r", 1), ("reach", 2)]
174            .into_iter()
175            .flat_map(|(name, arity)| {
176                (0..=5).map(move |n| {
177                    Atom::new(
178                        Predicate::new(name, arity).unwrap(),
179                        vec![Value::Number(n); arity],
180                    )
181                    .unwrap()
182                })
183            })
184            .collect();
185        let program = crate::oracle::relations::fixtures::program(&source);
186        let mut layouts = Layouts::default();
187        for (name, bounds) in [
188            (
189                "e",
190                vec![numbers(&program, 1..=4), numbers(&program, 2..=5)],
191            ),
192            ("q", vec![numbers(&program, 2..=5)]),
193            ("r", vec![numbers(&program, 0..=5)]),
194            (
195                "reach",
196                vec![numbers(&program, 1..=4), numbers(&program, 2..=5)],
197            ),
198        ] {
199            let predicate = Predicate::new(name, bounds.len()).unwrap();
200            layouts.push(
201                Layout::new(&program, (&predicate).into(), &bounds, 1 << 10)
202                    .unwrap()
203                    .unwrap(),
204            );
205        }
206        layouts
207    }
