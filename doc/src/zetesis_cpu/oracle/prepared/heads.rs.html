1//! Dense coordinates of ground source heads in one immutable preparation.
2//!
3//! A coordinate is a property of the entire ordered ground tuple and its
4//! exact layout, never candidate truth. The owning `PreparedQueries` keeps
5//! both source and layouts alive. Each candidate still visits the source,
6//! checks its guards and marks the derived position in its own workspace.
7
8use std::mem::size_of;
9
10use zetesis_core::{DomainPositions, PatternRef, Program, RuleRef, TemplateTerm};
11
12use super::{Layouts, Stop, Work, reserve};
13
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15struct Coordinate {
16    slot: usize,
17    position: usize,
18}
19
20/// One optional coordinate per template and per original table-fact occurrence. Nonground or tree heads keep the
21/// ordinary key-to-position operation. Retained storage is linear in source
22/// templates; ranking work is the sum of the ground arguments' bound lookups.
23#[derive(Default)]
24pub(in crate::oracle) struct Heads {
25    templates: Vec<Option<Coordinate>>,
26    facts: Vec<Option<Coordinate>>,
27}
28
29impl Heads {
30    /// Rank each ground head on its predicate's dense layout, by identity when
31    /// preparation holds the domain position table. `base` is the live
32    /// storage already admitted, the table's included while it is read.
33    pub(super) fn prepare(
34        program: &Program,
35        layouts: &Layouts,
36        positions: Option<&DomainPositions<'_>>,
37        base: u128,
38        work: &mut Work<'_>,
39    ) -> Result<Self, Stop> {
40        let mut result = Self::default();
41        let mut live = base;
42        reserve(
43            &mut result.templates,
44            program.templates().len(),
45            &mut live,
46            work,
47        )?;
48        reserve(&mut result.facts, program.facts().len(), &mut live, work)?;
49        for rule in program.rules() {
50            work.tick()?;
51            match rule {
52                RuleRef::Template { template, .. } => {
53                    result
54                        .templates
55                        .push(coordinate(template.head(), layouts, positions, work)?);
56                }
57                RuleRef::Fact(fact) => {
58                    result
59                        .facts
60                        .push(coordinate(Some(fact.pattern()), layouts, positions, work)?);
61                }
62            }
63        }
64        Ok(result)
65    }
66
67    pub(super) fn bytes(&self) -> u128 {
68        (self.templates.capacity() + self.facts.capacity()) as u128
69            * size_of::<Option<Coordinate>>() as u128
70    }
71
72    /// Each template's coordinate as its slot and position, for comparison.
73    #[cfg(test)]
74    pub(super) fn coordinates(&self) -> Vec<Option<(usize, usize)>> {
75        self.templates
76            .iter()
77            .map(|coordinate| coordinate.map(|Coordinate { slot, position }| (slot, position)))
78            .collect()
79    }
80
81    /// A prepared position belongs only to the template's own dense slot.
82    /// Missing coordinates retain ordinary key evaluation, including a
83    /// ground tuple absent from an empty inferred layout.
84    pub(in crate::oracle) fn position(
85        &self,
86        template: usize,
87        slot: usize,
88    ) -> Result<Option<usize>, Stop> {
89        position(&self.templates, template, slot)
90    }
91
92    pub(in crate::oracle) fn fact_position(
93        &self,
94        fact: usize,
95        slot: usize,
96    ) -> Result<Option<usize>, Stop> {
97        position(&self.facts, fact, slot)
98    }
99}
100
101fn position(
102    coordinates: &[Option<Coordinate>],
103    index: usize,
104    slot: usize,
105) -> Result<Option<usize>, Stop> {
106    coordinates
107        .get(index)
108        .copied()
109        .flatten()
110        .map(|coordinate| {
111            if coordinate.slot == slot {
112                Ok(coordinate.position)
113            } else {
114                Err(Stop::InvalidProgram)
115            }
116        })
117        .transpose()
118}
119
120fn coordinate(
121    head: Option<PatternRef<'_>>,
122    layouts: &Layouts,
123    positions: Option<&DomainPositions<'_>>,
124    work: &mut Work<'_>,
125) -> Result<Option<Coordinate>, Stop> {
126    let Some(head) = head else {
127        return Ok(None);
128    };
129    work.charge(head.terms().len())?;
130    let Some(slot) = layouts.slot(head.predicate()) else {
131        return Ok(None);
132    };
133    if !head
134        .terms()
135        .iter()
136        .all(|term| matches!(term, TemplateTerm::Constant(_)))
137    {
138        return Ok(None);
139    }
140    let layout = layouts.get(head.predicate()).ok_or(Stop::InvalidProgram)?;
141    let values = head.terms().iter().filter_map(|term| match term {
142        TemplateTerm::Constant(value) => Some(value),
143        TemplateTerm::Variable(_) => None,
144    });
145    let position = match positions {
146        Some(positions) => layout.position_by_identity(values, positions, work)?,
147        None => layout.position_of(values, work)?,
148    };
149    Ok(position.map(|position| Coordinate { slot, position }))
150}
