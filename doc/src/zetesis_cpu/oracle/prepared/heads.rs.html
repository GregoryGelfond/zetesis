1//! Dense coordinates of ground source heads in one immutable preparation.
2//!
3//! A coordinate is a property of the entire ordered ground tuple and its
4//! exact layout, never candidate truth. The owning `PreparedQueries` keeps
5//! both source and layouts alive. Each candidate still visits the source,
6//! checks its guards and marks the derived position in its own workspace.
7
8use std::mem::size_of;
9
10use zetesis_core::{Program, TemplateTerm};
11
12use super::{Layouts, Stop, Work, reserve};
13
14#[derive(Clone, Copy)]
15struct Coordinate {
16    slot: usize,
17    position: usize,
18}
19
20/// One optional coordinate per template. Nonground or tree heads keep the
21/// ordinary key-to-position operation. Retained storage is linear in source
22/// templates; ranking work is the sum of the ground arguments' bound lookups.
23#[derive(Default)]
24pub(in crate::oracle) struct Heads(Vec<Option<Coordinate>>);
25
26impl Heads {
27    pub(super) fn prepare(
28        program: &Program,
29        layouts: &Layouts,
30        base: u128,
31        work: &mut Work<'_>,
32    ) -> Result<Self, Stop> {
33        let mut coordinates = Vec::new();
34        let mut live = base;
35        reserve(&mut coordinates, program.templates().len(), &mut live, work)?;
36        for template in program.templates() {
37            work.tick()?;
38            let coordinate = if let Some(head) = template.head() {
39                work.charge(head.terms().len())?;
40                match layouts.slot(head.predicate()) {
41                    Some(slot)
42                        if head
43                            .terms()
44                            .iter()
45                            .all(|term| matches!(term, TemplateTerm::Constant(_))) =>
46                    {
47                        let layout = layouts.get(head.predicate()).ok_or(Stop::InvalidProgram)?;
48                        let values = head.terms().iter().filter_map(|term| match term {
49                            TemplateTerm::Constant(value) => Some(value),
50                            TemplateTerm::Variable(_) => None,
51                        });
52                        layout
53                            .position_of(values, work)?
54                            .map(|position| Coordinate { slot, position })
55                    }
56                    _ => None,
57                }
58            } else {
59                None
60            };
61            coordinates.push(coordinate);
62        }
63        Ok(Self(coordinates))
64    }
65
66    pub(super) fn bytes(&self) -> u128 {
67        self.0.capacity() as u128 * size_of::<Option<Coordinate>>() as u128
68    }
69
70    /// A prepared position belongs only to the template's own dense slot.
71    /// Missing coordinates retain ordinary key evaluation, including a
72    /// ground tuple absent from an empty inferred layout.
73    pub(in crate::oracle) fn position(
74        &self,
75        template: usize,
76        slot: usize,
77    ) -> Result<Option<usize>, Stop> {
78        self.0
79            .get(template)
80            .copied()
81            .flatten()
82            .map(|coordinate| {
83                if coordinate.slot == slot {
84                    Ok(coordinate.position)
85                } else {
86                    Err(Stop::InvalidProgram)
87                }
88            })
89            .transpose()
90    }
91}
