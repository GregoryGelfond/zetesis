1//! Relations in predicate order with the exact sum of their retained capacity.
2//!
3//! Accounting reads the sum on every storage decision, and a candidate can hold
4//! thousands of relations, so the sum is maintained rather than recomputed. The
5//! fields are private to this module: a relation's capacity can change only
6//! through [`RelationMut`], whose release records the change.
7
8use std::{
9    mem::size_of,
10    ops::{Deref, DerefMut},
11};
12
13use super::{Relation, Stop, Work, storage};
14
15/// The candidate's relations in semantic predicate order and the sum of their
16/// retained capacities, exact at every read.
17#[derive(Default)]
18pub(in crate::oracle) struct AccountedRelations {
19    relations: Vec<Relation>,
20    retained: u128,
21}
22
23impl AccountedRelations {
24    pub(super) fn as_slice(&self) -> &[Relation] {
25        &self.relations
26    }
27
28    pub(super) fn len(&self) -> usize {
29        self.relations.len()
30    }
31
32    /// Capacity of the directory itself: one relation header per slot.
33    pub(super) fn directory_bytes(&self) -> u128 {
34        self.relations.capacity() as u128 * size_of::<Relation>() as u128
35    }
36
37    /// Summed retained capacity of the relations, read without visiting them.
38    /// Debug builds recompute the sum and check it.
39    pub(super) fn retained_bytes(&self) -> u128 {
40        debug_assert_eq!(
41            self.retained,
42            self.relations
43                .iter()
44                .map(Relation::retained_bytes)
45                .sum::<u128>(),
46            "every relation capacity change is recorded"
47        );
48        self.retained
49    }
50
51    /// Reserve directory slots under the closure storage ceiling, as
52    /// [`storage::reserve`] does; relation contents are unchanged.
53    pub(super) fn reserve(
54        &mut self,
55        additional: usize,
56        live: u128,
57        work: &mut Work<'_>,
58    ) -> Result<u128, Stop> {
59        storage::reserve(&mut self.relations, additional, live, work)
60    }
61
62    /// Place a new relation at its predicate-order position and add its
63    /// capacity. The caller has reserved the slot.
64    pub(super) fn insert(&mut self, position: usize, relation: Relation) {
65        self.retained += relation.retained_bytes();
66        self.relations.insert(position, relation);
67    }
68
69    /// Borrow one relation for a change. Its capacity is measured now and again
70    /// when the borrow ends, whether by completion, an early return or
71    /// unwinding, and the sum takes the difference.
72    pub(super) fn get_mut(&mut self, index: usize) -> Option<RelationMut<'_>> {
73        let relation = self.relations.get_mut(index)?;
74        let before = relation.retained_bytes();
75        Some(RelationMut {
76            relation,
77            retained: &mut self.retained,
78            before,
79        })
80    }
81}
82
83/// One relation borrowed for a change, with the sum it contributes to.
84pub(in crate::oracle) struct RelationMut<'a> {
85    relation: &'a mut Relation,
86    retained: &'a mut u128,
87    before: u128,
88}
89
90impl Deref for RelationMut<'_> {
91    type Target = Relation;
92    fn deref(&self) -> &Relation {
93        self.relation
94    }
95}
96
97impl DerefMut for RelationMut<'_> {
98    fn deref_mut(&mut self) -> &mut Relation {
99        self.relation
100    }
101}
102
103impl Drop for RelationMut<'_> {
104    // The sum contains `before`, so replacing it with the current capacity
105    // cannot underflow.
106    fn drop(&mut self) {
107        *self.retained = *self.retained - self.before + self.relation.retained_bytes();
108    }
109}
