12pub(crate) struct Restrictions {
13    forbidden: Vec<Vec<CarrierAtom>>,
14    pub(crate) atoms: usize,
15    pub(crate) bytes: usize,
16    pub(crate) peak_bytes: usize,

18pub(crate) struct Attempt {
19    pub(crate) result: Result<Restrictions, Stop>,
20    pub(crate) work: u64,
21}
22pub(crate) enum Conflict {
23    Unconditional,
24    Selected(usize),
25}
26
27impl Restrictions {
28    pub(crate) fn conjunctions(&self) -> usize {
29        self.forbidden.len()
30    }
31    pub(crate) fn compile(
32        program: &Program,
33        limits: CandidateRestrictionLimits,
34        cancellation: &Cancellation,
35    ) -> Attempt {
36        let mut work = Work::source(cancellation, limits.max_work);
37        let result = compile(program, limits, &mut work);
38        Attempt {
39            result,
40            work: work.statistics.work,
41        }
42    }
43    pub(crate) fn conflict(
44        &self,
45        necessary: &[CarrierAtom],
46        atoms: &[Arc<GateAtom>],
47        bits: &[bool],
48        max_work: u64,
49        cancellation: &Cancellation,
50    ) -> (Result<Option<Conflict>, Stop>, u64) {
51        let mut work = Work::source(cancellation, max_work);
52        let result = self.find_conflict(necessary, atoms, bits, &mut work);
53        (result, work.statistics.work)
54    }
55    fn find_conflict(
56        &self,
57        necessary: &[CarrierAtom],
58        atoms: &[Arc<GateAtom>],
59        bits: &[bool],
60        work: &mut Work<'_>,
61    ) -> Result<Option<Conflict>, Stop> {
62        for forbidden in &self.forbidden {
63            work.tick()?;
64            let mut first = None;
65            let mut matched = true;
66            for premise in forbidden {
67                if held(premise, necessary, work)? {
68                    continue;
69                }
70                let Some(index) = selected_index(premise, atoms, bits, work)? else {
71                    matched = false;
72                    break;
73                };
74                first = Some(first.map_or(index, |previous: usize| previous.min(index)));
75            }
76            if matched {
77                return Ok(Some(
78                    first.map_or(Conflict::Unconditional, Conflict::Selected),
79                ));
80            }
81        }
82        Ok(None)
83    }
84}

86fn held(
87    premise: &CarrierAtom,
88    necessary: &[CarrierAtom],
89    work: &mut Work<'_>,
90) -> Result<bool, Stop> {
91    let (mut low, mut high) = (0, necessary.len());
92    while low < high {
93        let middle = low + (high - low) / 2;
94        work.tick()?;
95        match necessary[middle]
96            .atom()
97            .compare_ref_with(premise.atom(), || work.tick())?
98        {
99            std::cmp::Ordering::Less => low = middle + 1,
100            std::cmp::Ordering::Greater => high = middle,
101            std::cmp::Ordering::Equal => return Ok(true),
102        }
103    }
104    Ok(false)
105}
106fn selected_index(
107    premise: &CarrierAtom,
108    atoms: &[Arc<GateAtom>],
109    bits: &[bool],
110    work: &mut Work<'_>,
111) -> Result<Option<usize>, Stop> {
112    let (mut low, mut high) = (0, atoms.len());
113    while low < high {
114        let middle = low + (high - low) / 2;
115        work.tick()?;
116        match atoms[middle]
117            .atom()
118            .compare_ref_with(premise.atom(), || work.tick())?
119        {
120            std::cmp::Ordering::Less => low = middle + 1,
121            std::cmp::Ordering::Greater => high = middle,
122            std::cmp::Ordering::Equal => return Ok(bits[middle].then_some(middle)),
123        }
124    }
125    Ok(None)
126}

128/// Named live coordinate and metadata allocations. Program payload is shared
129/// once by the caller, never charged once per fact or premise. Buffer growth
130/// admits old/replacement overlap; allocator bookkeeping is excluded.
131struct Storage {
132    bytes: u128,
133    peak: u128,
134    limit: usize,
135}
136impl Storage {
137    fn admit(&mut self, bytes: u128) -> Result<(), Stop> {
138        if bytes > self.limit as u128 {
139            return Err(Stop::Allocation);
140        }
141        self.peak = self.peak.max(bytes);
142        Ok(())
143    }
144    fn reserve<T>(
145        &mut self,
146        values: &mut Vec<T>,
147        additional: usize,
148        work: &mut Work<'_>,
149    ) -> Result<(), Stop> {
150        let required = values
151            .len()
152            .checked_add(additional)
153            .ok_or(Stop::Allocation)?;
154        if required <= values.capacity() {
155            return Ok(());
156        }
157        let target = required.max(values.capacity().saturating_mul(2)).max(4);
158        let old = buffer_bytes(values);
159        self.admit(self.bytes + target as u128 * size_of::<T>() as u128)?;
160        work.charge(values.len().max(1))?;
161        values
162            .try_reserve_exact(target - values.len())
163            .map_err(|_| Stop::Allocation)?;
164        self.admit(self.bytes + buffer_bytes(values))?;
165        self.bytes = self.bytes - old + buffer_bytes(values);
166        Ok(())
167    }
168    fn release<T>(&mut self, values: Vec<T>) {
169        self.bytes -= buffer_bytes(&values);
170        drop(values);
171    }
172    fn remaining(&self) -> Result<usize, Stop> {
173        usize::try_from(
174            (self.limit as u128)
175                .checked_sub(self.bytes)
176                .ok_or(Stop::Allocation)?,
177        )
178        .map_err(|_| Stop::Allocation)
179    }
180}
181fn buffer_bytes<T>(values: &Vec<T>) -> u128 {
182    values.capacity() as u128 * size_of::<T>() as u128
183}
184fn located(
185    program: &Program,
186    pattern: PatternRef<'_>,
187    assignment: &[Option<TermRef<'_>>],
188    gates_only: bool,
189    remaining_atoms: &mut usize,
190    storage: &mut Storage,
191    work: &mut Work<'_>,
192) -> Result<CarrierAtom, Stop> {
193    if *remaining_atoms == 0 {
194        return Err(Stop::CarrierLimit);
195    }
196    work.charge(pattern.terms().len())?;
197    let key = pattern.key(assignment).map_err(|_| Stop::InvalidProgram)?;
198    let atom = program
199        .locate_key_with(&key, gates_only, storage.remaining()?, || work.tick())
200        .map_err(|error| match error {
201            CarrierFailure::Stopped(stop) => stop,
202            CarrierFailure::Storage(zetesis_core::CarrierError::Coordinates) => {
203                Stop::InvalidProgram
204            }
205            CarrierFailure::Storage(_) => Stop::Allocation,
206        })?
207        .ok_or(Stop::InvalidProgram)?;
208    storage.bytes += atom.coordinate_bytes();
209    storage.admit(storage.bytes)?;
210    *remaining_atoms -= 1;
211    Ok(atom)
212}

214/// Discover and deduplicate actual unconditional facts before they become the
215/// immutable input rows for constraint joins. Located coordinates and sorting
216/// retain the same storage account used by the later joins.
217fn unconditional_facts(
218    program: &Program,
219    remaining_atoms: &mut usize,
220    storage: &mut Storage,
221    work: &mut Work<'_>,
222) -> Result<Vec<CarrierAtom>, Stop> {
223    let mut facts: Vec<CarrierAtom> = Vec::new();
224    for template in program.templates() {
225        work.tick()?;
226        if template.positive().is_empty()
227            && template.gate_true().is_empty()
228            && template.gate_false().is_empty()
229            && template.filters().is_empty()
230            && let Some(head) = template.head()
231        {
232            let atom = located(program, head, &[], false, remaining_atoms, storage, work)?;
233            let (mut low, mut high) = (0, facts.len());
234            let mut duplicate = false;
235            while low < high {
236                let middle = low + (high - low) / 2;
237                work.tick()?;
238                match facts[middle]
239                    .atom()
240                    .compare_ref_with(atom.atom(), || work.tick())?
241                {
242                    std::cmp::Ordering::Less => low = middle + 1,
243                    std::cmp::Ordering::Greater => high = middle,
244                    std::cmp::Ordering::Equal => {
245                        duplicate = true;
246                        break;
247                    }
248                }
249            }
250            if duplicate {
251                storage.bytes -= atom.coordinate_bytes();
252            } else {
253                storage.reserve(&mut facts, 1, work)?;
254                work.charge(facts.len() - low + 1)?;
255                facts.insert(low, atom);
256            }
257        }
258    }
259    Ok(facts)
260}

262fn compile(
263    program: &Program,
264    limits: CandidateRestrictionLimits,
265    work: &mut Work<'_>,
266) -> Result<Restrictions, Stop> {
267    let mut eligible = false;
268    for template in program.templates() {
269        work.tick()?;
270        eligible |= template.head().is_none() && template.gate_false().is_empty();
271    }
272    if !eligible {
273        return Ok(Restrictions {
274            forbidden: Vec::new(),
275            atoms: 0,
276            bytes: 0,
277            peak_bytes: 0,
278        });
279    }
280    let mut storage = Storage {
281        bytes: 0,
282        peak: 0,
283        limit: limits.max_bytes,
284    };
285    let mut remaining_atoms = limits.max_atoms;
286    let facts = unconditional_facts(program, &mut remaining_atoms, &mut storage, work)?;
287    // Only actual unconditional facts enter the shared join; neither possible
288    // support nor the entire carrier is supplied as a source truth relation.
289    let mut relations = Relations::new();
290    storage.bytes += relations.retained_bytes();
291    storage.admit(storage.bytes)?;
292    for atom in &facts {
293        let previous = relations.retained_bytes();
294        relations.push_with(
295            atom.atom(),
296            storage.bytes - previous,
297            limits.max_bytes,
298            &mut storage.peak,
299            work,
300        )?;
301        storage.bytes = storage.bytes - previous + relations.retained_bytes();
302        storage.admit(storage.bytes)?;
303    }
304    let facts_bytes = storage.bytes;
305    let mut forbidden = Vec::new();
306    for template in program.templates() {
307        work.tick()?;
308        if template.head().is_some() || !template.gate_false().is_empty() {
309            continue;
310        }
311        let Some(partition) = lift(program, template, &mut storage, work)? else {
312            continue;
313        };
314        let query = template.with_patterns(&partition.positive, &partition.gates);
315        let scratch = visit(
316            query,
317            &relations,
318            super::Gates::Unjudged,
319            None,
320            super::QueryStorage {
321                retained_bytes: storage.bytes,
322                max_bytes: limits.max_bytes,
323            },
324            work,
325            |assignment, scratch, work| {
326                storage.bytes += scratch;
327                storage.admit(storage.bytes)?;
328                storage.reserve(&mut forbidden, 1, work)?;
329                let mut conjunction = Vec::new();
330                storage.reserve(&mut conjunction, query.gate_true().len(), work)?;
331                for premise in query.gate_true() {
332                    conjunction.push(located(
333                        program,
334                        premise,
335                        assignment,
336                        true,
337                        &mut remaining_atoms,
338                        &mut storage,
339                        work,
340                    )?);
341                }
342                work.tick()?;
343                forbidden.push(conjunction);
344                storage.bytes -= scratch;
345                Ok::<(), Stop>(())
346            },
347        )?;
348        storage.admit(storage.bytes + scratch)?;
349        storage.bytes -= buffer_bytes(&partition.positive) + buffer_bytes(&partition.gates);
350    }
351    storage.bytes -= facts_bytes;
352    Ok(Restrictions {
353        forbidden,
354        atoms: limits.max_atoms - remaining_atoms,
355        bytes: usize::try_from(storage.bytes).map_err(|_| Stop::Allocation)?,
356        peak_bytes: usize::try_from(storage.peak).map_err(|_| Stop::Allocation)?,
357    })
358}

360struct Partition<'a> {
361    positive: Vec<PatternRef<'a>>,
362    gates: Vec<PatternRef<'a>>,
363}
364/// Every variable must be bound by nongate facts. Other constraints remain for
365/// the membership oracle. The partition copies pattern handles, never payload.
366fn lift<'a>(
367    program: &Program,
368    template: TemplateRef<'a>,
369    storage: &mut Storage,
370    work: &mut Work<'_>,
371) -> Result<Option<Partition<'a>>, Stop> {
372    let mut bound = Vec::new();
373    storage.reserve(&mut bound, template.variable_count(), work)?;
374    work.charge(template.variable_count())?;
375    bound.resize(template.variable_count(), false);
376    let mut positive = Vec::new();
377    let mut gates = Vec::new();
378    for pattern in template.positive() {
379        work.tick()?;
380        if program
381            .gate_predicates()
382            .binary_search_with(pattern.predicate(), || work.tick())?
383            .is_err()
384        {
385            storage.reserve(&mut positive, 1, work)?;
386            for term in pattern.terms() {
387                work.tick()?;
388                if let TemplateTerm::Variable(variable) = term {
389                    *bound.get_mut(variable).ok_or(Stop::InvalidProgram)? = true;
390                }
391            }
392            work.tick()?;
393            positive.push(pattern);
394        } else {
395            storage.reserve(&mut gates, 1, work)?;
396            work.tick()?;
397            gates.push(pattern);
398        }
399    }
400    let mut complete = true;
401    for present in &bound {
402        work.tick()?;
403        complete &= *present;
404    }
405    storage.release(bound);
406    if !complete {
407        storage.bytes -= buffer_bytes(&positive) + buffer_bytes(&gates);
408        return Ok(None);
409    }
410    for pattern in template.gate_true() {
411        storage.reserve(&mut gates, 1, work)?;
412        work.tick()?;
413        gates.push(pattern);
414    }
415    Ok(Some(Partition { positive, gates }))
