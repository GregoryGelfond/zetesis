1//! Regions of a candidate space and the traversal that covers them.
2//!
3//! A region decides some atoms in, some out, and leaves the rest open; its
4//! candidates are the sets that agree with its decisions, the `Cube` of
5//! `Search.lean` indexed over the atoms of a root, as the oracle's `Cube`
6//! is the same cube over atoms. The traversal walks the coverage tree of that chapter
7//! with a stack: a region is narrowed by a caller-supplied narrowing, and
8//! is then refuted, a leaf when every atom is decided, split on the open
9//! atom its narrowing preferred or else its highest open atom, or counted
10//! when its narrowing decided nothing beyond the split and the caller
11//! counts such regions. The cut branch is visited before the held one, so
12//! with no preference the leaves come in the order of a binary counter over
13//! the atoms with atom zero as its low bit (`Cube.split_partition` and
14//! `Cube.split_disjoint` of `Search.lean`). The root is always split.
15//!
16//! The narrowing is the caller's: on the closure route it is the program's
17//! two closures under the region's gate readings, on the formula route the
18//! theory's readings. The traversal only requires that a narrowing never
19//! removes an accepted candidate from the region and never refutes a region
20//! holding one; each visit then keeps every accepted candidate of the root
21//! in exactly one region still to visit, visited as a leaf, or counted.
22//!
23//! Termination: a visit pops one region and pushes at most two, each with
24//! one more atom decided, so the number of regions ever pushed is bounded by
25//! twice the number of candidates of the root.
26

31/// A region of candidates: every atom held, cut or open, and, once
32/// narrowed, the open atom its narrowing would split on. Its decisions are two
33/// atom bitmasks, one for held and one for cut; a narrowing that carries
34/// knowledge from the region's parent applies only the decisions the parent
35/// had not yet seen, by diffing this region's decided mask against a snapshot.
36#[derive(Clone, Debug)]
37pub struct Region {
38    /// Bit set iff the atom is held. Disjoint from `cut` word for word, and
39    /// bits at `[atoms, 64*words)` are always zero and are not atoms.
40    held: Box<[u64]>,
41    /// Bit set iff the atom is cut.
42    cut: Box<[u64]>,
43    /// The atom universe; each mask is `words(atoms)` words long.
44    atoms: usize,
45    preferred: Option<usize>,
46}
47
48/// The number of 64-bit words a mask over `atoms` atoms needs.
49const fn words(atoms: usize) -> usize {
50    atoms.div_ceil(64)
51}
52
53/// The word index and single-bit mask of an atom.
54const fn locate(atom: usize) -> (usize, u64) {
55    (atom / 64, 1u64 << (atom % 64))
56}
57
58/// The set-bit positions of a mask, ascending, keeping only real atoms.
59fn set_bits(mask: &[u64], atoms: usize) -> impl Iterator<Item = usize> + '_ {
60    mask.iter().enumerate().flat_map(move |(word, &bits)| {
61        let base = word * 64;
62        SetBits { bits }
63            .map(move |bit| base + bit)
64            .filter(move |&atom| atom < atoms)
65    })
66}
67
68/// Iterates the set-bit positions of one word, ascending.
69struct SetBits {
70    bits: u64,
71}
72impl Iterator for SetBits {
73    type Item = usize;
74    fn next(&mut self) -> Option<usize> {
75        (self.bits != 0).then(|| {
76            let bit = self.bits.trailing_zeros() as usize;
77            self.bits &= self.bits - 1;
78            bit
79        })
80    }
81}

83/// Two regions are the same when they decide the same atoms the same way over
84/// the same universe and prefer the same split; the order the decisions were
85/// made in is history, not identity.
86impl PartialEq for Region {
87    fn eq(&self, other: &Self) -> bool {
88        self.atoms == other.atoms
89            && self.held == other.held
90            && self.cut == other.cut
91            && self.preferred == other.preferred
92    }
93}
94impl Eq for Region {}

96impl Region {
97    /// Header and owned vector capacities in bytes, including unused capacity.
98    /// Excludes allocator bookkeeping; this is not a process-memory estimate.
99    #[must_use]
100    pub fn retained_bytes(&self) -> u128 {
101        size_of::<Self>() as u128
102            + (self.held.len() + self.cut.len()) as u128 * size_of::<u64>() as u128
103    }
104
105    /// The region in which every atom is open: every candidate lies in it.
106    #[must_use]
107    pub fn all_open(atoms: usize) -> Self {
108        Self {
109            held: vec![0; words(atoms)].into_boxed_slice(),
110            cut: vec![0; words(atoms)].into_boxed_slice(),
111            atoms,
112            preferred: None,
113        }
114    }
115    /// The number of atoms the region decides over.
116    #[must_use]
117    pub fn len(&self) -> usize {
118        self.atoms
119    }
120    /// Whether the region decides over no atom at all.
121    #[must_use]
122    pub fn is_empty(&self) -> bool {
123        self.atoms == 0
124    }
125    /// Hold an atom in every candidate; `false` when the atom is cut or
126    /// outside the region.
127    pub fn hold(&mut self, atom: usize) -> bool {
128        self.decide(atom, true)
129    }
130    /// Cut an atom from every candidate; `false` when the atom is held or
131    /// outside the region.
132    pub fn cut(&mut self, atom: usize) -> bool {
133        self.decide(atom, false)
134    }
135    fn decide(&mut self, atom: usize, value: bool) -> bool {
136        if atom >= self.atoms {
137            return false;
138        }
139        let (word, bit) = locate(atom);
140        if self.held[word] & bit != 0 {
141            return value;
142        }
143        if self.cut[word] & bit != 0 {
144            return !value;
145        }
146        if value {
147            self.held[word] |= bit;
148        } else {
149            self.cut[word] |= bit;
150        }
151        true
152    }
153    /// The atom's decision: held, cut, or open; `None` outside the region too.
154    #[must_use]
155    pub fn decision(&self, atom: usize) -> Option<bool> {
156        if atom >= self.atoms {
157            return None;
158        }
159        let (word, bit) = locate(atom);
160        if self.held[word] & bit != 0 {
161            Some(true)
162        } else if self.cut[word] & bit != 0 {
163            Some(false)
164        } else {
165            None
166        }
167    }
168    /// Whether every candidate holds the atom.
169    #[must_use]
170    pub fn is_held(&self, atom: usize) -> bool {
171        self.decision(atom) == Some(true)
172    }
173    /// Whether no candidate holds the atom.
174    #[must_use]
175    pub fn is_cut(&self, atom: usize) -> bool {
176        self.decision(atom) == Some(false)
177    }
178    /// Whether the atom is open: some candidates hold it and some do not.
179    #[must_use]
180    pub fn is_open(&self, atom: usize) -> bool {
181        atom < self.atoms && {
182            let (word, bit) = locate(atom);
183            (self.held[word] | self.cut[word]) & bit == 0
184        }
185    }
186    /// The highest open atom, the one a split decides when no atom is
187    /// preferred.
188    #[must_use]
189    pub fn highest_open(&self) -> Option<usize> {
190        for word in (0..self.held.len()).rev() {
191            let mut open = !(self.held[word] | self.cut[word]);
192            // Ignore the unused high bits of the top partial word: they are not
193            // atoms and would otherwise read as open.
194            let valid = self.atoms - word * 64;
195            if valid < 64 {
196                open &= (1u64 << valid) - 1;
197            }
198            if open != 0 {
199                return Some(word * 64 + (63 - open.leading_zeros() as usize));
200            }
201        }
202        None
203    }
204    /// Prefer an open atom for the next split, as a narrowing may after
205    /// reading the region; a decided atom is not retained.
206    pub fn prefer(&mut self, atom: usize) {
207        self.preferred = Some(atom).filter(|&atom| self.is_open(atom));
208    }
209    /// The atom the next split decides: the preferred open atom, else the
210    /// highest open one.
211    #[must_use]
212    pub fn split_atom(&self) -> Option<usize> {
213        self.preferred
214            .filter(|&atom| self.is_open(atom))
215            .or_else(|| self.highest_open())
216    }
217    /// The atoms every candidate holds, ascending.
218    pub fn held(&self) -> impl Iterator<Item = usize> + '_ {
219        set_bits(&self.held, self.atoms)
220    }
221    /// The open atoms, ascending.
222    pub fn open(&self) -> impl Iterator<Item = usize> + '_ {
223        (0..self.atoms).filter(move |&atom| {
224            let (word, bit) = locate(atom);
225            (self.held[word] | self.cut[word]) & bit == 0
226        })
227    }
228    /// The atoms this region decides that are not set in `seen`, each paired
229    /// with whether it is held, ascending. `seen` is a decided-mask snapshot
230    /// over the same atoms; words missing from `seen` count as unseen. Reports
231    /// only real atoms, never a tail bit.
232    pub fn decided_since<'a>(
233        &'a self,
234        seen: &'a [u64],
235    ) -> impl Iterator<Item = (usize, bool)> + 'a {
236        (0..self.held.len()).flat_map(move |word| {
237            let seen_word = seen.get(word).copied().unwrap_or(0);
238            let fresh = (self.held[word] | self.cut[word]) & !seen_word;
239            let held = self.held[word];
240            let base = word * 64;
241            SetBits { bits: fresh }
242                .map(move |bit| (base + bit, held & (1u64 << bit) != 0))
243                .filter(move |&(atom, _)| atom < self.atoms)
244        })
245    }
246    /// Copy this region's decided mask (`held | cut`) into the available prefix
247    /// of `dst`. A complete snapshot needs `self.len().div_ceil(64)` words.
248    /// A shorter destination receives only its prefix; any excess destination
249    /// words remain unchanged. An empty destination is valid. Missing snapshot
250    /// words count as unseen when passed to [`Self::decided_since`].
251    pub fn snapshot_decided(&self, dst: &mut [u64]) {
252        for (word, (&held, &cut)) in dst.iter_mut().zip(self.held.iter().zip(self.cut.iter())) {
253            *word = held | cut;
254        }
255    }
256    /// Every decided atom with whether it is held, ascending.
257    pub fn decided(&self) -> impl Iterator<Item = (usize, bool)> + '_ {
258        self.decided_since(&[])
259    }
260    /// The two regions an open atom splits this one into: cut, then held.
261    /// Neither inherits a preference; their narrowing sets their own.
262    ///
263    /// # Panics
264    /// Panics unless `atom` is one of the region's open atoms.
265    #[must_use]
266    pub fn split(mut self, atom: usize) -> (Self, Self) {
267        // A region is split on one of its OPEN atoms: an out-of-range or
268        // already-decided atom fails fast, since deciding it would either touch
269        // a non-atom bit or leave a narrowing's knowledge out of step with the
270        // region's decision (a reader's seen-mask already covers a decided atom).
271        assert!(
272            self.is_open(atom),
273            "a region is split on one of its open atoms"
274        );
275        let (word, bit) = locate(atom);
276        // Clone once for the held child; reuse this region as the cut child
277        // rather than cloning it a second time.
278        let mut held = self.clone();
279        held.held[word] |= bit;
280        held.preferred = None;
281        self.cut[word] |= bit;
282        self.preferred = None;
283        (self, held)
284    }
285}

287/// The outcome of narrowing one region.
288#[derive(Clone, Copy, Debug, PartialEq, Eq)]
289pub enum Narrowing {
290    /// No accepted candidate lies in the region.
291    Refuted,
292    /// The fixed point, and whether the narrowing decided an atom.
293    Fixed {
294        /// Whether any atom was decided by the narrowing.
295        changed: bool,
296    },
297}
298
299/// When a region is counted instead of split.
300#[derive(Clone, Copy, Debug, PartialEq, Eq)]
301pub enum Counting {
302    /// Every region is split until decided: the traversal visits only leaves.
303    Never,
304    /// A region below the root whose narrowing decided nothing is counted:
305    /// its open atoms are offered to a flat counter by the caller.
306    Unchanged,
307}
308
309/// One region the traversal hands to the caller, with what the caller
310/// carries alongside it.
311#[derive(Clone, Debug, PartialEq, Eq)]
312pub enum Visit<S = ()> {
313    /// Every atom is decided: the region is one candidate.
314    Leaf(Region, S),
315    /// The region's open atoms are the caller's to count.
316    Counted(Region, S),
317}
318
319/// The regions a traversal visited, by outcome.
320#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
321pub struct RegionStatistics {
322    /// Regions popped and narrowed, the root included.
323    pub regions: usize,
324    /// Regions whose narrowing refuted them.
325    pub refuted: usize,
326    /// Regions visited as leaves: every atom decided, one candidate each.
327    pub leaves: usize,
328    /// Regions handed to the caller to count.
329    pub counted: usize,
330}
331
332impl RegionStatistics {
333    /// The regions split since `before`: every region visited that was
334    /// neither refuted nor a leaf nor counted.
335    #[must_use]
336    pub fn splits_since(self, before: Self) -> usize {
337        (self.regions - before.regions)
338            - (self.refuted - before.refuted)
339            - (self.leaves - before.leaves)
340            - (self.counted - before.counted)
341    }
342}

344/// The coverage tree of a root region, walked depth first. Each region
345/// carries the caller's state `S`, what the narrowing knows about the
346/// region; a split clones it into both children, so that a child starts
347/// from its parent's knowledge and the regions share nothing.
348#[derive(Clone, Debug)]
349pub struct Traversal<S = ()> {
350    /// The regions still to visit with their state, the next on top.
351    regions: Vec<(Region, S)>,
352    counting: Counting,
353    /// The root has not been visited yet; it is split whatever its narrowing.
354    root: bool,
355    /// The caller narrowed the root before the traversal began, so it is
356    /// not narrowed again.
357    narrowed_root: bool,
358    statistics: RegionStatistics,
359}
360
361impl Traversal {
362    /// A traversal of every candidate of the root, carrying nothing.
363    #[must_use]
364    pub fn new(root: Region, counting: Counting) -> Self {
365        Self::with_state(root, counting, ())
366    }
367}
368
369impl<S: Clone> Traversal<S> {
370    /// A traversal of every candidate of the root, carrying `state` with the
371    /// root and a clone of a region's state with each of its children.
372    #[must_use]
373    pub fn with_state(root: Region, counting: Counting, state: S) -> Self {
374        Self {
375            regions: vec![(root, state)],
376            counting,
377            root: true,
378            narrowed_root: false,
379            statistics: self::RegionStatistics::default(),
380        }
381    }
382
383    /// A traversal of every candidate of a root the caller has narrowed
384    /// already: the root is visited without a narrowing and split whatever
385    /// that narrowing reported, and every region below it is narrowed by
386    /// the caller's narrowing.
387    #[must_use]
388    pub fn with_narrowed_root(root: Region, counting: Counting, state: S) -> Self {
389        Self {
390            narrowed_root: true,
391            ..Self::with_state(root, counting, state)
392        }
393    }
394
395    /// The regions visited so far, by outcome.
396    #[must_use]
397    pub const fn statistics(&self) -> RegionStatistics {
398        self.statistics
399    }
400
401    /// Narrow regions until one is a leaf or is counted; `None` once every
402    /// region has been visited. The narrowing's failure type is the caller's;
403    /// a failed reservation for the split is reported as `Stop::Allocation`.
404    ///
405    /// # Errors
406    /// A stopped narrowing stops the traversal with its region kept on the
407    /// stack, so a later call narrows the same region again.
408    pub fn next<E: From<Stop>>(
409        &mut self,
410        mut narrow: impl FnMut(&mut Region, &mut S) -> Result<Narrowing, E>,
411    ) -> Result<Option<Visit<S>>, E> {
412        loop {
413            let Some((mut region, mut state)) = self.regions.pop() else {
414                return Ok(None);
415            };
416            let changed = if std::mem::take(&mut self.narrowed_root) {
417                true
418            } else {
419                match narrow(&mut region, &mut state) {
420                    Ok(Narrowing::Refuted) => {
421                        self.statistics.regions += 1;
422                        self.statistics.refuted += 1;
423                        self.root = false;
424                        continue;
425                    }
426                    Ok(Narrowing::Fixed { changed }) => changed,
427                    Err(stop) => {
428                        self.regions.push((region, state));
429                        return Err(stop);
430                    }
431                }
432            };
433            self.statistics.regions += 1;
434            let root = std::mem::replace(&mut self.root, false);
435            let Some(atom) = region.split_atom() else {
436                self.statistics.leaves += 1;
437                return Ok(Some(Visit::Leaf(region, state)));
438            };
439            if self.counting == Counting::Unchanged && !root && !changed {
440                self.statistics.counted += 1;
441                return Ok(Some(Visit::Counted(region, state)));
442            }
443            let (cut, held) = region.split(atom);
444            self.regions
445                .try_reserve(2)
446                .map_err(|_| E::from(Stop::Allocation))?;
447            self.regions.push((held, state.clone()));
448            self.regions.push((cut, state));
449        }
450    }
