1//! Exact finite-table selection and projection over an immutable typed relation.
2//!
3//! A surviving row is a tuple in the supplied table, not an ASP producer, a
4//! true atom, or an answer set. Source completeness and reduct membership remain
5//! separate obligations. Equality and aliasing use whole typed values.
6//!
7//! Preparation borrows the authoritative relation and builds value-to-row
8//! bitsets. Every projection starts from the same immutable coherent rows;
9//! domains may narrow, widen, or be restored in any order. No trail or previous
10//! domain cardinality is trusted. Row masks borrow only the authoritative
11//! relation; projected value domains also borrow the prepared table.
12

28/// Inclusive limits for one preparation, selection or projection operation.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub struct Limits {
31    /// Maximum distinct variable/value support entries across this table.
32    pub max_entries: usize,
33    /// Live operation capacity, including the relation and prepared input.
34    /// Borrowed source payload and other caller-owned results are excluded.
35    pub max_bytes: usize,
36    /// Charged cells, bitwise operations, typed descriptors/navigation and
37    /// compared text-byte prefixes.
38    pub max_work: u64,
39}
40
41impl Default for Limits {
42    fn default() -> Self {
43        Self {
44            max_entries: 1_048_576,
45            max_bytes: 134_217_728,
46            max_work: 100_000_000,
47        }
48    }
49}

51/// A finite table-operation resource.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum Resource {
54    /// Distinct variable/value entries, including the same value in different variables.
55    Entries,
56    /// Live operation-scoped capacity.
57    Bytes,
58    /// Charged logical operations and visited typed comparison steps.
59    Work,
60}

62/// An incomplete operation, never a conclusion about program consistency.
63#[derive(Clone, Debug, PartialEq, Eq)]
64pub enum Cause {
65    /// Column variables were not canonical consecutive first-occurrence labels.
66    Scope,
67    /// The supplied domain count differs from the variable count.
68    Domains,
69    /// A trusted relation could not decode one of its own rows.
70    Relation,
71    /// A capacity or work sum cannot be represented.
72    Overflow,
73    /// A fallible vector reservation failed.
74    Allocation,
75    /// Cancellation or deadline expiry was observed.
76    Interrupted(Stop),
77    /// An inclusive finite ceiling was exceeded.
78    Limit {
79        /// Exhausted resource.
80        resource: Resource,
81        /// Proposed amount, including the refused operation.
82        observed: u128,
83        /// Inclusive allowance.
84        limit: u128,
85    },
86}

88/// Failure with the completed work prefix and conservative capacity peak.
89/// No partial table or result is published. Existing borrowed objects remain valid.
90#[derive(Clone, Debug, PartialEq, Eq)]
91pub struct Failure {
92    /// Original cause, without converting an interruption into an empty table.
93    pub cause: Cause,
94    /// Work charged before the refused operation.
95    pub work: u64,
96    /// Largest accounted capacity, including conservatively coexisting reallocation buffers.
97    pub peak_bytes: usize,
98}

107/// A table refusal or an enclosing caller's admission/control refusal.
108#[derive(Clone, Debug, PartialEq, Eq)]
109pub enum MeteredCause<E> {

123/// A metered refusal with the actual admitted work and capacity prefix.
124/// No partial table or selection is published. Accepted caller permits are not
125/// refunded; `work` counts this operation only, not the caller's earlier history.
126#[derive(Clone, Debug, PartialEq, Eq)]
127pub struct MeteredFailure<E> {
128    /// Original local or caller refusal, retaining the caller's typed value.
129    pub cause: MeteredCause<E>,
130    /// Work admitted before the refused operation; a refused permit adds zero.
131    pub work: u64,
132    /// The same conservative operation-scoped capacity peak as [`Failure`].
133    pub peak_bytes: usize,
134}

136impl<E> From<Failure> for MeteredFailure<E> {
137    fn from(error: Failure) -> Self {
138        Self {
139            cause: MeteredCause::Table(error.cause),
140            work: error.work,
141            peak_bytes: error.peak_bytes,
142        }
143    }
144}

174/// Prepared support bitsets over one live immutable relation.
175///
176/// The scope labels argument columns by variable: `[0, 1, 0]` represents a
177/// repeated first variable. Labels must be consecutive in first-occurrence
178/// order. Rows disagreeing at aliased columns are excluded before projection.
179/// Equal row occurrences remain distinct positions, including catalog aliases.
180/// Value representatives borrow the relation's existing dictionary through
181/// [`TermRef`]; no logical value, atom or equality dictionary is duplicated.
182///
183/// With R rows, V distinct variable/value pairs, and W=ceil(R/32), support
184/// storage is V*W words. Preparation additionally sorts O(R) borrowed row cells
185/// per variable; typed comparisons include their payload costs. Large finite
186/// domains can make this index substantially larger than the original columns.
187pub struct Table<'owner, 'source> {
188    relation: &'owner Relation<'source>,
189    scope: Vec<usize>,
190    variables: Vec<Range<usize>>,
191    values: Vec<TermRef<'source>>,
192    supports: Vec<u32>,
193    coherent: Vec<u32>,
194    statistics: Statistics,
195}

197/// A complete finite-table projection for one supplied domain family.
198///
199/// Rows retain their original relation positions. A value is retained exactly
200/// when a surviving complete row carries it. Empty rows mean that this finite
201/// table/domain intersection is empty, not that an ASP program is inconsistent.
202pub struct Projection<'table, 'owner, 'source> {
203    table: &'table Table<'owner, 'source>,
204    rows: Vec<u32>,
205    supported: Vec<bool>,
206    statistics: Statistics,
207}

215impl<'owner, 'source> Table<'owner, 'source> {
216    /// Prepare a borrowed relation for repeated exact domain projection.
217    ///
218    /// # Errors
219    /// Refuses malformed scope, finite limits, allocation failure or interrupted
220    /// control. The relation is unchanged; no partial index is returned.
221    ///
222    /// # Examples
223    /// ```
224    /// use zetesis_core::{Atom, Predicate, Value, catalog::TermRef, relation::Relation};
225    /// use zetesis_cpu::{Cancellation, table::{Limits, Table}};
226    /// let predicate = Predicate::new("pair", 2)?;
227    /// let rows = [Atom::new(predicate.clone(), vec![Value::Number(1), Value::Number(2)])?];
228    /// let relation = Relation::from_atoms(&predicate, &rows, Default::default())?;
229    /// let table = Table::prepare(&relation, &[0, 1], Limits::default(), &Cancellation::default())?;
230    /// let first = [Value::Number(1), Value::Number(3)];
231    /// let second = [Value::Number(2)];
232    /// let projected = table.project(&[&first, &second], Limits::default(), &Cancellation::default())?;
233    /// assert!(projected.contains(0));
234    /// assert_eq!(projected.domain(0).unwrap().collect::<Vec<_>>(), vec![TermRef::from(&Value::Number(1))]);
235    /// # Ok::<(), Box<dyn std::error::Error>>(())
236    /// ```
237    pub fn prepare(
238        relation: &'owner Relation<'source>,
239        scope: &[usize],
240        limits: Limits,
241        cancellation: &Cancellation,
242    ) -> Result<Self, Failure> {
243        Self::prepare_with(relation, scope, limits, control(cancellation)).map_err(unmetered)
244    }
245
246    /// Prepare through a caller-owned admission and control callback.
247    ///
248    /// The same algorithm and local limits implement [`Self::prepare`]. The
249    /// callback receives zero at entry, allocation boundaries, and before each
250    /// local work-ceiling check: zero polls control and consumes no work. A
251    /// positive amount requests the entire next charged group after its local
252    /// ceiling check and before its effect or receipt increment. The callback
253    /// must admit that group atomically or return its original typed refusal.
254    /// Already accepted groups are never refunded. No callback is retained.
255    ///
256    /// This lets independent operations share a cumulative quota. Reading a
257    /// remaining allowance before the call alone cannot establish that guarantee.
258    /// Callers record the returned work receipt without charging it again.
259    ///
260    /// # Errors
261    /// Returns local shape/resource/allocation failures or the unchanged caller
262    /// admission/control error, with work and capacity retained on every outcome.
263    pub fn prepare_with<E>(
264        relation: &'owner Relation<'source>,
265        scope: &[usize],
266        limits: Limits,
267        before: impl FnMut(usize) -> Result<(), E>,
268    ) -> Result<Self, MeteredFailure<E>> {
269        let scratch_headers = 2 * size_of::<Vec<RowValue<'_>>>() + size_of::<Vec<usize>>();
270        let external = relation.storage().retained_bytes;
271        let mut work = Work::new(
272            limits,
273            before,
274            external,
275            size_of::<Self>() + scratch_headers,
276        )?;
277        let result = (|| {
278            if scope.len() != relation.predicate().arity() {
279                return Err(Cause::Scope.into());
280            }
281            let mut copied_scope = work.reserve(scope.len())?;
282            let mut first_columns = work.reserve(scope.len())?;
283            for (column, &variable) in scope.iter().enumerate() {
284                work.tick(1)?;
285                if variable > first_columns.len() {
286                    return Err(Cause::Scope.into());
287                }
288                if variable == first_columns.len() {
289                    work.tick(1)?;
290                    first_columns.push(column);
291                }
292                copied_scope.push(variable);
293            }
294            let words = relation.row_count().div_ceil(WORD_BITS);
295            let mut coherent = work.zeros(words)?;
296            for row in 0..relation.row_count() {
297                if coherent_row(relation, scope, &first_columns, row, &mut work)? {
298                    work.tick(1)?;
299                    coherent[row / WORD_BITS] |= 1 << (row % WORD_BITS);
300                }
301            }
302            let mut table = Self {
303                relation,
304                scope: copied_scope,
305                variables: work.reserve(first_columns.len())?,
306                values: Vec::new(),
307                supports: Vec::new(),
308                coherent,
309                statistics: Statistics {
310                    work: 0,
311                    retained_bytes: 0,
312                    peak_bytes: 0,
313                },
314            };
315            for &column in &first_columns {
316                table.prepare_variable(column, &mut work)?;
317            }
318            work.release(first_columns);
319            work.release_bytes(scratch_headers);
320            table.statistics = work.statistics(external);
321            Ok(table)
322        })();
323        result.map_err(|cause| work.failure(cause))
324    }

357    /// Intersect complete rows with the supplied domains and project their values.
358    ///
359    /// This ingress adapter borrows each owned value through [`TermRef`]; it
360    /// neither copies values nor builds another dictionary. Canonical domain
361    /// references use [`Self::project_domains`]. Duplicates have set meaning;
362    /// absent values have no support. Every call recomputes from the
363    /// immutable base, including after domain widening. For K variables, D
364    /// supplied domain values, V indexed entries, A allowed entries and W row
365    /// words, work is O(K + (K+1)*W + V + D*(1+log(V+1)+W) + A*W), plus typed
366    /// comparison payload costs. The first finite domain initializes the result
367    /// directly from its supports; later finite domains clear and intersect W
368    /// words even if empty. A nullary query still copies the base mask.
369    ///
370    /// # Errors
371    /// Refuses a wrong domain count, finite limits, allocation failure or
372    /// interrupted control. Existing projections and the table remain unchanged.
373    pub fn project(
374        &self,
375        domains: &[&[Value]],
376        limits: Limits,
377        cancellation: &Cancellation,
378    ) -> Result<Projection<'_, 'owner, 'source>, Failure> {
379        self.project_inner(
380            domains
381                .iter()
382                .map(|domain| Domain::Finite((*domain).into())),
383            limits,
384            cancellation,
385        )
386    }

388    /// Project borrowed finite, singleton or unrestricted domains.
389    ///
390    /// This shares row restriction and value projection with [`Self::project`].
391    /// The result borrows the prepared table, not the supplied domain storage.
392    /// An unrestricted variable permits every indexed value; a singleton needs
393    /// one typed lookup and no union scratch. Finite lists retain set meaning.
394    ///
395    /// # Errors
396    /// Refuses invalid domain count, finite limits, allocation failure or
397    /// interrupted control without publishing a partial projection.
398    pub fn project_domains(
399        &self,
400        domains: &[Domain<'_>],
401        limits: Limits,
402        cancellation: &Cancellation,
403    ) -> Result<Projection<'_, 'owner, 'source>, Failure> {
404        self.project_inner(domains.iter().copied(), limits, cancellation)
405    }

407    fn project_inner<'domain>(
408        &self,
409        domains: impl ExactSizeIterator<Item = Domain<'domain>>,
410        limits: Limits,
411        cancellation: &Cancellation,
412    ) -> Result<Projection<'_, 'owner, 'source>, Failure> {
413        let external = self.retained_inputs()?;
414        let scratch_header = size_of::<Vec<u32>>();
415        let mut work = Work::new(
416            limits,
417            control(cancellation),
418            external,
419            size_of::<Projection<'_, '_, '_>>() + scratch_header,
420        )
421        .map_err(unmetered)?;
422        let result = (|| {
423            self.check_domains(domains.len(), &work)?;
424            let mut supported = work.reserve(self.values.len())?;
425            work.tick(self.values.len())?;
426            supported.resize(self.values.len(), false);
427            let rows = self.restrict_rows(domains, Some(&mut supported), &mut work)?;
428            for (entry, allowed) in supported.iter_mut().enumerate() {
429                work.tick(1)?;
430                if *allowed {
431                    *allowed = intersects(&rows, self.support(entry), &mut work)?;
432                }
433            }
434            work.release_bytes(scratch_header);
435            Ok(Projection {
436                table: self,
437                rows,
438                supported,
439                statistics: work.statistics(external),
440            })
441        })();
442        result.map_err(|cause| unmetered(work.failure(cause)))
443    }

445    fn retained_inputs(&self) -> Result<usize, Failure> {
446        self.relation
447            .storage()
448            .retained_bytes
449            .checked_add(self.statistics.retained_bytes)
450            .ok_or(Failure {
451                cause: Cause::Overflow,
452                work: 0,
453                peak_bytes: usize::MAX,
454            })
455    }

457    fn check_domains<E>(
458        &self,
459        count: usize,
460        work: &Work<impl FnMut(usize) -> Result<(), E>>,
461    ) -> Result<(), MeteredCause<E>> {
462        work.entries(self.values.len())?;
463        if count == self.variable_count() {
464            Ok(())
465        } else {
466            Err(Cause::Domains.into())
467        }
468    }

470    fn prepare_variable<E>(
471        &mut self,
472        column: usize,
473        work: &mut Work<impl FnMut(usize) -> Result<(), E>>,
474    ) -> Result<(), MeteredCause<E>> {
475        let mut rows = work.reserve(self.relation.row_count())?;
476        for row in 0..self.relation.row_count() {
477            work.tick(1)?;
478            if self.coherent[row / WORD_BITS] & (1 << (row % WORD_BITS)) != 0 {
479                let value = self
480                    .relation
481                    .row(row)
482                    .and_then(|row| row.value(column))
483                    .ok_or(Cause::Relation)?;
484                rows.push(RowValue { row, value });
485            }
486        }
487        sort(&mut rows, work)?;
488        let start = self.values.len();
489        let mut previous: Option<TermRef<'_>> = None;
490        for row in &rows {
491            let distinct = match previous {
492                Some(value) => !value.equals_ref_with(row.value, || work.tick(1))?,
493                None => true,
494            };
495            if distinct {
496                let entries = self.values.len().checked_add(1).ok_or(Cause::Overflow)?;
497                work.entries(entries)?;
498                work.grow(&mut self.values, 1)?;
499                work.tick(1)?;
500                self.values.push(row.value);
501                work.grow(&mut self.supports, self.coherent.len())?;
502                for _ in 0..self.coherent.len() {
503                    work.tick(1)?;
504                    self.supports.push(0);
505                }
506            }
507            work.tick(1)?;
508            let entry = self.values.len() - 1;
509            self.supports[entry * self.coherent.len() + row.row / WORD_BITS] |=
510                1 << (row.row % WORD_BITS);
511            previous = Some(row.value);
512        }
513        work.tick(1)?;
514        self.variables.push(start..self.values.len());
515        work.release(rows);
516        Ok(())
517    }
518
519    fn support(&self, entry: usize) -> &[u32] {
520        let start = entry * self.coherent.len();
521        &self.supports[start..start + self.coherent.len()]
522    }
523
524    fn lookup<E>(
525        &self,
526        variable: usize,
527        value: TermRef<'_>,
528        work: &mut Work<impl FnMut(usize) -> Result<(), E>>,
529    ) -> Result<Option<usize>, MeteredCause<E>> {
530        let mut range = self.variables[variable].clone();
531        while range.start < range.end {
532            let middle = range.start + (range.end - range.start) / 2;
533            match work.compare(self.values[middle], value)? {
534                std::cmp::Ordering::Less => range.start = middle + 1,
535                std::cmp::Ordering::Greater => range.end = middle,
536                std::cmp::Ordering::Equal => return Ok(Some(middle)),
537            }
538        }
539        Ok(None)
540    }
541}

582fn coherent_row<E>(
583    relation: &Relation<'_>,
584    scope: &[usize],
585    first_columns: &[usize],
586    row: usize,
587    work: &mut Work<impl FnMut(usize) -> Result<(), E>>,
588) -> Result<bool, MeteredCause<E>> {
589    for (column, &variable) in scope.iter().enumerate() {
590        work.tick(1)?;
591        let previous = first_columns[variable];
592        if previous != column {
593            work.tick(1)?;
594            if relation.column(column).ok_or(Cause::Relation)?[row]
595                != relation.column(previous).ok_or(Cause::Relation)?[row]
596            {
597                return Ok(false);
598            }
599        }
600    }
601    Ok(true)
602}

604fn intersects<E>(
605    left: &[u32],
606    right: &[u32],
607    work: &mut Work<impl FnMut(usize) -> Result<(), E>>,
608) -> Result<bool, MeteredCause<E>> {
609    for (&left, &right) in left.iter().zip(right) {
610        work.tick(1)?;
611        if left & right != 0 {
612            return Ok(true);
613        }
614    }
615    Ok(false)
616}

618fn sort<E>(
619    rows: &mut Vec<RowValue<'_>>,
620    work: &mut Work<impl FnMut(usize) -> Result<(), E>>,
621) -> Result<(), MeteredCause<E>> {
622    let mut scratch = work.reserve(rows.len())?;
623    for &row in rows.iter() {
624        work.tick(1)?;
625        scratch.push(row);
626    }
627    let mut width = 1;
628    while width < rows.len() {
629        let mut start = 0;
630        while start < rows.len() {
631            let middle = start.saturating_add(width).min(rows.len());
632            let end = middle.saturating_add(width).min(rows.len());
633            let mut left = start;
634            let mut right = middle;
635            for output in &mut scratch[start..end] {
636                work.tick(1)?;
637                let take_left = left < middle
638                    && (right == end
639                        || work.compare(rows[left].value, rows[right].value)?
640                            != std::cmp::Ordering::Greater);
641                *output = rows[if take_left {
642                    let index = left;
643                    left += 1;
644                    index
645                } else {
646                    let index = right;
647                    right += 1;
648                    index
649                }];
650            }
651            start = end;
652        }
653        std::mem::swap(rows, &mut scratch);
654        width = width.saturating_mul(2);
655    }
656    work.release(scratch);
657    Ok(())
658}
