3/// Limits on retained abstract data and inspected source structure.
4#[derive(Clone, Copy, Debug)]
5pub struct Limits {
6    /// Source, transfer, and value-merge steps.
7    pub max_work: u64,
8    /// Distinct signed predicate signatures, including zero-arity predicates.
9    pub max_predicates: usize,
10    /// Distinct predicate argument positions.
11    pub max_positions: usize,
12    /// Transfer records, producer references, and dependency-column references.
13    pub max_links: usize,
14    /// Retained values in one argument; overflow widens only that argument.
15    pub max_values_per_argument: usize,
16    /// Retained finite value references across all arguments.
17    pub max_value_entries: usize,
18    /// Complete passes, including the final unchanged pass.
19    pub max_rounds: u64,
20    /// Cumulative name/string bytes inspected before comparison or copying.
21    pub max_inspected_bytes: u64,
22    /// Nodes in one finite symbol; larger values widen their output argument.
23    pub max_symbol_nodes: usize,
24    /// Nested symbol depth, with an atomic symbol at depth one.
25    pub max_symbol_depth: usize,
26    /// Name/string payload in one finite symbol.
27    pub max_symbol_bytes: u64,
28}
29impl Default for Limits {
30    fn default() -> Self {
31        Self {
32            max_work: 1_000_000,
33            max_predicates: 4_096,
34            max_positions: 16_384,

36            max_values_per_argument: 256,

39            max_inspected_bytes: 8_388_608,
40            max_symbol_nodes: 256,
41            max_symbol_depth: 64,
42            max_symbol_bytes: 16_384,
43        }
44    }
45}

47/// A global ceiling whose exhaustion prevents publishing any finite domains.
48#[derive(Clone, Copy, Debug, PartialEq, Eq)]
49pub enum Resource {
50    /// Source/transfer/merge steps.
51    Work,
52    /// Signed predicate signatures.
53    Predicates,
54    /// Argument positions.
55    Positions,
56    /// Retained transfer and provenance/dependency references.
57    Links,
58    /// Retained finite value references.
59    ValueEntries,
60    /// Complete fixed-point passes.
61    Rounds,
62    /// Inspected name/string bytes.
63    InspectedBytes,
64}
65
66/// The exact logical ceiling that stopped analysis.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68pub struct Stop {
69    /// Exhausted resource.
70    pub resource: Resource,
71    /// Configured inclusive maximum.
72    pub limit: u128,
73    /// Attempted count; the failed operation is not committed.
74    pub observed: u128,
75}
76impl std::fmt::Display for Stop {
77    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
78        write!(
79            f,
80            "{:?} ceiling {} exceeded by {}",
81            self.resource, self.limit, self.observed
82        )
83    }
84}
85impl std::error::Error for Stop {}

87/// Charged logical counts before completion or an explicit conservative stop.
88/// Reservations precede allocation; a later failure can leave a reservation unused.
89#[derive(Clone, Copy, Default, Debug, PartialEq, Eq)]
90pub struct Statistics {
91    /// Source/transfer/value steps.
92    pub work: u64,
93    /// Registered signed predicate signatures.
94    pub predicates: usize,
95    /// Registered argument positions.
96    pub positions: usize,
97    /// Submitted transfer and producer/dependency references.
98    pub links: usize,
99    /// Finite value references before any global fallback clears results.
100    pub value_entries: usize,
101    /// Started fixed-point passes.
102    pub rounds: u64,
103    /// Inspected name/string payload.
104    pub inspected_bytes: u64,
105    /// Arguments widened to Unknown before a possible global fallback.
106    pub widened: usize,
107}

109pub(crate) fn check(resource: Resource, observed: u128, limit: u128) -> Result<(), Stop> {
110    if observed > limit {
111        Err(Stop {
112            resource,
113            limit,
114            observed,
115        })
116    } else {
117        Ok(())
118    }
