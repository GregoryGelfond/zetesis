1//! Argument domains over neutral producer plans.
2//!
3//! One lowering borrows source symbols; another may submit authenticated catalog
4//! keys. Logical stops publish no bounds. Reader and allocation failures carry
5//! the committed receipt, without a partial analysis. Bounds describe defined
6//! values; they do not establish that arithmetic is total or that a binding exists.
7//! Requested integer certificates separately establish defined arithmetic over
8//! complete-binding domains, without proving a binding or a prefix extension.

24/// Whether every possible value is an integer.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub enum Integrality {
27    /// All values are numbers, including the vacuous empty domain.
28    Integer,
29    /// Nonnumeric values may occur.
30    Mixed,
31}
32/// A bound on only the numeric values in a domain.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum Interval {
35    /// No numeric value occurs.
36    Bottom,
37    /// Inclusive i32 endpoints.
38    Bounded {
39        /// Least permitted numeric value.
40        low: i32,
41        /// Greatest permitted numeric value.
42        high: i32,
43    },
44    /// No numeric bound has been established.
45    Unbounded,
46}
47/// Numeric components of one argument or local slot.
48#[derive(Clone, Copy, Debug, PartialEq, Eq)]
49pub struct Bound {
50    /// Whether nonnumeric values have been excluded.
51    pub integrality: Integrality,
52    /// Upper coverage of numeric values alone.
53    pub interval: Interval,
54}
55/// Completion and a conservative logical stop are distinct.
56#[derive(Clone, Copy, Debug, PartialEq, Eq)]
57pub enum Status {
58    /// Every producer is covered by the completed bounds.
59    FixedPoint,
60    /// All queries are unavailable; no partial certificate survived.
61    Stopped(Stop),
62}

87/// Borrow a compatible owner without importing or copying its term payload.
88/// Implementations charge actual inspected structure and semantic comparison
89/// work through the supplied budget. A refused read is never an unknown value.
90pub trait Values {
91    /// Authentic source key; neither Clone nor Ord is required.
92    type Key;
93    /// The owner's typed read/control failure.
94    type Error;
95    /// Classify one key and check the configured finite-symbol ceilings.
96    /// # Errors
97    /// Preserve owner/control, allocation and logical-limit failures.
98    fn classify(
99        &self,
100        key: &Self::Key,
101        limits: &Limits,
102        budget: &Budget,
103    ) -> Result<Classification, Error<Self::Error>>;
104    /// Compare keys by a consistent total order whose equality is semantic equality.
105    /// The engine does not interpret this order as the ASP relational order.
106    /// # Errors
107    /// Preserve owner/control, allocation and logical-limit failures.
108    fn compare(
109        &self,
110        left: &Self::Key,
111        right: &Self::Key,
112        budget: &Budget,
113    ) -> Result<Ordering, Error<Self::Error>>;
114}
115/// Shared logical meter for lowering, key inspection and abstract transfer.
116/// Finite-set entries are live reservations, released with their owning sets.
117pub struct Budget {
118    pub(super) limits: Limits,
119    pub(super) statistics: Cell<Statistics>,
120}
121impl Budget {
122    /// Admit work before performing it; a refused charge is not committed.
123    /// # Errors
124    /// Reports the inclusive work ceiling.
125    pub fn work(&self, amount: u64) -> Result<(), Stop> {
126        let mut statistics = self.statistics.get();
127        crate::limits::check(
128            crate::Resource::Work,
129            u128::from(statistics.work) + u128::from(amount),
130            u128::from(self.limits.max_work),
131        )?;
132        statistics.work += amount;
133        self.statistics.set(statistics);
134        Ok(())
135    }
136    /// Admit inspected name/string bytes before examining their payload.
137    /// # Errors
138    /// Reports the inclusive inspected-byte ceiling.
139    pub fn bytes(&self, amount: u64) -> Result<(), Stop> {
140        let mut statistics = self.statistics.get();
141        crate::limits::check(
142            crate::Resource::InspectedBytes,
143            u128::from(statistics.inspected_bytes) + u128::from(amount),
144            u128::from(self.limits.max_inspected_bytes),
145        )?;
146        statistics.inspected_bytes += amount;
147        self.statistics.set(statistics);
148        Ok(())
149    }
150    /// The committed logical receipt at this point.
151    #[must_use]
152    pub fn statistics(&self) -> Statistics {
153        self.statistics.get()
154    }
155}
156/// A refused operation; logical limits alone become a stopped analysis.
157#[derive(Debug)]
158pub enum Error<E> {
159    /// No part of the analysis may be used after this global stop.
160    Limit(Stop),
161    /// Host metadata allocation failed.
162    Allocation(TryReserveError),
163    /// The key owner refused a read or observed control.
164    Read(E),
165    /// A coordinate, interval or dependency in the neutral plan is invalid.
166    InvalidInput,
167}
168impl<E> From<Stop> for Error<E> {
169    fn from(stop: Stop) -> Self {
170        Self::Limit(stop)
171    }
172}
173impl<E> From<TryReserveError> for Error<E> {
174    fn from(error: TryReserveError) -> Self {
175        Self::Allocation(error)
176    }
177}

188/// Nodes reference only earlier expressions; the plan is acyclic.
189#[derive(Clone, Copy, Debug, PartialEq, Eq)]
190pub enum Expression {
191    /// An authenticated source constant.
192    Constant(ConstantId),
193    /// A local slot in the current rule.
194    Slot(usize),
195    /// Unary arithmetic on one expression.
196    Unary(UnaryOp, ExpressionId),
197    /// Absolute value; retained explicitly even when its static bound is unknown.
198    Absolute(ExpressionId),
199    /// Binary arithmetic preserving operand order.
200    Binary(BinaryOp, ExpressionId, ExpressionId),
201    /// A local transfer without a more precise bound. This is a summary, not
202    /// a denotational identity: even reusing its coordinate cannot establish
203    /// correlation or preserve an expression-specific guard refinement.
204    Unknown {
205        /// Every defined result is an integer.
206        integer: bool,
207        /// Why the finite component lost precision.
208        reason: Widening,
209    },
210}
211/// Complete-binding constraints; comparisons do not belong here.
212#[derive(Clone, Copy, Debug)]
213pub enum Binding {
214    /// One ordinary positive occurrence; repeated occurrences meet.
215    Positive {
216        /// Local variable slot.
217        slot: usize,
218        /// Predicate argument supplying the occurrence.
219        argument: ArgumentId,
220    },
221    /// A copy, constant or arithmetic assignment.
222    Assign {
223        /// Assigned slot.
224        slot: usize,
225        /// Assigned expression.
226        value: ExpressionId,
227    },
228    /// A generating interval; membership tests instead use `Guard::Membership`.
229    Range {
230        /// Generated slot.
231        slot: usize,
232        /// Lower endpoint expression.
233        low: ExpressionId,
234        /// Upper endpoint expression.
235        high: ExpressionId,
236    },
237    /// A conservative aggregate assignment, without a claimed tuple envelope.
238    Aggregate {
239        /// Assigned slot.
240        slot: usize,
241        /// Aggregate function.
242        function: AggregateFunction,
243    },
244}
245/// Ordered conditions refining only the values that can produce heads.
246#[derive(Clone, Copy, Debug)]
247pub enum Guard {
248    /// A comparison; neither operand establishes integrality by comparison alone.
249    Compare {
250        /// Left expression.
251        left: ExpressionId,
252        /// Source relation.
253        relation: Relation,
254        /// Right expression.
255        right: ExpressionId,
256    },
257    /// An already-bound slot tested against an interval.
258    Membership {
259        /// Bound target slot.
260        slot: usize,
261        /// Lower endpoint expression.
262        low: ExpressionId,
263        /// Upper endpoint expression.
264        high: ExpressionId,
265    },
266}

275/// One independent variable scope and its ordered producer transfer.
276#[derive(Debug, Default)]
277pub struct Rule {
278    /// Number of slots; every referenced slot must lie in this extent.
279    pub slots: usize,
280    /// Positive occurrences followed by assignments in dependency order.
281    pub bindings: Vec<Binding>,
282    /// Source-order guard refinements, applied once.
283    pub guards: Vec<Guard>,
284    /// Output argument occurrences; duplicates retain their provenance.
285    pub heads: Vec<Head>,
286    /// Expressions to certify against complete-binding domains after convergence.
287    /// Request order is retained, including duplicates. An empty list adds no
288    /// certificate work. Requests neither refine bounds nor change head transfer.
289    /// Covered arithmetic is negation, addition/subtraction, multiplication by a
290    /// literal, division by a literal and remainder by a bounded nonzero interval.
291    /// Every operand and intermediate must have a defined integer range.
292    pub integer_queries: Vec<ExpressionId>,
293}

324/// Defined integer arithmetic over one completed rule's binding domains.
325///
326/// Every operand and intermediate in the supported grammar is checked. This
327/// establishes neither a complete binding's existence nor a prefix guarantee,
328/// and it does not certify arithmetic in other expressions or in the bindings
329/// that supply a slot. The token borrows the exact rule result; callers retain
330/// responsibility for mapping that submitted scope to their original owner.
331#[derive(Clone, Copy, Debug)]
332pub struct IntegerCertificate<'rule> {
333    scope: &'rule RuleBounds,
334    query: usize,
335    range: crate::Range,
336}
337impl IntegerCertificate<'_> {
338    /// The requested expression coordinate, meaningful only in this analysis.
339    #[must_use]
340    pub fn expression(self) -> ExpressionId {
341        self.scope.integers[self.query].expression
342    }
343    /// Every complete binding has a defined integer result within this interval.
344    #[must_use]
345    pub fn range(self) -> crate::Range {
346        self.range
347    }
348}
349impl RuleBounds {
350    /// Number of original requests, including conservative declines.
351    #[must_use]
352    pub fn integer_query_count(&self) -> usize {
353        self.integers.len()
354    }
355    /// Certify one request by its position in this rule's submitted list.
356    ///
357    /// None means an absent request or a conservative decline, including empty,
358    /// mixed or unbounded operands, unsupported arithmetic and a possibly
359    /// undefined intermediate. It does not establish an impossible binding.
360    /// Guard-refined produced bounds never justify this result.
361    #[must_use]
362    pub fn integer(&self, query: usize) -> Option<IntegerCertificate<'_>> {
363        let range = self.integers.get(query)?.range?;
364        Some(IntegerCertificate {
365            scope: self,
366            query,
367            range,
368        })
369    }
370    /// Complete-binding coverage, without guard refinements or prefix guarantees.
371    #[must_use]
372    pub fn domains(&self) -> &[Bound] {
373        &self.domains
374    }
375    /// Bounds after one ordered guard pass, suitable only for head transfer.
376    #[must_use]
377    pub fn produced(&self) -> &[Bound] {
378        &self.produced
379    }
380}
381impl<K, C: Copy> Analysis<K, C> {
382    /// Whether all published bounds are complete or all were cleared.
383    #[must_use]
384    pub fn status(&self) -> Status {
385        self.status
386    }
387    /// Completed charges, including failed attempts' committed prefix.
388    #[must_use]
389    pub fn statistics(&self) -> Statistics {
390        self.statistics
391    }
392    /// Stop context; absent on a completed analysis.
393    #[must_use]
394    pub fn context(&self) -> Option<C> {
395        self.context
396    }
397    /// Numeric components of a valid completed argument.
398    #[must_use]
399    pub fn argument(&self, argument: ArgumentId) -> Option<Bound> {
400        self.arguments.get(argument.0).map(|value| value.bound)
401    }
402    /// Arguments in registration order.
403    pub fn arguments(&self) -> impl ExactSizeIterator<Item = (ArgumentId, Bound)> {
404        self.arguments
405            .iter()
406            .enumerate()
407            .map(|(index, value)| (ArgumentId(index), value.bound))
408    }
409    /// Finite values, or None for Unknown/unavailable. Order is registration
410    /// identity, not semantic term order; callers must not infer term order.
411    /// # Panics
412    /// Iteration panics if an internal finite entry names an unretained key.
413    /// Completed analyses maintain this invariant.
414    #[must_use]
415    pub fn values(&self, argument: ArgumentId) -> Option<impl ExactSizeIterator<Item = &K>> {
416        self.arguments
417            .get(argument.0)?
418            .values
419            .as_ref()
420            .map(|values| {
421                values.iter().map(|&index| {
422                    self.constants[index]
423                        .key
424                        .as_ref()
425                        .expect("only retained keys enter finite sets")
426                })
427            })
428    }
429    /// Finite entries in increasing analysis-local registration order. These
430    /// coordinates may compare sets from this same analysis without reading
431    /// payloads. They do not authenticate a different analysis or key owner.
432    /// # Panics
433    /// Iteration panics if an internal finite entry names an unretained key.
434    /// Completed analyses maintain this invariant.
435    #[must_use]
436    pub fn entries(
437        &self,
438        argument: ArgumentId,
439    ) -> Option<impl ExactSizeIterator<Item = (ConstantId, &K)>> {
440        self.arguments
441            .get(argument.0)?
442            .values
443            .as_ref()
444            .map(|values| {
445                values.iter().map(|&index| {
446                    (
447                        ConstantId(index),
448                        self.constants[index]
449                            .key
450                            .as_ref()
451                            .expect("only retained keys enter finite sets"),
452                    )
453                })
454            })
455    }
456    /// Direct original contexts for every submitted output occurrence.
457    #[must_use]
458    pub fn producers(&self, argument: ArgumentId) -> Option<&[C]> {
459        self.arguments
460            .get(argument.0)
461            .map(|value| value.producers.as_slice())
462    }
463    /// The first recorded loss of finite precision for this argument.
464    #[must_use]
465    pub fn widening(&self, argument: ArgumentId) -> Option<Widening> {
466        self.arguments
467            .get(argument.0)
468            .and_then(|value| value.widening)
469    }
470    /// Local bounds recomputed from the completed argument result.
471    #[must_use]
472    pub fn rule(&self, rule: RuleId) -> Option<&RuleBounds> {
473        self.rules.get(rule.0)
474    }
475}
476/// Lower and analyze one complete producer input with one shared logical meter.
477/// A logical stop returns a cleared Analysis; other failures return no Analysis.
478/// The reader and callback are synchronous and are not retained in the result.
479/// # Errors
480/// Returns owner/read, allocation or invalid-plan failures with their receipt.
481pub fn analyze<R: Values, C: Copy>(
482    reader: &R,
483    limits: Limits,
484    lower: impl FnOnce(&mut Input<'_, R, C>) -> Result<(), Error<R::Error>>,
485) -> Result<Analysis<R::Key, C>, Failure<R::Error, C>> {
486    let budget = Budget {
487        limits,
488        statistics: Cell::new(Statistics::default()),
489    };
490    let mut input = Input::new(reader, &budget);
491    let attempt = lower(&mut input).and_then(|()| engine::run(&mut input));
492    match attempt {
493        Ok(engine::Completed { arguments, rules }) => input.finish(arguments, rules),
494        Err(Error::Limit(stop)) => Ok(Analysis {
495            constants: Vec::new(),
496            arguments: Vec::new(),
497            rules: Vec::new(),
498            context: input.context,
499            statistics: budget.statistics(),
500            status: Status::Stopped(stop),
501        }),
502        Err(error) => Err(Failure {
503            error,
504            statistics: budget.statistics(),
505            context: input.context,
506        }),
507    }
508}
