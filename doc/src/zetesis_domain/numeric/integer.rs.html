1//! Total integer expressions over complete-binding domains.
2//!
3//! This is separate from defined-value transfer: a narrow result of that
4//! transfer does not imply that each intermediate is defined. The postorder
5//! stack checks every child before its parent, including both sides of e-e.
6//! Nodes refer only to earlier nodes, so traversal terminates; shared children
7//! may be revisited, and every visit is charged. Scratch is bounded by the
8//! admitted expression depth and work, without another expression graph.
9use super::{
10    BinaryOp, Bound, Error, Expression, ExpressionId, Input, Integrality, Interval, UnaryOp, Values,
11};
12use crate::Range;

14/// One requested expression's certificate or conservative decline.
15#[derive(Debug)]
16pub(super) struct Evaluation {
17    pub(super) expression: ExpressionId,
18    pub(super) range: Option<Range>,
19}

21#[derive(Clone, Copy)]
22struct Value {
23    range: Option<Range>,
24    literal: bool,
25}
26impl Value {
27    fn computed(range: Option<Range>) -> Self {
28        Self {
29            range,
30            literal: false,
31        }
32    }
33}

35fn slot(bound: Bound) -> Option<Range> {
36    if bound.integrality != Integrality::Integer {
37        return None;
38    }
39    match bound.interval {
40        Interval::Bounded { low, high } => Range::new(low, high),
41        Interval::Bottom | Interval::Unbounded => None,
42    }
43}
44
45fn binary(operator: BinaryOp, left: Value, right: Value) -> Option<Range> {
46    // Literal factors/divisors are syntactic constants, not merely slots whose
47    // current bound happens to be a singleton. Mod uses the interval-divisor
48    // law and Range's zero/MIN/-1 guards; variable Div remains outside scope.
49    let covered = match operator {
50        BinaryOp::Add | BinaryOp::Sub | BinaryOp::Mod => true,
51        BinaryOp::Mul => left.literal || right.literal,
52        BinaryOp::Div => right.literal,
53        BinaryOp::Pow | BinaryOp::BitAnd | BinaryOp::BitOr | BinaryOp::BitXor => false,
54    };
55    if !covered {
56        return None;
57    }
58    left.range?.binary(operator, right.range?)
59}

61fn expression<R: Values, C: Copy>(
62    input: &Input<'_, R, C>,
63    root: ExpressionId,
64    domains: &[Bound],
65) -> Result<Option<Range>, Error<R::Error>> {
66    let mut pending = Vec::new();
67    let mut values: Vec<Value> = Vec::new();
68    pending.try_reserve(1)?;
69    pending.push((root, false));
70    while let Some((id, ready)) = pending.pop() {
71        input.budget.work(1)?;
72        let node = *input.expressions.get(id.0).ok_or(Error::InvalidInput)?;
73        if !ready {
74            match node {
75                Expression::Unary(_, child) | Expression::Absolute(child) => {
76                    pending.try_reserve(2)?;
77                    pending.extend([(id, true), (child, false)]);
78                    continue;
79                }
80                Expression::Binary(_, left, right) => {
81                    pending.try_reserve(3)?;
82                    pending.extend([(id, true), (right, false), (left, false)]);
83                    continue;
84                }
85                Expression::Constant(_) | Expression::Slot(_) | Expression::Unknown { .. } => {}
86            }
87        }
88        let value = match node {
89            Expression::Constant(id) => Value {
90                range: input
91                    .constants
92                    .get(id.0)
93                    .ok_or(Error::InvalidInput)?
94                    .number
95                    .map(Range::point),
96                literal: true,
97            },
98            Expression::Slot(id) => {
99                Value::computed(slot(*domains.get(id).ok_or(Error::InvalidInput)?))
100            }
101            Expression::Unknown { .. } => Value::computed(None),
102            Expression::Unary(operator, _) => {
103                let child = values.pop().ok_or(Error::InvalidInput)?;
104                Value::computed(if operator == UnaryOp::Negate {
105                    child.range.and_then(|range| range.unary(operator))
106                } else {
107                    None
108                })
109            }
110            Expression::Absolute(_) => {
111                values.pop().ok_or(Error::InvalidInput)?;
112                Value::computed(None)
113            }
114            Expression::Binary(operator, _, _) => {
115                let right = values.pop().ok_or(Error::InvalidInput)?;
116                let left = values.pop().ok_or(Error::InvalidInput)?;
117                Value::computed(binary(operator, left, right))
118            }
119        };
120        values.try_reserve(1)?;
121        values.push(value);
122    }
123    Ok(values.pop().ok_or(Error::InvalidInput)?.range)
124}

126pub(super) fn queries<R: Values, C: Copy>(
127    input: &Input<'_, R, C>,
128    roots: &[ExpressionId],
129    domains: &[Bound],
130) -> Result<Vec<Evaluation>, Error<R::Error>> {
131    // Requests were admitted with their producer. Result cells are separate
132    // retained metadata, reserved before allocation even for declined queries.
133    input.links(roots.len())?;
134    let mut results = Vec::new();
135    results.try_reserve(roots.len())?;
136    for &root in roots {
137        input.budget.work(1)?;
138        let range = expression(input, root, domains)?;
139        input.budget.work(1)?;
140        results.push(Evaluation {
141            expression: root,
142            range,
143        });
144    }
145    Ok(results)
146}
