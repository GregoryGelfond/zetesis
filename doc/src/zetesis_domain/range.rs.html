1//! Conservative intervals for checked scalar arithmetic.
2//!
3//! Endpoints and every admitted intermediate lie within i32. Transfers are
4//! calculated in i64 and decline when they cannot establish a defined i32
5//! result for every pair of operands. Callers establish the numeric source
6//! domains independently; this operation supplies neither bindings nor truth.
7
8use themelios_program::term::{BinaryOp, UnaryOp};
9
10/// A nonempty interval of language integers. A successful transfer encloses
11/// every checked result over its inputs; a declined transfer makes no claim.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct Range {
14    low: i32,
15    high: i32,
16}
17
18impl Range {
19    /// Construct inclusive bounds, declining a descending interval.
20    #[must_use]
21    pub fn new(low: i32, high: i32) -> Option<Self> {
22        Self::checked(i64::from(low), i64::from(high))
23    }
24
25    /// Inclusive endpoints, each within the language's integer representation.
26    #[must_use]
27    pub fn bounds(self) -> (i32, i32) {
28        (self.low, self.high)
29    }
30
31    /// The interval containing exactly one integer.
32    #[must_use]
33    pub fn point(value: i32) -> Self {
34        Self {
35            low: value,
36            high: value,
37        }
38    }
39
40    fn checked(low: i64, high: i64) -> Option<Self> {
41        if low > high {
42            return None;
43        }
44        Some(Self {
45            low: i32::try_from(low).ok()?,
46            high: i32::try_from(high).ok()?,
47        })
48    }
49
50    /// Enlarge a numeric bound by one admitted source value.
51    pub fn include(&mut self, value: i32) {
52        self.low = self.low.min(value);
53        self.high = self.high.max(value);
54    }
55
56    /// Checked positive bounds, without changing scalar arithmetic semantics.
57    #[must_use]
58    pub fn positive_bounds(self) -> Option<(i32, i32)> {
59        (self.low > 0).then_some(self.bounds())
60    }
61
62    /// Enclose absolute values, declining when `i32::MIN` is possible.
63    #[must_use]
64    pub fn absolute(self) -> Option<Self> {
65        let (low, high) = (i64::from(self.low), i64::from(self.high));
66        let upper = low.abs().max(high.abs());
67        let lower = if low <= 0 && high >= 0 {
68            0
69        } else {
70            low.abs().min(high.abs())
71        };
72        Self::checked(lower, upper)
73    }
74
75    /// Enclose a unary operation when every input has a defined i32 result.
76    #[must_use]
77    pub fn unary(self, operator: UnaryOp) -> Option<Self> {
78        let (low, high) = (i64::from(self.low), i64::from(self.high));
79        match operator {
80            UnaryOp::Negate => Self::checked(-high, -low),
81            UnaryOp::BitwiseNot => Self::checked(!high, !low),
82        }
83    }
84
85    /// Enclose a binary operation over all pairs in the two intervals.
86    /// Decline undefined, overflowing or unsupported interval transfers.
87    #[must_use]
88    pub fn binary(self, operator: BinaryOp, right: Self) -> Option<Self> {
89        let (low, high) = (i64::from(self.low), i64::from(self.high));
90        let (right_low, right_high) = (i64::from(right.low), i64::from(right.high));
91        match operator {
92            BinaryOp::Add => Self::checked(low + right_low, high + right_high),
93            BinaryOp::Sub => Self::checked(low - right_high, high - right_low),
94            BinaryOp::Mul => self.corners(right, |left, right| Some(left * right)),
95            BinaryOp::Div | BinaryOp::Mod
96                if (right_low <= 0 && right_high >= 0)
97                    || (low == i64::from(i32::MIN) && right_low <= -1 && right_high >= -1) =>
98            {
99                None
100            }
101            BinaryOp::Div => self.corners(right, |left, right| Some(left / right)),
102            BinaryOp::Mod => {
103                let magnitude = right_low.abs().max(right_high.abs()) - 1;
104                Self::checked(low.min(0).max(-magnitude), high.max(0).min(magnitude))
105            }
106            BinaryOp::Pow if right_low == right_high && right_low >= 0 => {
107                let exponent = u32::try_from(right_low).ok()?;
108                let bound = self.corners(right, |left, _| left.checked_pow(exponent))?;
109                Some(Self {
110                    low: if exponent > 0 && low <= 0 && high >= 0 {
111                        bound.low.min(0)
112                    } else {
113                        bound.low
114                    },
115                    high: bound.high,
116                })
117            }
118            BinaryOp::Pow => None,
119            BinaryOp::BitAnd | BinaryOp::BitOr | BinaryOp::BitXor => {
120                Self::checked(i64::from(i32::MIN), i64::from(i32::MAX))
121            }
122        }
123    }
124
125    fn corners(self, right: Self, operation: impl Fn(i64, i64) -> Option<i64>) -> Option<Self> {
126        let (low, high) = (i64::from(self.low), i64::from(self.high));
127        let (right_low, right_high) = (i64::from(right.low), i64::from(right.high));
128        let corners = [
129            operation(low, right_low)?,
130            operation(low, right_high)?,
131            operation(high, right_low)?,
132            operation(high, right_high)?,
133        ];
134        Self::checked(*corners.iter().min()?, *corners.iter().max()?)
135    }
136}
