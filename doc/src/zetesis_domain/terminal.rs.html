1//! Terminal positive definitions over one exact borrowed program.
2//!
3//! Selected predicates have only flat, normal positive producers and no rule or
4//! constraint consumer anywhere in the input. All their producers are returned together.
5//! Their positive bodies therefore mention only unselected predicates. This is
6//! a structural certificate, not source admission, a grounding transformation,
7//! or a proof that a reconstructed interpretation is an answer set.
8//!
9//! The scan includes ordinary, conditional, choice, disjunctive and aggregate
10//! occurrences. A complementary strong-sign occurrence blocks selection. Show
11//! and diagnostic declarations do not control eligibility; objectives and
12//! projection make the whole result Unknown in this initial profile.
13//!
14//! Limits count inspected nodes, predicate-name/variable/symbol text, registered
15//! signed signatures and argument positions, and submitted occurrence/reference
16//! links. Only those fields of [`Limits`], plus its per-symbol bounds, apply;
17//! finite-domain widths and fixed-point rounds do not. Standard collection
18//! comparisons do not charge repeated text comparisons inside tree lookups;
19//! text accounting measures the logical text submitted to each operation.
20//! The classifier provides no cancellation callback. Collection
21//! allocations are bounded by these logical populations, not by a fallible
22//! physical-byte or RSS contract. Names and values remain borrowed. The scan
23//! does not recurse through input terms; closed `Symbol` traversal uses a bounded
24//! explicit stack. Unsupported source terms disqualify a producer without
25//! evaluating it. Unknown and Stopped discard every selected definition.
26//!
27//! ```
28//! use themelios_program::program::{Atom, Program, Rule};
29//! use themelios_program::symbol::Name;
30//! use zetesis_domain::{Limits, terminal};
31//!
32//! let atom = Atom::new(Name::new("completed").unwrap(), []);
33//! let program = Program::of([Rule::fact(atom)]);
34//! let result = terminal::analyze(&program, Limits::default());
35//! assert_eq!(result.status(), terminal::Status::Complete);
36//! assert!(result.belongs_to(&program));
37//! assert_eq!(result.definitions().len(), 1);
38//! ```

52/// Context outside this classifier's complete semantic-occurrence contract.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum UnknownReason {
55    /// Unresolved or unsupported source context, using the domain crate's reason.
56    Source(crate::UnknownReason),
57    /// Objective dependencies are outside the initial terminal profile.
58    Objectives,
59    /// Explicit projection requires a separate full-answer correspondence.
60    Projection,
61}

63/// Completion does not imply that any eligible terminal definition exists.
64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub enum Status {
66    /// Every semantic occurrence was classified; selection may be empty.
67    Complete,
68    /// Unsupported context; no partial selection is published.
69    Unknown(UnknownReason),
70    /// An inclusive logical ceiling stopped the scan before its next operation.
71    Stopped(Stop),
72}

74/// Selected original carriers, tied to the identical immutable input instance.
75#[derive(Debug)]
76pub struct Analysis<'p> {
77    program: &'p Program,
78    definitions: Vec<&'p WithProvenance<Statement>>,
79    status: Status,
80    context: Option<Context<'p>>,
81    statistics: Statistics,
82}
83
84impl<'p> Analysis<'p> {
85    /// The exact input; no normalization, source replay or copy is performed.
86    #[must_use]
87    pub fn program(&self) -> &'p Program {
88        self.program
89    }
90
91    /// Whether `program` is this input instance, not merely equal content.
92    #[must_use]
93    pub fn belongs_to(&self, program: &Program) -> bool {
94        std::ptr::eq(self.program, program)
95    }
96
97    /// Complete, unsupported and stopped results remain distinct.
98    #[must_use]
99    pub const fn status(&self) -> Status {
100        self.status
101    }
102
103    /// Source context at the first global fallback; absent on completion.
104    #[must_use]
105    pub const fn context(&self) -> Option<Context<'p>> {
106        self.context
107    }
108
109    /// Charged logical populations, including accepted work before a fallback.
110    #[must_use]
111    pub const fn statistics(&self) -> Statistics {
112        self.statistics
113    }
114
115    /// Every selected normal rule carrier, in the input Program's iteration
116    /// order, with its original provenance. Every producer of each selected
117    /// signature occurs here. Upstream-coalesced rules remain one carrier with
118    /// their combined provenance; this does not recreate textual occurrences.
119    /// Empty after Unknown/Stopped. Source/IR correspondence is a caller duty.
120    #[must_use]
121    pub fn definitions(&self) -> &[&'p WithProvenance<Statement>] {
122        &self.definitions
123    }
124}

126/// Classify complete terminal positive definitions without interpreting source.
127///
128/// A selected rule has a positive atomic head, positive ordinary body atoms,
129/// whole-variable or closed `Symbol` arguments, and a positive body binder for
130/// every head variable. Source arithmetic, ranges, pools and constructors that
131/// still need evaluation disqualify that producer. Facts and already closed
132/// compound symbols are allowed. No filename, spelling or shown predicate has
133/// special treatment. The result neither removes rules nor changes budgets.
134#[must_use]
135pub fn analyze(program: &Program, limits: Limits) -> Analysis<'_> {
136    let mut engine = Engine {
137        result: Analysis {
138            program,
139            definitions: Vec::new(),
140            status: Status::Complete,
141            context: None,
142            statistics: Statistics::default(),
143        },
144        limits,
145        signatures: BTreeMap::new(),
146        candidates: Vec::new(),
147    };
148    match engine.scan() {
149        Ok(()) => engine.result.context = None,
150        Err(Failure::Unknown(reason)) => {
151            engine.result.status = Status::Unknown(reason);
152            engine.result.definitions.clear();
153        }
154        Err(Failure::Stopped(stop)) => {
155            engine.result.status = Status::Stopped(stop);
156            engine.result.definitions.clear();
157        }
158    }
159    engine.result
160}

162#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
163struct Signature<'p> {
164    name: &'p str,
165    arity: u32,
166    sign: Sign,
167}
168
169struct Usage {
170    read: bool,
171    qualified: bool,
172}
173
174enum Failure {
175    Unknown(UnknownReason),
176    Stopped(Stop),
177}
178impl From<Stop> for Failure {
179    fn from(stop: Stop) -> Self {
180        Self::Stopped(stop)
181    }
182}
183
184struct Engine<'p> {
185    result: Analysis<'p>,
186    limits: Limits,
187    signatures: BTreeMap<Signature<'p>, Usage>,
188    candidates: Vec<(Signature<'p>, &'p WithProvenance<Statement>)>,
189}
190
191impl<'p> Engine<'p> {
192    fn work(&mut self) -> Result<(), Failure> {
193        let observed = u128::from(self.result.statistics.work) + 1;
194        check(Resource::Work, observed, u128::from(self.limits.max_work))?;
195        self.result.statistics.work += 1;
196        Ok(())
197    }
198
199    fn bytes(&mut self, count: usize) -> Result<(), Failure> {
200        let observed = u128::from(self.result.statistics.inspected_bytes) + count as u128;
201        check(
202            Resource::InspectedBytes,
203            observed,
204            u128::from(self.limits.max_inspected_bytes),
205        )?;
206        self.result.statistics.inspected_bytes =
207            u64::try_from(observed).expect("bounded by u64 limit");
208        Ok(())
209    }
210
211    fn link(&mut self) -> Result<(), Failure> {
212        let observed = self.result.statistics.links as u128 + 1;
213        check(Resource::Links, observed, self.limits.max_links as u128)?;
214        self.result.statistics.links += 1;
215        Ok(())
216    }
217
218    fn signature(&mut self, atom: &'p Atom, arity: usize) -> Result<Signature<'p>, Failure> {
219        self.work()?;
220        self.bytes(atom.name.as_str().len())?;
221        let arity = u32::try_from(arity)
222            .map_err(|_| Failure::Unknown(UnknownReason::Source(crate::UnknownReason::Head)))?;
223        let signature = Signature {
224            name: atom.name.as_str(),
225            arity,
226            sign: atom.sign,
227        };
228        if !self.signatures.contains_key(&signature) {
229            let predicates = self.result.statistics.predicates as u128 + 1;
230            let positions = self.result.statistics.positions as u128 + u128::from(arity);
231            check(
232                Resource::Predicates,
233                predicates,
234                self.limits.max_predicates as u128,
235            )?;
236            check(
237                Resource::Positions,
238                positions,
239                self.limits.max_positions as u128,
240            )?;
241            self.signatures.insert(
242                signature,
243                Usage {
244                    read: false,
245                    qualified: true,
246                },
247            );
248            self.result.statistics.predicates += 1;
249            self.result.statistics.positions =
250                usize::try_from(positions).expect("bounded by usize limit");
251        }
252        Ok(signature)
253    }
254}
