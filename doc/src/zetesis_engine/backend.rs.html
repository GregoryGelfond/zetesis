18/// A reusable CPU backend over canonical ASP programs.
19///
20/// Each successful `lower` replaces the previous program; a refusal preserves it.
21/// Lowering only validates bounded input structure and retains its canonical
22/// value and display policy. Each `solve` owns fresh preparation and search work,
23/// under a single request deadline including consumer pauses.
24///
25/// A live handle borrows this solver, so its program cannot be replaced during
26/// enumeration. Dropping that handle closes its cancellation window. Detached
27/// answer sets and snapshots remain independent values. Named/formal program
28/// parts are refused at lowering; agent-loop updates use ordinary replacements.
29///
30/// Models carry the complete answer set, including hidden atoms. Objectives and
31/// `#project` are excluded from solve preparation; `#show` affects display only.
32/// Native GPU and scored sessions remain available through `zetesis_solve`.
33pub struct Solver {
34    config: Config,
35    cancellation: CancellationSlot,
36    lowered: Option<Lowered>,
37    prepared: Option<Prepared>,
38}
39
40struct Lowered {
41    program: Arc<Program>,
42    show: ShowRule,
43}
44
45impl Solver {
46    /// Create an empty backend with explicit bounded preparation and execution
47    /// policy. This allocates a cancellation slot but starts no workers, timer,
48    /// grounding or search. Configuration ceilings are enforced where used.
49    #[must_use]
50    pub fn new(config: Config) -> Self {
51        Self {
52            config,
53            cancellation: CancellationSlot::default(),
54            lowered: None,
55            prepared: None,
56        }
57    }
58}
59
60impl Default for Solver {
61    fn default() -> Self {
62        Self::new(Config::default())
63    }
64}
65
66impl Drop for Solver {
67    fn drop(&mut self) {
68        self.cancellation.clear();
69    }
70}
71
72impl Backend for Solver {
73    fn capabilities(&self) -> Capabilities {
74        let mut capabilities = Capabilities::default();
75        capabilities.enumeration = true;
76        capabilities.cancellation = true;
77        capabilities.budgets.time = true;
78        capabilities
79    }
80
81    fn lower(&mut self, door: Door<'_>) -> Result<(), Fault> {
82        let program = door.program();
83        validate_program_formula(program, self.config.resources().program_admission_options())
84            .map_err(faults::admission)?;
85        let lowered = Lowered {
86            program: Arc::new(program.clone()),
87            show: ShowRule::of(program.statements().filter_map(
88                |statement| match statement.get() {
89                    Statement::Show(show) => Some(show),
90                    _ => None,
91                },
92            )),
93        };
94        self.lowered = Some(lowered);
95        self.prepared = None;
96        Ok(())
97    }
98
99    fn solve(&mut self, request: &SolveRequest) -> Result<Solved<'_>, Fault> {
100        let started = Instant::now();
101        let deadline = request
102            .time
103            .map(|duration| {
104                started.checked_add(duration).ok_or_else(|| {
105                    Fault::request(
106                        "the requested deadline cannot be represented on this platform",
107                        Presupposition::UnrealisableBudget,
108                    )
109                })
110            })
111            .transpose()?;
112        // Activate before releasing old preparation, copying display policy,
113        // compiling a program, or allocating native workers. A pull during
114        // setup therefore reaches this request, including its preparation.
115        let window = self.cancellation.open(deadline).map_err(window_fault)?;
116        self.prepared = None;
117        let lowered = self.lowered.as_ref().ok_or_else(|| {
118            Fault::request(
119                "lower a program before solving",
120                Presupposition::NeedsRebuild,
121            )
122        })?;
123        let show = lowered.show.clone();
124        let prepared = Prepared::new(
125            Arc::clone(&lowered.program),
126            &self.config,
127            window.cancellation(),
128        );
129        let prepared = match prepared {
130            Ok(prepared) => prepared,
131            Err(error) => {
132                let conclusion = faults::formula(error, &lowered.program)?;
133                return Ok(Solved::running(
134                    Box::new(Run::pending(conclusion, window)),
135                    Scenario::default(),
136                    show,
137                ));
138            }
139        };
140        self.prepared = Some(prepared);
141        let owner = self
142            .prepared
143            .as_ref()
144            .expect("successful preparation was retained");
145        let session = Session::enumerate(
146            owner.input(),
147            self.config.session(),
148            window.cancellation().clone(),
149        );
150        let run = match session {
151            Ok(session) => Run::new(
152                session,
153                &lowered.program,
154                owner.metadata(),
155                window,
156                self.config.output(),
157            ),
158            Err(error) => Run::pending(faults::session(error, &lowered.program)?, window),
159        };
160        Ok(Solved::running(Box::new(run), Scenario::default(), show))
161    }
162
163    fn interrupt(&self) -> Option<Box<dyn Cancel>> {
164        Some(Box::new(Interrupt(self.cancellation.clone())))
165    }
166}

168struct Interrupt(CancellationSlot);
169
170impl Cancel for Interrupt {
171    fn cancel(&self) {
172        self.0.cancel();
173    }
174}
175
176fn window_fault(error: CancellationSlotError) -> Fault {
177    Fault::resource(error.to_string()).caused_by(error)
