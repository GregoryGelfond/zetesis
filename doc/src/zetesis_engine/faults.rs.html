1//! Native run outcomes for the closed CPU, unscored, unprojected adapter profile.
2//!
3//! Only request-owned cancellation/deadline tokens may enter this bridge. Native
4//! resource ceilings are faults, never successful truncation. Source refusals
5//! resolve the actual original statement; there is no fabricated part/statement.
6
7use themelios_program::{Program, program::Statement, provenance::WithProvenance};
8use themelios_solve::{contract::Fault, outcome::Conclusion};
9use zetesis_cpu::Stop;
10use zetesis_solve::{Interruption, SearchState, SemanticOutcome, SolveError, SolveFailure};
11use zetesis_themelios::{FormulaFailure, ProgramSite, ProgramSubject, observation};
12
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14enum Class {
15    Control(Conclusion),
16    Resource,
17    Program,
18    Engine,
19    Adapter,
20}
21
22/// This assumes `Deadline` came from the acknowledged request, not an internal
23/// timeout configured independently of that request.
24pub(crate) fn control(stop: Stop) -> Option<Conclusion> {
25    match stop {
26        Stop::Cancelled => Some(Conclusion::Interrupted),
27        Stop::Deadline => Some(Conclusion::Budget),
28        _ => None,
29    }
30}
31
32fn stop_class(stop: Stop) -> Class {
33    match stop {
34        Stop::Cancelled => Class::Control(Conclusion::Interrupted),
35        Stop::Deadline => Class::Control(Conclusion::Budget),
36        Stop::WorkLimit
37        | Stop::RoundLimit
38        | Stop::StorageLimit
39        | Stop::DerivedAtomLimit
40        | Stop::CandidateLimit
41        | Stop::CarrierLimit
42        | Stop::Allocation => Class::Resource,
43        Stop::WrongProgram | Stop::InvalidProgram => Class::Engine,
44    }
45}
46
47fn fault(
48    class: Class,
49    message: String,
50    subject: Option<ProgramSubject<'_>>,
51) -> Result<Conclusion, Fault> {
52    match class {
53        Class::Control(conclusion) => Ok(conclusion),
54        Class::Resource => Err(Fault::resource(message)),
55        Class::Engine => Err(Fault::engine(message)),
56        Class::Adapter => Err(Fault::adapter_bug(message)),
57        Class::Program => Err(match subject {
58            Some(ProgramSubject::Statement(statement)) => Fault::program(message, statement),
59            Some(ProgramSubject::Part(part)) => Fault::program_part(message, part),
60            Some(ProgramSubject::Program) | None => Fault::adapter_bug(format!(
61                "native statement refusal omitted its original subject: {message}"
62            )),
63        }),
64    }
65}

67fn statement(program: &Program, site: Option<ProgramSite>) -> Option<&WithProvenance<Statement>> {
68    program.statements().nth(site?.statement_id()?.index())
69}

71pub(crate) fn interruption(reason: Interruption) -> Result<Conclusion, Fault> {
72    fault(interruption_class(reason), reason.to_string(), None)
73        .map_err(|fault| fault.caused_by(reason))
74}
75
76fn interruption_class(reason: Interruption) -> Class {
77    match reason {
78        Interruption::Preparation(stop)
79        | Interruption::Oracle(stop)
80        | Interruption::Constraint(stop)
81        | Interruption::Reconstruction(stop) => stop_class(stop),
82        Interruption::Countermodel(error) => sat_class(error),
83        Interruption::ModelConstruction(zetesis_solve::ModelConstructionStop::Control(stop)) => {
84            stop_class(stop)
85        }
86        Interruption::ModelConstruction(
87            zetesis_solve::ModelConstructionStop::Work { .. }
88            | zetesis_solve::ModelConstructionStop::Bytes { .. },
89        ) => Class::Resource,
90        // The adapter configures AnswerSets/All, without scoring or incumbents.
91        Interruption::Objective(_)
92        | Interruption::PreparedObjective(_)
93        | Interruption::Incumbent(_) => Class::Adapter,
94    }
95}

97pub(crate) fn completed(outcome: Option<&SemanticOutcome>) -> Result<Conclusion, Fault> {
98    match outcome.and_then(SemanticOutcome::search_state) {
99        Some(SearchState::Exhausted) => Ok(Conclusion::Exhausted),
100        Some(SearchState::Interrupted(reason)) => interruption(reason),
101        Some(SearchState::RequestedModels) => Err(Fault::adapter_bug(
102            "native requested-model stop in an uncapped adapter run",
103        )),
104        Some(SearchState::PendingInterruption(_)) => Err(Fault::adapter_bug(
105            "native run completed with checked answers still pending",
106        )),
107        None => Err(Fault::adapter_bug(
108            "native run completed without established search coverage",
109        )),
110    }
111}

113pub(crate) fn session(error: SolveFailure, original: &Program) -> Result<Conclusion, Fault> {
114    // Semantic evidence is not a replacement for the actual error: a later
115    // engine/publication failure must not become a clean semantic conclusion.
116    let result = match error.cause.as_ref() {
117        SolveError::Reconstruction(
118            zetesis_themelios::ReconstructionError::Source(source)
119            | zetesis_themelios::ReconstructionError::Model(zetesis_core::ModelFailure::Stopped(
120                source,
121            )),
122        ) => formula_ref(source, original),
123        SolveError::Constraint(failure) => match &failure.cause {
124            zetesis_themelios::ConstraintCheckCause::Source(source)
125            | zetesis_themelios::ConstraintCheckCause::Index(
126                zetesis_core::AtomIndexError::Stopped(source),
127            ) => formula_ref(source, original),
128            _ => fault(constraint_class(&failure.cause), error.to_string(), None),
129        },
130        cause => fault(solve_class(cause), error.to_string(), None),
131    };
132    result.map_err(|fault| fault.caused_by(error))
133}

135pub(crate) fn observation(
136    error: observation::Error,
137    original: &Program,
138) -> Result<Conclusion, Fault> {
139    fault(
140        observation_class(error.kind()),
141        error.to_string(),
142        statement(original, Some(error.site())).map(ProgramSubject::Statement),
143    )
144    .map_err(|fault| fault.caused_by(error))
145}

147fn solve_class(error: &SolveError) -> Class {
148    match error {
149        SolveError::TerminalStatisticsOverflow
150        | SolveError::HybridStatisticsOverflow
151        | SolveError::LazyStatisticsOverflow
152        | SolveError::ClosureStatisticsOverflow
153        | SolveError::ModelStatisticsOverflow
154        | SolveError::ClosureReservation { .. }
155        | SolveError::CompletionPool(_) => Class::Resource,
156        SolveError::Batch(error) => batch_class(error),
157        SolveError::QueryObservation(error) => batch_class(error),
158        SolveError::SharedCpu(error) => match error {
159            zetesis_cpu::lazy::shared::Cause::Source(stop)
160            | zetesis_cpu::lazy::shared::Cause::World { stop, .. } => stop_class(*stop),
161            zetesis_cpu::lazy::shared::Cause::InvalidOutput => Class::Engine,
162        },
163        SolveError::Static(error) => static_class(error),
164        SolveError::Words(zetesis_core::WordError::Model(error)) | SolveError::Model(error) => {
165            model_class(error)
166        }
167        SolveError::Reconstruction(zetesis_themelios::ReconstructionError::Model(
168            zetesis_core::ModelFailure::Model(error),
169        )) => model_class(error),
170        SolveError::ExecutionObservation(_)
171        | SolveError::ConstraintFailureMissing
172        | SolveError::Executor(_)
173        | SolveError::Words(_)
174        | SolveError::Reconstruction(_)
175        | SolveError::Constraint(_)
176        | SolveError::CandidateStreamNotExhausted
177        | SolveError::FormulaBatchShape { .. } => Class::Engine,
178        // The remaining CPU variants concern an incompatible configured profile:
179        // Projection, HybridBackend, BackendUnavailable, UnsupportedOracle,
180        // UnsupportedSourceBatching or PreparedInput. Cargo feature unification
181        // can also expose Gpu/LazyGpu errors without a matching adapter feature.
182        // This closed CPU, unprojected profile admits none of those paths. Any
183        // unclassified variant is a bridge configuration breach, never a clean
184        // conclusion or a guessed engine/resource classification. Widening the
185        // profile requires explicit classification of the newly reachable errors.
186        _ => Class::Adapter,
187    }
188}

200fn sat_class(error: zetesis_sat::Incomplete) -> Class {
201    use zetesis_sat::Incomplete as E;
202    match error {
203        E::Cancelled => Class::Control(Conclusion::Interrupted),
204        E::Deadline => Class::Control(Conclusion::Budget),
205        E::Verification(stop) => stop_class(stop),
206        E::Admission(error) => sat_admission_class(error),
207        E::Certificate(error) => certificate_class(error),
208        E::WorkLimit
209        | E::DecisionLimit
210        | E::CandidateLimit
211        | E::PendingBytes
212        | E::CompletionScratch
213        | E::ReductStorage { .. }
214        | E::BatchCandidateLimit
215        | E::Allocation
216        | E::ProjectionLimit { .. }
217        | E::CounterOverflow => Class::Resource,
218        E::WorkerPanicked
219        | E::PendingBatch
220        | E::WrongTheory
221        | E::RestrictionUniverse { .. }
222        | E::ClosedEnumerator
223        | E::LateCertificate
224        | E::RegionFilter
225        | E::RegionFilterUnsupported
226        | E::LateRegionFilter
227        | E::RegionFilterAlreadySet
228        | E::InvalidWitness => Class::Engine,
229    }
230}

232fn sat_admission_class(error: zetesis_sat::AdmissionError) -> Class {
233    match error {
234        zetesis_sat::AdmissionError::Limit { .. }
235        | zetesis_sat::AdmissionError::Overflow
236        | zetesis_sat::AdmissionError::Allocation => Class::Resource,
237        zetesis_sat::AdmissionError::Variable { .. } => Class::Engine,
238    }

241fn certificate_class(error: zetesis_sat::CertificateError) -> Class {
242    use zetesis_ferraris::{EvaluationError, PositiveError, TightError};
243    use zetesis_sat::CertificateError;
244    match error {
245        CertificateError::Restriction(error) => sat_admission_class(error),
246        CertificateError::Evaluation(EvaluationError::Stopped(stop))
247        | CertificateError::Tight(TightError::Stopped(stop))
248        | CertificateError::Positive(PositiveError::Stopped(stop)) => stop_class(stop),
249        CertificateError::Evaluation(EvaluationError::Storage { .. })
250        | CertificateError::Tight(TightError::Limit(_))
251        | CertificateError::Positive(PositiveError::Limit { .. } | PositiveError::Overflow) => {
252            Class::Resource
253        }
254        // Ordinary ineligibility must decline the optional native optimization;
255        // it cannot terminate the adapter's otherwise supported answer search.
256        CertificateError::Tight(_) | CertificateError::Positive(_) => Class::Engine,
257    }
258}

338fn observation_class(error: &observation::ErrorKind) -> Class {
339    match error {
340        observation::ErrorKind::Stopped(stop) => stop_class(*stop),

354/// Structural validation carries borrowed evidence and an owned typed cause.
355/// It has no cancellation or deadline effect; its classifier cannot return control.
356pub(crate) fn admission(error: zetesis_themelios::ProgramAdmissionFailure<'_>) -> Fault {
357    let class = program_class(&error.kind);
358    fault(class, error.to_string(), Some(error.subject))
359        .expect_err("structural admission has no runtime control")
360        .caused_by(error.kind)
361}

363pub(crate) fn formula(error: FormulaFailure, original: &Program) -> Result<Conclusion, Fault> {
364    formula_ref(&error, original).map_err(|fault| fault.caused_by(error))
365}
366
367fn formula_ref(error: &FormulaFailure, original: &Program) -> Result<Conclusion, Fault> {
368    let subject = error
369        .subject()
370        .or_else(|| statement(original, error.site()).map(ProgramSubject::Statement));
371    // Whole-input validation limits are resource faults. A semantic refusal
372    // must identify its actual statement or excluded program part; absent
373    // evidence is an internal invariant failure, never a guessed subject.
374    fault(formula_class(error), error.to_string(), subject)
375}

377fn program_class(error: &zetesis_themelios::ProgramFailureKind) -> Class {
378    match error {
379        zetesis_themelios::ProgramFailureKind::Limit(_) => Class::Resource,
380        zetesis_themelios::ProgramFailureKind::Core(error) => core_admission_class(error),
381        zetesis_themelios::ProgramFailureKind::Compilation(error) => match error {
382            zetesis_themelios::CompilationFailure::Profile(_) => Class::Program,
383            zetesis_themelios::CompilationFailure::Construction(error) => construction_class(error),
384        },
385    }
386}

388fn formula_class(error: &FormulaFailure) -> Class {
389    use zetesis_themelios::FormulaFailure as E;
390    match error.cause() {
391        E::Limit { .. } | E::MetadataAllocation { .. } | E::AtomAllocation { .. } => {
392            Class::Resource
393        }
394        E::Interrupted { reason, .. } => stop_class(*reason),
395        E::AtomCatalog { error, .. } => catalog_class(error),
396        E::TermAssignment { error, .. } => assignment_class(error),
397        E::TemplateCatalog { error, .. } => template_class(error),
398        E::Observation { error } => observation_class(error.kind()),
399        E::Expansion(error) => expansion_class(error),
400        E::Aggregate { error, .. } => aggregate_class(*error),
401        E::Theory { error, .. } => theory_class(*error),
402        E::SupportRelation { error, .. } => relation_class(error),
403        E::SupportTable { error, .. } => table_class(&error.cause),
404        E::Logical { error, .. } => program_class(error),
405        E::HybridUnsupported { .. }
406        | E::UnsafeVariable { .. }
407        | E::UnboundArgumentInput { .. }
408        | E::UnboundValueInput { .. }
409        | E::CyclicValueInput { .. } => Class::Program,
410        E::UnadmittedTerm { .. } | E::SourceActivity { .. } => Class::Engine,
411        E::ObjectiveCondition { .. } | E::Objective { .. } | E::Include(_) | E::Program { .. } => {
412            Class::Adapter
413        }
414    }
415}

417fn expansion_class(error: &zetesis_themelios::ExpansionFailure) -> Class {
418    use zetesis_themelios::ExpansionFailure as E;

431    use zetesis_themelios::AdmissionFailure as E;

460fn aggregate_class(error: zetesis_ferraris::AggregateError) -> Class {
461    use zetesis_ferraris::AggregateErrorKind as E;
462    match error.kind() {
463        E::Control(stop) => stop_class(stop),
464        E::InvalidPrefix { .. } | E::InvalidCondition { .. } => Class::Engine,
465        E::ElementLimit
466        | E::GuardLimit
467        | E::NodeLimit
468        | E::OperandLimit
469        | E::WorkLimit
470        | E::StateLimit
471        | E::SubsetLimit
472        | E::Allocation
473        | E::ArithmeticOverflow => Class::Resource,
474    }
475}
