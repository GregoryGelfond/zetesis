1//! Transfer a checked native answer to the canonical model without text or caches.
2//!
3//! Symbol construction has one independent byte allowance per atom and a
4//! cumulative symbol-walk allowance across this run. The atom count separately bounds
5//! the result container. Observation limits are per answer. These are named
6//! logical/construction limits, not a resident-memory ceiling. Upstream set
7//! insertion and Model display unions use infallible allocation and comparison;
8//! polls bracket those bounded operations, but cannot interrupt within them.
9

22/// Independent bounds on exporting complete atoms and evaluated display terms.
23///
24/// Atom count bounds each output set, byte capacity bounds one constructed atom,
25/// and symbol work accumulates across the whole run. Observation limits reset
26/// for each delivered model. These limits do not account for allocator overhead,
27/// upstream ordered-set comparisons or the caller's retained models.
28#[derive(Clone, Copy, Debug)]
29pub struct OutputLimits {
30    /// Full atom cardinality in one model.
31    pub max_atoms: usize,
32    /// Independent construction capacity for one signed atom.
33    pub max_atom_bytes: u128,
34    /// Cumulative symbol-walk callbacks; excludes ordered-set comparisons.
35    pub max_symbol_work: u64,
36    /// Evaluation allowances for each model’s `#show` terms.
37    pub observation: observation::Limits,
38    /// Construction allowances for each model’s displayed symbols.
39    pub construction: observation::ConstructionLimits,
40}
41
42impl OutputLimits {
43    /// Ordinary output capacities from the same resource policy as source and
44    /// solving. The symbol-work counter has only its representation maximum.
45    #[must_use]
46    pub fn for_resources(resources: zetesis_solve::Resources) -> Self {
47        Self {
48            max_atoms: resources.formula_limits().theory.max_atoms,
49            max_atom_bytes: resources.json_record_bytes() as u128,
50            max_symbol_work: u64::MAX,
51            observation: resources.observation_limits(),
52            construction: observation::ConstructionLimits {
53                max_bytes: resources.json_record_bytes(),
54            },
55        }
56    }
57}
58
59impl Default for OutputLimits {
60    fn default() -> Self {
61        Self::for_resources(zetesis_solve::Resources::default())
62    }
63}

65pub(crate) struct Exporter {
66    limits: OutputLimits,
67    work: u128,
68}
69
70impl Exporter {
71    pub(crate) fn new(limits: OutputLimits) -> Self {
72        Self { limits, work: 0 }
73    }
74
75    pub(crate) fn convert(
76        &mut self,
77        answer: &zetesis_solve::AnswerSet,
78        metadata: &SourceMetadata,
79        cancellation: &Cancellation,
80    ) -> Result<Model, OutputError> {
81        if answer.score().is_some() {
82            return Err(OutputError::ScoredAnswer);
83        }
84        cancellation.poll().map_err(OutputError::Control)?;
85        let native = answer.interpretation();
86        let count = native.atoms().len();
87        if count > self.limits.max_atoms {
88            return Err(OutputError::Atoms {
89                observed: count,
90                limit: self.limits.max_atoms,
91            });
92        }
93        let mut atoms = AnswerSet::new();
94        for atom in native.atoms() {
95            let symbol = symbols::atom_with(atom, self.limits.max_atom_bytes, || {
96                cancellation.poll().map_err(SymbolStop::Control)?;
97                self.work += 1;
98                if self.work > u128::from(self.limits.max_symbol_work) {
99                    return Err(SymbolStop::Work {
100                        observed: self.work,
101                        limit: self.limits.max_symbol_work,
102                    });
103                }
104                Ok(())
105            })
106            .map_err(OutputError::Symbol)?;
107            atoms.insert(symbol);
108        }
109        cancellation.poll().map_err(OutputError::Control)?;
110        let terms = metadata
111            .observations()
112            .evaluate_with_construction_limits(
113                native,
114                self.limits.observation,
115                self.limits.construction,
116                cancellation,
117            )
118            .map_err(OutputError::Observation)?
119            .into_symbols();
120        cancellation.poll().map_err(OutputError::Control)?;
121        let model = Model::of(atoms).with_terms(terms);
122        cancellation.poll().map_err(OutputError::Control)?;
123        Ok(model)
124    }
125}

127/// Why construction of one canonical output symbol stopped.
128#[derive(Debug)]
129pub enum SymbolStop {
130    /// Cancellation or the request's deadline was observed during construction.
131    Control(Stop),
132    /// Cumulative symbol traversal exceeded the run's configured work allowance.
133    Work {
134        /// Traversal callbacks attempted across the run, including this refusal.
135        observed: u128,
136        /// Configured maximum traversal callbacks.
137        limit: u64,
138    },
139}
140
141impl fmt::Display for SymbolStop {
142    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
143        match self {
144            Self::Control(stop) => stop.fmt(f),
145            Self::Work { observed, limit } => write!(
146                f,
147                "symbol export requires {observed} work units, allowance is {limit}"
148            ),
149        }
150    }
151}
152impl Error for SymbolStop {
153    fn source(&self) -> Option<&(dyn Error + 'static)> {
154        match self {
155            Self::Control(stop) => Some(stop),
156            Self::Work { .. } => None,
157        }
158    }
159}

161/// A typed failure while exporting a native answer through the Rust solve API.
162#[derive(Debug)]
163pub enum OutputError {
164    /// Request control observed at an output boundary.
165    Control(Stop),
166    /// The complete answer exceeds the configured output atom count.
167    Atoms {
168        /// Atoms in the native answer before export.
169        observed: usize,
170        /// Configured maximum atoms in one exported answer.
171        limit: usize,
172    },
173    /// Canonical symbol construction failed or reached its separate work limit.
174    Symbol(symbols::Failure<SymbolStop>),
175    /// Evaluation or construction of a `#show` term failed.
176    Observation(observation::Error),
177    /// The unscored backend received a scored native answer: an internal defect.
178    ScoredAnswer,
179}
180
181impl OutputError {
182    pub(crate) fn finish(self, original: &Program) -> Result<Conclusion, Fault> {
183        if let Self::Observation(error) = self {
184            return faults::observation(error, original);
185        }
186        let stop = match &self {
187            Self::Control(stop)
188            | Self::Symbol(symbols::Failure::Stopped(SymbolStop::Control(stop))) => Some(*stop),
189            _ => None,
190        };
191        if let Some(conclusion) = stop.and_then(faults::control) {
192            return Ok(conclusion);
193        }
194        let fault = match &self {
195            Self::Atoms { .. }
196            | Self::Symbol(
197                symbols::Failure::Bridge(
198                    symbols::Error::Allocation | symbols::Error::Storage { .. },
199                )
200                | symbols::Failure::Stopped(SymbolStop::Work { .. }),
201            ) => Fault::resource(self.to_string()),
202            _ => Fault::adapter_bug(self.to_string()),
203        };
204        Err(fault.caused_by(self))
205    }
206}
207
208impl fmt::Display for OutputError {
209    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
210        match self {
211            Self::Control(stop) => stop.fmt(f),
212            Self::Atoms { observed, limit } => {
213                write!(f, "model export has {observed} atoms, allowance is {limit}")
214            }
215            Self::Symbol(error) => error.fmt(f),
216            Self::Observation(error) => error.fmt(f),
217            Self::ScoredAnswer => f.write_str("unscored enumeration yielded an objective score"),
218        }
219    }
220}
221impl Error for OutputError {
222    fn source(&self) -> Option<&(dyn Error + 'static)> {
223        match self {
224            Self::Control(stop) => Some(stop),
225            Self::Symbol(error) => Some(error),
226            Self::Observation(error) => Some(error),
227            Self::Atoms { .. } | Self::ScoredAnswer => None,
228        }
229    }
