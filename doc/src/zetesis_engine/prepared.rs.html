1//! Owned native preparations, each borrowing one original canonical program.
2
3use std::sync::Arc;
4
5use themelios_program::program::Program;
6use zetesis_cpu::Cancellation;
7use zetesis_solve::PreparedInput;
8use zetesis_themelios::{
9    AutomaticAdmission, AutomaticAdmissionFailure, FormulaFailure, FormulaPurpose,
10    PreparedRelational, ProgramFormulaOptions, SourceMetadata, prepare_program_automatic,
11    prepare_program_formula_with,
12};
13
14use crate::{Config, Grounder};
15
16pub(crate) enum Prepared {
17    Relational(Box<PreparedRelational>),
18    /// Prepared by the shared execution policy, as the CLI's input is.
19    Execution(zetesis_solve::PreparedExecution),
20}
21
22impl Prepared {
23    pub(crate) fn new(
24        original: Arc<Program>,
25        config: &Config,
26        cancellation: &Cancellation,
27    ) -> Result<Self, FormulaFailure> {
28        let resources = config.resources();
29        let options = ProgramFormulaOptions {
30            admission: resources.program_admission_options(),
31            expansion: resources.expansion_limits(),
32            formula: resources.formula_limits(),
33            purpose: FormulaPurpose::AnswerSets,
34            cancellation: Some(cancellation.clone()),
35        };
36        let prepared = if config.grounder == Grounder::Lazy {
37            match prepare_program_automatic(original, options) {
38                Ok(AutomaticAdmission::Extended(relational)) => {
39                    return Ok(Self::Relational(Box::new(relational)));
40                }
41                Ok(AutomaticAdmission::Formula(prepared)) => prepared,
42                Err(
43                    AutomaticAdmissionFailure::Extended(error)
44                    | AutomaticAdmissionFailure::Formula(error),
45                ) => return Err(error),
46            }
47        } else {
48            prepare_program_formula_with(original, options)?
49        };
50        zetesis_solve::prepare_formula_execution(prepared, &config.session(), None)
51            .map(Self::Execution)
52    }
53
54    pub(crate) fn input(&self) -> PreparedInput<'_> {
55        match self {
56            Self::Relational(owner) => PreparedInput::relational(owner),
57            Self::Execution(prepared) => prepared.input(),
58        }
59    }
60
61    pub(crate) fn metadata(&self) -> &SourceMetadata {
62        match self {
63            Self::Relational(owner) => owner.metadata(),
64            Self::Execution(prepared) => prepared.metadata(),
65        }
66    }
67}
