10/// A finite extremum with the source language's empty-set convention.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum AggregateExtremum {
13    /// Minimum eligible value; the empty result is positive infinity (`#sup`).
14    Min,
15    /// Maximum eligible value; the empty result is negative infinity (`#inf`).
16    Max,
17}
18
19/// A numeric extremum guard, including the language's two ordered sentinels.
20/// Sentinels are distinct from every finite integer, including integer limits.
21#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
22pub enum ExtremumBound {
23    /// The least source term, `#inf`.
24    NegativeInfinity,
25    /// A finite signed numeric guard.
26    Number(i64),
27    /// The greatest source term, `#sup`.
28    PositiveInfinity,
29}
30impl From<i64> for ExtremumBound {
31    fn from(value: i64) -> Self {
32        Self::Number(value)
33    }
34}
35impl From<i32> for ExtremumBound {
36    fn from(value: i32) -> Self {
37        Self::Number(i64::from(value))
38    }
39}

41/// Append an exact finite numeric `#min` or `#max` comparison transactionally.
42/// Elements use their numeric `weight` as the first tuple component and retain
43/// arbitrary existing eligibility formulas. Equal complete tuples must already
44/// be OR-coalesced by the caller, as for [`super::append_aggregate`].
45///
46/// The empty maximum is `#inf`; the empty minimum is `#sup`. Numeric guards can
47/// be passed directly, or an explicit [`ExtremumBound`] can test the sentinels.
48/// Non-numeric tuple first components and explicit infinite element values are
49/// outside this numeric interface. Source assignment, origin tracking and guard
50/// expansion remain caller obligations. Default negation is applied afterward
51/// as implication to falsum, never by inverting the aggregate comparison.
52///
53/// Compilation uses two filtered disjunctions of eligibility formulas, linear
54/// in the input and existing prefix. It allocates no DP rows or subset carrier,
55/// so `max_states` and `max_subsets` may both be zero. It introduces no semantic
56/// atoms and retains an implication for not-equal. Prefix validation, inclusive
57/// node/work/element ceilings, cancellation, allocation and rollback follow
58/// [`super::append_aggregate`]. The original DAG length and nodes survive errors.
59///
60/// # Errors
61/// Returns a typed input, resource, allocation, cancellation or arithmetic error.
62pub fn append_extremum(
63    nodes: &mut FormulaNodes,
64    elements: &[AggregateElement],
65    extremum: AggregateExtremum,
66    comparison: AggregateComparison,
67    bound: impl Into<ExtremumBound>,
68    limits: AggregateLimits,
69    cancellation: &Cancellation,
70) -> Result<AggregateBuild, AggregateError> {
71    append(
72        Destination::retained(nodes.transaction()),
73        elements,
74        extremum,
75        comparison,
76        bound.into(),
77        limits,
78        cancellation,
79    )
80}

82pub(super) fn append(
83    destination: Destination<'_>,
84    elements: &[AggregateElement],
85    extremum: AggregateExtremum,
86    comparison: AggregateComparison,
87    bound: ExtremumBound,
88    limits: AggregateLimits,
89    cancellation: &Cancellation,
90) -> Result<AggregateBuild, AggregateError> {
91    transaction(destination, limits, cancellation, |builder| {
92        compile(builder, elements, extremum, comparison, bound)
93            .map(|root| (root, AggregateProfile::Extremum))
94    })
95}

97fn compile(
98    builder: &mut Builder<'_, '_>,
99    elements: &[AggregateElement],
100    extremum: AggregateExtremum,
101    comparison: AggregateComparison,
102    bound: ExtremumBound,
103) -> Result<usize, AggregateErrorKind> {
104    validate(builder, elements)?;
105    let falsum = builder.push(NodeView::False)?;
106    let truth = builder.push(NodeView::Implies(falsum, falsum))?;
107    // G is max >= bound / min <= bound. H is the corresponding strict
108    // comparison. Both are monotone in eligible tuples, including sentinels.
109    let (inclusive, strict) = witnesses(builder, elements, extremum, bound, falsum, truth)?;
110    comparison_root(builder, extremum, comparison, inclusive, strict, falsum)
111}

113pub(super) fn comparison_root(
114    builder: &mut Builder<'_, '_>,
115    extremum: AggregateExtremum,
116    comparison: AggregateComparison,
117    inclusive: usize,
118    strict: usize,
119    falsum: usize,
120) -> Result<usize, AggregateErrorKind> {
121    let (positive, negative) = match extremum {
122        AggregateExtremum::Max => (AggregateComparison::Ge, AggregateComparison::Lt),
123        AggregateExtremum::Min => (AggregateComparison::Le, AggregateComparison::Gt),
124    };
125    if comparison == positive {
126        return Ok(inclusive);
127    }
128    if comparison == negative {
129        return builder.push(NodeView::Implies(inclusive, falsum));
130    }
131    match comparison {
132        AggregateComparison::Eq => {
133            let below_strict = builder.push(NodeView::Implies(strict, falsum))?;
134            builder.push(NodeView::And(&[inclusive, below_strict]))
135        }
136        AggregateComparison::Ne => builder.push(NodeView::Implies(inclusive, strict)),
137        AggregateComparison::Gt | AggregateComparison::Lt => Ok(strict),
138        AggregateComparison::Ge | AggregateComparison::Le => {
139            builder.push(NodeView::Implies(strict, falsum))
140        }
141    }
142}

144fn witnesses(
145    builder: &mut Builder<'_, '_>,
146    elements: &[AggregateElement],
147    extremum: AggregateExtremum,
148    bound: ExtremumBound,
149    falsum: usize,
150    truth: usize,
151) -> Result<(usize, usize), AggregateErrorKind> {
152    let empty = match extremum {
153        AggregateExtremum::Max => ExtremumBound::NegativeInfinity,
154        AggregateExtremum::Min => ExtremumBound::PositiveInfinity,
155    };
156    let mut inclusive = Vec::new();
157    if bound == empty {
158        builder.operand(&mut inclusive, truth)?;
159    }
160    let mut strict = Vec::new();
161    for element in elements {
162        builder.tick()?;
163        let value = ExtremumBound::Number(i64::from(element.weight));
164        let order = match extremum {
165            AggregateExtremum::Max => value.cmp(&bound),
166            AggregateExtremum::Min => bound.cmp(&value),
167        };
168        if order.is_ge() {
169            builder.operand(&mut inclusive, element.condition)?;
170        }
171        if order.is_gt() {
172            builder.operand(&mut strict, element.condition)?;
173        }
174    }
175    Ok((
176        builder.group(&inclusive, falsum, false)?,
177        builder.group(&strict, falsum, false)?,
178    ))
