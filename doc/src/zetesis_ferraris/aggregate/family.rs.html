89/// Append an ordered family of finite `#count`/`#sum` comparisons atomically.
90/// All guards share exactly the same already OR-coalesced complete tuples and
91/// eligibility formulas. The caller establishes that this element family is
92/// independent of the changing guard, including its source bindings and support
93/// universe. This function checks indices and bounds, not that source premise.
94///
95/// For nonnegative weights, native threshold nodes share one contribution row.
96/// Constants simplify. Families needing no numeric thresholds also simplify to
97/// conjunctions and disjunctions; mixed families retain all nonconstant guards
98/// on their shared row so propagation can relate their bounds.
99/// Equality uses `G(k) AND NOT G(k+1)`; not-equal retains `G(k) -> G(k+1)`.
100/// Negative weights use the full failing-subset formula separately for each
101/// guard, with shared validation and cumulative work/subset ceilings.
102///
103/// Prefix and input validation occur once. An empty guard family still validates
104/// input and polls control, then returns no roots and appends no nodes. Output
105/// root storage is bounded by `max_guards`; the temporary-state budget counts numeric columns, bounds
106/// or subset bits and excludes this returned vector and DAG storage.
107/// Every loop visit and node append is charged and polls cancellation/deadlines.
108/// A limit, control, arithmetic or allocation error rolls back every new node;
109/// no partial root vector is returned. Existing [`super::append_aggregate`] limits
110/// and per-guard behavior are unchanged. Default negation remains a caller wrap.
111///
112/// # Errors
113/// Returns typed input, guard, resource, allocation, cancellation or arithmetic
114/// errors. Shared limits apply to the entire family, never independently per root.
115pub fn append_aggregate_family(
116    nodes: &mut FormulaNodes,
117    elements: &[AggregateElement],
118    guards: &[AggregateGuard],
119    limits: AggregateFamilyLimits,
120    cancellation: &Cancellation,
121) -> Result<AggregateFamilyBuild, AggregateError> {
122    append(
123        Destination::retained(nodes.transaction()),
124        elements,
125        guards,
126        limits,
127        cancellation,
128    )
129}

131pub(super) fn append(
132    destination: Destination<'_>,
133    elements: &[AggregateElement],
134    guards: &[AggregateGuard],
135    limits: AggregateFamilyLimits,
136    cancellation: &Cancellation,
137) -> Result<AggregateFamilyBuild, AggregateError> {
138    transaction_value(destination, limits.aggregate, cancellation, |builder| {
139        compile(builder, elements, guards, limits.max_guards)
140    })
141    .map(|((roots, profile), statistics)| AggregateFamilyBuild {
142        roots,
143        profile,
144        statistics,
145    })
146}

148type ResultKind<T> = Result<T, AggregateErrorKind>;
149
150pub(super) fn compile(
151    builder: &mut Builder<'_, '_>,
152    elements: &[AggregateElement],
153    guards: &[AggregateGuard],
154    max_guards: usize,
155) -> ResultKind<(Vec<usize>, AggregateProfile)> {
156    let (nonnegative, total) = validate(builder, elements)?;
157    if guards.len() > max_guards {
158        return Err(AggregateErrorKind::GuardLimit);
159    }
160    let profile = if nonnegative {
161        AggregateProfile::Threshold
162    } else {
163        AggregateProfile::SubsetImplications
164    };
165    if guards.is_empty() {
166        return Ok((Vec::new(), profile));
167    }
168    let falsum = builder.push(NodeView::False)?;
169    let truth = builder.push(NodeView::Implies(falsum, falsum))?;
170    let roots = if nonnegative {
171        super::monotone::compile(builder, elements, guards, total, falsum, truth)?
172    } else {
173        let mut roots = reserve(guards.len())?;
174        for guard in guards {
175            builder.tick()?;
176            roots.push(subsets(
177                builder,
178                elements,
179                guard.comparison,
180                i128::from(guard.bound),
181                falsum,
182                truth,
183            )?);
184        }
185        roots
186    };
187    Ok((roots, profile))
188}
