18pub(super) fn build(
19    theory: &Theory,
20    function: Function,
21    tuples: Vec<Tuple>,
22    guards: Vec<Guard>,
23    limits: AdmissionLimits,
24    cancellation: &Cancellation,
25) -> Result<Group, Error> {
26    let mut work = Work {
27        maximum: limits.max_work,
28        cancellation,
29        statistics: Statistics::default(),
30    };
31    let result = (|| {
32        work.poll()?;
33        let maximum = usize::try_from(limits.max_storage_bytes).unwrap_or(usize::MAX);
34        let owner = VocabularyBuilder::new(maximum).map_err(catalog)?;
35        let mut admission = Admission::new(
36            theory,
37            function,
38            owner,
39            tuples.len(),
40            guards.len(),
41            limits,
42            &mut work,
43        )?;
44        admission.logical(add(
45            bytes::<Tuple>(tuples.capacity())?,
46            bytes::<Guard>(guards.capacity())?,
47        )?)?;
48        for tuple in tuples {
49            admission.begin_tuple(tuple.key.len(), tuple.key.capacity(), tuple.condition)?;
50            for value in &tuple.key {
51                admission.owned_value(value)?;
52            }
53            admission.end_tuple(tuple.condition)?;
54        }
55        for guard in guards {
56            admission.work.charge(1)?;
57            let bound = match &guard.bound {
58                Bound::Integer(value) => {
59                    admission.guard_costs(1)?;
60                    BoundData::Integer(*value)
61                }
62                Bound::Term(value) => {
63                    let cost = term_work(value.into(), admission.work)?;
64                    let at = admission.owned_value(value)?;
65                    admission.guard_costs(cost)?;
66                    BoundData::Term(at)
67                }
68            };
69            admission.data.guards.push(GuardData {
70                comparison: guard.comparison,
71                bound,
72            });
73        }
74        let (mut data, mut owner) = admission.finish()?;
75        owner.ceiling(maximum).map_err(catalog)?;
76        let vocabulary = owner
77            .finish_with(data_bytes(&data), || work.charge(1))
78            .map_err(vocabulary)?;
79        work.statistics.storage_bytes = narrow(data_bytes(&data) + vocabulary.storage_bytes())?;
80        work.statistics.storage_peak_bytes = work
81            .statistics
82            .storage_peak_bytes
83            .max(narrow(vocabulary.publication_peak_bytes())?);
84        data.statistics = work.statistics;
85        Ok(Group { vocabulary, data })
86    })();
87    result.map_err(|kind| work.failure(kind))
88}

90pub(super) fn borrowed<'a>(
91    theory: &Theory,
92    function: Function,
93    read: CatalogRead<'a>,
94    tuples: Vec<Tuple<Vec<TermRef<'a>>>>,
95    guards: Vec<Guard<TermRef<'a>>>,
96    limits: AdmissionLimits,
97    cancellation: &Cancellation,
98) -> Result<GroupData, Error> {
99    let mut work = Work {
100        maximum: limits.max_work,
101        cancellation,
102        statistics: Statistics::default(),
103    };
104    let result = (|| {
105        work.poll()?;
106        let mut admission = Admission::new(
107            theory,
108            function,
109            read,
110            tuples.len(),
111            guards.len(),
112            limits,
113            &mut work,
114        )?;
115        // The logical carrier uses Value cells even though execution retains only IDs.
116        admission.logical(add(
117            bytes::<Tuple>(tuples.capacity())?,
118            bytes::<Guard>(guards.capacity())?,
119        )?)?;
120        for tuple in tuples {
121            admission.begin_tuple(tuple.key.len(), tuple.key.capacity(), tuple.condition)?;
122            for &value in &tuple.key {
123                admission.borrowed_value(value)?;
124            }
125            admission.end_tuple(tuple.condition)?;
126        }
127        for guard in guards {
128            admission.work.charge(1)?;
129            let bound = match guard.bound {
130                Bound::Integer(value) => {
131                    admission.guard_costs(1)?;
132                    BoundData::Integer(value)
133                }
134                Bound::Term(value) => {
135                    let cost = term_work(value, admission.work)?;
136                    let at = admission.borrowed_value(value)?;
137                    admission.guard_costs(cost)?;
138                    BoundData::Term(at)
139                }
140            };
141            admission.data.guards.push(GuardData {
142                comparison: guard.comparison,
143                bound,
144            });
145        }
146        let (mut data, _) = admission.finish()?;
147        data.statistics = work.statistics;
148        Ok(data)
149    })();
150    result.map_err(|kind| work.failure(kind))
151}

153/// The admission algorithm either imports into its owned vocabulary or borrows
154/// an already admitted prefix. The source type determines that capability;
155/// neither route allocates an indirection or retains a second payload owner.
156trait Authority {
157    fn read(&self) -> CatalogRead<'_>;
158    fn bytes(&self) -> u128;
159    fn import(
160        &mut self,
161        value: TermRef<'_>,
162        external: u128,
163        limits: AdmissionLimits,
164        work: &mut Work<'_>,
165    ) -> Result<Option<TermKey>, ErrorKind>;
166}
167impl Authority for VocabularyBuilder {
168    fn read(&self) -> CatalogRead<'_> {
169        Self::read(self)
170    }
171    fn bytes(&self) -> u128 {
172        self.storage_bytes()
173    }
174    fn import(
175        &mut self,
176        value: TermRef<'_>,
177        external: u128,
178        limits: AdmissionLimits,
179        work: &mut Work<'_>,
180    ) -> Result<Option<TermKey>, ErrorKind> {
181        self.ceiling(available(limits, external)?)
182            .map_err(catalog)?;
183        self.restart_storage_peak();
184        let result = self.import_term_with(
185            value,
186            Limits {
187                max_nodes: usize::MAX,
188                max_depth: usize::MAX,
189                max_bytes: usize::MAX,
190            },
191            || work.charge(1),
192        );
193        let current = external + self.storage_bytes();
194        work.statistics.storage_bytes = narrow(current)?;
195        work.statistics.storage_peak_bytes = work
196            .statistics
197            .storage_peak_bytes
198            .max(narrow(external + self.storage_peak_bytes())?);
199        result.map(Some).map_err(vocabulary)
200    }
201}
202impl Authority for CatalogRead<'_> {
203    fn read(&self) -> CatalogRead<'_> {
204        *self
205    }
206    fn bytes(&self) -> u128 {
207        0
208    }
209    fn import(
210        &mut self,
211        _value: TermRef<'_>,
212        _external: u128,
213        _limits: AdmissionLimits,
214        _work: &mut Work<'_>,
215    ) -> Result<Option<TermKey>, ErrorKind> {
216        Ok(None)
217    }
218}
219struct Admission<'w, 'c, A> {
220    data: GroupData,
221    authority: A,
222    limits: AdmissionLimits,
223    work: &'w mut Work<'c>,
224    key_costs: Vec<u64>,
225    logical_scratch: u64,
226    tuple_values: usize,
227    value_nodes: usize,
228    next_term: usize,
229    key_start: usize,
230    key_cost: u64,
231    first_cost: u64,
232    in_key: bool,
233}
234impl<'w, 'c, A: Authority> Admission<'w, 'c, A> {
235    fn new(
236        theory: &Theory,
237        function: Function,
238        authority: A,
239        tuples: usize,
240        guards: usize,
241        limits: AdmissionLimits,
242        work: &'w mut Work<'c>,
243    ) -> Result<Self, ErrorKind> {
244        ceiling(tuples, limits.max_tuples, Resource::Tuples)?;
245        ceiling(guards, limits.max_guards, Resource::Guards)?;
246        let allowance = available(
247            limits,
248            authority.bytes() + size_of::<GroupData>() as u128
249                - size_of::<TemplateComponents>() as u128,
250        )?;
251        let components =
252            TemplateComponents::new(authority.read(), allowance).map_err(component_never)?;
253        let data = GroupData {
254            theory: theory.clone(),
255            function,
256            components,
257            tuples: Vec::new(),
258            guards: Vec::new(),
259            first_costs: Vec::new(),
260            guard_costs: Vec::new(),
261            statistics: Statistics::default(),
262        };
263        let mut result = Self {
264            data,
265            authority,
266            limits,
267            work,
268            key_costs: Vec::new(),
269            logical_scratch: add(
270                bytes::<u64>(tuples)?,
271                bytes::<usize>(tuples)?
272                    .checked_mul(2)
273                    .ok_or(ErrorKind::Overflow)?,
274            )?,
275            tuple_values: 0,
276            value_nodes: 0,
277            next_term: 0,
278            key_start: 0,
279            key_cost: 1,
280            first_cost: 0,
281            in_key: false,
282        };
283        result.observe(0)?;
284        result.reserve_buffers(tuples, guards)?;
285        result.logical(add(bytes::<u64>(tuples)?, bytes::<u64>(guards)?)?)?;
286        Ok(result)
287    }
288    fn current(&self) -> u128 {
289        self.authority.bytes()
290            + data_bytes(&self.data)
291            + self.key_costs.capacity() as u128 * size_of::<u64>() as u128
292    }
293    fn observe(&mut self, overlap: u128) -> Result<(), ErrorKind> {
294        let current = self.current();
295        let peak = current.checked_add(overlap).ok_or(ErrorKind::Overflow)?;
296        self.work.statistics.storage_bytes = narrow(current)?;
297        self.work.statistics.storage_peak_bytes =
298            self.work.statistics.storage_peak_bytes.max(narrow(peak)?);
299        storage_limit(peak, self.limits)
300    }
301    fn reserve_buffers(&mut self, tuples: usize, guards: usize) -> Result<(), ErrorKind> {
302        self.data.tuples = reserve(tuples, self.current(), self.limits, self.work)?;
303        self.observe(0)?;
304        self.data.guards = reserve(guards, self.current(), self.limits, self.work)?;
305        self.observe(0)?;
306        self.data.first_costs = reserve(tuples, self.current(), self.limits, self.work)?;
307        self.observe(0)?;
308        self.data.guard_costs = reserve(guards, self.current(), self.limits, self.work)?;
309        self.observe(0)?;
310        self.key_costs = reserve(tuples, self.current(), self.limits, self.work)?;
311        self.observe(0)
312    }
313    fn logical(&mut self, added: u64) -> Result<(), ErrorKind> {
314        self.work.statistics.resident_bytes = add(self.work.statistics.resident_bytes, added)?;
315        self.work.statistics.peak_bytes =
316            add(self.work.statistics.resident_bytes, self.logical_scratch)?;
317        if self.work.statistics.peak_bytes > self.limits.max_bytes {
318            return Err(ErrorKind::Limit(Resource::Bytes));
319        }
320        Ok(())
321    }
322    fn begin_tuple(
323        &mut self,
324        len: usize,
325        capacity: usize,
326        condition: usize,
327    ) -> Result<(), ErrorKind> {
328        self.work.charge(1)?;
329        if condition >= self.data.theory.nodes().len() {
330            return Err(ErrorKind::Condition {
331                tuple: self.data.tuples.len(),
332                node: condition,
333            });
334        }
335        self.tuple_values = self
336            .tuple_values
337            .checked_add(len)
338            .ok_or(ErrorKind::Overflow)?;
339        ceiling(
340            self.tuple_values,
341            self.limits.max_tuple_values,
342            Resource::TupleValues,
343        )?;
344        self.logical(bytes::<Term>(capacity)?)?;
345        self.in_key = true;
346        self.key_start = self.next_term;
347        self.key_cost = 1;
348        self.first_cost = 0;
349        Ok(())
350    }
351    fn inspect(&mut self, value: TermRef<'_>, payload: u64) -> Result<u64, ErrorKind> {
352        self.work.charge(1)?;
353        self.value_nodes = self
354            .value_nodes
355            .checked_add(value.expanded_nodes())
356            .ok_or(ErrorKind::Overflow)?;
357        ceiling(
358            self.value_nodes,
359            self.limits.max_value_nodes,
360            Resource::ValueNodes,
361        )?;
362        let cost = term_work(value, self.work)?;
363        self.work.charge(cost)?;
364        self.logical(payload)?;
365        Ok(cost)
366    }
367    fn owned_value(&mut self, value: &Term) -> Result<usize, ErrorKind> {
368        self.work.charge(1)?;
369        let nodes = TermRef::from(value).expanded_nodes();
370        self.work
371            .charge(u64::try_from(nodes).map_err(|_| ErrorKind::Overflow)?)?;
372        let cost = self.inspect(value.into(), narrow(value.payload_bytes() as u128)?)?;
373        self.append(value.into(), cost)
374    }
375    fn borrowed_value(&mut self, value: TermRef<'_>) -> Result<usize, ErrorKind> {
376        self.work.charge(1)?;
377        self.authority
378            .read()
379            .term_key(value)
380            .map_err(ErrorKind::Canonical)?;
381        let payload = logical_payload(value, self.work)?;
382        let cost = self.inspect(value, payload)?;
383        self.append(value, cost)
384    }
385    fn append(&mut self, value: TermRef<'_>, cost: u64) -> Result<usize, ErrorKind> {
386        let external = self.current() - self.authority.bytes();
387        let imported = self
388            .authority
389            .import(value, external, self.limits, self.work)?;
390        let value = if let Some(key) = &imported {
391            self.authority
392                .read()
393                .term(key)
394                .map_err(ErrorKind::Canonical)?
395        } else {
396            value
397        };
398        let other = self.authority.bytes() + data_bytes(&self.data)
399            - self.data.components.storage_bytes()
400            + self.key_costs.capacity() as u128 * size_of::<u64>() as u128;
401        let maximum = available(self.limits, other)?;
402        self.data.components.restart_storage_peak();
403        let result = self.data.components.append_terms_with(
404            self.authority.read(),
405            [TemplateTerm::Constant(value)],
406            maximum,
407            || self.work.charge(1),
408        );
409        self.work.statistics.storage_peak_bytes = self
410            .work
411            .statistics
412            .storage_peak_bytes
413            .max(narrow(other + self.data.components.storage_peak_bytes())?);
414        let range = result.map_err(component)?;
415        self.observe(0)?;
416        self.next_term = range.end;
417        if self.in_key {
418            if range.start == self.key_start {
419                self.first_cost = cost;
420            }
421            self.key_cost = add(self.key_cost, cost)?;
422        }
423        Ok(range.start)
424    }
425    fn end_tuple(&mut self, condition: usize) -> Result<(), ErrorKind> {
426        self.work.charge(1)?;
427        self.data.tuples.push(TupleData {
428            key: self.key_start..self.next_term,
429            condition,
430        });
431        self.data.first_costs.push(self.first_cost);
432        self.key_costs.push(self.key_cost);
433        self.in_key = false;
434        Ok(())
435    }
436    fn guard_costs(&mut self, cost: u64) -> Result<(), ErrorKind> {
437        self.work.charge(cost)?;
438        self.data.guard_costs.push(cost);
439        Ok(())
440    }
441    fn finish(mut self) -> Result<(GroupData, A), ErrorKind> {
442        let view = self
443            .data
444            .bind_with(self.authority.read(), || self.work.charge(1))
445            .map_err(component)?;
446        unique(
447            view,
448            &self.key_costs,
449            self.current(),
450            self.limits,
451            self.work,
452        )?;
453        self.work.poll()?;
454        self.key_costs.clear();
455        drop(self.key_costs);
456        self.work.statistics.storage_bytes =
457            narrow(self.authority.bytes() + data_bytes(&self.data))?;
458        Ok((self.data, self.authority))
459    }
460}

462fn data_bytes(data: &GroupData) -> u128 {
463    size_of::<GroupData>() as u128 - size_of::<TemplateComponents>() as u128
464        + data.components.storage_bytes()
465        + data.tuples.capacity() as u128 * size_of::<TupleData>() as u128
466        + data.guards.capacity() as u128 * size_of::<GuardData>() as u128
467        + (data.first_costs.capacity() as u128 + data.guard_costs.capacity() as u128)
468            * size_of::<u64>() as u128
469}
470fn narrow(bytes: u128) -> Result<u64, ErrorKind> {
471    u64::try_from(bytes).map_err(|_| ErrorKind::Overflow)
472}
473fn storage_limit(required: u128, limits: AdmissionLimits) -> Result<(), ErrorKind> {
474    if required > u128::from(limits.max_storage_bytes) {
475        Err(ErrorKind::Limit(Resource::StorageBytes))
476    } else {
477        Ok(())
478    }
479}
480fn available(limits: AdmissionLimits, outside: u128) -> Result<usize, ErrorKind> {
481    storage_limit(outside, limits)?;
482    Ok(usize::try_from(u128::from(limits.max_storage_bytes) - outside).unwrap_or(usize::MAX))
483}
484fn reserve<T>(
485    count: usize,
486    current: u128,
487    limits: AdmissionLimits,
488    work: &mut Work<'_>,
489) -> Result<Vec<T>, ErrorKind> {
490    work.charge(1)?;
491    storage_limit(current + count as u128 * size_of::<T>() as u128, limits)?;
492    let result = super::storage(count)?;
493    let peak = current + result.capacity() as u128 * size_of::<T>() as u128;
494    work.statistics.storage_peak_bytes = work.statistics.storage_peak_bytes.max(narrow(peak)?);
495    storage_limit(peak, limits)?;
496    Ok(result)
497}
498fn catalog(error: CatalogError) -> ErrorKind {
499    match error {
500        CatalogError::Storage { .. } => ErrorKind::Limit(Resource::StorageBytes),
501        CatalogError::Allocation => ErrorKind::Stopped(zetesis_cpu::Stop::Allocation),
502        other => ErrorKind::Catalog(other),
503    }
504}
505fn vocabulary(error: VocabularyFailure<ErrorKind>) -> ErrorKind {
506    match error {
507        VocabularyFailure::Storage(error) => catalog(error),
508        VocabularyFailure::Stopped(error) => error,
509    }
510}
511fn component(error: TemplateCatalogFailure<ErrorKind>) -> ErrorKind {
512    match error {
513        TemplateCatalogFailure::Storage(error) => catalog(error),
514        TemplateCatalogFailure::Read(error) => ErrorKind::Canonical(error),
515        TemplateCatalogFailure::Stopped(error) => error,
516        TemplateCatalogFailure::Incomplete => ErrorKind::Overflow,
517    }
518}
519fn component_never(error: TemplateCatalogFailure) -> ErrorKind {
520    match error {
521        TemplateCatalogFailure::Storage(error) => catalog(error),
522        TemplateCatalogFailure::Read(error) => ErrorKind::Canonical(error),
523        TemplateCatalogFailure::Stopped(never) => match never {},
524        TemplateCatalogFailure::Incomplete => ErrorKind::Overflow,
525    }
526}
527fn logical_payload(value: TermRef<'_>, work: &mut Work<'_>) -> Result<u64, ErrorKind> {
528    work.charge(1)?;
529    match value.descriptor() {
530        ValueNodeRef::String(text) | ValueNodeRef::Symbol(text) => narrow(text.len() as u128),
531        ValueNodeRef::Function { .. } | ValueNodeRef::Tuple { .. } => {
532            let nodes = value.expanded_nodes();
533            let mut result = bytes::<zetesis_core::ValueNode>(nodes)?;
534            result = add(result, narrow(value.rendered_bytes() as u128)?)?;
535            for index in 0..nodes {
536                let term = value
537                    .subterm_with(index, || work.charge(1))?
538                    .expect("admitted expanded rank");
539                work.charge(1)?;
540                let text = match term.descriptor() {
541                    ValueNodeRef::String(text)
542                    | ValueNodeRef::Symbol(text)
543                    | ValueNodeRef::Function { name: text, .. } => text.len(),
544                    _ => 0,
545                };
546                result = add(result, narrow(text as u128)?)?;
547            }
548            Ok(result)
549        }
550        _ => Ok(0),
551    }
552}
553pub(super) fn term_work(value: TermRef<'_>, work: &mut Work<'_>) -> Result<u64, ErrorKind> {
554    work.charge(1)?;
555    let count = match value.descriptor() {
556        ValueNodeRef::Function { .. } | ValueNodeRef::Tuple { .. } => {
557            value.canonical_bytes_with(|| work.charge(1))?
558        }
559        ValueNodeRef::String(text) | ValueNodeRef::Symbol(text) => {
560            text.len().checked_add(1).ok_or(ErrorKind::Overflow)?
561        }
562        _ => 1,
563    };
564    u64::try_from(count).map_err(|_| ErrorKind::Overflow)
565}
566fn unique(
567    group: GroupRef<'_>,
568    costs: &[u64],
569    current: u128,
570    limits: AdmissionLimits,
571    work: &mut Work<'_>,
572) -> Result<(), ErrorKind> {
573    let count = group.tuples().len();
574    let mut order = reserve(count, current, limits, work)?;
575    let mut buffer = reserve(
576        count,
577        current + order.capacity() as u128 * size_of::<usize>() as u128,
578        limits,
579        work,
580    )?;
581    for index in 0..count {
582        work.charge(2)?;
583        order.push(index);
584        buffer.push(index);
585    }
586    let mut width = 1_usize;
587    while width < count {
588        let mut start = 0;
589        while start < count {
590            let middle = start.saturating_add(width).min(count);
591            let end = middle.saturating_add(width).min(count);
592            merge(
593                group,
594                costs,
595                &order,
596                &mut buffer[start..end],
597                start..middle,
598                middle..end,
599                work,
600            )?;
601            start = end;
602        }
603        std::mem::swap(&mut order, &mut buffer);
604        width = width.saturating_mul(2);
605    }
606    for pair in order.windows(2) {
607        work.charge(1)?;
608        if compare(group, costs, pair[0], pair[1], work)?.is_eq() {
609            return Err(ErrorKind::Duplicate {
610                first: pair[0],
611                second: pair[1],
612            });
613        }
614    }
615    Ok(())
616}
617fn merge(
618    group: GroupRef<'_>,
619    costs: &[u64],
620    order: &[usize],
621    output: &mut [usize],
622    mut left: std::ops::Range<usize>,
623    mut right: std::ops::Range<usize>,
624    work: &mut Work<'_>,
625) -> Result<(), ErrorKind> {
626    for slot in output {
627        work.charge(1)?;
628        let take_left = right.is_empty()
629            || (!left.is_empty()
630                && !compare(group, costs, order[left.start], order[right.start], work)?.is_gt());
631        let next = if take_left { &mut left } else { &mut right };
632        *slot = order[next.start];
633        next.start += 1;
634    }
635    Ok(())
636}
637fn compare(
638    group: GroupRef<'_>,
639    costs: &[u64],
640    left: usize,
641    right: usize,
642    work: &mut Work<'_>,
643) -> Result<Ordering, ErrorKind> {
644    work.charge(add(costs[left], costs[right])?)?;
645    let left = group.tuples().at(left).expect("sort coordinate").key;
646    let right = group.tuples().at(right).expect("sort coordinate").key;
647    for index in 0..left.len().min(right.len()) {
648        work.charge(1)?;
649        let order = left
650            .at(index)
651            .expect("key extent")
652            .compare_terms_with(right.at(index).expect("key extent"), || work.charge(1))?;
653        if !order.is_eq() {
654            return Ok(order);
655        }
656    }
657    Ok(left.len().cmp(&right.len()))
658}
659fn ceiling(actual: usize, maximum: usize, resource: Resource) -> Result<(), ErrorKind> {
660    if actual > maximum {
661        Err(ErrorKind::Limit(resource))
662    } else {
663        Ok(())
664    }
