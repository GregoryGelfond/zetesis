11/// Bounds for formula-prefix evaluation and retained tuple occurrence masks.
12#[derive(Clone, Copy, Debug)]
13pub struct EligibilityLimits {
14    /// Original/frozen node and operand visits, numeric-cache initialization and
15    /// packed-validity resets, plus copied tuple observations.
16    pub max_work: u64,
17    /// Simultaneous node truth, numeric scratch and output masks; borrowed theory, group,
18    /// interpretations, stack and allocator metadata are excluded.
19    pub max_bytes: u64,
20}
21impl Default for EligibilityLimits {
22    fn default() -> Self {
23        Self {
24            max_work: 100_000_000,
25            max_bytes: 64 * 1024 * 1024,
26        }
27    }
28}

30/// Actual eligibility observations bound to this group and borrowed M/J pair.
31/// Constructed only by [`Group::eligibility`]; no aggregate guard is evaluated
32/// during acquisition and no field asserts stable-model membership.
33#[derive(Debug)]
34pub struct Eligibility<'a> {
35    group: GroupRef<'a>,
36    candidate: &'a Interpretation,
37    tested: Option<&'a Interpretation>,
38    original: Vec<bool>,
39    frozen: Option<Vec<bool>>,
40    statistics: Statistics,
41}
42
43impl<'a> Eligibility<'a> {
44    /// Exact retained operation whose ordered condition IDs supplied the masks.
45    #[must_use]
46    pub const fn group(&self) -> GroupRef<'a> {
47        self.group
48    }
49
50    /// Original interpretation M, with the group's immutable theory identity.
51    #[must_use]
52    pub const fn candidate(&self) -> &'a Interpretation {
53        self.candidate
54    }
55
56    /// Tested J if frozen eligibility was requested; no subset claim is made.
57    #[must_use]
58    pub const fn tested(&self) -> Option<&'a Interpretation> {
59        self.tested
60    }
61
62    /// Actual original eligibility in exact tuple occurrence order.
63    #[must_use]
64    pub fn original(&self) -> &[bool] {
65        &self.original
66    }
67
68    /// Actual J-truth of each eligibility formula frozen in M, if requested.
69    #[must_use]
70    pub fn frozen(&self) -> Option<&[bool]> {
71        self.frozen.as_deref()
72    }
73
74    /// Completed acquisition accounting, excluding subsequent aggregate reduction.
75    #[must_use]
76    pub const fn statistics(&self) -> Statistics {
77        self.statistics
78    }
79
80    /// Reduce the acquired masks and evaluate all group guards under a fresh,
81    /// explicit reduction budget. No formula-prefix evaluation is repeated.
82    ///
83    /// # Errors
84    /// Refuses reduction work, checked arithmetic or control.
85    pub fn reduce(
86        &self,
87        limits: ReductionLimits,
88        cancellation: &Cancellation,
89    ) -> Result<Reduction<'a>, Error> {
90        self.group
91            .reduce(&self.original, self.frozen.as_deref(), limits, cancellation)
92    }
93}

95impl Group {
96    /// Acquire original/frozen formula truth through this canonical group.
97    /// # Errors
98    /// Refuses theory identity, work, allocation, storage or caller control.
99    pub fn eligibility<'a>(
100        &'a self,
101        candidate: &'a Interpretation,
102        tested: Option<&'a Interpretation>,
103        limits: EligibilityLimits,
104        cancellation: &Cancellation,
105    ) -> Result<Eligibility<'a>, Error> {
106        self.view()
107            .eligibility(candidate, tested, limits, cancellation)
108    }
109}

111impl<'a> GroupRef<'a> {
112    /// Acquire actual original and optional frozen eligibility without lowering
113    /// this aggregate or changing the retained theory.
114    ///
115    /// Each requested phase evaluates the complete existing formula prefix, then
116    /// copies each tuple's condition value. Repeated condition IDs remain distinct
117    /// tuple occurrences. Frozen evaluation uses original node truth from M and
118    /// atom truth from J. J need not be a subset of M; minimality is a separate
119    /// solver obligation. All prefix work is included in this acquisition's
120    /// budget, separate from reduction and guard evaluation.
121    ///
122    /// # Errors
123    /// Refuses foreign M/J identity, storage, work, cancellation or deadline.
124    /// No partial observation record escapes an interrupted acquisition.
125    pub fn eligibility(
126        self,
127        candidate: &'a Interpretation,
128        tested: Option<&'a Interpretation>,
129        limits: EligibilityLimits,
130        cancellation: &Cancellation,
131    ) -> Result<Eligibility<'a>, Error> {
132        let mut statistics = Statistics::default();
133        let metered = cancellation.metered(limits.max_work, |meter| {
134            self.acquire(candidate, tested, limits, &mut statistics, meter)
135        });
136        statistics.work = metered.charged;
137        metered
138            .result
139            .map(|(original, frozen)| Eligibility {
140                group: self,
141                candidate,
142                tested,
143                original,
144                frozen,
145                statistics,
146            })
147            .map_err(|kind| Error { kind, statistics })
148    }
149
150    fn acquire(
151        &self,
152        candidate: &Interpretation,
153        tested: Option<&Interpretation>,
154        limits: EligibilityLimits,
155        statistics: &mut Statistics,
156        meter: &mut Meter<'_>,
157    ) -> Result<(Vec<bool>, Option<Vec<bool>>), ErrorKind> {
158        meter.poll().map_err(ErrorKind::Stopped)?;
159        if !self.data.theory.same_instance(candidate.theory())
160            || tested.is_some_and(|tested| !self.data.theory.same_instance(tested.theory()))
161        {
162            return Err(ErrorKind::WrongTheory);
163        }
164        let phases = if tested.is_some() { 2 } else { 1 };
165        statistics.resident_bytes = bytes::<bool>(self.data.tuples.len())?
166            .checked_mul(phases)
167            .ok_or(ErrorKind::Overflow)?;
168        statistics.peak_bytes = add(
169            statistics.resident_bytes,
170            bytes::<bool>(self.data.theory.nodes().len())?
171                .checked_mul(phases)
172                .ok_or(ErrorKind::Overflow)?,
173        )?;
174        statistics.peak_bytes = add(
175            statistics.peak_bytes,
176            u64::try_from(NumericRowScratch::default().required_bytes(&self.data.theory))
177                .map_err(|_| ErrorKind::Overflow)?,
178        )?;
179        if statistics.peak_bytes > limits.max_bytes {
180            return Err(ErrorKind::Limit(Resource::Bytes));
181        }
182        let mut work = oracle::Work::new(0, meter);
183        let observations = observations(
184            *self,
185            candidate,
186            tested,
187            limits.max_bytes,
188            statistics,
189            &mut work,
190        )?;
191        work.meter.poll().map_err(ErrorKind::Stopped)?;
192        Ok(observations)
193    }
194}

196/// The full requested envelope is checked before this owner reserves anything.
197/// Actual capacity is observed after each reservation on success and refusal.
198#[derive(Default)]
199struct Observations {
200    original_nodes: Vec<bool>,
201    frozen_nodes: Vec<bool>,
202    original: Vec<bool>,
203    frozen: Vec<bool>,
204    numeric: NumericRowScratch,
205}
206
207impl Observations {
208    fn bytes(&self) -> u128 {
209        self.original_nodes.capacity() as u128
210            + self.frozen_nodes.capacity() as u128
211            + self.original.capacity() as u128
212            + self.frozen.capacity() as u128
213            + self.numeric.retained_bytes()
214    }
215
216    fn observe(
217        &self,
218        reservation: Result<(), ErrorKind>,
219        limit: u64,
220        statistics: &mut Statistics,
221    ) -> Result<(), ErrorKind> {
222        observe_capacity(self.bytes(), reservation, limit, statistics)
223    }
224
225    fn reserve(
226        &mut self,
227        theory: &Theory,
228        tuples: usize,
229        frozen: bool,
230        limit: u64,
231        statistics: &mut Statistics,
232    ) -> Result<(), ErrorKind> {
233        let result = reserve_mask(&mut self.original_nodes, theory.nodes().len());
234        self.observe(result, limit, statistics)?;
235        let base = self.bytes() - self.numeric.retained_bytes();
236        self.numeric
237            .reserve_observed(theory, |actual| {
238                observe_capacity(base + actual, Ok(()), limit, statistics).map_err(Observed)
239            })
240            .map_err(|error| error.0)?;
241        let result = reserve_mask(&mut self.original, tuples);
242        self.observe(result, limit, statistics)?;
243        if frozen {
244            let result = reserve_mask(&mut self.frozen_nodes, theory.nodes().len());
245            self.observe(result, limit, statistics)?;
246            let result = reserve_mask(&mut self.frozen, tuples);
247            self.observe(result, limit, statistics)?;
248        }
249        Ok(())
250    }
251}

253// Adapt the native group's existing error vocabulary to generic scratch control.
254struct Observed(ErrorKind);
255impl From<Stop> for Observed {
256    fn from(stop: Stop) -> Self {
257        Self(stopped(stop))
258    }
259}
260
261fn observe_capacity(
262    bytes: u128,
263    reservation: Result<(), ErrorKind>,
264    limit: u64,
265    statistics: &mut Statistics,
266) -> Result<(), ErrorKind> {
267    let actual = u64::try_from(bytes).map_err(|_| ErrorKind::Overflow);
268    if let Ok(actual) = &actual {
269        statistics.storage_peak_bytes = statistics.storage_peak_bytes.max(*actual);
270    }
271    // Preserve the original allocation failure if observation also refuses.
272    reservation?;
273    if actual? > limit {
274        return Err(ErrorKind::Limit(Resource::Bytes));
275    }
276    Ok(())
277}
278
279fn reserve_mask(target: &mut Vec<bool>, count: usize) -> Result<(), ErrorKind> {
280    target
281        .try_reserve_exact(count)
282        .map_err(|_| ErrorKind::Stopped(Stop::Allocation))
283}

285fn observations(
286    group: GroupRef<'_>,
287    candidate: &Interpretation,
288    tested: Option<&Interpretation>,
289    limit: u64,
290    statistics: &mut Statistics,
291    work: &mut oracle::Work<'_, '_>,
292) -> Result<(Vec<bool>, Option<Vec<bool>>), ErrorKind> {
293    let mut buffers = Observations::default();
294    buffers.reserve(
295        &group.data.theory,
296        group.data.tuples.len(),
297        tested.is_some(),
298        limit,
299        statistics,
300    )?;
301    oracle::evaluate(
302        &group.data.theory,
303        candidate,
304        None,
305        &mut buffers.original_nodes,
306        &mut buffers.numeric,
307        work,
308    )
309    .map_err(stopped)?;
310    project(group, &buffers.original_nodes, &mut buffers.original, work)?;
311    if let Some(tested) = tested {
312        oracle::evaluate(
313            &group.data.theory,
314            tested,
315            Some(&buffers.original_nodes),
316            &mut buffers.frozen_nodes,
317            &mut buffers.numeric,
318            work,
319        )
320        .map_err(stopped)?;
321        project(group, &buffers.frozen_nodes, &mut buffers.frozen, work)?;
322    }
323    statistics.storage_bytes = add(
324        bytes::<bool>(buffers.original.capacity())?,
325        bytes::<bool>(buffers.frozen.capacity())?,
326    )?;
327    Ok((buffers.original, tested.map(|_| buffers.frozen)))
328}

330fn project(
331    group: GroupRef<'_>,
332    nodes: &[bool],
333    mask: &mut Vec<bool>,
334    work: &mut oracle::Work<'_, '_>,
335) -> Result<(), ErrorKind> {
336    for tuple in &group.data.tuples {
337        work.tick().map_err(stopped)?;
338        mask.push(nodes[tuple.condition]);
339    }
340    Ok(())
341}
342
343fn stopped(stop: Stop) -> ErrorKind {
344    match stop {
345        Stop::WorkLimit => ErrorKind::Limit(Resource::Work),
346        _ => ErrorKind::Stopped(stop),
347    }
