11/// Bounds for formula-prefix evaluation and retained tuple occurrence masks.
12#[derive(Clone, Copy, Debug)]
13pub struct EligibilityLimits {
14    /// Original/frozen node and operand visits, plus copied tuple observations.
15    pub max_work: u64,
16    /// Simultaneous node-truth scratch and output masks; borrowed theory, group,
17    /// interpretations, stack and allocator metadata are excluded.
18    pub max_bytes: u64,
19}
20impl Default for EligibilityLimits {
21    fn default() -> Self {
22        Self {
23            max_work: 100_000_000,
24            max_bytes: 64 * 1024 * 1024,
25        }
26    }
27}

29/// Actual eligibility observations bound to this group and borrowed M/J pair.
30/// Constructed only by [`Group::eligibility`]; no aggregate guard is evaluated
31/// during acquisition and no field asserts stable-model membership.
32#[derive(Debug)]
33pub struct Eligibility<'a> {
34    group: GroupRef<'a>,
35    candidate: &'a Interpretation,
36    tested: Option<&'a Interpretation>,
37    original: Vec<bool>,
38    frozen: Option<Vec<bool>>,
39    statistics: Statistics,
40}
41
42impl<'a> Eligibility<'a> {
43    /// Exact retained operation whose ordered condition IDs supplied the masks.
44    #[must_use]
45    pub const fn group(&self) -> GroupRef<'a> {
46        self.group
47    }
48
49    /// Original interpretation M, with the group's immutable theory identity.
50    #[must_use]
51    pub const fn candidate(&self) -> &'a Interpretation {
52        self.candidate
53    }
54
55    /// Tested J if frozen eligibility was requested; no subset claim is made.
56    #[must_use]
57    pub const fn tested(&self) -> Option<&'a Interpretation> {
58        self.tested
59    }
60
61    /// Actual original eligibility in exact tuple occurrence order.
62    #[must_use]
63    pub fn original(&self) -> &[bool] {
64        &self.original
65    }
66
67    /// Actual J-truth of each eligibility formula frozen in M, if requested.
68    #[must_use]
69    pub fn frozen(&self) -> Option<&[bool]> {
70        self.frozen.as_deref()
71    }
72
73    /// Completed acquisition accounting, excluding subsequent aggregate reduction.
74    #[must_use]
75    pub const fn statistics(&self) -> Statistics {
76        self.statistics
77    }
78
79    /// Reduce the acquired masks and evaluate all group guards under a fresh,
80    /// explicit reduction budget. No formula-prefix evaluation is repeated.
81    ///
82    /// # Errors
83    /// Refuses reduction work, checked arithmetic or control.
84    pub fn reduce(
85        &self,
86        limits: ReductionLimits,
87        cancellation: &Cancellation,
88    ) -> Result<Reduction<'a>, Error> {
89        self.group
90            .reduce(&self.original, self.frozen.as_deref(), limits, cancellation)
91    }
92}

94impl Group {
95    /// Acquire original/frozen formula truth through this canonical group.
96    /// # Errors
97    /// Refuses theory identity, work, allocation, storage or caller control.
98    pub fn eligibility<'a>(
99        &'a self,
100        candidate: &'a Interpretation,
101        tested: Option<&'a Interpretation>,
102        limits: EligibilityLimits,
103        cancellation: &Cancellation,
104    ) -> Result<Eligibility<'a>, Error> {
105        self.view()
106            .eligibility(candidate, tested, limits, cancellation)
107    }

110impl<'a> GroupRef<'a> {
111    /// Acquire actual original and optional frozen eligibility without lowering
112    /// this aggregate or changing the retained theory.
113    ///
114    /// Each requested phase evaluates the complete existing formula prefix, then
115    /// copies each tuple's condition value. Repeated condition IDs remain distinct
116    /// tuple occurrences. Frozen evaluation uses original node truth from M and
117    /// atom truth from J. J need not be a subset of M; minimality is a separate
118    /// solver obligation. All prefix work is included in this acquisition's
119    /// budget, separate from reduction and guard evaluation.
120    ///
121    /// # Errors
122    /// Refuses foreign M/J identity, storage, work, cancellation or deadline.
123    /// No partial observation record escapes an interrupted acquisition.
124    pub fn eligibility(
125        self,
126        candidate: &'a Interpretation,
127        tested: Option<&'a Interpretation>,
128        limits: EligibilityLimits,
129        cancellation: &Cancellation,
130    ) -> Result<Eligibility<'a>, Error> {
131        let mut statistics = Statistics::default();
132        let result = self.acquire(candidate, tested, limits, cancellation, &mut statistics);
133        result
134            .map(|(original, frozen)| Eligibility {
135                group: self,
136                candidate,
137                tested,
138                original,
139                frozen,
140                statistics,
141            })
142            .map_err(|kind| Error { kind, statistics })
143    }
144
145    fn acquire(
146        &self,
147        candidate: &Interpretation,
148        tested: Option<&Interpretation>,
149        limits: EligibilityLimits,
150        cancellation: &Cancellation,
151        statistics: &mut Statistics,
152    ) -> Result<(Vec<bool>, Option<Vec<bool>>), ErrorKind> {
153        cancellation.poll().map_err(ErrorKind::Stopped)?;
154        if !self.data.theory.same_instance(candidate.theory())
155            || tested.is_some_and(|tested| !self.data.theory.same_instance(tested.theory()))
156        {
157            return Err(ErrorKind::WrongTheory);
158        }
159        let phases = if tested.is_some() { 2 } else { 1 };
160        statistics.resident_bytes = bytes::<bool>(self.data.tuples.len())?
161            .checked_mul(phases)
162            .ok_or(ErrorKind::Overflow)?;
163        statistics.peak_bytes = add(
164            statistics.resident_bytes,
165            bytes::<bool>(self.data.theory.nodes().len())?
166                .checked_mul(phases)
167                .ok_or(ErrorKind::Overflow)?,
168        )?;
169        if statistics.peak_bytes > limits.max_bytes {
170            return Err(ErrorKind::Limit(Resource::Bytes));
171        }
172        let mut work = oracle::Work {
173            limits: oracle::Limits {
174                max_work: limits.max_work,
175                max_subsets: 0,
176            },
177            cancellation,
178            statistics: oracle::Statistics::default(),
179        };
180        let result = observations(*self, candidate, tested, &mut work);
181        statistics.work = work.statistics.work;
182        let observations = result?;
183        cancellation.poll().map_err(ErrorKind::Stopped)?;
184        Ok(observations)
185    }
186}

188fn observations(
189    group: GroupRef<'_>,
190    candidate: &Interpretation,
191    tested: Option<&Interpretation>,
192    work: &mut oracle::Work<'_>,
193) -> Result<(Vec<bool>, Option<Vec<bool>>), ErrorKind> {
194    let mut original_nodes = storage(group.data.theory.nodes().len())?;
195    oracle::evaluate(
196        &group.data.theory,
197        candidate,
198        None,
199        &mut original_nodes,
200        work,
201    )
202    .map_err(stopped)?;
203    let original = project(group, &original_nodes, work)?;
204    let frozen = if let Some(tested) = tested {
205        let mut frozen_nodes = storage(group.data.theory.nodes().len())?;
206        oracle::evaluate(
207            &group.data.theory,
208            tested,
209            Some(&original_nodes),
210            &mut frozen_nodes,
211            work,
212        )
213        .map_err(stopped)?;
214        Some(project(group, &frozen_nodes, work)?)
215    } else {
216        None
217    };
218    Ok((original, frozen))
219}
220
221fn project(
222    group: GroupRef<'_>,
223    nodes: &[bool],
224    work: &mut oracle::Work<'_>,
225) -> Result<Vec<bool>, ErrorKind> {
226    let mut mask = storage(group.data.tuples.len())?;
227    for tuple in &group.data.tuples {
228        work.tick().map_err(stopped)?;
229        mask.push(nodes[tuple.condition]);
230    }
231    Ok(mask)
232}
233
234fn stopped(stop: Stop) -> ErrorKind {
235    match stop {
236        Stop::WorkLimit => ErrorKind::Limit(Resource::Work),
237        _ => ErrorKind::Stopped(stop),
238    }
