14/// Owned formula nodes with reusable, completed aggregate-prefix checks.
15///
16/// Construction and [`Self::push`] accept unchecked nodes. Aggregate compilation
17/// validates backward edges in the unchecked suffix before using any condition;
18/// atom-universe and root admission remain [`crate::Theory::new`]'s responsibility.
19/// The buffer owns one vector and a scan frontier. Read-only slice access cannot
20/// change checked nodes, and removing a suffix clamps the frontier to what remains.
21/// Extracting the vector discards this evidence; wrapping it again starts unchecked.
22///
23/// The compiler methods have the corresponding free functions' reduct semantics,
24/// input obligations, node ceilings and transactional failure behavior. They save
25/// one work unit per previously checked node. Each call still charges its initial
26/// control poll/node-ceiling check, scans every unchecked node, and charges all
27/// aggregate-specific work. A scan publishes its frontier only on completion;
28/// that completed check can survive a later compilation failure. Appended compiler
29/// nodes are checked on the next call, with no privileged topology assumption.
30/// Refusals retain their actual work and restore the original nodes and length;
31/// reserved vector capacity may change.
32///
33/// The ordinary vector editing operations below have vector allocation behavior
34/// and do not consume an aggregate budget. Compiler appends remain fallible and
35/// metered. No mutable node access is exposed:
36///
37/// ```compile_fail
38/// use zetesis_ferraris::{FormulaNodes, Node};
39/// let mut nodes = FormulaNodes::new(vec![Node::False]);
40/// nodes[0] = Node::And(0, 0);
41/// ```
42#[derive(Debug, Default)]
43pub struct FormulaNodes {
44    nodes: Vec<Node>,
45    validated: usize,
46}
47
48impl FormulaNodes {
49    /// Take ownership without scanning or copying the node vector.
50    #[must_use]
51    pub const fn new(nodes: Vec<Node>) -> Self {
52        Self {
53            nodes,
54            validated: 0,
55        }
56    }
57
58    /// Allocated node capacity, excluding this owner's inline scan frontier.
59    #[must_use]
60    pub fn capacity(&self) -> usize {
61        self.nodes.capacity()
62    }
63
64    /// Append one unchecked node, preserving the unchanged checked prefix.
65    pub fn push(&mut self, node: Node) {
66        self.nodes.push(node);
67    }
68
69    /// Remove nodes after `len`, discarding their validation evidence.
70    pub fn truncate(&mut self, len: usize) {
71        self.nodes.truncate(len);
72        self.validated = self.validated.min(self.nodes.len());
73    }
74
75    /// Move the suffix out, retaining evidence only for the unchanged prefix.
76    /// The returned nodes retain their original absolute child indices.
77    ///
78    /// # Panics
79    /// Panics if `at` exceeds the current length, as [`Vec::split_off`] does.
80    #[must_use]
81    pub fn split_off(&mut self, at: usize) -> Vec<Node> {
82        let suffix = self.nodes.split_off(at);
83        self.validated = self.validated.min(at);
84        suffix
85    }
86
87    /// Return the vector without copying, discarding its validation frontier.
88    #[must_use]
89    pub fn into_vec(self) -> Vec<Node> {
90        self.nodes
91    }
92
93    /// Append the scalar formula of [`super::append_aggregate`], reusing checked
94    /// prefix nodes under this owner's exclusive mutation boundary.
95    ///
96    /// # Errors
97    /// Returns the free function's typed failures with actual work accounting.
98    pub fn append_aggregate(
99        &mut self,
100        elements: &[AggregateElement],
101        comparison: AggregateComparison,
102        bound: i64,
103        limits: AggregateLimits,
104        cancellation: &Cancellation,
105    ) -> Result<AggregateBuild, AggregateError> {
106        lower::append(
107            self.destination(),
108            elements,
109            comparison,
110            bound,
111            limits,
112            cancellation,
113        )
114    }
115
116    /// Append the ordered roots of [`super::append_aggregate_family`], reusing
117    /// completed prefix validation across independent families.
118    ///
119    /// # Errors
120    /// Returns the free function's typed failures. Its limits remain cumulative
121    /// across this family; completed earlier calls do not spend this call's budget.
122    pub fn append_aggregate_family(
123        &mut self,
124        elements: &[AggregateElement],
125        guards: &[AggregateGuard],
126        limits: AggregateFamilyLimits,
127        cancellation: &Cancellation,
128    ) -> Result<AggregateFamilyBuild, AggregateError> {
129        family::append(self.destination(), elements, guards, limits, cancellation)
130    }
131
132    /// Append the numeric extremum of [`super::append_extremum`], reusing
133    /// completed prefix validation.
134    ///
135    /// # Errors
136    /// Returns the free function's typed failures with actual work accounting.
137    pub fn append_extremum(
138        &mut self,
139        elements: &[AggregateElement],
140        extremum: AggregateExtremum,
141        comparison: AggregateComparison,
142        bound: impl Into<ExtremumBound>,
143        limits: AggregateLimits,
144        cancellation: &Cancellation,
145    ) -> Result<AggregateBuild, AggregateError> {
146        extremum::append(
147            self.destination(),
148            elements,
149            extremum,
150            comparison,
151            bound.into(),
152            limits,
153            cancellation,
154        )
155    }
156
157    /// Append the typed extremum of [`super::append_value_extremum`], reusing
158    /// completed prefix validation.
159    ///
160    /// # Errors
161    /// Returns the free function's typed failures with actual work accounting.
162    pub fn append_value_extremum(
163        &mut self,
164        elements: &[ValueExtremumElement],
165        extremum: AggregateExtremum,
166        comparison: AggregateComparison,
167        bound: &Value,
168        limits: AggregateLimits,
169        cancellation: &Cancellation,
170    ) -> Result<AggregateBuild, AggregateError> {
171        self.append_value_extremum_refs(
172            elements.iter().map(|element| ValueExtremumElement {
173                value: (&element.value).into(),
174                condition: element.condition,
175            }),
176            extremum,
177            comparison,
178            bound.into(),
179            limits,
180            cancellation,
181        )
182    }
183
184    /// Append the borrowed typed extremum of [`super::append_value_extremum_refs`].
185    /// Canonical terms keep their original authority; only node-prefix validation
186    /// is reused, with every typed comparison still charged.
187    ///
188    /// # Errors
189    /// Returns the free function's typed failures with actual work accounting.
190    pub fn append_value_extremum_refs<'a>(
191        &mut self,
192        elements: impl Iterator<Item = ValueExtremumElement<TermRef<'a>>>,
193        extremum: AggregateExtremum,
194        comparison: AggregateComparison,
195        bound: TermRef<'_>,
196        limits: AggregateLimits,
197        cancellation: &Cancellation,
198    ) -> Result<AggregateBuild, AggregateError> {
199        value_extremum::append(
200            self.destination(),
201            elements,
202            extremum,
203            comparison,
204            bound,
205            limits,
206            cancellation,
207        )
208    }
209
210    fn destination(&mut self) -> Destination<'_> {
211        Destination::retained(&mut self.nodes, &mut self.validated)
212    }
213}
