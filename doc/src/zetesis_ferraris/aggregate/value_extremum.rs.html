12/// One complete tuple's first value and OR-coalesced eligibility formula.
13/// Equal first values do not identify equal complete tuples.
14#[derive(Clone, Debug, PartialEq, Eq)]
15pub struct ValueExtremumElement<V = Value> {
16    /// Complete first tuple value, compared using ASP term order.
17    pub value: V,
18    /// Absolute condition index in the existing DAG prefix.
19    pub condition: usize,
20}
21
22/// Append an exact min/max comparison over complete ASP values.
23///
24/// The caller groups equal complete tuples by disjoining eligibility. Empty
25/// minimum is `#sup`; empty maximum is `#inf`. Explicit sentinel elements are
26/// allowed. Default negation is applied to the resulting formula by the caller.
27/// This uses the numeric path's reduct-preserving witness connectives, including
28/// implication for not-equal, and introduces no semantic atoms. All appends are
29/// transactional; work, element and node limits and cancellation remain explicit.
30/// Structural comparisons charge their visited term components.
31/// No threshold states or subsets are allocated.
32///
33/// # Errors
34/// Returns typed prefix, condition, resource, allocation or cancellation errors.
35pub fn append_value_extremum(
36    nodes: &mut Vec<Node>,
37    elements: &[ValueExtremumElement],
38    extremum: AggregateExtremum,
39    comparison: AggregateComparison,
40    bound: &Value,
41    limits: AggregateLimits,
42    cancellation: &Cancellation,
43) -> Result<AggregateBuild, AggregateError> {
44    append_value_extremum_refs(
45        nodes,
46        elements.iter().map(|element| ValueExtremumElement {
47            value: (&element.value).into(),
48            condition: element.condition,
49        }),
50        extremum,
51        comparison,
52        bound.into(),
53        limits,
54        cancellation,
55    )
56}

58/// Append the same reduct-preserving extremum comparison over borrowed values.
59///
60/// One traversal checks every actual condition against the original DAG prefix
61/// before using it. Borrowed canonical terms share their original authority;
62/// no typed payload is materialized. Prefix validation, element visits and typed
63/// comparison steps consume the aggregate work allowance.
64///
65/// # Errors
66/// Returns the same prefix, condition, resource, allocation and cancellation
67/// failures as [`append_value_extremum`], preserving the original DAG on refusal.
68pub fn append_value_extremum_refs<'a>(
69    nodes: &mut Vec<Node>,
70    elements: impl Iterator<Item = ValueExtremumElement<TermRef<'a>>>,
71    extremum: AggregateExtremum,
72    comparison: AggregateComparison,
73    bound: TermRef<'_>,
74    limits: AggregateLimits,
75    cancellation: &Cancellation,
76) -> Result<AggregateBuild, AggregateError> {
77    transaction(nodes, limits, cancellation, |builder| {
78        let prefix = validate_prefix(builder)?;
79        let falsum = builder.push(Node::False)?;
80        let truth = builder.push(Node::Implies(falsum, falsum))?;
81        let empty = match extremum {
82            AggregateExtremum::Min => Value::Supremum,
83            AggregateExtremum::Max => Value::Infimum,
84        };
85        let mut inclusive = (bound == TermRef::from(&empty)).then_some(truth);
86        let mut strict = None;
87        for (index, element) in elements.enumerate() {
88            builder.tick()?;
89            if index >= limits.max_elements {
90                return Err(AggregateErrorKind::ElementLimit);
91            }
92            if element.condition >= prefix {
93                return Err(AggregateErrorKind::InvalidCondition { element: index });
94            }
95            let order = element.value.compare_terms_with(bound, || builder.tick())?;
96            let order = if extremum == AggregateExtremum::Min {
97                order.reverse()
98            } else {
99                order
100            };
101            if order.is_ge() {
102                inclusive = Some(builder.join(inclusive, element.condition, false)?);
103            }
104            if order.is_gt() {
105                strict = Some(builder.join(strict, element.condition, false)?);
106            }
107        }
108        comparison_root(
109            builder,
110            extremum,
111            comparison,
112            inclusive.unwrap_or(falsum),
113            strict.unwrap_or(falsum),
114            falsum,
115        )
116        .map(|root| (root, AggregateProfile::Extremum))
117    })
118}
