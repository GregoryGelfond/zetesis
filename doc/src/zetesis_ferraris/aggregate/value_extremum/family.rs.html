24/// One typed-family outcome and its actual auxiliary capacity receipt.
25///
26/// Auxiliary bytes describe family preparation and returned roots separately
27/// from graph storage and the aggregate's logical work statistics.
28#[derive(Debug)]
29#[must_use]
30pub struct ValueExtremumFamilyAttempt {
31    /// Ordered roots on success, or the original typed failure with completed
32    /// work accounting. A failed result has rolled back the entire graph suffix.
33    pub result: Result<AggregateFamilyBuild, AggregateError>,
34    /// Peak actual scratch/output vector bytes, including all vector headers
35    /// and capacity from failed reservations. Zero means no auxiliary allocation
36    /// was attempted; a zero-capacity attempt still reports its vector headers.
37    /// Borrowed inputs and DAG storage belong to separate owners and are excluded.
38    pub auxiliary_peak_bytes: u128,
39}

41/// Checked requested auxiliary storage for one typed extremum family.
42///
43/// Preparation owns dictionary indexes (E + 2), conditions (E), u64 ranks (E),
44/// and two cut rules per guard (2G). These cells share the state ceiling; root
45/// cells obey the separate guard ceiling. The dictionary depends only on the
46/// value row, includes both endpoints and preserves full ASP term order.
47#[derive(Clone, Copy, Debug, PartialEq, Eq)]
48pub struct ValueExtremumFamilySpace {
49    states: usize,
50    bytes: usize,
51}
52impl ValueExtremumFamilySpace {
53    /// Checked requested peak before selecting shared preparation. Includes all
54    /// five vector headers and requested capacities, excluding borrowed input
55    /// and graph storage. Empty guard families allocate nothing. Allocator excess
56    /// is reported by [`ValueExtremumFamilyAttempt::auxiliary_peak_bytes`].
57    #[must_use]
58    pub fn new(elements: usize, guards: usize) -> Option<Self> {
59        if guards == 0 {
60            return Some(Self {
61                states: 0,
62                bytes: 0,
63            });
64        }
65        let dictionary = elements.checked_add(2)?;
66        let cuts = guards.checked_mul(2)?;
67        let states = dictionary
68            .checked_add(elements.checked_mul(2)?)?
69            .checked_add(cuts)?;
70        let bytes = dictionary
71            .checked_add(elements)?
72            .checked_add(guards)?
73            .checked_mul(size_of::<usize>())?
74            .checked_add(elements.checked_mul(size_of::<u64>())?)?
75            .checked_add(cuts.checked_mul(size_of::<ExtremumCutRule>())?)?
76            .checked_add(size_of::<Preparation>())?;
77        Some(Self { states, bytes })
78    }
79
80    /// Peak temporary cells charged against `AggregateLimits::max_states`.
81    #[must_use]
82    pub const fn states(self) -> usize {
83        self.states
84    }
85
86    /// Requested auxiliary bytes, including roots while preparation is live.
87    #[must_use]
88    pub const fn storage_bytes(self) -> usize {
89        self.bytes
90    }
91}

93/// Append typed min/max guards over one shared, guard-independent ordinal row.
94///
95/// The complete value dictionary includes #inf and #sup and is sorted using
96/// charged ASP term comparisons. Equal dictionary values share a rank; every
97/// complete tuple occurrence retains its condition and rank, including repeated
98/// condition IDs. Two native inclusive/strict cuts are published per guard.
99/// Not-equal remains the inclusive-to-strict implication, and each cut and
100/// wrapper retains its own frozen truth gate for arbitrary inner worlds.
101///
102/// The caller coalesces equal complete tuples and establishes row independence
103/// from the changing guard. Borrowed term authorities stay unchanged. Preparation
104/// checks and reuses nondecreasing input order in O(E + G log E) comparisons;
105/// otherwise its existing in-place sort uses O(E log E + G log E). Both use
106/// O(E + G) auxiliary cells. Guard cut pairs share one dictionary search. All
107/// conditions are validated even with no guards. Work, state, guard and graph
108/// ceilings apply to the whole family; refusal rolls back the entire suffix.
109/// Scalar extrema retain their separate zero-state compilation contract.
110///
111/// # Errors
112/// Returns typed input, guard, state, allocation, work, node, operand, arithmetic
113/// or control failures, with actual auxiliary capacity on every outcome.
114pub fn append_value_extremum_family_refs(
115    nodes: &mut FormulaNodes,
116    elements: &[ValueExtremumElement<TermRef<'_>>],
117    extremum: AggregateExtremum,
118    guards: &[ValueExtremumGuard<TermRef<'_>>],
119    limits: AggregateFamilyLimits,
120    cancellation: &Cancellation,
121) -> ValueExtremumFamilyAttempt {
122    append(
123        Destination::retained(nodes.transaction()),
124        elements,
125        extremum,
126        guards,
127        limits,
128        cancellation,
129    )
130}

132pub(in crate::aggregate) fn append(
133    destination: Destination<'_>,
134    elements: &[ValueExtremumElement<TermRef<'_>>],
135    extremum: AggregateExtremum,
136    guards: &[ValueExtremumGuard<TermRef<'_>>],
137    limits: AggregateFamilyLimits,
138    cancellation: &Cancellation,
139) -> ValueExtremumFamilyAttempt {
140    let mut auxiliary_peak_bytes = 0;
141    let result = transaction_value(destination, limits.aggregate, cancellation, |builder| {
142        compile(
143            builder,
144            elements,
145            extremum,
146            guards,
147            limits,
148            &mut auxiliary_peak_bytes,
149        )
150    })
151    .map(|(roots, statistics)| {
152        AggregateFamilyBuild::new(roots, AggregateProfile::Extremum, statistics)
153    });
154    ValueExtremumFamilyAttempt {
155        result,
156        auxiliary_peak_bytes,
157    }
158}

160fn compile(
161    builder: &mut Builder<'_, '_>,
162    elements: &[ValueExtremumElement<TermRef<'_>>],
163    extremum: AggregateExtremum,
164    guards: &[ValueExtremumGuard<TermRef<'_>>],
165    limits: AggregateFamilyLimits,
166    auxiliary_peak_bytes: &mut u128,
167) -> Result<Vec<usize>, AggregateErrorKind> {
168    let prefix = validate_prefix(builder)?;
169    for (index, element) in elements.iter().enumerate() {
170        builder.tick()?;
171        if index >= limits.aggregate.max_elements {
172            return Err(AggregateErrorKind::ElementLimit);
173        }
174        if element.condition >= prefix {
175            return Err(AggregateErrorKind::InvalidCondition { element: index });
176        }
177    }
178    if guards.len() > limits.max_guards {
179        return Err(AggregateErrorKind::GuardLimit);
180    }
181    if guards.is_empty() {
182        return Ok(Vec::new());
183    }
184    let space = ValueExtremumFamilySpace::new(elements.len(), guards.len())
185        .ok_or(AggregateErrorKind::ArithmeticOverflow)?;
186    builder.states(space.states())?;
187    let mut preparation = Preparation::default();
188    preparation.reserve(builder, elements.len(), guards.len(), auxiliary_peak_bytes)?;
189    let rank_count = preparation.classify(builder, elements, extremum, guards)?;
190    let cut_nodes = builder.extremum_family(
191        ExtremumRow {
192            conditions: &preparation.conditions,
193            ranks: &preparation.ranks,
194            rank_count,
195        },
196        &preparation.cuts,
197    )?;
198    let falsum = builder.push(NodeView::False)?;
199    for (index, guard) in guards.iter().enumerate() {
200        builder.tick()?;
201        let inclusive = cut_nodes.start + index * 2;
202        let root = comparison_root(
203            builder,
204            extremum,
205            guard.comparison,
206            inclusive,
207            inclusive + 1,
208            falsum,
209        )?;
210        builder.tick()?;
211        preparation.roots.push(root);
212    }
213    Ok(preparation.roots)
214}

216#[derive(Default)]
217struct Preparation {
218    dictionary: Vec<usize>,
219    conditions: Vec<usize>,
220    ranks: Vec<u64>,
221    cuts: Vec<ExtremumCutRule>,
222    roots: Vec<usize>,
223}
224impl Preparation {
225    fn classify(
226        &mut self,
227        builder: &mut Builder<'_, '_>,
228        elements: &[ValueExtremumElement<TermRef<'_>>],
229        extremum: AggregateExtremum,
230        guards: &[ValueExtremumGuard<TermRef<'_>>],
231    ) -> Result<u64, AggregateErrorKind> {
232        for index in 0..elements.len() + 2 {
233            builder.tick()?;
234            self.dictionary.push(index);
235        }
236        for element in elements {
237            builder.tick()?;
238            self.conditions.push(element.condition);
239            builder.tick()?;
240            self.ranks.push(0);
241        }
242        let distinct = ranked::prepare(builder, &mut self.dictionary, &mut self.ranks, elements)?;
243        for guard in guards {
244            let (lower, upper) =
245                ranked::cuts(builder, guard.bound, &self.dictionary[..distinct], elements)?;
246            let (inclusive, strict) = match extremum {
247                AggregateExtremum::Min => (
248                    ExtremumCutRule::Prefix(upper),
249                    ExtremumCutRule::Prefix(lower),
250                ),
251                AggregateExtremum::Max => (
252                    ExtremumCutRule::Suffix(lower),
253                    ExtremumCutRule::Suffix(upper),
254                ),
255            };
256            builder.tick()?;
257            self.cuts.push(inclusive);
258            builder.tick()?;
259            self.cuts.push(strict);
260        }
261        u64::try_from(distinct).map_err(|_| AggregateErrorKind::ArithmeticOverflow)
262    }
263
264    fn reserve(
265        &mut self,
266        builder: &mut Builder<'_, '_>,
267        elements: usize,
268        guards: usize,
269        peak: &mut u128,
270    ) -> Result<(), AggregateErrorKind> {
271        let reserved = self.dictionary.try_reserve_exact(elements + 2);
272        self.observe(builder, peak, reserved)?;
273        let reserved = self.conditions.try_reserve_exact(elements);
274        self.observe(builder, peak, reserved)?;
275        let reserved = self.ranks.try_reserve_exact(elements);
276        self.observe(builder, peak, reserved)?;
277        let reserved = self.cuts.try_reserve_exact(guards * 2);
278        self.observe(builder, peak, reserved)?;
279        let reserved = self.roots.try_reserve_exact(guards);
280        self.observe(builder, peak, reserved)
281    }
282
283    fn observe(
284        &self,
285        builder: &mut Builder<'_, '_>,
286        peak: &mut u128,
287        reserved: Result<(), std::collections::TryReserveError>,
288    ) -> Result<(), AggregateErrorKind> {
289        let bytes = size_of::<Self>() as u128
290            + (self.dictionary.capacity() as u128
291                + self.conditions.capacity() as u128
292                + self.roots.capacity() as u128)
293                * size_of::<usize>() as u128
294            + self.ranks.capacity() as u128 * size_of::<u64>() as u128
295            + self.cuts.capacity() as u128 * size_of::<ExtremumCutRule>() as u128;
296        *peak = (*peak).max(bytes);
297        let states = self.dictionary.capacity() as u128
298            + self.conditions.capacity() as u128
299            + self.ranks.capacity() as u128
300            + self.cuts.capacity() as u128;
301        builder.allocated_states(usize::try_from(states).unwrap_or(usize::MAX));
302        reserved.map_err(|_| AggregateErrorKind::Allocation)?;
303        let states = usize::try_from(states).map_err(|_| AggregateErrorKind::StateLimit)?;
304        builder.states(states)
305    }
306}
