9type Result<T> = std::result::Result<T, AggregateErrorKind>;

11static INFIMUM: Value = Value::Infimum;
12static SUPREMUM: Value = Value::Supremum;

14/// Sort occurrence indexes plus two virtual endpoints, compact equal values in
15/// the dictionary only, and assign every original occurrence its complete rank.
16/// The caller supplies initialized index/rank cells and retains both buffers.
17pub(super) fn prepare(
18    builder: &mut Builder<'_, '_>,
19    order: &mut [usize],
20    ranks: &mut [u64],
21    elements: &[ValueExtremumElement<TermRef<'_>>],
22) -> Result<usize> {
23    if let Some(distinct) = ordered_ranks(builder, ranks, elements)? {
24        compact_ordered(builder, order, ranks, elements.len(), distinct)?;
25        return Ok(distinct);
26    }
27    sort(builder, order, elements)?;
28    let mut distinct = 0;
29    for position in 0..order.len() {
30        builder.tick()?;
31        let index = order[position];
32        if distinct == 0
33            || !compare(
34                builder,
35                value(order[distinct - 1], elements),
36                value(index, elements),
37            )?
38            .is_eq()
39        {
40            builder.tick()?;
41            order[distinct] = index;
42            distinct += 1;
43        }
44        if index < elements.len() {
45            let rank =
46                u64::try_from(distinct - 1).map_err(|_| AggregateErrorKind::ArithmeticOverflow)?;
47            builder.tick()?;
48            ranks[index] = rank;
49        }
50    }
51    Ok(distinct)
52}

54fn value<'a>(index: usize, elements: &[ValueExtremumElement<TermRef<'a>>]) -> TermRef<'a> {
55    match index.cmp(&elements.len()) {
56        Ordering::Less => elements[index].value,
57        Ordering::Equal => (&INFIMUM).into(),
58        Ordering::Greater => (&SUPREMUM).into(),
59    }
60}

62fn compare(
63    builder: &mut Builder<'_, '_>,
64    left: TermRef<'_>,
65    right: TermRef<'_>,
66) -> Result<Ordering> {
67    builder.tick()?;
68    left.compare_terms_with(right, || builder.tick())
69}

71/// Check ASP term order while assigning complete ranks. Numeric contribution
72/// order already agrees; mixed storage-identity order can differ. The public
73/// family API verifies every comparison rather than trusting the input order.
74/// An inversion leaves the dictionary untouched and its incomplete ranks are
75/// overwritten by the ordinary sorting path before anything is published.
76fn ordered_ranks(
77    builder: &mut Builder<'_, '_>,
78    ranks: &mut [u64],
79    elements: &[ValueExtremumElement<TermRef<'_>>],
80) -> Result<Option<usize>> {
81    let mut previous = (&INFIMUM).into();
82    let mut distinct = 1usize;
83    for (rank, element) in ranks.iter_mut().zip(elements) {
84        match compare(builder, previous, element.value)? {
85            Ordering::Greater => return Ok(None),
86            Ordering::Less => distinct += 1,
87            Ordering::Equal => {}
88        }
89        builder.tick()?;
90        *rank = u64::try_from(distinct - 1).map_err(|_| AggregateErrorKind::ArithmeticOverflow)?;
91        previous = element.value;
92    }
93    if !compare(builder, previous, (&SUPREMUM).into())?.is_eq() {
94        distinct += 1;
95    }
96    Ok(Some(distinct))
97}

99/// Reuse the checked ranks to compact the dictionary without comparing typed
100/// values again. Rank zero and the last rank include the virtual endpoints;
101/// repeated first values still retain every original condition occurrence.
102fn compact_ordered(
103    builder: &mut Builder<'_, '_>,
104    order: &mut [usize],
105    ranks: &[u64],
106    elements: usize,
107    distinct: usize,
108) -> Result<()> {
109    builder.tick()?;
110    order[0] = elements;
111    let mut retained = 1usize;
112    for (index, &rank) in ranks.iter().enumerate() {
113        builder.tick()?;
114        if rank == u64::try_from(retained).map_err(|_| AggregateErrorKind::ArithmeticOverflow)? {
115            builder.tick()?;
116            order[retained] = index;
117            retained += 1;
118        }
119    }
120    if retained < distinct {
121        builder.tick()?;
122        order[retained] = elements + 1;
123    }
124    Ok(())
125}

127/// Both insertion points in one search of the unique complete dictionary.
128/// Equality has exactly one position; an absent bound has identical lower and
129/// upper positions. Guard bounds keep their own typed authority and need not
130/// occur in the row.
131pub(super) fn cuts(
132    builder: &mut Builder<'_, '_>,
133    bound: TermRef<'_>,
134    order: &[usize],
135    elements: &[ValueExtremumElement<TermRef<'_>>],
136) -> Result<(u64, u64)> {
137    let mut low = 0;
138    let mut high = order.len();
139    while low < high {
140        let middle = low + (high - low) / 2;
141        match compare(builder, value(order[middle], elements), bound)? {
142            Ordering::Less => low = middle + 1,
143            Ordering::Greater => high = middle,
144            Ordering::Equal => {
145                let lower =
146                    u64::try_from(middle).map_err(|_| AggregateErrorKind::ArithmeticOverflow)?;
147                return Ok((lower, lower + 1));
148            }
149        }
150    }
151    let boundary = u64::try_from(low).map_err(|_| AggregateErrorKind::ArithmeticOverflow)?;
152    Ok((boundary, boundary))
153}

155/// In-place heap sorting charges every fallible comparison and swap before its
156/// mutation; an infallible comparator never conceals control/work failures.
157fn sort(
158    builder: &mut Builder<'_, '_>,
159    order: &mut [usize],
160    elements: &[ValueExtremumElement<TermRef<'_>>],
161) -> Result<()> {
162    for root in (0..order.len() / 2).rev() {
163        builder.tick()?;
164        sift(builder, order, root, elements)?;
165    }
166    for end in (1..order.len()).rev() {
167        builder.tick()?;
168        order.swap(0, end);
169        sift(builder, &mut order[..end], 0, elements)?;
170    }
171    Ok(())
172}

174fn sift(
175    builder: &mut Builder<'_, '_>,
176    order: &mut [usize],
177    mut root: usize,
178    elements: &[ValueExtremumElement<TermRef<'_>>],
179) -> Result<()> {
180    while root < order.len() / 2 {
181        builder.tick()?;
182        let mut child = root * 2 + 1;
183        if child + 1 < order.len()
184            && compare(
185                builder,
186                value(order[child], elements),
187                value(order[child + 1], elements),
188            )?
189            .is_lt()
190        {
191            child += 1;
192        }
193        if !compare(
194            builder,
195            value(order[root], elements),
196            value(order[child], elements),
197        )?
198        .is_lt()
199        {
200            break;
201        }
202        builder.tick()?;
203        order.swap(root, child);
204        root = child;
205    }
206    Ok(())
