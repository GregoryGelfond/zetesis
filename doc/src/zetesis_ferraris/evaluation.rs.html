9/// Per-evaluation bounds, including storage retained from earlier evaluations.
10#[derive(Clone, Copy, Debug)]
11pub struct EvaluationLimits {
12    /// One unit per formula node, operand occurrence and tested asserted root.
13    pub max_work: u64,
14    /// Workspace header plus actual truth-vector capacity, in bytes.
15    /// The borrowed interpretation and theory are excluded.
16    pub max_bytes: usize,
17}
18
19impl Default for EvaluationLimits {
20    fn default() -> Self {
21        Self {
22            max_work: Limits::default().max_work,
23            max_bytes: 64 * 1024 * 1024,
24        }
25    }
26}

28/// Reusable storage for evaluating classical formula satisfaction.
29///
30/// Each evaluation visits every node in topological order, then asserted roots
31/// through the first false root. Work is linear in those visits; retained space
32/// is linear in the largest admitted node count. Reuse preserves capacity, not
33/// truth: every node is recomputed for the supplied interpretation. A returned
34/// [`FormulaEvaluation`] borrows both inputs and prevents workspace mutation
35/// while its truth values remain observable.
36#[derive(Debug, Default)]
37pub struct EvaluationWorkspace {
38    values: Vec<bool>,
39}
40
41impl EvaluationWorkspace {
42    /// Header and actual truth capacity, excluding allocator bookkeeping.
43    #[must_use]
44    pub fn retained_bytes(&self) -> u128 {
45        Self::bytes(self.values.capacity())
46    }
47
48    /// Reserve workspace for a theory without evaluating an interpretation.
49    /// Clears previous scratch values, keeps existing capacity and establishes
50    /// no subject or truth capability. A batch coordinator can use this to
51    /// inspect actual worker capacity before admitting candidate work.
52    ///
53    /// # Errors
54    /// Cancellation and deadlines precede storage checks. Byte ceilings include
55    /// existing capacity; allocation failure or excess actual capacity is typed.
56    /// A failure may retain increased capacity, reported by [`Self::retained_bytes`].
57    pub fn reserve(
58        &mut self,
59        theory: &Theory,
60        max_bytes: usize,
61        cancellation: &Cancellation,
62    ) -> Result<(), EvaluationError> {
63        cancellation.poll()?;
64        Self::check_storage(theory.nodes().len().max(self.values.capacity()), max_bytes)?;
65        self.values.clear();
66        self.values
67            .try_reserve_exact(theory.nodes().len())
68            .map_err(|_| Stop::Allocation)?;
69        Self::check_storage(self.values.capacity(), max_bytes)
70    }
71
72    /// Evaluate original satisfaction and retain the truth of every node.
73    ///
74    /// A successful result authenticates node truth for this exact candidate;
75    /// it does not establish answer-set membership. A stopped node or root scan
76    /// exposes no truth capability. Cancellation is checked before storage
77    /// admission, including for an empty theory. Storage refuses a lowered
78    /// ceiling below existing capacity; this operation does not shrink it.
79    /// Work and retained bytes are reported on every outcome. A failed attempt
80    /// may have grown capacity or evaluated a prefix, which the next call resets.
81    pub fn evaluate<'a>(
82        &'a mut self,
83        interpretation: &'a Interpretation,
84        limits: EvaluationLimits,
85        cancellation: &Cancellation,
86    ) -> EvaluationAttempt<'a> {
87        let mut work = Work {
88            limits: Limits {
89                max_work: limits.max_work,
90                max_subsets: 0,
91            },
92            cancellation,
93            statistics: Statistics::default(),
94        };
95        let result = self.evaluate_into(interpretation, limits.max_bytes, &mut work);
96        EvaluationAttempt {
97            work: work.statistics.work,
98            retained_bytes: self.retained_bytes(),
99            result: result.map(|failed_root| FormulaEvaluation {
100                interpretation,
101                values: &self.values,
102                failed_root,
103            }),
104        }
105    }
106
107    fn evaluate_into(
108        &mut self,
109        interpretation: &Interpretation,
110        max_bytes: usize,
111        work: &mut Work<'_>,
112    ) -> Result<Option<usize>, EvaluationError> {
113        let theory = interpretation.theory();
114        self.reserve(theory, max_bytes, work.cancellation)?;
115        evaluate(theory, interpretation, None, &mut self.values, work)?;
116        Ok(failed_root(theory, &self.values, work)?)
117    }
118
119    fn check_storage(capacity: usize, limit: usize) -> Result<(), EvaluationError> {
120        let required = Self::bytes(capacity);
121        if required > limit as u128 {
122            Err(EvaluationError::Storage { required, limit })
123        } else {
124            Ok(())
125        }
126    }
127
128    fn bytes(capacity: usize) -> u128 {
129        size_of::<Self>() as u128 + capacity as u128 * size_of::<bool>() as u128
130    }
131}

133/// Truth of all nodes and the completed classical satisfaction decision.
134///
135/// Only [`EvaluationWorkspace::evaluate`] constructs this view. Its candidate
136/// carries the exact theory owner; equal-looking foreign theories cannot be
137/// substituted. Consumers using it with another prepared object must check that
138/// object's theory identity. No public constructor accepts caller-supplied bits.
139///
140/// The borrow prevents stale truth from surviving workspace reuse:
141///
142/// ```compile_fail
143/// # use zetesis_ferraris::{EvaluationLimits, EvaluationWorkspace, Interpretation};
144/// # use zetesis_cpu::Cancellation;
145/// # fn stale(candidate: &Interpretation) {
146/// let mut workspace = EvaluationWorkspace::default();
147/// let control = Cancellation::default();
148/// let truth = workspace.evaluate(candidate, EvaluationLimits::default(), &control)
149///     .result.unwrap();
150/// let next = workspace.evaluate(candidate, EvaluationLimits::default(), &control);
151/// assert!(truth.is_model());
152/// # }
153/// ```
154#[derive(Debug)]
155pub struct FormulaEvaluation<'a> {
156    interpretation: &'a Interpretation,
157    values: &'a [bool],
158    failed_root: Option<usize>,
159}
160
161impl FormulaEvaluation<'_> {
162    /// Exact interpretation whose original truth was evaluated.
163    #[must_use]
164    pub fn interpretation(&self) -> &Interpretation {
165        self.interpretation
166    }
167
168    /// Exact admitted theory of the evaluated interpretation.
169    #[must_use]
170    pub fn theory(&self) -> &Theory {
171        self.interpretation.theory()
172    }
173
174    /// Whether every asserted root is classically satisfied.
175    #[must_use]
176    pub fn is_model(&self) -> bool {
177        self.failed_root.is_none()
178    }
179
180    /// First false asserted root, or absence after a complete successful scan.
181    #[must_use]
182    pub fn failed_root(&self) -> Option<usize> {
183        self.failed_root
184    }
185
186    /// Original truth at a valid node index; absence denotes an invalid index.
187    #[must_use]
188    pub fn node_truth(&self, node: usize) -> Option<bool> {
189        self.values.get(node).copied()
190    }
191
192    /// The truth of every node in index order: the frozen mask of the
193    /// interpretation's reduct, a node false here being falsum there.
194    #[must_use]
195    pub fn truth(&self) -> &[bool] {
196        self.values
197    }
198}

200/// Completed truth or a typed failure, with its exact consumed work prefix.
201#[derive(Debug)]
202#[must_use]
203pub struct EvaluationAttempt<'a> {
204    /// A truth view exists only after node evaluation and satisfaction complete.
205    pub result: Result<FormulaEvaluation<'a>, EvaluationError>,
206    /// Charged node, operand and root visits, including on interruption.
207    pub work: u64,
208    /// Workspace header plus actual capacity after this attempt.
209    pub retained_bytes: u128,
210}
211
212/// Failure to complete evaluation; never evidence of formula falsity.
213#[derive(Clone, Copy, Debug, PartialEq, Eq)]
214pub enum EvaluationError {
215    /// Cancellation, deadline, allocation or charged-work exhaustion.
216    Stopped(Stop),
217    /// Existing or requested workspace capacity exceeds its ceiling.
218    Storage {
219        /// Complete workspace charge, including its header.
220        required: u128,
221        /// Requested ceiling; zero is a real limit.
222        limit: usize,
223    },
224}
225
226impl From<Stop> for EvaluationError {
227    fn from(stop: Stop) -> Self {
228        Self::Stopped(stop)
229    }
230}
231
232impl fmt::Display for EvaluationError {
233    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
234        match self {
235            Self::Stopped(stop) => stop.fmt(formatter),
236            Self::Storage { required, limit } => write!(
237                formatter,
238                "formula evaluation needs {required} workspace bytes; limit is {limit}"
239            ),
240        }
241    }
242}
