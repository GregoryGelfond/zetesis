1//! Ordinal extrema reduction over already-computed predecessor truth.
2//!
3//! A row's ranks belong to one guard-independent ordered carrier. Rank zero
4//! represents its least endpoint and `rank_count - 1` its greatest endpoint.
5//! Typed value ordering and the preservation of guard comparisons are separate
6//! preparation obligations; this primitive does not compare source values.
7

13/// A borrowed ordinal row, retaining every original condition occurrence.
14///
15/// Columns have equal lengths, `rank_count` is at least two, and each rank is
16/// smaller than `rank_count`. Repeated conditions may occur at different ranks;
17/// reducing both extrema must retain all those occurrences. Child identity and
18/// preceding-node validation belong to the enclosing formula owner.
19#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
20pub struct ExtremumRow<'a> {
21    /// Ordered formula child IDs, one per contribution occurrence.
22    pub conditions: &'a [usize],
23    /// Ordinal ranks in the same occurrence order.
24    pub ranks: &'a [u64],
25    /// Size of this row's ordered carrier, including both endpoints.
26    pub rank_count: u64,
27}
28
29/// An ordinal boundary tested against the corresponding selected extremum.
30///
31/// A boundary lies in `0..=rank_count`. Formula admission validates that range;
32/// a rule and a measure must refer to the same row's carrier.
33#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
34pub enum ExtremumCutRule {
35    /// The selected minimum is strictly smaller than this boundary.
36    Prefix(u64),
37    /// The selected maximum is at least this boundary.
38    Suffix(u64),
39}
40
41/// A borrowed cut descriptor and its authoritative ordinal row.
42#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
43pub struct ExtremumCut<'a> {
44    /// Contribution occurrences and their row-local rank authority.
45    pub row: ExtremumRow<'a>,
46    /// Validated boundary in this exact row's carrier.
47    pub rule: &'a ExtremumCutRule,
48}
49
50/// Both selected extrema within one row's ordered carrier.
51///
52/// The empty selection has minimum `rank_count - 1` and maximum zero. These are
53/// the actual endpoint ranks, so an empty selection and a selected endpoint can
54/// agree on one extremum without agreeing on the other.
55#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
56pub struct ExtremaMeasure {
57    /// Least selected rank, or the greatest endpoint for an empty selection.
58    pub minimum: u64,
59    /// Greatest selected rank, or the least endpoint for an empty selection.
60    pub maximum: u64,
61}
62
63impl ExtremumRow<'_> {
64    /// Reduce both extrema from predecessor truth without allocation.
65    ///
66    /// A successful call reads every occurrence exactly once in row order,
67    /// including repeated conditions and either endpoint rank. It does not stop
68    /// after finding either extreme. The callback owns truth acquisition and
69    /// work/cancellation charges, including validation of child IDs. This
70    /// operation retains two ranks and performs linear work in the row length.
71    /// A frozen consumer supplies reduced predecessor truth and separately
72    /// applies the original cut's truth gate; this primitive does neither.
73    ///
74    /// # Errors
75    /// Propagates callback errors without further reads. Unequal column lengths,
76    /// a carrier smaller than two, or any out-of-carrier rank (even for a false
77    /// condition) returns [`Stop::InvalidProgram`]. No partial measure is returned.
78    #[inline]
79    pub fn reduce<E: From<Stop>>(
80        self,
81        mut read: impl FnMut(usize) -> Result<bool, E>,
82    ) -> Result<ExtremaMeasure, E> {
83        if self.conditions.len() != self.ranks.len() || self.rank_count < 2 {
84            return Err(Stop::InvalidProgram.into());
85        }
86        let mut measure = ExtremaMeasure {
87            minimum: self.rank_count - 1,
88            maximum: 0,
89        };
90        for (&condition, &rank) in self.conditions.iter().zip(self.ranks) {
91            if rank >= self.rank_count {
92                return Err(Stop::InvalidProgram.into());
93            }
94            if read(condition)? {
95                measure.minimum = measure.minimum.min(rank);
96                measure.maximum = measure.maximum.max(rank);
97            }
98        }
99        Ok(measure)
100    }
101}

103impl ExtremumCutRule {
104    /// Test this boundary in constant time against a measure from the same row.
105    ///
106    /// The caller validates `boundary <= rank_count` and carrier identity. This
107    /// operation does not validate an arbitrary externally constructed measure.
108    #[must_use]
109    pub const fn holds(self, measure: ExtremaMeasure) -> bool {
110        match self {
111            Self::Prefix(boundary) => measure.minimum < boundary,
112            Self::Suffix(boundary) => measure.maximum >= boundary,
113        }
114    }
115}

117impl ExtremumCut<'_> {
118    /// Recognize only endpoint-bound constants in an admitted carrier.
119    ///
120    /// The caller establishes `rank_count >= 2` and a boundary in
121    /// `0..=rank_count`. Interior cuts, including rows with no qualifying
122    /// occurrence, need a caller-metered witness scan before they can be
123    /// classified as constant. This does not inspect conditions or apply a
124    /// frozen original-node gate.
125    #[must_use]
126    pub fn endpoint_truth(self) -> Option<bool> {
127        match *self.rule {
128            ExtremumCutRule::Prefix(0) => Some(false),
129            ExtremumCutRule::Suffix(0) => Some(true),
130            ExtremumCutRule::Prefix(boundary) if boundary == self.row.rank_count => Some(true),
131            ExtremumCutRule::Suffix(boundary) if boundary == self.row.rank_count => Some(false),
132            _ => None,
133        }
134    }
135
136    /// Select one qualifying witness from this exact row's admitted carrier.
137    ///
138    /// The caller establishes `rank < rank_count` and a valid cut boundary.
139    /// Handle [`Self::endpoint_truth`] first: true endpoint cuts need no witness,
140    /// so this predicate alone does not characterize whole-cut truth. A frozen
141    /// consumer still applies the original cut gate separately.
142    #[must_use]
143    pub fn qualifies(self, rank: u64) -> bool {
144        match *self.rule {
145            ExtremumCutRule::Prefix(boundary) => rank < boundary,
146            ExtremumCutRule::Suffix(boundary) => rank >= boundary,
147        }
148    }
149}
