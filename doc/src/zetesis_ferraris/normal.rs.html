10/// Translate an already bounded static normal program into Ferraris formulas.
11/// Atom indices remain the input graph's dense indices, including unused atoms.
12/// Each rule becomes `body -> head`; constraints imply falsum. Frozen true gates
13/// become double negations, false gates single negations. Complete bodies retain
14/// one native conjunction; choice gates never become positive antecedents.
15///
16/// This parses no source. Inclusive node and operand dimensions admit all graph
17/// appends. Temporary body rows are bounded by the same operand ceiling. The
18/// returned theory has its own interpretation identity.
19///
20/// # Errors
21/// Refuses excessive dimensions, invalid edges, overflow or failed allocation.
22pub fn from_ground_program(
23    program: &GroundProgram,
24    limits: AdmissionLimits,
25) -> Result<Theory, AdmissionError> {
26    if program.atom_count() > limits.max_atoms || program.rules().len() > limits.max_roots {
27        return Err(AdmissionError::Limit);
28    }
29    let mut nodes = FormulaNodes::default();
30    let mut transaction = nodes.transaction();
31    for atom in 0..program.atom_count() {
32        push(&mut transaction, NodeView::Atom(atom), limits)?;
33    }
34    let falsum = push(&mut transaction, NodeView::False, limits)?;
35    let truth = push(&mut transaction, NodeView::Implies(falsum, falsum), limits)?;
36    let mut roots = reserve(program.rules().len())?;
37    for rule in program.rules() {
38        let count = rule
39            .positive()
40            .len()
41            .checked_add(rule.gate_true().len())
42            .and_then(|count| count.checked_add(rule.gate_false().len()))
43            .ok_or(AdmissionError::Limit)?;
44        if count > limits.max_operands {
45            return Err(AdmissionError::Limit);
46        }
47        let mut body = reserve(count)?;
48        body.extend(rule.positive().iter().map(|&atom| atom as usize));
49        for &atom in rule.gate_true() {
50            let negative = push(
51                &mut transaction,
52                NodeView::Implies(atom as usize, falsum),
53                limits,
54            )?;
55            body.push(push(
56                &mut transaction,
57                NodeView::Implies(negative, falsum),
58                limits,
59            )?);
60        }
61        for &atom in rule.gate_false() {
62            body.push(push(
63                &mut transaction,
64                NodeView::Implies(atom as usize, falsum),
65                limits,
66            )?);
67        }
68        let body = if body.is_empty() {
69            truth
70        } else {
71            push(&mut transaction, NodeView::And(&body), limits)?
72        };
73        let head = rule.head().map_or(falsum, |atom| atom as usize);
74        roots.push(push(
75            &mut transaction,
76            NodeView::Implies(body, head),
77            limits,
78        )?);
79    }
80    transaction.commit();
81    Theory::new(program.atom_count(), nodes.into_parts(), roots, limits)
82}

84fn reserve<T>(count: usize) -> Result<Vec<T>, AdmissionError> {
85    let mut values = Vec::new();
86    values
87        .try_reserve_exact(count)
88        .map_err(|_| AdmissionError::Allocation)?;
89    Ok(values)
90}

92fn push(
93    transaction: &mut FormulaTransaction<'_>,
94    node: NodeView<'_>,
95    limits: AdmissionLimits,
96) -> Result<usize, AdmissionError> {
97    transaction.push(node, limits.max_nodes, limits.max_operands)
98}

100/// Translate a static normal program and add candidate-only support conditions.
101/// Each atom `a` gets `not not (a -> OR producer_bodies)`. Every normal stable
102/// model has such a producer in its least-closure derivation. A passing guard's
103/// reduct is true for every tested subset; self-supported cycles still need the
104/// minimality oracle. Complete producer disjunctions retain native groups.
105///
106/// This specialization uses normal-rule structure, not arbitrary formulas.
107/// Admission includes all support nodes and every logical operand occurrence.
108/// The paired original graph and the copied construction owner overlap briefly.
109///
110/// # Errors
111/// Refuses the same conditions as [`from_ground_program`], including additional
112/// support graph dimensions and temporary producer incidence allocation.
113pub fn from_ground_program_supported(
114    program: &GroundProgram,
115    limits: AdmissionLimits,
116) -> Result<Theory, AdmissionError> {
117    let original = from_ground_program(program, limits)?;
118    let root_count = original
119        .roots()
120        .len()
121        .checked_add(original.atom_count())
122        .ok_or(AdmissionError::Limit)?;
123    if root_count > limits.max_roots {
124        return Err(AdmissionError::Limit);
125    }
126    let (offsets, bodies) = producers(program, &original)?;
127    let mut raw_nodes = reserve(original.nodes().len())?;
128    raw_nodes.extend_from_slice(original.nodes());
129    let mut operands = reserve(original.operands().len())?;
130    operands.extend_from_slice(original.operands());
131    let mut nodes = FormulaNodes::new(FormulaParts::new(raw_nodes, operands)?);
132    let mut roots = reserve(root_count)?;
133    roots.extend_from_slice(original.roots());
134    let mut transaction = nodes.transaction();
135    let falsum = original.atom_count();
136    for atom in 0..original.atom_count() {
137        let row = &bodies[offsets[atom]..offsets[atom + 1]];
138        let body = if row.is_empty() {
139            falsum
140        } else {
141            push(&mut transaction, NodeView::Or(row), limits)?
142        };
143        let condition = push(&mut transaction, NodeView::Implies(atom, body), limits)?;
144        let negative = push(
145            &mut transaction,
146            NodeView::Implies(condition, falsum),
147            limits,
148        )?;
149        roots.push(push(
150            &mut transaction,
151            NodeView::Implies(negative, falsum),
152            limits,
153        )?);
154    }
155    transaction.commit();
156    Theory::new(original.atom_count(), nodes.into_parts(), roots, limits)
157}

159/// One flat producer incidence row per atom, preserving original rule order.
160fn producers(
161    program: &GroundProgram,
162    original: &Theory,
163) -> Result<(Vec<usize>, Vec<usize>), AdmissionError> {
164    let length = original
165        .atom_count()
166        .checked_add(1)
167        .ok_or(AdmissionError::Limit)?;
168    let mut offsets = reserve(length)?;
169    offsets.resize(length, 0usize);
170    for rule in program.rules() {
171        if let Some(head) = rule.head() {
172            offsets[head as usize + 1] += 1;
173        }
174    }
175    for atom in 0..original.atom_count() {
176        offsets[atom + 1] += offsets[atom];
177    }
178    let count = offsets[original.atom_count()];
179    let mut bodies = reserve(count)?;
180    bodies.resize(count, 0);
181    let mut cursors = reserve(original.atom_count())?;
182    cursors.extend_from_slice(&offsets[..original.atom_count()]);
183    for (rule, &root) in program.rules().iter().zip(original.roots()) {
184        if let Some(head) = rule.head() {
185            let NodeView::Implies(body, _) = original.view().node(root)? else {
186                unreachable!("normal translation constructs implication roots");
187            };
188            let cursor = &mut cursors[head as usize];
189            bodies[*cursor] = body;
190            *cursor += 1;
191        }
192    }
193    Ok((offsets, bodies))
194}
