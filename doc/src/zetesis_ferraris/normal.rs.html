7/// Translate an already bounded static normal program into Ferraris formulas.
8/// Atom indices are exactly the input graph's dense indices, including unused
9/// carrier atoms. Each rule becomes `body -> head`; constraints imply falsum.
10/// Frozen true guards become double negations and false guards become single
11/// negations. In particular, a choice's true guard must not become an ordinary
12/// positive antecedent: `not not p -> p` supports either choice, while `p -> p`
13/// cannot support a nonempty stable model on its own.
14///
15/// This performs no source parsing or grounding. All formula storage is counted
16/// and admitted before construction; the input graph remains caller-owned.
17/// The returned theory has its own interpretation identity.
18///
19/// # Errors
20/// Returns [`AdmissionError::Limit`] for excessive formula dimensions or count
21/// overflow, and [`AdmissionError::Allocation`] if storage cannot be reserved.
22pub fn from_ground_program(
23    program: &GroundProgram,
24    limits: AdmissionLimits,
25) -> Result<Theory, AdmissionError> {
26    if program.atom_count() > limits.max_atoms || program.rules().len() > limits.max_roots {
27        return Err(AdmissionError::Limit);
28    }
29    let mut count = program.atom_count() as u128 + 2;
30    for rule in program.rules() {
31        let literals = rule.positive().len() as u128
32            + rule.gate_true().len() as u128
33            + rule.gate_false().len() as u128;
34        count += literals.saturating_sub(1)
35            + 2 * rule.gate_true().len() as u128
36            + rule.gate_false().len() as u128
37            + 1;
38        if count > limits.max_nodes as u128 {
39            return Err(AdmissionError::Limit);
40        }
41    }
42    if count > limits.max_nodes as u128 {
43        return Err(AdmissionError::Limit);
44    }
45    let count = usize::try_from(count).map_err(|_| AdmissionError::Limit)?;
46    let mut nodes = Vec::new();
47    nodes
48        .try_reserve_exact(count)
49        .map_err(|_| AdmissionError::Allocation)?;
50    let mut roots = Vec::new();
51    roots
52        .try_reserve_exact(program.rules().len())
53        .map_err(|_| AdmissionError::Allocation)?;
54    nodes.extend((0..program.atom_count()).map(Node::Atom));
55    let falsum = push(&mut nodes, Node::False);
56    let truth = push(&mut nodes, Node::Implies(falsum, falsum));
57    for rule in program.rules() {
58        let mut body = None;
59        for &atom in rule.positive() {
60            conjunct(&mut nodes, &mut body, atom as usize);
61        }
62        for &atom in rule.gate_true() {
63            let negative = push(&mut nodes, Node::Implies(atom as usize, falsum));
64            let positive = push(&mut nodes, Node::Implies(negative, falsum));
65            conjunct(&mut nodes, &mut body, positive);
66        }
67        for &atom in rule.gate_false() {
68            let negative = push(&mut nodes, Node::Implies(atom as usize, falsum));
69            conjunct(&mut nodes, &mut body, negative);
70        }
71        let head = rule.head().map_or(falsum, |atom| atom as usize);
72        roots.push(push(&mut nodes, Node::Implies(body.unwrap_or(truth), head)));
73    }
74    debug_assert_eq!(nodes.len(), count);
75    Theory::new(program.atom_count(), nodes, roots, limits)
76}

78fn push(nodes: &mut Vec<Node>, node: Node) -> usize {
79    let index = nodes.len();
80    nodes.push(node);
81    index
82}
83
84fn conjunct(nodes: &mut Vec<Node>, body: &mut Option<usize>, literal: usize) {
85    *body = Some(match *body {
86        None => literal,
87        Some(previous) => push(nodes, Node::And(previous, literal)),
88    });
89}

91/// Translate a static normal program and add candidate-only support conditions.
92/// Each atom `a` gets the guard `not not (a -> OR producer_bodies)`. Every normal
93/// stable model has such a producer in its least-closure derivation. For a
94/// candidate satisfying the guard, its frozen reduct is true for every tested
95/// subset; the added guard therefore cannot remove a countermodel or create
96/// support. Self-supported positive cycles still need the minimality oracle.
97///
98/// This is a specialization justified by the input's normal-rule structure,
99/// not a transformation for arbitrary formula theories. It avoids searching
100/// classical assignments containing carrier atoms with no possible producer.
101/// Formula dimensions include all support guards and producer disjunctions.
102///
103/// # Errors
104/// Returns [`AdmissionError`] for the same reasons as [`from_ground_program`],
105/// with additional guard storage included in the admission limits.
106pub fn from_ground_program_supported(
107    program: &GroundProgram,
108    limits: AdmissionLimits,
109) -> Result<Theory, AdmissionError> {
110    let original = from_ground_program(program, limits)?;
111    let roots_count = original.roots().len() as u128 + original.atom_count() as u128;
112    if roots_count > limits.max_roots as u128 {
113        return Err(AdmissionError::Limit);
114    }
115    let mut heads = Vec::new();
116    heads
117        .try_reserve_exact(original.atom_count())
118        .map_err(|_| AdmissionError::Allocation)?;
119    heads.resize(original.atom_count(), 0usize);
120    for rule in program.rules() {
121        if let Some(head) = rule.head() {
122            heads[head as usize] += 1;
123        }
124    }
125    let count = original.nodes().len() as u128
126        + 3 * original.atom_count() as u128
127        + heads
128            .iter()
129            .map(|count| count.saturating_sub(1) as u128)
130            .sum::<u128>();
131    if count > limits.max_nodes as u128 {
132        return Err(AdmissionError::Limit);
133    }
134    let count = usize::try_from(count).map_err(|_| AdmissionError::Limit)?;
135    let mut nodes = Vec::new();
136    nodes
137        .try_reserve_exact(count)
138        .map_err(|_| AdmissionError::Allocation)?;
139    nodes.extend_from_slice(original.nodes());
140    let mut roots = Vec::new();
141    roots
142        .try_reserve_exact(usize::try_from(roots_count).map_err(|_| AdmissionError::Limit)?)
143        .map_err(|_| AdmissionError::Allocation)?;
144    roots.extend_from_slice(original.roots());
145    let mut support = Vec::new();
146    support
147        .try_reserve_exact(original.atom_count())
148        .map_err(|_| AdmissionError::Allocation)?;
149    support.resize(original.atom_count(), None);
150    for (rule, &root) in program.rules().iter().zip(original.roots()) {
151        if let Some(head) = rule.head() {
152            let Node::Implies(body, _) = original.nodes()[root] else {
153                unreachable!("normal translation constructs implication roots");
154            };
155            let disjunction = &mut support[head as usize];
156            *disjunction = Some(match *disjunction {
157                None => body,
158                Some(previous) => push(&mut nodes, Node::Or(previous, body)),
159            });
160        }
161    }
162    // The direct translator puts falsum immediately after the atom nodes.
163    let falsum = original.atom_count();
164    for (atom, body) in support.into_iter().enumerate() {
165        let condition = push(&mut nodes, Node::Implies(atom, body.unwrap_or(falsum)));
166        let negative = push(&mut nodes, Node::Implies(condition, falsum));
167        roots.push(push(&mut nodes, Node::Implies(negative, falsum)));
168    }
169    debug_assert_eq!(nodes.len(), count);
170    Theory::new(original.atom_count(), nodes, roots, limits)
171}
