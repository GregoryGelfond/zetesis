1//! Per-pass sums and extrema indexed by one admitted numeric row authority.
2//!
3//! One 16-byte slot holds either an `i128` sum or two `u64` extrema. The immutable
4//! `NumericRow` domain determines its interpretation; the slot is never a second
5//! row authority. Byte conversions are safe Rust, with no union or casts.
6//! One invariant governs reads: a row whose validity bit is set holds its exact
7//! measure under the node truth its owner currently holds. A complete original
8//! or frozen evaluation pass clears every bit first and sets each as it reduces
9//! the row. An update of a retained tight evaluation clears the bit of every row
10//! with a changed condition before any of its guards is re-evaluated, since a
11//! row's conditions precede its guards; every surviving bit is therefore still
12//! exact.
13
14use crate::{ExtremaMeasure, ExtremumCut, NumericDomain, Theory, Threshold};
15use zetesis_cpu::Stop;
16
17/// Capacity survives passes; valid bits survive only into an update of the
18/// evaluation that published them, under the invariant the module states.
19/// Measure cells remain safely initialized after interruption, but a cell is
20/// read only while its bit is set. The cache borrows no subject or source data.
21#[derive(Debug, Default)]
22pub(crate) struct NumericRowScratch {
23    measures: Vec<Measure>,
24    valid: Vec<u64>,
25}
26
27impl NumericRowScratch {
28    fn rows(theory: &Theory) -> usize {
29        theory.evaluation_rows()
30    }
31
32    /// Maximum charged reset work, attained by a fresh scratch owner. The dense
33    /// measure column initializes every numeric row when any numeric node exists,
34    /// while each pass clears one validity word per 64 rows. Existing initialized
35    /// cells can only reduce this bound, including after an interrupted pass.
36    pub(crate) fn reset_work_bound(theory: &Theory) -> u128 {
37        let rows = Self::rows(theory);
38        rows as u128 + rows.div_ceil(64) as u128
39    }
40
41    /// Requested columns for a fresh owner, before observing allocator capacity.
42    pub(crate) fn requested_bytes(theory: &Theory) -> u128 {
43        let rows = Self::rows(theory);
44        Self::bytes(rows, rows.div_ceil(64))
45    }
46
47    /// Actual vector payload only; owning workspaces account their own headers.
48    pub(crate) fn retained_bytes(&self) -> u128 {
49        Self::bytes(self.measures.capacity(), self.valid.capacity())
50    }
51
52    /// Complete requested payload, including all capacity retained from prior
53    /// subjects. No cache allocation is requested for a theory without live numeric nodes.
54    pub(crate) fn required_bytes(&self, theory: &Theory) -> u128 {
55        let rows = Self::rows(theory);
56        Self::bytes(
57            self.measures.capacity().max(rows),
58            self.valid.capacity().max(rows.div_ceil(64)),
59        )
60    }
61
62    fn bytes(rows: usize, words: usize) -> u128 {
63        rows as u128 * size_of::<Measure>() as u128 + words as u128 * size_of::<u64>() as u128
64    }
65
66    /// Reserve columns without initializing cells or asserting pass validity.
67    /// A refused reservation can leave increased capacity for the caller's
68    /// actual-byte observation. Initialized lengths remain unchanged.
69    pub(crate) fn reserve(&mut self, theory: &Theory) -> Result<(), Stop> {
70        self.reserve_observed(theory, |_| Ok::<_, Stop>(()))
71    }
72
73    /// Observe complete actual payload after each column reservation, including
74    /// its refused attempt, before attempting another allocation. Preserve the
75    /// original allocation error when observation also refuses.
76    pub(crate) fn reserve_observed<E: From<Stop>>(
77        &mut self,
78        theory: &Theory,
79        mut observe: impl FnMut(u128) -> Result<(), E>,
80    ) -> Result<(), E> {
81        let rows = Self::rows(theory);
82        let reserved = self
83            .measures
84            .try_reserve_exact(rows.saturating_sub(self.measures.len()))
85            .map_err(|_| Stop::Allocation);
86        let observed = observe(self.retained_bytes());
87        reserved?;
88        observed?;
89        let reserved = self
90            .valid
91            .try_reserve_exact(rows.div_ceil(64).saturating_sub(self.valid.len()))
92            .map_err(|_| Stop::Allocation);
93        let observed = observe(self.retained_bytes());
94        reserved?;
95        observed
96    }
97
98    /// Start one complete topological pass. Charge every validity-word reset
99    /// and every newly initialized measure cell before its write. Tail bits
100    /// remain zero. Old initialized cells need no write and remain inaccessible
101    /// until this pass publishes their individual bit. A stopped reset is not
102    /// usable; the next reset starts by discarding every validity word again.
103    pub(crate) fn reset<E: From<Stop>>(
104        &mut self,
105        theory: &Theory,
106        mut tick: impl FnMut() -> Result<(), E>,
107    ) -> Result<(), E> {
108        let rows = Self::rows(theory);
109        let words = rows.div_ceil(64);
110        self.valid.clear();
111        if self.measures.capacity() < rows || self.valid.capacity() < words {
112            return Err(Stop::InvalidProgram.into());
113        }
114        for _ in 0..words {
115            tick()?;
116            self.valid.push(0);
117        }
118        while self.measures.len() < rows {
119            tick()?;
120            self.measures.push(Measure([0; 2]));
121        }
122        Ok(())
123    }
124
125    /// Compare a guard with its row's one complete sum for this pass. Each
126    /// occurrence, including zero weights and repeats, is read on the first
127    /// guard only. The caller applies this node's own frozen truth gate.
128    #[inline]
129    pub(crate) fn threshold<E: From<Stop>>(
130        &mut self,
131        theory: &Theory,
132        node: usize,
133        threshold: Threshold<'_>,
134        read: impl FnMut(usize) -> Result<bool, E>,
135    ) -> Result<bool, E> {
136        let descriptor = theory
137            .nodes()
138            .get(node)
139            .and_then(|node| node.threshold_descriptor())
140            .ok_or(Stop::InvalidProgram)?;
141        let row = theory
142            .parts()
143            .thresholds()
144            .get(descriptor)
145            .ok_or(Stop::InvalidProgram)?
146            .row;
147        if theory.parts().rows()[row].domain != NumericDomain::Weights {
148            return Err(Stop::InvalidProgram.into());
149        }
150        let measure = self.cached(row, || threshold.selected_sum(read).map(Measure::sum))?;
151        Ok(measure.selected_sum() >= threshold.bound)
152    }
153
154    /// Reduce an ordinal row once per pass. The node and borrowed view come
155    /// from this exact admitted theory; repeated guards use the same row ID.
156    #[inline]
157    pub(crate) fn extremum<E: From<Stop>>(
158        &mut self,
159        theory: &Theory,
160        node: usize,
161        cut: ExtremumCut<'_>,
162        read: impl FnMut(usize) -> Result<bool, E>,
163    ) -> Result<ExtremaMeasure, E> {
164        let descriptor = theory
165            .nodes()
166            .get(node)
167            .and_then(|node| node.extremum_cut_descriptor())
168            .ok_or(Stop::InvalidProgram)?;
169        let row = theory
170            .parts()
171            .extremum_cuts()
172            .get(descriptor)
173            .ok_or(Stop::InvalidProgram)?
174            .row;
175        if !matches!(
176            theory.parts().rows()[row].domain,
177            NumericDomain::Ordinal { .. }
178        ) {
179            return Err(Stop::InvalidProgram.into());
180        }
181        self.cached(row, || cut.row.reduce(read).map(Measure::ordinal))
182            .map(Measure::extrema)
183    }
184
185    /// Withdraw one row's measure, so that its next guard reduces it again. An
186    /// update of a retained evaluation withdraws exactly the rows whose
187    /// conditions changed; every other row keeps a measure equal to its
188    /// reduction under the current candidate.
189    pub(crate) fn invalidate(&mut self, row: usize) -> Result<(), Stop> {
190        let word = self.valid.get_mut(row / 64).ok_or(Stop::InvalidProgram)?;
191        *word &= !(1_u64 << (row % 64));
192        Ok(())
193    }
194
195    /// Domain interpretation is fixed by the admitted descriptor before this
196    /// call. A set bit holds the row's exact measure under the current truth
197    /// (the module's invariant); otherwise the row is reduced now. All children
198    /// precede the first referencing guard, so its complete measure remains
199    /// valid for later guards even when unrelated rows are evaluated between
200    /// them.
201    #[inline]
202    fn cached<E: From<Stop>>(
203        &mut self,
204        row: usize,
205        reduce: impl FnOnce() -> Result<Measure, E>,
206    ) -> Result<Measure, E> {
207        let word = self.valid.get_mut(row / 64).ok_or(Stop::InvalidProgram)?;
208        let bit = 1_u64 << (row % 64);
209        let measure = self.measures.get_mut(row).ok_or(Stop::InvalidProgram)?;
210        if *word & bit == 0 {
211            let completed = reduce()?;
212            *measure = completed;
213            *word |= bit;
214        }
215        Ok(*measure)
216    }
217}

219/// A compact safely initialized slot; `NumericRow` supplies its only domain tag.
220#[derive(Clone, Copy, Debug)]
221struct Measure([u64; 2]);
222
223impl Measure {
224    fn sum(total: i128) -> Self {
225        let bytes = total.to_le_bytes();
226        Self([
227            u64::from_le_bytes(bytes[..8].try_into().expect("eight low bytes")),
228            u64::from_le_bytes(bytes[8..].try_into().expect("eight high bytes")),
229        ])
230    }
231
232    fn selected_sum(self) -> i128 {
233        let mut bytes = [0; 16];
234        bytes[..8].copy_from_slice(&self.0[0].to_le_bytes());
235        bytes[8..].copy_from_slice(&self.0[1].to_le_bytes());
236        i128::from_le_bytes(bytes)
237    }
238
239    fn ordinal(measure: ExtremaMeasure) -> Self {
240        Self([measure.minimum, measure.maximum])
241    }
242
243    fn extrema(self) -> ExtremaMeasure {
244        ExtremaMeasure {
245            minimum: self.0[0],
246            maximum: self.0[1],
247        }
248    }
249}
