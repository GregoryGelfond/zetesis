1//! Per-pass sums and extrema indexed by one admitted numeric row authority.
2//!
3//! One 16-byte slot holds either an `i128` sum or two `u64` extrema. The immutable
4//! `NumericRow` domain determines its interpretation; the slot is never a second
5//! row authority. Byte conversions are safe Rust, with no union or casts.
6//! One invariant governs guard reads: a row whose validity bit is set holds its
7//! exact measure under the node truth its owner currently holds. A complete original
8//! or frozen evaluation pass clears every bit first and sets each as it reduces
9//! the row. A retained tight evaluation adjusts a valid sum once per changed
10//! occurrence and withdraws a changed extremum. All conditions precede every
11//! guard, so a guard sees the final updated sum or reduces an invalid row.
12//! Applying one changed node's incidence list is indivisible with respect to
13//! guard evaluation, including when that list repeats the same numeric row.
14//! Interrupted updates withdraw their owning evaluation before any later read.
15
16use crate::{ExtremaMeasure, ExtremumCut, NumericDomain, Theory, Threshold};
17use zetesis_algebra::{Stop, words::Words};

19/// Capacity survives passes; valid bits survive only into an update of the
20/// evaluation that published them, under the invariant the module states.
21/// Measure cells remain safely initialized after interruption, but a cell is
22/// read only while its bit is set. The cache borrows no subject or source data.
23#[derive(Debug, Default)]
24pub(crate) struct NumericRowScratch {
25    measures: Vec<Measure>,
26    valid: Words<u64>,
27}
28
29impl NumericRowScratch {
30    fn rows(theory: &Theory) -> usize {
31        theory.evaluation_rows()
32    }
33
34    /// Maximum charged reset work, attained by a fresh scratch owner. The dense
35    /// measure column initializes every numeric row when any numeric node exists,
36    /// while each pass clears one validity word per 64 rows. Existing initialized
37    /// cells can only reduce this bound, including after an interrupted pass.
38    pub(crate) fn reset_work_bound(theory: &Theory) -> u128 {
39        let rows = Self::rows(theory);
40        rows as u128 + rows.div_ceil(64) as u128
41    }
42
43    /// Requested columns for a fresh owner, before observing allocator capacity.
44    pub(crate) fn requested_bytes(theory: &Theory) -> u128 {
45        let rows = Self::rows(theory);
46        Self::bytes(rows, rows.div_ceil(64))
47    }
48
49    /// Actual vector payload only; owning workspaces account their own headers.
50    pub(crate) fn retained_bytes(&self) -> u128 {
51        Self::bytes(self.measures.capacity(), self.valid.capacity())
52    }
53
54    /// Complete requested payload, including all capacity retained from prior
55    /// subjects. No cache allocation is requested for a theory without live numeric nodes.
56    pub(crate) fn required_bytes(&self, theory: &Theory) -> u128 {
57        let rows = Self::rows(theory);
58        Self::bytes(
59            self.measures.capacity().max(rows),
60            self.valid.capacity().max(rows.div_ceil(64)),
61        )
62    }
63
64    fn bytes(rows: usize, words: usize) -> u128 {
65        rows as u128 * size_of::<Measure>() as u128 + words as u128 * size_of::<u64>() as u128
66    }
67
68    /// Reserve columns without initializing cells or asserting pass validity.
69    /// A refused reservation can leave increased capacity for the caller's
70    /// actual-byte observation. Initialized lengths remain unchanged.
71    pub(crate) fn reserve(&mut self, theory: &Theory) -> Result<(), Stop> {
72        self.reserve_observed(theory, |_| Ok::<_, Stop>(()))
73    }
74
75    /// Observe complete actual payload after each column reservation, including
76    /// its refused attempt, before attempting another allocation. Preserve the
77    /// original allocation error when observation also refuses.
78    pub(crate) fn reserve_observed<E: From<Stop>>(
79        &mut self,
80        theory: &Theory,
81        mut observe: impl FnMut(u128) -> Result<(), E>,
82    ) -> Result<(), E> {
83        let rows = Self::rows(theory);
84        let reserved = self
85            .measures
86            .try_reserve_exact(rows.saturating_sub(self.measures.len()))
87            .map_err(|_| Stop::Allocation);
88        let observed = observe(self.retained_bytes());
89        reserved?;
90        observed?;
91        let additional = rows.div_ceil(64).saturating_sub(self.valid.len());
92        let reserved = self
93            .valid
94            .storage_mut()
95            .try_reserve_exact(additional)
96            .map_err(|_| Stop::Allocation);
97        let observed = observe(self.retained_bytes());
98        reserved?;
99        observed
100    }
101
102    /// Start one complete topological pass. Charge every validity-word reset
103    /// and every newly initialized measure cell before its write. Tail bits
104    /// remain zero. Old initialized cells need no write and remain inaccessible
105    /// until this pass publishes their individual bit. A stopped reset is not
106    /// usable; the next reset starts by discarding every validity word again.
107    pub(crate) fn reset<E: From<Stop>>(
108        &mut self,
109        theory: &Theory,
110        mut tick: impl FnMut() -> Result<(), E>,
111    ) -> Result<(), E> {
112        let rows = Self::rows(theory);
113        let words = rows.div_ceil(64);
114        self.valid.storage_mut().clear();
115        if self.measures.capacity() < rows || self.valid.capacity() < words {
116            return Err(Stop::InvalidProgram.into());
117        }
118        for _ in 0..words {
119            tick()?;
120            self.valid.storage_mut().push(0);
121        }
122        while self.measures.len() < rows {
123            tick()?;
124            self.measures.push(Measure([0; 2]));
125        }
126        Ok(())
127    }
128
129    /// Compare a guard with its row's one complete sum for this pass. Each
130    /// occurrence, including zero weights and repeats, is read on the first
131    /// guard only. The caller applies this node's own frozen truth gate.
132    #[inline]
133    pub(crate) fn threshold<E: From<Stop>>(
134        &mut self,
135        theory: &Theory,
136        node: usize,
137        threshold: Threshold<'_>,
138        read: impl FnMut(usize) -> Result<bool, E>,
139    ) -> Result<bool, E> {
140        let descriptor = theory
141            .nodes()
142            .get(node)
143            .and_then(|node| node.threshold_descriptor())
144            .ok_or(Stop::InvalidProgram)?;
145        let row = theory
146            .parts()
147            .thresholds()
148            .get(descriptor)
149            .ok_or(Stop::InvalidProgram)?
150            .row;
151        if theory.parts().rows()[row].domain != NumericDomain::Weights {
152            return Err(Stop::InvalidProgram.into());
153        }
154        let measure = self.cached(row, || threshold.selected_sum(read).map(Measure::sum))?;
155        Ok(measure.selected_sum() >= threshold.bound)
156    }
157
158    /// Reduce an ordinal row once per pass. The node and borrowed view come
159    /// from this exact admitted theory; repeated guards use the same row ID.
160    #[inline]
161    pub(crate) fn extremum<E: From<Stop>>(
162        &mut self,
163        theory: &Theory,
164        node: usize,
165        cut: ExtremumCut<'_>,
166        read: impl FnMut(usize) -> Result<bool, E>,
167    ) -> Result<ExtremaMeasure, E> {
168        let descriptor = theory
169            .nodes()
170            .get(node)
171            .and_then(|node| node.extremum_cut_descriptor())
172            .ok_or(Stop::InvalidProgram)?;
173        let row = theory
174            .parts()
175            .extremum_cuts()
176            .get(descriptor)
177            .ok_or(Stop::InvalidProgram)?
178            .row;
179        if !matches!(
180            theory.parts().rows()[row].domain,
181            NumericDomain::Ordinal { .. }
182        ) {
183            return Err(Stop::InvalidProgram.into());
184        }
185        self.cached(row, || cut.row.reduce(read).map(Measure::ordinal))
186            .map(Measure::extrema)
187    }
188
189    /// Apply one changed condition occurrence. The caller owns the same theory
190    /// and visits each occurrence once after its truth changes; `selected` is
191    /// its new truth. Row identity and weight come only from the authoritative
192    /// columns. Repeated conditions remain distinct contributions, including
193    /// zero weights. The caller charges the reader visit before this operation.
194    ///
195    /// A valid sum is adjusted exactly; an ordinal measure is withdrawn. An
196    /// invalid measure stays invalid, so the next guard performs a full read.
197    /// No guard may run until every earlier changed condition has been handled.
198    pub(crate) fn change_condition(
199        &mut self,
200        theory: &Theory,
201        row: usize,
202        occurrence: usize,
203        selected: bool,
204    ) -> Result<(), Stop> {
205        let descriptor = theory.parts().rows().get(row).ok_or(Stop::InvalidProgram)?;
206        if occurrence >= descriptor.length {
207            return Err(Stop::InvalidProgram);
208        }
209        let word = self.valid.get_mut(row / 64).ok_or(Stop::InvalidProgram)?;
210        let bit = Words::<u64>::locate(row).1;
211        if *word & bit == 0 {
212            return Ok(());
213        }
214        if descriptor.domain != NumericDomain::Weights {
215            *word &= !bit;
216            return Ok(());
217        }
218        let position = descriptor
219            .start
220            .checked_add(occurrence)
221            .ok_or(Stop::InvalidProgram)?;
222        let weight = i128::from(
223            *theory
224                .parts()
225                .numeric_values()
226                .get(position)
227                .ok_or(Stop::InvalidProgram)?,
228        );
229        let measure = self.measures.get_mut(row).ok_or(Stop::InvalidProgram)?;
230        let sum = measure.selected_sum();
231        let updated = if selected {
232            sum.checked_add(weight)
233        } else {
234            sum.checked_sub(weight).filter(|&value| value >= 0)
235        }
236        .ok_or(Stop::InvalidProgram)?;
237        *measure = Measure::sum(updated);
238        Ok(())
239    }
240
241    /// Domain interpretation is fixed by the admitted descriptor before this
242    /// call. A set bit holds the row's exact measure under the current truth
243    /// (the module's invariant); otherwise the row is reduced now. All children
244    /// precede the first referencing guard, so its complete measure remains
245    /// valid for later guards even when unrelated rows are evaluated between
246    /// them.
247    #[inline]
248    fn cached<E: From<Stop>>(
249        &mut self,
250        row: usize,
251        reduce: impl FnOnce() -> Result<Measure, E>,
252    ) -> Result<Measure, E> {
253        let word = self.valid.get_mut(row / 64).ok_or(Stop::InvalidProgram)?;
254        let bit = Words::<u64>::locate(row).1;
255        let measure = self.measures.get_mut(row).ok_or(Stop::InvalidProgram)?;
256        if *word & bit == 0 {
257            let completed = reduce()?;
258            *measure = completed;
259            *word |= bit;
260        }
261        Ok(*measure)
262    }
263}

265/// A compact safely initialized slot; `NumericRow` supplies its only domain tag.
266#[derive(Clone, Copy, Debug)]
267struct Measure([u64; 2]);
268
269impl Measure {
270    fn sum(total: i128) -> Self {
271        let bytes = total.to_le_bytes();
272        Self([
273            u64::from_le_bytes(bytes[..8].try_into().expect("eight low bytes")),
274            u64::from_le_bytes(bytes[8..].try_into().expect("eight high bytes")),
275        ])
276    }
277
278    fn selected_sum(self) -> i128 {
279        let mut bytes = [0; 16];
280        bytes[..8].copy_from_slice(&self.0[0].to_le_bytes());
281        bytes[8..].copy_from_slice(&self.0[1].to_le_bytes());
282        i128::from_le_bytes(bytes)
283    }
284
285    fn ordinal(measure: ExtremaMeasure) -> Self {
286        Self([measure.minimum, measure.maximum])
287    }
288
289    fn extrema(self) -> ExtremaMeasure {
290        ExtremaMeasure {
291            minimum: self.0[0],
292            maximum: self.0[1],
293        }
294    }
295}
