8/// Per-call bounds for exact finite checking. Exceeding a bound is incomplete,
9/// never a proof of stability or nonminimality.
10#[derive(Clone, Copy, Debug)]
11pub struct Limits {
12    /// Charged node/operand evaluations, numeric-cache initialization/reset,
13    /// root tests, atom scans and subset-bit operations. A shared numeric row
14    /// is reduced once per complete evaluation pass.
15    pub max_work: u64,
16    /// Maximum number of proper subsets checked against a frozen reduct.
17    pub max_subsets: u64,
18}
19impl Default for Limits {
20    fn default() -> Self {
21        Self {
22            max_work: 100_000_000,
23            max_subsets: 1_048_576,
24        }
25    }
26}

28/// Logical work performed before a completed decision.
29#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
30pub struct Statistics {
31    /// Charged primitive operations, including classical candidate evaluation.
32    pub work: u64,
33    /// Proper subsets evaluated; the original candidate is never counted.
34    pub subsets: u64,

37/// Completed membership verdict data with a checkable rejection witness. The
38/// public variants do not retain the original checked subject; use
39/// [`crate::check_interpretation`] for an owned subject-bound native decision.
40#[derive(Clone, Debug)]
41pub enum Verdict {
42    /// No proper subset satisfies the candidate's frozen formula reduct.
43    Stable,
44    /// The original interpretation fails this asserted root node.
45    NotModel {
46        /// Circuit node index of a false asserted formula.
47        root: usize,
48    },
49    /// This proper subset models the frozen reduct.
50    NonMinimal {
51        /// Program-bound counterexample to subset minimality.
52        witness: Interpretation,
53    },
54}

56/// Completed decision and exact logical-work accounting.
57#[derive(Clone, Debug)]
58pub struct Check {
59    verdict: Verdict,
60    statistics: Statistics,
61}
62impl Check {
63    /// Stable-model membership decision, or the reason for rejection.
64    #[must_use]
65    pub fn verdict(&self) -> &Verdict {
66        &self.verdict
67    }
68    /// True only for a completed minimality proof.
69    #[must_use]
70    pub fn accepted(&self) -> bool {
71        matches!(self.verdict, Verdict::Stable)
72    }
73    /// Work consumed by this completed call.
74    #[must_use]
75    pub fn statistics(&self) -> Statistics {
76        self.statistics
77    }
78}

80/// Formula evaluation's work: its meter, its subset ceiling and the subsets
81/// tested so far.
82pub(super) struct Work<'m, 'c> {
83    pub(super) max_subsets: u64,
84    pub(super) meter: &'m mut Meter<'c>,
85    pub(super) subsets: u64,

87impl<'m, 'c> Work<'m, 'c> {
88    pub(super) fn new(max_subsets: u64, meter: &'m mut Meter<'c>) -> Self {
89        Self {
90            max_subsets,
91            meter,
92            subsets: 0,
93        }
94    }

96    pub(super) fn tick(&mut self) -> Result<(), Stop> {
97        self.meter.charge().map_err(ChargeError::into_stop)
98    }
99
100    pub(super) fn statistics(&self) -> Statistics {
101        Statistics {
102            work: self.meter.charged(),
103            subsets: self.subsets,
104        }
105    }

108pub(super) fn identities(theory: &Theory, interpretation: &Interpretation) -> Result<(), Stop> {
109    if theory.same_instance(interpretation.theory()) {
110        Ok(())
111    } else {
112        Err(Stop::WrongProgram)
113    }

116pub(super) fn reserve<T>(count: usize) -> Result<Vec<T>, Stop> {
117    let mut vector = Vec::new();
118    vector
119        .try_reserve_exact(count)
120        .map_err(|_| Stop::Allocation)?;
121    Ok(vector)
122}

124pub(super) fn evaluate(
125    theory: &Theory,
126    interpretation: &Interpretation,
127    frozen: Option<&[bool]>,
128    output: &mut Vec<bool>,
129    numeric: &mut NumericRowScratch,
130    work: &mut Work<'_, '_>,
131) -> Result<(), Stop> {
132    output.clear();
133    numeric.reset(theory, || work.tick())?;
134    let view = theory.view();
135    for index in 0..view.len() {
136        work.tick()?;
137        let node = view.node(index).map_err(|_| Stop::InvalidProgram)?;
138        let value = evaluate_node(index, node, interpretation, output, numeric, work)?;
139        // A maximal M-false subformula becomes falsum. Masking every M-false
140        // node has the same root meaning and avoids materializing a new DAG.
141        output.push(value && frozen.is_none_or(|mask| mask[index]));
142    }
143    Ok(())
144}

146/// Compute one node from its complete predecessor values. The caller publishes
147/// its result only after this read-only phase finishes, including all child work.
148fn evaluate_node(
149    index: usize,
150    node: NodeView<'_>,
151    interpretation: &Interpretation,
152    values: &[bool],
153    numeric: &mut NumericRowScratch,
154    work: &mut Work<'_, '_>,
155) -> Result<bool, Stop> {
156    match node {
157        NodeView::Atom(atom) => Ok(interpretation.contains(atom)),
158        NodeView::False => Ok(false),
159        NodeView::And(operands) => evaluate_operands(operands, values, true, work),
160        NodeView::Or(operands) => evaluate_operands(operands, values, false, work),
161        NodeView::Threshold(threshold) => {
162            numeric.threshold(interpretation.theory(), index, threshold, |child| {
163                work.tick()?;
164                Ok(values[child])
165            })
166        }
167        NodeView::ExtremumCut(cut) => numeric
168            .extremum(interpretation.theory(), index, cut, |child| {
169                work.tick()?;
170                Ok(values[child])
171            })
172            .map(|measure| cut.rule.holds(measure)),
173        NodeView::Implies(a, b) => {
174            work.tick()?;
175            let antecedent = values[a];
176            work.tick()?;
177            let consequent = values[b];
178            Ok(!antecedent || consequent)
179        }
180    }
181}

183/// Read every operand, including duplicates and operands after the result is
184/// decided. A failed tick leaves the enclosing node unpublished in the output.
185fn evaluate_operands(
186    operands: &[usize],
187    values: &[bool],
188    conjunction: bool,
189    work: &mut Work<'_, '_>,
190) -> Result<bool, Stop> {
191    let mut value = conjunction;
192    for &operand in operands {
193        work.tick()?;
194        let child = values[operand];
195        value = if conjunction {
196            value & child
197        } else {
198            value | child
199        };
200    }
201    Ok(value)
202}

204pub(super) fn failed_root(
205    theory: &Theory,
206    values: &[bool],
207    work: &mut Work<'_, '_>,
208) -> Result<Option<usize>, Stop> {
209    for root in theory.roots() {
210        work.tick()?;
211        if !values[*root] {
212            return Ok(Some(*root));
213        }
214    }
215    Ok(None)
216}

218/// Classical satisfaction of a finite theory.
219///
220/// # Errors
221/// Refuses foreign interpretations, cancellation, deadlines, allocation, or work limits.
222pub fn models(
223    theory: &Theory,
224    interpretation: &Interpretation,
225    limits: Limits,
226    cancellation: &Cancellation,
227) -> Result<bool, Stop> {
228    identities(theory, interpretation)?;
229    EvaluationWorkspace::default()
230        .evaluate(
231            interpretation,
232            EvaluationLimits {
233                max_work: limits.max_work,
234                max_bytes: usize::MAX,
235            },
236            cancellation,
237        )
238        .result
239        .map(|evaluation| evaluation.is_model())
240        .map_err(|error| match error {
241            EvaluationError::Stopped(stop) => stop,
242            // The convenience operation imposes no finite byte ceiling. A
243            // charge beyond usize is an unrepresentable allocation request.
244            EvaluationError::Storage { .. } => Stop::Allocation,
245        })
246}

248/// Satisfaction in `tested` of the formula reduct frozen in `candidate`.
249/// No subset relation is required; minimality search imposes that separately.
250/// This call freezes the candidate anew and shares one work budget between
251/// freezing and testing. Use [`FrozenReduct`] to reuse a freeze across tests,
252/// with a separate budget for construction and each satisfaction query.
253///
254/// # Errors
255/// Refuses foreign interpretations, cancellation, deadlines, allocation, or work limits.
256pub fn models_reduct(
257    theory: &Theory,
258    candidate: &Interpretation,
259    tested: &Interpretation,
260    limits: Limits,
261    cancellation: &Cancellation,
262) -> Result<bool, Stop> {
263    cancellation
264        .metered(limits.max_work, |meter| {
265            reduct_satisfaction(
266                theory,
267                candidate,
268                tested,
269                Work::new(limits.max_subsets, meter),
270            )
271        })
272        .result
273}

275/// [`models_reduct`] charged to `work`.
276fn reduct_satisfaction(
277    theory: &Theory,
278    candidate: &Interpretation,
279    tested: &Interpretation,
280    mut work: Work<'_, '_>,
281) -> Result<bool, Stop> {
282    identities(theory, candidate)?;
283    identities(theory, tested)?;
284    work.meter.poll()?;
285    let mut values = reserve(theory.nodes().len())?;
286    let mut numeric = NumericRowScratch::default();
287    numeric.reserve(theory)?;
288    let reduct = FrozenReduct::freeze(candidate, &mut numeric, &mut work)?;
289    reduct.satisfied_by(tested, &mut values, &mut numeric, &mut work)
290}

292/// Append the candidate's atom coordinates in increasing order. The caller has
293/// checked theory identity and reserved an empty destination for the universe.
294/// A stop retains the selected prefix and work charged before the refused tick.
295fn select_atoms(
296    theory: &Theory,
297    candidate: &Interpretation,
298    selected: &mut Vec<usize>,
299    work: &mut Work<'_, '_>,
300) -> Result<(), Stop> {
301    for atom in 0..theory.atom_count() {
302        work.tick()?;
303        if candidate.contains(atom) {
304            selected.push(atom);
305        }
306    }
307    Ok(())
308}

310/// Advance the selected coordinates as a binary counter. The caller supplies
311/// distinct in-universe coordinates, a subset supported on those coordinates,
312/// and its population. `check` calls this only for a proper subset. A stop retains
313/// preceding bit and population updates; the refused tick performs no update.
314#[inline]
315fn advance_subset(
316    selected: &[usize],
317    subset: &mut Interpretation,
318    present: &mut usize,
319    work: &mut Work<'_, '_>,
320) -> Result<(), Stop> {
321    for atom in selected {
322        work.tick()?;
323        let packed = &mut subset.words[*atom / 64];
324        let bit = 1 << (*atom % 64);
325        if *packed & bit == 0 {
326            *packed |= bit;
327            *present += 1;
328            break;
329        }
330        *packed &= !bit;
331        *present -= 1;
332    }
333    Ok(())
334}

336/// Test one proper subset against the frozen reduct. Poll before charging the
337/// subset, then share the remaining work allowance between evaluation and roots.
338/// A true result is a countermodel; a false result permits the next carry.
339#[inline]
340fn check_subset(
341    theory: &Theory,
342    subset: &Interpretation,
343    frozen: &[bool],
344    values: &mut Vec<bool>,
345    numeric: &mut NumericRowScratch,
346    work: &mut Work<'_, '_>,
347) -> Result<bool, Stop> {
348    work.meter.poll()?;
349    if work.subsets >= work.max_subsets {
350        return Err(Stop::CandidateLimit);
351    }
352    work.subsets += 1;
353    evaluate(theory, subset, Some(frozen), values, numeric, work)?;
354    Ok(failed_root(theory, values, work)?.is_none())
355}

357/// Search the candidate's proper subsets in binary-counter order. The caller
358/// supplies its exact original truth mask, exact increasing selected coordinates
359/// and an empty, same-theory subset. On success, true leaves the first reduct
360/// countermodel in `subset`; false establishes that no proper subset satisfies
361/// the frozen reduct. A stop preserves the work and partial carry.
362#[inline]
363fn find_countermodel(
364    theory: &Theory,
365    frozen: &[bool],
366    selected: &[usize],
367    subset: &mut Interpretation,
368    values: &mut Vec<bool>,
369    numeric: &mut NumericRowScratch,
370    work: &mut Work<'_, '_>,
371) -> Result<bool, Stop> {
372    // Empty is the first proper subset unless M itself is empty. Incrementing
373    // over selected atom indices avoids machine-word cardinality restrictions.
374    let mut present = 0;
375    let mut countermodel = Ok(false);
376    while present < selected.len() {
377        countermodel = check_subset(theory, subset, frozen, values, numeric, work);
378        match countermodel {
379            Ok(false) => {
380                if let Err(stop) = advance_subset(selected, subset, &mut present, work) {
381                    countermodel = Err(stop);
382                    break;
383                }
384            }
385            Ok(true) | Err(_) => break,
386        }
387    }
388    countermodel
389}

391/// Decide stable-model membership by classical satisfaction and exhaustive
392/// proper-subset checking of the Ferraris formula reduct. This reference kernel
393/// is exponential in the candidate size and does not assume a least reduct model.
394///
395/// # Errors
396/// Refuses foreign candidates, cancellation, deadlines, allocation, or exhausted
397/// work/subset bounds. A subset bound uses `Stop::CandidateLimit` because it
398/// bounds candidate countermodels; no partial search returns `Stable`.
399pub fn check(
400    theory: &Theory,
401    candidate: &Interpretation,
402    limits: Limits,
403    cancellation: &Cancellation,
404) -> Result<Check, Stop> {
405    cancellation
406        .metered(limits.max_work, |meter| {
407            exhaustive_check(theory, candidate, Work::new(limits.max_subsets, meter))
408        })
409        .result
410}

412/// [`check`] charged to `work`.
413fn exhaustive_check(
414    theory: &Theory,
415    candidate: &Interpretation,
416    mut work: Work<'_, '_>,
417) -> Result<Check, Stop> {
418    identities(theory, candidate)?;
419    work.meter.poll()?;
420    let mut frozen = reserve(theory.nodes().len())?;
421    let mut numeric = NumericRowScratch::default();
422    numeric.reserve(theory)?;
423    evaluate(
424        theory,
425        candidate,
426        None,
427        &mut frozen,
428        &mut numeric,
429        &mut work,
430    )?;
431    if let Some(root) = failed_root(theory, &frozen, &mut work)? {
432        return Ok(Check {
433            verdict: Verdict::NotModel { root },
434            statistics: work.statistics(),
435        });
436    }
437    let mut selected = reserve(theory.atom_count())?;
438    select_atoms(theory, candidate, &mut selected, &mut work)?;
439    let mut words = reserve(candidate.words.len())?;
440    words.resize(candidate.words.len(), 0);
441    let mut subset = Interpretation {
442        theory: theory.clone(),
443        words,
444    };
445    let mut values = reserve(theory.nodes().len())?;
446    let verdict = if find_countermodel(
447        theory,
448        &frozen,
449        &selected,
450        &mut subset,
451        &mut values,
452        &mut numeric,
453        &mut work,
454    )? {
455        Verdict::NonMinimal { witness: subset }
456    } else {
457        Verdict::Stable
458    };
459    Ok(Check {
460        verdict,
461        statistics: work.statistics(),
462    })
463}

471    /// Run `test` on work whose meter has admitted seven operations, between
472    /// metered boundaries, under a limit `allowance` operations further, with
473    /// three subsets already tested. With `stop`, a stop is requested once the
474    /// meter is there. The test's own value is returned unsettled.
475    fn with_work<R>(allowance: u64, stop: bool, test: impl FnOnce(&mut Work<'_, '_>) -> R) -> R {
476        let cancellation = Cancellation::default();
477        let limits = Limits {
478            max_work: 7 + allowance,
479            max_subsets: 5,
480        };
481        cancellation
482            .metered(limits.max_work, |meter| {
483                for _ in 0..7 {
484                    meter.charge().unwrap();
485                }
486                if stop {
487                    cancellation.cancel();
488                }
489                let mut work = Work {
490                    max_subsets: limits.max_subsets,
491                    meter,
492                    subsets: 3,
493                };
494                Ok::<_, Stop>(test(&mut work))
495            })
496            .result
497            .unwrap()
498    }

501    fn a_requested_stop_precedes_the_work_limit_between_boundaries() {
502        // A refused charge leaves the count where it was.
503        let tick = |stop| with_work(0, stop, |work| (work.tick(), work.statistics().work));
504        assert_eq!(tick(false), (Err(Stop::WorkLimit), 7));
505        assert_eq!(tick(true), (Err(Stop::Cancelled), 7));
