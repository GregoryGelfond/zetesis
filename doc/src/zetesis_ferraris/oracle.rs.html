7/// Per-call bounds for exact finite checking. Exceeding a bound is incomplete,
8/// never a proof of stability or nonminimality.
9#[derive(Clone, Copy, Debug)]
10pub struct Limits {
11    /// Charged node and operand evaluations, root tests, atom scans and subset-bit operations.
12    pub max_work: u64,
13    /// Maximum number of proper subsets checked against a frozen reduct.
14    pub max_subsets: u64,
15}
16impl Default for Limits {
17    fn default() -> Self {
18        Self {
19            max_work: 100_000_000,
20            max_subsets: 1_048_576,
21        }
22    }
23}

25/// Logical work performed before a completed decision.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
27pub struct Statistics {
28    /// Charged primitive operations, including classical candidate evaluation.
29    pub work: u64,
30    /// Proper subsets evaluated; the original candidate is never counted.
31    pub subsets: u64,
32}

34/// Completed membership verdict data with a checkable rejection witness. The
35/// public variants do not retain the original checked subject; use
36/// [`crate::check_interpretation`] for an owned subject-bound native decision.
37#[derive(Clone, Debug)]
38pub enum Verdict {
39    /// No proper subset satisfies the candidate's frozen formula reduct.
40    Stable,
41    /// The original interpretation fails this asserted root node.
42    NotModel {
43        /// Circuit node index of a false asserted formula.
44        root: usize,
45    },
46    /// This proper subset models the frozen reduct.
47    NonMinimal {
48        /// Program-bound counterexample to subset minimality.
49        witness: Interpretation,
50    },
51}

53/// Completed decision and exact logical-work accounting.
54#[derive(Clone, Debug)]
55pub struct Check {
56    verdict: Verdict,
57    statistics: Statistics,
58}
59impl Check {
60    /// Stable-model membership decision, or the reason for rejection.
61    #[must_use]
62    pub fn verdict(&self) -> &Verdict {
63        &self.verdict
64    }
65    /// True only for a completed minimality proof.
66    #[must_use]
67    pub fn accepted(&self) -> bool {
68        matches!(self.verdict, Verdict::Stable)
69    }
70    /// Work consumed by this completed call.
71    #[must_use]
72    pub fn statistics(&self) -> Statistics {
73        self.statistics
74    }
75}

77pub(super) struct Work<'a> {
78    pub(super) limits: Limits,
79    pub(super) cancellation: &'a Cancellation,
80    pub(super) statistics: Statistics,
81}
82impl Work<'_> {
83    pub(super) fn tick(&mut self) -> Result<(), Stop> {
84        self.cancellation.poll()?;
85        if self.statistics.work >= self.limits.max_work {
86            return Err(Stop::WorkLimit);
87        }
88        self.statistics.work += 1;
89        Ok(())
90    }
91}

93pub(super) fn identities(theory: &Theory, interpretation: &Interpretation) -> Result<(), Stop> {
94    if theory.same_instance(interpretation.theory()) {
95        Ok(())
96    } else {
97        Err(Stop::WrongProgram)
98    }
99}

101pub(super) fn reserve<T>(count: usize) -> Result<Vec<T>, Stop> {
102    let mut vector = Vec::new();
103    vector
104        .try_reserve_exact(count)
105        .map_err(|_| Stop::Allocation)?;
106    Ok(vector)
107}

109pub(super) fn evaluate(
110    theory: &Theory,
111    interpretation: &Interpretation,
112    frozen: Option<&[bool]>,
113    output: &mut Vec<bool>,
114    work: &mut Work<'_>,
115) -> Result<(), Stop> {
116    output.clear();
117    let view = theory.view();
118    for index in 0..view.len() {
119        work.tick()?;
120        let node = view.node(index).map_err(|_| Stop::InvalidProgram)?;
121        let value = evaluate_node(node, interpretation, output, work)?;
122        // A maximal M-false subformula becomes falsum. Masking every M-false
123        // node has the same root meaning and avoids materializing a new DAG.
124        output.push(value && frozen.is_none_or(|mask| mask[index]));
125    }
126    Ok(())
127}

129/// Compute one node from its complete predecessor values. The caller publishes
130/// its result only after this read-only phase finishes, including all child work.
131fn evaluate_node(
132    node: NodeView<'_>,
133    interpretation: &Interpretation,
134    values: &[bool],
135    work: &mut Work<'_>,
136) -> Result<bool, Stop> {
137    match node {
138        NodeView::Atom(atom) => Ok(interpretation.contains(atom)),
139        NodeView::False => Ok(false),
140        NodeView::And(operands) => evaluate_operands(operands, values, true, work),
141        NodeView::Or(operands) => evaluate_operands(operands, values, false, work),
142        NodeView::Implies(a, b) => {
143            work.tick()?;
144            let antecedent = values[a];
145            work.tick()?;
146            let consequent = values[b];
147            Ok(!antecedent || consequent)
148        }
149    }
150}

152/// Read every operand, including duplicates and operands after the result is
153/// decided. A failed tick leaves the enclosing node unpublished in the output.
154fn evaluate_operands(
155    operands: &[usize],
156    values: &[bool],
157    conjunction: bool,
158    work: &mut Work<'_>,
159) -> Result<bool, Stop> {
160    let mut value = conjunction;
161    for &operand in operands {
162        work.tick()?;
163        let child = values[operand];
164        value = if conjunction {
165            value & child
166        } else {
167            value | child
168        };
169    }
170    Ok(value)
171}

173pub(super) fn failed_root(
174    theory: &Theory,
175    values: &[bool],
176    work: &mut Work<'_>,
177) -> Result<Option<usize>, Stop> {
178    for root in theory.roots() {
179        work.tick()?;
180        if !values[*root] {
181            return Ok(Some(*root));
182        }
183    }
184    Ok(None)
185}

187/// Classical satisfaction of a finite theory.
188///
189/// # Errors
190/// Refuses foreign interpretations, cancellation, deadlines, allocation, or work limits.
191pub fn models(
192    theory: &Theory,
193    interpretation: &Interpretation,
194    limits: Limits,
195    cancellation: &Cancellation,
196) -> Result<bool, Stop> {
197    identities(theory, interpretation)?;
198    EvaluationWorkspace::default()
199        .evaluate(
200            interpretation,
201            EvaluationLimits {
202                max_work: limits.max_work,
203                max_bytes: usize::MAX,
204            },
205            cancellation,
206        )
207        .result
208        .map(|evaluation| evaluation.is_model())
209        .map_err(|error| match error {
210            EvaluationError::Stopped(stop) => stop,
211            // The convenience operation imposes no finite byte ceiling. A
212            // charge beyond usize is an unrepresentable allocation request.
213            EvaluationError::Storage { .. } => Stop::Allocation,
214        })
215}

217/// Satisfaction in `tested` of the formula reduct frozen in `candidate`.
218/// No subset relation is required; minimality search imposes that separately.
219/// This call freezes the candidate anew and shares one work budget between
220/// freezing and testing. Use [`FrozenReduct`] to reuse a freeze across tests,
221/// with a separate budget for construction and each satisfaction query.
222///
223/// # Errors
224/// Refuses foreign interpretations, cancellation, deadlines, allocation, or work limits.
225pub fn models_reduct(
226    theory: &Theory,
227    candidate: &Interpretation,
228    tested: &Interpretation,
229    limits: Limits,
230    cancellation: &Cancellation,
231) -> Result<bool, Stop> {
232    identities(theory, candidate)?;
233    identities(theory, tested)?;
234    cancellation.poll()?;
235    let mut work = Work {
236        limits,
237        cancellation,
238        statistics: Statistics::default(),
239    };
240    let mut values = reserve(theory.nodes().len())?;
241    let reduct = FrozenReduct::freeze(candidate, &mut work)?;
242    reduct.satisfied_by(tested, &mut values, &mut work)
243}

245/// Append the candidate's atom coordinates in increasing order. The caller has
246/// checked theory identity and reserved an empty destination for the universe.
247/// A stop retains the selected prefix and work charged before the refused tick.
248fn select_atoms(
249    theory: &Theory,
250    candidate: &Interpretation,
251    selected: &mut Vec<usize>,
252    work: &mut Work<'_>,
253) -> Result<(), Stop> {
254    for atom in 0..theory.atom_count() {
255        work.tick()?;
256        if candidate.contains(atom) {
257            selected.push(atom);
258        }
259    }
260    Ok(())
261}

263/// Advance the selected coordinates as a binary counter. The caller supplies
264/// distinct in-universe coordinates, a subset supported on those coordinates,
265/// and its population. `check` calls this only for a proper subset. A stop retains
266/// preceding bit and population updates; the refused tick performs no update.
267#[inline]
268fn advance_subset(
269    selected: &[usize],
270    subset: &mut Interpretation,
271    present: &mut usize,
272    work: &mut Work<'_>,
273) -> Result<(), Stop> {
274    for atom in selected {
275        work.tick()?;
276        let packed = &mut subset.words[*atom / 64];
277        let bit = 1 << (*atom % 64);
278        if *packed & bit == 0 {
279            *packed |= bit;
280            *present += 1;
281            break;
282        }
283        *packed &= !bit;
284        *present -= 1;
285    }
286    Ok(())
287}

289/// Test one proper subset against the frozen reduct. Poll before charging the
290/// subset, then share the remaining work allowance between evaluation and roots.
291/// A true result is a countermodel; a false result permits the next carry.
292#[inline]
293fn check_subset(
294    theory: &Theory,
295    subset: &Interpretation,
296    frozen: &[bool],
297    values: &mut Vec<bool>,
298    work: &mut Work<'_>,
299) -> Result<bool, Stop> {
300    work.cancellation.poll()?;
301    if work.statistics.subsets >= work.limits.max_subsets {
302        return Err(Stop::CandidateLimit);
303    }
304    work.statistics.subsets += 1;
305    evaluate(theory, subset, Some(frozen), values, work)?;
306    Ok(failed_root(theory, values, work)?.is_none())
307}

309/// Search the candidate's proper subsets in binary-counter order. The caller
310/// supplies its exact original truth mask, exact increasing selected coordinates
311/// and an empty, same-theory subset. On success, true leaves the first reduct
312/// countermodel in `subset`; false establishes that no proper subset satisfies
313/// the frozen reduct. A stop preserves the work and partial carry.
314#[inline]
315fn find_countermodel(
316    theory: &Theory,
317    frozen: &[bool],
318    selected: &[usize],
319    subset: &mut Interpretation,
320    values: &mut Vec<bool>,
321    work: &mut Work<'_>,
322) -> Result<bool, Stop> {
323    // Empty is the first proper subset unless M itself is empty. Incrementing
324    // over selected atom indices avoids machine-word cardinality restrictions.
325    let mut present = 0;
326    let mut countermodel = Ok(false);
327    while present < selected.len() {
328        countermodel = check_subset(theory, subset, frozen, values, work);
329        match countermodel {
330            Ok(false) => {
331                if let Err(stop) = advance_subset(selected, subset, &mut present, work) {
332                    countermodel = Err(stop);
333                    break;
334                }
335            }
336            Ok(true) | Err(_) => break,
337        }
338    }
339    countermodel
340}

342/// Decide stable-model membership by classical satisfaction and exhaustive
343/// proper-subset checking of the Ferraris formula reduct. This reference kernel
344/// is exponential in the candidate size and does not assume a least reduct model.
345///
346/// # Errors
347/// Refuses foreign candidates, cancellation, deadlines, allocation, or exhausted
348/// work/subset bounds. A subset bound uses `Stop::CandidateLimit` because it
349/// bounds candidate countermodels; no partial search returns `Stable`.
350pub fn check(
351    theory: &Theory,
352    candidate: &Interpretation,
353    limits: Limits,
354    cancellation: &Cancellation,
355) -> Result<Check, Stop> {
356    identities(theory, candidate)?;
357    cancellation.poll()?;
358    let mut work = Work {
359        limits,
360        cancellation,
361        statistics: Statistics::default(),
362    };
363    let mut frozen = reserve(theory.nodes().len())?;
364    evaluate(theory, candidate, None, &mut frozen, &mut work)?;
365    if let Some(root) = failed_root(theory, &frozen, &mut work)? {
366        return Ok(Check {
367            verdict: Verdict::NotModel { root },
368            statistics: work.statistics,
369        });
370    }
371    let mut selected = reserve(theory.atom_count())?;
372    select_atoms(theory, candidate, &mut selected, &mut work)?;
373    let mut words = reserve(candidate.words.len())?;
374    words.resize(candidate.words.len(), 0);
375    let mut subset = Interpretation {
376        theory: theory.clone(),
377        words,
378    };
379    let mut values = reserve(theory.nodes().len())?;
380    let verdict = if find_countermodel(
381        theory,
382        &frozen,
383        &selected,
384        &mut subset,
385        &mut values,
386        &mut work,
387    )? {
388        Verdict::NonMinimal { witness: subset }
389    } else {
390        Verdict::Stable
391    };
392    Ok(Check {
393        verdict,
394        statistics: work.statistics,
395    })
396}
