16/// Bounds for emitting a separate consequence theory, excluding the borrowed plan.
17/// The existing aggregate translator also bounds each temporary state frontier.
18#[derive(Clone, Copy, Debug)]
19pub struct RestrictionLimits {
20    /// Atom, final node and root ceilings; node limits also constrain translation.
21    pub theory: AdmissionLimits,
22    /// Per-group element/frontier ceilings. Its work ceiling applies separately
23    /// to each translation, within the cumulative ceiling below.
24    pub aggregate: AggregateLimits,
25    /// Cumulative atom-node, root and aggregate-translation operations.
26    pub max_work: u64,
27}
28
29impl Default for RestrictionLimits {
30    fn default() -> Self {
31        Self {
32            theory: AdmissionLimits::default(),
33            aggregate: AggregateLimits::default(),
34            max_work: 10_000_000,
35        }
36    }
37}

39/// An incomplete consequence-theory construction; no partial theory is exposed.
40#[derive(Clone, Copy, Debug, PartialEq, Eq)]
41pub enum RestrictionErrorKind {
42    /// The declared semantic universe exceeds the emission ceiling.
43    Atoms,
44    /// A formula node would exceed the emission ceiling.
45    Nodes,
46    /// An asserted consequence would exceed the emission ceiling.
47    Roots,
48    /// One group's temporary element vector exceeds its ceiling.
49    Elements,
50    /// Cumulative emission work reached its inclusive ceiling.
51    Work,
52    /// A bound or accounting quantity cannot be represented.
53    Overflow,
54    /// The independent exact cardinality translator refused its bounded work.
55    Aggregate(AggregateError),
56    /// Final theory admission refused the constructed shape.
57    Theory(crate::AdmissionError),
58    /// Cancellation or fallible allocation stopped emission.
59    Stopped(Stop),
60}
61
62impl Refusal for RestrictionErrorKind {
63    fn is_resource_refusal(&self) -> bool {
64        match self {
65            Self::Atoms
66            | Self::Nodes
67            | Self::Roots
68            | Self::Elements
69            | Self::Work
70            | Self::Overflow => true,
71            Self::Aggregate(error) => error.kind().is_resource_refusal(),
72            Self::Theory(error) => error.is_resource_refusal(),
73            Self::Stopped(stop) => stop.is_resource_refusal(),
74        }
75    }
76
77    fn from_stop(stop: Stop) -> Self {
78        Self::Stopped(stop)
79    }
80}

82/// Typed emission refusal with its charged work prefix.
83#[derive(Clone, Copy, Debug, PartialEq, Eq)]
84pub struct RestrictionError {
85    kind: RestrictionErrorKind,
86    work: u64,
87}

89impl RestrictionError {
90    /// Specific refusal, never an accepted consequence or logical UNSAT.
91    #[must_use]
92    pub const fn kind(self) -> RestrictionErrorKind {
93        self.kind
94    }
95
96    /// Charged emission and translation operations before refusal.
97    #[must_use]
98    pub const fn work(self) -> u64 {
99        self.work
100    }
101}
102
103impl fmt::Display for RestrictionError {
104    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
105        match self.kind {
106            RestrictionErrorKind::Aggregate(error) => error.fmt(f),
107            RestrictionErrorKind::Theory(error) => error.fmt(f),
108            RestrictionErrorKind::Stopped(stop) => stop.fmt(f),
109            kind => write!(f, "partition consequence emission refused: {kind:?}"),
110        }
111    }
112}
113
114impl std::error::Error for RestrictionError {
115    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
116        match &self.kind {
117            RestrictionErrorKind::Aggregate(error) => Some(error),
118            RestrictionErrorKind::Theory(error) => Some(error),
119            RestrictionErrorKind::Stopped(stop) => Some(stop),
120            _ => None,
121        }
122    }
123}

125/// An emitted lower-bound view, with no original-theory entailment certificate.
126#[derive(Debug)]
127pub struct Restriction {
128    theory: Theory,
129    work: u64,
130}
131
132impl Restriction {
133    /// Consequence formulas over the caller's declared semantic atom indices.
134    #[must_use]
135    pub fn theory(&self) -> &Theory {
136        &self.theory
137    }
138
139    /// Charged atom-node, root and aggregate-translation operations.
140    #[must_use]
141    pub const fn work(&self) -> u64 {
142        self.work
143    }
144}
145
146impl Plan {
147    /// Emit only the derived local lower bounds as a separate Boolean theory.
148    ///
149    /// Inconsistent stated premises emit falsum, even for an empty partition.
150    /// Otherwise zero lower bounds emit no root. The original theory is neither
151    /// read nor changed. Applying this view to a candidate generator preserves
152    /// complete original enumeration only if the original theory entails the
153    /// supplied premises with exactly the same atom index meanings. The original
154    /// theory must remain the subject of every frozen-reduct check.
155    ///
156    /// Emission reuses exact cardinality lowering. Temporary elements are bounded
157    /// per group; nodes, roots, translator states and total work have independent
158    /// ceilings. Translation costs include revalidating its existing DAG prefix.
159    /// Final theory shape validation scans those bounded nodes and roots once
160    /// more; it is not included in the logical work counter. No search occurs.
161    ///
162    /// # Errors
163    /// Refuses emission/translation bounds, allocation or control. Failure drops
164    /// the entire partial output and retains its charged work count.
165    pub fn restriction(
166        &self,
167        limits: RestrictionLimits,
168        cancellation: &Cancellation,
169    ) -> Result<Restriction, RestrictionError> {
170        let metered = cancellation.metered(limits.max_work, |meter| {
171            Builder {
172                nodes: FormulaNodes::default(),
173                roots: Vec::new(),
174                limits,
175                cancellation,
176                meter,
177            }
178            .build(self)
179        });
180        let work = metered.charged;
181        metered
182            .result
183            .map(|theory| Restriction { theory, work })
184            .map_err(|kind| RestrictionError { kind, work })
185    }
186}

188/// One emission's output under construction, its limits and its meter.
189/// Cardinality lowering, an operation of another arrangement that polls before
190/// every step, runs under the smaller of its per-translation work ceiling and
191/// the meter's remaining allowance, and its reported work is counted afterwards.
192struct Builder<'a, 'c> {
193    nodes: FormulaNodes,
194    roots: Vec<usize>,
195    limits: RestrictionLimits,
196    cancellation: &'c Cancellation,
197    meter: &'a mut Meter<'c>,
198}
199
200impl Builder<'_, '_> {
201    fn build(&mut self, plan: &Plan) -> Result<Theory, RestrictionErrorKind> {
202        self.poll()?;
203        if plan.atom_count() > self.limits.theory.max_atoms {
204            return Err(RestrictionErrorKind::Atoms);
205        }
206        if plan.inconsistent() {
207            let falsum = self.node(NodeView::False)?;
208            self.root(falsum)?;
209        } else {
210            for group in plan.consequences() {
211                self.tick()?;
212                if group.lower != 0 {
213                    let root = self.lower(group.members, group.lower)?;
214                    self.root(root)?;
215                }
216            }
217        }
218        self.poll()?;
219        let theory = Theory::new(
220            plan.atom_count(),
221            std::mem::take(&mut self.nodes).into_parts(),
222            std::mem::take(&mut self.roots),
223            self.limits.theory,
224        )
225        .map_err(RestrictionErrorKind::Theory)?;
226        self.poll()?;
227        Ok(theory)
228    }
229
230    fn lower(&mut self, members: &[usize], lower: usize) -> Result<usize, RestrictionErrorKind> {
231        if members.len() > self.limits.aggregate.max_elements {
232            return Err(RestrictionErrorKind::Elements);
233        }
234        let mut elements = Vec::new();
235        elements
236            .try_reserve_exact(members.len())
237            .map_err(|_| RestrictionErrorKind::Stopped(Stop::Allocation))?;
238        for &atom in members {
239            let condition = self.node(NodeView::Atom(atom))?;
240            elements.push(AggregateElement {
241                condition,
242                weight: 1,
243            });
244        }
245        let bound = i64::try_from(lower).map_err(|_| RestrictionErrorKind::Overflow)?;
246        let mut limits = self.limits.aggregate;
247        limits.max_nodes = limits.max_nodes.min(self.limits.theory.max_nodes);
248        limits.max_operands = limits.max_operands.min(self.limits.theory.max_operands);
249        limits.max_work = limits.max_work.min(self.meter.remaining());
250        let result = self.nodes.append_aggregate(
251            &elements,
252            AggregateComparison::Ge,
253            bound,
254            limits,
255            self.cancellation,
256        );
257        let work = match result {
258            Ok(build) => build.statistics().work,
259            Err(error) => error.statistics().work,
260        };
261        self.meter.absorb(work).map_err(refused)?;
262        result
263            .map(crate::AggregateBuild::root)
264            .map_err(RestrictionErrorKind::Aggregate)
265    }
266
267    fn poll(&self) -> Result<(), RestrictionErrorKind> {
268        self.meter.poll().map_err(RestrictionErrorKind::Stopped)
269    }
270
271    fn tick(&mut self) -> Result<(), RestrictionErrorKind> {
272        self.meter.charge().map_err(refused)
273    }
274
275    fn node(&mut self, node: NodeView<'_>) -> Result<usize, RestrictionErrorKind> {
276        self.tick()?;
277        if self.nodes.parts().nodes().len() == self.limits.theory.max_nodes {
278            return Err(RestrictionErrorKind::Nodes);
279        }
280        let mut transaction = self.nodes.transaction();
281        let index = transaction
282            .push(
283                node,
284                self.limits.theory.max_nodes,
285                self.limits.theory.max_operands,
286            )
287            .map_err(RestrictionErrorKind::Theory)?;
288        transaction.commit();
289        Ok(index)
290    }
291
292    fn root(&mut self, root: usize) -> Result<(), RestrictionErrorKind> {
293        self.tick()?;
294        if self.roots.len() == self.limits.theory.max_roots {
295            return Err(RestrictionErrorKind::Roots);
296        }
297        self.roots
298            .try_reserve(1)
299            .map_err(|_| RestrictionErrorKind::Stopped(Stop::Allocation))?;
300        self.roots.push(root);
301        Ok(())
302    }
303}

305/// A refused charge in the emission's vocabulary: a requested stop, or the
306/// emission's work limit.
307fn refused(error: ChargeError) -> RestrictionErrorKind {
308    match error {
309        ChargeError::Stopped(stop) => RestrictionErrorKind::Stopped(stop),
310        ChargeError::WorkLimit => RestrictionErrorKind::Work,
311    }
