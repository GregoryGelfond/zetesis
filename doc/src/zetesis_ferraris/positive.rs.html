1//! Least consequences of positive producers, checked against original constraints.

3//! This optional complete-root certificate admits positive cycles and monotone

5//! dependency projection as the original theory. Construction computes the least

19/// Independent limits for complete-root classification and least closure.
20#[derive(Clone, Copy, Debug)]
21pub struct PositivePlanLimits {
22    /// Forward incidences, counting every ordered child occurrence, including
23    /// repeated IDs in native conjunction and disjunction rows.
24    pub max_dependencies: usize,
25    /// Named plan header, construction-vector headers and actual vector capacity.
26    /// Excludes the shared theory, allocator overhead and other stack temporaries.
27    /// This is not RSS or an allocator hard cap; allocation slack is checked after
28    /// reservation, and a refused actual allocation contributes to the peak.
29    pub max_bytes: usize,
30    /// Charged node/root, initialization, incidence and queue operations.
31    pub max_work: u64,
32}
33
34impl Default for PositivePlanLimits {
35    fn default() -> Self {
36        Self {
37            max_dependencies: 4_194_304,
38            max_bytes: 256 * 1024 * 1024,
39            max_work: 100_000_000,
40        }
41    }
42}

44/// Which explicit preparation resource could not be admitted.
45#[derive(Clone, Copy, Debug, PartialEq, Eq)]
46pub enum PositiveResource {
47    /// Forward graph incidences.
48    Dependencies,
49    /// Named simultaneous construction/retained storage.
50    Bytes,
51    /// Charged preparation and propagation work.
52    Work,
53}

55/// Refused grammar or incomplete preparation, never a satisfiability result.
56#[derive(Clone, Copy, Debug, PartialEq, Eq)]
57pub enum PositiveError {
58    /// This original root is not an atomic fact, positive atomic-head rule or
59    /// arbitrary constraint. In particular, choices are not ordinary producers.
60    UnsupportedRoot {
61        /// Original asserted formula node.
62        root: usize,
63    },
64    /// An original atomic-head producer body contains an unsupported implication.
65    UnsupportedBody {
66        /// Original asserted formula node.
67        root: usize,
68        /// Original body node whose complete nested classification failed.
69        body: usize,
70    },
71    /// Completed propagation failed an original producer. This is an invariant
72    /// refusal, never a constraint-based conclusion that no answer set exists.
73    InvalidClosure {
74        /// Original producer root reported false by exact evaluation.
75        root: usize,
76    },
77    /// A named resource's complete next request exceeds its ceiling.
78    Limit {
79        /// Refused resource.
80        resource: PositiveResource,
81        /// Complete requested amount, rather than only its increment.
82        observed: u128,
83        /// Configured ceiling.
84        limit: u128,
85    },
86    /// Checked shape/count arithmetic cannot represent the required value.
87    Overflow,
88    /// Original cancellation, deadline or allocation failure.
89    Stopped(Stop),
90}
91
92impl fmt::Display for PositiveError {
93    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
94        match self {
95            Self::UnsupportedRoot { root } => {
96                write!(f, "root {root} has no positive atomic-head plan")
97            }
98            Self::UnsupportedBody { root, body } => {
99                write!(f, "positive root {root} has an unsupported body at {body}")
100            }
101            Self::InvalidClosure { root } => {
102                write!(f, "least consequences fail original producer {root}")
103            }
104            Self::Limit {
105                resource,
106                observed,
107                limit,
108            } => write!(
109                f,
110                "positive plan {resource:?} needs {observed}; limit is {limit}"
111            ),
112            Self::Overflow => f.write_str("positive plan arithmetic overflowed"),
113            Self::Stopped(stop) => stop.fmt(f),
114        }
115    }
116}
117impl std::error::Error for PositiveError {}
118impl From<Stop> for PositiveError {
119    fn from(stop: Stop) -> Self {
120        Self::Stopped(stop)
121    }
122}

124/// Work and named-storage observations, including a failed preparation prefix.
125#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
126pub struct PositivePlanStatistics {
127    /// Admitted primitive operations. A refused next operation is not included.
128    pub work: u64,
129    /// Admitted forward incidences during graph construction.
130    pub dependencies: usize,
131    /// Atom vertices that became true, including facts.
132    pub derived_atoms: usize,
133    /// Formula vertices that became true, including canonical truth constants.
134    pub activated_nodes: usize,
135    /// Outgoing incidences visited during propagation.
136    pub propagated_dependencies: usize,
137    /// Maximum named simultaneous bytes, using observed vector capacities and
138    /// reserving the final plan header throughout construction. Refused proposed
139    /// allocations do not increase this; actual slack before refusal does.
140    pub peak_bytes: u128,
141    /// Final plan header and actual interpretation-word capacity on success;
142    /// zero on failure, because no prepared owner is returned.
143    pub retained_bytes: u128,
144}

146/// A complete least-consequence computation for one exact original theory.
147///
148/// Every root was checked. The least interpretation satisfies all positive
149/// producers. If no constraint failed, it is the unique answer set of this
150/// admitted grammar; a failed constraint rules out every answer set, but need
151/// not rule out larger classical models when its body is nonmonotone. This
152/// class certificate is not the solve crate's subject-associated `AnswerSet`.
153/// It does not establish source completeness or verify the Rust implementation.
154///
155/// ```
156/// use zetesis_cpu::Cancellation;
157/// use zetesis_ferraris::{AdmissionLimits, Node, PositivePlan, PositivePlanLimits, Theory};
158///
159/// // a. b :- a. a :- b. The positive cycle is seeded by the fact a.
160/// let theory = Theory::new(
161///     2,
162///     zetesis_ferraris::FormulaParts::new(vec![Node::atom(0), Node::atom(1), Node::implies(0, 1), Node::implies(1, 0)], vec![]).unwrap(),
163///     vec![0, 2, 3],
164///     AdmissionLimits::default(),
165/// )?;
166/// let plan = PositivePlan::compile(&theory, PositivePlanLimits::default(), &Cancellation::default())?;
167/// assert_eq!(plan.least_consequences().atoms().collect::<Vec<_>>(), [0, 1]);
168/// assert_eq!(plan.failed_constraint(), None);
169/// # Ok::<(), Box<dyn std::error::Error>>(())
170/// ```
171#[derive(Debug)]
172pub struct PositivePlan {
173    least: Interpretation,
174    failed_constraint: Option<usize>,
175    statistics: PositivePlanStatistics,
176}

178impl PositivePlan {
179    /// Compute once using the same implementation as [`Self::compile_accounted`].
180    ///
181    /// # Errors
182    /// Returns unsupported-root/body, resource, arithmetic, control or allocation
183    /// refusal. No partial closure or certificate is returned.
184    pub fn compile(
185        theory: &Theory,
186        limits: PositivePlanLimits,
187        cancellation: &Cancellation,
188    ) -> Result<Self, PositiveError> {
189        Self::compile_accounted(theory, limits, cancellation).result
190    }
191
192    /// Inspect all original roots and compute least consequences with forward
193    /// incidences. Producer bodies admit atoms, falsum, And, Or and exactly
194    /// False→False truth. Roots admit atomic facts, positive body→atom,
195    /// arbitrary body→False and False. Other producer implication bodies,
196    /// default negation in producers and choice heads are refused.
197    ///
198    /// Each atom/formula vertex becomes true at most once; each outgoing edge
199    /// is then visited once. Total work is linear in atoms, nodes, roots and
200    /// incidences; no DNF expansion or repeated full fixpoint scan occurs.
201    /// After propagation, exact original evaluation checks the least result.
202    /// Evaluation shares the remaining work and simultaneous byte allowance;
203    /// its temporary truth vector replaces the released incidence/work vectors.
204    /// Failed attempts retain actual work and peak observations, not a closure.
205    #[must_use]
206    pub fn compile_accounted(
207        theory: &Theory,
208        limits: PositivePlanLimits,
209        cancellation: &Cancellation,
210    ) -> PositiveAttempt {
211        let mut budget = Budget {
212            limits,
213            cancellation,
214            statistics: PositivePlanStatistics::default(),
215            current_bytes: 0,
216        };
217        let result = compile::build(theory, &mut budget);
218        let result = result.map(|(least, failed_constraint)| {
219            budget.statistics.retained_bytes = size_of::<Self>() as u128
220                + least.words.capacity() as u128 * size_of::<u64>() as u128;
221            Self {
222                least,
223                failed_constraint,
224                statistics: budget.statistics,
225            }
226        });
227        PositiveAttempt {
228            result,
229            statistics: budget.statistics,
230        }
231    }
232
233    /// Exact immutable original theory; equality of independently built theories
234    /// does not authorize substituting their local atom IDs.
235    #[must_use]
236    pub fn theory(&self) -> &Theory {
237        self.least.theory()
238    }
239
240    /// Least consequences of the positive producers. When a constraint failed,
241    /// this interpretation is not a model of the complete theory.
242    #[must_use]
243    pub const fn least_consequences(&self) -> &Interpretation {
244        &self.least
245    }
246
247    /// First violated original constraint in root order, after complete closure.
248    #[must_use]
249    pub const fn failed_constraint(&self) -> Option<usize> {
250        self.failed_constraint
251    }
252
253    /// Complete construction/propagation/original-evaluation and owner observations.
254    #[must_use]
255    pub const fn statistics(&self) -> PositivePlanStatistics {
256        self.statistics
257    }
258}

260/// One preparation attempt; an error supplies no positive semantic certificate.
261#[derive(Debug)]
262pub struct PositiveAttempt {
263    /// Completed plan or original typed refusal.
264    pub result: Result<PositivePlan, PositiveError>,
265    /// Actual admitted work and named capacity, including a failure prefix.
266    pub statistics: PositivePlanStatistics,
267}
268
269struct Budget<'a> {
270    limits: PositivePlanLimits,
271    cancellation: &'a Cancellation,
272    statistics: PositivePlanStatistics,
273    current_bytes: u128,
274}
275
276impl Budget<'_> {
277    fn tick(&mut self) -> Result<(), PositiveError> {
278        self.cancellation.poll()?;
279        Self::ceiling(
280            PositiveResource::Work,
281            u128::from(self.statistics.work) + 1,
282            u128::from(self.limits.max_work),
283        )?;
284        self.statistics.work += 1;
285        Ok(())
286    }
287
288    fn dependency(&mut self) -> Result<(), PositiveError> {
289        let next = self
290            .statistics
291            .dependencies
292            .checked_add(1)
293            .ok_or(PositiveError::Overflow)?;
294        Self::ceiling(
295            PositiveResource::Dependencies,
296            next as u128,
297            self.limits.max_dependencies as u128,
298        )?;
299        self.statistics.dependencies = next;
300        Ok(())
301    }
302
303    fn observe_bytes(&mut self, bytes: u128) -> Result<(), PositiveError> {
304        self.current_bytes = bytes;
305        self.statistics.peak_bytes = self.statistics.peak_bytes.max(bytes);
306        self.check_bytes(bytes)
307    }
308
309    fn check_bytes(&self, bytes: u128) -> Result<(), PositiveError> {
310        Self::ceiling(
311            PositiveResource::Bytes,
312            bytes,
313            self.limits.max_bytes as u128,
314        )
315    }
316
317    fn reserve<T>(&mut self, count: usize) -> Result<Vec<T>, PositiveError> {
318        self.cancellation.poll()?;
319        let requested = self
320            .current_bytes
321            .checked_add(count as u128 * size_of::<T>() as u128)
322            .ok_or(PositiveError::Overflow)?;
323        self.check_bytes(requested)?;
324        let mut result = Vec::new();
325        result
326            .try_reserve_exact(count)
327            .map_err(|_| Stop::Allocation)?;
328        let actual = self
329            .current_bytes
330            .checked_add(result.capacity() as u128 * size_of::<T>() as u128)
331            .ok_or(PositiveError::Overflow)?;
332        self.observe_bytes(actual)?;
333        Ok(result)
334    }
335
336    fn release<T>(&mut self, vector: Vec<T>) -> Result<(), PositiveError> {
337        let bytes = vector.capacity() as u128 * size_of::<T>() as u128;
338        drop(vector);
339        self.current_bytes = self
340            .current_bytes
341            .checked_sub(bytes)
342            .ok_or(PositiveError::Overflow)?;
343        Ok(())
344    }
345
346    fn ceiling(
347        resource: PositiveResource,
348        observed: u128,
349        limit: u128,
350    ) -> Result<(), PositiveError> {
351        if observed > limit {
352            return Err(PositiveError::Limit {
353                resource,
354                observed,
355                limit,
356            });
357        }
358        Ok(())
359    }
