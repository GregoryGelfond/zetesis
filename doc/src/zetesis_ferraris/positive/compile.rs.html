3use std::mem::size_of;

5use super::{Budget, PositiveError, PositivePlan, propagate, validation};
6use crate::{Interpretation, Node, Theory};

8#[derive(Clone, Copy)]
9enum Kind {
10    Unsupported,
11    False,
12    True,
13    Atom,
14    And,
15    Or,
16}
17
18impl Kind {
19    fn monotone(self) -> bool {
20        !matches!(self, Self::Unsupported)
21    }
22    fn waiting(self) -> State {
23        match self {
24            Self::Atom | Self::Or => State::AwaitingOne,
25            Self::And => State::AwaitingTwo,
26            Self::Unsupported | Self::False | Self::True => State::Inactive,
27        }
28    }
29}

31/// A vertex awaits its remaining incidences or has been queued exactly once.
32/// Inactive truth constants are activated explicitly during seeding; false and
33/// unsupported nodes remain inactive.
34#[derive(Clone, Copy, PartialEq, Eq)]
35pub(super) enum State {
36    Inactive,
37    AwaitingOne,
38    AwaitingTwo,
39    True,
40}
41
42pub(super) struct Graph {
43    pub offsets: Vec<usize>,
44    pub targets: Vec<usize>,
45    pub states: Vec<State>,
46    pub queue: Vec<usize>,
47}
48
49pub(super) fn build(
50    theory: &Theory,
51    budget: &mut Budget<'_>,
52) -> Result<(Interpretation, Option<usize>), PositiveError> {
53    budget.cancellation.poll()?;
54    let headers = size_of::<PositivePlan>()
55        + size_of::<Graph>()
56        + size_of::<Vec<Kind>>()
57        + size_of::<Vec<usize>>();
58    budget.observe_bytes(headers as u128)?;
59    let vertices = theory
60        .atom_count()
61        .checked_add(theory.nodes().len())
62        .ok_or(PositiveError::Overflow)?;
63    let kinds = classify(theory, budget)?;
64    let mut offsets = initialized(
65        vertices.checked_add(1).ok_or(PositiveError::Overflow)?,
66        0_usize,
67        budget,
68    )?;
69    edges(theory, &kinds, budget, |source, _, budget| {
70        budget.dependency()?;
71        offsets[source + 1] = offsets[source + 1]
72            .checked_add(1)
73            .ok_or(PositiveError::Overflow)?;
74        Ok(())
75    })?;
76    for index in 0..vertices {
77        budget.tick()?;
78        offsets[index + 1] = offsets[index + 1]
79            .checked_add(offsets[index])
80            .ok_or(PositiveError::Overflow)?;
81    }
82    let mut targets = initialized(budget.statistics.dependencies, 0_usize, budget)?;
83    let mut cursors = budget.reserve(vertices)?;
84    for &offset in &offsets[..vertices] {
85        budget.tick()?;
86        cursors.push(offset);
87    }
88    edges(theory, &kinds, budget, |source, target, _| {
89        targets[cursors[source]] = target;
90        cursors[source] += 1;
91        Ok(())
92    })?;
93    let mut states = budget.reserve(vertices)?;
94    for _ in 0..theory.atom_count() {
95        budget.tick()?;
96        states.push(State::AwaitingOne);
97    }
98    for kind in &kinds {
99        budget.tick()?;
100        states.push(kind.waiting());
101    }
102    budget.release(kinds)?;
103    budget.release(cursors)?;
104    let queue = budget.reserve(vertices)?;
105    let mut words = initialized(theory.atom_count().div_ceil(64), 0_u64, budget)?;
106    let mut graph = Graph {
107        offsets,
108        targets,
109        states,
110        queue,
111    };
112    propagate::complete(theory, &mut graph, &mut words, budget)?;
113    drop(graph);
114    let least = Interpretation {
115        theory: theory.clone(),
116        words,
117    };
118    let failed = validation::complete(&least, budget)?;
119    Ok((least, failed))
120}

122fn initialized<T: Copy>(
123    count: usize,
124    value: T,
125    budget: &mut Budget<'_>,
126) -> Result<Vec<T>, PositiveError> {
127    let mut values = budget.reserve(count)?;
128    for _ in 0..count {
129        budget.tick()?;
130        values.push(value);
131    }
132    Ok(values)
133}

135fn classify(theory: &Theory, budget: &mut Budget<'_>) -> Result<Vec<Kind>, PositiveError> {
136    let mut kinds: Vec<Kind> = budget.reserve(theory.nodes().len())?;
137    for node in theory.nodes() {
138        budget.tick()?;
139        kinds.push(match *node {
140            Node::Atom(_) => Kind::Atom,
141            Node::False => Kind::False,
142            Node::And(a, b) if kinds[a].monotone() && kinds[b].monotone() => Kind::And,
143            Node::Or(a, b) if kinds[a].monotone() && kinds[b].monotone() => Kind::Or,
144            Node::Implies(a, b)
145                if theory.nodes()[a] == Node::False && theory.nodes()[b] == Node::False =>
146            {
147                Kind::True
148            }
149            Node::And(_, _) | Node::Or(_, _) | Node::Implies(_, _) => Kind::Unsupported,
150        });
151    }
152    Ok(kinds)
153}

155/// The same finite edge stream is counted then written into reserved CSR slots.
156/// Original roots are checked on both passes against the same immutable theory.
157/// Unsupported unreachable nodes do not supply edges; every accepted body has
158/// a complete monotone classification. No formula or atom occurrence is merged.
159fn edges(
160    theory: &Theory,
161    kinds: &[Kind],
162    budget: &mut Budget<'_>,
163    mut visit: impl FnMut(usize, usize, &mut Budget<'_>) -> Result<(), PositiveError>,
164) -> Result<(), PositiveError> {
165    let atoms = theory.atom_count();
166    for (index, node) in theory.nodes().iter().enumerate() {
167        budget.tick()?;
168        match (kinds[index], *node) {
169            (Kind::Atom, Node::Atom(atom)) => {
170                budget.tick()?;
171                visit(atom, atoms + index, budget)?;
172            }
173            (Kind::And, Node::And(a, b)) | (Kind::Or, Node::Or(a, b)) => {
174                for child in [a, b] {
175                    budget.tick()?;
176                    visit(atoms + child, atoms + index, budget)?;
177                }
178            }
179            _ => {}
180        }
181    }
182    for &root in theory.roots() {
183        budget.tick()?;
184        if let Some((body, head)) = producer(theory, root, kinds)? {
185            budget.tick()?;
186            visit(atoms + body, head, budget)?;
187        }
188    }
189    Ok(())
190}
191
192fn producer(
193    theory: &Theory,
194    root: usize,
195    kinds: &[Kind],
196) -> Result<Option<(usize, usize)>, PositiveError> {
197    match theory.nodes()[root] {
198        Node::Atom(_) | Node::False => Ok(None),
199        Node::Implies(_, head) if theory.nodes()[head] == Node::False => Ok(None),
200        Node::Implies(body, head) => {
201            let Node::Atom(atom) = theory.nodes()[head] else {
202                return Err(PositiveError::UnsupportedRoot { root });
203            };
204            if !kinds[body].monotone() {
205                return Err(PositiveError::UnsupportedBody { root, body });
206            }
207            Ok(Some((body, atom)))
208        }
209        _ => Err(PositiveError::UnsupportedRoot { root }),
210    }
