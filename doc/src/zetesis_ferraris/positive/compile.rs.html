10#[derive(Clone, Copy)]
11enum Kind {
12    Unsupported,
13    False,
14    True,
15    Atom,
16    And(usize),
17    Or,
18    Threshold(i128),
19}
20
21impl Kind {
22    fn monotone(self) -> bool {
23        !matches!(self, Self::Unsupported)
24    }
25    fn waiting(self) -> State {
26        match self {
27            Self::Atom | Self::Or => State::Awaiting(1),
28            Self::And(arity) => State::Awaiting(arity),
29            Self::Threshold(bound) => State::Weighted(bound),
30            Self::Unsupported | Self::False | Self::True => State::Inactive,
31        }
32    }
33}

35/// A vertex awaits its remaining incidences or has been queued exactly once.
36/// Inactive truth constants are activated explicitly during seeding; false and
37/// unsupported nodes remain inactive.
38#[derive(Clone, Copy, PartialEq, Eq)]
39pub(super) enum State {
40    Inactive,
41    Awaiting(usize),
42    Weighted(i128),
43    True,
44}

59pub(super) fn build(
60    theory: &Theory,
61    budget: &mut Budget<'_, '_>,
62) -> Result<(Interpretation, Option<usize>), PositiveError> {
63    budget.meter.poll()?;
64    let headers = size_of::<PositivePlan>()
65        + size_of::<Graph>()
66        + size_of::<Vec<Kind>>()
67        + size_of::<Vec<usize>>();
68    budget.observe_bytes(headers as u128)?;
69    let vertices = theory
70        .atom_count()
71        .checked_add(theory.nodes().len())
72        .ok_or(PositiveError::Overflow)?;
73    let kinds = classify(theory, budget)?;
74    let mut offsets = initialized(
75        vertices.checked_add(1).ok_or(PositiveError::Overflow)?,
76        0_usize,
77        budget,
78    )?;
79    edges(theory, &kinds, budget, |source, _, budget| {
80        budget.dependency()?;
81        offsets[source + 1] = offsets[source + 1]
82            .checked_add(1)
83            .ok_or(PositiveError::Overflow)?;
84        Ok(())
85    })?;
86    for index in 0..vertices {
87        budget.tick()?;
88        offsets[index + 1] = offsets[index + 1]
89            .checked_add(offsets[index])
90            .ok_or(PositiveError::Overflow)?;
91    }
92    let mut forward = initialized(
93        budget.statistics.dependencies,
94        Edge {
95            target: 0,
96            weight: 1,
97        },
98        budget,
99    )?;
100    let mut cursors = budget.reserve(vertices)?;
101    for &offset in &offsets[..vertices] {
102        budget.tick()?;
103        cursors.push(offset);
104    }
105    edges(theory, &kinds, budget, |source, edge, _| {
106        forward[cursors[source]] = edge;
107        cursors[source] += 1;
108        Ok(())
109    })?;
110    let mut states = budget.reserve(vertices)?;
111    for _ in 0..theory.atom_count() {
112        budget.tick()?;
113        states.push(State::Awaiting(1));
114    }
115    for kind in &kinds {
116        budget.tick()?;
117        states.push(kind.waiting());
118    }
119    budget.release(kinds)?;
120    budget.release(cursors)?;
121    let queue = budget.reserve(vertices)?;
122    let mut words = initialized(theory.atom_count().div_ceil(64), 0_u64, budget)?;
123    let mut graph = Graph {
124        offsets,
125        edges: forward,
126        states,
127        queue,
128    };
129    propagate::complete(theory, &mut graph, &mut words, budget)?;
130    drop(graph);
131    let least = Interpretation {
132        theory: theory.clone(),
133        words,
134    };
135    let failed = validation::complete(&least, budget)?;
136    Ok((least, failed))
137}

139fn initialized<T: Copy>(
140    count: usize,
141    value: T,
142    budget: &mut Budget<'_, '_>,
143) -> Result<Vec<T>, PositiveError> {
144    let mut values = budget.reserve(count)?;
145    for _ in 0..count {
146        budget.tick()?;
147        values.push(value);
148    }
149    Ok(values)
150}

152fn classify(theory: &Theory, budget: &mut Budget<'_, '_>) -> Result<Vec<Kind>, PositiveError> {
153    let mut kinds: Vec<Kind> = budget.reserve(theory.nodes().len())?;
154    let view = theory.view();
155    for index in 0..view.len() {
156        budget.tick()?;
157        kinds.push(match view.node(index).map_err(|_| Stop::InvalidProgram)? {
158            NodeView::Atom(_) => Kind::Atom,
159            NodeView::False => Kind::False,
160            NodeView::And(operands) => {
161                if monotone_operands(operands, &kinds, budget)? {
162                    Kind::And(operands.len())
163                } else {
164                    Kind::Unsupported
165                }
166            }
167            NodeView::Or(operands) => {
168                if monotone_operands(operands, &kinds, budget)? {
169                    Kind::Or
170                } else {
171                    Kind::Unsupported
172                }
173            }
174            NodeView::Threshold(threshold) => classify_threshold(threshold, &kinds, budget)?,
175            NodeView::ExtremumCut(cut) => classify_cut(cut, &kinds, budget)?,
176            NodeView::Implies(a, b) => {
177                budget.tick()?;
178                let left = view.node(a).map_err(|_| Stop::InvalidProgram)?;
179                budget.tick()?;
180                let right = view.node(b).map_err(|_| Stop::InvalidProgram)?;
181                if left == NodeView::False && right == NodeView::False {
182                    Kind::True
183                } else {
184                    Kind::Unsupported
185                }
186            }
187        });
188    }
189    Ok(kinds)
190}

192fn classify_cut(
193    cut: ExtremumCut<'_>,
194    kinds: &[Kind],
195    budget: &mut Budget<'_, '_>,
196) -> Result<Kind, PositiveError> {
197    if let Some(truth) = cut.endpoint_truth() {
198        return Ok(if truth { Kind::True } else { Kind::False });
199    }
200    let mut monotone = true;
201    for (&child, &rank) in cut.row.conditions.iter().zip(cut.row.ranks) {
202        budget.tick()?;
203        if cut.qualifies(rank) {
204            monotone &= kinds[child].monotone();
205        }
206    }
207    Ok(if monotone {
208        Kind::Or
209    } else {
210        Kind::Unsupported
211    })
212}

214fn classify_threshold(
215    threshold: Threshold<'_>,
216    kinds: &[Kind],
217    budget: &mut Budget<'_, '_>,
218) -> Result<Kind, PositiveError> {
219    let mut total = 0_i128;
220    let mut monotone = true;
221    for (&child, &weight) in threshold.conditions.iter().zip(threshold.weights) {
222        budget.tick()?;
223        total = total
224            .checked_add(i128::from(weight))
225            .ok_or(Stop::InvalidProgram)?;
226        if weight != 0 {
227            monotone &= kinds[child].monotone();
228        }
229    }
230    Ok(if threshold.bound <= 0 {
231        Kind::True
232    } else if threshold.bound > total {
233        Kind::False
234    } else if monotone {
235        Kind::Threshold(threshold.bound)
236    } else {
237        Kind::Unsupported
238    })
239}

241fn monotone_operands(
242    operands: &[usize],
243    kinds: &[Kind],
244    budget: &mut Budget<'_, '_>,
245) -> Result<bool, PositiveError> {
246    let mut monotone = true;
247    for &child in operands {
248        budget.tick()?;
249        monotone &= kinds[child].monotone();
250    }
251    Ok(monotone)
252}

254/// The same finite edge stream is counted then written into reserved CSR slots.
255/// Original roots are checked on both passes against the same immutable theory.
256/// Unsupported unreachable nodes do not supply edges; every accepted body has
257/// a complete monotone classification. No formula or atom occurrence is merged.
258fn edges(
259    theory: &Theory,
260    kinds: &[Kind],
261    budget: &mut Budget<'_, '_>,
262    mut visit: impl FnMut(usize, Edge, &mut Budget<'_, '_>) -> Result<(), PositiveError>,
263) -> Result<(), PositiveError> {
264    let atoms = theory.atom_count();
265    for (index, &kind) in kinds.iter().enumerate() {
266        budget.tick()?;
267        match (
268            kind,
269            theory
270                .view()
271                .node(index)
272                .map_err(|_| Stop::InvalidProgram)?,
273        ) {
274            (Kind::Atom, NodeView::Atom(atom)) => {
275                budget.tick()?;
276                visit(
277                    atom,
278                    Edge {
279                        target: atoms + index,
280                        weight: 1,
281                    },
282                    budget,
283                )?;
284            }
285            (Kind::And(_), NodeView::And(operands)) | (Kind::Or, NodeView::Or(operands)) => {
286                for &child in operands {
287                    budget.tick()?;
288                    visit(
289                        atoms + child,
290                        Edge {
291                            target: atoms + index,
292                            weight: 1,
293                        },
294                        budget,
295                    )?;
296                }
297            }
298            (Kind::Threshold(_), NodeView::Threshold(threshold)) => {
299                for (&child, &weight) in threshold.conditions.iter().zip(threshold.weights) {
300                    budget.tick()?;
301                    if weight != 0 {
302                        visit(
303                            atoms + child,
304                            Edge {
305                                target: atoms + index,
306                                weight,
307                            },
308                            budget,
309                        )?;
310                    }
311                }
312            }
313            (Kind::Or, NodeView::ExtremumCut(cut)) => {
314                for (&child, &rank) in cut.row.conditions.iter().zip(cut.row.ranks) {
315                    budget.tick()?;
316                    if cut.qualifies(rank) {
317                        visit(
318                            atoms + child,
319                            Edge {
320                                target: atoms + index,
321                                weight: 1,
322                            },
323                            budget,
324                        )?;
325                    }
326                }
327            }
328            _ => {}
329        }
330    }
331    for &root in theory.roots() {
332        budget.tick()?;
333        if let Some((body, head)) = producer(theory, root, kinds)? {
334            budget.tick()?;
335            visit(
336                atoms + body,
337                Edge {
338                    target: head,
339                    weight: 1,
340                },
341                budget,
342            )?;
343        }
344    }
345    Ok(())
346}

348fn producer(
349    theory: &Theory,
350    root: usize,
351    kinds: &[Kind],
352) -> Result<Option<(usize, usize)>, PositiveError> {
353    match theory.view().node(root).map_err(|_| Stop::InvalidProgram)? {
354        NodeView::Atom(_) | NodeView::False => Ok(None),
355        NodeView::Implies(_, head)
356            if theory.view().node(head).map_err(|_| Stop::InvalidProgram)? == NodeView::False =>
357        {
358            Ok(None)
359        }
360        NodeView::Implies(body, head) => {
361            let NodeView::Atom(atom) =
362                theory.view().node(head).map_err(|_| Stop::InvalidProgram)?
363            else {
364                return Err(PositiveError::UnsupportedRoot { root });
365            };
366            if !kinds[body].monotone() {
367                return Err(PositiveError::UnsupportedBody { root, body });
368            }
369            Ok(Some((body, atom)))
370        }
371        _ => Err(PositiveError::UnsupportedRoot { root }),
372    }
