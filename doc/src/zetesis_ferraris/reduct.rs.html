1//! A formula reduct bound to the interpretation that determines its falsum mask.
2
3use zetesis_cpu::{Cancellation, Stop};
4
5use crate::numeric_evaluation::NumericRowScratch;
6use crate::oracle::{Work, evaluate, failed_root, identities, reserve};
7use crate::{Interpretation, Limits, Theory};
8
9/// The Ferraris reduct of an immutable theory in one fixed interpretation.
10///
11/// This value borrows the exact candidate, whose theory handle fixes the original
12/// formula DAG. It owns one Boolean per DAG node: that node's classical truth in
13/// the candidate. Testing the reduct masks every candidate-false subformula to
14/// falsum, without constructing another DAG or reevaluating the candidate.
15///
16/// Freezing does not assert that the candidate models the original theory or is
17/// an answer set. A satisfaction query does not establish subset minimality.
18/// The candidate and mask remain immutable, so independent callers can share the
19/// reduct; each query reserves its own temporary evaluation workspace.
20///
21/// The mask cannot be supplied separately from the checked subject:
22/// ```compile_fail
23/// use zetesis_ferraris::{FrozenReduct, Interpretation};
24/// fn attach(candidate: &Interpretation, truth: Vec<bool>) -> FrozenReduct<'_> {
25///     FrozenReduct { candidate, truth }
26/// }
27/// ```
28#[derive(Debug)]
29pub struct FrozenReduct<'a> {
30    candidate: &'a Interpretation,
31    truth: Vec<bool>,
32}
33
34impl<'a> FrozenReduct<'a> {
35    /// Freeze the candidate's own theory with one topological evaluation.
36    ///
37    /// For N DAG nodes, this retains N Boolean values. Each Boolean
38    /// child occurrence is read; each referenced numeric row is read once.
39    /// Repeated positions within that row still count separately. If numeric nodes exist,
40    /// temporary dense measures cover R numeric rows with ceil(R/64) validity
41    /// words; initialization and each validity-word reset are charged. The
42    /// temporary cache is released after freezing; it is not part of the mask. The candidate's packed words
43    /// and theory are borrowed, not cloned. No root or proper subset is tested;
44    /// `limits.max_subsets` has no effect. Even an empty DAG polls control.
45    ///
46    /// # Errors
47    /// Returns cancellation, deadline, allocation, or work-limit stops. A stopped
48    /// freeze returns no reduct. Storage reservation is fallible and precedes
49    /// node evaluation, after the initial control poll.
50    pub fn new(
51        candidate: &'a Interpretation,
52        limits: Limits,
53        cancellation: &Cancellation,
54    ) -> Result<Self, Stop> {
55        cancellation
56            .metered(limits.max_work, |meter| {
57                Self::freeze_with(candidate, Work::new(limits.max_subsets, meter))
58            })
59            .result
60    }
61
62    /// [`Self::new`] charged to `work`.
63    fn freeze_with(candidate: &'a Interpretation, mut work: Work<'_, '_>) -> Result<Self, Stop> {
64        work.meter.poll()?;
65        let mut numeric = NumericRowScratch::default();
66        numeric.reserve(candidate.theory())?;
67        Self::freeze(candidate, &mut numeric, &mut work)
68    }
69
70    /// The exact interpretation determining this reduct, without allocation.
71    #[must_use]
72    pub const fn candidate(&self) -> &'a Interpretation {
73        self.candidate
74    }
75
76    /// The immutable original theory, without allocation or evaluation.
77    #[must_use]
78    pub fn theory(&self) -> &'a Theory {
79        self.candidate.theory()
80    }
81
82    /// Whether `tested` satisfies every asserted root of this frozen reduct.
83    ///
84    /// `tested` may be any interpretation of the same admitted theory instance;
85    /// it need not be a subset of the candidate. An independently admitted equal
86    /// theory has a different identity and is refused before control is polled.
87    ///
88    /// A query charges every node, Boolean child occurrence, and each
89    /// referenced numeric row once, then roots through the first failure. It
90    /// also charges fresh dense-measure initialization and packed-validity reset
91    /// when numeric nodes exist, as freezing does. Temporary truth and row-cache storage
92    /// are released on completion or stop. The
93    /// freeze is reused unchanged. This is a fresh per-query budget, excluding
94    /// construction and previous queries; `limits.max_subsets` has no effect.
95    ///
96    /// # Errors
97    /// Returns wrong-theory, cancellation, deadline, allocation, or work-limit
98    /// stops. A stop is not a logical rejection and leaves this reduct reusable.
99    pub fn is_satisfied_by(
100        &self,
101        tested: &Interpretation,
102        limits: Limits,
103        cancellation: &Cancellation,
104    ) -> Result<bool, Stop> {
105        cancellation
106            .metered(limits.max_work, |meter| {
107                self.satisfaction(tested, Work::new(limits.max_subsets, meter))
108            })
109            .result
110    }
111
112    /// [`Self::is_satisfied_by`] charged to `work`.
113    fn satisfaction(&self, tested: &Interpretation, mut work: Work<'_, '_>) -> Result<bool, Stop> {
114        identities(self.theory(), tested)?;
115        work.meter.poll()?;
116        let mut values = reserve(self.truth.len())?;
117        let mut numeric = NumericRowScratch::default();
118        numeric.reserve(self.theory())?;
119        self.satisfied_by(tested, &mut values, &mut numeric, &mut work)
120    }
121
122    // The one-shot wrapper reserves an equally sized tested workspace first,
123    // keeping both fallible reservations before candidate evaluation. Sharing
124    // Work preserves its combined freeze/query ceiling and stop precedence.
125    pub(super) fn freeze(
126        candidate: &'a Interpretation,
127        numeric: &mut NumericRowScratch,
128        work: &mut Work<'_, '_>,
129    ) -> Result<Self, Stop> {
130        let mut truth = reserve(candidate.theory().nodes().len())?;
131        evaluate(
132            candidate.theory(),
133            candidate,
134            None,
135            &mut truth,
136            numeric,
137            work,
138        )?;
139        Ok(Self { candidate, truth })
140    }
141
142    // Callers establish theory identity and reserve N values before this scan.
143    // The immutable truth mask and tested workspace refer to the same DAG order.
144    pub(super) fn satisfied_by(
145        &self,
146        tested: &Interpretation,
147        values: &mut Vec<bool>,
148        numeric: &mut NumericRowScratch,
149        work: &mut Work<'_, '_>,
150    ) -> Result<bool, Stop> {
151        evaluate(
152            self.theory(),
153            tested,
154            Some(&self.truth),
155            values,
156            numeric,
157            work,
158        )?;
159        Ok(failed_root(self.theory(), values, work)?.is_none())
160    }
161}
