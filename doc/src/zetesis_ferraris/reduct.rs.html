1//! A formula reduct bound to the interpretation that determines its falsum mask.
2
3use zetesis_cpu::{Cancellation, Stop};
4
5use crate::oracle::{Work, evaluate, failed_root, identities, reserve};
6use crate::{Interpretation, Limits, Statistics, Theory};
7
8/// The Ferraris reduct of an immutable theory in one fixed interpretation.
9///
10/// This value borrows the exact candidate, whose theory handle fixes the original
11/// formula DAG. It owns one Boolean per DAG node: that node's classical truth in
12/// the candidate. Testing the reduct masks every candidate-false subformula to
13/// falsum, without constructing another DAG or reevaluating the candidate.
14///
15/// Freezing does not assert that the candidate models the original theory or is
16/// an answer set. A satisfaction query does not establish subset minimality.
17/// The candidate and mask remain immutable, so independent callers can share the
18/// reduct; each query reserves its own temporary evaluation workspace.
19///
20/// The mask cannot be supplied separately from the checked subject:
21/// ```compile_fail
22/// use zetesis_ferraris::{FrozenReduct, Interpretation};
23/// fn attach(candidate: &Interpretation, truth: Vec<bool>) -> FrozenReduct<'_> {
24///     FrozenReduct { candidate, truth }
25/// }
26/// ```
27#[derive(Debug)]
28pub struct FrozenReduct<'a> {
29    candidate: &'a Interpretation,
30    truth: Vec<bool>,
31}
32
33impl<'a> FrozenReduct<'a> {
34    /// Freeze the candidate's own theory with one topological evaluation.
35    ///
36    /// For N DAG nodes and E logical child occurrences, this charges exactly
37    /// N + E operations and retains N Boolean values: O(N + E) work and O(N)
38    /// storage. Repeated children count separately. The candidate's packed words
39    /// and theory are borrowed, not cloned. No root or proper subset is tested;
40    /// `limits.max_subsets` has no effect. Even an empty DAG polls control.
41    ///
42    /// # Errors
43    /// Returns cancellation, deadline, allocation, or work-limit stops. A stopped
44    /// freeze returns no reduct. Storage reservation is fallible and precedes
45    /// node evaluation, after the initial control poll.
46    pub fn new(
47        candidate: &'a Interpretation,
48        limits: Limits,
49        cancellation: &Cancellation,
50    ) -> Result<Self, Stop> {
51        cancellation.poll()?;
52        let mut work = Work {
53            limits,
54            cancellation,
55            statistics: Statistics::default(),
56        };
57        Self::freeze(candidate, &mut work)
58    }
59
60    /// The exact interpretation determining this reduct, without allocation.
61    #[must_use]
62    pub const fn candidate(&self) -> &'a Interpretation {
63        self.candidate
64    }
65
66    /// The immutable original theory, without allocation or evaluation.
67    #[must_use]
68    pub fn theory(&self) -> &'a Theory {
69        self.candidate.theory()
70    }
71
72    /// Whether `tested` satisfies every asserted root of this frozen reduct.
73    ///
74    /// `tested` may be any interpretation of the same admitted theory instance;
75    /// it need not be a subset of the candidate. An independently admitted equal
76    /// theory has a different identity and is refused before control is polled.
77    ///
78    /// For N DAG nodes, E child occurrences and R asserted roots, a query charges
79    /// N + E evaluations and then one test per root through the first failure,
80    /// at most N + E + R work.
81    /// It uses N temporary Boolean values, released on completion or stop. The
82    /// freeze is reused unchanged. This is a fresh per-query budget, excluding
83    /// construction and previous queries; `limits.max_subsets` has no effect.
84    ///
85    /// # Errors
86    /// Returns wrong-theory, cancellation, deadline, allocation, or work-limit
87    /// stops. A stop is not a logical rejection and leaves this reduct reusable.
88    pub fn is_satisfied_by(
89        &self,
90        tested: &Interpretation,
91        limits: Limits,
92        cancellation: &Cancellation,
93    ) -> Result<bool, Stop> {
94        identities(self.theory(), tested)?;
95        cancellation.poll()?;
96        let mut work = Work {
97            limits,
98            cancellation,
99            statistics: Statistics::default(),
100        };
101        let mut values = reserve(self.truth.len())?;
102        self.satisfied_by(tested, &mut values, &mut work)
103    }
104
105    // The one-shot wrapper reserves an equally sized tested workspace first,
106    // keeping both fallible reservations before candidate evaluation. Sharing
107    // Work preserves its combined freeze/query ceiling and stop precedence.
108    pub(super) fn freeze(candidate: &'a Interpretation, work: &mut Work<'_>) -> Result<Self, Stop> {
109        let mut truth = reserve(candidate.theory().nodes().len())?;
110        evaluate(candidate.theory(), candidate, None, &mut truth, work)?;
111        Ok(Self { candidate, truth })
112    }
113
114    // Callers establish theory identity and reserve N values before this scan.
115    // The immutable truth mask and tested workspace refer to the same DAG order.
116    pub(super) fn satisfied_by(
117        &self,
118        tested: &Interpretation,
119        values: &mut Vec<bool>,
120        work: &mut Work<'_>,
121    ) -> Result<bool, Stop> {
122        evaluate(self.theory(), tested, Some(&self.truth), values, work)?;
123        Ok(failed_root(self.theory(), values, work)?.is_none())
124    }
125}
