1//! Regions of candidates over a formula theory, narrowed by its readings.
2//!
3//! A region (`zetesis_cpu::regions::Region`) holds some atoms in, cuts some
4//! out and leaves the rest open; here it decides over the theory's atoms. A
5//! formula has two readings under a region: it is *sure* when every seed of
6//! the region satisfies it and *never* when no seed does, with a held atom
7//! sure, a cut atom never, and the connectives combining the readings as
8//! the closure route's definite and possible gates do (`FormulaBounds.read`).
9//! The readings are what the knowledge below means: a node known to hold
10//! is sure, and a node known never is never.
11//!
12//! One narrowing closes, to a fixed point, what every candidate of the
13//! region must make of each node and each atom: known to hold, known to
14//! fail, or open. The rules are sound for stable models. Every root is
15//! known to hold and falsum to fail; a node learns from its operands as
16//! the connective dictates, and teaches its operands what its own
17//! knowledge leaves them, a conjunction known to hold both operands, a
18//! disjunction known to hold with one operand failing the other, an
19//! implication known to hold its consequent when its antecedent holds and
20//! the failure of its antecedent when its consequent fails, and the duals
21//! for a node known to fail; an atom node and its atom know the same,
22//! so a node shared by several parents is known once for all of them.
23//! A maximal tree of one connective, a clause or a body, is read as one
24//! node over its operands with the n-ary rules (`FormulaChains`), so a
25//! decision costs one step per occurrence of its atom.
26//! This is unit propagation over a clause form of the theory, read on the
27//! theory itself (`FormulaBounds.Known`, `known_sound`). In the producer
28//! fragment of the support restriction, an atom none of whose producers
29//! can support it, each having a body known to fail or another head known
30//! to hold, is known to fail (`unsupported_cut`), and an atom known to
31//! hold with exactly one producer able to support it forces that
32//! producer's body (`sole_support_forces`). A node or atom known both to
33//! hold and to fail refutes the region. The open atoms known are then
34//! held or cut.
35//!
36//! Work is charged per node read, per root tested and per producer checked,
37//! against `RegionLimits`, and control is polled once per narrowing. Metered
38//! entry points instead acquire each permit from the caller's quota, which can
39//! also poll control, and retain the admitted prefix on every returned failure.

110/// Extract the producers when every root is an implication, or a bare head,
111/// whose head is a disjunction of atoms with falsum, an atomic choice, or
112/// falsum: the fragment `DisjunctiveSupport.Covered` names, with a fact's
113/// head standing alone. No producers when a root lies outside it, in which
114/// case the support rule does not apply.
115///
116/// # Errors
117/// Returns the stop when extraction exceeds `limits.max_work`, control stops,
118/// or compact-adjacency storage cannot be represented or reserved. Other
119/// extraction storage retains its existing infallible allocation behavior.
120pub fn producers(
121    theory: &Theory,
122    limits: RegionLimits,
123    cancellation: &Cancellation,
124) -> Result<Extraction, Stop> {
125    cancellation.poll()?;
126    let mut work = Work::new(limits.max_work);
127    let producers = extract(theory, &mut work)?;
128    Ok(Extraction {
129        producers,
130        work: work.spent,
131    })
132}

134fn extract(theory: &Theory, work: &mut Work) -> Result<Option<Producers>, Stop> {
135    let nodes = theory.nodes();
136    let mut ordinary = Vec::with_capacity(nodes.len());
137    for node in nodes {
138        work.tick()?;
139        ordinary.push(match *node {
140            Node::Atom(_) | Node::False => true,
141            Node::Or(left, right) => ordinary[left] && ordinary[right],
142            _ => false,
143        });
144    }
145    let mut rules = Vec::new();
146    for &root in theory.roots() {
147        work.tick()?;
148        let (body, head) = match nodes[root] {
149            Node::Implies(body, head) => (Some(body), head),
150            _ => (None, root),
151        };
152        if matches!(nodes[head], Node::False) {
153            continue;
154        }
155        let (heads, choice) = if ordinary[head] {
156            let mut heads = BTreeSet::new();
157            let mut stack = vec![head];
158            while let Some(node) = stack.pop() {
159                work.tick()?;
160                match nodes[node] {
161                    Node::Atom(atom) => {
162                        heads.insert(atom);
163                    }
164                    Node::Or(left, right) => stack.extend([left, right]),
165                    _ => {}
166                }
167            }
168            (heads.into_iter().collect::<Vec<_>>(), false)
169        } else if let Some(atom) = crate::atomic_choice::atom(theory, head) {
170            (vec![atom], true)
171        } else {
172            return Ok(None);
173        };
174        rules.push(Producer {
175            body,
176            heads,
177            choice,
178        });
179    }
180    let by_head = Adjacency::build(
181        theory.atom_count(),
182        rules
183            .iter()
184            .enumerate()
185            .flat_map(|(index, producer)| producer.heads.iter().map(move |&atom| (atom, index))),
186    )?;
187    let by_body = Adjacency::build(
188        nodes.len(),
189        rules
190            .iter()
191            .enumerate()
192            .filter_map(|(index, producer)| producer.body.map(|body| (body, index))),
193    )?;
194    Ok(Some(Producers {
195        rules,
196        by_head,
197        by_body,
198    }))
199}

226struct Work<F = fn() -> Result<(), Stop>> {
227    spent: u64,
228    ceiling: u64,
229    charge: F,
230}
231impl Work {
232    fn new(ceiling: u64) -> Self {
233        Self {
234            spent: 0,
235            ceiling,
236            charge: || Ok(()),
237        }
238    }
239}
240impl<E: From<Stop>, F: FnMut() -> Result<(), E>> Work<F> {
241    fn metered(charge: F) -> Self {
242        Self {
243            spent: 0,
244            ceiling: u64::MAX,
245            charge,
246        }
247    }
248    fn tick(&mut self) -> Result<(), E> {
249        if self.spent >= self.ceiling {
250            return Err(Stop::WorkLimit.into());
251        }
252        (self.charge)()?;
253        self.spent += 1;
254        Ok(())
255    }
256}

258/// The shape of a theory a narrowing walks: each maximal tree of one
259/// connective read as one node, a *chain*, with its operands; the parents
260/// of each node that is not inside a chain, an implication or a chain root
261/// reading it as an operand; the nodes carrying each atom; and the atoms
262/// among each node's operands. Built once per theory from its nodes. The three
263/// immutable incidence maps store ordered rows as offsets
264/// into contiguous entry vectors, retaining every occurrence already chosen by
265/// chain formation. Compaction adds only linear construction passes.
266///
267/// A node is absorbed into its parent's chain when it has that one parent,
268/// the same connective, and is not a root of the theory; every other
269/// conjunction or disjunction is the root of its own chain, of two operands
270/// at least. An absorbed node has no knowledge of its own: the chain's
271/// readings are the n-ary readings of its operands (`FormulaChains`), and
272/// a node false under a frozen mask is either an operand, which fails, or
273/// an absorbed node whose operands all fail (disjunction) or one of which
274/// fails (conjunction), so the operands' masks already read it.
275#[derive(Clone, Debug)]
276pub struct Narrower {
277    parents: Adjacency,
278    atom_nodes: Adjacency,
279    chains: Vec<Chain>,
280    /// The chain a node roots, stored one-based so absence needs no extra word.
281    chain_of: Vec<Option<NonZeroUsize>>,
282    /// Whether a node is inside a chain, with no knowledge of its own.
283    absorbed: Vec<bool>,
284    /// The atoms among a node's operands, for the split ranking.
285    atom_operands: Adjacency,
286}

288/// A maximal tree of one connective, read as one node over its operands.

290struct Chain {
291    disjunction: bool,
292    root: usize,
293    /// The leaves, distinct nodes, in operand order.
294    operands: Vec<usize>,
295}

297/// A one-based chain link. Both construction maps point into allocated chain
298/// vectors, whose positions cannot reach `usize::MAX`; checked encoding retains
299/// that invariant without imposing a smaller theory-size admission limit.
300fn encoded_chain(position: usize) -> NonZeroUsize {
301    NonZeroUsize::new(position.checked_add(1).expect("allocated chain position"))
302        .expect("positive chain link")
303}

305/// Recover the zero-based position used by chain operands and counters.
306fn chain_position(link: NonZeroUsize) -> usize {
307    link.get() - 1
308}

310/// The chains of a theory: each conjunction or disjunction not absorbed
311/// into its parent is a root, with the chain it roots and, per node,
312/// whether the node is absorbed. Operands precede their parents, so a
313/// node's chain is complete when its parent is reached: a same-connective
314/// operand that is not a root of the theory and has this one parent joins
315/// the parent's chain, its own dissolving into it.
316fn chains(theory: &Theory) -> (Vec<Chain>, Vec<Option<NonZeroUsize>>, Vec<bool>) {
317    let nodes = theory.nodes();
318    let mut roots = vec![false; nodes.len()];
319    for &root in theory.roots() {
320        roots[root] = true;
321    }
322    let mut parent_count = vec![0usize; nodes.len()];
323    for node in nodes {
324        match *node {
325            Node::Atom(_) | Node::False => {}
326            Node::And(a, b) | Node::Or(a, b) | Node::Implies(a, b) => {
327                parent_count[a] += 1;
328                if b != a {
329                    parent_count[b] += 1;
330                }
331            }
332        }
333    }
334    let mut built: Vec<Chain> = Vec::new();
335    let mut chain_of = vec![None; nodes.len()];
336    let mut absorbed = vec![false; nodes.len()];
337    for (index, node) in nodes.iter().enumerate() {
338        let (disjunction, a, b) = match *node {
339            Node::Or(a, b) => (true, a, b),
340            Node::And(a, b) => (false, a, b),
341            _ => continue,
342        };
343        let mut operands = Vec::new();
344        // A node reached through both sides is one operand, seen once: seen
345        // twice, an absorbed one would return as an opaque second operand,
346        // its chain already dissolved, and the chain would wait for
347        // knowledge it never gets.
348        let sides: &[usize] = if b == a { &[a] } else { &[a, b] };
349        for &operand in sides {
350            let inner = if roots[operand] || parent_count[operand] != 1 {
351                None
352            } else {
353                chain_of[operand]
354                    .map(chain_position)
355                    .filter(|&k| built[k].disjunction == disjunction)
356            };
357            // The operands stay distinct: a node reached twice, through
358            // both sides, is one operand, as it is one node of the DAG.
359            if let Some(inner) = inner {
360                chain_of[operand] = None;
361                absorbed[operand] = true;
362                for leaf in std::mem::take(&mut built[inner].operands) {
363                    if !operands.contains(&leaf) {
364                        operands.push(leaf);
365                    }
366                }
367            } else if !operands.contains(&operand) {
368                operands.push(operand);
369            }
370        }
371        chain_of[index] = Some(encoded_chain(built.len()));
372        built.push(Chain {
373            disjunction,
374            root: index,
375            operands,
376        });
377    }
378    // Dissolved chains keep their slot, emptied; renumber the live ones.
379    let mut live = Vec::with_capacity(built.len());
380    let mut renumbered = vec![None; built.len()];
381    for (old, chain) in built.into_iter().enumerate() {
382        if !absorbed[chain.root] {
383            renumbered[old] = Some(encoded_chain(live.len()));
384            live.push(chain);
385        }
386    }
387    for entry in &mut chain_of {
388        *entry = entry.and_then(|old| renumbered[chain_position(old)]);
389    }
390    (live, chain_of, absorbed)
391}

393/// What a narrowing knows about a region, carried from a region to its
394/// children: the knowledge of a region holds in every region inside it
395/// (`FormulaBounds.known_mono`), so a child's narrowing starts from its
396/// parent's knowledge and learns only what the split decided. A fresh
397/// value knows nothing and is seeded in full on first use.
398///
399/// A value is linear in the theory: a bit pair over the nodes and one over
400/// the atoms, two counters per chain, a count per atom and the worklists.
401/// A split that offers one child to another worker clones it, so the
402/// clone is the split's cost. A knowledge belongs to the narrower that
403/// made it and to the region it was closed for: narrowing a region with a
404/// knowledge from another narrower, theory, producer set or frozen mask,
405/// or from a region not enclosing it, is unsound, and nothing checks it.
406/// The proposers keep the invariant by carrying each region's knowledge
407/// from its parent and by narrowing with the narrower that made it.
408#[derive(Clone, Debug)]
409pub struct Knowledge {
410    known: Known,
411}
412
413impl Knowledge {
414    /// Header, owned flag, seen and counter arrays, and worklist capacities in
415    /// bytes, including empty worklists' retained capacity. The shared theory, narrower
416    /// and producer index are excluded, as are allocator bookkeeping and
417    /// temporary clones.
418    #[must_use]
419    pub fn retained_bytes(&self) -> u128 {
420        let known = &self.known;
421        let flag_words = [
422            known.sure.len(),
423            known.never.len(),
424            known.atom_sure.len(),
425            known.atom_never.len(),
426        ];
427        let indices = [known.learned.capacity(), known.heads.capacity()];
428        let counters = [&known.sure_operands, &known.never_operands, &known.unknown];
429        size_of::<Self>() as u128
430            + flag_words.into_iter().map(|n| n as u128).sum::<u128>() * size_of::<u64>() as u128
431            + indices.into_iter().map(|n| n as u128).sum::<u128>() * size_of::<usize>() as u128
432            + counters
433                .into_iter()
434                .map(Counters::allocated_bytes)
435                .sum::<u128>()
436            + known.nodes.capacity() as u128 * size_of::<(usize, bool)>() as u128
437            + known.seen.len() as u128 * size_of::<u64>() as u128
438    }
439}

444/// The ordered operand-to-parent incidences already chosen by chain formation.
445/// Chains precede implications as in the original index; a repeated implication
446/// operand is one incidence. Distinct atom nodes carrying the same atom remain
447/// distinct occurrences when this stream is projected to atom operands.
448fn dependencies<'a>(
449    nodes: &'a [Node],
450    chains: &'a [Chain],
451) -> impl Iterator<Item = (usize, usize)> + Clone + 'a {
452    chains
453        .iter()
454        .flat_map(|chain| {
455            chain
456                .operands
457                .iter()
458                .map(move |&operand| (operand, chain.root))
459        })
460        .chain(nodes.iter().enumerate().flat_map(|(index, node)| {
461            let operands = match *node {
462                Node::Implies(a, b) => [Some(a), (b != a).then_some(b)],
463                _ => [None, None],
464            };
465            operands
466                .into_iter()
467                .flatten()
468                .map(move |operand| (operand, index))
469        }))
470}

472impl Narrower {
473    /// Index the theory's DAG for narrowing. Use [`Self::try_new`] to receive
474    /// compact-adjacency capacity and allocation refusals as [`Stop`].
475    ///
476    /// # Panics
477    /// Panics if compact adjacency cannot be represented or reserved. Other
478    /// index construction retains its existing infallible allocation behavior.
479    #[must_use]
480    pub fn new(theory: &Theory) -> Self {
481        Self::try_new(theory).expect("region adjacency storage could not be reserved")
482    }
483
484    /// Index the same DAG using checked compact adjacency construction.
485    /// Each incidence map reads its immutable edge stream twice and uses linear
486    /// row scans, preserving row order and duplicates. The existing logical
487    /// index receipt remains one visit per theory node; construction passes and
488    /// storage initialization are not additional charged propagation reads.
489    ///
490    /// # Errors
491    /// Returns [`Stop::Allocation`] for compact-adjacency offset overflow or
492    /// reservation failure. Chain construction and later knowledge allocation
493    /// retain their existing infallible behavior; this is not universal OOM
494    /// recovery. This constructor has no independent cancellation contract.
495    pub fn try_new(theory: &Theory) -> Result<Self, Stop> {
496        let nodes = theory.nodes();
497        let (chains, chain_of, absorbed) = chains(theory);
498        let parents = Adjacency::build(nodes.len(), dependencies(nodes, &chains))?;
499        let atom_operands = Adjacency::build(
500            nodes.len(),
501            dependencies(nodes, &chains).filter_map(|(operand, parent)| match nodes[operand] {
502                Node::Atom(atom) => Some((parent, atom)),
503                _ => None,
504            }),
505        )?;
506        let atom_nodes = Adjacency::build(
507            theory.atom_count(),
508            nodes
509                .iter()
510                .enumerate()
511                .filter_map(|(index, node)| match *node {
512                    Node::Atom(atom) => Some((atom, index)),
513                    _ => None,
514                }),
515        )?;
516        Ok(Self {
517            parents,
518            atom_nodes,
519            chains,
520            chain_of,
521            absorbed,
522            atom_operands,
523        })
524    }

526    /// The work indexing charged: one visit per node.
527    #[must_use]
528    pub fn work(&self) -> u64 {
529        self.parents.len() as u64
530    }

532    /// Knowledge of nothing, for the root of a tree over this theory.
533    #[must_use]
534    pub fn knowledge(&self) -> Knowledge {
535        // Every chain operand and every parent counted for an atom is an
536        // occurrence in this same incidence stream. Its total bounds all
537        // three counter arrays; no theory-size or language cap is imposed.
538        let incidences = self.parents.entry_count();
539        let mut unknown = Counters::zeros(self.atom_nodes.len(), incidences);
540        for (atom, nodes) in self.atom_nodes.iter().enumerate() {
541            for &node in nodes {
542                unknown.add(atom, self.parents[node].len());
543            }
544        }
545        Knowledge {
546            known: Known::empty(self.parents.len(), self.chains.len(), incidences, unknown),
547        }
548    }

550    /// Narrow the region to the fixed point of the closure's rules from
551    /// what is already known about it, [`Self::knowledge`] for the root and
552    /// the parent's knowledge for a child, learning only the decisions the
553    /// knowledge has not seen, and leave the knowledge closed for the
554    /// region's children.
555    ///
556    /// `knowledge` is this narrower's, closed for a region enclosing this
557    /// one under the same `producers` and `frozen` mask, as [`Knowledge`]
558    /// states; the narrowing cannot tell a foreign knowledge from its own.
559    ///
560    /// # Errors
561    /// Returns the stop when the narrowing exceeds its work or propagation
562    /// ceiling, or control stops it. The region then holds the decisions the
563    /// closure had learned before the stop, and the knowledge is partly
564    /// closed: both sound, neither to be reused; the proposers abandon the
565    /// region.
566    pub fn narrow_known(
567        &self,
568        theory: &Theory,
569        producers: Option<&Producers>,
570        region: &mut Region,
571        knowledge: &mut Knowledge,
572        limits: RegionLimits,
573        cancellation: &Cancellation,
574    ) -> Result<(Narrowing, NarrowingStatistics), Stop> {
575        let subject = Subject {
576            theory,
577            producers,
578            frozen: None,
579        };
580        let attempt = self.narrow_with(
581            subject,
582            region,
583            knowledge,
584            Work::new(limits.max_work),
585            cancellation,
586        );
587        attempt
588            .result
589            .map(|narrowing| (narrowing, attempt.statistics))
590    }

645        limits: RegionLimits,
646        cancellation: &Cancellation,
647    ) -> Result<(Narrowing, NarrowingStatistics), Stop> {
648        let subject = Subject {
649            theory,
650            producers: None,
651            frozen: Some(truth),
652        };
653        let attempt = self.narrow_with(
654            subject,
655            region,
656            knowledge,
657            Work::new(limits.max_work),
658            cancellation,
659        );
660        attempt
661            .result
662            .map(|narrowing| (narrowing, attempt.statistics))
663    }

665    /// Narrow a frozen reduct with the quota and failure receipt contract of
666    /// [`Self::narrow_known_metered`]. The frozen mask and ancestor knowledge
667    /// retain the preconditions of [`Self::narrow_frozen_known`].
668    pub fn narrow_frozen_known_metered<E: From<Stop>>(
669        &self,
670        theory: &Theory,
671        truth: &[bool],
672        region: &mut Region,
673        knowledge: &mut Knowledge,
674        cancellation: &Cancellation,
675        charge: impl FnMut() -> Result<(), E>,
676    ) -> NarrowingAttempt<E> {
677        let subject = Subject {
678            theory,
679            producers: None,
680            frozen: Some(truth),
681        };
682        self.narrow_with(
683            subject,
684            region,
685            knowledge,
686            Work::metered(charge),
687            cancellation,
688        )

691    fn narrow_with<E: From<Stop>>(
692        &self,
693        subject: Subject<'_>,
694        region: &mut Region,
695        knowledge: &mut Knowledge,
696        mut work: Work<impl FnMut() -> Result<(), E>>,
697        cancellation: &Cancellation,
698    ) -> NarrowingAttempt<E> {
699        let mut statistics = NarrowingStatistics::default();
700        let result = (|| {
701            cancellation.poll().map_err(E::from)?;
702            let known = &mut knowledge.known;
703            if known.close(subject, self, region, &mut work, &mut statistics)?
704                == Step::Contradiction
705            {
706                return Ok(Narrowing::Refuted);
707            }
708            // The atoms this closure learned decide the region; the region's own
709            // decisions, the split's and those made here, are then all seen.
710            let mut changed = false;
711            for atom in known.learned.drain(..) {
712                let was_open = region.is_open(atom);
713                let decided = if bit(&known.atom_sure, atom) {
714                    statistics.held += u64::from(was_open);
715                    region.hold(atom)
716                } else {
717                    statistics.cut += u64::from(was_open);
718                    region.cut(atom)
719                };
720                debug_assert!(decided, "a learned atom agrees with the region");
721                changed |= was_open;
722            }
723            region.snapshot_decided(&mut known.seen);
724            if let Some(atom) = most_constrained(region, known, &mut work)? {
725                region.prefer(atom);
726            }
727            Ok(Narrowing::Fixed { changed })
728        })();
729        statistics.work = work.spent;
730        NarrowingAttempt { result, statistics }
731    }
732}

734/// The open atom with the most parents still unknown, the one whose
735/// decision the theory is most sensitive to; ties go to the lower atom.
736/// This chooses the split, as the clause search branches on the variable
737/// with the most unresolved occurrences. The counts are kept as parents
738/// become known, so the ranking is one read per open atom.
739fn most_constrained<E: From<Stop>>(
740    region: &Region,
741    known: &Known,
742    work: &mut Work<impl FnMut() -> Result<(), E>>,
743) -> Result<Option<usize>, E> {
744    let mut best: Option<(usize, usize)> = None;
745    for atom in region.open() {
746        work.tick()?;
747        let unknown = known.unknown.get(atom);
748        if best.is_none_or(|(_, count)| unknown > count) {
749            best = Some((atom, unknown));
750        }
751    }
752    Ok(best.map(|(atom, _)| atom))
753}

755/// What every candidate of the region must make of each node and each
756/// atom: known to hold, known to fail, or open. One bit pair over the
757/// nodes, one over the atoms, and two counters over each chain, the
758/// operands known to hold and the operands known to fail, closed under the
759/// upward rules from a node's operands and the downward rules from a node's
760/// own knowledge until nothing changes: unit propagation on the theory
761/// itself, with a node shared by several parents known once for all of
762/// them (`FormulaBounds.Known`, `known_sound`; the chain rules are
763/// `FormulaChains`). The closure is driven by a worklist: a node that
764/// learns something is revisited once, and only its parents, operands and
765/// dependent producers are read, a chain learning from an operand by one
766/// counter step.
767#[derive(Clone, Debug)]
768struct Known {
769    sure: Box<[u64]>,
770    never: Box<[u64]>,
771    atom_sure: Box<[u64]>,
772    atom_never: Box<[u64]>,
773    /// Per chain, the operands known to hold.
774    sure_operands: Counters,
775    /// Per chain, the operands known to fail.
776    never_operands: Counters,
777    /// Per atom, the parents of its nodes not yet known: the split ranking.
778    /// A parent is counted once here and taken off once when it is
779    /// revisited, so the count never goes below zero.
780    unknown: Counters,
781    /// The atoms this closure decided, not yet told to the region.
782    learned: Vec<usize>,
783    /// Nodes that learned something, with what, and have not been revisited.
784    nodes: Vec<(usize, bool)>,
785    /// Atoms whose support must be rechecked; queued only when producers
786    /// are known, since only they say what supports an atom.
787    heads: Vec<usize>,
788    /// The region's decided-mask snapshot already told to this closure; new
789    /// decisions are the region's decided atoms not set here.
790    seen: Box<[u64]>,
791    /// The roots, falsum and every atom's support have been seeded once;
792    /// later closures learn only decisions not yet known.
793    seeded: bool,
794}
795
796/// What a step of the closure did.
797#[derive(Clone, Copy, PartialEq, Eq)]
798enum Step {
799    Unchanged,
800    Changed,
801    /// A node or atom became known both to hold and to fail.
802    Contradiction,
803}
804
805impl Step {
806    fn join(self, other: Self) -> Self {
807        match (self, other) {
808            (Self::Contradiction, _) | (_, Self::Contradiction) => Self::Contradiction,
809            (Self::Changed, _) | (_, Self::Changed) => Self::Changed,
810            (Self::Unchanged, Self::Unchanged) => Self::Unchanged,
811        }
812    }
813}
814
815/// The number of 64-bit words a bitset over `n` flags needs.
816const fn flag_words(n: usize) -> usize {
817    n.div_ceil(64)
818}
819
820/// Whether flag `index` is set in a bitset.
821#[inline]
822fn bit(mask: &[u64], index: usize) -> bool {
823    mask[index / 64] & (1u64 << (index % 64)) != 0
824}
825
826fn learn(known: &mut [u64], opposite: &[u64], index: usize) -> Step {
827    let (word, flag) = (index / 64, 1u64 << (index % 64));
828    if known[word] & flag != 0 {
829        Step::Unchanged
830    } else if opposite[word] & flag != 0 {
831        Step::Contradiction
832    } else {
833        known[word] |= flag;
834        Step::Changed
835    }
836}

838impl Known {
839    fn empty(nodes: usize, chains: usize, incidences: usize, unknown: Counters) -> Self {
840        let seen = vec![0u64; unknown.len().div_ceil(64)].into_boxed_slice();
841        Self {
842            sure: vec![0; flag_words(nodes)].into_boxed_slice(),
843            never: vec![0; flag_words(nodes)].into_boxed_slice(),
844            atom_sure: vec![0; flag_words(unknown.len())].into_boxed_slice(),
845            atom_never: vec![0; flag_words(unknown.len())].into_boxed_slice(),
846            sure_operands: Counters::zeros(chains, incidences),
847            never_operands: Counters::zeros(chains, incidences),
848            unknown,
849            learned: Vec::new(),
850            nodes: Vec::new(),
851            heads: Vec::new(),
852            seen,
853            seeded: false,
854        }
855    }
856
857    /// A node learns to hold; it is revisited if that is new.
858    fn sure(&mut self, index: usize) -> Step {
859        let step = learn(&mut self.sure, &self.never, index);
860        if step == Step::Changed {
861            self.nodes.push((index, true));
862        }
863        step
864    }
865
866    /// A node learns to fail; it is revisited if that is new.
867    fn never(&mut self, index: usize) -> Step {
868        let step = learn(&mut self.never, &self.sure, index);
869        if step == Step::Changed {
870            self.nodes.push((index, false));
871        }
872        step
873    }

875    /// An atom learns to hold or to fail, and every node carrying it
876    /// learns the same. A newly held atom has its support rechecked when
877    /// producers are known.
878    fn atom(
879        &mut self,
880        index: &Narrower,
881        producers: Option<&Producers>,
882        atom: usize,
883        value: bool,
884    ) -> Step {
885        let step = if value {
886            learn(&mut self.atom_sure, &self.atom_never, atom)
887        } else {
888            learn(&mut self.atom_never, &self.atom_sure, atom)
889        };
890        if step != Step::Changed {
891            return step;
892        }
893        self.learned.push(atom);
894        let mut step = step;
895        for &node in &index.atom_nodes[atom] {
896            step = step.join(if value {
897                self.sure(node)
898            } else {
899                self.never(node)
900            });
901        }
902        if value && producers.is_some() {
903            self.heads.push(atom);
904        }
905        step
906    }

908    /// Close the knowledge from the region's decisions, falsum and the
909    /// roots. Each event on the worklist follows a new bit, or is one of
910    /// the initial seeds, so the events are bounded by the bits.
911    fn close<E: From<Stop>>(
912        &mut self,
913        subject: Subject<'_>,
914        index: &Narrower,
915        region: &Region,
916        work: &mut Work<impl FnMut() -> Result<(), E>>,
917        statistics: &mut NarrowingStatistics,
918    ) -> Result<Step, E> {
919        let Subject {
920            theory,
921            producers,
922            frozen,
923        } = subject;
924        let nodes = theory.nodes();
925        let mut step = Step::Unchanged;
926        if !self.seeded {
927            for (node, kind) in nodes.iter().enumerate() {
928                if index.absorbed[node] {
929                    continue;
930                }
931                let falsum =
932                    matches!(kind, Node::False) || frozen.is_some_and(|truth| !truth[node]);
933                if falsum {
934                    step = step.join(self.never(node));
935                }
936            }
937            for &root in theory.roots() {
938                step = step.join(self.sure(root));
939            }
940            if producers.is_some() {
941                self.heads.extend(0..theory.atom_count());
942            }
943            self.seeded = true;
944        }
945        // Take the seen mask out so the new-decision iterator borrows the local
946        // rather than `self`, leaving `self.atom` free to mutate the closure;
947        // the swap moves a box pointer and copies nothing.
948        let mut seen = std::mem::take(&mut self.seen);
949        for (atom, value) in region.decided_since(&seen) {
950            step = step.join(self.atom(index, producers, atom, value));
951        }
952        region.snapshot_decided(&mut seen);
953        self.seen = seen;
954        if step == Step::Contradiction {
955            return Ok(step);
956        }
957        loop {
958            let step = if let Some((node, value)) = self.nodes.pop() {
959                statistics.propagations += 1;
960                self.revisit(subject, index, node, value, work)?
961            } else if let Some(atom) = self.heads.pop() {
962                statistics.propagations += 1;
963                let producers =
964                    producers.expect("a support recheck is queued only when producers are known");
965                self.recheck(index, producers, atom, work)?
966            } else {
967                return Ok(Step::Changed);
968            };
969            if step == Step::Contradiction {
970                return Ok(step);
971            }
972        }
973    }

975    /// A node that learned something teaches its operands, and lets each
976    /// parent learn from it: a chain by one counter step, an implication
977    /// from both its operands and, when already known, by teaching them
978    /// again, since what it leaves them may have narrowed. A node false
979    /// under a frozen mask is falsum in the reduct, a constant with no
980    /// operands: it teaches nothing and learns nothing from them, and a
981    /// parent under the mask likewise.
982    fn revisit<E: From<Stop>>(
983        &mut self,
984        subject: Subject<'_>,
985        index: &Narrower,
986        node: usize,
987        value: bool,
988        work: &mut Work<impl FnMut() -> Result<(), E>>,
989    ) -> Result<Step, E> {
990        work.tick()?;
991        let Subject {
992            theory,
993            producers,
994            frozen,
995        } = subject;
996        let nodes = theory.nodes();
997        let masked = |node: usize| frozen.is_some_and(|truth| !truth[node]);
998        for &atom in &index.atom_operands[node] {
999            self.unknown.decrement(atom);
1000        }
1001        let mut step = Step::Unchanged;
1002        if !masked(node) {
1003            step = step.join(self.teach_operands(nodes, index, producers, node));
1004        }
1005        for &parent in &index.parents[node] {
1006            work.tick()?;
1007            if masked(parent) {
1008                continue;
1009            }
1010            step = step.join(if let Some(chain) = index.chain_of[parent] {
1011                self.operand_changed(index, chain_position(chain), value)
1012            } else {
1013                let up = self.learn_from_operands(nodes, parent);
1014                if bit(&self.sure, parent) || bit(&self.never, parent) {
1015                    up.join(self.teach_operands(nodes, index, producers, parent))
1016                } else {
1017                    up
1018                }
1019            });
1020        }
1021        Ok(step)
1022    }

1024    /// A chain learns from an operand that became known: one more operand
1025    /// holds or fails. A disjunction holds with one, fails with all, and a
1026    /// disjunction known to hold with all but one failing forces that one;
1027    /// a conjunction dually (`disj_chain_sure`, `disj_chain_never`,
1028    /// `disj_chain_unit` and the conjunction laws).
1029    fn operand_changed(&mut self, index: &Narrower, chain: usize, value: bool) -> Step {
1030        let Chain {
1031            disjunction,
1032            root,
1033            ref operands,
1034        } = index.chains[chain];
1035        let total = operands.len();
1036        if value {
1037            self.sure_operands.add(chain, 1);
1038        } else {
1039            self.never_operands.add(chain, 1);
1040        }
1041        let (sure, never) = (
1042            self.sure_operands.get(chain),
1043            self.never_operands.get(chain),
1044        );
1045        match (disjunction, value) {
1046            (true, true) => self.sure(root),
1047            (true, false) if never == total => self.never(root),
1048            (true, false) if bit(&self.sure, root) && never + 1 == total => self.unit(index, chain),
1049            (false, false) => self.never(root),
1050            (false, true) if sure == total => self.sure(root),
1051            (false, true) if bit(&self.never, root) && sure + 1 == total => self.unit(index, chain),
1052            _ => Step::Unchanged,
1053        }
1054    }

1056    /// The one operand of a chain not yet known learns what the chain's
1057    /// own knowledge leaves it: to hold, in a disjunction known to hold
1058    /// whose others fail; to fail, in a conjunction known to fail whose
1059    /// others hold. The operands are scanned once for it; a node's bit is
1060    /// set when it learns and counted when it is revisited, so the scan
1061    /// may find none, every operand being known with one count pending,
1062    /// and then the pending step decides the chain.
1063    fn unit(&mut self, index: &Narrower, chain: usize) -> Step {
1064        let Chain {
1065            disjunction,
1066            ref operands,
1067            ..
1068        } = index.chains[chain];
1069        let open = operands.iter().copied().find(|&operand| {
1070            if disjunction {
1071                !bit(&self.never, operand)
1072            } else {
1073                !bit(&self.sure, operand)
1074            }
1075        });
1076        match open {
1077            Some(operand) if disjunction => self.sure(operand),
1078            Some(operand) => self.never(operand),
1079            None => Step::Unchanged,
1080        }
1081    }

1083    /// A known node teaches its operands what its knowledge leaves them,
1084    /// tells its atom, and, when it is a body that fails, has the
1085    /// producers' heads rechecked. A chain known to hold forces its one
1086    /// open operand (disjunction) or every operand (conjunction); known to
1087    /// fail, every operand (disjunction) or its one open operand
1088    /// (conjunction).
1089    fn teach_operands(
1090        &mut self,
1091        nodes: &[Node],
1092        index: &Narrower,
1093        producers: Option<&Producers>,
1094        node: usize,
1095    ) -> Step {
1096        let mut step = Step::Unchanged;
1097        if let Some(chain) = index.chain_of[node] {
1098            step = step.join(self.teach_chain(index, chain_position(chain)));
1099        }
1100        if bit(&self.sure, node) {
1101            step = step.join(match nodes[node] {
1102                Node::Atom(atom) => {
1103                    // A held head blocks the other heads of its producers.
1104                    if let Some(producers) = producers {
1105                        for &producer in &producers.by_head[atom] {
1106                            self.heads.extend(
1107                                producers.rules[producer]
1108                                    .heads
1109                                    .iter()
1110                                    .copied()
1111                                    .filter(|&head| head != atom),
1112                            );
1113                        }
1114                    }
1115                    self.atom(index, producers, atom, true)
1116                }
1117                Node::False => Step::Contradiction,
1118                Node::And(..) | Node::Or(..) => Step::Unchanged,
1119                Node::Implies(a, b) => {
1120                    if bit(&self.sure, a) {
1121                        self.sure(b)
1122                    } else if bit(&self.never, b) {
1123                        self.never(a)
1124                    } else {
1125                        Step::Unchanged
1126                    }
1127                }
1128            });
1129        }
1130        if bit(&self.never, node) {
1131            step = step.join(match nodes[node] {
1132                Node::Atom(atom) => self.atom(index, producers, atom, false),
1133                Node::False | Node::And(..) | Node::Or(..) => Step::Unchanged,
1134                Node::Implies(a, b) => self.sure(a).join(self.never(b)),
1135            });
1136            if let Some(producers) = producers {
1137                for &producer in &producers.by_body[node] {
1138                    self.heads
1139                        .extend(producers.rules[producer].heads.iter().copied());
1140                }
1141            }
1142        }
1143        step
1144    }

1146    /// What a chain's own knowledge leaves its operands.
1147    fn teach_chain(&mut self, index: &Narrower, chain: usize) -> Step {
1148        let Chain {
1149            disjunction,
1150            root,
1151            ref operands,
1152        } = index.chains[chain];
1153        let total = operands.len();
1154        let mut step = Step::Unchanged;
1155        if bit(&self.sure, root) {
1156            if disjunction {
1157                if self.never_operands.get(chain) + 1 == total {
1158                    step = step.join(self.unit(index, chain));
1159                }
1160            } else {
1161                for &operand in operands {
1162                    step = step.join(self.sure(operand));
1163                }
1164            }
1165        }
1166        if bit(&self.never, root) {
1167            if disjunction {
1168                for &operand in operands {
1169                    step = step.join(self.never(operand));
1170                }
1171            } else if self.sure_operands.get(chain) + 1 == total {
1172                step = step.join(self.unit(index, chain));
1173            }
1174        }
1175        step
1176    }

1178    /// An implication learns from its operands what the connective
1179    /// dictates; chains learn by their counters.
1180    fn learn_from_operands(&mut self, nodes: &[Node], node: usize) -> Step {
1181        match nodes[node] {
1182            Node::Implies(a, b) => {
1183                let mut up = Step::Unchanged;
1184                if bit(&self.never, a) || bit(&self.sure, b) {
1185                    up = up.join(self.sure(node));
1186                }
1187                if bit(&self.sure, a) && bit(&self.never, b) {
1188                    up = up.join(self.never(node));
1189                }
1190                up
1191            }
1192            Node::Atom(_) | Node::False | Node::And(..) | Node::Or(..) => Step::Unchanged,
1193        }
1194    }

1196    /// An atom none of its producers can support is known to fail, and an
1197    /// atom known to hold with exactly one producer able to support it
1198    /// forces that producer's body (`unsupported_cut`,
1199    /// `sole_support_forces`). A producer can support its atom when its
1200    /// body is not known to fail and, unless it is a choice, no other of
1201    /// its heads is known to hold.
1202    fn recheck<E: From<Stop>>(
1203        &mut self,
1204        index: &Narrower,
1205        producers: &Producers,
1206        atom: usize,
1207        work: &mut Work<impl FnMut() -> Result<(), E>>,
1208    ) -> Result<Step, E> {
1209        if bit(&self.atom_never, atom) {
1210            return Ok(Step::Unchanged);
1211        }
1212        let mut supporters = 0;
1213        let mut sole = None;
1214        for &producer in &producers.by_head[atom] {
1215            work.tick()?;
1216            let producer = &producers.rules[producer];
1217            let body_impossible = producer.body.is_some_and(|body| bit(&self.never, body));
1218            let other_held = !producer.choice
1219                && producer
1220                    .heads
1221                    .iter()
1222                    .any(|&head| head != atom && bit(&self.atom_sure, head));
1223            if !body_impossible && !other_held {
1224                supporters += 1;
1225                sole = producer.body;
1226            }
1227        }
1228        Ok(match (supporters, sole) {
1229            (0, _) => self.atom(index, Some(producers), atom, false),
1230            (1, Some(body)) if bit(&self.atom_sure, atom) => self.sure(body),
1231            _ => Step::Unchanged,
1232        })
1233    }
