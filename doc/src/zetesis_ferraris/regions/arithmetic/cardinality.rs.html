75        size_of::<Self>() as u128
76            + match self {
77                Self::RootRefuted { .. } => 0,
78                Self::Families(families) => {
79                    families.pairs.capacity() as u128 * size_of::<Pair>() as u128
80                        + families.occurrences.capacity() as u128 * size_of::<Occurrence>() as u128
81                        + families.index.as_ref().map_or(0, |index| {
82                            index.words.capacity() as u128 * size_of::<u64>() as u128
83                        })
84                }
85            }
86    }

89/// Independent of discovery: authenticate directions and rebuild both unions.
90/// Equality of union cardinality and the sum of row cardinalities establishes
91/// disjointness, including duplicate row occurrences. XOR checks equal carriers.
92fn check_pair<Q: Quota + ?Sized>(
93    bounds: &[Bound],
94    lower: &[usize],
95    upper: &[usize],
96    unions: (&mut [u64], &mut [u64]),
97    budget: &mut Budget<'_, '_, Q>,
98) -> Result<bool, BudgetError<Q::Error>> {
99    let (left, right) = unions;
100    clear(left, budget)?;
101    clear(right, budget)?;
102    let Some(lower_cardinality) = union_side(bounds, lower, BoundSide::Lower, left, budget)? else {
103        return Ok(false);
104    };
105    let Some(upper_cardinality) = union_side(bounds, upper, BoundSide::Upper, right, budget)?
106    else {
107        return Ok(false);
108    };
109    let cardinality = u64::try_from(words::count(left, &mut |work| budget.charge(work))?)
110        .map_err(|_| BudgetError::Stopped(Stop::WorkLimit))?;
111    budget.charge(1)?;
112    if lower_cardinality != cardinality {
113        return Ok(false);
114    }
115    budget.charge(1)?;
116    if upper_cardinality != cardinality {
117        return Ok(false);
118    }
119    words::xor_assign(left, right, &mut || budget.charge(1))?;
120    let remaining = words::first(left, &mut |work| budget.charge(work))?;
121    budget.charge(1)?;
122    if remaining.is_some() {
123        return Ok(false);
124    }
125    Ok(true)
126}

128fn union_side<Q: Quota + ?Sized>(
129    bounds: &[Bound],
130    rows: &[usize],
131    side: BoundSide,
132    union: &mut [u64],
133    budget: &mut Budget<'_, '_, Q>,
134) -> Result<Option<u64>, BudgetError<Q::Error>> {
135    budget.charge(1)?;
136    let mut cardinality = 0_u64;
137    for &row in rows {
138        budget.charge(1)?;
139        let bound = bounds
140            .get(row)
141            .ok_or(BudgetError::Source(AdmissionError::Edge))?;
142        budget.charge(1)?;
143        if bound.side != side {
144            return Ok(None);
145        }
146        words::union_assign(union, &bound.support, &mut |work| budget.charge(work))?;
147        budget.charge(2)?;
148        let Some(next) = cardinality.checked_add(bound.cardinality) else {
149            return Ok(None);
150        };
151        cardinality = next;
152    }
153    budget.charge(1)?;
154    Ok(Some(cardinality))
155}

157fn effective<Q: Quota + ?Sized>(
158    bound: &Bound,
159    budget: &mut Budget<'_, '_, Q>,
160) -> Result<u64, BudgetError<Q::Error>> {
161    budget.charge(2)?; // original guard emptiness and authenticated side
162    let fallback = match bound.side {
163        BoundSide::Lower => 0,
164        BoundSide::Upper => bound.cardinality,
165    };
166    if bound.guard.is_empty() {
167        budget.charge(2)?; // bound comparison and selected value
168        Ok(match bound.side {
169            BoundSide::Lower => bound.value.max(fallback),
170            BoundSide::Upper => bound.value.min(fallback),
171        })
172    } else {
173        budget.charge(1)?;
174        Ok(fallback)
175    }
176}

178fn reduce_side<Q: Quota + ?Sized>(
179    bounds: &[Bound],
180    occurrences: &mut [Occurrence],
181    pair: usize,
182    side: BoundSide,
183    budget: &mut Budget<'_, '_, Q>,
184) -> Result<Option<u64>, BudgetError<Q::Error>> {
185    budget.charge(1)?;
186    let mut sum = 0_u64;
187    for occurrence in occurrences {
188        budget.charge(1)?;
189        let bound = bounds
190            .get(occurrence.row)
191            .ok_or(BudgetError::Source(AdmissionError::Edge))?;
192        budget.charge(1)?;
193        if occurrence.pair != pair {
194            return Err(BudgetError::Stopped(Stop::InvalidProgram));
195        }
196        budget.charge(1)?;
197        if occurrence.side != side {
198            return Err(BudgetError::Stopped(Stop::InvalidProgram));
199        }
200        let value = effective(bound, budget)?;
201        budget.charge(2)?;
202        occurrence.value = value;
203        if occurrence.value > bound.cardinality {
204            return Ok(None);
205        }
206        budget.charge(2)?;
207        sum = sum
208            .checked_add(occurrence.value)
209            .ok_or(BudgetError::Stopped(Stop::InvalidProgram))?;
210    }
211    budget.charge(1)?;
212    Ok(Some(sum))
213}

215/// C0 reads the same occurrence cache and pair ranges retained for later C1.
216fn reduce_pair<Q: Quota + ?Sized>(
217    families: &mut Families,
218    bounds: &[Bound],
219    pair: usize,
220    budget: &mut Budget<'_, '_, Q>,
221) -> Result<bool, BudgetError<Q::Error>> {
222    budget.charge(4)?;
223    let lower = families.pairs[pair].lower.clone();
224    let upper = families.pairs[pair].upper.clone();
225    let Some(lower) = reduce_side(
226        bounds,
227        &mut families.occurrences[lower],
228        pair,
229        BoundSide::Lower,
230        budget,
231    )?
232    else {
233        return Ok(true);
234    };
235    let Some(upper) = reduce_side(
236        bounds,
237        &mut families.occurrences[upper],
238        pair,
239        BoundSide::Upper,
240        budget,
241    )?
242    else {
243        return Ok(true);
244    };
245    budget.charge(4)?;
246    families.pairs[pair].lower_sum = lower;
247    families.pairs[pair].upper_sum = upper;
248    Ok(families.pairs[pair].lower_sum > families.pairs[pair].upper_sum)
249}

282/// Stable bottom-up merging, reusing one initialized caller-owned buffer.
283/// Each level moves each record once; an odd final level copies it back once.
284pub(super) fn sort<'a, T: Copy, Q: Quota + ?Sized>(
285    values: &'a mut [T],
286    scratch: &'a mut [T],
287    words: u64,
288    mut compare: impl FnMut(&T, &T, &mut Budget<'_, '_, Q>) -> Result<Ordering, BudgetError<Q::Error>>,
289    budget: &mut Budget<'_, '_, Q>,
290) -> Result<(), BudgetError<Q::Error>> {
291    let len = values.len();
292    if len < 2 {
293        return Ok(());
294    }
295    let mut source = values;
296    let mut target = &mut scratch[..len];
297    budget.charge(1)?;
298    let mut copied = false;
299    let mut width = 1usize;
300    while width < len {
301        for first in (0..len).step_by(width.saturating_mul(2)) {
302            let middle = first.saturating_add(width).min(len);
303            let end = middle.saturating_add(width).min(len);
304            let (mut left, mut right) = (first, middle);
305            for slot in &mut target[first..end] {
306                let take_left = if right == end {
307                    true
308                } else if left == middle {
309                    false
310                } else {
311                    let order = compare(&source[left], &source[right], budget)?;
312                    budget.charge(1)?;
313                    order != Ordering::Greater
314                };
315                budget.charge(words)?;
316                if take_left {
317                    *slot = source[left];
318                    left += 1;
319                } else {
320                    *slot = source[right];
321                    right += 1;
322                }
323            }
324        }
325        budget.charge(5)?; // two borrowed slice handles and the parity flag
326        std::mem::swap(&mut source, &mut target);
327        copied = !copied;
328        width = width.saturating_mul(2);
329    }
330    if copied {
331        for (target, source) in target.iter_mut().zip(source.iter()) {
332            budget.charge(words)?;
333            *target = *source;
334        }
335    }
336    Ok(())
337}

339fn scalar<T: Copy + Ord, Q: Quota + ?Sized>(
340    left: T,
341    right: T,
342    budget: &mut Budget<'_, '_, Q>,
343) -> Result<Ordering, BudgetError<Q::Error>> {
344    budget.charge(1)?;
345    if left < right {
346        return Ok(Ordering::Less);
347    }
348    budget.charge(1)?;
349    Ok(if left > right {
350        Ordering::Greater
351    } else {
352        Ordering::Equal
353    })
354}

356fn literal_order<Q: Quota + ?Sized>(
357    left: &Literal,
358    right: &Literal,
359    budget: &mut Budget<'_, '_, Q>,
360) -> Result<Ordering, BudgetError<Q::Error>> {
361    let atom = scalar(left.atom, right.atom, budget)?;
362    budget.charge(1)?;
363    if atom == Ordering::Equal {
364        scalar(left.positive, right.positive, budget)
365    } else {
366        Ok(atom)
367    }
368}

370/// Sorting keys alone remove repeated signed occurrences. Source guards stay
371/// untouched, including both polarities of an atom and all original repeats.
372fn guard_keys<Q: Quota + ?Sized>(
373    bounds: &[Bound],
374    storage: &mut Storage,
375    budget: &mut Budget<'_, '_, Q>,
376) -> Result<GuardKeys, BudgetError<Q::Error>> {
377    budget.charge(2)?;
378    let (mut total, mut maximum) = (0usize, 0usize);
379    for row in bounds {
380        budget.charge(2)?;
381        total = total
382            .checked_add(row.guard.len())
383            .ok_or(BudgetError::Stopped(Stop::Allocation))?;
384        budget.charge(2)?;
385        maximum = maximum.max(row.guard.len());
386    }
387    let mut keys = storage.reserve(bounds.len(), budget)?;
388    let mut literals = storage.reserve(total, budget)?;
389    let mut scratch = storage.filled(
390        maximum,
391        Literal {
392            atom: 0,
393            positive: false,
394        },
395        2,
396        budget,
397    )?;
398    for row in bounds {
399        budget.charge(1)?;
400        let start = literals.len();
401        for literal in &row.guard {
402            budget.charge(3)?;
403            literals.push(*literal);
404        }
405        sort(
406            &mut literals[start..],
407            &mut scratch,
408            2,
409            literal_order,
410            budget,
411        )?;
412        budget.charge(1)?;
413        let mut count = 0usize;
414        for index in start..literals.len() {
415            budget.charge(3)?; // copy the pair and test whether a predecessor exists
416            let literal = literals[index];
417            let duplicate = if count == 0 {
418                false
419            } else {
420                budget.charge(2)?;
421                let previous = literals[start + count - 1];
422                budget.charge(1)?;
423                if previous.atom == literal.atom {
424                    budget.charge(1)?;
425                    previous.positive == literal.positive
426                } else {
427                    false
428                }
429            };
430            budget.charge(1)?;
431            if !duplicate {
432                budget.charge(3)?;
433                literals[start + count] = literal;
434                count += 1;
435            }
436        }
437        budget.charge(3)?;
438        keys.push(GuardKey { start, len: count });
439    }
440    storage.release(scratch)?;
441    Ok(GuardKeys { keys, literals })
442}

444fn find<Q: Quota + ?Sized>(
445    parents: &[usize],
446    row: usize,
447    budget: &mut Budget<'_, '_, Q>,
448) -> Result<usize, BudgetError<Q::Error>> {
449    budget.charge(1)?;
450    let mut current = row;
451    loop {
452        budget.charge(2)?;
453        let next = parents[current];
454        if next == current {
455            return Ok(current);
456        }
457        budget.charge(1)?;
458        current = next;
459    }
460}

462fn unite<Q: Quota + ?Sized>(
463    parents: &mut [usize],
464    sizes: &mut [usize],
465    left: usize,
466    right: usize,
467    budget: &mut Budget<'_, '_, Q>,
468) -> Result<(), BudgetError<Q::Error>> {
469    let left = find(parents, left, budget)?;
470    let right = find(parents, right, budget)?;
471    budget.charge(1)?;
472    if left == right {
473        return Ok(());
474    }
475    budget.charge(3)?;
476    let (winner, loser) = if sizes[left] < sizes[right] {
477        (right, left)
478    } else {
479        (left, right)
480    };
481    budget.charge(2)?;
482    parents[loser] = winner;
483    sizes[winner] += sizes[loser]; // components partition at most bounds.len() rows
484    Ok(())
485}

487fn components<Q: Quota + ?Sized>(
488    bounds: &[Bound],
489    atom_count: usize,
490    storage: &mut Storage,
491    budget: &mut Budget<'_, '_, Q>,
492) -> Result<Vec<usize>, BudgetError<Q::Error>> {
493    let mut first = storage.filled(atom_count, 0usize, 1, budget)?;
494    let mut parents = storage.reserve(bounds.len(), budget)?;
495    for row in 0..bounds.len() {
496        budget.charge(2)?;
497        parents.push(row);
498    }
499    let mut sizes = storage.filled(bounds.len(), 1usize, 1, budget)?;
500    for (row, bound) in bounds.iter().enumerate() {
501        budget.charge(2)?;
502        let encoded = row
503            .checked_add(1)
504            .ok_or(BudgetError::Stopped(Stop::Allocation))?;
505        words::visit_set_bits(
506            &bound.support,
507            budget,
508            &mut |budget, work| budget.charge(work),
509            &mut |budget, atom| {
510                budget.charge(1)?;
511                if atom >= atom_count {
512                    return Err(BudgetError::Source(AdmissionError::Atom));
513                }
514                budget.charge(1)?;
515                if first[atom] == 0 {
516                    budget.charge(1)?;
517                    first[atom] = encoded;
518                } else {
519                    budget.charge(1)?;
520                    unite(&mut parents, &mut sizes, row, first[atom] - 1, budget)?;
521                }
522                Ok(())
523            },
524        )?;
525    }
526    storage.release(first)?;
527    storage.release(sizes)?;
528    let mut result = storage.reserve(bounds.len(), budget)?;
529    for row in 0..bounds.len() {
530        let component = find(&parents, row, budget)?;
531        budget.charge(2)?;
532        result.push(component);
533    }
534    storage.release(parents)?;
535    Ok(result)
536}

554/// Equal-cardinality supports compare by their increasing atom sequences.
555/// At the first differing word, the support containing its least differing bit
556/// comes first. No secondary atom-sequence representation is constructed.
557fn support_order<Q: Quota + ?Sized>(
558    left: &[u64],
559    right: &[u64],
560    budget: &mut Budget<'_, '_, Q>,
561) -> Result<Ordering, BudgetError<Q::Error>> {
562    for (&left, &right) in left.iter().zip(right) {
563        budget.charge(1)?;
564        if left != right {
565            budget.charge(5)?;
566            let first = 1u64 << Words::<u64>::first_bit(left ^ right);
567            return Ok(if left & first != 0 {
568                Ordering::Less
569            } else {
570                Ordering::Greater
571            });
572        }
573    }
574    Ok(Ordering::Equal)
575}

577struct Keys<'a> {
578    bounds: &'a [Bound],
579    guards: &'a [GuardKey],
580    literals: &'a [Literal],
581}
582
583impl Keys<'_> {
584    fn compare<Q: Quota + ?Sized>(
585        &self,
586        left: usize,
587        right: usize,
588        descending: bool,
589        budget: &mut Budget<'_, '_, Q>,
590    ) -> Result<Ordering, BudgetError<Q::Error>> {
591        let a = &self.bounds[left];
592        let b = &self.bounds[right];
593        let count = scalar(a.cardinality, b.cardinality, budget)?;
594        budget.charge(1)?;
595        if count != Ordering::Equal {
596            return Ok(if descending { count.reverse() } else { count });
597        }
598        let support = support_order(&a.support, &b.support, budget)?;
599        budget.charge(1)?;
600        if support != Ordering::Equal {
601            return Ok(support);
602        }
603        budget.charge(4)?;
604        let a_key = self.guards[left];
605        let b_key = self.guards[right];
606        for (a, b) in self.literals[a_key.start..a_key.start + a_key.len]
607            .iter()
608            .zip(&self.literals[b_key.start..b_key.start + b_key.len])
609        {
610            let order = literal_order(a, b, budget)?;
611            budget.charge(1)?;
612            if order != Ordering::Equal {
613                return Ok(order);
614            }
615        }
616        let guard = scalar(a_key.len, b_key.len, budget)?;
617        budget.charge(1)?;
618        if guard != Ordering::Equal {
619            return Ok(guard);
620        }
621        let value = scalar(a.value, b.value, budget)?;
622        budget.charge(1)?;
623        if value != Ordering::Equal {
624            return Ok(value);
625        }
626        let kind = scalar(kind(a.source.kind), kind(b.source.kind), budget)?;
627        budget.charge(1)?;
628        if kind != Ordering::Equal {
629            return Ok(kind);
630        }
631        let root = scalar(a.source.root, b.source.root, budget)?;
632        budget.charge(1)?;
633        if root != Ordering::Equal {
634            return Ok(root);
635        }
636        scalar(a.source.node, b.source.node, budget)
637    }
638}

640fn greedy<Q: Quota + ?Sized>(
641    bounds: &[Bound],
642    rows: &[usize],
643    selected: &mut Vec<usize>,
644    union: &mut [u64],
645    storage: &mut Storage,
646    budget: &mut Budget<'_, '_, Q>,
647) -> Result<Range<usize>, BudgetError<Q::Error>> {
648    clear(union, budget)?;
649    budget.charge(1)?;
650    let first = selected.len();
651    for &row in rows {
652        let disjoint =
653            words::disjoint(union, &bounds[row].support, &mut |work| budget.charge(work))?;
654        budget.charge(1)?;
655        if disjoint {
656            words::union_assign(union, &bounds[row].support, &mut |work| budget.charge(work))?;
657            storage.push(selected, row, 1, budget)?;
658        }
659    }
660    budget.charge(2)?;
661    Ok(first..selected.len())
662}

664/// Build the four lists in lower-ascending, lower-descending, upper-ascending,
665/// upper-descending order, then publish their prescribed Cartesian attempts.
666/// The caller already admitted the four-range workspace; all borrowed order
667/// slices refer to the same carrier and this helper retains no new owner.
668fn propose_carrier<Q: Quota + ?Sized>(
669    bounds: &[Bound],
670    orders: [&[usize]; 4],
671    selected: &mut Vec<usize>,
672    proposals: &mut Vec<Proposal>,
673    union: &mut [u64],
674    storage: &mut Storage,
675    budget: &mut Budget<'_, '_, Q>,
676) -> Result<(), BudgetError<Q::Error>> {
677    let lists = [
678        greedy(bounds, orders[0], selected, union, storage, budget)?,
679        greedy(bounds, orders[1], selected, union, storage, budget)?,
680        greedy(bounds, orders[2], selected, union, storage, budget)?,
681        greedy(bounds, orders[3], selected, union, storage, budget)?,
682    ];
683    for (lower, upper) in [(0, 2), (0, 3), (1, 2), (1, 3)] {
684        storage.push(
685            proposals,
686            Proposal {
687                lower: lists[lower].clone(),
688                upper: lists[upper].clone(),
689            },
690            4,
691            budget,
692        )?;
693    }
694    Ok(())
695}

711/// Complete the four deterministic proposals for every support carrier before
712/// checking them. Neither a guarded atom nor greedy success changes a carrier.
713fn discover<Q: Quota + ?Sized>(
714    bounds: &[Bound],
715    atom_count: usize,
716    counts: &mut Counts,
717    storage: &mut Storage,
718    budget: &mut Budget<'_, '_, Q>,
719) -> Result<PartitionProposals, BudgetError<Q::Error>> {
720    let components = components(bounds, atom_count, storage, budget)?;
721    let guards = guard_keys(bounds, storage, budget)?;
722    let mut members = row_indices(bounds.len(), storage, budget)?;
723    let mut scratch = storage.filled(bounds.len(), 0usize, 1, budget)?;
724    sort(
725        &mut members,
726        &mut scratch,
727        1,
728        |&a, &b, budget| {
729            let order = scalar(components[a], components[b], budget)?;
730            budget.charge(1)?;
731            if order == Ordering::Equal {
732                scalar(side(bounds[a].side), side(bounds[b].side), budget)
733            } else {
734                Ok(order)
735            }
736        },
737        budget,
738    )?;
739    let mut descending = storage.reserve(bounds.len(), budget)?;
740    for &row in &members {
741        budget.charge(2)?;
742        descending.push(row);
743    }
744    let mut selected = storage.reserve(0, budget)?;
745    let mut proposals = storage.reserve(0, budget)?;
746    let mut union = Words::new(storage.filled(atom_count.div_ceil(128) * 2, 0u64, 1, budget)?);
747    storage.admit_record(size_of::<[Range<usize>; 4]>() as u128, budget)?;
748    let keys = Keys {
749        bounds,
750        guards: &guards.keys,
751        literals: &guards.literals,
752    };
753    let mut first = 0;
754    while first < members.len() {
755        budget.charge(2)?;
756        counts.carriers += 1;
757        let component = components[members[first]];
758        let mut end = first;
759        while end < members.len() {
760            budget.charge(1)?;
761            if components[members[end]] != component {
762                break;
763            }
764            end += 1;
765        }
766        let mut middle = first;
767        while middle < end {
768            budget.charge(1)?;
769            if bounds[members[middle]].side != BoundSide::Lower {
770                break;
771            }
772            middle += 1;
773        }
774        for range in [first..middle, middle..end] {
775            sort(
776                &mut members[range.clone()],
777                &mut scratch,
778                1,
779                |&a, &b, budget| keys.compare(a, b, false, budget),
780                budget,
781            )?;
782            sort(
783                &mut descending[range],
784                &mut scratch,
785                1,
786                |&a, &b, budget| keys.compare(a, b, true, budget),
787                budget,
788            )?;
789        }
790        propose_carrier(
791            bounds,
792            [
793                &members[first..middle],
794                &descending[first..middle],
795                &members[middle..end],
796                &descending[middle..end],
797            ],
798            &mut selected,
799            &mut proposals,
800            &mut union,
801            storage,
802            budget,
803        )?;
804        first = end;
805    }
806    storage.release_record(size_of::<[Range<usize>; 4]>() as u128)?;
807    storage.release(union.into_storage())?;
808    storage.release(descending)?;
809    storage.release(members)?;
810    storage.release(scratch)?;
811    storage.release(components)?;
812    storage.release(guards.keys)?;
813    storage.release(guards.literals)?;
814    Ok(PartitionProposals {
815        selected,
816        proposals,
817    })
818}

820fn retain<Q: Quota + ?Sized>(
821    families: &mut Families,
822    lower: &[usize],
823    upper: &[usize],
824    storage: &mut Storage,
825    budget: &mut Budget<'_, '_, Q>,
826) -> Result<usize, BudgetError<Q::Error>> {
827    budget.charge(2)?;
828    let pair = families.pairs.len();
829    let first = families.occurrences.len();
830    for &row in lower {
831        storage.push(
832            &mut families.occurrences,
833            Occurrence {
834                row,
835                pair,
836                side: BoundSide::Lower,
837                value: 0,
838            },
839            4,
840            budget,
841        )?;
842    }
843    budget.charge(1)?;
844    let middle = families.occurrences.len();
845    for &row in upper {
846        storage.push(
847            &mut families.occurrences,
848            Occurrence {
849                row,
850                pair,
851                side: BoundSide::Upper,
852                value: 0,
853            },
854            4,
855            budget,
856        )?;
857    }
858    budget.charge(1)?;
859    let end = families.occurrences.len();
860    storage.push(
861        &mut families.pairs,
862        Pair {
863            lower: first..middle,
864            upper: middle..end,
865            lower_sum: 0,
866            upper_sum: 0,
867        },
868        6,
869        budget,
870    )?;
871    Ok(pair)
872}

874/// Prepare the approved two-order greedy policy, then independently check each
875/// of its four pair attempts per carrier. This is deliberately incomplete for
876/// general partition discovery; rejected proposals have no semantic effect.
877///
878/// Union by size without compression bounds each find by log2(row count).
879/// Sorting uses stable merges and exact canonical ties. Temporary guard keys
880/// are flat offsets over copied literals. Every copy, numeric reduction and
881/// allocated capacity is charged; no unmetered library sort or search runs.
882/// Shared disjointness stops at its first overlap, so rejected scans can cost
883/// less than the reference's complete-word overlap reduction. Effective values
884/// and pair sums are computed directly in the retained occurrence records.
885///
886/// Input rows are independently source-checked, with support cardinalities and
887/// zero padding for this atom universe. Any error discards the whole result.
888/// The first checked root contradiction returns before later checks or index
889/// construction. All remaining family occurrences get the shared incidence
890/// index, including every original guard occurrence and each repeated pair use.
891pub(super) fn prepare<Q: Quota + ?Sized>(
892    bounds: &[Bound],
893    atom_count: usize,
894    storage: &mut Storage,
895    budget: &mut Budget<'_, '_, Q>,
896) -> Result<Prepared, BudgetError<Q::Error>> {
897    let width = atom_count.div_ceil(128) * 2;
898    for bound in bounds {
899        budget.charge(1)?;
900        if bound.support.len() != width {
901            return Err(BudgetError::Source(AdmissionError::Span));
902        }
903    }
904    storage.admit_record(size_of::<Prepared>() as u128, budget)?;
905    let pairs = storage.reserve(0, budget)?;
906    let occurrences = storage.reserve(0, budget)?;
907    storage.embed::<Pair>()?;
908    storage.embed::<Occurrence>()?;
909    budget.charge(5)?; // index absence and four observation counters
910    let mut families = Families {
911        pairs,
912        occurrences,
913        index: None,
914        counts: Counts::default(),
915    };
916    budget.charge(1)?;
917    if bounds.is_empty() {
918        return Ok(Prepared::Families(families));
919    }
920    let PartitionProposals {
921        selected,
922        proposals,
923    } = budget.in_phase(super::work::Phase::Discovery, |budget| {
924        discover(bounds, atom_count, &mut families.counts, storage, budget)
925    })?;
926    let mut lower_union = Words::new(storage.filled(width, 0u64, 1, budget)?);
927    let mut upper_union = Words::new(storage.filled(width, 0u64, 1, budget)?);
928    for proposal in &proposals {
929        budget.charge(1)?;
930        families.counts.attempts += 1;
931        let lower = &selected[proposal.lower.clone()];
932        let upper = &selected[proposal.upper.clone()];
933        let valid = budget.in_phase(super::work::Phase::CandidateChecks, |budget| {
934            check_pair(
935                bounds,
936                lower,
937                upper,
938                (&mut lower_union, &mut upper_union),
939                budget,
940            )
941        })?;
942        budget.charge(1)?;
943        if !valid {
944            budget.charge(1)?;
945            families.counts.rejected += 1;
946            continue;
947        }
948        budget.charge(1)?;
949        families.counts.accepted += 1;
950        let pair = retain(&mut families, lower, upper, storage, budget)?;
951        let refuted = reduce_pair(&mut families, bounds, pair, budget)?;
952        budget.charge(1)?;
953        if refuted {
954            budget.charge(1)?;
955            storage.release(lower_union.into_storage())?;
956            storage.release(upper_union.into_storage())?;
957            storage.release(selected)?;
958            storage.release(proposals)?;
959            storage.release_embedded(families.pairs)?;
960            storage.release_embedded(families.occurrences)?;
961            return Ok(Prepared::RootRefuted {
962                counts: families.counts,
963            });
964        }
965    }
966    storage.release(lower_union.into_storage())?;
967    storage.release(upper_union.into_storage())?;
968    storage.release(selected)?;
969    storage.release(proposals)?;
970    let index = budget.in_phase(super::work::Phase::Index, |budget| {
971        let mut index = parity::build_index(
972            atom_count,
973            families
974                .occurrences
975                .iter()
976                .map(|occurrence| &bounds[occurrence.row].support[..]),
977            storage,
978            budget,
979        )?;
980        for (position, occurrence) in families.occurrences.iter().enumerate() {
981            for literal in &bounds[occurrence.row].guard {
982                budget.charge(1)?;
983                if literal.atom >= atom_count {
984                    return Err(BudgetError::Source(AdmissionError::Atom));
985                }
986                index.insert(literal.atom, position, budget)?;
987            }
988        }
989        Ok(index)
990    })?;
991    storage.release_record(size_of::<Index>() as u128)?;
992    budget.charge(2)?;
993    families.index = Some(index);
994    Ok(Prepared::Families(families))
995}
