1//! Incremental exclusion by the immutable checked cardinality families.
2//!
3//! A lineage retains its last complete input snapshot and two sums per pair.
4//! Each selected occurrence is reconstructed at that snapshot and the current
5//! region. Replacing a prefix mixes bounds that all apply to the descendant;
6//! their first contradiction excludes it. Prepared occurrences stay immutable.

23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24struct PairSums {
25    lower: u64,
26    upper: u64,
27}
28
29/// Last complete input and pair sums for one prepared cardinality system.
30///
31/// Knowledge may be reused only with the same prepared system, atom universe
32/// and a descendant region. It is not transferable merely because two systems
33/// have the same theory. Its private snapshot detects width and descent faults,
34/// but does not authenticate the identity of a different prepared system.
35/// After a failed or refuted attempt, abandon the partial knowledge. A sibling
36/// starts from its parent's complete copy, never the other sibling's progress.
37#[derive(Clone, Debug)]
38pub struct CardinalityKnowledge {
39    snapshot: Snapshot,
40    sums: Vec<PairSums>,
41}
42
43impl CardinalityKnowledge {
44    fn empty() -> Self {
45        Self {
46            snapshot: Snapshot::default(),
47            sums: Vec::new(),
48        }
49    }
50
51    /// Copy initialized lineage values into an independent destination,
52    /// reusing sufficient existing capacity. The source is never changed.
53    /// On failure, `destination` is `None`; partial copied values are unusable.
54    ///
55    /// This helper polls at entry but neither refunds nor adds an exit poll.
56    /// The enclosing copy attempt accounts the Meter delta and refunds its
57    /// unused permits. Ordinary `Clone` has no metered-work guarantee.
58    ///
59    /// # Errors
60    /// Preserves cancellation, allocation and the caller's typed quota error.
61    pub fn copy_reserved<Q: Quota + ?Sized>(
62        &self,
63        destination: &mut Option<Self>,
64        cancellation: &Cancellation,
65        work: &mut Meter<'_, Q>,
66    ) -> Result<(), Failure<Q::Error>> {
67        copy_attempt(
68            self,
69            destination,
70            cancellation,
71            work,
72            |work| {
73                charge(work, 7)?; // snapshot Vec/flag and sums Vec headers
74                Ok(Self::empty())
75            },
76            |target, source, work| {
77                target.snapshot.copy_from(&source.snapshot, work)?;
78                copy_values(&mut target.sums, &source.sums, 2, work)
79            },
80        )
81    }
82
83    /// This record and actual owned capacities, excluding the immutable system
84    /// and the per-worker scratch. Embedded vector headers are counted once.
85    #[must_use]
86    pub fn retained_bytes(&self) -> u128 {
87        size_of::<Self>() as u128
88            + self.snapshot.allocated_bytes()
89            + self.sums.capacity() as u128 * size_of::<PairSums>() as u128
90    }
91}

93impl CardinalitySystem {
94    /// Allocate no storage and inspect no rows. The first metered attempt
95    /// initializes this system's input snapshot and prepared pair sums.
96    #[must_use]
97    pub fn knowledge(&self) -> CardinalityKnowledge {
98        CardinalityKnowledge::empty()
99    }
100
101    /// Inspect the affected checked family occurrences once, in stored order.
102    ///
103    /// A complete nonrefuting pass returns `Unchanged`: updating private sums
104    /// changes no region decision. Refutation returns at the first pair whose
105    /// lower sum exceeds its upper sum. No optional allowance or new family is
106    /// introduced, and no source recognition is repeated.
107    ///
108    /// `known` must belong to this exact system and a monotone ancestor of
109    /// `region`; see [`CardinalityKnowledge`]. `scratch` is worker-owned and
110    /// may be reused sequentially by either arithmetic subject.
111    ///
112    /// # Errors
113    /// Preserves caller quota refusals and control/allocation stops. Detected
114    /// width or descent violations are invariant failures, never refutations.
115    /// Every outcome returns its performed work and refunds unused grants once.
116    pub fn inspect_reserved<Q: Quota + ?Sized>(
117        &self,
118        region: &mut Region,
119        known: &mut CardinalityKnowledge,
120        scratch: &mut NarrowingScratch,
121        cancellation: &Cancellation,
122        work: &mut Meter<'_, Q>,
123    ) -> ArithmeticAttempt<Failure<Q::Error>> {
124        let start = work.charged();
125        let result = cancellation
126            .poll()
127            .map_err(Failure::Stopped)
128            .and_then(|()| self.inspect(region, known, &mut scratch.arithmetic, work));
129        work.refund();
130        ArithmeticAttempt {
131            result,
132            statistics: NarrowingStatistics {
133                work: work.charged() - start,
134                ..NarrowingStatistics::default()
135            },
136        }
137    }
138
139    fn inspect<Q: Quota + ?Sized>(
140        &self,
141        region: &Region,
142        known: &mut CardinalityKnowledge,
143        scratch: &mut ArithmeticScratch,
144        work: &mut Meter<'_, Q>,
145    ) -> Result<SubjectStep, Failure<Q::Error>> {
146        known
147            .snapshot
148            .validate(region, self.owner.atom_count(), work)?;
149        charge(work, 1)?;
150        let Prepared::Families(families) = &self.prepared else {
151            return Ok(SubjectStep::Refuted);
152        };
153        charge(work, 1)?;
154        if !known.snapshot.initialized() {
155            charge(work, 1)?; // clear the inherited vector length before initialization
156            known.sums.clear();
157            for pair in &families.pairs {
158                push(
159                    &mut known.sums,
160                    PairSums {
161                        lower: pair.lower_sum,
162                        upper: pair.upper_sum,
163                    },
164                    2,
165                    work,
166                )?;
167            }
168        }
169        charge(work, 1)?;
170        if let Some(index) = &families.index {
171            known.snapshot.select(region, index, scratch, work)?;
172            if self.inspect_selected(families, region, known, &scratch.touched, work)? {
173                return Ok(SubjectStep::Refuted);
174            }
175        } else {
176            known.snapshot.prepare(region.mask_words(), work)?;
177        }
178        known.snapshot.publish(region, work)?;
179        Ok(SubjectStep::Unchanged)
180    }
181
182    fn inspect_selected<Q: Quota + ?Sized>(
183        &self,
184        families: &Families,
185        region: &Region,
186        known: &mut CardinalityKnowledge,
187        touched: &[u64],
188        work: &mut Meter<'_, Q>,
189    ) -> Result<bool, Failure<Q::Error>> {
190        for (word_index, word) in touched.iter().enumerate() {
191            charge(work, 1)?;
192            let mut bits = Words::<u64>::bits(*word);
193            while let Some(bit) = bits.next_with(|| charge(work, 5))? {
194                let position = word_index * 64 + bit;
195                let occurrence = &families.occurrences[position];
196                charge(work, 1)?;
197                let bound = &self.bounds[occurrence.row];
198                let old = values::effective(bound, known.snapshot.old_masks(), work)?;
199                let current = values::effective(bound, region.decision_words(), work)?;
200                charge(work, 1)?;
201                let sums = &mut known.sums[occurrence.pair];
202                if values::replace(sums, occurrence.side, old, current, work)? {
203                    return Ok(true);
204                }
205            }
206        }
207        Ok(false)
208    }
209}
