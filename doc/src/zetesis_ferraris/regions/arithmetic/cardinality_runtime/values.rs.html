1//! Reconstruct one retained occurrence at a particular partial interpretation.
2//!
3//! Prepared supports are padded, whereas snapshots have the region's logical
4//! width. The omitted support suffix is authenticated zero. Guard occurrences
5//! stay ordered and are never cached, canonicalized or skipped after a count.
6

18pub(super) fn effective<Q: Quota + ?Sized>(
19    bound: &Bound,
20    masks: (&[u64], &[u64]),
21    work: &mut Meter<'_, Q>,
22) -> Result<u64, Failure<Q::Error>> {
23    charge(work, 1)?; // select the authenticated bound direction
24    let selected = match bound.side {
25        BoundSide::Lower => masks.0,
26        BoundSide::Upper => masks.1,
27    };
28    let count = counted_intersection(&bound.support[..selected.len()], selected, work)?;
29    charge(work, 1)?; // materialize the fixed-width count
30    let count = u64::try_from(count).map_err(|_| Stop::InvalidProgram)?;
31    charge(work, 1)?; // dispatch the effective-count construction
32    let value = match bound.side {
33        BoundSide::Lower => count,
34        BoundSide::Upper => {
35            charge(work, 2)?; // subtraction and checked-result inspection
36            bound
37                .cardinality
38                .checked_sub(count)
39                .ok_or(Stop::InvalidProgram)?
40        }
41    };
42    let active = guard(&bound.guard, masks, work)?;
43    charge(work, 1)?; // inspect the completed guard result
44    if active {
45        charge(work, 3)?; // direction, clamp comparison and selected value
46        let value = match bound.side {
47            BoundSide::Lower => bound.value.max(value),
48            BoundSide::Upper => bound.value.min(value),
49        };
50        Ok(value)
51    } else {
52        Ok(value)
53    }
54}

56fn guard<Q: Quota + ?Sized>(
57    literals: &[Literal],
58    masks: (&[u64], &[u64]),
59    work: &mut Meter<'_, Q>,
60) -> Result<bool, Failure<Q::Error>> {
61    for literal in literals {
62        charge(work, 3)?;
63        let (slot, bit) = Words::<u64>::locate(literal.atom);
64        charge(work, 1)?;
65        let selected = if literal.positive { masks.0 } else { masks.1 };
66        charge(work, 2)?; // AND and membership comparison
67        if selected[slot] & bit == 0 {
68            charge(work, 1)?;
69            return Ok(false);
70        }
71    }
72    charge(work, 1)?;
73    Ok(true)
74}

76/// Replace one summand before testing the mixed old/current family bounds.
77/// Both old and current bounds apply to the descendant, so a contradiction
78/// already excludes it without visiting a later occurrence.
79pub(super) fn replace<Q: Quota + ?Sized>(
80    sums: &mut PairSums,
81    side: BoundSide,
82    old: u64,
83    current: u64,
84    work: &mut Meter<'_, Q>,
85) -> Result<bool, Failure<Q::Error>> {
86    charge(work, 1)?;
87    let sum = match side {
88        BoundSide::Lower => &mut sums.lower,
89        BoundSide::Upper => &mut sums.upper,
90    };
91    charge(work, 2)?;
92    let rest = sum.checked_sub(old).ok_or(Stop::InvalidProgram)?;
93    charge(work, 2)?;
94    let next = rest.checked_add(current).ok_or(Stop::InvalidProgram)?;
95    charge(work, 1)?;
96    *sum = next;
97    charge(work, 2)?; // comparison and its complete Boolean result
98    Ok(sums.lower > sums.upper)
99}
