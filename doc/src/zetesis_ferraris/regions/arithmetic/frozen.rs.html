1//! A candidate's inner subjects: parity rows and count bounds of its
2//! countermodel class, recognized by probing the closed query root under the
3//! candidate's frozen reduct.
4//!
5//! Proposals are syntactic (`proposals`). A row's provenance is a refuted
6//! probe (`probes`): a frozen narrowing from a copy of the closed root with
7//! exactly a guard and a pattern decided, the premise of
8//! `Zetesis.Algebra.entailed_by_refutations`. A pattern a clause of its group
9//! forbids is refuted without a probe, on the proposal pass's reading of the
10//! clause, a trusted computation that discharges the same premise
11//! (`known_refutations`). A clause too wide for a group whose pattern cuts its
12//! whole support bounds it from below by one on the same reading
13//! (`Zetesis.AlgebraicRows.lower_of_clause`). A saturation guard whose probe
14//! reaches the candidate is a unit (`Zetesis.Algebra.saturation_unit`). The
15//! preparation is built from the theory and the candidate, under its own
16//! lease of the caller's account, and belongs to that candidate alone.

45/// A candidate's query root after its first frozen closure, with what reads it.
46#[derive(Clone, Copy)]
47pub struct ClosedQuery<'a> {
48    /// The index of the candidate's theory.
49    pub narrower: &'a Narrower,
50    /// The theory's producers when its roots are rules; none proposes nothing.
51    pub producers: Option<&'a Producers>,
52    /// The candidate's frozen reduct.
53    pub subject: FrozenSubject<'a>,
54    /// The candidate whose countermodels the query walks.
55    pub candidate: &'a Interpretation,
56    /// The query root, closed by the frozen narrowing.
57    pub region: &'a Region,
58    /// The closed root's knowledge.
59    pub knowledge: &'a Knowledge<Countermodels>,
60    /// What the root's first closure charged: with one pass over the index,
61    /// what one probe is leased, since a probe narrows that closed root again
62    /// with a few more decisions.
63    pub closure_work: u64,
64}

66/// The lease of one candidate's preparation: a floor, and one allowance per
67/// probe run; crossing the lease declines the whole preparation and installs
68/// nothing.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70pub struct FrozenPreparationLimits {
71    /// Word operations, charged to the caller's account as performed: the
72    /// floor of the lease, which must cover the proposal pass, the
73    /// refutations read off the clauses before probing, and the rows'
74    /// loading. Before probing, the lease grants an allowance per scheduled
75    /// probe equal to the root closure's work ([`ClosedQuery::closure_work`])
76    /// plus one pass over the index. The floor and these allowances bound
77    /// work across all phases. The receipt records the operations performed.
78    pub work: u64,
79    /// Peak simultaneous named bytes.
80    pub bytes: usize,
81}
82
83impl FrozenPreparationLimits {
84    /// The default preparation work allowance, per candidate, with `bytes`.
85    /// The original theory's own preparation scales its allowance with the
86    /// theory; a candidate's does not.
87    #[must_use]
88    pub const fn per_candidate(bytes: usize) -> Self {
89        Self {
90            work: PreparationLimits::DEFAULT_WORK,
91            bytes,
92        }
93    }
94}

158/// One candidate's inner subjects: the units its query root cuts, and the
159/// parity and counting systems derived for that candidate within the supplied
160/// closed region. Its descendants may reuse them throughout this query.
161#[derive(Debug)]
162pub struct FrozenPreparation {
163    owner: Theory,
164    candidate: Interpretation,
165    units: Vec<usize>,
166    parity: Option<ParitySystem>,
167    cardinality: Option<CardinalitySystem>,
168}

170impl FrozenPreparation {
171    /// Prepare `query`'s candidate's inner subjects under `limits`, charging
172    /// `quota` as performed. The query root and its knowledge are read, never
173    /// changed. Optional limits publish nothing; the account's refusal and
174    /// cancellation keep their typed error.
175    #[must_use]
176    pub fn prepare_reserved<Q: Quota + ?Sized>(
177        query: ClosedQuery<'_>,
178        scratch: &mut NarrowingScratch,
179        limits: FrozenPreparationLimits,
180        cancellation: &Cancellation,
181        quota: &mut Q,
182    ) -> FrozenPreparationAttempt<Failure<Q::Error>> {
183        let mut lease = Leased {
184            account: quota,
185            remaining: limits.work,
186            used: 0,
187            extended: 0,
188        };
189        let mut statistics = FrozenPreparationStatistics::default();
190        let mut peak = 0_u128;
191        let result = build(
192            query,
193            scratch,
194            limits,
195            cancellation,
196            &mut lease,
197            &mut statistics,
198            &mut peak,
199        );
200        let result = match result {
201            Ok(prepared) => {
202                statistics.status = if prepared.is_some() {
203                    PreparationStatus::Prepared
204                } else {
205                    PreparationStatus::NotPrepared
206                };
207                Ok(prepared)
208            }
209            Err(error) => settle(error, &mut statistics),
210        };
211        statistics.preparation_work = lease.used;
212        statistics.leased = lease.extended;
213        statistics.preparation_bytes = peak;
214        FrozenPreparationAttempt {
215            result,
216            work: lease.used,
217            peak_bytes: peak,
218            statistics,
219        }
220    }
221
222    /// Check the theory instance and candidate atoms. To reuse the prepared
223    /// systems, the caller must also stay within the original closed region.
224    #[must_use]
225    pub fn belongs_to(&self, theory: &Theory, candidate: &Interpretation) -> bool {
226        self.owner.same_instance(theory)
227            && candidate.theory.same_instance(theory)
228            && self.candidate.words.storage() == candidate.words.storage()
229    }
230
231    /// The atoms the query root cuts: the saturation units.
232    #[must_use]
233    pub fn units(&self) -> &[usize] {
234        &self.units
235    }
236
237    /// The parity system of the countermodels, when any row was found.
238    #[must_use]
239    pub const fn parity(&self) -> Option<&ParitySystem> {
240        self.parity.as_ref()
241    }
242
243    /// The counting system of the countermodels, when any bound was found.
244    #[must_use]
245    pub const fn cardinality(&self) -> Option<&CardinalitySystem> {
246        self.cardinality.as_ref()
247    }
248
249    /// Whether a checked row refuted the closed root under the units.
250    #[must_use]
251    pub fn root_refuted(&self) -> bool {
252        self.parity.as_ref().is_some_and(ParitySystem::root_refuted)
253            || self
254                .cardinality
255                .as_ref()
256                .is_some_and(CardinalitySystem::root_refuted)
257    }

260/// The lease's refusal: crossed, or the account's own.
261#[derive(Clone, Copy, Debug, PartialEq, Eq)]
262enum LeaseRefusal<E> {
263    Crossed,
264    Account(E),
265}
266
267impl<E: Refusal> Refusal for LeaseRefusal<E> {
268    fn is_resource_refusal(&self) -> bool {
269        match self {
270            Self::Crossed => true,
271            Self::Account(error) => error.is_resource_refusal(),
272        }
273    }
274
275    fn from_stop(stop: Stop) -> Self {
276        Self::Account(E::from_stop(stop))
277    }
278}

280/// The caller's account under a lease: grants at most what the lease leaves,
281/// forwarding every grant and refund to the account so the account's receipt
282/// is exact.
283struct Leased<'a, Q: Quota + ?Sized> {
284    account: &'a mut Q,
285    remaining: u64,
286    used: u64,
287    extended: u64,
288}
289
290impl<Q: Quota + ?Sized> Quota for Leased<'_, Q> {
291    type Error = LeaseRefusal<Q::Error>;
292
293    fn reserve(&mut self, wanted: u64) -> Result<u64, Self::Error> {
294        let capped = wanted.min(self.remaining);
295        if capped == 0 {
296            return Err(LeaseRefusal::Crossed);
297        }
298        let mut granted = self
299            .account
300            .reserve(capped)
301            .map_err(LeaseRefusal::Account)?;
302        if granted > capped {
303            self.account.refund(granted - capped);
304            granted = capped;
305        }
306        self.remaining -= granted;
307        self.used += granted;
308        Ok(granted)
309    }
310
311    fn refund(&mut self, unspent: u64) {
312        self.account.refund(unspent);
313        self.remaining += unspent;
314        self.used -= unspent;
315    }
316}
317
318impl<Q: Quota + ?Sized> Leased<'_, Q> {
319    /// Lease `additional` operations more, as a probing round earns them.
320    fn extend(&mut self, additional: u64) {
321        self.remaining = self.remaining.saturating_add(additional);
322        self.extended = self.extended.saturating_add(additional);
323    }
324}

326/// Settle a refusal: a crossed lease declines, the rest keep their error.
327fn settle<E>(
328    error: BudgetError<LeaseRefusal<E>>,
329    statistics: &mut FrozenPreparationStatistics,
330) -> Result<Option<FrozenPreparation>, PreparationError<Failure<E>>> {
331    match error {
332        BudgetError::WorkDeclined | BudgetError::Refused(LeaseRefusal::Crossed) => {
333            statistics.status = PreparationStatus::WorkDeclined;
334            Ok(None)
335        }
336        BudgetError::StorageDeclined { .. } => {
337            statistics.status = PreparationStatus::StorageDeclined;
338            Ok(None)
339        }
340        BudgetError::Stopped(stop) => {
341            statistics.status = PreparationStatus::Incomplete;
342            Err(PreparationError::Stopped(Failure::Stopped(stop)))
343        }
344        BudgetError::Refused(LeaseRefusal::Account(error)) => {
345            statistics.status = PreparationStatus::Incomplete;
346            Err(PreparationError::Stopped(Failure::Refused(error)))
347        }
348        BudgetError::Source(source) => {
349            statistics.status = PreparationStatus::Incomplete;
350            Err(PreparationError::Source(source))
351        }
352    }
353}

355/// Propose, probe, and load the rows: three metered phases over one lease.
356fn build<Q: Quota + ?Sized>(
357    query: ClosedQuery<'_>,
358    scratch: &mut NarrowingScratch,
359    limits: FrozenPreparationLimits,
360    cancellation: &Cancellation,
361    lease: &mut Leased<'_, Q>,
362    statistics: &mut FrozenPreparationStatistics,
363    peak: &mut u128,
364) -> Result<Option<FrozenPreparation>, BudgetError<LeaseRefusal<Q::Error>>> {
365    let theory = query.subject.theory();
366    let Some(producers) = query.producers else {
367        return Ok(None);
368    };
369    let mut storage = Storage::default();
370    let mut probing = Probing {
371        lease,
372        allowance: query.closure_work.saturating_add(query.narrower.work()),
373        bytes: limits.bytes,
374        cancellation,
375        storage: &mut storage,
376    };
377    let proposals = probing.proposals(query, producers, statistics, peak)?;
378    if proposals.guards.is_empty() {
379        return Ok(None);
380    }
381    let proposal_bytes = probing.storage.bytes();
382    let open_guards = proposals
383        .guards
384        .iter()
385        .filter(|&&guard| query.region.is_open(guard))
386        .count() as u64;
387    probing
388        .lease
389        .extend(open_guards.saturating_mul(probing.allowance));
390    let (mut probe, units, admitted) = probing.with_budget(peak, |storage, budget| {
391        let copies = probes::Probe::copy_bytes(query.region, query.knowledge);
392        storage.admit_record(copies, budget)?;
393        let mut probe =
394            probes::Probe::new(query.narrower, query.subject, query.region, query.knowledge);
395        storage.release_record(copies)?;
396        storage.admit_record(probe.live_bytes(), budget)?;
397        let mut units = storage.reserve_bytes(0, budget)?;
398        let mut admitted = Admitted {
399            groups: storage.reserve_bytes(0, budget)?,
400            wide: storage.reserve_bytes(0, budget)?,
401        };
402        for &guard in &proposals.guards {
403            // A guard the root already cuts needs no probe: its clauses hold
404            // there unconditionally. An open guard is probed held; one whose
405            // narrowing reaches the candidate, or is refuted, is a unit.
406            if query.region.is_open(guard) {
407                let refuted = probe.run(
408                    &[(guard, true)],
409                    None,
410                    scratch,
411                    cancellation,
412                    storage,
413                    budget,
414                )?;
415                statistics.probes += 1;
416                let reaches = refuted
417                    || query
418                        .candidate
419                        .atoms()
420                        .all(|atom| probe.region().is_held(atom));
421                if !reaches {
422                    continue;
423                }
424                storage.push_bytes(&mut units, guard, budget)?;
425                statistics.saturation_units += 1;
426            }
427            admitted.admit_guard(&proposals, guard, storage, budget)?;
428        }
429        Ok((probe, units, admitted))
430    })?;
431    statistics.probed_groups = admitted.groups.len() as u64;
432    let (parity, cardinality) = probe_rounds(
433        theory,
434        &admitted,
435        &mut probe,
436        scratch,
437        &mut probing,
438        statistics,
439        peak,
440    )?;
441    statistics.root_refuted = parity.as_ref().is_some_and(ParitySystem::root_refuted)
442        || cardinality
443            .as_ref()
444            .is_some_and(CardinalitySystem::root_refuted);
445    if units.is_empty() && parity.is_none() && cardinality.is_none() {
446        // Nothing to install: no unit, no row, no bound.
447        return Ok(None);
448    }
449    probing.storage.release(admitted.groups)?;
450    probing.storage.release(admitted.wide)?;
451    probing
452        .storage
453        .release_record(probe.live_bytes() + proposal_bytes)?;
454    drop(probe);
455    drop(proposals);
456    probing.with_budget(peak, |storage, budget| {
457        publish(query, units, (parity, cardinality), storage, budget).map(Some)
458    })
459}

461/// Transfer the accounted units and systems into the published record and
462/// reserve an owned copy of the candidate's words. The theory stays shared.
463fn publish<Q: Quota + ?Sized>(
464    query: ClosedQuery<'_>,
465    units: Vec<usize>,
466    (parity, cardinality): Systems,
467    storage: &mut Storage,
468    budget: &mut Budget<'_, '_, Q>,
469) -> Result<FrozenPreparation, BudgetError<Q::Error>> {
470    let mut embedded = size_of::<Vec<usize>>();
471    if parity.is_some() {
472        embedded += size_of::<ParitySystem>();
473    }
474    if cardinality.is_some() {
475        embedded += size_of::<CardinalitySystem>();
476    }
477    storage.admit_record((size_of::<FrozenPreparation>() - embedded) as u128, budget)?;
478    let requested = query.candidate.words.len() as u128 * size_of::<u64>() as u128;
479    storage.admit_record(requested, budget)?;
480    let candidate = query.candidate.try_clone().map_err(BudgetError::Source)?;
481    storage.release_record(requested)?;
482    storage.admit_record(
483        candidate.words.capacity() as u128 * size_of::<u64>() as u128,
484        budget,
485    )?;
486    Ok(FrozenPreparation {
487        owner: query.subject.theory().clone(),
488        candidate,
489        units,
490        parity,
491        cardinality,
492    })
493}

499/// What a probing round may spend: the lease it draws on, the allowance each
500/// probe is leased, the floor's byte limit for the loading, and the
501/// cancellation the loading polls. The lease is the one ceiling on work; the
502/// loading's budget bounds bytes.
503struct Probing<'l, 'q, Q: Quota + ?Sized> {
504    lease: &'l mut Leased<'q, Q>,
505    allowance: u64,
506    bytes: usize,
507    cancellation: &'l Cancellation,
508    storage: &'l mut Storage,
509}
510
511impl<'q, Q: Quota + ?Sized> Probing<'_, 'q, Q> {
512    /// Publish the census after successful recognition, including when a
513    /// later phase declines. Only the returned proposals remain live.
514    fn proposals(
515        &mut self,
516        query: ClosedQuery<'_>,
517        producers: &Producers,
518        statistics: &mut FrozenPreparationStatistics,
519        peak: &mut u128,
520    ) -> Result<proposals::Proposals, BudgetError<LeaseRefusal<Q::Error>>> {
521        let proposals = self.with_budget(peak, |storage, budget| {
522            budget.in_phase(Phase::Recognition, |budget| {
523                proposals::propose(
524                    query.subject.theory(),
525                    producers,
526                    query.candidate,
527                    query.subject.truth(),
528                    query.region,
529                    storage,
530                    budget,
531                )
532            })
533        })?;
534        statistics.guards = proposals.counts.guards;
535        statistics.clauses = proposals.counts.clauses;
536        statistics.unsupported_clauses = proposals.counts.unsupported_clauses;
537        statistics.groups = proposals.counts.groups;
538        Ok(proposals)
539    }
540
541    /// Settle each phase's work and observed peak, including a refused
542    /// request. Live storage crosses the phase boundary with its owners.
543    fn with_budget<T>(
544        &mut self,
545        peak: &mut u128,
546        operation: impl FnOnce(
547            &mut Storage,
548            &mut Budget<'_, '_, Leased<'q, Q>>,
549        ) -> Result<T, BudgetError<LeaseRefusal<Q::Error>>>,
550    ) -> Result<T, BudgetError<LeaseRefusal<Q::Error>>> {
551        let mut meter = Meter::reserved(self.lease);
552        let mut budget = Budget::new(&mut meter, u64::MAX, self.bytes, self.cancellation);
553        let result = operation(self.storage, &mut budget);
554        *peak = (*peak).max(budget.peak_bytes());
555        budget.finish();
556        result
557    }
558}

560/// What the root admits under the guards it cuts: the groups to probe, and
561/// the clauses too wide for a group whose pattern cuts their whole support,
562/// each a lower bound of one over it.
563struct Admitted<'p> {
564    groups: Vec<&'p proposals::Group>,
565    wide: Vec<&'p proposals::Wide>,
566}
567
568impl<'p> Admitted<'p> {
569    /// Admit the groups and wide clauses of a guard the root cuts.
570    fn admit_guard<Q: Quota + ?Sized>(
571        &mut self,
572        proposals: &'p proposals::Proposals,
573        guard: usize,
574        storage: &mut Storage,
575        budget: &mut Budget<'_, '_, Q>,
576    ) -> Result<(), BudgetError<Q::Error>> {
577        for group in proposals.parity.iter().filter(|group| group.guard == guard) {
578            storage.push_bytes(&mut self.groups, group, budget)?;
579        }
580        for clause in proposals.wide.iter().filter(|clause| clause.guard == guard) {
581            storage.push_bytes(&mut self.wide, clause, budget)?;
582        }
583        Ok(())
584    }
585}

587/// A probed group's table over its support: the patterns found forbidden,
588/// those known before any probe among them.
589struct ProbedGroup {
590    support: Support,

594/// The patterns `table` forbids.
595fn count_forbidden(table: &probes::Table) -> u64 {
596    table
597        .storage()
598        .iter()
599        .map(|word| u64::from(word.count_ones()))
600        .sum()
601}

603/// The patterns of each group refuted before any probe runs: those a clause
604/// of the group forbids, and those whose restriction to a group whose
605/// support this one contains is a pattern that group's clauses forbid. The
606/// region deciding such a pattern holds no member of the class: a member in
607/// it lies in the closed root, so under the pattern it satisfies the clause's
608/// body (the clause's rule and the chains to its base standing in the reduct,
609/// their bodies true in the candidate, as the proposal pass reads them) and
610/// holds the clause's guard, which no countermodel in the root holds, the
611/// root having cut it already or the guard being a unit
612/// (`saturation_unit`); the narrowing of that region, inside it, holds none
613/// either, which is the premise `entailed_by_refutations` asks of a refuted
614/// pattern, discharged here by the proposal pass's reading of the clause
615/// rather than by a probe. The table records it without running the probe.
616/// One word operation is charged to the floor per clause read, per pair of
617/// groups compared and per pattern examined.
618///
619/// # Errors
620/// The lease's refusal.
621fn known_refutations<Q: Quota + ?Sized>(
622    groups: &[&proposals::Group],
623    storage: &mut Storage,
624    budget: &mut Budget<'_, '_, Q>,
625) -> Result<Vec<probes::Table>, BudgetError<Q::Error>> {
626    let mut known = storage.reserve_bytes(groups.len(), budget)?;
627    for group in groups {
628        let support = &group.support.atoms[..group.support.len];
629        known.push(Words::<u64, _>::new([0; 4]));
630        let refuted = known.last_mut().expect("the current table was appended");
631        for &pattern in &group.patterns {
632            budget.charge(1)?;
633            refuted.insert(usize::from(pattern));
634        }
635        for other in groups {
636            budget.charge(1)?;
637            if other.support.len > group.support.len || std::ptr::eq(*other, *group) {
638                continue;
639            }
640            // Where `other`'s atoms sit in this support, if all of them do.
641            storage.admit_record(
642                size_of::<[usize; super::source::MAX_SUPPORT]>() as u128,
643                budget,
644            )?;
645            let mut positions = [0_usize; super::source::MAX_SUPPORT];
646            let contained = other.support.atoms[..other.support.len]
647                .iter()
648                .enumerate()
649                .all(|(k, atom)| {
650                    support.iter().position(|a| a == atom).is_some_and(|here| {
651                        positions[k] = here;
652                        true
653                    })
654                });
655            if !contained {
656                storage.release_record(size_of::<[usize; super::source::MAX_SUPPORT]>() as u128)?;
657                continue;
658            }
659            let mask = positions[..other.support.len]
660                .iter()
661                .fold(0_usize, |mask, &here| mask | (1 << here));
662            for &forbidden in &other.patterns {
663                let value = (0..other.support.len)
664                    .filter(|&k| forbidden & (1 << k) != 0)
665                    .fold(0_usize, |value, k| value | (1 << positions[k]));
666                for pattern in 0..(1_usize << group.support.len) {
667                    budget.charge(1)?;
668                    if pattern & mask == value {
669                        refuted.insert(pattern);
670                    }
671                }
672            }
673            storage.release_record(size_of::<[usize; super::source::MAX_SUPPORT]>() as u128)?;
674        }
675    }
676    Ok(known)
677}

679/// Retain complete tables from this same query in pending-group order. Each
680/// four-word replacement admits its whole charge before changing that slot;
681/// a refusal leaves that group and its suffix untouched. Prior completed
682/// groups remain charged. These slots already belong to the preparation.
683fn retain_refutations<Q: Quota + ?Sized>(
684    pending: &[usize],
685    tables: &[ProbedGroup],
686    known: &mut [probes::Table],
687    budget: &mut Budget<'_, '_, Q>,
688) -> Result<(), BudgetError<Q::Error>> {
689    for &index in pending {
690        budget.bounded(4, || {
691            known[index] = Words::new(*tables[index].forbidden.storage());
692            ((), 4)
693        })?;
694    }
695    Ok(())
696}

698/// The probing rounds a preparation may run: the first by the frozen reading
699/// alone, the second under the rows it established. A later round tightens
700/// a hull the round before left inexact, one probe per pattern of each such
701/// group at the lease allowance each, and the cap bounds a preparation's
702/// probing at two rounds' probes; what the rounds established is loaded,
703/// sound for the class as before, the remaining hulls inexact.
704const MAX_ROUNDS: u64 = 2;
705
706/// Probe the groups in rounds and load their rows and bounds: round one by
707/// the frozen reading alone, leaving a group of one clause cutting its whole
708/// support to that clause, whose table already bounds the support from below
709/// by one; each later round under the parity rows the round before
710/// established, sound for the class as their joint closure is,
711/// while a round refutes more patterns than the one before, some hull is
712/// still inexact, a parity row was loaded for a later round to narrow under
713/// and no system refutes the root; each round leases its
714/// probes at `allowance` each. The systems of the last round are the
715/// preparation's.
716fn probe_rounds<Q: Quota + ?Sized>(
717    theory: &Theory,
718    admitted: &Admitted<'_>,
719    probe: &mut probes::Probe<'_>,
720    scratch: &mut NarrowingScratch,
721    probing: &mut Probing<'_, '_, Q>,
722    statistics: &mut FrozenPreparationStatistics,
723    peak: &mut u128,
724) -> Result<Systems, BudgetError<LeaseRefusal<Q::Error>>> {
725    let groups = &admitted.groups[..];
726    let mut parity: Option<ParitySystem> = None;
727    let mut cardinality: Option<CardinalitySystem> = None;
728    let (mut tables, mut refuted, mut known, mut pending) =
729        probing.with_budget(peak, |storage, budget| {
730            let mut tables = storage.reserve_bytes(groups.len(), budget)?;
731            tables.extend(groups.iter().map(|group| ProbedGroup {
732                support: group.support,
733                forbidden: Words::new([0; 4]),
734            }));
735            let mut refuted = storage.reserve_bytes(groups.len(), budget)?;
736            refuted.resize(groups.len(), 0);
737            let known = known_refutations(groups, storage, budget)?;
738            let mut pending = storage.reserve_bytes(groups.len(), budget)?;
739            pending.extend(0..groups.len());
740            Ok((tables, refuted, known, pending))
741        })?;
742    while !pending.is_empty() && statistics.rounds < MAX_ROUNDS {
743        if statistics.rounds != 0 {
744            // Every earlier refutation belongs to this same group's ordered
745            // support and cut guard in this candidate's closed query. Adding
746            // its checked rows cannot make an excluded pattern possible.
747            // Retain those exact table entries before leasing later probes.
748            probing.with_budget(peak, |_, budget| {
749                retain_refutations(&pending, &tables, &mut known, budget)
750            })?;
751        }
752        statistics.rounds += 1;
753        // The first round leaves a group of one clause cutting its whole
754        // support to that clause: its table bounds the support from below by
755        // one, and only a later round's rows could refute more of it.
756        let settled_by_clause =
757            |index: usize| statistics.rounds == 1 && groups[index].patterns == [0];
758        // The probes this round runs: the patterns not already refuted by
759        // their clauses or a completed probe of this same group.
760        let probes: u64 = pending
761            .iter()
762            .filter(|&&index| !settled_by_clause(index))
763            .map(|&index| (1_u64 << groups[index].support.len) - count_forbidden(&known[index]))
764            .sum();
765        probing
766            .lease
767            .extend(probes.saturating_mul(probing.allowance));
768        let mut progressed = false;
769        let cancellation = probing.cancellation;
770        probing.with_budget(peak, |storage, budget| {
771            for &index in &pending {
772                if settled_by_clause(index) {
773                    tables[index].forbidden = Words::new(*known[index].storage());
774                    refuted[index] = count_forbidden(&known[index]);
775                    progressed = true;
776                    continue;
777                }
778                let found = probe.table(
779                    probes::Target {
780                        group: groups[index],
781                        known: &known[index],
782                        forbidden: &mut tables[index].forbidden,
783                    },
784                    parity.as_ref(),
785                    scratch,
786                    cancellation,
787                    storage,
788                    budget,
789                )?;
790                statistics.probes += found.run;
791                progressed |= found.refuted > refuted[index];
792                refuted[index] = found.refuted;
793            }
794            Ok(())
795        })?;
796        // The old systems and worklist have served their last probe. End their
797        // lifetimes before loading the replacement, and retain the tables.
798        probing.storage.release(pending)?;
799        if let Some(previous) = parity.take() {
800            probing.storage.release_record(previous.retained_bytes())?;
801        }
802        if let Some(previous) = cardinality.take() {
803            probing.storage.release_record(previous.retained_bytes())?;
804        }
805        let loaded = load_round(theory, &tables, &admitted.wide, probing, statistics, peak)?;
806        parity = loaded.parity;
807        cardinality = loaded.cardinality;
808        pending = loaded.inexact;
809        if !progressed
810            || parity.as_ref().is_none_or(ParitySystem::root_refuted)
811            || cardinality
812                .as_ref()
813                .is_some_and(CardinalitySystem::root_refuted)
814        {
815            break;
816        }
817    }
818    if statistics.rounds == 0 && !admitted.wide.is_empty() {
819        // No group to probe: the wide clauses' bounds load alone.
820        let loaded = load_round(theory, &tables, &admitted.wide, probing, statistics, peak)?;
821        parity = loaded.parity;
822        cardinality = loaded.cardinality;
823        probing.storage.release(loaded.inexact)?;
824    }
825    statistics.refuted_patterns = refuted.iter().sum();
826    probing.storage.release(pending)?;
827    probing.storage.release(known)?;
828    probing.storage.release(refuted)?;
829    probing.storage.release(tables)?;
830    Ok((parity, cardinality))
831}

833/// Load a round's systems from `tables` and the `wide` clauses' bounds, the
834/// lease bounding work and the budget bytes.
835fn load_round<Q: Quota + ?Sized>(
836    theory: &Theory,
837    tables: &[ProbedGroup],
838    wide: &[&proposals::Wide],
839    probing: &mut Probing<'_, '_, Q>,
840    statistics: &mut FrozenPreparationStatistics,
841    peak: &mut u128,
842) -> Result<Loaded, BudgetError<LeaseRefusal<Q::Error>>> {
843    probing.with_budget(peak, |storage, budget| {
844        load(theory, tables, wide, statistics, storage, budget)
845    })
846}

848/// What a round loaded: the parity and counting systems, when any row or
849/// bound was found, and the indices of the tables whose hull is inexact.
850struct Loaded {
851    parity: Option<ParitySystem>,
852    cardinality: Option<CardinalitySystem>,
853    inexact: Vec<usize>,
854}
855
856/// The systems of the probed groups: the parity system of each table's
857/// affine hull, every equation checked against its table and eliminated to
858/// echelon form, and the counting system of the tables' bounds and the wide
859/// clauses'; with the indices of the tables whose hull is inexact.
860fn load<Q: Quota + ?Sized>(
861    theory: &Theory,
862    tables: &[ProbedGroup],
863    wide: &[&proposals::Wide],
864    statistics: &mut FrozenPreparationStatistics,
865    outer: &mut Storage,
866    budget: &mut Budget<'_, '_, Q>,
867) -> Result<Loaded, BudgetError<Q::Error>> {
868    let mut inexact = outer.reserve_bytes(tables.len(), budget)?;
869    if tables.is_empty() && wide.is_empty() {
870        return Ok(Loaded {
871            parity: None,
872            cardinality: None,
873            inexact,
874        });
875    }
876    let parity = load_parity(theory, tables, &mut inexact, statistics, outer, budget)?;
877    // Transfer only the surviving parity owner. Its builder can also have
878    // held temporary empty evidence and vector headers on a row-free input.
879    if let Some(system) = &parity {
880        outer.admit_record(system.retained_bytes(), budget)?;
881    }
882    let mut storage = Storage::with_live_bytes(outer.bytes());
883    let cardinality = load_counts(theory, tables, wide, statistics, &mut storage, budget)?;
884    if let Some(system) = &cardinality {
885        outer.admit_record(system.retained_bytes(), budget)?;
886    }
887    Ok(Loaded {
888        parity,
889        cardinality,
890        inexact,
891    })
892}

894/// Load the tables' implied equations into a parity system. The caller has
895/// reserved one inexact-index slot per input table.
896fn load_parity<Q: Quota + ?Sized>(
897    theory: &Theory,
898    tables: &[ProbedGroup],
899    inexact: &mut Vec<usize>,
900    statistics: &mut FrozenPreparationStatistics,
901    outer: &Storage,
902    budget: &mut Budget<'_, '_, Q>,
903) -> Result<Option<ParitySystem>, BudgetError<Q::Error>> {
904    let mut storage = Storage::with_live_bytes(outer.bytes());
905    let (facts, mut recognized_equations, counts) =
906        budget.in_phase(Phase::Recognition, |budget| {
907            storage.admit_record(
908                (size_of::<rows::Recognized>()
909                    - size_of::<CheckedRootFacts>()
910                    - size_of::<Vec<rows::Equation>>()
911                    - size_of::<Vec<source::Clause>>()) as u128,
912                budget,
913            )?;
914            let facts = CheckedRootFacts::none(theory, &mut storage, budget)?;
915            let mut equations = storage.reserve(0, budget)?;
916            let mut counts = rows::Counts::default();
917            for (group, table) in tables.iter().enumerate() {
918                let fixed = size_of::<hull::Hull>() as u128;
919                storage.admit_record(fixed, budget)?;
920                let proposed =
921                    storage.with_record(hull::SCRATCH_BYTES as u128, budget, |_, budget| {
922                        rows::propose(
923                            table.support,
924                            table.forbidden.storage(),
925                            &mut counts,
926                            budget,
927                        )
928                    })?;
929                if !proposed.exact {
930                    inexact.push(group);
931                }
932                for &local in proposed.equations() {
933                    let words =
934                        rows::materialize(theory, table.support, local, &mut storage, budget)?;
935                    budget.charge(1)?; // the constant
936                    let candidate = rows::Equation {
937                        words,
938                        constant: local.constant,
939                    };
940                    let valid = source::implied_by_table(
941                        table.forbidden.storage(),
942                        table.support.len,
943                        local.coefficients,
944                        local.constant,
945                        budget,
946                    )?;
947                    rows::retain(
948                        valid,
949                        candidate,
950                        local.coefficients,
951                        &mut equations,
952                        &mut counts,
953                        &mut storage,
954                        budget,
955                    )?;
956                }
957                storage.release_record(fixed)?;
958            }
959            Ok((facts, equations, counts))
960        })?;
961    statistics.inexact_hulls = counts.inexact_hulls;
962    budget.charge(1)?;
963    if recognized_equations.is_empty() {
964        return Ok(None);
965    }
966
967    let prepared = budget.in_phase(Phase::Reduction, |budget| {
968        parity::prepare(
969            &mut recognized_equations,
970            theory.atom_count(),
971            &mut storage,
972            budget,
973        )
974    })?;
975    storage.admit_record(size_of::<Vec<source::Clause>>() as u128, budget)?;
976    let recognized = rows::Recognized {
977        facts,
978        clauses: Vec::new(),
979        equations: recognized_equations,
980        counts,
981    };
982    statistics.parity = ParityStatistics::from_preparation(&recognized, &prepared, budget);
983    Ok(Some(ParitySystem::from_rows(recognized, prepared)))
984}

986/// The counting system of the tables' bounds and the wide clauses', when
987/// any was found: the bounds proposed, their families prepared and the
988/// root's double count taken, as the outer preparation takes them.
989fn load_counts<Q: Quota + ?Sized>(
990    theory: &Theory,
991    tables: &[ProbedGroup],
992    wide: &[&proposals::Wide],
993    statistics: &mut FrozenPreparationStatistics,
994    storage: &mut Storage,
995    budget: &mut Budget<'_, '_, Q>,
996) -> Result<Option<CardinalitySystem>, BudgetError<Q::Error>> {
997    let mut counted = CardinalityStatistics::default();
998    let start = budget.work();
999    let bounds = budget.in_phase(Phase::Recognition, |budget| {
1000        counts::propose(theory, tables, wide, &mut counted, storage, budget)
1001    })?;
1002    budget.charge(1)?;
1003    if bounds.is_empty() {
1004        storage.release(bounds)?;
1005        statistics.cardinality = counted;
1006        return Ok(None);
1007    }
1008    let prepared = budget.in_phase(Phase::Reduction, |budget| {
1009        cardinality::prepare(&bounds, theory.atom_count(), storage, budget)
1010    })?;
1011    counted.status = PreparationStatus::Prepared;
1012    counted.rows = bounds.len();
1013    counted.carriers = prepared.counts().carriers;
1014    counted.attempts = prepared.counts().attempts;
1015    counted.families = prepared.counts().accepted;
1016    counted.refused = prepared.counts().rejected;
1017    match &prepared {
1018        cardinality::Prepared::RootRefuted { .. } => counted.root_refuted = true,
1019        cardinality::Prepared::Families(families) => {
1020            counted.occurrences = families.occurrences.len();
1021        }
1022    }
1023    counted.preparation_work = budget.work() - start;
1024    counted.preparation_bytes = budget.peak_bytes();
1025    storage.admit_record(
1026        (size_of::<CardinalitySystem>()
1027            - size_of::<cardinality::Prepared>()
1028            - size_of::<Vec<super::cardinality_source::Bound>>()) as u128,
1029        budget,
1030    )?;
1031    let system = CardinalitySystem::from_bounds(theory.clone(), bounds, prepared);
1032    counted.retained_bytes = system.retained_bytes();
1033    statistics.cardinality = counted;
1034    Ok(Some(system))
1035}
