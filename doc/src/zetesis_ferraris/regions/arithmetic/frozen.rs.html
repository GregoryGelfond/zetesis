1//! A candidate's inner subjects: parity rows and count bounds of its
2//! countermodel class, recognized by probing the closed query root under the
3//! candidate's frozen reduct.
4//!
5//! Proposals are syntactic (`proposals`). A row's provenance is a refuted
6//! probe (`probes`): a frozen narrowing from a copy of the closed root with
7//! exactly a guard and a pattern decided, the premise of
8//! `Zetesis.Algebra.entailed_by_refutations`. A pattern a clause of its group
9//! forbids is refuted without a probe, on the proposal pass's reading of the
10//! clause, a trusted computation that discharges the same premise
11//! (`known_refutations`). A clause too wide for a group whose pattern cuts its
12//! whole support bounds it from below by one on the same reading
13//! (`Zetesis.AlgebraicRows.lower_of_clause`). A saturation guard whose probe
14//! reaches the candidate is a unit (`Zetesis.Algebra.saturation_unit`). The
15//! preparation is built from the theory and the candidate, under its own
16//! lease of the caller's account, and belongs to that candidate alone.

45/// A candidate's query root after its first frozen closure, with what reads it.
46#[derive(Clone, Copy)]
47pub struct ClosedQuery<'a> {
48    /// The index of the candidate's theory.
49    pub narrower: &'a Narrower,
50    /// The theory's producers when its roots are rules; none proposes nothing.
51    pub producers: Option<&'a Producers>,
52    /// The candidate's frozen reduct.
53    pub subject: FrozenSubject<'a>,
54    /// The candidate whose countermodels the query walks.
55    pub candidate: &'a Interpretation,
56    /// The query root, closed by the frozen narrowing.
57    pub region: &'a Region,
58    /// The closed root's knowledge.
59    pub knowledge: &'a Knowledge<Countermodels>,
60    /// What the root's first closure charged: with one pass over the index,
61    /// what one probe is leased, since a probe narrows that closed root again
62    /// with a few more decisions.
63    pub closure_work: u64,

66/// The lease of one candidate's preparation: a floor, and one allowance per
67/// probe run; crossing the lease declines the whole preparation and installs
68/// nothing.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70pub struct FrozenPreparationLimits {
71    /// Word operations, charged to the caller's account as performed: the
72    /// floor of the lease, which must cover the proposal pass, the
73    /// refutations read off the clauses before probing, and the rows'
74    /// loading. Before probing, the lease grants an allowance per scheduled
75    /// probe equal to the root closure's work ([`ClosedQuery::closure_work`])
76    /// plus one pass over the index. The floor and these allowances bound
77    /// work across all phases. The receipt records the operations performed.
78    pub work: u64,
79    /// Peak simultaneous named bytes.
80    pub bytes: usize,
81}

83impl FrozenPreparationLimits {
84    /// The root preparation's work allowance, per candidate, with `bytes`.
85    #[must_use]
86    pub const fn per_candidate(bytes: usize) -> Self {
87        Self {
88            work: PreparationLimits::DEFAULT_WORK,
89            bytes,
90        }
91    }
92}

156/// One candidate's inner subjects: the units its query root cuts, and the
157/// parity and counting systems derived for that candidate within the supplied
158/// closed region. Its descendants may reuse them throughout this query.
159#[derive(Debug)]
160pub struct FrozenPreparation {
161    owner: Theory,
162    candidate: Interpretation,
163    units: Vec<usize>,

168impl FrozenPreparation {
169    /// Prepare `query`'s candidate's inner subjects under `limits`, charging
170    /// `quota` as performed. The query root and its knowledge are read, never
171    /// changed. Optional limits publish nothing; the account's refusal and
172    /// cancellation keep their typed error.
173    #[must_use]
174    pub fn prepare_reserved<Q: Quota + ?Sized>(
175        query: ClosedQuery<'_>,
176        scratch: &mut NarrowingScratch,
177        limits: FrozenPreparationLimits,
178        cancellation: &Cancellation,
179        quota: &mut Q,
180    ) -> FrozenPreparationAttempt<Failure<Q::Error>> {
181        let mut lease = Leased {
182            account: quota,
183            remaining: limits.work,
184            used: 0,
185            extended: 0,
186        };
187        let mut statistics = FrozenPreparationStatistics::default();
188        let mut peak = 0_u128;
189        let result = build(
190            query,
191            scratch,
192            limits,
193            cancellation,
194            &mut lease,
195            &mut statistics,
196            &mut peak,
197        );
198        let result = match result {
199            Ok(prepared) => {
200                statistics.status = if prepared.is_some() {
201                    PreparationStatus::Prepared
202                } else {
203                    PreparationStatus::NotPrepared
204                };
205                Ok(prepared)
206            }
207            Err(error) => settle(error, &mut statistics),
208        };
209        statistics.preparation_work = lease.used;
210        statistics.leased = lease.extended;
211        statistics.preparation_bytes = peak;
212        FrozenPreparationAttempt {
213            result,
214            work: lease.used,
215            peak_bytes: peak,
216            statistics,
217        }
218    }

224        self.owner.same_instance(theory)
225            && candidate.theory.same_instance(theory)
226            && self.candidate.words.storage() == candidate.words.storage()
227    }

233    }

238        self.parity.as_ref()
239    }

241    /// The counting system of the countermodels, when any bound was found.
242    #[must_use]
243    pub const fn cardinality(&self) -> Option<&CardinalitySystem> {

245    }
246
247    /// Whether a checked row refuted the closed root under the units.
248    #[must_use]
249    pub fn root_refuted(&self) -> bool {
250        self.parity.as_ref().is_some_and(ParitySystem::root_refuted)

252                .cardinality
253                .as_ref()
254                .is_some_and(CardinalitySystem::root_refuted)

258/// The lease's refusal: crossed, or the account's own.
259#[derive(Clone, Copy, Debug, PartialEq, Eq)]
260enum LeaseRefusal<E> {
261    Crossed,
262    Account(E),
263}
264
265impl<E: Refusal> Refusal for LeaseRefusal<E> {
266    fn is_resource_refusal(&self) -> bool {
267        match self {
268            Self::Crossed => true,
269            Self::Account(error) => error.is_resource_refusal(),
270        }
271    }
272
273    fn from_stop(stop: Stop) -> Self {
274        Self::Account(E::from_stop(stop))
275    }
276}

278/// The caller's account under a lease: grants at most what the lease leaves,
279/// forwarding every grant and refund to the account so the account's receipt
280/// is exact.
281struct Leased<'a, Q: Quota + ?Sized> {
282    account: &'a mut Q,
283    remaining: u64,
284    used: u64,
285    extended: u64,
286}
287
288impl<Q: Quota + ?Sized> Quota for Leased<'_, Q> {
289    type Error = LeaseRefusal<Q::Error>;
290
291    fn reserve(&mut self, wanted: u64) -> Result<u64, Self::Error> {
292        let capped = wanted.min(self.remaining);
293        if capped == 0 {
294            return Err(LeaseRefusal::Crossed);
295        }
296        let mut granted = self
297            .account
298            .reserve(capped)
299            .map_err(LeaseRefusal::Account)?;
300        if granted > capped {
301            self.account.refund(granted - capped);
302            granted = capped;
303        }
304        self.remaining -= granted;
305        self.used += granted;
306        Ok(granted)
307    }
308
309    fn refund(&mut self, unspent: u64) {
310        self.account.refund(unspent);
311        self.remaining += unspent;
312        self.used -= unspent;
313    }
314}
315
316impl<Q: Quota + ?Sized> Leased<'_, Q> {
317    /// Lease `additional` operations more, as a probing round earns them.
318    fn extend(&mut self, additional: u64) {
319        self.remaining = self.remaining.saturating_add(additional);
320        self.extended = self.extended.saturating_add(additional);
321    }
322}

324/// Settle a refusal: a crossed lease declines, the rest keep their error.
325fn settle<E>(
326    error: BudgetError<LeaseRefusal<E>>,
327    statistics: &mut FrozenPreparationStatistics,
328) -> Result<Option<FrozenPreparation>, PreparationError<Failure<E>>> {
329    match error {
330        BudgetError::WorkDeclined | BudgetError::Refused(LeaseRefusal::Crossed) => {
331            statistics.status = PreparationStatus::WorkDeclined;
332            Ok(None)
333        }
334        BudgetError::StorageDeclined { .. } => {
335            statistics.status = PreparationStatus::StorageDeclined;
336            Ok(None)
337        }
338        BudgetError::Stopped(stop) => {
339            statistics.status = PreparationStatus::Incomplete;
340            Err(PreparationError::Stopped(Failure::Stopped(stop)))
341        }
342        BudgetError::Refused(LeaseRefusal::Account(error)) => {
343            statistics.status = PreparationStatus::Incomplete;
344            Err(PreparationError::Stopped(Failure::Refused(error)))
345        }
346        BudgetError::Source(source) => {
347            statistics.status = PreparationStatus::Incomplete;
348            Err(PreparationError::Source(source))
349        }
350    }
351}

353/// Propose, probe, and load the rows: three metered phases over one lease.
354fn build<Q: Quota + ?Sized>(
355    query: ClosedQuery<'_>,
356    scratch: &mut NarrowingScratch,
357    limits: FrozenPreparationLimits,
358    cancellation: &Cancellation,
359    lease: &mut Leased<'_, Q>,
360    statistics: &mut FrozenPreparationStatistics,
361    peak: &mut u128,
362) -> Result<Option<FrozenPreparation>, BudgetError<LeaseRefusal<Q::Error>>> {
363    let theory = query.subject.theory();
364    let Some(producers) = query.producers else {
365        return Ok(None);
366    };
367    let mut storage = Storage::default();
368    let mut probing = Probing {
369        lease,
370        allowance: query.closure_work.saturating_add(query.narrower.work()),
371        bytes: limits.bytes,
372        cancellation,
373        storage: &mut storage,
374    };
375    let proposals = probing.proposals(query, producers, statistics, peak)?;
376    if proposals.guards.is_empty() {
377        return Ok(None);
378    }
379    let proposal_bytes = probing.storage.bytes();
380    let open_guards = proposals
381        .guards
382        .iter()
383        .filter(|&&guard| query.region.is_open(guard))
384        .count() as u64;
385    probing
386        .lease
387        .extend(open_guards.saturating_mul(probing.allowance));
388    let (mut probe, units, admitted) = probing.with_budget(peak, |storage, budget| {
389        let copies = probes::Probe::copy_bytes(query.region, query.knowledge);
390        storage.admit_record(copies, budget)?;
391        let mut probe =
392            probes::Probe::new(query.narrower, query.subject, query.region, query.knowledge);
393        storage.release_record(copies)?;
394        storage.admit_record(probe.live_bytes(), budget)?;
395        let mut units = storage.reserve_bytes(0, budget)?;
396        let mut admitted = Admitted {
397            groups: storage.reserve_bytes(0, budget)?,
398            wide: storage.reserve_bytes(0, budget)?,
399        };
400        for &guard in &proposals.guards {
401            // A guard the root already cuts needs no probe: its clauses hold
402            // there unconditionally. An open guard is probed held; one whose
403            // narrowing reaches the candidate, or is refuted, is a unit.
404            if query.region.is_open(guard) {
405                let refuted = probe.run(
406                    &[(guard, true)],
407                    None,
408                    scratch,
409                    cancellation,
410                    storage,
411                    budget,
412                )?;
413                statistics.probes += 1;
414                let reaches = refuted
415                    || query
416                        .candidate
417                        .atoms()
418                        .all(|atom| probe.region().is_held(atom));
419                if !reaches {
420                    continue;
421                }
422                storage.push_bytes(&mut units, guard, budget)?;
423                statistics.saturation_units += 1;
424            }
425            admitted.admit_guard(&proposals, guard, storage, budget)?;
426        }
427        Ok((probe, units, admitted))
428    })?;
429    statistics.probed_groups = admitted.groups.len() as u64;
430    let (parity, cardinality) = probe_rounds(
431        theory,
432        &admitted,
433        &mut probe,
434        scratch,
435        &mut probing,
436        statistics,
437        peak,
438    )?;
439    statistics.root_refuted = parity.as_ref().is_some_and(ParitySystem::root_refuted)
440        || cardinality
441            .as_ref()
442            .is_some_and(CardinalitySystem::root_refuted);
443    if units.is_empty() && parity.is_none() && cardinality.is_none() {
444        // Nothing to install: no unit, no row, no bound.
445        return Ok(None);
446    }
447    probing.storage.release(admitted.groups)?;
448    probing.storage.release(admitted.wide)?;
449    probing
450        .storage
451        .release_record(probe.live_bytes() + proposal_bytes)?;
452    drop(probe);
453    drop(proposals);
454    probing.with_budget(peak, |storage, budget| {
455        publish(query, units, (parity, cardinality), storage, budget).map(Some)
456    })
457}

459/// Transfer the accounted units and systems into the published record and
460/// reserve an owned copy of the candidate's words. The theory stays shared.
461fn publish<Q: Quota + ?Sized>(
462    query: ClosedQuery<'_>,
463    units: Vec<usize>,
464    (parity, cardinality): Systems,
465    storage: &mut Storage,
466    budget: &mut Budget<'_, '_, Q>,
467) -> Result<FrozenPreparation, BudgetError<Q::Error>> {
468    let mut embedded = size_of::<Vec<usize>>();
469    if parity.is_some() {
470        embedded += size_of::<ParitySystem>();
471    }
472    if cardinality.is_some() {
473        embedded += size_of::<CardinalitySystem>();
474    }
475    storage.admit_record((size_of::<FrozenPreparation>() - embedded) as u128, budget)?;
476    let requested = query.candidate.words.len() as u128 * size_of::<u64>() as u128;
477    storage.admit_record(requested, budget)?;
478    let candidate = query.candidate.try_clone().map_err(BudgetError::Source)?;
479    storage.release_record(requested)?;
480    storage.admit_record(
481        candidate.words.capacity() as u128 * size_of::<u64>() as u128,
482        budget,
483    )?;
484    Ok(FrozenPreparation {
485        owner: query.subject.theory().clone(),
486        candidate,
487        units,
488        parity,
489        cardinality,
490    })
491}

497/// What a probing round may spend: the lease it draws on, the allowance each
498/// probe is leased, the floor's byte limit for the loading, and the
499/// cancellation the loading polls. The lease is the one ceiling on work; the
500/// loading's budget bounds bytes.
501struct Probing<'l, 'q, Q: Quota + ?Sized> {
502    lease: &'l mut Leased<'q, Q>,
503    allowance: u64,
504    bytes: usize,
505    cancellation: &'l Cancellation,
506    storage: &'l mut Storage,
507}
508
509impl<'q, Q: Quota + ?Sized> Probing<'_, 'q, Q> {
510    /// Publish the census after successful recognition, including when a
511    /// later phase declines. Only the returned proposals remain live.
512    fn proposals(
513        &mut self,
514        query: ClosedQuery<'_>,
515        producers: &Producers,
516        statistics: &mut FrozenPreparationStatistics,
517        peak: &mut u128,
518    ) -> Result<proposals::Proposals, BudgetError<LeaseRefusal<Q::Error>>> {
519        let proposals = self.with_budget(peak, |storage, budget| {
520            budget.in_phase(Phase::Recognition, |budget| {
521                proposals::propose(
522                    query.subject.theory(),
523                    producers,
524                    query.candidate,
525                    query.subject.truth(),
526                    query.region,
527                    storage,
528                    budget,
529                )
530            })
531        })?;
532        statistics.guards = proposals.counts.guards;
533        statistics.clauses = proposals.counts.clauses;
534        statistics.unsupported_clauses = proposals.counts.unsupported_clauses;
535        statistics.groups = proposals.counts.groups;
536        Ok(proposals)
537    }
538
539    /// Settle each phase's work and observed peak, including a refused
540    /// request. Live storage crosses the phase boundary with its owners.
541    fn with_budget<T>(
542        &mut self,
543        peak: &mut u128,
544        operation: impl FnOnce(
545            &mut Storage,
546            &mut Budget<'_, '_, Leased<'q, Q>>,
547        ) -> Result<T, BudgetError<LeaseRefusal<Q::Error>>>,
548    ) -> Result<T, BudgetError<LeaseRefusal<Q::Error>>> {
549        let mut meter = Meter::reserved(self.lease);
550        let mut budget = Budget::new(&mut meter, u64::MAX, self.bytes, self.cancellation);
551        let result = operation(self.storage, &mut budget);
552        *peak = (*peak).max(budget.peak_bytes());
553        budget.finish();
554        result
555    }
556}

558/// What the root admits under the guards it cuts: the groups to probe, and
559/// the clauses too wide for a group whose pattern cuts their whole support,
560/// each a lower bound of one over it.
561struct Admitted<'p> {
562    groups: Vec<&'p proposals::Group>,
563    wide: Vec<&'p proposals::Wide>,
564}
565
566impl<'p> Admitted<'p> {
567    /// Admit the groups and wide clauses of a guard the root cuts.
568    fn admit_guard<Q: Quota + ?Sized>(
569        &mut self,
570        proposals: &'p proposals::Proposals,
571        guard: usize,
572        storage: &mut Storage,
573        budget: &mut Budget<'_, '_, Q>,
574    ) -> Result<(), BudgetError<Q::Error>> {
575        for group in proposals.parity.iter().filter(|group| group.guard == guard) {
576            storage.push_bytes(&mut self.groups, group, budget)?;
577        }
578        for clause in proposals.wide.iter().filter(|clause| clause.guard == guard) {
579            storage.push_bytes(&mut self.wide, clause, budget)?;
580        }
581        Ok(())
582    }
583}

594    table

596        .iter()
597        .map(|word| u64::from(word.count_ones()))
598        .sum()

601/// The patterns of each group refuted before any probe runs: those a clause
602/// of the group forbids, and those whose restriction to a group whose
603/// support this one contains is a pattern that group's clauses forbid. The
604/// region deciding such a pattern holds no member of the class: a member in
605/// it lies in the closed root, so under the pattern it satisfies the clause's
606/// body (the clause's rule and the chains to its base standing in the reduct,
607/// their bodies true in the candidate, as the proposal pass reads them) and
608/// holds the clause's guard, which no countermodel in the root holds, the
609/// root having cut it already or the guard being a unit
610/// (`saturation_unit`); the narrowing of that region, inside it, holds none
611/// either, which is the premise `entailed_by_refutations` asks of a refuted
612/// pattern, discharged here by the proposal pass's reading of the clause
613/// rather than by a probe. The table records it without running the probe.
614/// One word operation is charged to the floor per clause read, per pair of
615/// groups compared and per pattern examined.
616///
617/// # Errors
618/// The lease's refusal.
619fn known_refutations<Q: Quota + ?Sized>(
620    groups: &[&proposals::Group],
621    storage: &mut Storage,
622    budget: &mut Budget<'_, '_, Q>,
623) -> Result<Vec<probes::Table>, BudgetError<Q::Error>> {
624    let mut known = storage.reserve_bytes(groups.len(), budget)?;
625    for group in groups {
626        let support = &group.support.atoms[..group.support.len];
627        known.push(Words::<u64, _>::new([0; 4]));
628        let refuted = known.last_mut().expect("the current table was appended");
629        for &pattern in &group.patterns {
630            budget.charge(1)?;
631            refuted.insert(usize::from(pattern));
632        }
633        for other in groups {
634            budget.charge(1)?;
635            if other.support.len > group.support.len || std::ptr::eq(*other, *group) {
636                continue;
637            }
638            // Where `other`'s atoms sit in this support, if all of them do.
639            storage.admit_record(
640                size_of::<[usize; super::source::MAX_SUPPORT]>() as u128,
641                budget,
642            )?;
643            let mut positions = [0_usize; super::source::MAX_SUPPORT];
644            let contained = other.support.atoms[..other.support.len]
645                .iter()
646                .enumerate()
647                .all(|(k, atom)| {
648                    support.iter().position(|a| a == atom).is_some_and(|here| {
649                        positions[k] = here;
650                        true
651                    })
652                });
653            if !contained {
654                storage.release_record(size_of::<[usize; super::source::MAX_SUPPORT]>() as u128)?;
655                continue;
656            }
657            let mask = positions[..other.support.len]
658                .iter()
659                .fold(0_usize, |mask, &here| mask | (1 << here));
660            for &forbidden in &other.patterns {
661                let value = (0..other.support.len)
662                    .filter(|&k| forbidden & (1 << k) != 0)
663                    .fold(0_usize, |value, k| value | (1 << positions[k]));
664                for pattern in 0..(1_usize << group.support.len) {
665                    budget.charge(1)?;
666                    if pattern & mask == value {
667                        refuted.insert(pattern);
668                    }
669                }
670            }
671            storage.release_record(size_of::<[usize; super::source::MAX_SUPPORT]>() as u128)?;
672        }
673    }
674    Ok(known)
675}

677/// The probing rounds a preparation may run: the first by the frozen reading
678/// alone, the second under the rows it established. A later round tightens
679/// a hull the round before left inexact, one probe per pattern of each such
680/// group at the lease allowance each, and the cap bounds a preparation's
681/// probing at two rounds' probes; what the rounds established is loaded,
682/// sound for the class as before, the remaining hulls inexact.
683const MAX_ROUNDS: u64 = 2;
684
685/// Probe the groups in rounds and load their rows and bounds: round one by
686/// the frozen reading alone, leaving a group of one clause cutting its whole
687/// support to that clause, whose table already bounds the support from below
688/// by one; each later round under the parity rows the round before
689/// established, sound for the class as their joint closure is,
690/// while a round refutes more patterns than the one before, some hull is
691/// still inexact, a parity row was loaded for a later round to narrow under
692/// and no system refutes the root; each round leases its
693/// probes at `allowance` each. The systems of the last round are the
694/// preparation's.
695fn probe_rounds<Q: Quota + ?Sized>(
696    theory: &Theory,
697    admitted: &Admitted<'_>,
698    probe: &mut probes::Probe<'_>,
699    scratch: &mut NarrowingScratch,
700    probing: &mut Probing<'_, '_, Q>,
701    statistics: &mut FrozenPreparationStatistics,
702    peak: &mut u128,
703) -> Result<Systems, BudgetError<LeaseRefusal<Q::Error>>> {
704    let groups = &admitted.groups[..];
705    let mut parity: Option<ParitySystem> = None;
706    let mut cardinality: Option<CardinalitySystem> = None;
707    let (mut tables, mut refuted, known, mut pending) =
708        probing.with_budget(peak, |storage, budget| {
709            let mut tables = storage.reserve_bytes(groups.len(), budget)?;
710            tables.extend(groups.iter().map(|group| ProbedGroup {
711                support: group.support,
712                forbidden: Words::new([0; 4]),
713            }));
714            let mut refuted = storage.reserve_bytes(groups.len(), budget)?;
715            refuted.resize(groups.len(), 0);
716            let known = known_refutations(groups, storage, budget)?;
717            let mut pending = storage.reserve_bytes(groups.len(), budget)?;
718            pending.extend(0..groups.len());
719            Ok((tables, refuted, known, pending))
720        })?;
721    while !pending.is_empty() && statistics.rounds < MAX_ROUNDS {
722        statistics.rounds += 1;
723        // The first round leaves a group of one clause cutting its whole
724        // support to that clause: its table bounds the support from below by
725        // one, and only a later round's rows could refute more of it.
726        let settled_by_clause =
727            |index: usize| statistics.rounds == 1 && groups[index].patterns == [0];
728        // The probes this round runs: the patterns not known refuted.
729        let probes: u64 = pending
730            .iter()
731            .filter(|&&index| !settled_by_clause(index))
732            .map(|&index| (1_u64 << groups[index].support.len) - count_forbidden(&known[index]))
733            .sum();
734        probing
735            .lease
736            .extend(probes.saturating_mul(probing.allowance));
737        let mut progressed = false;
738        let cancellation = probing.cancellation;
739        probing.with_budget(peak, |storage, budget| {
740            for &index in &pending {
741                if settled_by_clause(index) {
742                    tables[index].forbidden = Words::new(*known[index].storage());
743                    refuted[index] = count_forbidden(&known[index]);
744                    progressed = true;
745                    continue;
746                }
747                let found = probe.table(
748                    probes::Target {
749                        group: groups[index],
750                        known: &known[index],
751                        forbidden: &mut tables[index].forbidden,
752                    },
753                    parity.as_ref(),
754                    scratch,
755                    cancellation,
756                    storage,
757                    budget,
758                )?;
759                statistics.probes += found.run;
760                progressed |= found.refuted > refuted[index];
761                refuted[index] = found.refuted;
762            }
763            Ok(())
764        })?;
765        // The old systems and worklist have served their last probe. End their
766        // lifetimes before loading the replacement, and retain the tables.
767        probing.storage.release(pending)?;
768        if let Some(previous) = parity.take() {
769            probing.storage.release_record(previous.retained_bytes())?;
770        }
771        if let Some(previous) = cardinality.take() {
772            probing.storage.release_record(previous.retained_bytes())?;
773        }
774        let loaded = load_round(theory, &tables, &admitted.wide, probing, statistics, peak)?;
775        parity = loaded.parity;
776        cardinality = loaded.cardinality;
777        pending = loaded.inexact;
778        if !progressed
779            || parity.as_ref().is_none_or(ParitySystem::root_refuted)
780            || cardinality
781                .as_ref()
782                .is_some_and(CardinalitySystem::root_refuted)
783        {
784            break;
785        }
786    }
787    if statistics.rounds == 0 && !admitted.wide.is_empty() {
788        // No group to probe: the wide clauses' bounds load alone.
789        let loaded = load_round(theory, &tables, &admitted.wide, probing, statistics, peak)?;
790        parity = loaded.parity;
791        cardinality = loaded.cardinality;
792        probing.storage.release(loaded.inexact)?;
793    }
794    statistics.refuted_patterns = refuted.iter().sum();
795    probing.storage.release(pending)?;
796    probing.storage.release(known)?;
797    probing.storage.release(refuted)?;
798    probing.storage.release(tables)?;
799    Ok((parity, cardinality))
800}

802/// Load a round's systems from `tables` and the `wide` clauses' bounds, the
803/// lease bounding work and the budget bytes.
804fn load_round<Q: Quota + ?Sized>(
805    theory: &Theory,
806    tables: &[ProbedGroup],
807    wide: &[&proposals::Wide],
808    probing: &mut Probing<'_, '_, Q>,
809    statistics: &mut FrozenPreparationStatistics,
810    peak: &mut u128,
811) -> Result<Loaded, BudgetError<LeaseRefusal<Q::Error>>> {
812    probing.with_budget(peak, |storage, budget| {
813        load(theory, tables, wide, statistics, storage, budget)
814    })
815}

817/// What a round loaded: the parity and counting systems, when any row or
818/// bound was found, and the indices of the tables whose hull is inexact.
819struct Loaded {
820    parity: Option<ParitySystem>,
821    cardinality: Option<CardinalitySystem>,
822    inexact: Vec<usize>,
823}
824
825/// The systems of the probed groups: the parity system of each table's
826/// affine hull, every equation checked against its table and eliminated to
827/// echelon form, and the counting system of the tables' bounds and the wide
828/// clauses'; with the indices of the tables whose hull is inexact.
829fn load<Q: Quota + ?Sized>(
830    theory: &Theory,
831    tables: &[ProbedGroup],
832    wide: &[&proposals::Wide],
833    statistics: &mut FrozenPreparationStatistics,
834    outer: &mut Storage,
835    budget: &mut Budget<'_, '_, Q>,
836) -> Result<Loaded, BudgetError<Q::Error>> {
837    let mut inexact = outer.reserve_bytes(tables.len(), budget)?;
838    if tables.is_empty() && wide.is_empty() {
839        return Ok(Loaded {
840            parity: None,
841            cardinality: None,
842            inexact,
843        });
844    }
845    let parity = load_parity(theory, tables, &mut inexact, statistics, outer, budget)?;
846    // Transfer only the surviving parity owner. Its builder can also have
847    // held temporary empty evidence and vector headers on a row-free input.
848    if let Some(system) = &parity {
849        outer.admit_record(system.retained_bytes(), budget)?;
850    }
851    let mut storage = Storage::with_live_bytes(outer.bytes());
852    let cardinality = load_counts(theory, tables, wide, statistics, &mut storage, budget)?;
853    if let Some(system) = &cardinality {
854        outer.admit_record(system.retained_bytes(), budget)?;
855    }
856    Ok(Loaded {
857        parity,
858        cardinality,
859        inexact,
860    })
861}

863/// Load the tables' implied equations into a parity system. The caller has
864/// reserved one inexact-index slot per input table.
865fn load_parity<Q: Quota + ?Sized>(
866    theory: &Theory,
867    tables: &[ProbedGroup],
868    inexact: &mut Vec<usize>,
869    statistics: &mut FrozenPreparationStatistics,
870    outer: &Storage,
871    budget: &mut Budget<'_, '_, Q>,
872) -> Result<Option<ParitySystem>, BudgetError<Q::Error>> {
873    let mut storage = Storage::with_live_bytes(outer.bytes());
874    let (facts, mut recognized_equations, counts) =
875        budget.in_phase(Phase::Recognition, |budget| {
876            storage.admit_record(
877                (size_of::<rows::Recognized>()
878                    - size_of::<CheckedRootFacts>()
879                    - size_of::<Vec<rows::Equation>>()
880                    - size_of::<Vec<source::Clause>>()) as u128,
881                budget,
882            )?;
883            let facts = CheckedRootFacts::none(theory, &mut storage, budget)?;
884            let mut equations = storage.reserve(0, budget)?;
885            let mut counts = rows::Counts::default();
886            for (group, table) in tables.iter().enumerate() {
887                let fixed = size_of::<hull::Hull>() as u128;
888                storage.admit_record(fixed, budget)?;
889                let proposed =
890                    storage.with_record(hull::SCRATCH_BYTES as u128, budget, |_, budget| {
891                        rows::propose(
892                            table.support,
893                            table.forbidden.storage(),
894                            &mut counts,
895                            budget,
896                        )
897                    })?;
898                if !proposed.exact {
899                    inexact.push(group);
900                }
901                for &local in proposed.equations() {
902                    let words =
903                        rows::materialize(theory, table.support, local, &mut storage, budget)?;
904                    budget.charge(1)?; // the constant
905                    let candidate = rows::Equation {
906                        words,
907                        constant: local.constant,
908                    };
909                    let valid = source::implied_by_table(
910                        table.forbidden.storage(),
911                        table.support.len,
912                        local.coefficients,
913                        local.constant,
914                        budget,
915                    )?;
916                    rows::retain(
917                        valid,
918                        candidate,
919                        local.coefficients,
920                        &mut equations,
921                        &mut counts,
922                        &mut storage,
923                        budget,
924                    )?;
925                }
926                storage.release_record(fixed)?;
927            }
928            Ok((facts, equations, counts))
929        })?;
930    statistics.inexact_hulls = counts.inexact_hulls;
931    budget.charge(1)?;
932    if recognized_equations.is_empty() {
933        return Ok(None);
934    }
935
936    let prepared = budget.in_phase(Phase::Reduction, |budget| {
937        parity::prepare(
938            &mut recognized_equations,
939            theory.atom_count(),
940            &mut storage,
941            budget,
942        )
943    })?;
944    storage.admit_record(size_of::<Vec<source::Clause>>() as u128, budget)?;
945    let recognized = rows::Recognized {
946        facts,
947        clauses: Vec::new(),
948        equations: recognized_equations,
949        counts,
950    };
951    statistics.parity = ParityStatistics::from_preparation(&recognized, &prepared, budget);
952    Ok(Some(ParitySystem::from_rows(recognized, prepared)))
953}

955/// The counting system of the tables' bounds and the wide clauses', when
956/// any was found: the bounds proposed, their families prepared and the
957/// root's double count taken, as the outer preparation takes them.
958fn load_counts<Q: Quota + ?Sized>(
959    theory: &Theory,
960    tables: &[ProbedGroup],
961    wide: &[&proposals::Wide],
962    statistics: &mut FrozenPreparationStatistics,
963    storage: &mut Storage,
964    budget: &mut Budget<'_, '_, Q>,
965) -> Result<Option<CardinalitySystem>, BudgetError<Q::Error>> {
966    let mut counted = CardinalityStatistics::default();
967    let start = budget.work();
968    let bounds = budget.in_phase(Phase::Recognition, |budget| {
969        counts::propose(theory, tables, wide, &mut counted, storage, budget)
970    })?;
971    budget.charge(1)?;
972    if bounds.is_empty() {
973        storage.release(bounds)?;
974        statistics.cardinality = counted;
975        return Ok(None);
976    }
977    let prepared = budget.in_phase(Phase::Reduction, |budget| {
978        cardinality::prepare(&bounds, theory.atom_count(), storage, budget)
979    })?;
980    counted.status = PreparationStatus::Prepared;
981    counted.rows = bounds.len();
982    counted.carriers = prepared.counts().carriers;
983    counted.attempts = prepared.counts().attempts;
984    counted.families = prepared.counts().accepted;
985    counted.refused = prepared.counts().rejected;
986    match &prepared {
987        cardinality::Prepared::RootRefuted { .. } => counted.root_refuted = true,
988        cardinality::Prepared::Families(families) => {
989            counted.occurrences = families.occurrences.len();
990        }
991    }
992    counted.preparation_work = budget.work() - start;
993    counted.preparation_bytes = budget.peak_bytes();
994    storage.admit_record(
995        (size_of::<CardinalitySystem>()
996            - size_of::<cardinality::Prepared>()
997            - size_of::<Vec<super::cardinality_source::Bound>>()) as u128,
998        budget,
999    )?;
1000    let system = CardinalitySystem::from_bounds(theory.clone(), bounds, prepared);
1001    counted.retained_bytes = system.retained_bytes();
1002    statistics.cardinality = counted;
1003    Ok(Some(system))
1004}
