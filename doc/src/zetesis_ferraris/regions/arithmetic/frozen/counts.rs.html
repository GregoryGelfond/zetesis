1//! Count bounds of the countermodel class from the probed tables and the
2//! clauses too wide for a group.
3//!
4//! A table's least and greatest allowed counts bound its support
5//! (`bounds_of_allowed`), the least one when the empty pattern is forbidden,
6//! a clause's (`lower_of_clause`), the greatest one when it is two or more
7//! and below the support's size; and the pairs no allowed pattern holds
8//! together, read from every table, grow into cliques, each an upper bound
9//! of one (`upper_of_clique`), which carry a greatest count of one, as the
10//! hull's unit rows carry one of nothing. A clause too wide for a group
11//! whose pattern cuts its whole support bounds that support from below by
12//! one with no table (`lower_of_clause`): no countermodel in the root cuts
13//! it whole, by the static refutations' argument. Atoms one side covers and
14//! the other does not get the other side's trivial bound, nothing at least
15//! or everything at most, so that the families partition one atom set both
16//! ways. The bounds carry no guard: a table's guard is cut at the closed
17//! root, by a saturation unit or already, which makes the bound a row of the
18//! class as the parity rows are.
19

37/// The bounds the tables establish, the tables' first, the cliques' after
38/// and the completing trivial bounds last, counted into `statistics` as
39/// lower and upper bounds.
40pub(super) fn propose<Q: Quota + ?Sized>(
41    theory: &Theory,
42    tables: &[ProbedGroup],
43    wide: &[&Wide],
44    statistics: &mut CardinalityStatistics,
45    storage: &mut Storage,
46    budget: &mut Budget<'_, '_, Q>,
47) -> Result<Vec<Bound>, BudgetError<Q::Error>> {
48    let width = theory.atom_count().div_ceil(128) * 2;
49    let mut bounds = storage.reserve(0, budget)?;
50    let mut exclusions: Vec<(usize, usize)> = storage.reserve(0, budget)?;
51    for (group, table) in tables.iter().enumerate() {
52        let atoms = &table.support.atoms[..table.support.len];
53        storage.admit_record(size_of::<Counted>() as u128, budget)?;
54        let Some(Counted {
55            least,
56            greatest,
57            together,
58        }) = counts(atoms.len(), &table.forbidden, budget)?
59        else {
60            storage.release_record(size_of::<Counted>() as u128)?;
61            continue;
62        };
63        if least >= 1 {
64            let bound = bound(
65                (BoundSide::Lower, least),
66                atoms.iter().copied(),
67                (group, 0),
68                width,
69                storage,
70                budget,
71            )?;
72            push_bound(&mut bounds, bound, storage, budget)?;
73            statistics.lower += 1;
74        }
75        budget.charge(1)?;
76        // A greatest count below two is carried elsewhere: of one, every pair
77        // of the support is an exclusion, so the cliques bound the support,
78        // or a superset of it, by one; of nothing, the one allowed pattern's
79        // hull is the unit rows cutting every atom. The table's own bound
80        // would duplicate them.
81        if greatest >= 2 && greatest < atoms.len() as u64 {
82            let bound = bound(
83                (BoundSide::Upper, greatest),
84                atoms.iter().copied(),
85                (group, 1),
86                width,
87                storage,
88                budget,
89            )?;
90            push_bound(&mut bounds, bound, storage, budget)?;
91            statistics.upper += 1;
92        }
93        for first in 0..atoms.len() {
94            for second in first + 1..atoms.len() {
95                budget.charge(1)?;
96                if together[first] & (1 << second) == 0 {
97                    storage.push(&mut exclusions, (atoms[first], atoms[second]), 1, budget)?;
98                    storage.push(&mut exclusions, (atoms[second], atoms[first]), 1, budget)?;
99                }
100            }
101        }
102        storage.release_record(size_of::<Counted>() as u128)?;
103    }
104    // A clause too wide for a group, its pattern cutting its whole support:
105    // no countermodel in the root cuts every atom of it (`lower_of_clause`).
106    for (index, clause) in wide.iter().enumerate() {
107        budget.charge(1)?;
108        let bound = bound(
109            (BoundSide::Lower, 1),
110            clause.support.iter().copied(),
111            (tables.len() + index, 2),
112            width,
113            storage,
114            budget,
115        )?;
116        push_bound(&mut bounds, bound, storage, budget)?;
117        statistics.lower += 1;
118    }
119    cliques(
120        &mut exclusions,
121        &mut bounds,
122        statistics,
123        width,
124        storage,
125        budget,
126    )?;
127    storage.release(exclusions)?;
128    complete(theory, &mut bounds, statistics, width, storage, budget)?;
129    Ok(bounds)
130}

132/// The trivial bounds completing each side's cover of the other's atoms: a
133/// lower bound of nothing over the atoms only upper bounds cover, an upper
134/// bound of everything over the atoms only lower bounds cover.
135fn complete<Q: Quota + ?Sized>(
136    theory: &Theory,
137    bounds: &mut Vec<Bound>,
138    statistics: &mut CardinalityStatistics,
139    width: usize,
140    storage: &mut Storage,
141    budget: &mut Budget<'_, '_, Q>,
142) -> Result<(), BudgetError<Q::Error>> {
143    let mut lower = Words::new(storage.filled(width, 0_u64, 1, budget)?);
144    let mut upper = Words::new(storage.filled(width, 0_u64, 1, budget)?);
145    for bound in bounds.iter() {
146        let side = match bound.side {
147            BoundSide::Lower => &mut lower,
148            BoundSide::Upper => &mut upper,
149        };
150        words::union_assign(side.storage_mut(), &bound.support, &mut |work| {
151            budget.charge(work)
152        })?;
153    }
154    for (side, covered, other) in [
155        (BoundSide::Lower, &lower, &upper),
156        (BoundSide::Upper, &upper, &lower),
157    ] {
158        // Two passes over the atoms, each charged: the count names the
159        // bound's value, the second pass its support.
160        let missing = || {
161            (0..theory.atom_count()).filter(|&atom| other.contains(atom) && !covered.contains(atom))
162        };
163        budget.charge(theory.atom_count() as u64)?;
164        let count = missing().count();
165        if count == 0 {
166            continue;
167        }
168        let value = match side {
169            BoundSide::Lower => 0,
170            BoundSide::Upper => count as u64,
171        };
172        budget.charge(theory.atom_count() as u64)?;
173        let bound = bound(
174            (side, value),
175            missing(),
176            (bounds.len(), 0),
177            width,
178            storage,
179            budget,
180        )?;
181        push_bound(bounds, bound, storage, budget)?;
182        match side {
183            BoundSide::Lower => statistics.lower += 1,
184            BoundSide::Upper => statistics.upper += 1,
185        }
186    }
187    storage.release(lower.into_storage())?;
188    storage.release(upper.into_storage())?;
189    Ok(())
190}

192/// What a table's allowed patterns count: the least and greatest held, and
193/// per position the positions some allowed pattern holds with it.
194struct Counted {
195    least: u64,
196    greatest: u64,
197    together: [u8; MAX_SUPPORT],
198}
199
200/// The counts of a table's allowed patterns; `None` when no pattern is
201/// allowed, which the parity rows refute.
202fn counts<Q: Quota + ?Sized>(
203    len: usize,
204    forbidden: &Table,
205    budget: &mut Budget<'_, '_, Q>,
206) -> Result<Option<Counted>, BudgetError<Q::Error>> {
207    let mut least = u64::MAX;
208    let mut greatest = 0;
209    let mut together = [0_u8; MAX_SUPPORT];
210    for pattern in 0..(1_usize << len) {
211        budget.charge(1)?;
212        if forbidden.contains(pattern) {
213            continue;
214        }
215        let count = u64::from(pattern.count_ones());
216        least = least.min(count);
217        greatest = greatest.max(count);
218        let held = u8::try_from(pattern).map_err(|_| BudgetError::Stopped(Stop::InvalidProgram))?;
219        for position in (0..len).filter(|&position| held & (1 << position) != 0) {
220            together[position] |= held;
221        }
222    }
223    Ok((least != u64::MAX).then_some(Counted {
224        least,
225        greatest,
226        together,
227    }))
228}

230/// A bound over `atoms`, unguarded, with its probed source.
231fn bound<Q: Quota + ?Sized>(
232    (side, value): (BoundSide, u64),
233    atoms: impl Iterator<Item = usize>,
234    (root, node): (usize, usize),
235    width: usize,
236    storage: &mut Storage,
237    budget: &mut Budget<'_, '_, Q>,
238) -> Result<Bound, BudgetError<Q::Error>> {
239    let mut support = Words::new(storage.filled(width, 0_u64, 1, budget)?);
240    let mut cardinality = 0;
241    for atom in atoms {
242        budget.charge(1)?;
243        support.insert(atom);
244        cardinality += 1;
245    }
246    let guard = storage.reserve(0, budget)?;
247    Ok(Bound {
248        support,
249        guard,
250        side,
251        value,
252        cardinality,
253        source: Source {
254            kind: Kind::Probed,
255            root,
256            node,
257        },
258    })
259}

261/// The exclusions, both ways round and sorted, grown into cliques: each pair
262/// seeds one, joined in turn by every excluded neighbour of its first atom
263/// that every member excludes, so a hole's pairs grow into the hole's clique
264/// and a pigeon's own into the pigeon's. Each clique found once is an upper
265/// bound of one.
266fn cliques<Q: Quota + ?Sized>(
267    edges: &mut Vec<(usize, usize)>,
268    bounds: &mut Vec<Bound>,
269    statistics: &mut CardinalityStatistics,
270    width: usize,
271    storage: &mut Storage,
272    budget: &mut Budget<'_, '_, Q>,
273) -> Result<(), BudgetError<Q::Error>> {
274    let mut scratch = storage.filled(edges.len(), (0, 0), 2, budget)?;
275    sort(
276        edges,
277        &mut scratch,
278        2,
279        |a, b, budget| {
280            budget.charge(1)?;
281            Ok(a.cmp(b))
282        },
283        budget,
284    )?;
285    storage.release(scratch)?;
286    budget.charge(edges.len() as u64)?;
287    edges.dedup();
288    let mut found: Vec<Vec<usize>> = storage.reserve(0, budget)?;
289    let mut start = 0;
290    while start < edges.len() {
291        let atom = edges[start].0;
292        let mut end = start;
293        while end < edges.len() && edges[end].0 == atom {
294            budget.charge(1)?;
295            end += 1;
296        }
297        let neighbours = &edges[start..end];
298        for &(_, seed) in neighbours {
299            budget.charge(1)?;
300            if seed < atom {
301                continue;
302            }
303            let mut clique: Vec<usize> = storage.reserve(2, budget)?;
304            storage.push(&mut clique, atom, 1, budget)?;
305            storage.push(&mut clique, seed, 1, budget)?;
306            for &(_, next) in neighbours {
307                budget.charge(1)?;
308                if next == seed {
309                    continue;
310                }
311                let mut excluded = true;
312                for &member in &clique {
313                    budget.charge(1)?;
314                    if edges.binary_search(&(member, next)).is_err() {
315                        excluded = false;
316                        break;
317                    }
318                }
319                if excluded {
320                    storage.push(&mut clique, next, 1, budget)?;
321                }
322            }
323            let mut scratch = storage.filled(clique.len(), 0, 1, budget)?;
324            sort(
325                &mut clique,
326                &mut scratch,
327                1,
328                |a, b, budget| {
329                    budget.charge(1)?;
330                    Ok(a.cmp(b))
331                },
332                budget,
333            )?;
334            storage.release(scratch)?;
335            let mut known = false;
336            for earlier in &found {
337                budget.charge(earlier.len() as u64)?;
338                if *earlier == clique {
339                    known = true;
340                    break;
341                }
342            }
343            if known {
344                storage.release(clique)?;
345                continue;
346            }
347            let bound = bound(
348                (BoundSide::Upper, 1),
349                clique.iter().copied(),
350                (clique[0], clique[1]),
351                width,
352                storage,
353                budget,
354            )?;
355            push_bound(bounds, bound, storage, budget)?;
356            statistics.upper += 1;
357            storage.push(&mut found, clique, 3, budget)?;
358            storage.embed::<usize>()?;
359        }
360        start = end;
361    }
362    for clique in found.drain(..) {
363        storage.release_embedded(clique)?;
364    }
365    storage.release(found)?;
366    Ok(())
