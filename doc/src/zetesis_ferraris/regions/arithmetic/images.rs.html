25/// Exact original-root image preparation observations. No per-entry activity is included.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
27pub struct RootImageStatistics {
28    /// Outcome of this image transaction, independently of earlier preparation.
29    pub status: PreparationStatus,
30    /// Authenticated unguarded mandatory support occurrences selected.
31    pub groups: usize,
32    /// Independently checked original forbidden-pair occurrences indexed.
33    pub pairs: usize,
34    /// Performed selection, source-checking and index construction work.
35    pub preparation_work: u64,
36    /// Performed work constructing and copying one private root.
37    pub copying_work: u64,
38    /// Performed pivot closure and final publication work.
39    pub closure_work: u64,
40    /// Shared logical requirement high-water through this attempt, including
41    /// earlier live owners and refused requirements; not incremental bytes or RSS.
42    pub peak_bytes: u128,
43    /// Fully completed rounds; a partial interrupted or absorbing round is excluded.
44    pub completed_rounds: u64,
45    /// Cuts performed in the private root, including a discarded prefix.
46    /// Publication is determined by `status` and the enclosing preparation result.
47    pub applied_cuts: u64,
48    /// A completed checked semantic contradiction, retained even if its later
49    /// publication refuses. This is not a retained root or solve-completion claim;
50    /// consult `status` and the enclosing preparation result for publication.
51    pub root_refuted: bool,
52}

67pub(super) fn apply<Q: Quota + ?Sized>(
68    result: Result<Option<RootPreparation>, PreparationError<zetesis_algebra::Failure<Q::Error>>>,
69    theory: &Theory,
70    policy: Policy,
71    parity: &mut ParityStatistics,
72    cardinality: &mut CardinalityStatistics,
73    statistics: &mut RootImageStatistics,
74    budget: &mut Budget<'_, '_, Q>,
75) -> Result<Option<RootPreparation>, PreparationError<zetesis_algebra::Failure<Q::Error>>> {
76    let Some(mut prepared) = result? else {
77        return Ok(None);
78    };
79    if !policy.enabled {
80        return Ok(Some(prepared));
81    }
82    if prepared.root_refuted() {
83        return Ok(Some(prepared));
84    }
85    let outcome = transaction::run(theory, &mut prepared, statistics, budget);
86    statistics.peak_bytes = budget.peak_bytes();
87    match outcome {
88        Ok(entered) => {
89            statistics.status = if entered {
90                PreparationStatus::Prepared
91            } else {
92                PreparationStatus::NotPrepared
93            };
94            Ok(Some(prepared))
95        }
96        Err(error) => {
97            statistics.status = status(&error);
98            let optional = matches!(
99                error,
100                BudgetError::WorkDeclined | BudgetError::StorageDeclined { .. }
101            );
102            if optional && policy.retain_on_decline {
103                return Ok(Some(prepared));
104            }
105            parity.status = statistics.status;
106            cardinality.status = statistics.status;
107            parity.retained_bytes = 0;
108            cardinality.retained_bytes = 0;
109            match error {
110                BudgetError::WorkDeclined | BudgetError::StorageDeclined { .. } => Ok(None),
111                BudgetError::Stopped(stop) => Err(PreparationError::Stopped(
112                    zetesis_algebra::Failure::Stopped(stop),
113                )),
114                BudgetError::Refused(error) => Err(PreparationError::Stopped(
115                    zetesis_algebra::Failure::Refused(error),
116                )),
117                BudgetError::Source(error) => Err(PreparationError::Source(error)),
118            }
119        }
120    }
121}

123fn status<E>(error: &BudgetError<E>) -> PreparationStatus {
124    match error {
125        BudgetError::WorkDeclined => PreparationStatus::WorkDeclined,
126        BudgetError::StorageDeclined { .. } => PreparationStatus::StorageDeclined,
127        BudgetError::Stopped(_) | BudgetError::Refused(_) | BudgetError::Source(_) => {
128            PreparationStatus::Incomplete
129        }
130    }
