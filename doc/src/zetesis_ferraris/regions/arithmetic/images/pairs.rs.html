1//! Independently reread original forbidden-pair evidence and its sparse index.
2
3use super::super::source;
4use super::{Adjacency, Budget, BudgetError, IndexConstruction, Quota, Stop, Storage, Theory};
5use zetesis_algebra::reduction::{Any, Reduction};
6
7#[derive(Clone, Copy)]
8struct Pair {
9    left: usize,
10    right: usize,
11}
12
13pub(in crate::regions::arithmetic) fn read<Q: Quota + ?Sized>(
14    theory: &Theory,
15    storage: &mut Storage,
16    budget: &mut Budget<'_, '_, Q>,
17) -> Result<(Option<Adjacency>, usize), BudgetError<Q::Error>> {
18    let mut pairs = storage.reserve::<Pair, _>(0, budget)?;
19    for ordinal in 0..theory.roots().len() {
20        if let Some(pair) = pair(theory, ordinal, storage, budget)? {
21            storage.push(&mut pairs, pair, 2, budget)?;
22        }
23    }
24    budget.charge(1)?; // inspect the complete authenticated pair population
25    if pairs.is_empty() {
26        storage.release(pairs)?;
27        // None owns its representation even though it has no index buffers.
28        storage.admit_record(size_of::<Option<Adjacency>>() as u128, budget)?;
29        budget.charge(1)?; // publish the empty-index tag
30        return Ok((None, 0));
31    }
32    let edges = pairs.iter().flat_map(|pair| {
33        [
34            Ok(Some((pair.left, pair.right))),
35            Ok(Some((pair.right, pair.left))),
36        ]
37    });
38    let forbidden = Adjacency::try_build_in(
39        theory.atom_count(),
40        edges,
41        &mut IndexConstruction { storage, budget },
42    )?;
43    let count = pairs.len();
44    storage.release(pairs)?;
45    // The two vector headers already own Adjacency. Account only any extra
46    // metadata that Option adds, rather than counting those headers twice.
47    storage.admit_record(index_tag_bytes(), budget)?;
48    budget.charge(1)?;
49    Ok((Some(forbidden), count))
50}
51
52const fn index_tag_bytes() -> u128 {
53    (size_of::<Option<Adjacency>>() - size_of::<Adjacency>()) as u128
54}
55
56/// Release the exact owner returned by the completed pair scan.
57pub(in crate::regions::arithmetic) fn release<Q: Quota + ?Sized>(
58    pairs: Option<Adjacency>,
59    storage: &mut Storage,
60    budget: &mut Budget<'_, '_, Q>,
61) -> Result<(), BudgetError<Q::Error>> {
62    budget.charge(1)?;
63    match pairs {
64        Some(pairs) => {
65            pairs.release_in(&mut IndexConstruction { storage, budget })?;
66            storage.release_record(index_tag_bytes())?;
67        }
68        None => storage.release_record(size_of::<Option<Adjacency>>() as u128)?,
69    }
70    Ok(())
71}

73fn pair<Q: Quota + ?Sized>(
74    theory: &Theory,
75    ordinal: usize,
76    storage: &mut Storage,
77    budget: &mut Budget<'_, '_, Q>,
78) -> Result<Option<Pair>, BudgetError<Q::Error>> {
79    // One retained proposal exists alongside the independently reconstructed
80    // clause and its literal scratch. The original source is read both times.
81    storage.with_record(
82        size_of::<source::Clause>() as u128,
83        budget,
84        |storage, budget| {
85            let proposed = storage.with_record(
86                source::CLAUSE_SCRATCH_BYTES as u128,
87                budget,
88                |storage, budget| source::positive_pair_clause(theory, ordinal, storage, budget),
89            )?;
90            let Some(proposed) = proposed else {
91                return Ok(None);
92            };
93            budget.charge(3)?;
94            if Any.combine(proposed.support.len != 2, proposed.forbidden != 0b11) {
95                return Ok(None);
96            }
97            let checked = storage.with_record(
98                source::CLAUSE_SCRATCH_BYTES as u128,
99                budget,
100                |storage, budget| source::clause(theory, proposed.root, storage, budget),
101            )?;
102            let Some(checked) = checked else {
103                return Err(BudgetError::Stopped(Stop::InvalidProgram));
104            };
105            budget.charge(5)?;
106            if Any.combine(
107                Any.combine(
108                    proposed.root != checked.root,
109                    proposed.support.len != checked.support.len,
110                ),
111                proposed.forbidden != checked.forbidden,
112            ) {
113                return Err(BudgetError::Stopped(Stop::InvalidProgram));
114            }
115            for (&left, &right) in proposed.support.atoms.iter().zip(&checked.support.atoms) {
116                budget.charge(1)?;
117                if left != right {
118                    return Err(BudgetError::Stopped(Stop::InvalidProgram));
119                }
120            }
121            budget.charge(3)?;
122            Ok(Some(Pair {
123                left: proposed.support.atoms[0],
124                right: proposed.support.atoms[1],
125            }))
126        },
127    )
128}
