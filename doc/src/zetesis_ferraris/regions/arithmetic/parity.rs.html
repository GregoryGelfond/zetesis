1//! Root echelon preparation over independently checked owner-coordinate rows.
2//!
3//! This is forward elimination, not reduced row echelon form. Source fields
4//! describe each row's initial check; trusted addition establishes subsequent
5//! consequences of those checked premises. No region or owner is changed.
6//!
7//! The original theory's one-atom equations arrive as units and are no row
8//! ([`prepare_after_units`]). Each decides its atom, so it is added to every
9//! row over that atom before elimination: afterwards no row names a decided
10//! atom, and the units and the rows are independent parts of one system.

37#[derive(Debug)]
38pub(super) struct Prepared {
39    pub basis: Vec<usize>,
40    /// The atoms the system decides: the direct units, one for each atom
41    /// they name and in the order recognized, then the one-atom echelon rows
42    /// in basis order. No two name one atom. Empty when the root is refuted.
43    pub units: Vec<Unit>,
44    pub index: Option<Index>,
45    pub root_refuted: bool,
46    pub counts: Counts,
47}
48
49impl Prepared {
50    /// The units that were recognized as one-atom equations and are no row:
51    /// those before the `counts.units` collected from the echelon rows.
52    pub(super) fn direct(&self) -> &[Unit] {
53        let collected = usize::try_from(self.counts.units).expect("units were pushed");
54        &self.units[..self.units.len() - collected]
55    }
56
57    /// The rank of the checked system: its direct units and echelon pivots.
58    /// A refuted root has no unit, so this is the pivots retained before the
59    /// refutation.
60    pub(super) fn rank(&self) -> usize {
61        self.direct().len() + self.basis.len()
62    }
63
64    /// The one retained record and actual payload capacities, with embedded
65    /// vector/index headers already included in the record layout.
66    pub(super) fn retained_bytes(&self) -> u128 {
67        size_of::<Self>() as u128
68            + self.basis.capacity() as u128 * size_of::<usize>() as u128
69            + self.units.capacity() as u128 * size_of::<Unit>() as u128
70            + self.index.as_ref().map_or(0, |index| {
71                index.words.capacity() as u128 * size_of::<u64>() as u128
72            })
73    }
74}

76/// Atom-major incidence bits; row identifiers are the input enumeration order.
77#[derive(Debug)]
78pub(super) struct Index {
79    pub words: Words<u64>,
80    pub row_words: usize,
81    pub rows: usize,
82    /// Executed insertions, including repeated insertion of the same bit.
83    pub insertions: u64,
84}
85
86impl Index {
87    /// Insert an admitted atom/row incidence: locate costs three, OR and its
88    /// observation counter cost two. Callers establish both coordinate bounds.
89    pub(super) fn insert<Q: Quota + ?Sized>(
90        &mut self,
91        atom: usize,
92        row: usize,
93        budget: &mut Budget<'_, '_, Q>,
94    ) -> Result<(), BudgetError<Q::Error>> {
95        assert!(row < self.rows, "index row is admitted");
96        budget.charge(3)?;
97        let (word, bit) = Words::<u64>::locate(row);
98        self.insert_bit(atom, word, bit, budget)
99    }
100
101    fn insert_bit<Q: Quota + ?Sized>(
102        &mut self,
103        atom: usize,
104        word: usize,
105        bit: u64,
106        budget: &mut Budget<'_, '_, Q>,
107    ) -> Result<(), BudgetError<Q::Error>> {
108        budget.charge(2)?;
109        self.words[atom * self.row_words + word] |= bit;
110        self.insertions += 1;
111        Ok(())
112    }
113}

115/// Eliminate checked rows in place, preserving their original atom coordinates.
116///
117/// All rows have `W = 2*ceil(atom_count/128)` words, with zero universe padding,
118/// and belong to the caller's one checked owner. A directory entry is a row
119/// identifier plus one; zero denotes absence. Each addition clears the least
120/// atom, so it strictly increases the next pivot and there are at most
121/// `r*min(r,n)` additions. Only `add` changes coefficients or constants.
122///
123/// A conclusive zero-row test publishes refutation in that same admitted chunk.
124/// It returns before counting units or building an index. Units otherwise stay
125/// private in this result until the entire caller preparation succeeds. On any
126/// error the caller must discard this preparation; no result prefix is returned.
127/// Full-chunk additions still preserve the checked equations on a refused run.
128///
129/// The input rows are borrowed, so there is no reference-algorithm row-copy
130/// term. Setup costs `15+2n`: three vector headers, directory cells plus each
131/// pushed length, four counters and two result flags. With `A` additions and
132/// rank `rho`, scans cost at most `(r+A)*(W+4)`, additions and directory decoding
133/// `A*(W+3)`, and retaining basis identifiers `6rho`. A complete nonrefuting
134/// path adds `rho*(2W+3)`, at most `u*(W+23)` for units, the index bound below,
135/// and three final result writes. The unit bound includes at most `u` vector
136/// growths and fewer than `2u` moved two-field records, under Storage's doubling
137/// schedule. Actual capacities, including simultaneous old/new buffers, enter
138/// Storage; actual charges are made by that helper, never by this upper bound.
139/// A refuting path omits those suffix terms. The shared first-bit primitive's
140/// result tag, counter increments, vector lengths and headers explain the
141/// differences from the scalar reference coefficients.
142pub(super) fn prepare<Q: Quota + ?Sized>(
143    rows: &mut [Equation],
144    atom_count: usize,
145    storage: &mut Storage,
146    budget: &mut Budget<'_, '_, Q>,
147) -> Result<Prepared, BudgetError<Q::Error>> {
148    eliminate(rows, None, atom_count, storage, budget)
149}

151/// Prepare `rows` after the one-atom equations `direct`, which are checked
152/// consequences of the same owner and whose vector `storage` already
153/// accounts.
154///
155/// The units are read in order into two masks of the rows' width: the atoms
156/// decided and, of those, the atoms held. A unit that repeats an earlier one
157/// is dropped; one that contradicts an earlier one refutes the root. Each
158/// row then has the units over its atoms added to it: its decided atoms are
159/// cleared and its constant takes the parity of those held, in one chunk of
160/// `W + 2`. That is the sum of checked equations, so the rows and the kept
161/// units have exactly the answer sets' consequences the rows and units given
162/// had. [`prepare`]'s elimination follows on rows that name no decided atom,
163/// so the units it collects from the echelon are over other atoms than the
164/// direct ones.
165///
166/// A refuted root publishes no unit, and its rank counts no unit either: the
167/// pivots retained before the refutation, and none when two units disagree.
168/// As in [`prepare`], a refutation is published in the chunk that proves it:
169/// the result's record is admitted before the first unit is read, and a
170/// unit's chunk covers the refutation it may prove.
171///
172/// Without a unit this is [`prepare`] after one test, less the units'
173/// header, which its reservation initialized. Otherwise the masks cost
174/// `2 * (3 + 2W)`, reading `u` units `13u` and adding them to `r` rows
175/// `r * (W + 2)`: the work grows with the units and with the rows, and no
176/// longer with the product of the units and the theory's width, as it did
177/// when each unit was a row.
178///
179/// On any error the caller must discard this preparation and `storage`.
180///
181/// # Panics
182/// Panics if a unit names an atom outside `atom_count`.
183pub(super) fn prepare_after_units<Q: Quota + ?Sized>(
184    direct: Vec<Unit>,
185    rows: &mut [Equation],
186    atom_count: usize,
187    storage: &mut Storage,
188    budget: &mut Budget<'_, '_, Q>,
189) -> Result<Prepared, BudgetError<Q::Error>> {
190    eliminate(rows, Some(direct), atom_count, storage, budget)
191}

193/// Read the direct units of `prepared` and add them to `rows`, as
194/// [`prepare_after_units`] describes. Returns whether the root survives: two
195/// units that disagree set `root_refuted` and leave no unit.
196fn read_units<Q: Quota + ?Sized>(
197    rows: &mut [Equation],
198    prepared: &mut Prepared,
199    atom_count: usize,
200    storage: &mut Storage,
201    budget: &mut Budget<'_, '_, Q>,
202) -> Result<bool, BudgetError<Q::Error>> {
203    budget.charge(1)?;
204    if prepared.units.is_empty() {
205        return Ok(true);
206    }
207    let width = atom_count.div_ceil(128) * 2;
208    let mut decided = Words::new(storage.filled(width, 0_u64, 1, budget)?);
209    let mut held = Words::new(storage.filled(width, 0_u64, 1, budget)?);
210    let mut kept = 0;
211    for position in 0..prepared.units.len() {
212        // One chunk a unit: read it; locate its atom; test the decided bit;
213        // then either compare the held bit and, when they disagree, publish
214        // the refutation with no unit, or set both bits and store the kept
215        // unit. The lengths the vector is cut to are part of the same chunk.
216        budget.charge(13)?;
217        let unit = prepared.units[position];
218        assert!(unit.atom < atom_count, "a unit names an admitted atom");
219        if decided.contains(unit.atom) {
220            if held.contains(unit.atom) != unit.value {
221                prepared.root_refuted = true;
222                prepared.units.clear();
223                storage.release(held.into_storage())?;
224                storage.release(decided.into_storage())?;
225                return Ok(false);
226            }
227            continue;
228        }
229        decided.insert(unit.atom);
230        if unit.value {
231            held.insert(unit.atom);
232        }
233        prepared.units[kept] = unit;
234        kept += 1;
235    }
236    prepared.units.truncate(kept);
237    for row in rows.iter_mut() {
238        add_units(row, &decided, &held, budget)?;
239    }
240    storage.release(held.into_storage())?;
241    storage.release(decided.into_storage())?;
242    Ok(true)
243}

245/// Add to a checked row every unit over one of its atoms: clear the row's
246/// decided atoms and add the parity of those held to its constant. One
247/// `W + 2` chunk covers every word's two ANDs and its parity, the constant
248/// and the chunk's admission, so a refusal leaves the row whole.
249fn add_units<Q: Quota + ?Sized>(
250    row: &mut Equation,
251    decided: &[u64],
252    held: &[u64],
253    budget: &mut Budget<'_, '_, Q>,
254) -> Result<(), BudgetError<Q::Error>> {
255    assert_eq!(
256        row.words.len(),
257        decided.len(),
258        "a row has the owner's width"
259    );
260    let width = u64::try_from(row.words.len()).expect("a row length fits a work word");
261    budget.charge(width + 2)?;
262    let mut odd = false;
263    for ((word, decided), held) in row.words.iter_mut().zip(decided).zip(held) {
264        odd ^= (*word & held).count_ones() & 1 == 1;
265        *word &= !decided;
266    }
267    row.constant ^= odd;
268    Ok(())
269}

271/// Forward elimination of `rows`. With `direct`, those units are read and
272/// added to the rows first and are published before the units the echelon
273/// yields; without it the units' vector is reserved here.
274fn eliminate<Q: Quota + ?Sized>(
275    rows: &mut [Equation],
276    direct: Option<Vec<Unit>>,
277    atom_count: usize,
278    storage: &mut Storage,
279    budget: &mut Budget<'_, '_, Q>,
280) -> Result<Prepared, BudgetError<Q::Error>> {
281    let mut pivots = storage.filled(atom_count, 0usize, 1, budget)?;
282    let basis = storage.reserve(rows.len().min(atom_count), budget)?;
283    let (units, given) = match direct {
284        Some(units) => (units, true),
285        None => (storage.reserve(0, budget)?, false),
286    };
287    storage.admit_record(
288        (size_of::<Prepared>() - size_of::<Vec<usize>>() - size_of::<Vec<Unit>>()) as u128,
289        budget,
290    )?;
291    budget.charge(6)?;
292    let mut prepared = Prepared {
293        basis,
294        units,
295        index: None,
296        root_refuted: false,
297        counts: Counts::default(),
298    };
299    if given && !read_units(rows, &mut prepared, atom_count, storage, budget)? {
300        storage.release(pivots)?;
301        return Ok(prepared);
302    }
303    for target in 0..rows.len() {
304        loop {
305            let Some(atom) = words::first(&rows[target].words, &mut |work| budget.charge(work))?
306            else {
307                let refuted = budget.bounded(3, || {
308                    if rows[target].constant {
309                        prepared.root_refuted = true;
310                        (true, 3) // constant test, refutation flag, terminal tag
311                    } else {
312                        (false, 1)
313                    }
314                })?;
315                if refuted {
316                    storage.release(pivots)?;
317                    // A refuted root publishes no unit: the refutation's
318                    // chunk covers the length of the direct units' vector,
319                    // which keeps its capacity.
320                    prepared.units.clear();
321                    return Ok(prepared);
322                }
323                break;
324            };
325            budget.charge(1)?;
326            let previous = pivots[atom];
327            if previous == 0 {
328                budget.charge(3)?;
329                pivots[atom] = target + 1;
330                prepared.counts.rank += 1;
331                storage.push(&mut prepared.basis, target, 1, budget)?;
332                break;
333            }
334            budget.charge(1)?;
335            let source = previous - 1;
336            add(rows, target, source, &mut prepared.counts, budget)?;
337        }
338    }
339    storage.release(pivots)?;
340    collect_units(rows, &mut prepared, storage, budget)?;
341    let index = budget.in_phase(super::work::Phase::Index, |budget| {
342        build_index(
343            atom_count,
344            prepared.basis.iter().map(|&row| &rows[row].words[..]),
345            storage,
346            budget,
347        )
348    })?;
349    budget.charge(2)?;
350    prepared.counts.incidences = index.insertions;
351    prepared.index = Some(index);
352    // The standalone index moved into the already-admitted Option slot. Its
353    // payload remains live; its header and scalar record now overlap that slot.
354    storage.release_record(size_of::<Index>() as u128)?;
355    budget.charge(1)?; // terminal result tag
356    Ok(prepared)
357}

359/// The sole mutation of a checked row: a complete equation addition. Its
360/// `W+2` chunk covers every XOR, the constant XOR and the observation counter.
361/// Equal padded lengths and an earlier source are established by preparation;
362/// the shared kernel's callbacks cannot refuse inside this admitted chunk.
363fn add<Q: Quota + ?Sized>(
364    rows: &mut [Equation],
365    target: usize,
366    source: usize,
367    counts: &mut Counts,
368    budget: &mut Budget<'_, '_, Q>,
369) -> Result<(), BudgetError<Q::Error>> {
370    assert!(source < target, "a pivot belongs to an earlier checked row");
371    let (earlier, remaining) = rows.split_at_mut(target);
372    let source = &earlier[source];
373    let target = &mut remaining[0];
374    let width = u64::try_from(target.words.len()).expect("a row length fits a work word");
375    budget.charge(width + 2)?;
376    words::xor_assign(&mut target.words, &source.words, &mut || {
377        Ok::<_, Infallible>(())
378    })
379    .unwrap_or_else(|never| match never {});
380    target.constant ^= source.constant;
381    counts.additions += 1;
382    Ok(())
383}

385fn collect_units<Q: Quota + ?Sized>(
386    rows: &[Equation],
387    prepared: &mut Prepared,
388    storage: &mut Storage,
389    budget: &mut Budget<'_, '_, Q>,
390) -> Result<(), BudgetError<Q::Error>> {
391    for row in &prepared.basis {
392        budget.charge(1)?;
393        let row = &rows[*row];
394        let count = words::count(&row.words, &mut |work| budget.charge(work))?;
395        budget.charge(1)?;
396        if count == 1 {
397            let atom = words::first(&row.words, &mut |work| budget.charge(work))?
398                .expect("the unchanged row has one atom");
399            storage.push(
400                &mut prepared.units,
401                Unit {
402                    atom,
403                    value: row.constant,
404                },
405                2,
406                budget,
407            )?;
408            budget.charge(1)?;
409            prepared.counts.units += 1;
410        }
411    }
412    Ok(())
413}

415/// Build the same padded incidence representation for parity rows or family
416/// occurrences. Supports use the exact owner's `W` words with cleared padding.
417/// Their iterator supplies borrowed views in constant time; it performs no
418/// source rereads or uncharged semantic transformation.
419///
420/// For `t` supports and `I` set bits, the complete bound is
421/// `2*n*B(t) + 6 + 3*t + t*W + 7*I`: initialized payload plus vector lengths,
422/// header/metadata, one row location, shared set-bit enumeration, and OR plus
423/// an insertion counter per incidence. Guard insertions later cost five each.
424/// Matrix/byte dimension checks precede allocation; unrepresentable allocation
425/// dimensions remain an allocation failure, never a logical contradiction.
426pub(super) fn build_index<'a, Q: Quota + ?Sized>(
427    atom_count: usize,
428    supports: impl ExactSizeIterator<Item = &'a [u64]>,
429    storage: &mut Storage,
430    budget: &mut Budget<'_, '_, Q>,
431) -> Result<Index, BudgetError<Q::Error>> {
432    budget.charge(0)?; // global control precedes allocation-dimension refusal
433    let rows = supports.len();
434    let row_words = rows.div_ceil(128) * 2;
435    let length = atom_count
436        .checked_mul(row_words)
437        .ok_or(BudgetError::Stopped(Stop::Allocation))?;
438    let words = Words::new(storage.filled(length, 0, 1, budget)?);
439    storage.admit_record((size_of::<Index>() - size_of::<Vec<u64>>()) as u128, budget)?;
440    budget.charge(3)?;
441    let mut index = Index {
442        words,
443        row_words,
444        rows,
445        insertions: 0,
446    };
447    for (row, support) in supports.enumerate() {
448        budget.charge(3)?;
449        let (word, bit) = Words::<u64>::locate(row);
450        words::visit_set_bits(
451            support,
452            budget,
453            &mut |budget, work| budget.charge(work),
454            &mut |budget, atom| index.insert_bit(atom, word, bit, budget),
455        )?;
456    }
457    Ok(index)
