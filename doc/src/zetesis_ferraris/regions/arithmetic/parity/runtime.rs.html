1//! One parity inspection batch over checked original-coordinate equations.
2//!
3//! Inspect touched basis rows against one unchanged region, save that input,
4//! then apply collected consequences in basis order. A decision made by this
5//! batch remains new to the snapshot, so the next application revisits its
6//! incident rows. A changed batch does not claim a local fixed point.
7

26/// The last completely inspected input of one original parity system.
27///
28/// Reuse requires the same system and a descendant region. Cloning is an exact
29/// independent value copy; production search copies must use the charged
30/// copying boundary rather than treating that copy as free work.
31#[derive(Clone, Debug)]
32pub struct ParityKnowledge {
33    snapshot: Snapshot,
34}
35
36impl ParityKnowledge {
37    fn empty() -> Self {
38        Self {
39            snapshot: Snapshot::default(),
40        }
41    }
42
43    /// This record and its actual buffer capacity, excluding shared equations.
44    #[must_use]
45    pub fn retained_bytes(&self) -> u128 {
46        size_of::<Self>() as u128 + self.snapshot.allocated_bytes()
47    }
48
49    /// Copy initialized knowledge with metered payload work and capacity reuse.
50    ///
51    /// Cancellation is polled before any semantic work. Any failure leaves the
52    /// source unchanged and the destination unavailable. Invalidating a failed
53    /// destination is attempt cleanup, not a semantic copy. This helper does
54    /// not refund: its caller settles the encompassing copy attempt.
55    ///
56    /// # Errors
57    /// Returns the original quota/control failure or typed allocation stop.
58    pub fn copy_reserved<Q: Quota + ?Sized>(
59        &self,
60        destination: &mut Option<Self>,
61        cancellation: &Cancellation,
62        work: &mut Meter<'_, Q>,
63    ) -> Result<(), Failure<Q::Error>> {
64        runtime::copy_attempt(
65            self,
66            destination,
67            cancellation,
68            work,
69            |work| {
70                charge(work, 4)?; // empty vector header3 and initialization flag1
71                Ok(Self::empty())
72            },
73            |target, source, work| target.snapshot.copy_from(&source.snapshot, work),
74        )
75    }
76}

78impl ParitySystem {
79    /// Create uninitialized knowledge without allocation or inspecting rows.
80    #[must_use]
81    pub fn knowledge(&self) -> ParityKnowledge {
82        ParityKnowledge::empty()
83    }

85    /// Reserve the fresh knowledge and its replacement buffer before inspection.
86    /// The empty snapshot and the buffer each own a vector header. The buffer's
87    /// payload passes to the snapshot on its first metered initialization.
88    pub(in super::super) fn probe_knowledge<Q: Quota + ?Sized>(
89        &self,
90        region: &Region,
91        storage: &mut Storage,
92        budget: &mut Budget<'_, '_, Q>,
93    ) -> Result<(ParityKnowledge, Vec<u64>), BudgetError<Q::Error>> {
94        storage.admit_record(size_of::<ParityKnowledge>() as u128, budget)?;
95        let words = if self.prepared.root_refuted || self.prepared.basis.is_empty() {
96            0
97        } else {
98            region.mask_words().checked_mul(2).ok_or(Stop::Allocation)?
99        };
100        Ok((self.knowledge(), storage.reserve_bytes(words, budget)?))
101    }
102
103    /// Whether the rows are consistent under the region's decisions: the
104    /// decided atoms substituted, the rows over the open atoms eliminated
105    /// to echelon form, and no row left reading `0 = 1`. Complete for the
106    /// rows as a linear system, where [`Self::inspect_reserved`] applies one
107    /// row at a time. Charged per word read and combined; the elimination's
108    /// scratch is allocated for the call.
109    ///
110    /// # Errors
111    /// The meter's refusal or a stop.
112    pub fn consistent_in_reserved<Q: Quota + ?Sized>(
113        &self,
114        region: &Region,
115        work: &mut Meter<'_, Q>,
116    ) -> Result<bool, Failure<Q::Error>> {
117        // The standalone operation has a work account and no optional byte
118        // ceiling. Both callers use the same elimination and fallible storage.
119        let cancellation = Cancellation::default();
120        let mut budget = Budget::new(work, u64::MAX, usize::MAX, &cancellation);
121        self.probe_consistent(region, &Storage::default(), &mut budget)
122            .map_err(|error| match error {
123                BudgetError::Refused(error) => Failure::Refused(error),
124                BudgetError::Stopped(stop) => Failure::Stopped(stop),
125                BudgetError::StorageDeclined { .. } => Failure::Stopped(Stop::Allocation),
126                BudgetError::WorkDeclined => Failure::Stopped(Stop::WorkLimit),
127                BudgetError::Source(_) => Failure::Stopped(Stop::InvalidProgram),
128            })
129    }

131    /// The elimination workspace lives alongside the query's other probe
132    /// owners. Growing the pivot directory admits old and new capacities; a
133    /// retained row transfers its vector header into that directory.
134    pub(in super::super) fn probe_consistent<Q: Quota + ?Sized>(
135        &self,
136        region: &Region,
137        outer: &Storage,
138        budget: &mut Budget<'_, '_, Q>,
139    ) -> Result<bool, BudgetError<Q::Error>> {
140        let (held, cut) = region.decision_words();
141        let width = held.len();
142        let mut storage = Storage::with_live_bytes(outer.bytes());
143        let mut pivots: Vec<(Vec<u64>, bool)> = storage.reserve_bytes(0, budget)?;
144        for &source in &self.prepared.basis {
145            let row = &self.recognized.equations[source];
146            let coefficients = &row.words[..width];
147            charge(budget.meter_mut(), 2 * width as u64 + 1)?;
148            // Substitute the decided atoms: held ones flip the constant.
149            let mut constant = row.constant;
150            let mut reduced = storage.reserve_bytes(width, budget)?;
151            for ((&word, &held), &cut) in coefficients.iter().zip(held).zip(cut) {
152                constant ^= (word & held).count_ones() % 2 == 1;
153                reduced.push(word & !(held | cut));
154            }
155            // Eliminate against the pivots found so far.
156            for (pivot, pivot_constant) in &pivots {
157                charge(budget.meter_mut(), width as u64 + 1)?;
158                let Some(bit) = lowest_bit(pivot) else {
159                    continue;
160                };
161                if reduced[bit / 64] & (1 << (bit % 64)) != 0 {
162                    for (target, source) in reduced.iter_mut().zip(pivot) {
163                        *target ^= source;
164                    }
165                    constant ^= pivot_constant;
166                }
167            }
168            charge(budget.meter_mut(), width as u64 + 1)?;
169            if lowest_bit(&reduced).is_none() {
170                if constant {
171                    return Ok(false);
172                }
173                storage.release(reduced)?;
174                continue;
175            }
176            storage.push_bytes(&mut pivots, (reduced, constant), budget)?;
177            storage.embed::<u64>()?;
178        }
179        Ok(true)
180    }

182    /// Inspect applicable rows once and apply their consequences in row order.
183    ///
184    /// The region uses this system's original atom coordinates. Initialized
185    /// knowledge belongs to this exact system and an ancestor of the region;
186    /// the same-system requirement is a caller precondition, not an identity
187    /// token checked by this operation. Detected width or lineage violations
188    /// retain an internal-instantiation failure, never a semantic refutation.
189    ///
190    /// Cancellation is checked before mutation. The operation refunds unused
191    /// permits once and reports all performed work and fresh decisions on
192    /// every outcome. After a failure, abandon both region and knowledge.
193    /// Worker scratch can be reused: its next invocation resets live contents.
194    /// `Changed` means progress, not a fixed point of this or other subjects.
195    pub fn inspect_reserved<Q: Quota + ?Sized>(
196        &self,
197        region: &mut Region,
198        known: &mut ParityKnowledge,
199        scratch: &mut NarrowingScratch,
200        cancellation: &Cancellation,
201        work: &mut Meter<'_, Q>,
202    ) -> ArithmeticAttempt<Failure<Q::Error>> {
203        self.inspect_with_buffer(region, known, scratch, cancellation, None, work)
204    }
205
206    /// A probe supplies its accounted buffer at the ordinary snapshot
207    /// replacement step. Both paths charge the same initialization prefix.
208    pub(in super::super) fn inspect_with_buffer<Q: Quota + ?Sized>(
209        &self,
210        region: &mut Region,
211        known: &mut ParityKnowledge,
212        scratch: &mut NarrowingScratch,
213        cancellation: &Cancellation,
214        supplied: Option<&mut Vec<u64>>,
215        work: &mut Meter<'_, Q>,
216    ) -> ArithmeticAttempt<Failure<Q::Error>> {
217        let before = work.charged();
218        let mut statistics = NarrowingStatistics::default();
219        let result = cancellation.poll().map_err(Failure::from).and_then(|()| {
220            known
221                .snapshot
222                .validate(region, self.recognized.facts.owner().atom_count(), work)?;
223            inspect(
224                self,
225                region,
226                &mut known.snapshot,
227                (&mut scratch.arithmetic, supplied),
228                work,
229                &mut statistics,
230            )
231        });
232        work.refund();
233        statistics.work = work.charged() - before;
234        ArithmeticAttempt { result, statistics }
235    }
236}

238/// The lowest set bit of a word vector, as an atom index.
239fn lowest_bit(words: &[u64]) -> Option<usize> {
240    words
241        .iter()
242        .enumerate()
243        .find(|(_, word)| **word != 0)
244        .map(|(index, word)| index * 64 + word.trailing_zeros() as usize)
245}

247fn inspect<Q: Quota + ?Sized>(
248    system: &ParitySystem,
249    region: &mut Region,
250    snapshot: &mut Snapshot,
251    (scratch, supplied): (&mut ArithmeticScratch, Option<&mut Vec<u64>>),
252    work: &mut Meter<'_, Q>,
253    statistics: &mut NarrowingStatistics,
254) -> Result<SubjectStep, Failure<Q::Error>> {
255    charge(work, 1)?;
256    scratch.consequences.clear();
257    charge(work, 1)?;
258    if system.prepared.root_refuted {
259        return Ok(SubjectStep::Refuted);
260    }
261    charge(work, 1)?;
262    if system.prepared.basis.is_empty() {
263        return Ok(SubjectStep::Unchanged);
264    }
265    charge(work, 1)?;
266    let index = system
267        .prepared
268        .index
269        .as_ref()
270        .expect("complete nonrefuting preparation has its index");
271    snapshot.select_using(region, index, scratch, supplied, work)?;
272    if words::first(&scratch.touched, &mut |amount| charge(work, amount))?.is_none() {
273        // No row can read an open mask or collect a consequence. Retain this
274        // inspected input even when its new decisions touch no parity row.
275        snapshot.publish(region, work)?;
276        charge(work, 1)?; // the reached unchanged result
277        return Ok(SubjectStep::Unchanged);
278    }
279    prepare_open(region, &mut scratch.open, work)?;
280    let (held, _) = region.decision_words();
281    let ArithmeticScratch {
282        touched,
283        open,
284        consequences,
285    } = scratch;
286    for (word, mask) in touched.iter().enumerate() {
287        charge(work, 1)?;
288        let mut remaining = Words::<u64>::bits(*mask);
289        while let Some(bit) = remaining.next_with(|| charge(work, 3))? {
290            charge(work, 2)?;
291            let ordinal = word * 64 + bit;
292            charge(work, 2)?;
293            let source = system.prepared.basis[ordinal];
294            let row = &system.recognized.equations[source];
295            let consistent = collect(row, held, open, consequences, work)?;
296            charge(work, 1)?;
297            if !consistent {
298                return Ok(SubjectStep::Refuted);
299            }
300        }
301    }
302    // This input is the basis of every row just inspected. Saving the region
303    // after publication instead would lose triggers created by this batch.
304    snapshot.publish(region, work)?;
305    apply(consequences, region, work, statistics)
306}

308fn prepare_open<Q: Quota + ?Sized>(
309    region: &Region,
310    open: &mut Words<u64>,
311    work: &mut Meter<'_, Q>,
312) -> Result<(), Failure<Q::Error>> {
313    runtime::prepare_words(open, region.mask_words(), work)?;
314    let (held, cut) = region.decision_words();
315    for ((output, held), cut) in open.iter_mut().zip(held).zip(cut) {
316        charge(work, 2)?;
317        *output = !(*held | *cut);
318    }
319    charge(work, 2)?;
320    let tail = region.len() % 64;
321    if tail != 0 {
322        charge(work, 3)?;
323        *open
324            .last_mut()
325            .expect("a partial word belongs to a nonempty region") &= u64::MAX >> (64 - tail);
326    }
327    Ok(())
328}

330fn collect<Q: Quota + ?Sized>(
331    row: &Equation,
332    held: &[u64],
333    open: &[u64],
334    consequences: &mut Vec<(usize, bool)>,
335    work: &mut Meter<'_, Q>,
336) -> Result<bool, Failure<Q::Error>> {
337    // Admitted rows have the same universe with zero 128-bit padding. Region
338    // masks use the shorter 64-bit extent; do not read the extra padding word.
339    let coefficients = &row.words[..held.len()];
340    let held_count = runtime::counted_intersection(coefficients, held, work)?;
341    let open_count = runtime::counted_intersection(coefficients, open, work)?;
342    charge(work, 3)?; // remainder, Boolean conversion and zero-open test
343    let odd = held_count % 2 != 0;
344    if open_count == 0 {
345        charge(work, 1)?;
346        return Ok(odd == row.constant);
347    }
348    charge(work, 1)?;
349    if open_count == 1 {
350        let atom = first_common(coefficients, open, work)?;
351        charge(work, 1)?;
352        let value = row.constant ^ odd;
353        runtime::push(consequences, (atom, value), 2, work)?;
354    }
355    Ok(true)
356}

358/// Read only as far as the established unique common bit. The singleton test
359/// belongs to the preceding immutable intersection count, not this locator.
360fn first_common<Q: Quota + ?Sized>(
361    left: &[u64],
362    right: &[u64],
363    work: &mut Meter<'_, Q>,
364) -> Result<usize, Failure<Q::Error>> {
365    for (index, (left, right)) in left.iter().zip(right).enumerate() {
366        charge(work, 1)?;
367        let common = *left & *right;
368        if let Some(bit) = words::first(&[common], &mut |amount| charge(work, amount))? {
369            charge(work, 2)?;
370            return Ok(index * 64 + bit);
371        }
372    }
373    unreachable!("the unchanged intersection has exactly one bit")
374}

376fn apply<Q: Quota + ?Sized>(
377    consequences: &[(usize, bool)],
378    region: &mut Region,
379    work: &mut Meter<'_, Q>,
380    statistics: &mut NarrowingStatistics,
381) -> Result<SubjectStep, Failure<Q::Error>> {
382    charge(work, 1)?;
383    let mut changed = false;
384    for consequence in consequences {
385        charge(work, 2)?; // the two initialized consequence fields
386        let (atom, value) = *consequence;
387        let decision = work.bounded(9, || {
388            let decision = region.decision(atom);
389            // Admitted coordinate: bound test1, locate3, held AND/test2;
390            // optionally cut AND/test2; then the returned decision tag1.
391            (decision, if decision == Some(true) { 7 } else { 9 })
392        })?;
393        if let Some(previous) = decision {
394            charge(work, 1)?;
395            if previous != value {
396                return Ok(SubjectStep::Refuted);
397            }
398        } else {
399            publish(region, atom, value, statistics, work)?;
400            charge(work, 1)?;
401            changed = true;
402        }
403    }
404    charge(work, 1)?; // the reached final progress selection
405    Ok(if changed {
406        SubjectStep::Changed
407    } else {
408        SubjectStep::Unchanged
409    })
410}

412fn publish<Q: Quota + ?Sized>(
413    region: &mut Region,
414    atom: usize,
415    value: bool,
416    statistics: &mut NarrowingStatistics,
417    work: &mut Meter<'_, Q>,
418) -> Result<(), Failure<Q::Error>> {
419    charge(work, 5)?; // counter selection, two additions and overflow tests
420    let decisions = if value {
421        &mut statistics.held
422    } else {
423        &mut statistics.cut
424    };
425    let propagated = statistics
426        .propagations
427        .checked_add(1)
428        .expect("each original atom changes at most once");
429    let decided = decisions
430        .checked_add(1)
431        .expect("a polarity count is bounded by original atoms");
432    // The open-coordinate Region mutation costs 11: bounds1, locate3, two
433    // AND/tests4, polarity choice1, OR1 and result tag1. The caller selects
434    // hold/cut1 and writes the two already checked receipt fields2.
435    charge(work, 14)?;
436    let accepted = if value {
437        region.hold(atom)
438    } else {
439        region.cut(atom)
440    };
441    assert!(
442        accepted,
443        "an admitted open coordinate accepts either decision"
444    );
445    statistics.propagations = propagated;
446    *decisions = decided;
447    Ok(())
448}
