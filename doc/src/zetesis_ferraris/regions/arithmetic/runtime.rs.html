2//!
3//! These helpers borrow the existing Meter. They neither poll nor refund;
4//! the outer attempted operation owns both. Allocation internals and ABI

18/// One complete arithmetic inspection/application batch and its real prefix.
19/// A changed batch is local progress, not a joint or local fixed-point claim.
20#[derive(Debug)]
21#[must_use = "account the receipt and inspect the result before reusing state"]
22pub struct ArithmeticAttempt<E = Stop> {
23    /// Local progress or the unchanged caller/control failure.
24    pub result: Result<SubjectStep, E>,
25    /// Performed work and applied decisions, including failed prefixes.
26    pub statistics: NarrowingStatistics,
27}

29/// Sequentially shared worker storage; never copied with a region.
30#[derive(Debug, Default)]
31pub(in crate::regions) struct ArithmeticScratch {

33    pub(super) open: Words<u64>,
34    pub(super) consequences: Vec<(usize, bool)>,

37impl ArithmeticScratch {
38    /// Payload capacities only: the enclosing worker counts this record once.
39    pub(in crate::regions) fn allocated_bytes(&self) -> u128 {
40        (self.touched.capacity() as u128 + self.open.capacity() as u128) * size_of::<u64>() as u128
41            + self.consequences.capacity() as u128 * size_of::<(usize, bool)>() as u128
42    }

45/// Common copy boundary for opaque arithmetic knowledge.
46///
47/// Poll before inspecting the optional destination. The initializer charges its
48/// own real empty record fields; this helper charges the presence test and a
49/// new Some tag. The copy closure charges all initialized semantic payload.
50/// Any failure makes the destination unavailable. That finalization and Drop
51/// are failure cleanup, not another semantic read or copy after refusal.
52/// No refund or exit poll occurs here; the outer owner publishes the Meter
53/// delta and refunds at its existing copy-attempt boundary.
54pub(super) fn copy_attempt<T, Q: Quota + ?Sized>(
55    source: &T,
56    destination: &mut Option<T>,
57    cancellation: &Cancellation,
58    work: &mut Meter<'_, Q>,
59    initialize: impl FnOnce(&mut Meter<'_, Q>) -> Result<T, Failure<Q::Error>>,
60    copy: impl FnOnce(&mut T, &T, &mut Meter<'_, Q>) -> Result<(), Failure<Q::Error>>,
61) -> Result<(), Failure<Q::Error>> {
62    let result = cancellation
63        .poll()
64        .map_err(Failure::Stopped)
65        .and_then(|()| {
66            charge(work, 1)?;
67            if destination.is_none() {
68                let empty = initialize(work)?;
69                charge(work, 1)?;
70                *destination = Some(empty);
71            }
72            copy(
73                destination
74                    .as_mut()
75                    .expect("copy destination is initialized"),
76                source,
77                work,
78            )
79        });
80    if result.is_err() {
81        *destination = None;
82    }
83    result
84}

86/// Admit a guaranteed primitive chunk immediately before its execution.
87/// The caller runs all of it after success; use `Meter::bounded` directly when
88/// the operation may itself return a shorter performed prefix.
89pub(super) fn charge<Q: Quota + ?Sized>(
90    work: &mut Meter<'_, Q>,
91    count: u64,
92) -> Result<(), Failure<Q::Error>> {
93    work.bounded(count, || ((), count))
94}

96fn reserved<T, Q: Quota + ?Sized>(
97    length: usize,
98    work: &mut Meter<'_, Q>,
99) -> Result<Vec<T>, Failure<Q::Error>> {
100    charge(work, 3)?;
101    let mut values = Vec::new();
102    values
103        .try_reserve_exact(length)
104        .map_err(|_| Failure::Stopped(Stop::Allocation))?;
105    Ok(values)
106}

108/// Replace initialized values, reusing capacity only when sufficient.
109/// A refused replacement may have overwritten a prefix; discard the target.
110/// The source accessor is an internal infallible value/read, not a public hook.
111fn overwrite<T: Copy, Q: Quota + ?Sized>(
112    target: &mut Vec<T>,
113    length: usize,
114    fields: u64,
115    mut value: impl FnMut(usize) -> T,
116    acquire: impl FnOnce(usize, &mut Meter<'_, Q>) -> Result<Vec<T>, Failure<Q::Error>>,
117    work: &mut Meter<'_, Q>,
118) -> Result<(), Failure<Q::Error>> {
119    charge(work, 1)?;
120    if target.capacity() < length {
121        let mut next = acquire(length, work)?;
122        for index in 0..length {
123            charge(
124                work,
125                fields
126                    .checked_add(1)
127                    .ok_or(Failure::Stopped(Stop::WorkLimit))?,
128            )?;
129            next.push(value(index));
130        }
131        charge(work, 6)?;
132        std::mem::swap(target, &mut next);
133    } else {
134        let previous = target.len();
135        for (index, destination) in target.iter_mut().enumerate().take(length.min(previous)) {
136            charge(work, fields)?;
137            *destination = value(index);
138        }
139        for index in previous..length {
140            charge(
141                work,
142                fields
143                    .checked_add(1)
144                    .ok_or(Failure::Stopped(Stop::WorkLimit))?,
145            )?;
146            target.push(value(index));
147        }
148        charge(work, 1)?;
149        if previous > length {
150            charge(work, 1)?;
151            target.truncate(length);
152        }
153    }
154    Ok(())
155}

157/// Set the exact logical word length to zero, preserving sufficient capacity.
158pub(super) fn prepare_words<Q: Quota + ?Sized>(
159    target: &mut Words<u64>,
160    length: usize,
161    work: &mut Meter<'_, Q>,
162) -> Result<(), Failure<Q::Error>> {
163    overwrite(target.storage_mut(), length, 1, |_| 0, reserved, work)
164}

166/// Initialize the snapshot through the same metered replacement operation
167/// with storage already reserved by the caller. Moving the three-word header
168/// takes the place of building it in the ordinary reservation.
169pub(super) fn prepare_supplied_words<Q: Quota + ?Sized>(
170    target: &mut Words<u64>,
171    length: usize,
172    supplied: Vec<u64>,
173    work: &mut Meter<'_, Q>,
174) -> Result<(), Failure<Q::Error>> {
175    overwrite(
176        target.storage_mut(),
177        length,
178        1,
179        |_| 0,
180        |length, work| {
181            charge(work, 3)?;
182            debug_assert!(supplied.is_empty() && supplied.capacity() >= length);
183            Ok(supplied)
184        },
185        work,
186    )
187}

189/// Copy initialized words without allocation, preserving the next-word refusal.
190/// Bulk only an already-admitted run; its body has no callbacks or early exit.
191pub(super) fn copy_words<Q: Quota + ?Sized, S: AsRef<[u64]> + AsMut<[u64]>>(
192    target: &mut Words<u64, S>,
193    source: &[u64],
194    work: &mut Meter<'_, Q>,
195) -> Result<(), Failure<Q::Error>> {
196    assert_eq!(target.len(), source.len(), "snapshot word shapes agree");
197    // Acquire the backing once, as the original mutable iterator did. A
198    // generic backing's AsMut need not be free of observable effects.
199    let target = &mut **target;
200    let length = target.len().min(source.len());
201    let mut copied = 0;
202    while copied < length {
203        let available = usize::try_from(work.remaining()).unwrap_or(usize::MAX);
204        let covered = available.min(length - copied);
205        if covered == 0 {
206            // Replenish only for the next original one-word primitive.
207            charge(work, 1)?;
208            target[copied] = source[copied];
209            copied += 1;
210        } else {
211            // This complete, infallible slice is already covered by the grant:
212            // no quota callback or poll can occur between its word copies.
213            let count = u64::try_from(covered).expect("covered words fit the permit balance");
214            charge(work, count)?;
215            let end = copied + covered;
216            for (target, source) in target[copied..end].iter_mut().zip(&source[copied..end]) {
217                *target = *source;
218            }
219            copied = end;
220        }
221    }
222    Ok(())
223}

225/// Copy the declared initialized scalar fields, never unused capacity.
226pub(super) fn copy_values<T: Copy, Q: Quota + ?Sized>(
227    target: &mut Vec<T>,
228    source: &[T],
229    fields: u64,
230    work: &mut Meter<'_, Q>,
231) -> Result<(), Failure<Q::Error>> {
232    overwrite(
233        target,
234        source.len(),
235        fields,
236        |index| source[index],
237        reserved,
238        work,
239    )
240}

242/// Append a scalar record under the existing preparation-style doubling rule.
243/// Actual initialized transfers are charged; a failed destination is unusable.
244pub(super) fn push<T: Copy, Q: Quota + ?Sized>(
245    values: &mut Vec<T>,
246    value: T,
247    fields: u64,
248    work: &mut Meter<'_, Q>,
249) -> Result<(), Failure<Q::Error>> {
250    charge(work, 1)?;
251    if values.len() == values.capacity() {
252        let capacity = values.capacity().saturating_mul(2).max(1);
253        let mut next = reserved(capacity, work)?;
254        for old in values.iter() {
255            charge(
256                work,
257                fields
258                    .checked_add(1)
259                    .ok_or(Failure::Stopped(Stop::WorkLimit))?,
260            )?;
261            next.push(*old);
262        }
263        charge(work, 6)?;
264        std::mem::swap(values, &mut next);
265    }
266    charge(
267        work,
268        fields
269            .checked_add(1)
270            .ok_or(Failure::Stopped(Stop::WorkLimit))?,
271    )?;
272    values.push(value);
273    Ok(())
274}

276/// AND/popcount plus its caller-owned initialization and accumulator additions.
277/// On success the charge is 3*length+1. Each AND admits one; each following
278/// popcount and guaranteed addition admits two. No whole-row precharge occurs.
279pub(super) fn counted_intersection<Q: Quota + ?Sized>(
280    left: &[u64],
281    right: &[u64],
282    work: &mut Meter<'_, Q>,
283) -> Result<usize, Failure<Q::Error>> {
284    assert_eq!(left.len(), right.len(), "intersection word shapes agree");
285    charge(work, 1)?;
286    let mut popcount = false;
287    words::intersection_count(left, right, &mut || {
288        charge(work, if popcount { 2 } else { 1 })?;
289        popcount = !popcount;
290        Ok(())
291    })
292}
