1//! Bounded source patterns for checked candidate equations.
2//!
3//! A clause names an original root ordinal and its canonical atom coordinates.
4//! It is a proposal, not an entailment certificate. The checker rereads those
5//! ordinals from the exact admitted owner before trusting an equation. No
6//! formula, atom coordinate or reduct is rewritten.

21/// The finite pattern table has at most 256 bits, in four words. This limit
22/// bounds recognition only; a larger formula remains available to narrowing.
23pub(super) const MAX_SUPPORT: usize = 8;

25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub(super) struct Literal {
27    pub atom: usize,
28    pub positive: bool,
29}

31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub(super) struct Support {
33    pub atoms: [usize; MAX_SUPPORT],
34    pub len: usize,
35}

37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub(super) struct Clause {
39    pub support: Support,
40    /// The body-true pattern forbidden by this original constraint.
41    pub forbidden: u8,
42    pub root: usize,
43}

45/// Fixed literal scratch and the output record while decoding one clause.
46pub(super) const CLAUSE_SCRATCH_BYTES: usize =

57/// Read a signed atom without recursively interpreting another formula shape.
58/// A previously read view is consumed directly, avoiding another source read.
59pub(super) fn literal<Q: Quota + ?Sized>(
60    theory: &Theory,
61    view: NodeView<'_>,
62    budget: &mut Budget<'_, '_, Q>,
63) -> Result<Option<Literal>, BudgetError<Q::Error>> {
64    budget.charge(1)?; // inspect the source kind
65    let (atom, positive) = match view {
66        NodeView::Atom(atom) => (atom, true),
67        NodeView::Implies(left, right) => {
68            let right = budget.node(theory, right)?;
69            budget.charge(1)?;
70            if !matches!(right, NodeView::False) {
71                budget.charge(1)?;
72                return Ok(None);
73            }
74            let left = budget.node(theory, left)?;
75            budget.charge(1)?;
76            let NodeView::Atom(atom) = left else {
77                budget.charge(1)?;
78                return Ok(None);
79            };
80            (atom, false)
81        }
82        _ => {
83            budget.charge(1)?;
84            return Ok(None);
85        }
86    };
87    budget.charge(1)?;
88    if atom >= theory.atom_count() {
89        budget.charge(1)?;
90        return Err(BudgetError::Source(AdmissionError::Atom));
91    }
92    budget.charge(3)?; // atom, polarity and result tag
93    Ok(Some(Literal { atom, positive }))
94}

96/// Authenticate only the canonical truth formula, falsum implying falsum.
97/// Other tautologies are deliberately outside this source view.
98pub(super) fn canonical_truth<Q: Quota + ?Sized>(
99    theory: &Theory,
100    view: NodeView<'_>,
101    budget: &mut Budget<'_, '_, Q>,
102) -> Result<bool, BudgetError<Q::Error>> {
103    budget.charge(1)?;
104    let NodeView::Implies(left, right) = view else {
105        budget.charge(1)?;
106        return Ok(false);
107    };
108    let left = budget.node(theory, left)?;
109    budget.charge(1)?;
110    if !matches!(left, NodeView::False) {
111        budget.charge(1)?;
112        return Ok(false);
113    }
114    let right = budget.node(theory, right)?;
115    budget.charge(2)?;
116    Ok(matches!(right, NodeView::False))
117}

119/// Read a signed clause leaf through exact canonical-TRUE antecedents.
120/// Each accepted wrapper moves to a strictly earlier node, so the loop is
121/// bounded by the admitted DAG. It neither flattens a hidden conjunction nor
122/// treats a different tautology as canonical truth.
123pub(super) fn clause_literal<'a, Q: Quota + ?Sized>(
124    theory: &'a Theory,
125    mut node: usize,
126    mut view: NodeView<'a>,
127    budget: &mut Budget<'_, '_, Q>,
128) -> Result<Option<Literal>, BudgetError<Q::Error>> {
129    loop {
130        budget.charge(1)?;
131        let NodeView::Implies(left, right) = view else {
132            break;
133        };
134        budget.charge(3)?;
135        if left >= node || right >= node {
136            budget.charge(1)?;
137            return Err(BudgetError::Source(AdmissionError::Edge));
138        }
139        let antecedent = budget.node(theory, left)?;
140        if !canonical_truth(theory, antecedent, budget)? {
141            break;
142        }
143        budget.charge(1)?;
144        node = right;
145        view = budget.node(theory, node)?;
146    }
147    literal(theory, view, budget)
148}

150/// Authenticate a constraint's False consequent from an already-charged view.
151/// The body remains unread; callers retain their root ordinal and source grammar.
152pub(super) fn constraint_body<Q: Quota + ?Sized>(
153    theory: &Theory,
154    view: NodeView<'_>,
155    budget: &mut Budget<'_, '_, Q>,
156) -> Result<Option<usize>, BudgetError<Q::Error>> {
157    budget.charge(1)?;
158    let NodeView::Implies(body, consequent) = view else {
159        budget.charge(1)?;
160        return Ok(None);
161    };
162    let consequent = budget.node(theory, consequent)?;
163    budget.charge(1)?;
164    if !matches!(consequent, NodeView::False) {
165        budget.charge(1)?;
166        return Ok(None);
167    }
168    Ok(Some(body))
169}

171/// Recognize a constraint through its nested conjunction occurrences and
172/// declared signed-clause views. Duplicate atoms still decline the source,
173/// including opposite polarities. The exact constant-true body has no atoms
174/// and forbids its sole empty pattern.
175pub(super) fn clause<Q: Quota + ?Sized>(
176    theory: &Theory,
177    ordinal: usize,
178    storage: &mut Storage,
179    budget: &mut Budget<'_, '_, Q>,
180) -> Result<Option<Clause>, BudgetError<Q::Error>> {
181    budget.charge(2)?; // authenticate root ordinal and copy its node identity
182    let root = *theory
183        .roots()
184        .get(ordinal)
185        .ok_or(BudgetError::Source(AdmissionError::Root))?;
186    let root_view = budget.node(theory, root)?;
187    let Some(body) = constraint_body(theory, root_view, budget)? else {
188        return Ok(None);
189    };
190    let body_view = budget.node(theory, body)?;
191    budget.charge(17)?; // fixed literal scratch and its initialized length
192    let mut literals = [Literal {
193        atom: 0,
194        positive: false,
195    }; MAX_SUPPORT];
196    let mut len = 0;
197    if !canonical_truth(theory, body_view, budget)? {
198        let mut conjunction = Conjunction::new(storage, budget)?;
199        let accepted = conjunction.walk_view(
200            theory,
201            (body, body_view),
202            storage,
203            budget,
204            |node, view, _, budget| {
205                let Some(value) = clause_literal(theory, node, view, budget)? else {
206                    budget.charge(1)?;
207                    return Ok(false);
208                };
209                budget.charge(1)?;
210                if len == MAX_SUPPORT {
211                    budget.charge(1)?;
212                    return Ok(false);
213                }
214                budget.charge(3)?;
215                literals[len] = value;
216                len += 1;
217                budget.charge(1)?;
218                Ok(true)
219            },
220        );
221        let released = conjunction.release(storage);
222        if !accepted? {
223            released?;
224            budget.charge(1)?;
225            return Ok(None);
226        }
227        released?;
228    }
229    for position in 1..len {
230        budget.charge(2)?;
231        let value = literals[position];
232        let mut destination = position;
233        while destination > 0 {
234            budget.charge(1)?;
235            if literals[destination - 1].atom <= value.atom {
236                break;
237            }
238            budget.charge(2)?;
239            literals[destination] = literals[destination - 1];
240            destination -= 1;
241        }
242        budget.charge(2)?;
243        literals[destination] = value;
244    }
245    for pair in literals[..len].windows(2) {
246        budget.charge(1)?;
247        if pair[0].atom == pair[1].atom {
248            budget.charge(1)?;
249            return Ok(None);
250        }
251    }
252    budget.charge(9)?; // independent output coordinates and pattern accumulator
253    let mut atoms = [0; MAX_SUPPORT];
254    let mut forbidden = 0;
255    for (position, value) in literals[..len].iter().enumerate() {
256        budget.charge(1)?;
257        atoms[position] = value.atom;
258        budget.charge(2)?;
259        forbidden |= u8::from(value.positive) << position;
260    }
261    budget.charge(4)?; // length, pattern, root ordinal and result tag
262    Ok(Some(Clause {
263        support: Support { atoms, len },
264        forbidden,
265        root: ordinal,
266    }))
267}

269/// Condition a recognized original clause on independently checked facts.
270/// Raw shape, distinctness and size checks precede conditioning. Thus known
271/// facts cannot admit an otherwise unsupported clause. A known negative leaf
272/// makes the body false; unknown atoms retain both possible truth values.
273///
274/// Callers account for `CONDITIONED_SCRATCH_BYTES` while this record is built.
275pub(super) fn conditioned_clause<Q: Quota + ?Sized>(
276    theory: &Theory,
277    facts: &CheckedRootFacts,
278    ordinal: usize,
279    storage: &mut Storage,
280    budget: &mut Budget<'_, '_, Q>,
281) -> Result<Option<Clause>, BudgetError<Q::Error>> {
282    let Some(original) = clause(theory, ordinal, storage, budget)? else {
283        return Ok(None);
284    };
285    budget.charge(1)?;
286    if !facts.belongs_to(theory) {
287        budget.charge(1)?;
288        return Ok(None);
289    }
290    budget.charge(11)?; // eight coordinates, length, pattern and original root
291    let mut residual = Clause {
292        support: Support {
293            atoms: [0; MAX_SUPPORT],
294            len: 0,
295        },
296        forbidden: 0,
297        root: original.root,
298    };
299    for (position, &atom) in original.support.atoms[..original.support.len]
300        .iter()
301        .enumerate()
302    {
303        budget.charge(3)?;
304        let positive = original.forbidden & (1 << position) != 0;
305        if facts.contains(atom, budget)? {
306            budget.charge(1)?;
307            if !positive {
308                budget.charge(1)?;
309                return Ok(None);
310            }
311        } else {
312            budget.charge(4)?; // coordinate, shift/union and length
313            residual.support.atoms[residual.support.len] = atom;
314            residual.forbidden |= u8::from(positive) << residual.support.len;
315            residual.support.len += 1;
316        }
317    }
318    budget.charge(1)?;
319    Ok(Some(residual))
320}

322/// Reread the claimed source roots and check the equation at every
323/// allowed pattern. The proposal's cached forbidden patterns are never read.
324pub(super) fn check_clause_equation<Q: Quota + ?Sized>(
325    theory: &Theory,
326    facts: &CheckedRootFacts,
327    roots: &[Clause],
328    support: Support,
329    (words, constant): (&[u64], bool),
330    storage: &mut Storage,
331    budget: &mut Budget<'_, '_, Q>,
332) -> Result<bool, BudgetError<Q::Error>> {
333    budget.charge(1)?;
334    if !facts.belongs_to(theory) {
335        budget.charge(1)?;
336        return Ok(false);
337    }
338    let Some(coefficients) = local_coefficients(theory, support, words, budget)? else {
339        return Ok(false);
340    };
341    budget.charge(4)?;
342    let mut forbidden = Words::<u64, _>::new([0; 4]);
343    for proposed in roots {
344        budget.charge(1)?;
345        let Some(checked) = conditioned_clause(theory, facts, proposed.root, storage, budget)?
346        else {
347            budget.charge(1)?;
348            return Ok(false);
349        };
350        budget.charge(1)?;
351        if checked.support.len != support.len {
352            budget.charge(1)?;
353            return Ok(false);
354        }
355        for (actual, expected) in checked.support.atoms.iter().zip(support.atoms) {
356            budget.charge(1)?;
357            if *actual != expected {
358                budget.charge(1)?;
359                return Ok(false);
360            }
361        }
362        budget.charge(4)?;
363        forbidden.insert(usize::from(checked.forbidden));
364    }
365    implied_by_table(
366        forbidden.storage(),
367        support.len,
368        coefficients,
369        constant,
370        budget,
371    )
372}

374/// Check dense coefficients against an authenticated bounded coordinate map.
375/// Every padded word is inspected; no coefficient outside that map is accepted.
376pub(super) fn local_coefficients<Q: Quota + ?Sized>(
377    theory: &Theory,
378    support: Support,
379    words: &[u64],
380    budget: &mut Budget<'_, '_, Q>,
381) -> Result<Option<u8>, BudgetError<Q::Error>> {
382    budget.charge(1)?;
383    if support.len > MAX_SUPPORT {
384        budget.charge(1)?;
385        return Ok(None);
386    }
387    budget.charge(1)?;
388    if words.len() != theory.atom_count().div_ceil(128) * 2 {
389        budget.charge(1)?;
390        return Ok(None);
391    }
392    budget.charge(1)?;
393    let mut coefficients = 0_u8;
394    for (word_id, &word) in words.iter().enumerate() {
395        budget.charge(1)?;
396        let mut permitted = 0_u64;
397        for (local, &atom) in support.atoms[..support.len].iter().enumerate() {
398            budget.charge(2)?; // coordinate word and membership comparison
399            if atom / 64 == word_id {
400                budget.charge(3)?;
401                let bit = 1_u64 << (atom % 64);
402                permitted |= bit;
403                budget.charge(2)?;
404                if word & bit != 0 {
405                    budget.charge(2)?;
406                    coefficients |= 1 << local;
407                }
408            }
409        }
410        budget.charge(3)?;
411        if word & !permitted != 0 {
412            budget.charge(1)?;
413            return Ok(None);
414        }
415    }
416    budget.charge(1)?;
417    Ok(Some(coefficients))
418}

420/// The independent finite-pattern check shared by both parity sources.
421pub(super) fn implied_by_table<Q: Quota + ?Sized>(
422    forbidden: &[u64; 4],
423    support_len: usize,
424    coefficients: u8,
425    constant: bool,
426    budget: &mut Budget<'_, '_, Q>,
427) -> Result<bool, BudgetError<Q::Error>> {
428    budget.charge(1)?;
429    let extent = 1_u16 << support_len;
430    for pattern in 0..extent {
431        budget.charge(5)?;
432        let value = (pattern & u16::from(coefficients)).count_ones() & 1 != 0;
433        if value != constant {
434            budget.charge(5)?;
435            if !Words::<u64, _>::new(forbidden).contains(usize::from(pattern)) {
436                budget.charge(1)?;
437                return Ok(false);
438            }
439        }
440    }
441    budget.charge(1)?;
442    Ok(true)
