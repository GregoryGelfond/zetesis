1//! Bounded source patterns for checked candidate equations.
2//!
3//! A clause names an original root ordinal and its canonical atom coordinates.
4//! It is a proposal, not an entailment certificate. The checker rereads those
5//! ordinals from the exact admitted owner before trusting an equation. No
6//! formula, atom coordinate or reduct is rewritten.

21/// The finite pattern table has at most 256 bits, in four words. This limit
22/// bounds recognition only; a larger formula remains available to narrowing.
23pub(super) const MAX_SUPPORT: usize = 8;

25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub(super) struct Literal {

28    pub positive: bool,
29}

31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub(super) struct Support {
33    pub atoms: [usize; MAX_SUPPORT],
34    pub len: usize,
35}

38pub(super) struct Clause {

43}

57/// Read a signed atom without recursively interpreting another formula shape.
58/// A previously read view is consumed directly, avoiding another source read.
59pub(super) fn literal<Q: Quota + ?Sized>(
60    theory: &Theory,
61    view: NodeView<'_>,
62    budget: &mut Budget<'_, '_, Q>,
63) -> Result<Option<Literal>, BudgetError<Q::Error>> {
64    budget.charge(1)?; // inspect the source kind
65    let (atom, positive) = match view {
66        NodeView::Atom(atom) => (atom, true),
67        NodeView::Implies(left, right) => {
68            let right = budget.node(theory, right)?;
69            budget.charge(1)?;
70            if !matches!(right, NodeView::False) {
71                budget.charge(1)?;
72                return Ok(None);
73            }
74            let left = budget.node(theory, left)?;
75            budget.charge(1)?;
76            let NodeView::Atom(atom) = left else {
77                budget.charge(1)?;
78                return Ok(None);
79            };
80            (atom, false)
81        }
82        _ => {
83            budget.charge(1)?;
84            return Ok(None);
85        }
86    };
87    budget.charge(1)?;
88    if atom >= theory.atom_count() {
89        budget.charge(1)?;
90        return Err(BudgetError::Source(AdmissionError::Atom));
91    }
92    budget.charge(3)?; // atom, polarity and result tag
93    Ok(Some(Literal { atom, positive }))
94}

96/// Authenticate only the canonical truth formula, falsum implying falsum.
97/// Other tautologies are deliberately outside this source view.
98pub(super) fn canonical_truth<Q: Quota + ?Sized>(
99    theory: &Theory,
100    view: NodeView<'_>,
101    budget: &mut Budget<'_, '_, Q>,
102) -> Result<bool, BudgetError<Q::Error>> {
103    budget.charge(1)?;
104    let NodeView::Implies(left, right) = view else {
105        budget.charge(1)?;
106        return Ok(false);
107    };
108    let left = budget.node(theory, left)?;
109    budget.charge(1)?;
110    if !matches!(left, NodeView::False) {
111        budget.charge(1)?;
112        return Ok(false);
113    }
114    let right = budget.node(theory, right)?;
115    budget.charge(2)?;
116    Ok(matches!(right, NodeView::False))
117}

119/// Read a signed clause leaf through exact canonical-TRUE antecedents.
120/// Each accepted wrapper moves to a strictly earlier node, so the loop is
121/// bounded by the admitted DAG. It neither flattens a hidden conjunction nor
122/// treats a different tautology as canonical truth.
123pub(super) fn clause_literal<'a, Q: Quota + ?Sized>(
124    theory: &'a Theory,
125    mut node: usize,
126    mut view: NodeView<'a>,
127    budget: &mut Budget<'_, '_, Q>,
128) -> Result<Option<Literal>, BudgetError<Q::Error>> {
129    loop {
130        budget.charge(1)?;
131        let NodeView::Implies(left, right) = view else {
132            break;
133        };
134        budget.charge(3)?;
135        if left >= node || right >= node {
136            budget.charge(1)?;
137            return Err(BudgetError::Source(AdmissionError::Edge));
138        }
139        let antecedent = budget.node(theory, left)?;
140        if !canonical_truth(theory, antecedent, budget)? {
141            break;
142        }
143        budget.charge(1)?;
144        node = right;
145        view = budget.node(theory, node)?;
146    }
147    literal(theory, view, budget)
148}

150/// Authenticate a constraint's False consequent from an already-charged view.
151/// The body remains unread; callers retain their root ordinal and source grammar.
152pub(super) fn constraint_body<Q: Quota + ?Sized>(
153    theory: &Theory,
154    view: NodeView<'_>,
155    budget: &mut Budget<'_, '_, Q>,
156) -> Result<Option<usize>, BudgetError<Q::Error>> {
157    budget.charge(1)?;
158    let NodeView::Implies(body, consequent) = view else {
159        budget.charge(1)?;
160        return Ok(None);
161    };
162    let consequent = budget.node(theory, consequent)?;
163    budget.charge(1)?;
164    if !matches!(consequent, NodeView::False) {
165        budget.charge(1)?;
166        return Ok(None);
167    }
168    Ok(Some(body))
169}

171/// Recognize a constraint through its nested conjunction occurrences and
172/// declared signed-clause views. Duplicate atoms still decline the source,
173/// including opposite polarities. The exact constant-true body has no atoms
174/// and forbids its sole empty pattern.
175pub(super) fn clause<Q: Quota + ?Sized>(
176    theory: &Theory,
177    ordinal: usize,
178    storage: &mut Storage,
179    budget: &mut Budget<'_, '_, Q>,
180) -> Result<Option<Clause>, BudgetError<Q::Error>> {
181    read_clause::<MAX_SUPPORT, false, Q>(theory, ordinal, storage, budget)
182}

184/// Read at most two positive occurrences through the same raw grammar.
185/// A negative leaf or a third occurrence cannot pass the final pair filter.
186/// No source tail is read after that determination. The caller still rereads
187/// every successful proposal with the general signed-clause checker.
188pub(super) fn positive_pair_clause<Q: Quota + ?Sized>(
189    theory: &Theory,
190    ordinal: usize,
191    storage: &mut Storage,
192    budget: &mut Budget<'_, '_, Q>,
193) -> Result<Option<Clause>, BudgetError<Q::Error>> {
194    // Exact length/pattern authentication remains with the pair consumer.
195    read_clause::<2, true, Q>(theory, ordinal, storage, budget)
196}

198fn read_clause<const MAXIMUM: usize, const POSITIVE_ONLY: bool, Q: Quota + ?Sized>(
199    theory: &Theory,
200    ordinal: usize,
201    storage: &mut Storage,
202    budget: &mut Budget<'_, '_, Q>,
203) -> Result<Option<Clause>, BudgetError<Q::Error>> {
204    budget.charge(2)?; // authenticate root ordinal and copy its node identity
205    let root = *theory
206        .roots()
207        .get(ordinal)
208        .ok_or(BudgetError::Source(AdmissionError::Root))?;
209    let root_view = budget.node(theory, root)?;
210    let Some(body) = constraint_body(theory, root_view, budget)? else {
211        return Ok(None);
212    };
213    let body_view = budget.node(theory, body)?;
214    budget.charge(17)?; // fixed literal scratch and its initialized length
215    let mut literals = [Literal {
216        atom: 0,
217        positive: false,
218    }; MAX_SUPPORT];
219    let mut len = 0;
220    if !canonical_truth(theory, body_view, budget)? {
221        let mut conjunction = Conjunction::new(storage, budget)?;
222        let accepted = conjunction.walk_view(
223            theory,
224            (body, body_view),
225            storage,
226            budget,
227            |node, view, _, budget| {
228                let Some(value) = clause_literal(theory, node, view, budget)? else {
229                    budget.charge(1)?;
230                    return Ok(false);
231                };
232                if POSITIVE_ONLY {
233                    budget.charge(1)?; // inspect the authenticated polarity
234                    if !value.positive {
235                        budget.charge(1)?;
236                        return Ok(false);
237                    }
238                }
239                budget.charge(1)?;
240                if len == MAXIMUM {
241                    budget.charge(1)?;
242                    return Ok(false);
243                }
244                budget.charge(3)?;
245                literals[len] = value;
246                len += 1;
247                budget.charge(1)?;
248                Ok(true)
249            },
250        );
251        let released = conjunction.release(storage);
252        if !accepted? {
253            released?;
254            budget.charge(1)?;
255            return Ok(None);
256        }
257        released?;
258    }
259    for position in 1..len {
260        budget.charge(2)?;
261        let value = literals[position];
262        let mut destination = position;
263        while destination > 0 {
264            budget.charge(1)?;
265            if literals[destination - 1].atom <= value.atom {
266                break;
267            }
268            budget.charge(2)?;
269            literals[destination] = literals[destination - 1];
270            destination -= 1;
271        }
272        budget.charge(2)?;
273        literals[destination] = value;
274    }
275    for pair in literals[..len].windows(2) {
276        budget.charge(1)?;
277        if pair[0].atom == pair[1].atom {
278            budget.charge(1)?;
279            return Ok(None);
280        }
281    }
282    budget.charge(9)?; // independent output coordinates and pattern accumulator
283    let mut atoms = [0; MAX_SUPPORT];
284    let mut forbidden = 0;
285    for (position, value) in literals[..len].iter().enumerate() {
286        budget.charge(1)?;
287        atoms[position] = value.atom;
288        budget.charge(2)?;
289        forbidden |= u8::from(value.positive) << position;
290    }
291    budget.charge(4)?; // length, pattern, root ordinal and result tag
292    Ok(Some(Clause {
293        support: Support { atoms, len },
294        forbidden,
295        root: ordinal,
296    }))
297}

299/// Condition a recognized original clause on independently checked facts.
300/// Raw shape, distinctness and size checks precede conditioning. Thus known
301/// facts cannot admit an otherwise unsupported clause. A known negative leaf
302/// makes the body false; unknown atoms retain both possible truth values.
303///
304/// Callers account for `CONDITIONED_SCRATCH_BYTES` while this record is built.
305pub(super) fn conditioned_clause<Q: Quota + ?Sized>(
306    theory: &Theory,
307    facts: &CheckedRootFacts,
308    ordinal: usize,
309    storage: &mut Storage,
310    budget: &mut Budget<'_, '_, Q>,
311) -> Result<Option<Clause>, BudgetError<Q::Error>> {
312    let Some(original) = clause(theory, ordinal, storage, budget)? else {
313        return Ok(None);
314    };
315    budget.charge(1)?;
316    if !facts.belongs_to(theory) {
317        budget.charge(1)?;
318        return Ok(None);
319    }
320    budget.charge(11)?; // eight coordinates, length, pattern and original root
321    let mut residual = Clause {
322        support: Support {
323            atoms: [0; MAX_SUPPORT],
324            len: 0,
325        },
326        forbidden: 0,
327        root: original.root,
328    };
329    for (position, &atom) in original.support.atoms[..original.support.len]
330        .iter()
331        .enumerate()
332    {
333        budget.charge(3)?;
334        let positive = original.forbidden & (1 << position) != 0;
335        if facts.contains(atom, budget)? {
336            budget.charge(1)?;
337            if !positive {
338                budget.charge(1)?;
339                return Ok(None);
340            }
341        } else {
342            budget.charge(4)?; // coordinate, shift/union and length
343            residual.support.atoms[residual.support.len] = atom;
344            residual.forbidden |= u8::from(positive) << residual.support.len;
345            residual.support.len += 1;
346        }
347    }
348    budget.charge(1)?;
349    Ok(Some(residual))
350}

352/// Reread the claimed source roots and check the equation at every
353/// allowed pattern. The proposal's cached forbidden patterns are never read.
354pub(super) fn check_clause_equation<Q: Quota + ?Sized>(
355    theory: &Theory,
356    facts: &CheckedRootFacts,
357    roots: &[Clause],
358    support: Support,
359    (words, constant): (&[u64], bool),
360    storage: &mut Storage,
361    budget: &mut Budget<'_, '_, Q>,
362) -> Result<bool, BudgetError<Q::Error>> {
363    budget.charge(1)?;
364    if !facts.belongs_to(theory) {
365        budget.charge(1)?;
366        return Ok(false);
367    }
368    let Some(coefficients) = local_coefficients(theory, support, words, budget)? else {
369        return Ok(false);
370    };
371    reread_clauses_imply(
372        theory,
373        facts,
374        roots,
375        support,
376        (coefficients, constant),
377        storage,
378        budget,
379    )
380}

382/// As [`check_clause_equation`], for the equation that gives `atom` the
383/// value `value`, which is kept as a unit and has no row. The unit's atom is
384/// read against `support`, as a row's words are: an atom the support lacks
385/// declines it. The facts' owner is authenticated where each clause is read
386/// again, and without a clause no unit follows.
387pub(super) fn check_clause_unit<Q: Quota + ?Sized>(
388    theory: &Theory,
389    facts: &CheckedRootFacts,
390    roots: &[Clause],
391    support: Support,
392    (atom, value): (usize, bool),
393    storage: &mut Storage,
394    budget: &mut Budget<'_, '_, Q>,
395) -> Result<bool, BudgetError<Q::Error>> {
396    let Some(coefficients) = unit_coefficients(support, atom, budget)? else {
397        return Ok(false);
398    };
399    reread_clauses_imply(
400        theory,
401        facts,
402        roots,
403        support,
404        (coefficients, value),
405        storage,
406        budget,
407    )
408}

410/// The coefficients over `support` of an equation whose one atom is `atom`:
411/// that atom's coordinate alone, and none when the support lacks the atom.
412/// A support's atoms are distinct, so the coordinate is the only one.
413pub(super) fn unit_coefficients<Q: Quota + ?Sized>(
414    support: Support,
415    atom: usize,
416    budget: &mut Budget<'_, '_, Q>,
417) -> Result<Option<u8>, BudgetError<Q::Error>> {
418    budget.charge(1)?;
419    if support.len > MAX_SUPPORT {
420        budget.charge(1)?;
421        return Ok(None);
422    }
423    for (coordinate, &known) in support.atoms[..support.len].iter().enumerate() {
424        budget.charge(1)?;
425        if known == atom {
426            budget.charge(2)?; // the shifted coefficient and the result tag
427            return Ok(Some(1 << coordinate));
428        }
429    }
430    budget.charge(1)?;
431    Ok(None)
432}

434/// Whether the clauses `roots`, read again and conditioned on the checked
435/// facts, each have the support `support` and together forbid every pattern
436/// at which the equation fails.
437fn reread_clauses_imply<Q: Quota + ?Sized>(
438    theory: &Theory,
439    facts: &CheckedRootFacts,
440    roots: &[Clause],
441    support: Support,
442    (coefficients, constant): (u8, bool),
443    storage: &mut Storage,
444    budget: &mut Budget<'_, '_, Q>,
445) -> Result<bool, BudgetError<Q::Error>> {
446    budget.charge(4)?;
447    let mut forbidden = Words::<u64, _>::new([0; 4]);
448    for proposed in roots {
449        budget.charge(1)?;
450        let Some(checked) = conditioned_clause(theory, facts, proposed.root, storage, budget)?
451        else {
452            budget.charge(1)?;
453            return Ok(false);
454        };
455        budget.charge(1)?;
456        if checked.support.len != support.len {
457            budget.charge(1)?;
458            return Ok(false);
459        }
460        for (actual, expected) in checked.support.atoms.iter().zip(support.atoms) {
461            budget.charge(1)?;
462            if *actual != expected {
463                budget.charge(1)?;
464                return Ok(false);
465            }
466        }
467        budget.charge(4)?;
468        forbidden.insert(usize::from(checked.forbidden));
469    }
470    implied_by_table(
471        forbidden.storage(),
472        support.len,
473        coefficients,
474        constant,
475        budget,
476    )
477}

479/// Check dense coefficients against an authenticated bounded coordinate map.
480/// Every padded word is inspected; no coefficient outside that map is accepted.
481pub(super) fn local_coefficients<Q: Quota + ?Sized>(
482    theory: &Theory,
483    support: Support,
484    words: &[u64],
485    budget: &mut Budget<'_, '_, Q>,
486) -> Result<Option<u8>, BudgetError<Q::Error>> {
487    budget.charge(1)?;
488    if support.len > MAX_SUPPORT {
489        budget.charge(1)?;
490        return Ok(None);
491    }
492    budget.charge(1)?;
493    if words.len() != theory.atom_count().div_ceil(128) * 2 {
494        budget.charge(1)?;
495        return Ok(None);
496    }
497    budget.charge(1)?;
498    let mut coefficients = 0_u8;
499    for (word_id, &word) in words.iter().enumerate() {
500        budget.charge(1)?;
501        let mut permitted = 0_u64;
502        for (local, &atom) in support.atoms[..support.len].iter().enumerate() {
503            budget.charge(2)?; // coordinate word and membership comparison
504            if atom / 64 == word_id {
505                budget.charge(3)?;
506                let bit = 1_u64 << (atom % 64);
507                permitted |= bit;
508                budget.charge(2)?;
509                if word & bit != 0 {
510                    budget.charge(2)?;
511                    coefficients |= 1 << local;
512                }
513            }
514        }
515        budget.charge(3)?;
516        if word & !permitted != 0 {
517            budget.charge(1)?;
518            return Ok(None);
519        }
520    }
521    budget.charge(1)?;
522    Ok(Some(coefficients))
523}

525/// The independent finite-pattern check shared by both parity sources.
526pub(super) fn implied_by_table<Q: Quota + ?Sized>(
527    forbidden: &[u64; 4],
528    support_len: usize,
529    coefficients: u8,
530    constant: bool,
531    budget: &mut Budget<'_, '_, Q>,
532) -> Result<bool, BudgetError<Q::Error>> {
533    budget.charge(1)?;
534    let extent = 1_u16 << support_len;
535    for pattern in 0..extent {
536        budget.charge(5)?;
537        let value = (pattern & u16::from(coefficients)).count_ones() & 1 != 0;
538        if value != constant {
539            budget.charge(5)?;
540            if !Words::<u64, _>::new(forbidden).contains(usize::from(pattern)) {
541                budget.charge(1)?;
542                return Ok(false);
543            }
544        }
545    }
546    budget.charge(1)?;
547    Ok(true)
