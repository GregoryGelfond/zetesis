1//! Named simultaneous capacities during optional row preparation.
2//!
3//! There is one allowance, owned by `Budget`. This ledger only tracks which
4//! buffers are live. A move reserves both buffers, charges the moved logical
5//! fields, and releases the old capacity. Allocator internals are excluded.
6
7use zetesis_algebra::Quota;
8
9use std::mem::size_of;
10
11use zetesis_algebra::Stop;
12
13use super::budget::{Budget, BudgetError};
14
15#[derive(Default)]
16pub(super) struct Storage {
17    bytes: u128,
18}
19
20impl Storage {
21    /// Account for caller-owned live data alongside this scope's buffers.
22    /// The caller admits this baseline before constructing the ledger; buffers
23    /// reserved here are released here, while this borrowed baseline remains.
24    pub(super) fn with_live_bytes(bytes: u128) -> Self {
25        Self { bytes }
26    }
27
28    pub(super) fn bytes(&self) -> u128 {
29        self.bytes
30    }
31
32    /// Admit named non-buffer metadata or fixed scratch while it is live.
33    /// Contained vector headers already tracked by `reserve` are excluded.
34    pub(super) fn admit_record<Q: Quota + ?Sized>(
35        &mut self,
36        bytes: u128,
37        budget: &mut Budget<'_, '_, Q>,
38    ) -> Result<(), BudgetError<Q::Error>> {
39        let next = self
40            .bytes
41            .checked_add(bytes)
42            .ok_or(BudgetError::Stopped(Stop::Allocation))?;
43        budget.admit_bytes(next)?;
44        self.bytes = next;
45        Ok(())
46    }
47
48    /// End the lifetime of fixed scratch; it is absent from later peaks.
49    pub(super) fn release_record(&mut self, bytes: u128) -> Result<(), Stop> {
50        self.bytes = self.bytes.checked_sub(bytes).ok_or(Stop::InvalidProgram)?;
51        Ok(())
52    }
53
54    /// Borrow fixed workspace for one operation, including its failure prefix.
55    pub(super) fn with_record<T, Q: Quota + ?Sized>(
56        &mut self,
57        bytes: u128,
58        budget: &mut Budget<'_, '_, Q>,
59        operation: impl FnOnce(&mut Self, &mut Budget<'_, '_, Q>) -> Result<T, BudgetError<Q::Error>>,
60    ) -> Result<T, BudgetError<Q::Error>> {
61        self.admit_record(bytes, budget)?;
62        let result = operation(self, budget);
63        let released = self.release_record(bytes);
64        result.and_then(|value| released.map(|()| value).map_err(BudgetError::Stopped))
65    }

67    /// Transfer a vector header into an already-accounted outer allocation.
68    /// Its payload stays live; the outer element already includes the header.
69    pub(super) fn embed<T>(&mut self) -> Result<(), Stop> {
70        self.bytes = self
71            .bytes
72            .checked_sub(size_of::<Vec<T>>() as u128)
73            .ok_or(Stop::InvalidProgram)?;
74        Ok(())
75    }
76
77    /// Release a payload whose header belongs to its enclosing allocation.
78    /// Moving that header through a call is ABI transport, not another owner.
79    pub(super) fn release_embedded<T>(&mut self, values: Vec<T>) -> Result<(), Stop> {
80        self.bytes = self
81            .bytes
82            .checked_sub(values.capacity() as u128 * size_of::<T>() as u128)
83            .ok_or(Stop::InvalidProgram)?;
84        drop(values);
85        Ok(())
86    }
87
88    pub(super) fn reserve<T, Q: Quota + ?Sized>(
89        &mut self,
90        count: usize,
91        budget: &mut Budget<'_, '_, Q>,
92    ) -> Result<Vec<T>, BudgetError<Q::Error>> {
93        budget.charge(3)?; // initialize the vector's three-word header
94        self.reserve_bytes(count, budget)
95    }
96
97    /// Reserve and account for a vector when its producer meters the logical
98    /// work itself. Both the request and the actual capacity must fit.
99    pub(super) fn reserve_bytes<T, Q: Quota + ?Sized>(
100        &mut self,
101        count: usize,
102        budget: &mut Budget<'_, '_, Q>,
103    ) -> Result<Vec<T>, BudgetError<Q::Error>> {
104        let mut values = Vec::new();
105        let header = size_of::<Vec<T>>() as u128;
106        let requested = count as u128 * size_of::<T>() as u128;
107        let request = self
108            .bytes
109            .checked_add(header)
110            .and_then(|bytes| bytes.checked_add(requested))
111            .ok_or(BudgetError::Stopped(Stop::Allocation))?;
112        budget.admit_bytes(request)?;
113        values
114            .try_reserve_exact(count)
115            .map_err(|_| BudgetError::Stopped(Stop::Allocation))?;
116        let retained = header + values.capacity() as u128 * size_of::<T>() as u128;
117        let next = self
118            .bytes
119            .checked_add(retained)
120            .ok_or(BudgetError::Stopped(Stop::Allocation))?;
121        budget.admit_bytes(next)?;
122        self.bytes = next;
123        Ok(values)
124    }

126    /// Append an already-metered value, admitting the simultaneous old and
127    /// new buffers before a move.
128    pub(super) fn push_bytes<T, Q: Quota + ?Sized>(
129        &mut self,
130        values: &mut Vec<T>,
131        value: T,
132        budget: &mut Budget<'_, '_, Q>,
133    ) -> Result<(), BudgetError<Q::Error>> {
134        if values.len() == values.capacity() {
135            let capacity = values.capacity().saturating_mul(2).max(1);
136            let mut next = self.reserve_bytes(capacity, budget)?;
137            next.append(values);
138            self.release(std::mem::replace(values, next))?;
139        }
140        values.push(value);
141        Ok(())
142    }
143
144    /// Grow by doubling the current capacity, with one cell as the first step.
145    /// Actual capacities govern bytes; every moved logical field is charged.
146    /// The finite geometric schedule moves fewer than twice the final capacity
147    /// cells when the allocator returns the requested exact capacities.
148    pub(super) fn push<T, Q: Quota + ?Sized>(
149        &mut self,
150        values: &mut Vec<T>,
151        value: T,
152        words: u64,
153        budget: &mut Budget<'_, '_, Q>,
154    ) -> Result<(), BudgetError<Q::Error>> {
155        budget.charge(1)?;
156        if values.len() == values.capacity() {
157            let capacity = values.capacity().saturating_mul(2).max(1);
158            let mut next = self.reserve(capacity, budget)?;
159            let transfer = u64::try_from(values.len())
160                .ok()
161                .and_then(|count| count.checked_mul(words + 1))
162                .and_then(|work| work.checked_add(1)); // clear source length
163            let admission = transfer
164                .ok_or(BudgetError::Stopped(Stop::WorkLimit))
165                .and_then(|work| budget.charge(work));
166            if let Err(error) = admission {
167                self.release(next)?;
168                return Err(error);
169            }
170            // Capacity is already available and the whole bounded transfer was
171            // admitted before append changes the source's header or moves a cell.
172            next.append(values);
173            if let Err(error) = budget.charge(6) {
174                self.release(next)?;
175                return Err(error);
176            }
177            std::mem::swap(values, &mut next);
178            self.release(next)?;
179        }
180        budget.charge(words + 1)?;
181        values.push(value);
182        Ok(())
183    }
184
185    pub(super) fn filled<T: Copy, Q: Quota + ?Sized>(
186        &mut self,
187        count: usize,
188        value: T,
189        words: u64,
190        budget: &mut Budget<'_, '_, Q>,
191    ) -> Result<Vec<T>, BudgetError<Q::Error>> {
192        let mut values = self.reserve(count, budget)?;
193        for _ in 0..count {
194            budget.charge(words + 1)?;
195            values.push(value);
196        }
197        Ok(values)
198    }
199
200    pub(super) fn release<T>(&mut self, values: Vec<T>) -> Result<(), Stop> {
201        let capacity =
202            size_of::<Vec<T>>() as u128 + values.capacity() as u128 * size_of::<T>() as u128;
203        self.bytes = self
204            .bytes
205            .checked_sub(capacity)
206            .ok_or(Stop::InvalidProgram)?;
207        drop(values);
208        Ok(())
209    }
