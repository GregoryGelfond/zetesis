1//! Uniform positive weights reduce a threshold to a bound on its atom count.
2//!
3//! This helper does not authenticate conditions, distinctness or ownership.
4//! Its caller supplies those source checks and the operation-admission callback.
5
6/// The threshold itself gives a lower bound; its negation gives an upper bound.
7#[derive(Clone, Copy, Debug, Eq, PartialEq)]
8pub(super) enum BoundSide {
9    Lower,
10    Upper,
11}
12
13/// A source shape left to the ordinary narrower, rather than a new refutation.
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub(super) enum Decline {
16    NonPositive,
17    ZeroWeight,
18    BeyondCardinality,
19}
20
21/// The normalized bound or the reason this source does not supply one.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub(super) enum UniformBound {
24    Declined(Decline),
25    Valid { value: u64 },
26}
27
28/// Normalize `weight * count >= bound`, or its negation, without wide division.
29///
30/// Positive bounds use `ceil(bound / weight) = (bound - 1) / weight + 1`.
31/// A ceiling outside `1..=cardinality` is trivial and declines; it never becomes
32/// a refutation here. Upper bounds subtract one from that positive ceiling.
33/// Callback failures propagate unchanged, before any operation in their chunk.
34///
35/// Work counts logical 64-bit operations, not instructions or meter overhead.
36/// The input-only bound is
37/// `4 + 1 + 3 + 4 + 128 * (13 + 4) + 3 + 2 + 2 = 2195`:
38/// sign/zero, weight, predecessor, division initialization and bits, successor,
39/// cardinality checks, then value and tag. Refusals spend only their prefix,
40/// including one result tag. Each bit spends 13, plus 4 only if it subtracts.
41pub(super) fn uniform_bound<E>(
42    bound: i128,
43    weight: u64,
44    cardinality: u64,
45    side: BoundSide,
46    before: &mut impl FnMut(u64) -> Result<(), E>,
47) -> Result<UniformBound, E> {
48    // This is a bit-preserving register representation, independent of host
49    // endianness, not an absolute value or a narrowing numeric conversion.
50    let [
51        lo0,
52        lo1,
53        lo2,
54        lo3,
55        lo4,
56        lo5,
57        lo6,
58        lo7,
59        hi0,
60        hi1,
61        hi2,
62        hi3,
63        hi4,
64        hi5,
65        hi6,
66        hi7,
67    ] = bound.to_le_bytes();
68    let mut numerator = [
69        u64::from_le_bytes([lo0, lo1, lo2, lo3, lo4, lo5, lo6, lo7]),
70        u64::from_le_bytes([hi0, hi1, hi2, hi3, hi4, hi5, hi6, hi7]),
71    ];
72    before(2)?;
73    if numerator[1] >> 63 != 0 {
74        before(1)?;
75        return Ok(UniformBound::Declined(Decline::NonPositive));
76    }
77    before(2)?;
78    if numerator[0] | numerator[1] == 0 {
79        before(1)?;
80        return Ok(UniformBound::Declined(Decline::NonPositive));
81    }
82    before(1)?;
83    if weight == 0 {
84        before(1)?;
85        return Ok(UniformBound::Declined(Decline::ZeroWeight));
86    }
87
88    // The positive two-word value has a predecessor even when its low limb is
89    // zero. The low subtraction wraps; its borrow is taken from the high limb.
90    before(3)?;
91    let borrow = u64::from(numerator[0] == 0);
92    numerator[0] = numerator[0].wrapping_sub(1);
93    numerator[1] -= borrow;
94    let mut quotient = divide(numerator, weight, before)?;
95
96    // The predecessor is below 2^127, so this two-word increment cannot overflow.
97    before(3)?;
98    quotient[0] = quotient[0].wrapping_add(1);
99    let carry = u64::from(quotient[0] == 0);
100    quotient[1] += carry;
101    before(1)?;
102    if quotient[1] != 0 {
103        before(1)?;
104        return Ok(UniformBound::Declined(Decline::BeyondCardinality));
105    }
106    before(1)?;
107    if quotient[0] > cardinality {
108        before(1)?;
109        return Ok(UniformBound::Declined(Decline::BeyondCardinality));
110    }
111    before(2)?;
112    let value = match side {
113        BoundSide::Lower => quotient[0],
114        BoundSide::Upper => quotient[0] - 1,
115    };
116    Ok(UniformBound::Valid { value })
117}
118
119/// Restoring division of a two-word numerator by a nonzero word.
120///
121/// Before each bit, the remainder is below the divisor. Shifting in a bit makes
122/// it smaller than twice the divisor, so one subtraction restores the invariant.
123/// The quotient is the processed prefix divided by the divisor. The remainder
124/// needs 65 bits; neither a wide division nor a hidden multiplication is used.
125fn divide<E>(
126    numerator: [u64; 2],
127    divisor: u64,
128    before: &mut impl FnMut(u64) -> Result<(), E>,
129) -> Result<[u64; 2], E> {
130    before(4)?;
131    let mut quotient = [0_u64; 2];
132    let mut remainder = [0_u64; 2];
133    for word in numerator.iter().rev() {
134        for offset in (0..64).rev() {
135            before(13)?;
136            let bit = (word >> offset) & 1;
137            let carry = remainder[0] >> 63;
138            remainder[0] = (remainder[0] << 1) | bit;
139            remainder[1] = carry;
140            quotient[1] = (quotient[1] << 1) | (quotient[0] >> 63);
141            quotient[0] <<= 1;
142            let wide = remainder[1] != 0;
143            let at_least_divisor = remainder[0] >= divisor;
144            let take = wide || at_least_divisor;
145            if take {
146                before(4)?;
147                let borrow = u64::from(remainder[0] < divisor);
148                remainder[0] = remainder[0].wrapping_sub(divisor);
149                remainder[1] -= borrow;
150                quotient[0] |= 1;
151            }
152        }
153    }
154    Ok(quotient)
155}
