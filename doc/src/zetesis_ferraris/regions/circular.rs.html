1//! Producers that cannot support their own head.
2//!
3//! A body requires the atoms that are its conjuncts: an atom body requires its
4//! atom, and a conjunction requires each operand that is an atom. An atom is
5//! derived only through `a` when it has producers and every one of them has a
6//! body requiring `a`. A producer whose one head is `a` is circular when its
7//! body requires `a` itself or an atom derived only through `a`: its body can
8//! hold only where its head already does.
9//!
10//! No answer set rests on a circular producer. Every atom of an answer set has
11//! an enabled producer, with that atom as its only true head, whose body
12//! requires neither the atom nor any atom derived only through it
13//! (`ExternalSupport.answer_set_noncircular_support`). The support cut and the
14//! positive-dependency test of an original theory therefore leave circular
15//! producers out. Every root stays in the theory and propagates as the
16//! implication it is; nothing here rewrites a rule or decides membership.
17//!
18//! Only producers with one head are marked. A producer with several heads may
19//! be circular for one of them and a support of another, so it is never
20//! marked: the restriction stays unused there, which is sound. Operands that
21//! are themselves conjunctions are not opened either; a body then requires
22//! fewer atoms than the law would grant it, and fewer producers are marked.
23//!
24//! The pass lists each distinct body's required atoms once and turns that list
25//! around in one transient adjacency. For each head it stamps the bodies
26//! requiring that head; a required atom is then derived only through the head
27//! exactly when every producer of the atom has a stamped body. Each head and
28//! atom pair is decided once, and a decision ends at the first producer whose
29//! body is not stamped. Every node, head, operand, pair, stamp and producer
30//! read is one unit of the caller's work; a body shared by several producers
31//! is read once for each, as the ground rules it stands for would be. Placing
32//! the pairs in their adjacency and zeroing the stamp, memo and mark vectors
33//! are not charged, as the inventory's own adjacencies are not.
34//!
35//! The cost is not linear in the extraction's visits. The extraction visits
36//! a conjunction once, whatever its width; the pass reads its operands once
37//! for every producer of the body, `k * m` reads for `m` atoms under `k`
38//! rules, and each head and required atom pair reads the atom's producers up
39//! to the first unstamped one. The extraction therefore gives the pass an
40//! allowance, a fixed multiple of its own visits within what its ceiling
41//! leaves, and keeps an unmarked inventory when the pass exhausts it
42//! ([`super::Producers::mark`]). No mark is ever needed for soundness.
43

53/// No producer marked, for an inventory of `producers` rules.
54///
55/// # Errors
56/// An allocation failure.
57pub(super) fn none(producers: usize) -> Result<Words<u64>, Stop> {
58    Ok(Words::<u64>::new(zeros(flag_words(producers))?))
59}

61/// One mark per circular producer, in the inventory's producer order, and
62/// whether any producer is marked.
63///
64/// # Errors
65/// The caller's stop when its work is refused, or an allocation failure. No
66/// partial marking is returned.
67pub(super) fn mark<Q: Quota + ?Sized>(
68    theory: &Theory,
69    rules: &[Producer],
70    by_head: &Adjacency,
71    by_body: &Adjacency,
72    work: &mut Meter<'_, Q>,
73) -> Result<(Words<u64>, bool), Failure<Q::Error>> {
74    let mut marks = none(rules.len())?;
75    let mut any = false;
76    let requirements = Requirements::read(theory, by_body, work)?;
77    if requirements.pairs == 0 {
78        return Ok((marks, any));
79    }
80    // The head whose requiring bodies are stamped, as `head + 1`.
81    let mut stamped = zeros(theory.nodes().len())?;
82    // Per atom, the head it was last decided for and whether it is derived
83    // only through that head.
84    let mut decided = zeros(theory.atom_count())?;
85    let mut derived = Words::<u64>::new(zeros(flag_words(theory.atom_count()))?);
86    for head in 0..by_head.len() {
87        work.tick()?;
88        let mut candidates = false;
89        for &row in &by_head[head] {
90            work.tick()?;
91            if rules[row].heads.len() == 1 && rules[row].body.is_some() {
92                candidates = true;
93                break;
94            }
95        }
96        if !candidates {
97            continue;
98        }
99        let stamp = head + 1;
100        for &body in &requirements.bodies[head] {
101            work.tick()?;
102            stamped[body] = stamp;
103        }
104        for &row in &by_head[head] {
105            work.tick()?;
106            let producer = &rules[row];
107            let (Some(body), [_]) = (producer.body, producer.heads.as_slice()) else {
108                continue;
109            };
110            let mut circular = false;
111            for atom in required(theory, body) {
112                work.tick()?;
113                let Some(atom) = atom? else { continue };
114                if atom == head {
115                    circular = true;
116                    break;
117                }
118                if decided[atom] != stamp {
119                    decided[atom] = stamp;
120                    let only = only_through(&by_head[atom], rules, &stamped, stamp, work)?;
121                    if only {
122                        derived.insert(atom);
123                    } else {
124                        derived.remove(atom);
125                    }
126                }
127                if derived.contains(atom) {
128                    circular = true;
129                    break;
130                }
131            }
132            if circular {
133                marks.insert(row);
134                any = true;
135            }
136        }
137    }
138    Ok((marks, any))
139}

141/// Whether an atom has producers and the body of every one of them is stamped
142/// for the head under examination.
143fn only_through<Q: Quota + ?Sized>(
144    producers: &[usize],
145    rules: &[Producer],
146    stamped: &[usize],
147    stamp: usize,
148    work: &mut Meter<'_, Q>,
149) -> Result<bool, Failure<Q::Error>> {
150    for &row in producers {
151        work.tick()?;
152        if rules[row].body.is_none_or(|body| stamped[body] != stamp) {
153            return Ok(false);
154        }
155    }
156    Ok(!producers.is_empty())
157}

159/// Each inspected requirement: the body when it is an atom, or every operand
160/// when it is a conjunction. A non-atomic operand yields `None` so the caller
161/// charges its inspection too. Repeated operands repeat.
162fn required(
163    theory: &Theory,
164    body: usize,
165) -> impl Iterator<Item = Result<Option<usize>, Stop>> + '_ {
166    let nodes = theory.view();
167    let (own, operands) = match nodes.node(body) {
168        Ok(NodeView::Atom(atom)) => (Some(Ok(Some(atom))), &[][..]),
169        Ok(NodeView::And(operands)) => (None, operands),
170        Ok(_) => (None, &[][..]),
171        Err(_) => (Some(Err(Stop::InvalidProgram)), &[][..]),
172    };
173    own.into_iter().chain(
174        operands
175            .iter()
176            .map(move |&operand| match nodes.node(operand) {
177                Ok(NodeView::Atom(atom)) => Ok(Some(atom)),
178                Ok(_) => Ok(None),
179                Err(_) => Err(Stop::InvalidProgram),
180            }),
181    )
182}

184/// Per atom, the distinct producer bodies that require it.
185struct Requirements {
186    bodies: Adjacency,
187    pairs: usize,
188}
189
190impl Requirements {
191    fn read<Q: Quota + ?Sized>(
192        theory: &Theory,
193        by_body: &Adjacency,
194        work: &mut Meter<'_, Q>,
195    ) -> Result<Self, Failure<Q::Error>> {
196        let mut pairs = Vec::new();
197        for body in 0..by_body.len() {
198            work.tick()?;
199            if by_body[body].is_empty() {
200                continue;
201            }
202            for atom in required(theory, body) {
203                work.tick()?;
204                let Some(atom) = atom? else { continue };
205                pairs.try_reserve(1).map_err(|_| Stop::Allocation)?;
206                pairs.push((atom, body));
207            }
208        }
209        let bodies = Adjacency::build(theory.atom_count(), pairs.iter().copied())?;
210        Ok(Self {
211            bodies,
212            pairs: pairs.len(),
213        })
214    }
215}

217fn zeros<T: Clone + Default>(length: usize) -> Result<Vec<T>, Stop> {
218    let mut values = Vec::new();
219    values
220        .try_reserve_exact(length)
221        .map_err(|_| Stop::Allocation)?;
222    values.resize(length, T::default());
223    Ok(values)
224}
