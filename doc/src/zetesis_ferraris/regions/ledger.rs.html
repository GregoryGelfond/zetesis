1//! Learned rows: bounded, exact-original reuse of consequences of the theory.
2//!
3//! Every row says that every answer set of its owner's consequence class that
4//! satisfies its signed atom conditions satisfies each of its signed outputs
5//! (`LearnedRegions.RowOf`). Rows are independent of the region that supplied
6//! them. Deleting a row loses pruning only. No rejected candidate, partial
7//! closure, objective bound or frozen reduct supplies a certificate, and no row
8//! enters a membership certificate.
9//!
10//! A ledger is a tier: one walk's or worker's rows. A [`LearningStore`] holds
11//! the rows one solve's streams share, by class; tiers are seeded from it and
12//! deliver to it at their walks' boundaries, never inside a narrowing. Rows
13//! reach a ledger only from the narrower's own sources, today the founded
14//! collector, or from the store those sources filled: no caller supplies,
15//! edits or reads a row.
16

33/// Up to this many rows the matching step scans every row; above it, matching
34/// reads the occurrence index for the atoms decided since the last step.
35const SCAN_ROWS: usize = u64::BITS as usize;
36
37/// The largest tier [`LearningLimits::for_allowance`] derives.
38const MAX_TIER_ROWS: usize = 4_096;
39
40/// Allowances for learned rows: per tier (one walk's or worker's ledger), per
41/// store, per source, and what and when a tier shares.
42///
43/// The defaults are the limits the founded source has always had: 64 rows,
44/// 256 capture attempts, 131,072 capture reads over a tier's lifetime, 4,096
45/// reads and 65,536 bytes per capture, and rows of at most 16 conditions and
46/// 64 outputs. A tier shares rows of at most 8 conditions, or rows that have
47/// hit twice, and delivers every 4,096 regions its walk visits.
48/// [`LearningLimits::for_allowance`] raises the tier and store capacities
49/// with the memory allowance and never lowers any limit below these.
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub struct LearningLimits {
52    /// Rows one tier retains. A full tier evicts by decayed hits.
53    pub tier_rows: usize,
54    /// Rows the store retains per consequence class.
55    pub store_rows: usize,
56    /// The founded source's reserved share of every tier: eviction never takes
57    /// a founded row while the tier holds this many or fewer of them.
58    pub founded_floor: usize,
59    /// Signed conditions one founded row may require.
60    pub founded_conditions: usize,
61    /// Atoms one founded row may exclude.
62    pub founded_outputs: usize,
63    /// Founded capture attempts over a tier's lifetime.
64    pub founded_attempts: usize,
65    /// Founded discovery, collection and comparison reads over a tier's life.
66    pub founded_work: u64,
67    /// Reads of one founded capture.
68    pub founded_event_work: u64,
69    /// Workspace bytes of one founded capture.
70    pub founded_event_bytes: u64,
71    /// Seeding and delivery reads over a tier's lifetime. Exhausting them
72    /// stops the tier's sharing; its own rows still prune its walk.
73    pub share_work: u64,
74    /// The quality filter: a tier shares a row with at most this many
75    /// conditions, or one that has hit at least `share_hits` times.
76    pub share_conditions: usize,
77    /// See `share_conditions`.
78    pub share_hits: u64,
79    /// The publication trigger: a walk delivers its tier each time it has
80    /// visited this many regions since the tier last delivered, as well as at
81    /// the walk's end. Zero never triggers.
82    pub publish_regions: u64,
83}

85impl LearningLimits {
86    /// The defaults: the founded source's limits as they have always been.
87    pub const DEFAULT: Self = Self {
88        tier_rows: 64,
89        store_rows: 256,
90        founded_floor: 64,
91        founded_conditions: 16,
92        founded_outputs: 64,
93        founded_attempts: 256,
94        founded_work: 131_072,
95        founded_event_work: 4_096,
96        founded_event_bytes: 65_536,
97        share_work: 131_072,
98        share_conditions: 8,
99        share_hits: 2,
100        publish_regions: 4_096,
101    };
102
103    /// The limits for a candidate-memory allowance of `bytes`: one tier row
104    /// for every MiB of it, between the default's 64 rows and 4,096, and a
105    /// store four times a tier. Every other limit keeps its default, so an
106    /// allowance of 64 MiB or less keeps every default.
107    #[must_use]
108    pub const fn for_allowance(bytes: usize) -> Self {
109        let defaults = Self::DEFAULT;
110        let rows = bytes >> 20;
111        let rows = if rows < defaults.tier_rows {
112            defaults.tier_rows
113        } else if rows > MAX_TIER_ROWS {
114            MAX_TIER_ROWS
115        } else {
116            rows
117        };
118        Self {
119            tier_rows: rows,
120            store_rows: rows.saturating_mul(4),
121            ..defaults
122        }
123    }
124}

126impl Default for LearningLimits {
127    fn default() -> Self {
128        Self::DEFAULT
129    }

176/// Identities of stores, so a tier's seeding mark names the store it read.
177static STORES: AtomicU64 = AtomicU64::new(0);

179/// The rows one solve's streams share, by consequence class.
180///
181/// Its public surface is creating an empty store and its counts: rows enter
182/// only by a tier's delivery and leave only by seeding a tier, and no caller
183/// supplies, edits or reads one. The store charges no work of its own; the
184/// tier that seeds or delivers pays for every read.
185#[derive(Debug)]
186pub struct LearningStore {
187    /// Distinct among the stores of a process.
188    id: u64,
189    classes: Vec<Ledger>,
190    deliveries: u64,
191    seeded: u64,
192}
193
194impl Default for LearningStore {
195    fn default() -> Self {
196        Self {
197            id: STORES.fetch_add(1, Ordering::Relaxed),
198            classes: Vec::new(),
199            deliveries: 0,
200            seeded: 0,
201        }
202    }
203}

205impl LearningStore {
206    /// An empty store.
207    #[must_use]
208    pub fn new() -> Self {
209        Self::default()
210    }
211
212    /// Rows held, over every class.
213    #[must_use]
214    pub fn len(&self) -> usize {
215        self.classes.iter().map(Ledger::len).sum()
216    }
217
218    /// True when the store holds no row.
219    #[must_use]
220    pub fn is_empty(&self) -> bool {
221        self.len() == 0
222    }
223
224    /// Rows tiers delivered and the store retained, over its life.
225    #[must_use]
226    pub const fn deliveries(&self) -> u64 {
227        self.deliveries
228    }
229
230    /// Rows seeding copied into tiers, over the store's life.
231    #[must_use]
232    pub const fn seeded(&self) -> u64 {
233        self.seeded
234    }
235
236    fn class(&self, owner: &LearningOwner) -> Option<&Ledger> {
237        self.classes.iter().find(|ledger| {
238            ledger
239                .owner
240                .as_ref()
241                .is_some_and(|o| o.same_class(&owner.theory))
242        })
243    }
244
245    fn class_mut(&mut self, owner: &LearningOwner) -> Result<&mut Ledger, Stop> {
246        let position = self.classes.iter().position(|ledger| {
247            ledger
248                .owner
249                .as_ref()
250                .is_some_and(|o| o.same_class(&owner.theory))
251        });
252        if let Some(position) = position {
253            return Ok(&mut self.classes[position]);
254        }
255        self.classes
256            .try_reserve_exact(1)
257            .map_err(|_| Stop::Allocation)?;
258        // A class holds up to the store's rows, all of them evictable.
259        let mut limits = owner.limits;
260        limits.tier_rows = limits.store_rows;
261        limits.founded_floor = 0;
262        let mut ledger = Ledger::default();
263        ledger.prepare(&LearningOwner {
264            theory: owner.theory.clone(),
265            limits,
266        });
267        self.classes.push(ledger);
268        let last = self.classes.len() - 1;
269        Ok(&mut self.classes[last])
270    }

276    None,
277    /// Every condition holds: each output must hold.
278    ForceOutputs,
279    /// Exactly one condition is open and an output's negation is decided: the

283        value: bool,

287/// Sorted, duplicate-free signed conditions and signed outputs of one row.
288#[derive(Debug, PartialEq, Eq)]
289pub(in crate::regions) struct Literals {
290    pub(in crate::regions) conditions: Vec<(usize, bool)>,
291    pub(in crate::regions) outputs: Vec<(usize, bool)>,
292}
293
294impl Literals {
295    /// Owned buffers; the caller accounts for its containing row/header.
296    pub(in crate::regions) fn retained_bytes(&self) -> u128 {
297        (self.conditions.capacity() + self.outputs.capacity()) as u128
298            * size_of::<(usize, bool)>() as u128
299    }
300
301    /// The founded collector's unsupported atoms as absent outputs under its
302    /// conditions (`LearnedRegions.founded_row`).
303    fn from_founded(explanation: Explanation, peak: &mut u128) -> Result<Self, Stop> {
304        let mut outputs = Vec::new();
305        outputs
306            .try_reserve_exact(explanation.unsupported.len())
307            .map_err(|_| Stop::Allocation)?;
308        *peak = (*peak).max(
309            explanation.retained_bytes()
310                + outputs.capacity() as u128 * size_of::<(usize, bool)>() as u128,
311        );
312        outputs.extend(explanation.unsupported.iter().map(|&atom| (atom, false)));
313        Ok(Self {
314            conditions: explanation.conditions,
315            outputs,
316        })
317    }
318}

342/// A row offered to retention, with what the ledger keeps about it.
343struct Proposed {
344    literals: Literals,
345    source: Source,
346    /// The hit count the row starts with.
347    hits: u64,
348    /// The most outputs a union with an existing row may give it.

352/// What comparing a proposed row with the retained rows established.
353enum Compared {
354    /// The retained row at this position gives the proposed row.
355    Subsumed(usize),
356    /// An allowance ran out before every comparison finished.
357    Declined,
358    /// The rows, one bit per position, the proposed row replaces: identical
359    /// conditions whose outputs it now includes, or rows it dominates.
360    Replaces(Words<u64>),
361}

363/// Where a retention left a proposed row.
364enum Retained {
365    /// Added, or replaced dominated rows, at this position.
366    Added(usize),
367    /// An existing row at this position already gives it.
368    Subsumed(usize),
369    /// Declined: an allowance ran out, or the tier is full of protected rows.
370    Declined,
371}

431impl Ledger {
432    /// Only an owner's carrier subject, seeding and delivery call this.
433    /// Restriction, bound and frozen visits suspend use rather than
434    /// discarding the owner's rows; an owner of a different class clears
435    /// them, and an owner of the same class supplies its limits.
436    pub(in crate::regions) fn prepare(&mut self, owner: &LearningOwner) {
437        let Some(current) = &mut self.owner else {
438            self.adopt(owner);
439            return;
440        };
441        if !current.same_class(&owner.theory) {
442            self.adopt(owner);
443        } else if current.limits != owner.limits {
444            current.limits = owner.limits;
445        }
446    }

448    /// Clear the tier for a new class.
449    fn adopt(&mut self, owner: &LearningOwner) {
450        self.rows.clear();
451        self.postings.clear();
452        self.founded = 0;
453        self.evicted = 0;
454        self.seeded_from = None;
455        self.owner = Some(owner.clone());
456        self.attempts = 0;
457        self.epoch = 0;
458        self.remaining = owner.limits.founded_work;
459        self.sharing = owner.limits.share_work;
460        self.stale = true;
461        self.full = true;
462        self.peak_bytes = self.retained_bytes();
463    }

511    /// Retain a founded row under the capture allowance, as capture does
512    /// after collection; true when the row was added.

542    /// Whether atom decisions must be recorded for the occurrence index.

545    }

547    /// The rows a matching step tests: every row when the tier is small, at
548    /// a narrowing's first step and after the rows changed; otherwise the
549    /// rows that mention an atom decided since the last step. The first
550    /// indexed step after a change rebuilds the index, at one read per
551    /// posting. Positions are ascending and distinct.
552    ///
553    /// Within one narrowing knowledge only grows, and a row's deduction reads
554    /// only its own atoms. A row none of whose atoms was decided since the
555    /// last step therefore has the deduction that step already applied, so
556    /// the index loses no deduction.
557    pub(in crate::regions) fn candidates<Q: Quota + ?Sized>(
558        &mut self,
559        decided: &[usize],
560        rows: &mut Vec<usize>,
561        work: &mut Meter<'_, Q>,
562    ) -> Result<(), Failure<Q::Error>> {
563        rows.clear();
564        if self.rows.len() <= SCAN_ROWS || self.full {
565            self.full = false;
566            rows.try_reserve(self.rows.len())
567                .map_err(|_| Stop::Allocation)?;
568            rows.extend(0..self.rows.len());
569            return Ok(());
570        }
571        if self.stale {
572            self.rebuild_postings(work)?;
573        }
574        #[cfg(test)]
575        {
576            self.indexed_steps += 1;
577        }
578        for &atom in decided {
579            work.tick()?;
580            let start = self.postings.partition_point(|&(a, _)| a < atom);
581            for &(a, position) in &self.postings[start..] {
582                if a != atom {
583                    break;
584                }
585                work.tick()?;
586                rows.try_reserve(1).map_err(|_| Stop::Allocation)?;
587                rows.push(position);
588            }
589        }
590        rows.sort_unstable();
591        rows.dedup();
592        Ok(())
593    }

651    pub(in crate::regions) fn capture<Q: Quota + ?Sized>(
652        &mut self,
653        input: Input<'_>,
654        atom: usize,
655        work: &mut Meter<'_, Q>,
656        counts: &mut SupportExplanationStatistics,
657    ) -> Result<(), Failure<Q::Error>> {
658        if !self.can_capture() {
659            return Ok(());
660        }
661        let limits = self.limits();
662        self.attempts += 1;
663        counts.attempts += 1;
664        let mut temporary_peak = 0;
665        let before = work.charged();
666        let result = explanations::collect(
667            input,
668            atom,
669            Limits {
670                atoms: limits.founded_outputs,
671                literals: limits.founded_conditions,
672                visits: self.remaining.min(limits.founded_event_work),
673                bytes: u128::from(limits.founded_event_bytes),
674            },
675            work,
676            &mut temporary_peak,
677        );
678        self.remaining = self.remaining.saturating_sub(work.charged() - before);
679        self.peak_bytes = self.peak_bytes.max(self.retained_bytes() + temporary_peak);
680        let Some(explanation) = result? else {
681            counts.declined += 1;
682            return Ok(());
683        };
684        let literals = Literals::from_founded(explanation, &mut temporary_peak)?;
685        let retained = self.retained_bytes();
686        let mut remaining = self.remaining;
687        let result = self.retain(
688            Proposed {
689                literals,
690                source: Source::Founded,
691                hits: 0,
692                maximum_outputs: limits.founded_outputs,
693            },
694            &mut remaining,
695            work,
696            &mut temporary_peak,
697        );
698        self.remaining = remaining;
699        self.peak_bytes = self
700            .peak_bytes
701            .max(retained + temporary_peak)
702            .max(self.retained_bytes());
703        match result? {
704            Retained::Added(_) => counts.certified += 1,
705            Retained::Subsumed(_) | Retained::Declined => counts.declined += 1,
706        }
707        Ok(())
708    }

710    /// Copy the store's rows of this tier's class into the tier by retention,
711    /// so a row the tier holds or subsumes is not added twice. Each copied
712    /// row starts at the store's hit count. A tier that read this store
713    /// completely before reads only the rows retained there since, at one
714    /// read for each row it passes over. Charges the tier's sharing
715    /// allowance; its exhaustion ends seeding, never the run.
716    pub(in crate::regions) fn seed<Q: Quota + ?Sized>(
717        &mut self,
718        store: &mut LearningStore,
719        work: &mut Meter<'_, Q>,
720    ) -> Result<(), Failure<Q::Error>> {
721        let Some(owner) = self.owner.clone() else {
722            return Ok(());
723        };
724        let Some(class) = store.class(&owner) else {
725            return Ok(());
726        };
727        let read = match self.seeded_from {
728            Some((id, born)) if id == store.id => born,
729            _ => 0,
730        };
731        let mark = (store.id, class.born);
732        let mut transfer = Transfer {
733            sharing: self.sharing,
734            moved: 0,
735            peak: 0,
736        };
737        let result = self.seed_rows(
738            &class.rows,
739            read,
740            owner.limits.founded_outputs,
741            &mut transfer,
742            work,
743        );
744        // A stopped seeding keeps the rows it completed and pays for its
745        // prefix, as a finished one does.
746        self.sharing = transfer.sharing;
747        self.peak_bytes = self.peak_bytes.max(self.retained_bytes() + transfer.peak);
748        store.seeded += transfer.moved;
749        if result? {
750            self.seeded_from = Some(mark);
751        }
752        Ok(())
753    }

755    /// Retain each of `rows` retained after `read`, each starting at its
756    /// store hit count; true when every row was read.
757    fn seed_rows<Q: Quota + ?Sized>(
758        &mut self,
759        rows: &[Entry],
760        read: u64,
761        founded_outputs: usize,
762        transfer: &mut Transfer,
763        work: &mut Meter<'_, Q>,
764    ) -> Result<bool, Failure<Q::Error>> {
765        for entry in rows {
766            if entry.born <= read {
767                work.tick()?;
768                continue;
769            }
770            if !capture_read(&mut transfer.sharing, work)? {
771                return Ok(false);
772            }
773            let Some(literals) =
774                entry
775                    .row
776                    .to_literals(&mut transfer.sharing, work, &mut transfer.peak)?
777            else {
778                return Ok(false);
779            };
780            let maximum_outputs = literals.outputs.len().max(founded_outputs);
781            match self.retain(
782                Proposed {
783                    literals,
784                    source: entry.source,
785                    hits: entry.hits,
786                    maximum_outputs,
787                },
788                &mut transfer.sharing,
789                work,
790                &mut transfer.peak,
791            )? {
792                Retained::Added(position) => {
793                    // A seeded row is the store's; the tier delivers only
794                    // what it learns or what hits after seeding.
795                    self.rows[position].delivered = true;
796                    transfer.moved += 1;
797                }
798                Retained::Subsumed(_) => {}
799                Retained::Declined => return Ok(false),
800            }
801        }
802        Ok(true)
803    }

805    /// Deliver the rows retained or hit since the last delivery that pass the
806    /// quality filter to the store, under this tier's class. A delivered
807    /// row's hits since its last delivery add to the store's count for the
808    /// row that keeps them. Each row examined costs one read; delivering one
809    /// is charged to the tier's sharing allowance, whose exhaustion ends
810    /// delivery, never the run.
811    pub(in crate::regions) fn deliver<Q: Quota + ?Sized>(
812        &mut self,
813        store: &mut LearningStore,
814        work: &mut Meter<'_, Q>,
815    ) -> Result<(), Failure<Q::Error>> {
816        let Some(owner) = self.owner.clone() else {
817            return Ok(());
818        };
819        let class = store.class_mut(&owner)?;
820        let mut transfer = Transfer {
821            sharing: self.sharing,
822            moved: 0,
823            peak: 0,
824        };
825        let result = deliver_rows(&mut self.rows, class, owner.limits, &mut transfer, work);
826        // A stopped delivery keeps the rows it completed and pays for its
827        // prefix, as a finished one does.
828        self.sharing = transfer.sharing;
829        self.peak_bytes = self.peak_bytes.max(self.retained_bytes() + transfer.peak);
830        store.deliveries += transfer.moved;
831        result
832    }

834    /// The row a full tier gives up for a new row: the fewest decayed hits,
835    /// oldest first among ties, never a founded row while the tier holds no
836    /// more founded rows than its floor.
837    fn evictable(&self) -> Option<usize> {
838        let protected = self.founded <= self.limits().founded_floor;
839        self.rows
840            .iter()
841            .enumerate()
842            .filter(|(_, entry)| !protected || entry.source != Source::Founded)
843            .min_by_key(|(_, entry)| (entry.hits, entry.born))
844            .map(|(position, _)| position)
845    }

847    /// Consolidate a completed row without changing any retained row until
848    /// every comparison, allocation and publication visit has succeeded. Row
849    /// positions are private to a matching pass; retention runs between
850    /// passes. `remaining` is the allowance the caller charges: the founded
851    /// capture's, or the tier's sharing allowance.
852    fn retain<Q: Quota + ?Sized>(
853        &mut self,
854        proposed: Proposed,
855        remaining: &mut u64,
856        work: &mut Meter<'_, Q>,
857        peak: &mut u128,
858    ) -> Result<Retained, Failure<Q::Error>> {
859        let Proposed {
860            mut literals,
861            source,
862            hits,
863            maximum_outputs,
864        } = proposed;
865        let rows = self.rows.len();
866        let replaced = match self.compare(&mut literals, maximum_outputs, remaining, work, peak)? {
867            Compared::Replaces(replaced) => replaced,
868            Compared::Subsumed(position) => return Ok(Retained::Subsumed(position)),
869            Compared::Declined => return Ok(Retained::Declined),
870        };
871        let any = replaced.iter().any(|&word| word != 0);
872        // A full tier with nothing to replace evicts before it adds.
873        let victim = if !any && rows >= self.limits().tier_rows {
874            let Some(victim) = self.evictable() else {
875                return Ok(Retained::Declined);
876            };
877            Some(victim)
878        } else {
879            None
880        };
881        let Some(row) = Row::prepare(literals, remaining, work, peak)? else {
882            return Ok(Retained::Declined);
883        };
884        // Pay for the bounded compaction before its first mutation. A stop
885        // during this admission preserves all previous rows.
886        if any || victim.is_some() {
887            for _ in 0..rows {
888                if !capture_read(remaining, work)? {
889                    return Ok(Retained::Declined);
890                }
891            }
892        }
893        self.born += 1;
894        let entry = Entry {
895            row,
896            source,
897            hits,
898            unshared: 0,
899            born: self.born,
900            delivered: false,
901        };
902        let position = self.place(entry, &replaced, victim)?;
903        self.settle();
904        Ok(Retained::Added(position))
905    }

907    /// Compare a proposed row with every retained row, changing neither: an
908    /// existing row that subsumes it, or the rows it replaces once it holds
909    /// the outputs of every row with identical conditions.
910    fn compare<Q: Quota + ?Sized>(
911        &self,
912        literals: &mut Literals,
913        maximum_outputs: usize,
914        remaining: &mut u64,
915        work: &mut Meter<'_, Q>,
916        peak: &mut u128,
917    ) -> Result<Compared, Failure<Q::Error>> {
918        let rows = self.rows.len();
919        let mut replaced = Words::new(Vec::new());
920        replaced
921            .storage_mut()
922            .try_reserve_exact(rows.div_ceil(64))
923            .map_err(|_| Stop::Allocation)?;
924        replaced.storage_mut().resize(rows.div_ceil(64), 0_u64);
925        for position in 0..rows {
926            if !capture_read(remaining, work)? {
927                return Ok(Compared::Declined);
928            }
929            let row = &self.rows[position].row;
930            let Some(subsumed) = row.subsumes(literals, remaining, work)? else {
931                return Ok(Compared::Declined);
932            };
933            if subsumed {
934                return Ok(Compared::Subsumed(position));
935            }
936            let Some(combined) =
937                row.union_outputs(literals, maximum_outputs, remaining, work, peak)?
938            else {
939                return Ok(Compared::Declined);
940            };
941            if combined {
942                replaced.insert(position);
943            }
944        }
945        // A union can dominate a row examined before its outputs were added.
946        // Compare the completed proposed row against every other row.
947        for position in 0..rows {
948            if replaced.contains(position) {
949                continue;
950            }
951            if !capture_read(remaining, work)? {
952                return Ok(Compared::Declined);
953            }
954            let Some(dominated) = self.rows[position]
955                .row
956                .subsumed_by(literals, remaining, work)?
957            else {
958                return Ok(Compared::Declined);
959            };
960            if dominated {
961                replaced.insert(position);
962            }
963        }
964        Ok(Compared::Replaces(replaced))
965    }

967    /// Publish a completed entry in the evicted row's position, in the first
968    /// replaced row's position with the other replaced rows removed, or after
969    /// every row.
970    fn place(
971        &mut self,
972        entry: Entry,
973        replaced: &Words<u64>,
974        victim: Option<usize>,
975    ) -> Result<usize, Stop> {
976        if let Some(victim) = victim {
977            self.rows[victim] = entry;
978            self.evicted += 1;
979            return Ok(victim);
980        }
981        let first = replaced.next_from(0);
982        if let Some(first) = first {
983            self.rows[first] = entry;
984            let mut position = 0;
985            self.rows.retain(|_| {
986                let keep = position == first || !replaced.contains(position);
987                position += 1;
988                keep
989            });
990            return Ok(first);
991        }
992        self.rows
993            .try_reserve_exact(1)
994            .map_err(|_| Stop::Allocation)?;
995        self.peak_bytes = self
996            .peak_bytes
997            .max(self.retained_bytes() + size_of::<Entry>() as u128);
998        self.rows.push(entry);
999        Ok(self.rows.len() - 1)
1000    }

1002    /// After a retention: recount the founded rows, halve every row's hits once
1003    /// the tier has retained as many rows as it may hold, and send the next
1004    /// matching step through every row, since positions changed.
1005    fn settle(&mut self) {
1006        self.founded = self
1007            .rows
1008            .iter()
1009            .filter(|entry| entry.source == Source::Founded)
1010            .count();
1011        self.epoch += 1;
1012        if self.epoch >= self.limits().tier_rows {
1013            self.epoch = 0;
1014            for entry in &mut self.rows {
1015                entry.hits /= 2;
1016            }
1017        }
1018        // The first indexed step after the full scan rebuilds the index.
1019        self.postings.clear();
1020        self.stale = true;
1021        self.full = true;
1022    }
1023
1024    /// Rebuild the occurrence index, one read per posting. A stop leaves
1025    /// `stale` set, so a partial index is never read.
1026    fn rebuild_postings<Q: Quota + ?Sized>(
1027        &mut self,
1028        work: &mut Meter<'_, Q>,
1029    ) -> Result<(), Failure<Q::Error>> {
1030        self.postings.clear();
1031        let count = self.rows.iter().fold(0_usize, |count, entry| {
1032            let (conditions, outputs) = entry.row.lengths();
1033            count.saturating_add(conditions).saturating_add(outputs)
1034        });
1035        self.postings
1036            .try_reserve_exact(count)
1037            .map_err(|_| Stop::Allocation)?;
1038        self.peak_bytes = self.peak_bytes.max(self.retained_bytes());
1039        for (position, entry) in self.rows.iter().enumerate() {
1040            for atom in entry.row.atoms() {
1041                work.tick()?;
1042                self.postings.push((atom, position));
1043            }
1044        }
1045        self.postings.sort_unstable();
1046        self.stale = false;
1047        Ok(())
1048    }
1049}

1051/// What one seeding or delivery has spent and moved so far: its sharing
1052/// allowance, the rows it added and its transient peak. The tier records it
1053/// whether the transfer finishes or stops.
1054struct Transfer {
1055    sharing: u64,
1056    moved: u64,
1057    peak: u128,
1058}

1060/// Deliver each of a tier's `rows` new or hit since its last delivery that
1061/// passes the quality filter into `class`, one read per row examined.
1062fn deliver_rows<Q: Quota + ?Sized>(
1063    rows: &mut [Entry],
1064    class: &mut Ledger,
1065    limits: LearningLimits,
1066    transfer: &mut Transfer,
1067    work: &mut Meter<'_, Q>,
1068) -> Result<(), Failure<Q::Error>> {
1069    for entry in rows {
1070        work.tick()?;
1071        if entry.delivered && entry.unshared == 0 {
1072            continue;
1073        }
1074        let (conditions, _) = entry.row.lengths();
1075        if conditions > limits.share_conditions && entry.hits < limits.share_hits {
1076            continue;
1077        }
1078        if !capture_read(&mut transfer.sharing, work)? {
1079            return Ok(());
1080        }
1081        let Some(literals) =
1082            entry
1083                .row
1084                .to_literals(&mut transfer.sharing, work, &mut transfer.peak)?
1085        else {
1086            return Ok(());
1087        };
1088        let maximum_outputs = literals.outputs.len().max(limits.founded_outputs);
1089        let hits = entry.unshared;
1090        match class.retain(
1091            Proposed {
1092                literals,
1093                source: entry.source,
1094                hits,
1095                maximum_outputs,
1096            },
1097            &mut transfer.sharing,
1098            work,
1099            &mut transfer.peak,
1100        )? {
1101            Retained::Added(_) => transfer.moved += 1,
1102            Retained::Subsumed(position) => {
1103                let kept = &mut class.rows[position];
1104                kept.hits = kept.hits.saturating_add(hits);
1105            }
1106            Retained::Declined => return Ok(()),
1107        }
1108        entry.delivered = true;
1109        entry.unshared = 0;
1110    }
1111    Ok(())
1112}
