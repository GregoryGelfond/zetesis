1//! Split ordering over already-maintained unknown-parent counts, and over the
2//! chains a region has shortened.
3//!
4//! A chain is a conjunction or disjunction read as one node over its
5//! operands. In a region it has *members* still to be decided: its open
6//! operands, and its own truth while that is not yet known. A disjunction
7//! known to hold needs one of its open operands to hold, and a conjunction
8//! known to fail needs one to fail; while the chain's truth is open it is one
9//! more thing the operands decide, as the body of `h :- a, b` with `a` held
10//! leaves `b` and `h`.
11//!
12//! The region has *shortened* a chain when its decisions settled an operand
13//! without settling the chain and two or more members remain. Its members are
14//! where the next decision does the most: one more settled the same way
15//! brings the chain to the single member the closure decides. A chain with
16//! every operand open says nothing the program's structure does not already
17//! say, so it counts nothing here; that structure is read by the generator
18//! order ([`super::GeneratorOrder`]) where one is prepared, and by the
19//! unknown-parent counts otherwise.
20//!
21//! An atom's score adds, for each occurrence of the atom in a shortened chain,
22//! the chain's weight now less its weight with every operand open. A weight
23//! is a quarter for each member, so one chain down to two members outweighs
24//! three down to three, and a chain's first settled operand counts least. The
25//! atom is found as an operand of the chain, or under at most two enclosing
26//! implications whose truth is not yet known, which reaches the chains that
27//! hold its negation or double negation and a conditional it is part of.
28//!
29//! The closure marks a region's knowledge when it shortens a chain
30//! ([`Knowledge::shortened`]), and descendants inherit the mark. Without it
31//! every score is zero and no ranking reads one. The score is read where no
32//! restriction is active: under a generator order among the heads of least
33//! choice depth, otherwise among the eligible atoms before their
34//! unknown-parent counts. Under a candidate restriction or an objective bound
35//! the ranking subject is the restriction's own knowledge, whose chains are
36//! not the original's, and the degree ranking is unchanged. The score orders
37//! an exhaustive split and is no part of any membership decision.
38

46/// Members beyond which every chain weighs the same. A chain that long is far
47/// from deciding anything, and sixteen quarters keep the heaviest weight at
48/// `2^40`, so the weights of a million chains still add without saturating.
49const WEIGHED_MEMBERS: usize = 16;
50
51/// Enclosing implications, not yet known, an occurrence is read through: two
52/// reach an atom under `not` and under `not not`.
53const ENCLOSING_IMPLICATIONS: u8 = 2;
54
55/// A chain's weight with `members` members: a quarter for each, down to 256
56/// for [`WEIGHED_MEMBERS`] or more.
57const fn weight(members: usize) -> u64 {
58    let members = if members < WEIGHED_MEMBERS {
59        members
60    } else {
61        WEIGHED_MEMBERS
62    };
63    1 << (8 + 2 * (WEIGHED_MEMBERS - members))
64}

66impl Narrower {
67    /// Whether this index was built for a theory of `atoms` atoms.
68    pub(super) fn indexes(&self, atoms: usize) -> bool {
69        self.atom_nodes.len() == atoms
70    }
71
72    /// What the chains the region has shortened weigh for `atom` (the module
73    /// header states the score). `knowledge` must be this index's, closed for
74    /// the region.
75    ///
76    /// One unit per parent read: the parents of the atom's nodes, and those
77    /// of each enclosing implication read through. Constant additional space.
78    ///
79    /// # Errors
80    /// The meter's refusal, with the units charged before it.
81    pub(super) fn shortened_weight<C: Class, Q: Quota + ?Sized>(
82        &self,
83        knowledge: &Knowledge<C>,
84        atom: usize,
85        work: &mut Meter<'_, Q>,
86    ) -> Result<u64, Failure<Q::Error>> {
87        match &knowledge.width {
88            Width::Compact16(known) => self.shortened_weight_at(known, atom, work),
89            Width::Compact32(known) => self.shortened_weight_at(known, atom, work),
90            Width::Native(known) => self.shortened_weight_at(known, atom, work),
91        }
92    }
93
94    /// [`Self::shortened_weight`] at the knowledge's own counter width.
95    pub(super) fn shortened_weight_at<W: Count, Q: Quota + ?Sized>(
96        &self,
97        known: &Known<W>,
98        atom: usize,
99        work: &mut Meter<'_, Q>,
100    ) -> Result<u64, Failure<Q::Error>> {
101        let mut score = 0_u64;
102        for &node in &self.atom_nodes[atom] {
103            score = score.saturating_add(self.weight_above(
104                known,
105                node,
106                ENCLOSING_IMPLICATIONS,
107                work,
108            )?);
109        }
110        Ok(score)
111    }
112
113    /// The weight of the shortened chains that hold `node` as an operand, and
114    /// of those above each implication that reads `node` and is not yet known,
115    /// through `enclosing` of them.
116    fn weight_above<W: Count, Q: Quota + ?Sized>(
117        &self,
118        known: &Known<W>,
119        node: usize,
120        enclosing: u8,
121        work: &mut Meter<'_, Q>,
122    ) -> Result<u64, Failure<Q::Error>> {
123        let mut score = 0_u64;
124        for parent in self.logical_parents(node) {
125            work.tick()?;
126            let root_known = known.masks.node_known(parent);
127            if let Some(chain) = self.chain_of[parent] {
128                let chain = chain_position(chain);
129                if known.masks.witnessed(chain) {
130                    continue;
131                }
132                let written = self.chains[chain].leaves(&self.extrema).len();
133                let open = written.saturating_sub(known.neutral_operands.get(chain));
134                // The chain's own truth is a member while it is unknown.
135                let own = usize::from(!root_known);
136                if open == written || open + own < 2 {
137                    continue;
138                }
139                score = score.saturating_add(weight(open + own) - weight(written + own));
140            } else if enclosing > 0 && !root_known {
141                score =
142                    score.saturating_add(self.weight_above(known, parent, enclosing - 1, work)?);
143            }
144        }
145        Ok(score)
146    }
147}

149impl<C: Class> Knowledge<C> {
150    /// Whether a closure of the region or of an ancestor shortened a chain.
151    /// Without the mark every [`Narrower::shortened_weight`] is zero. A chain
152    /// settled since is not forgotten, so with it every weight may be zero
153    /// too.
154    pub(super) fn shortened(&self) -> bool {
155        match &self.width {
156            Width::Compact16(known) => known.shortened,
157            Width::Compact32(known) => known.shortened,
158            Width::Native(known) => known.shortened,
159        }
160    }
161
162    fn unknown_count(&self, atom: usize) -> usize {
163        match &self.width {
164            Width::Compact16(known) => known.unknown.get(atom),
165            Width::Compact32(known) => known.unknown.get(atom),
166            Width::Native(known) => known.unknown.get(atom),
167        }
168    }
169
170    /// Select by original choice marks, this subject's unknown-parent count,
171    /// then `secondary`'s count; remaining ties select the lowest atom. Both
172    /// knowledge values must use the region's atom identities and be closed
173    /// for it. This only orders an exhaustive split and changes neither value.
174    ///
175    /// The scan takes `O(A / 64 + E)` time for `A` atoms and `E` eligible atoms,
176    /// with constant additional space. Eligible atoms are the open choice heads
177    /// when any remain, otherwise all open atoms. Each primary count is charged
178    /// once; a secondary count is read at most once, only when a primary tie
179    /// requires it. An uncontested winner needs no secondary read. Packed mask
180    /// traversal is not separately charged.
181    /// Cancellation and quota reservations follow
182    /// [`Self::preferred_atom_reserved`]; every failure retains its exact work
183    /// prefix and returns no partial preference.
184    ///
185    /// # Errors
186    /// Retains cancellation, a quota refusal or a zero-grant work refusal.
187    ///
188    /// # Panics
189    /// Panics if an open atom is outside either knowledge's universe.
190    pub fn preferred_atom_with_secondary_reserved<Q: Quota + ?Sized>(
191        &self,
192        region: &Region,
193        choices: &ChoiceHeads,
194        secondary: &Self,
195        cancellation: &Cancellation,
196        work: &mut Meter<'_, Q>,
197    ) -> PreferenceAttempt<Failure<Q::Error>> {
198        let before = work.charged();
199        let result = cancellation
200            .poll()
201            .map_err(Failure::from)
202            .and_then(|()| self.preferred_with_secondary(region, choices, secondary, work));
203        work.refund();
204        PreferenceAttempt {
205            result,
206            work: work.charged() - before,
207        }
208    }
209
210    fn read_unknown<Q: Quota + ?Sized>(
211        &self,
212        atom: usize,
213        work: &mut Meter<'_, Q>,
214    ) -> Result<usize, Failure<Q::Error>> {
215        work.tick()?;
216        Ok(self.unknown_count(atom))
217    }
218
219    fn preferred_with_secondary<Q: Quota + ?Sized>(
220        &self,
221        region: &Region,
222        choices: &ChoiceHeads,
223        secondary: &Self,
224        work: &mut Meter<'_, Q>,
225    ) -> Result<Option<usize>, Failure<Q::Error>> {
226        let mut best: Option<(usize, usize, Option<usize>)> = None;
227        choices.try_for_each_eligible(region, |atom| -> Result<(), Failure<Q::Error>> {
228            let primary = self.read_unknown(atom, work)?;
229            let Some((previous, degree, cached)) = best.as_mut() else {
230                best = Some((atom, primary, None));
231                return Ok(());
232            };
233            match primary.cmp(degree) {
234                std::cmp::Ordering::Greater => {
235                    *previous = atom;
236                    *degree = primary;
237                    *cached = None;
238                }
239                std::cmp::Ordering::Equal => {
240                    let previous_count = if let Some(count) = *cached {
241                        count
242                    } else {
243                        let count = secondary.read_unknown(*previous, work)?;
244                        *cached = Some(count);
245                        count
246                    };
247                    let count = secondary.read_unknown(atom, work)?;
248                    if count > previous_count {
249                        *previous = atom;
250                        *cached = Some(count);
251                    }
252                }
253                std::cmp::Ordering::Less => {}
254            }
255            Ok(())
256        })?;
257        Ok(best.map(|(atom, _, _)| atom))
258    }
