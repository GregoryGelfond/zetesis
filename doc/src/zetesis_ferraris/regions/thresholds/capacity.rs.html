1//! Necessary counted contributions of a conjunction under an upper bound.
2//!
3//! If B entails every distinct condition in C, then B also entails at least
4//! |H union C| contributions, where H are the currently true unit-row terms.
5//! An exclusive bound not exceeding that union makes B false. Ordinary Boolean
6//! propagation then removes that alternative from any required disjunction.
7//!
8//! Preparation joins direct conjunction operands with unit-row incidences.
9//! Original root implications add necessary aliases: A -> c supplies A => c;
10//! (A or B) -> c supplies both aliases. Double-negated root implications have
11//! the same original truth. No non-root implication or frozen truth is used.
12//! Only direct operands are projected; an unrecognized structure loses a
13//! deduction, never a possible interpretation. Repeated conditions count once.
14//!
15//! The immutable projection stores only conjunction/row intersections with at
16//! least two distinct conditions. There is no row-by-conjunction Cartesian
17//! table and no per-region cache. Preparing it sorts sparse incidences; storage
18//! is linear in row offsets, retained intersections and their distinct conditions.
19//! A dense incidence join can still produce a dense projection. Each
20//! dirty row scans its intersections, stopping a condition scan once the bound
21//! is reached or exceeded. All runtime visits are charged and cancellable.
22//! Original-only row notifications are coalesced by the existing queue.
23

46impl Capacity {
47    pub(super) fn prepare(
48        theory: &Theory,
49        rows: &[Row],
50        by_condition: Option<&Adjacency>,
51        incidences: &[super::Incidence],
52        chains: &[Chain],
53        extrema: &crate::regions::extrema::Extrema,
54    ) -> Result<Option<Self>, Stop> {
55        if !rows.iter().any(eligible) || !chains.iter().any(|chain| !chain.disjunction) {
56            return Ok(None);
57        }
58        let by_condition = by_condition.expect("unit rows have incidences");
59        let aliases = aliases(theory, rows, by_condition, incidences)?;
60        let mut requirements = Vec::new();
61        let mut conditions = Vec::new();
62        let mut postings = Vec::new();
63        let mut projected = Vec::new();
64        for chain in chains.iter().filter(|chain| !chain.disjunction) {
65            projected.clear();
66            for &leaf in chain.leaves(extrema) {
67                project(leaf, rows, by_condition, incidences, &mut projected)?;
68                let start = aliases.partition_point(|&(source, _)| source < leaf);
69                for &(source, condition) in &aliases[start..] {
70                    if source != leaf {
71                        break;
72                    }
73                    project(condition, rows, by_condition, incidences, &mut projected)?;
74                }
75            }
76            projected.sort_unstable();
77            projected.dedup();
78            let mut start = 0;
79            while start < projected.len() {
80                let row = projected[start].0;
81                let mut end = start + 1;
82                while end < projected.len() && projected[end].0 == row {
83                    end += 1;
84                }
85                if end - start >= 2 {
86                    push(&mut postings, (row, requirements.len()))?;
87                    let first = conditions.len();
88                    conditions
89                        .try_reserve(end - start)
90                        .map_err(|_| Stop::Allocation)?;
91                    conditions.extend(
92                        projected[start..end]
93                            .iter()
94                            .map(|&(_, condition)| condition),
95                    );
96                    push(
97                        &mut requirements,
98                        Requirement {
99                            root: chain.root,
100                            start: first,
101                            end: conditions.len(),
102                        },
103                    )?;
104                }
105                start = end;
106            }
107        }
108        if requirements.is_empty() {
109            return Ok(None);
110        }
111        Ok(Some(Self {
112            by_row: Adjacency::build(rows.len(), postings.iter().copied())?,
113            requirements,
114            conditions,
115        }))
116    }

118    pub(super) fn has_row(&self, row: usize) -> bool {
119        !self.by_row[row].is_empty()
120    }
121
122    pub(super) fn retained_bytes(&self) -> u128 {
123        size_of::<Self>() as u128 + self.by_row.retained_bytes() - size_of::<Adjacency>() as u128
124            + self.requirements.capacity() as u128 * size_of::<Requirement>() as u128
125            + self.conditions.capacity() as u128 * size_of::<usize>() as u128
126    }
127}

129fn eligible(row: &Row) -> bool {
130    // Terms are distinct and sorted by descending weight. A singleton cannot
131    // supply the two necessary contributions retained by this projection.
132    row.terms.len() >= 2 && row.unit_first().is_some()
133}
134
135fn push<T>(values: &mut Vec<T>, value: T) -> Result<(), Stop> {
136    values.try_reserve(1).map_err(|_| Stop::Allocation)?;
137    values.push(value);
138    Ok(())
139}
140
141fn project(
142    condition: usize,
143    rows: &[Row],
144    by_condition: &Adjacency,
145    incidences: &[super::Incidence],
146    projected: &mut Vec<(usize, usize)>,
147) -> Result<(), Stop> {
148    for &edge in &by_condition[condition] {
149        let row = incidences[edge].row;
150        if eligible(&rows[row]) {
151            push(projected, (row, condition))?;
152        }
153    }
154    Ok(())
155}

157/// Double negation has classical original truth; stripping it here creates no
158/// rule or permission to use the inner implication as a frozen-reduct root.
159fn original_root(theory: &Theory, mut node: usize) -> Result<usize, Stop> {
160    let view = theory.view();
161    while let NodeView::Implies(inner, falsum) =
162        view.node(node).map_err(|_| Stop::InvalidProgram)?
163    {
164        if !view
165            .node(falsum)
166            .map_err(|_| Stop::InvalidProgram)?
167            .is_false()
168        {
169            break;
170        }
171        let NodeView::Implies(value, other_false) =
172            view.node(inner).map_err(|_| Stop::InvalidProgram)?
173        else {
174            break;
175        };
176        if !view
177            .node(other_false)
178            .map_err(|_| Stop::InvalidProgram)?
179            .is_false()
180        {
181            break;
182        }
183        node = value;
184    }
185    Ok(node)
186}

188fn aliases(
189    theory: &Theory,
190    rows: &[Row],
191    by_condition: &Adjacency,
192    incidences: &[super::Incidence],
193) -> Result<Vec<(usize, usize)>, Stop> {
194    let mut result = reserve(0)?;
195    let view = theory.view();
196    for &root in theory.roots() {
197        let root = original_root(theory, root)?;
198        let NodeView::Implies(source, target) =
199            view.node(root).map_err(|_| Stop::InvalidProgram)?
200        else {
201            continue;
202        };
203        if !by_condition[target]
204            .iter()
205            .any(|&edge| eligible(&rows[incidences[edge].row]))
206        {
207            continue;
208        }
209        push(&mut result, (source, target))?;
210        if let NodeView::Or(children) = view.node(source).map_err(|_| Stop::InvalidProgram)? {
211            for &child in children {
212                push(&mut result, (child, target))?;
213            }
214        }
215    }
216    result.sort_unstable();
217    result.dedup();
218    Ok(result)
219}

221impl<W: Count, C: crate::regions::ReadingClass> Closure<'_, W, C> {
222    pub(in crate::regions) fn threshold_capacity<Q: Quota + ?Sized>(
223        &mut self,
224        index: &Narrower,
225        row: usize,
226        work: &mut Meter<'_, Q>,
227    ) -> Result<Step, Failure<Q::Error>> {
228        let Some(capacity) = &index.thresholds.capacity else {
229            return Ok(Step::Unchanged);
230        };
231        let state = self.known.thresholds[row];
232        let Some(upper) = state.upper else {
233            return Ok(Step::Unchanged);
234        };
235        let mut step = Step::Unchanged;
236        for &entry in &capacity.by_row[row] {
237            work.tick()?;
238            let required = &capacity.requirements[entry];
239            if self.known.never.contains(required.root) {
240                continue;
241            }
242            let mut minimum = state.known;
243            for &condition in &capacity.conditions[required.start..required.end] {
244                work.tick()?;
245                // known counts processed events. A pending true condition can
246                // make this an underestimate, never an overestimate. Its later
247                // processed event schedules this row again.
248                minimum += i128::from(!self.known.sure.contains(condition));
249                if minimum >= upper {
250                    break;
251                }
252            }
253            if minimum >= upper {
254                step = step.join(self.never(required.root));
255                if step == Step::Contradiction {
256                    return Ok(step);
257                }
258            }
259        }
260        Ok(step)
261    }
262}
