1//! Unit-count bounds across Boolean witnesses.
2//!
3//! Flat theories use the sparse least-condition anchor join and exact-prefix
4//! authentication. Preparation takes O(N + R + U log U + L + (R + C + L)
5//! log(A + 1)) work for N nodes, R native rows, U unit rows, C chains, L chain
6//! leaves and A distinct row anchors, including the cone summary below.
7//! Retained storage is O(U + C); there is no row-by-chain table. With no unit
8//! row, preparation allocates nothing and never examines chains.
9//!
10//! Preparation also records, in one bit per chain, whether the chain's
11//! same-kind cone contains a row anchor, the O(N + (C + L) log(A + 1)) part of
12//! that total. Every row either discovery below can reach is posted under an
13//! anchor of that cone, so a chain whose cone holds none returns at once,
14//! without reading its cone or postings.
15//!
16//! A shared associative DAG can hide an intermediate cut such as [C,b,d]
17//! inside ((C or b) or d). Opaque and fully unfolded minima do not identify
18//! every such cut. For a nested Boolean event, walk its same-kind cone and
19//! filter its distinct node IDs against the existing row anchors. For a native
20//! event, follow the anchor's existing immediate parent edges through same-kind
21//! chains. Row discovery uses existing local edge roles: an unrelated shared
22//! component does not move a flat anchor off its immutable posting route.
23//! Both discoveries are temporary joins, without an ancestor table.
24//!
25//! Candidates are then authenticated in their actual direction. A Boolean
26//! witness implies a row witness when every terminal path meets a row condition;
27//! stop at those conditions, including compounds. A row witness implies a
28//! OR witness when every row condition occurs in its same-kind cone. A true
29//! AND entails all k contained conditions, so a native exclusive upper bound
30//! at most k refutes that AND, even when the row has additional terms.
31//! For OR the witness is truth; for AND it is falsity. These implications do
32//! not require independent conditions. Native rows remain distinct unit terms.
33//!
34//! One requested cone or ancestor walk visits O(V + E) reachable nodes/edges.
35//! Cover authentication adds O(V log(Ur + 1)) comparisons for Ur row terms;
36//! containment adds O(Ur) marked-node reads. Discovery adds anchor searches and
37//! posting visits. All runtime visits/comparisons are metered and cancellable.
38//! Dense buckets may still be expensive. Worker scratch is O(N + E + R + C),
39//! reserved lazily with checked allocation and actual-capacity accounting;
40//! cloned region knowledge and the flat counter path remain unchanged.
41//!
42//! Frozen subjects skip this bridge: these are original-subject consequences,
43//! never an assumption about arbitrary frozen-reduct interpretations.

61/// One immutable anchor relation projected to native rows and Boolean chains.
62/// Anchor slots cover only distinct first conditions of grouped unit rows;
63/// no node-sized offsets or authenticated row/chain pairs are stored.
64#[derive(Clone, Debug)]
65pub(super) struct Connectives {
66    anchors: Vec<usize>,
67    rows: Adjacency,
68    chains: Adjacency,
69    /// Sparse reverse projection, ordered by original chain ID.
70    by_chain: Vec<(usize, usize)>,
71    /// One bit per chain: set when its same-kind cone, the nodes an expanded
72    /// discovery walk collects, contains a row anchor.
73    anchored_cones: Vec<u64>,
74}
75
76impl Connectives {
77    /// Retained anchor projections; the enclosing Thresholds owns the header.
78    pub(super) fn retained_bytes(&self) -> u128 {
79        self.rows.retained_bytes()
80            + self.chains.retained_bytes()
81            + self.anchors.capacity() as u128 * size_of::<usize>() as u128
82            + self.by_chain.capacity() as u128 * size_of::<(usize, usize)>() as u128
83            + self.anchored_cones.capacity() as u128 * size_of::<u64>() as u128
84    }
85
86    pub(super) fn prepare(
87        rows: &[Row],
88        chains: &[Chain],
89        chain_of: &[Option<NonZeroUsize>],
90        extrema: &crate::regions::extrema::Extrema,
91        shared_connectives: bool,
92    ) -> Result<Option<Self>, Stop> {
93        let anchors = anchor_keys(rows)?;
94        if anchors.is_empty() {
95            // In particular, a Boolean-only theory does not inspect chains.
96            return Ok(None);
97        }
98        let by_chain = chain_anchors(chains, extrema, &anchors)?;
99        if by_chain.is_empty() && !shared_connectives {
100            return Ok(None);
101        }
102        let row_postings = Adjacency::build(
103            anchors.len(),
104            rows.iter().enumerate().filter_map(|(row, data)| {
105                unit_first(data).map(|first| {
106                    let anchor = anchors.binary_search(&first).expect("prepared row anchor");
107                    (anchor, row)
108                })
109            }),
110        )?;
111        let chain_postings = Adjacency::build(
112            anchors.len(),
113            by_chain.iter().map(|&(chain, anchor)| (anchor, chain)),
114        )?;
115        let anchored_cones = anchored_cones(chains, chain_of, extrema, &anchors)?;
116        Ok(Some(Self {
117            anchors,
118            rows: row_postings,
119            chains: chain_postings,
120            by_chain,
121            anchored_cones,
122        }))
123    }

125    /// Find a processed chain's one anchor, charging each reverse-index probe.
126    fn rows(&self, chain: usize, work: &mut Work<'_>) -> Result<&[usize], Stop> {
127        let Some(position) = find(&self.by_chain, chain, |&(chain, _)| chain, work)? else {
128            return Ok(&[]);
129        };
130        Ok(&self.rows[self.by_chain[position].1])
131    }
132
133    /// Whether the chain's same-kind cone contains any row anchor. When it
134    /// does not, neither discovery from this chain can reach a row.
135    fn anchored_cone(&self, chain: usize) -> bool {
136        bit(&self.anchored_cones, chain)
137    }
138
139    /// A cone node is only a discovery key, not proof of a row relation.
140    fn rows_at(&self, node: usize, work: &mut Work<'_>) -> Result<&[usize], Stop> {
141        let Some(anchor) = find(&self.anchors, node, |&condition| condition, work)? else {
142            return Ok(&[]);
143        };
144        Ok(&self.rows[anchor])
145    }
146
147    /// Find a native row's anchor, charging each immutable key probe.
148    fn chains(&self, first: usize, work: &mut Work<'_>) -> Result<&[usize], Stop> {
149        let Some(anchor) = find(&self.anchors, first, |&condition| condition, work)? else {
150            return Ok(&[]);
151        };
152        Ok(&self.chains[anchor])
153    }
154}

156/// Mark each chain whose same-kind cone contains a row anchor. A cone is the
157/// chain's root together with, for each leaf, the cone of the same-kind chain
158/// that leaf roots, or else the leaf itself: exactly the nodes an expanded
159/// discovery walk collects. A flat chain has no same-kind inner chain, so its
160/// cone holds its least leaf, the key of its posting. Operands precede their
161/// parents, so ascending node order completes every inner chain before an
162/// outer one reads it. With N nodes, C chains, L chain leaves and A anchors
163/// this takes O(N + (C + L) log(A + 1)) work and retains one bit per chain. A
164/// node absent from `chain_of` roots no chain.
165fn anchored_cones(
166    chains: &[Chain],
167    chain_of: &[Option<NonZeroUsize>],
168    extrema: &crate::regions::extrema::Extrema,
169    anchors: &[usize],
170) -> Result<Vec<u64>, Stop> {
171    let mut marks = reserve(chains.len().div_ceil(64))?;
172    marks.resize(chains.len().div_ceil(64), 0);
173    for &link in chain_of.iter().flatten() {
174        let chain = chain_position(link);
175        let source = &chains[chain];
176        let reaches = anchors.binary_search(&source.root).is_ok()
177            || source.leaves(extrema).iter().any(|&leaf| {
178                match chain_of.get(leaf).copied().flatten().map(chain_position) {
179                    Some(inner) if chains[inner].disjunction == source.disjunction => {
180                        bit(&marks, inner)
181                    }
182                    _ => anchors.binary_search(&leaf).is_ok(),
183                }
184            });
185        if reaches {
186            marks[chain / 64] |= 1u64 << (chain % 64);
187        }
188    }
189    Ok(marks)
190}

192/// Allocate only after finding a native unit row. Sorting and deduplication
193/// retain one key per distinct anchor; the allocation is bounded by unit rows.
194fn anchor_keys(rows: &[Row]) -> Result<Vec<usize>, Stop> {
195    let count = rows.iter().filter(|row| unit_first(row).is_some()).count();
196    let mut anchors = reserve(count)?;
197    anchors.extend(rows.iter().filter_map(unit_first));
198    anchors.sort_unstable();
199    anchors.dedup();
200    Ok(anchors)
201}

203/// Scan each nonempty chain's borrowed leaves once, without reordering them.
204/// Only a minimum shared with some native unit row receives a reverse entry.
205/// Increasing chain IDs preserve the old parent-posting order at every anchor.
206fn chain_anchors(
207    chains: &[Chain],
208    extrema: &crate::regions::extrema::Extrema,
209    anchors: &[usize],
210) -> Result<Vec<(usize, usize)>, Stop> {
211    let mut reverse = Vec::new();
212    for (chain, data) in chains.iter().enumerate() {
213        let Some(first) = data.leaves(extrema).iter().min() else {
214            continue;
215        };
216        if let Ok(anchor) = anchors.binary_search(first) {
217            reverse.try_reserve(1).map_err(|_| Stop::Allocation)?;
218            reverse.push((chain, anchor));
219        }
220    }
221    Ok(reverse)
222}

224/// A shrinking binary-search interval; charge before reading each candidate.
225fn find<T>(
226    values: &[T],
227    wanted: usize,
228    key: impl Fn(&T) -> usize,
229    work: &mut Work<'_>,
230) -> Result<Option<usize>, Stop> {
231    let mut start = 0;
232    let mut end = values.len();
233    while start < end {
234        work.tick()?;
235        let middle = start + (end - start) / 2;
236        match key(&values[middle]).cmp(&wanted) {
237            Ordering::Less => start = middle + 1,
238            Ordering::Equal => return Ok(Some(middle)),
239            Ordering::Greater => end = middle,
240        }
241    }
242    Ok(None)
243}

245fn unit_first(row: &Row) -> Option<usize> {
246    let &(first, 1) = row.terms.first()? else {
247        return None;
248    };
249    Some(first)
250}

252/// Authenticate the shorter canonical row without reordering borrowed leaves.
253/// Distinct leaves at or below the prefix's last condition must all belong to
254/// the prefix, and their count must equal its length. Thus no earlier condition
255/// is missing or replaced by an unrelated leaf. Larger leaves are extensions.
256fn exact_prefix(
257    terms: &[(usize, i128)],
258    leaves: &[usize],
259    work: &mut Work<'_>,
260) -> Result<bool, Stop> {
261    let prefix = &terms[..terms.len().min(leaves.len())];
262    let Some(&(last, _)) = prefix.last() else {
263        return Ok(false);
264    };
265    let mut matched = 0;
266    for &leaf in leaves {
267        work.tick()?;
268        if leaf <= last {
269            if !contains(prefix, leaf, work)? {
270                return Ok(false);
271            }
272            matched += 1;
273        }
274    }
275    Ok(matched == prefix.len())
276}

278/// The candidate interval shrinks on every comparison. Each actual term read
279/// consumes work, including the comparison that finds or rejects membership.
280fn contains(terms: &[(usize, i128)], leaf: usize, work: &mut Work<'_>) -> Result<bool, Stop> {
281    Ok(find(terms, leaf, |&(condition, _)| condition, work)?.is_some())
282}

284/// A row anchor needs ancestor discovery only if it can enter a shared
285/// same-kind path. A connective anchor is kept conservatively because it can
286/// itself be a candidate. Otherwise, any longer path begins at an immediate
287/// chain parent marked as participating in a same-kind edge. Ordinal parents
288/// conservatively keep their existing streamed candidate order. Otherwise,
289/// all candidates are immediate flat chains; an authenticated exact prefix must
290/// have the same least condition and is already in the immutable postings.
291///
292/// This is a metered read of existing geometry, not a retained reachability
293/// index. It allocates nothing and leaves coverage/containment authentication
294/// unchanged. An interrupted applicability query changes no logical state.
295fn needs_ancestor_discovery(
296    index: &Narrower,
297    anchor: usize,
298    work: &mut Work<'_>,
299) -> Result<bool, Stop> {
300    if index.witness_chains.is_empty() {
301        return Ok(false);
302    }
303    work.tick()?;
304    if index.chain_of[anchor].is_some() {
305        return Ok(true);
306    }
307    for parent in index.logical_parents(anchor) {
308        work.tick()?;
309        if index.chain_of[parent].is_some_and(|link| {
310            let chain = crate::regions::chain_position(link);
311            index.chains[chain].ordinal() || crate::regions::witnesses::participates(index, chain)
312        }) {
313            return Ok(true);
314        }
315    }
316    Ok(false)
317}
318
319fn bound(disjunction: bool, leaves: &[usize]) -> i128 {
320    if disjunction { 1 } else { leaves.len() as i128 }
321}

323/// Lower requirements only rise and exclusive upper requirements only fall.
324/// A proposal already implied now cannot become useful later in this closure.
325/// Keep the same predicate at discovery and before authenticated transport.
326fn useful_bound(state: &State, value: bool, bound: i128) -> bool {
327    if value {
328        state.lower < bound
329    } else {
330        state.upper.is_none_or(|upper| bound < upper)
331    }
332}

341impl<C: Count> Closure<'_, C> {
342    /// A processed true Boolean prefix supplies a lower bound to its native
343    /// extension; a false Boolean extension supplies an exclusive upper bound
344    /// to its native prefix. Each candidate is authenticated before transport.
345    pub(in crate::regions) fn threshold_connective(
346        &mut self,
347        index: &Narrower,
348        chain: usize,
349        value: bool,
350        work: &mut Work<'_>,
351    ) -> Result<Step, Stop> {
352        let source = &index.chains[chain];
353        if value != source.disjunction {
354            // teach_chain already forced every leaf of a true And or false Or.
355            // Processed condition events establish the same lower bound (all
356            // prefix leaves true) or a stronger upper bound (all terms false)
357            // wherever that Boolean requirement could be transported.
358            return Ok(Step::Unchanged);
359        }
360        let Some(anchors) = &index.thresholds.connectives else {
361            return Ok(Step::Unchanged);
362        };
363        if !anchors.anchored_cone(chain) {
364            // Both discoveries below reach rows only through an anchor in this
365            // chain's same-kind cone: an expanded walk reads the anchors of the
366            // cone's nodes, and a flat chain's posting is keyed by its least
367            // leaf. With no anchor there is no candidate row to transport.
368            return Ok(Step::Unchanged);
369        }
370        let expanded = crate::regions::witnesses::expands(index, chain);
371        let rows = if expanded {
372            // Each distinct cone node has at most one anchor slot, and each
373            // row has one anchor. Thus this temporary join emits no row twice.
374            self.lists.witnesses.cone(index, chain, work)?;
375            self.lists.witnesses.clear_candidates();
376            for position in 0..self.lists.witnesses.prepared().len() {
377                let node = self.lists.witnesses.prepared()[position];
378                for &row in anchors.rows_at(node, work)? {
379                    work.tick()?;
380                    let proposed = if value {
381                        1
382                    } else {
383                        index.thresholds.rows[row].terms.len() as i128
384                    };
385                    // Discovery still visits every posting. Charge this extra
386                    // state read before retaining only potentially useful IDs;
387                    // authentication and its later recheck keep survivor order.
388                    work.tick()?;
389                    if useful_bound(&self.known.thresholds[row], value, proposed) {
390                        self.lists.witnesses.push_candidate(row)?;
391                    }
392                }
393            }
394            &[][..]
395        } else {
396            anchors.rows(chain, work)?
397        };
398        let count = if expanded {
399            self.lists.witnesses.candidates().len()
400        } else {
401            rows.len()
402        };
403        let mut step = Step::Unchanged;
404        // Authentication reuses witness scratch, so release each candidate
405        // borrow before walking its cone. Candidate IDs remain unchanged.
406        let mut position = 0;
407        while position < count {
408            let row = if expanded {
409                self.lists.witnesses.candidates()[position]
410            } else {
411                rows[position]
412            };
413            position += 1;
414            work.tick()?;
415            let data = &index.thresholds.rows[row];
416            let length = data.terms.len();
417            let bound = if expanded && !source.disjunction {
418                length as i128
419            } else {
420                bound(source.disjunction, source.leaves(&index.extrema))
421            };
422            if !useful_bound(&self.known.thresholds[row], value, bound) {
423                continue;
424            }
425            let authenticated = if expanded {
426                self.lists
427                    .witnesses
428                    .covered_by(index, chain, work, |node, work| {
429                        contains(&data.terms, node, work)
430                    })?
431            } else {
432                let compatible = if value {
433                    source.leaves(&index.extrema).len() <= length
434                } else {
435                    source.leaves(&index.extrema).len() >= length
436                };
437                compatible && exact_prefix(&data.terms, source.leaves(&index.extrema), work)?
438            };
439            if !authenticated {
440                continue;
441            }
442            let state = &mut self.known.thresholds[row];
443            if value {
444                state.lower = bound;
445            } else {
446                state.upper = Some(bound);
447            }
448            self.lists.queue_threshold(row);
449            step = step.join(self.threshold_row(index, row, None, work)?);
450            if step == Step::Contradiction {
451                return Ok(step);
452            }
453            step = Step::Changed;
454        }
455        Ok(step)
456    }

458    /// Compose native-prefix transport with Boolean endpoints discovered for
459    /// the same dirty requirement. Both consume original row state and stop at
460    /// the first contradiction, without charging the remaining neighbours.
461    pub(in crate::regions) fn threshold_requirements(
462        &mut self,
463        index: &Narrower,
464        row: usize,
465        work: &mut Work<'_>,
466    ) -> Result<Step, Stop> {
467        let mut step = self.threshold_prefixes(index, row, work)?;
468        if step == Step::Contradiction {
469            return Ok(step);
470        }
471        let Some(anchors) = &index.thresholds.connectives else {
472            return Ok(step);
473        };
474        let data = &index.thresholds.rows[row];
475        let Some(first) = unit_first(data) else {
476            return Ok(step);
477        };
478        let source = self.known.thresholds[row];
479        let length = data.terms.len();
480        if source.lower < 1 && source.upper.is_none_or(|upper| upper > length as i128) {
481            return Ok(step);
482        }
483        let shared = needs_ancestor_discovery(index, first, work)?;
484        let chains = if shared {
485            // Discovery follows one connective per branch. An irrelevant
486            // polarity stays irrelevant along that branch. A known absorbing
487            // truth (OR sure / AND never) already entails that same truth at
488            // every ancestor; ordinary node events complete it if still pending.
489            // Keep the opposite-known polarity: authentication may refute it.
490            let sure = &*self.known.sure;
491            let never = &*self.known.never;
492            self.lists
493                .witnesses
494                .parent_chains(index, first, work, |chain| {
495                    if chain.disjunction {
496                        source.lower >= 1 && !bit(sure, chain.root)
497                    } else {
498                        source.upper.is_some_and(|upper| upper <= length as i128)
499                            && !bit(never, chain.root)
500                    }
501                })?;
502            &[][..]
503        } else {
504            anchors.chains(first, work)?
505        };
506        let count = if shared {
507            self.lists.witnesses.candidates().len()
508        } else {
509            chains.len()
510        };
511        // Cone walks mutate scratch without changing this candidate sequence.
512        let mut position = 0;
513        while position < count {
514            let chain_index = if shared {
515                self.lists.witnesses.candidates()[position]
516            } else {
517                chains[position]
518            };
519            position += 1;
520            work.tick()?;
521            let chain = &index.chains[chain_index];
522            // Keep the cheap truth and bound filters before any cone walk.
523            let sure = chain.disjunction && source.lower >= 1 && !bit(self.known.sure, chain.root);
524            let never = !chain.disjunction
525                && source.upper.is_some_and(|upper| upper <= length as i128)
526                && !bit(self.known.never, chain.root);
527            if !(sure || never) {
528                continue;
529            }
530            let expanded = crate::regions::witnesses::expands(index, chain_index);
531            let authenticated = if expanded {
532                self.lists.witnesses.mark_cone(index, chain_index, work)?;
533                let mut contained = 0;
534                for &(condition, _) in &data.terms {
535                    work.tick()?;
536                    contained += usize::from(self.lists.witnesses.visited(condition));
537                }
538                if sure {
539                    contained == length
540                } else {
541                    // A true AND makes every contained row condition true.
542                    // Retain the exact-prefix inference when this native row
543                    // has extra terms but its exclusive upper bound is tight.
544                    source.upper.is_some_and(|upper| upper <= contained as i128)
545                }
546            } else {
547                let bound = bound(chain.disjunction, chain.leaves(&index.extrema));
548                let compatible =
549                    (sure && length <= chain.leaves(&index.extrema).len() && source.lower >= bound)
550                        || (never
551                            && length >= chain.leaves(&index.extrema).len()
552                            && source.upper.is_some_and(|upper| upper <= bound));
553                compatible && exact_prefix(&data.terms, chain.leaves(&index.extrema), work)?
554            };
555            if !authenticated {
556                continue;
557            }
558            step = step.join(if sure {
559                self.sure(chain.root)
560            } else {
561                self.never(chain.root)
562            });
563            if step == Step::Contradiction {
564                return Ok(step);
565            }
566        }
567        Ok(step)
568    }
