1//! A residual divisibility check over the existing grouped coefficients.
2//!
3//! After mandatory propagation drains, every compatible complete sum is the processed
4//! base plus a multiple of the gcd of undecided coefficients. Missing that
5//! lattice refutes a region; hitting it does not establish a possible model.
6//! This original-only check neither publishes bounds nor changes formula truth.
7//! A partial gcd admitting an interval point cannot lead to a refutation: every
8//! later gcd divides it, so the final lattice still contains that point. Scan
9//! small coefficients first to expose these exits without another term order.
10//! It allocates nothing, charging terms, Euclidean steps and interval checks.
11//! Dirty events coalesce across mandatory waves; each row is scanned at most
12//! once per closure. Deferral preserves refutations: deciding conditions only
13//! restricts the old residue lattice, and requirements only tighten its interval.
14
15use crate::regions::{Closure, Count, Narrower, Step, Work, bit};
16use zetesis_cpu::Stop;

23impl State {
24    /// Either endpoint is already a lattice witness unless both requirements
25    /// exclude it. This filter leaves one-sided requirements unchanged.
26    pub(super) fn has_residual_interval(self) -> bool {
27        self.lower > self.known && self.upper.is_some_and(|upper| upper <= self.possible)
28    }
29}
30
31impl<C: Count> Closure<'_, C> {
32    /// Called only for original subjects after all mandatory propagation drains.
33    /// Thus every sure/never term has updated the processed sums. This check
34    /// publishes no knowledge, so later residual checks cannot dirty the row.
35    pub(in crate::regions) fn threshold_residual(
36        &self,
37        index: &Narrower,
38        row: usize,
39        work: &mut Work<'_>,
40    ) -> Result<Step, Stop> {
41        let data = &index.thresholds.rows[row];
42        // Existing unit-row transport acquires no arithmetic scan or work tick.
43        if !data.has_nonunit_coefficient() {
44            return Ok(Step::Unchanged);
45        }
46        work.tick()?;
47        debug_assert!(
48            self.lists.nodes.is_empty()
49                && self.lists.threshold_rows.is_empty()
50                && self.lists.heads.is_empty(),
51            "residual sums require drained mandatory events"
52        );
53        let state = self.known.thresholds[row];
54        if !state.has_residual_interval() {
55            return Ok(Step::Unchanged);
56        }
57        let upper = state.upper.expect("two-sided residual interval");
58        let lower = state.known.max(state.lower);
59        if lower >= upper || lower > state.possible {
60            return Ok(Step::Contradiction);
61        }
62        let width = upper.checked_sub(lower).ok_or(Stop::InvalidProgram)?;
63        let upper = state
64            .possible
65            .min(upper.checked_sub(1).ok_or(Stop::InvalidProgram)?);
66        let mut divisor = 0;
67        // Forcing keeps its descending order; this independent reduction reads
68        // the same immutable coefficients from the small end.
69        for &(condition, weight) in data.terms.iter().rev() {
70            work.tick()?;
71            if bit(self.known.sure, condition) || bit(self.known.never, condition) {
72                continue;
73            }
74            // This first undecided term was already read and charged. A
75            // coefficient fitting the exclusive width supplies a point for
76            // every final divisor, without a separate probe or modulo test.
77            // Later terms are no smaller, so test this witness only once.
78            if divisor == 0 && weight <= width {
79                return Ok(Step::Unchanged);
80            }
81            let previous = divisor;
82            divisor = gcd(divisor, weight, work)?;
83            if divisor == 1 {
84                return Ok(Step::Unchanged);
85            }
86            if divisor != previous {
87                work.tick()?;
88                if !misses(state.known, lower, upper, divisor)? {
89                    return Ok(Step::Unchanged);
90                }
91            }
92        }
93        // Every positive divisor was tested when it last changed. Reaching
94        // the end means that final progression already missed the interval.
95        if divisor != 0 {
96            return Ok(Step::Contradiction);
97        }
98        work.tick()?;
99        Ok(if misses(state.known, lower, upper, divisor)? {
100            Step::Contradiction
101        } else {
102            Step::Unchanged
103        })
104    }
105}

107/// Positive grouped coefficients fit i128, even when they exceed u64. Each
108/// Euclidean step strictly decreases the nonnegative right operand.
109fn gcd(mut left: i128, mut right: i128, work: &mut Work<'_>) -> Result<i128, Stop> {
110    if left == 0 || left == right {
111        return Ok(right);
112    }
113    while right != 0 {
114        work.tick()?;
115        let remainder = left.checked_rem(right).ok_or(Stop::InvalidProgram)?;
116        left = right;
117        right = remainder;
118    }
119    Ok(left)
120}

122/// Invariant: 0 <= base <= lower <= upper. Distance to the next lattice point
123/// is compared with interval width, without adding to an extreme bound.
124fn misses(base: i128, lower: i128, upper: i128, divisor: i128) -> Result<bool, Stop> {
125    if divisor == 0 {
126        return Ok(base < lower || base > upper);
127    }
128    let offset = lower.checked_sub(base).ok_or(Stop::InvalidProgram)?;
129    let remainder = offset.checked_rem(divisor).ok_or(Stop::InvalidProgram)?;
130    let distance = if remainder == 0 {
131        0
132    } else {
133        divisor - remainder
134    };
135    let width = upper.checked_sub(lower).ok_or(Stop::InvalidProgram)?;
136    Ok(distance > width)
137}
