1//! Residual divisibility and condition forcing over grouped coefficients.
2//!
3//! After mandatory propagation drains, every compatible complete sum is the
4//! processed base plus a multiple of the gcd of undecided coefficients. Missing
5//! that lattice refutes a region; hitting it does not establish a possible model.
6//! Removing one coefficient gives the corresponding necessary test for each
7//! truth value of its condition. If one branch misses, the other is required.
8//!
9//! A cheap whole-row scan runs first. One fitting open coefficient and a small
10//! positive gcd of other open coefficients exclude useful complement tests.
11//! This ascending reduction can stop before reading the whole row; it stores no
12//! condition-dependent state. Otherwise one reusable suffix reduction
13//! and a forward prefix reduction test all complements without quadratic scans.
14//! Each batch uses one drained snapshot: conditions are unique within a grouped
15//! row, and publishing one merely queues its node. Pending events cannot change
16//! another condition or the processed sums during this reduction. The prefix
17//! retains each originally open coefficient even after its decision is queued.
18//! Mandatory propagation drains those decisions before another residual batch.
19//! Every new wave fixes an open condition; no numeric bound or reason is cached.
20

31impl State {
32    /// Either endpoint is already a lattice witness unless both requirements
33    /// exclude it. This filter leaves one-sided requirements unchanged.
34    pub(super) fn has_residual_interval(self) -> bool {
35        self.lower > self.known && self.upper.is_some_and(|upper| upper <= self.possible)
36    }
37}

39impl<W: Count, C: crate::regions::ReadingClass> Closure<'_, W, C> {
40    /// Called only for original subjects after all mandatory propagation drains.
41    /// Thus every sure/never term has updated the processed sums.
42    pub(in crate::regions) fn threshold_residual<Q: Quota + ?Sized>(
43        &mut self,
44        index: &Narrower,
45        row: usize,
46        work: &mut Meter<'_, Q>,
47    ) -> Result<Step, Failure<Q::Error>> {
48        let checked = self.threshold_residual_check(index, row, work)?;
49        if checked == Step::Contradiction {
50            return Ok(checked);
51        }
52        let data = &index.thresholds.rows[row];
53        let state = self.known.thresholds[row];
54        if !data.has_nonunit_coefficient() || !state.has_residual_interval() {
55            return Ok(Step::Unchanged);
56        }
57        let upper = state.upper.expect("two-sided residual interval");
58        let width = upper.checked_sub(state.lower).ok_or(Stop::InvalidProgram)?;
59        // Every complement except the first fitting coefficient's retains
60        // that coefficient. Its own complement has a lattice witness once a
61        // positive partial gcd fits the width: later gcds only divide it.
62        // Clipped branch endpoints are all-off/all-on witnesses after the
63        // mandatory bound rules drain. The empty gcd (zero) is no witness.
64        let mut fitting = false;
65        let mut complement = 0;
66        for &(condition, weight) in data.terms.iter().rev() {
67            work.tick()?;
68            if !fitting && weight > width {
69                break;
70            }
71            if self.known.sure.contains(condition) || self.known.never.contains(condition) {
72                continue;
73            }
74            if !fitting {
75                fitting = true;
76                continue;
77            }
78            if weight <= width {
79                return Ok(Step::Unchanged);
80            }
81            complement = gcd(complement, weight, work)?;
82            if complement > 0 && complement <= width {
83                return Ok(Step::Unchanged);
84            }
85        }
86        prepare_suffix(
87            &data.terms,
88            &self.known.sure,
89            &self.known.never,
90            &mut self.lists.residual_suffix,
91            work,
92        )?;
93        let upper = upper.checked_sub(1).ok_or(Stop::InvalidProgram)?;
94        let mut prefix = 0;
95        let mut step = Step::Unchanged;
96        for (position, &(condition, weight)) in data.terms.iter().enumerate() {
97            work.tick()?;
98            // Earlier publications touch only earlier, distinct conditions.
99            // This condition still has its initial snapshot truth.
100            if self.known.sure.contains(condition) || self.known.never.contains(condition) {
101                continue;
102            }
103            let suffix = self.lists.residual_suffix[data.terms.len() - position - 1];
104            let divisor = gcd(prefix, suffix, work)?;
105            prefix = gcd(prefix, weight, work)?;
106            // A unit lattice cannot improve the already drained bound rules.
107            if divisor == 1 {
108                continue;
109            }
110            let false_maximum = state
111                .possible
112                .checked_sub(weight)
113                .ok_or(Stop::InvalidProgram)?;
114            let true_base = state
115                .known
116                .checked_add(weight)
117                .ok_or(Stop::InvalidProgram)?;
118            work.tick()?;
119            let false_excluded =
120                branch_misses(state.known, false_maximum, state.lower, upper, divisor)?;
121            work.tick()?;
122            let true_excluded =
123                branch_misses(true_base, state.possible, state.lower, upper, divisor)?;
124            if false_excluded && true_excluded {
125                return Ok(Step::Contradiction);
126            }
127            let decision = if false_excluded {
128                self.sure(condition)
129            } else if true_excluded {
130                self.never(condition)
131            } else {
132                Step::Unchanged
133            };
134            if decision == Step::Contradiction {
135                return Ok(decision);
136            }
137            step = step.join(decision);
138        }
139        Ok(step)
140    }

142    /// The allocation-free whole-row check precedes complement preparation.
143    /// A partial gcd admitting an interval point cannot lead to refutation:
144    /// every later gcd divides it and retains that point. Reading the small
145    /// coefficients first exposes this exit without another immutable order.
146    fn threshold_residual_check<Q: Quota + ?Sized>(
147        &self,
148        index: &Narrower,
149        row: usize,
150        work: &mut Meter<'_, Q>,
151    ) -> Result<Step, Failure<Q::Error>> {
152        let data = &index.thresholds.rows[row];
153        // Existing unit-row transport acquires no arithmetic scan or work tick.
154        if !data.has_nonunit_coefficient() {
155            return Ok(Step::Unchanged);
156        }
157        work.tick()?;
158        debug_assert!(
159            self.lists.nodes.is_empty()
160                && self.lists.threshold_rows.is_empty()
161                && self.lists.heads.is_empty(),
162            "residual sums require drained mandatory events"
163        );
164        let state = self.known.thresholds[row];
165        if !state.has_residual_interval() {
166            return Ok(Step::Unchanged);
167        }
168        let upper = state.upper.expect("two-sided residual interval");
169        let lower = state.known.max(state.lower);
170        if lower >= upper || lower > state.possible {
171            return Ok(Step::Contradiction);
172        }
173        let width = upper.checked_sub(lower).ok_or(Stop::InvalidProgram)?;
174        let upper = state
175            .possible
176            .min(upper.checked_sub(1).ok_or(Stop::InvalidProgram)?);
177        let mut divisor = 0;
178        // Forcing keeps its descending order; this independent reduction reads
179        // the same immutable coefficients from the small end.
180        for &(condition, weight) in data.terms.iter().rev() {
181            work.tick()?;
182            if self.known.sure.contains(condition) || self.known.never.contains(condition) {
183                continue;
184            }
185            // This first undecided term was already read and charged. A
186            // coefficient fitting the exclusive width supplies a point for
187            // every final divisor, without a separate probe or modulo test.
188            // Later terms are no smaller, so test this witness only once.
189            if divisor == 0 && weight <= width {
190                return Ok(Step::Unchanged);
191            }
192            let previous = divisor;
193            divisor = gcd(divisor, weight, work)?;
194            if divisor == 1 {
195                return Ok(Step::Unchanged);
196            }
197            if divisor != previous {
198                work.tick()?;
199                if !misses(state.known, lower, upper, divisor)? {
200                    return Ok(Step::Unchanged);
201                }
202            }
203        }
204        // Every positive divisor was tested when it last changed. Reaching
205        // the end means that final progression already missed the interval.
206        if divisor != 0 {
207            return Ok(Step::Contradiction);
208        }
209        work.tick()?;
210        Ok(if misses(state.known, lower, upper, divisor)? {
211            Step::Contradiction
212        } else {
213            Step::Unchanged
214        })
215    }
216}

218/// Rebuild one reversed reduction from the drained condition masks. Capacity is
219/// retained by the walker, but no truth-dependent entry survives a new batch.
220/// The initial empty suffix and every copied/reduced position are charged.
221fn prepare_suffix<Q: Quota + ?Sized>(
222    terms: &[(usize, i128)],
223    sure: &[u64],
224    never: &[u64],
225    suffixes: &mut Vec<i128>,
226    work: &mut Meter<'_, Q>,
227) -> Result<(), Failure<Q::Error>> {
228    suffixes.clear();
229    let length = terms.len().checked_add(1).ok_or(Stop::Allocation)?;
230    work.tick()?;
231    suffixes
232        .try_reserve_exact(length)
233        .map_err(|_| Stop::Allocation)?;
234    suffixes.push(0);
235    let mut suffix = 0;
236    for &(condition, weight) in terms.iter().rev() {
237        work.tick()?;
238        if !bit(sure, condition) && !bit(never, condition) {
239            suffix = gcd(suffix, weight, work)?;
240        }
241        suffixes.push(suffix);
242    }
243    Ok(())
244}

246/// Intersect a hypothetical branch's attainable bounds with the required
247/// interval before testing its necessary lattice. Empty intersections refute
248/// directly; divisor zero represents the singleton base, not a modulo operation.
249fn branch_misses(
250    base: i128,
251    maximum: i128,
252    lower: i128,
253    upper: i128,
254    divisor: i128,
255) -> Result<bool, Stop> {
256    let lower = base.max(lower);
257    let upper = maximum.min(upper);
258    if lower > upper {
259        return Ok(true);
260    }
261    misses(base, lower, upper, divisor)
262}

264/// Positive grouped coefficients fit i128, even when they exceed u64. Each
265/// Euclidean step strictly decreases the nonnegative right operand.
266fn gcd<Q: Quota + ?Sized>(
267    mut left: i128,
268    mut right: i128,
269    work: &mut Meter<'_, Q>,
270) -> Result<i128, Failure<Q::Error>> {
271    if left == 0 || left == right {
272        return Ok(right);
273    }
274    while right != 0 {
275        work.tick()?;
276        let remainder = left.checked_rem(right).ok_or(Stop::InvalidProgram)?;
277        left = right;
278        right = remainder;
279    }
280    Ok(left)
281}

283/// Invariant: 0 <= base <= lower <= upper. Distance to the next lattice point
284/// is compared with interval width, without adding to an extreme bound.
285fn misses(base: i128, lower: i128, upper: i128, divisor: i128) -> Result<bool, Stop> {
286    if divisor == 0 {
287        return Ok(base < lower || base > upper);
288    }
289    let offset = lower.checked_sub(base).ok_or(Stop::InvalidProgram)?;
290    let remainder = offset.checked_rem(divisor).ok_or(Stop::InvalidProgram)?;
291    let distance = if remainder == 0 {
292        0
293    } else {
294        divisor - remainder
295    };
296    let width = upper.checked_sub(lower).ok_or(Stop::InvalidProgram)?;
297    Ok(distance > width)
298}
