1//! Lazy distinct-leaf views of shared associative connectives.
2//!
3//! Single-parent chain counters remain the ordinary path. A shared same-kind
4//! child is a compact DAG boundary, not a logical boundary for associativity or
5//! idempotence. The optional view unfolds only those same-kind links. It keeps
6//! no leaf-to-ancestor transpose and no flattened prefix table.
7//!
8//! Preparation marks affected chains in O(C + L), with C chain roots and L
9//! existing chain leaves. A requested view costs O(V + E) for its reachable
10//! same-kind cone, using generation marks, continuation frames and an optional
11//! collected node/leaf vector. Cover authentication stops at its first failure;
12//! neither it nor a witness fold schedules unread sibling edges.
13//! Retained walker scratch is O(N + E + C), with N theory nodes. Reservations are
14//! fallible, each initialized mark and visited node/edge is charged, and the
15//! caller's normal work quota/cancellation contract applies. The scratch never
16//! enters cloned region knowledge; every narrowing clears abandoned dirty work.
17//!
18//! When an inner leaf changes without deciding its shared parent, dirty notices
19//! follow only existing immediate same-kind parent edges. Coalescing bounds the
20//! pending queue by C. Only a required nested witness scans its distinct leaves;
21//! no unrelated flat chain changes its counter algorithm. Repeated notices can
22//! revisit cones, so this trades runtime work for compact retained storage.
23//! Settled roots do not relay possibility notices: ordinary truth events have
24//! already informed their parents and forced any required descendants. Their own
25//! required witness check remains independent of that notification rule.
26//! Enqueue sites apply the same relevance rule after updating operand counters.
27//! A settled root enters the queue only for a remaining expanded unit obligation;
28//! a later root truth event seeds that obligation if it becomes required later.
29//! Frozen closure enables associative witness views for native extrema too;
30//! a masked same-kind child is an opaque falsum. Numeric bound bridges remain
31//! original-only and are never transplanted into a frozen reduct.
32

40const EXPAND: u8 = 1;
41const DIRTY: u8 = 2;
42
43pub(super) fn nested(
44    chains: &[Chain],
45    chain_of: &[Option<NonZeroUsize>],
46    extrema: &super::extrema::Extrema,
47) -> Result<Vec<u8>, Stop> {
48    let has_child = |chain: &Chain| {
49        chain.leaves(extrema).iter().any(|&operand| {
50            chain_of[operand]
51                .is_some_and(|link| chains[chain_position(link)].disjunction == chain.disjunction)
52        })
53    };
54    if !chains.iter().any(has_child) {
55        return Ok(Vec::new());
56    }
57    let mut roles = Vec::new();
58    roles
59        .try_reserve_exact(chains.len())
60        .map_err(|_| Stop::Allocation)?;
61    roles.resize(chains.len(), 0);
62    for (parent, chain) in chains.iter().enumerate() {
63        for &operand in chain.leaves(extrema) {
64            if let Some(link) = chain_of[operand] {
65                let child = chain_position(link);
66                if chains[child].disjunction == chain.disjunction {
67                    roles[parent] |= EXPAND | DIRTY;
68                    roles[child] |= DIRTY;
69                }
70            }
71        }
72    }
73    Ok(roles)
74}

90/// A suspended same-kind chain: children before `next_child` were scheduled in
91/// order; the remaining borrowed operands have not been read. Only IDs and an
92/// offset persist, so no borrow or formula authority enters worker scratch.
93#[derive(Debug)]
94struct Frame {
95    chain: usize,
96    next_child: usize,
97}

99/// A visitor either descends, prunes this node's children, or stops the walk.
100/// Pruning a covered compound is different from stopping at an uncovered leaf.
101enum Visit {
102    Descend,
103    Prune,
104    Stop,
105}

108pub(super) struct Scratch {

119impl Scratch {
120    pub(super) fn retained_bytes(&self) -> u128 {
121        (self.stack.capacity()
122            + self.leaves.capacity()
123            + self.candidates.capacity()
124            + self.dirty.capacity()) as u128
125            * size_of::<usize>() as u128
126            + self.frames.capacity() as u128 * size_of::<Frame>() as u128
127            + self.seen.capacity() as u128 * size_of::<u32>() as u128
128            + self.pending.capacity() as u128 * size_of::<u64>() as u128
129    }

131    pub(super) fn clear(&mut self) {
132        for &chain in &self.dirty {
133            self.pending[chain / 64] &= !(1u64 << (chain % 64));
134        }
135        self.dirty.clear();
136        self.candidates.clear();
137        self.stack.clear();
138        self.frames.clear();
139        self.leaves.clear();
140    }
141
142    pub(super) fn prepare_dirty(&mut self, chains: usize) -> Result<(), Stop> {
143        self.dirty
144            .try_reserve_exact(chains.saturating_sub(self.dirty.len()))
145            .map_err(|_| Stop::Allocation)?;
146        let words = flag_words(chains);
147        if self.pending.len() < words {
148            self.pending
149                .try_reserve_exact(words - self.pending.len())
150                .map_err(|_| Stop::Allocation)?;
151            self.pending.resize(words, 0);
152        }
153        Ok(())
154    }
155
156    pub(super) fn queue_dirty(&mut self, chain: usize) {
157        let (word, flag) = (chain / 64, 1u64 << (chain % 64));
158        if self.pending[word] & flag == 0 {
159            self.pending[word] |= flag;
160            self.dirty.push(chain);
161        }
162    }
163
164    pub(super) fn next_dirty(&mut self) -> Option<usize> {
165        let chain = self.dirty.pop()?;
166        self.pending[chain / 64] &= !(1u64 << (chain % 64));
167        Some(chain)
168    }

170    /// Borrow the ordinary leaves or prepare one expanded view. Every view has
171    /// distinct node IDs in first left-to-right occurrence order. A shared child
172    /// may occur on several paths, but its descendants are expanded only once.
173    pub(super) fn read<'a>(
174        &'a mut self,
175        index: &'a Narrower,
176        chain: usize,
177        frozen: Option<&[bool]>,
178        work: &mut Work<'_>,
179    ) -> Result<&'a [usize], Stop> {
180        if !expands(index, chain) {
181            return Ok(index.chains[chain].leaves(&index.extrema));
182        }
183        // The visitor's begin clears the scratch's leaf vector. Lend its
184        // reusable allocation to the collecting adapter, then restore it on
185        // every outcome, including a refused walk or fallible leaf push.
186        let mut leaves = std::mem::take(&mut self.leaves);
187        leaves.clear();
188        let result = self.visit_expanded(index, chain, frozen, work, |leaf, _| {
189            push(&mut leaves, leaf)?;
190            Ok(ControlFlow::Continue(()))
191        });
192        self.leaves = leaves;
193        result.map(|_| ())?;
194        Ok(&self.leaves)
195    }

197    /// Fold distinct witnesses in the collecting view's traversal order.
198    /// The caller has selected an expanding chain. One continuation frame per
199    /// active same-kind ancestor reads child edges only as they are reached;
200    /// stopping after a witness or two open leaves does not prepare later siblings.
201    /// Node and edge visits are charged here, while the visitor charges its own
202    /// leaf reads. A complete walk retains its node/edge receipt; an early stop
203    /// omits unread edges as well as nodes. First-use marks still cost O(N).
204    /// Frames use O(D) scratch for active depth D, counted at actual capacity.
205    /// Breaks and errors may leave frames; the next walk clears them and advances
206    /// the same generation marks. A fold never allocates a leaf-result vector.
207    fn visit_expanded(
208        &mut self,
209        index: &Narrower,
210        chain: usize,
211        frozen: Option<&[bool]>,
212        work: &mut Work<'_>,
213        mut leaf: impl FnMut(usize, &mut Work<'_>) -> Result<ControlFlow<()>, Stop>,
214    ) -> Result<ControlFlow<()>, Stop> {
215        self.visit_nodes(index, chain, frozen, work, |node, branch, work| {
216            if branch {
217                Ok(Visit::Descend)
218            } else {
219                leaf(node, work).map(|flow| match flow {
220                    ControlFlow::Continue(()) => Visit::Prune,
221                    ControlFlow::Break(()) => Visit::Stop,
222                })
223            }
224        })
225    }
226
227    /// Visit distinct nodes in first-occurrence depth-first order. A branch is
228    /// an active same-kind chain; the visitor may prune it before any child edge
229    /// is read. Generation marks coalesce repeated paths. The one frame stack
230    /// serves witness folds, cover authentication and complete cone collection.
231    /// Every encountered node and consumed edge is charged, including duplicate
232    /// node encounters. Failure leaves only scratch; the next begin discards it.
233    fn visit_nodes(
234        &mut self,
235        index: &Narrower,
236        chain: usize,
237        frozen: Option<&[bool]>,
238        work: &mut Work<'_>,
239        mut visit: impl FnMut(usize, bool, &mut Work<'_>) -> Result<Visit, Stop>,
240    ) -> Result<ControlFlow<()>, Stop> {
241        self.begin(index.parents.len(), work)?;
242        let mut node = index.chains[chain].root;
243        let disjunction = index.chains[chain].disjunction;
244        loop {
245            work.tick()?;
246            if !self.visited(node) {
247                self.seen[node] = self.generation;
248                let inner = index.chain_of[node].map(chain_position).filter(|&inner| {
249                    index.chains[inner].disjunction == disjunction
250                        && frozen.is_none_or(|truth| truth[node])
251                });
252                match visit(node, inner.is_some(), work)? {
253                    Visit::Descend => {
254                        if let Some(inner) = inner {
255                            self.frames.try_reserve(1).map_err(|_| Stop::Allocation)?;
256                            self.frames.push(Frame {
257                                chain: inner,
258                                next_child: 0,
259                            });
260                        }
261                    }
262                    Visit::Prune => {}
263                    Visit::Stop => return Ok(ControlFlow::Break(())),
264                }
265            }
266            let Some(next) = self.next_child(index, work)? else {
267                return Ok(ControlFlow::Continue(()));
268            };
269            node = next;
270        }
271    }
272
273    /// Resume the deepest unfinished chain, discarding completed frames.
274    /// Each advance consumes one previously unread edge; a completed frame is
275    /// popped once. A refusal precedes the edge read and leaves its cursor intact.
276    fn next_child(&mut self, index: &Narrower, work: &mut Work<'_>) -> Result<Option<usize>, Stop> {
277        while let Some(frame) = self.frames.last_mut() {
278            let children = index.chains[frame.chain].leaves(&index.extrema);
279            if frame.next_child == children.len() {
280                self.frames.pop();
281                continue;
282            }
283            work.tick()?;
284            let child = children[frame.next_child];
285            frame.next_child += 1;
286            return Ok(Some(child));
287        }
288        Ok(None)
289    }

307    /// Collect the same-kind cone, including compound conditions and its root.
308    /// A native row can name a cut through this cone, rather than just its
309    /// opaque or fully unfolded boundary. Distinct nodes occur once.
310    pub(super) fn cone(
311        &mut self,
312        index: &Narrower,
313        chain: usize,
314        work: &mut Work<'_>,
315    ) -> Result<(), Stop> {
316        // Keep the collecting allocation observable even after a refused walk.
317        let mut nodes = std::mem::take(&mut self.leaves);
318        nodes.clear();
319        let result = self.visit_nodes(index, chain, None, work, |node, _, _| {
320            push(&mut nodes, node)?;
321            Ok(Visit::Descend)
322        });
323        self.leaves = nodes;
324        result.map(|_| ())
325    }

327    /// Mark the complete same-kind cone without collecting a node sequence.
328    /// Membership-only consumers use the same traversal, generation marks and
329    /// node/edge receipt as `cone`; no result-vector capacity is requested.
330    /// The next walk invalidates these marks, including after a refused prefix.
331    pub(super) fn mark_cone(
332        &mut self,
333        index: &Narrower,
334        chain: usize,
335        work: &mut Work<'_>,
336    ) -> Result<(), Stop> {
337        self.visit_nodes(index, chain, None, work, |_, _, _| Ok(Visit::Descend))
338            .map(|_| ())
339    }

345    /// Every terminal path must meet a caller-approved condition. Stop at such
346    /// a condition before unfolding it: a row may name a compound condition.
347    /// An uncovered opaque endpoint refutes this structural implication only.
348    /// Its unread siblings are neither scheduled nor charged. Complete coverage
349    /// retains the previous node/edge receipt and first-occurrence callback order.
350    pub(super) fn covered_by(
351        &mut self,
352        index: &Narrower,
353        chain: usize,
354        work: &mut Work<'_>,
355        mut covers: impl FnMut(usize, &mut Work<'_>) -> Result<bool, Stop>,
356    ) -> Result<bool, Stop> {
357        self.visit_nodes(index, chain, None, work, |node, branch, work| {
358            Ok(if covers(node, work)? {
359                Visit::Prune
360            } else if branch {
361                Visit::Descend
362            } else {
363                Visit::Stop
364            })
365        })
366        .map(|flow| flow.is_continue())
367    }

369    /// Discover candidate chain roots from one row anchor using the existing
370    /// immediate-parent transpose. Initial parents may use either connective;
371    /// each subsequent path retains that connective. The anchor itself is also
372    /// a candidate when it names a live chain. Results are distinct, with no
373    /// retained row-by-chain or leaf-by-ancestor relation. The caller may prune
374    /// a branch only when no same-kind ancestor needs this requirement: either
375    /// its polarity is irrelevant, or this root already has the absorbing truth
376    /// that ordinary node events propagate to every same-kind ancestor.
377    pub(super) fn parent_chains(
378        &mut self,
379        index: &Narrower,
380        condition: usize,
381        work: &mut Work<'_>,
382        mut relevant: impl FnMut(&Chain) -> bool,
383    ) -> Result<(), Stop> {
384        self.begin(index.parents.len(), work)?;
385        self.candidates.clear();
386        if index.chain_of[condition].is_some() {
387            push(&mut self.stack, condition)?;
388        }
389        for parent in index.logical_parents(condition).rev() {
390            work.tick()?;
391            if index.chain_of[parent].is_some() {
392                push(&mut self.stack, parent)?;
393            }
394        }
395        while let Some(node) = self.stack.pop() {
396            work.tick()?;
397            if self.visited(node) {
398                continue;
399            }
400            self.seen[node] = self.generation;
401            let chain = chain_position(index.chain_of[node].expect("queued live chain"));
402            if !relevant(&index.chains[chain]) {
403                continue;
404            }
405            self.push_candidate(chain)?;
406            for parent in index.logical_parents(node).rev() {
407                work.tick()?;
408                if let Some(link) = index.chain_of[parent]
409                    && index.chains[chain_position(link)].disjunction
410                        == index.chains[chain].disjunction
411                {
412                    push(&mut self.stack, parent)?;
413                }
414            }
415        }
416        Ok(())
417    }
418
419    fn begin(&mut self, nodes: usize, work: &mut Work<'_>) -> Result<(), Stop> {
420        self.stack.clear();
421        self.frames.clear();
422        self.leaves.clear();
423        self.seen
424            .try_reserve_exact(nodes.saturating_sub(self.seen.len()))
425            .map_err(|_| Stop::Allocation)?;
426        while self.seen.len() < nodes {
427            work.tick()?;
428            self.seen.push(0);
429        }
430        if self.generation == u32::MAX {
431            for seen in &mut self.seen {
432                work.tick()?;
433                *seen = 0;
434            }
435            self.generation = 0;
436        }
437        self.generation += 1;
438        Ok(())
439    }
440}

442fn push(values: &mut Vec<usize>, value: usize) -> Result<(), Stop> {
443    values.try_reserve(1).map_err(|_| Stop::Allocation)?;
444    values.push(value);
445    Ok(())

448impl<C: Count> Closure<'_, C> {
449    /// Select a change that can still inform a shared ancestor or establish
450    /// an expanded unit consequence. Counts describe processed events, so callers
451    /// first account for a changed operand. Truth bits may precede their events:
452    /// a newly required root also reaches this operation from `teach_operands`.
453    /// The caller has selected a participating shared chain before consulting
454    /// dynamic truth: `participates`, `expands` or a shared same-kind edge
455    /// establishes that immutable role. Unrelated chains never need this test.
456    /// Frozen falsum roots and unrelated chains have no associative obligation.
457    #[inline]
458    pub(super) fn needs_witness_notification(
459        &self,
460        index: &Narrower,
461        chain: usize,
462        frozen: Option<&[bool]>,
463    ) -> bool {
464        if frozen.is_some() && index.extrema.is_empty() {
465            return false;
466        }
467        debug_assert!(participates(index, chain));
468        let role = index.witness_chains[chain];
469        // A processed witness establishes the required root truth in the same
470        // operand event. Contradictory events stop before reaching this helper;
471        // otherwise that truth persists, so neither relay nor unit work remains.
472        if bit(self.known.witnessed, chain) {
473            return false;
474        }
475        let source = &index.chains[chain];
476        if frozen.is_some_and(|truth| !truth[source.root]) {
477            return false;
478        }
479        let (required, neutral) = if source.disjunction {
480            (&self.known.sure, &self.known.never)
481        } else {
482            (&self.known.never, &self.known.sure)
483        };
484        !bit(neutral, source.root) && (role & EXPAND != 0 || !bit(required, source.root))
485    }
486
487    pub(super) fn witness_changed(
488        &mut self,
489        index: &Narrower,
490        chain: usize,
491        frozen: Option<&[bool]>,
492        work: &mut Work<'_>,
493    ) -> Result<Step, Stop> {
494        work.tick()?;
495        let source = &index.chains[chain];
496        if frozen.is_some_and(|truth| !truth[source.root]) {
497            return Ok(Step::Unchanged);
498        }
499        // Ordinary node events drain first. A settled source has therefore
500        // already informed its parents: an absorbing value settles them, and a
501        // neutral value forces this source's operands. Only an unknown source
502        // must relay a change in its possible witnesses through the shared DAG.
503        if !bit(self.known.sure, source.root) && !bit(self.known.never, source.root) {
504            for parent in index.logical_parents(source.root) {
505                work.tick()?;
506                if let Some(link) = index.chain_of[parent] {
507                    let parent = chain_position(link);
508                    if index.chains[parent].disjunction == source.disjunction
509                        && self.needs_witness_notification(index, parent, frozen)
510                    {
511                        self.lists.witnesses.queue_dirty(parent);
512                    }
513                }
514            }
515        }
516        if !expands(index, chain)
517            || !if source.disjunction {
518                bit(self.known.sure, source.root)
519            } else {
520                bit(self.known.never, source.root)
521            }
522            || bit(self.known.witnessed, chain)
523        {
524            return Ok(Step::Unchanged);
525        }
526        let mut candidate = None;
527        let known = &self.known;
528        let outcome =
529            self.lists
530                .witnesses
531                .visit_expanded(index, chain, frozen, work, |leaf, work| {
532                    work.tick()?;
533                    let (witness, neutral) = if source.disjunction {
534                        (bit(known.sure, leaf), bit(known.never, leaf))
535                    } else {
536                        (bit(known.never, leaf), bit(known.sure, leaf))
537                    };
538                    if witness {
539                        return Ok(ControlFlow::Break(()));
540                    }
541                    if !neutral {
542                        if candidate.is_some() {
543                            return Ok(ControlFlow::Break(()));
544                        }
545                        candidate = Some(leaf);
546                    }
547                    Ok(ControlFlow::Continue(()))
548                })?;
549        if outcome.is_break() {
550            return Ok(Step::Unchanged);
551        }
552        Ok(match candidate {
553            None => Step::Contradiction,
554            Some(leaf) if source.disjunction => self.sure(leaf),
555            Some(leaf) => self.never(leaf),
556        })
557    }
