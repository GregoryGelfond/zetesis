1//! Direct evaluation of a completely checked stratified normal theory.
2//!
3//! Positive dependency components are saturated in dependency order. Negative
4//! literals read only completed predecessors. Original constraints then filter
5//! the unique resulting interpretation; they never provide support. This is a
6//! ground-theory certificate, independent of source classification.

19/// Bounds for checked classification, stratification and direct evaluation.
20#[derive(Clone, Copy, Debug)]
21pub struct StratifiedPlanLimits {
22    /// Signed dependency incidences, counting repeated child occurrences.
23    pub max_dependencies: usize,
24    /// Named simultaneous headers and actual vector capacities. Shared theory,
25    /// allocator overhead and other stack temporaries are excluded; this is not RSS.
26    pub max_bytes: usize,
27    /// Charged classification, graph, component, closure and validation work.
28    pub max_work: u64,
29}
30
31impl Default for StratifiedPlanLimits {
32    fn default() -> Self {
33        Self {
34            max_dependencies: 4_194_304,
35            max_bytes: 256 * 1024 * 1024,
36            max_work: 100_000_000,
37        }
38    }
39}

41/// Explicit resource whose next complete request could not be admitted.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub enum StratifiedResource {
44    /// Signed graph incidences.
45    Dependencies,
46    /// Named simultaneous storage.
47    Bytes,
48    /// Charged primitive work.
49    Work,
50}

52/// An unsupported class or incomplete attempt, never a satisfiability verdict.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54pub enum StratifiedError {
55    /// An asserted root is not an atomic fact, normal atomic-head rule or constraint.
56    UnsupportedRoot {
57        /// Original root node.
58        root: usize,
59    },
60    /// A producer body is outside the conjunction-of-signed-atoms grammar.
61    UnsupportedBody {
62        /// Original producer root.
63        root: usize,
64        /// Original body node.
65        body: usize,
66    },
67    /// A negative dependency belongs to a recursive component.
68    NegativeCycle {
69        /// Negated atom whose truth would depend on this component.
70        atom: usize,
71        /// Original default-negation node.
72        body: usize,
73    },
74    /// Final exact evaluation rejected a producer rather than a constraint.
75    InvalidClosure {
76        /// Original producer root.
77        root: usize,
78    },
79    /// An explicit resource allowance was insufficient.
80    Limit {
81        /// Refused resource.
82        resource: StratifiedResource,
83        /// Complete requested amount.
84        observed: u128,
85        /// Configured ceiling.
86        limit: u128,
87    },
88    /// A checked count cannot represent the required value.
89    Overflow,
90    /// Cancellation, deadline or allocation failure.
91    Stopped(Stop),
92}
93
94impl fmt::Display for StratifiedError {
95    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
96        match self {
97            Self::UnsupportedRoot { root } => {
98                write!(f, "root {root} has no stratified normal plan")
99            }
100            Self::UnsupportedBody { root, body } => {
101                write!(f, "normal root {root} has an unsupported body at {body}")
102            }
103            Self::NegativeCycle { atom, body } => {
104                write!(
105                    f,
106                    "negative dependency from atom {atom} to body {body} is recursive"
107                )
108            }
109            Self::InvalidClosure { root } => {
110                write!(f, "stratified consequences fail producer {root}")
111            }
112            Self::Limit {
113                resource,
114                observed,
115                limit,
116            } => {
117                write!(
118                    f,
119                    "stratified plan {resource:?} needs {observed}; limit is {limit}"
120                )
121            }
122            Self::Overflow => f.write_str("stratified plan arithmetic overflowed"),
123            Self::Stopped(stop) => stop.fmt(f),
124        }
125    }
126}
127impl std::error::Error for StratifiedError {}
128impl From<Stop> for StratifiedError {
129    fn from(stop: Stop) -> Self {
130        Self::Stopped(stop)
131    }
132}

147    /// Outgoing incidences visited from true vertices during evaluation.

155/// A directly evaluated, completely checked stratified normal theory.
156///
157/// Each original producer has one atomic head and a conjunction of atoms,
158/// single default-negated atoms, falsum or canonical truth. Positive recursion
159/// is permitted; recursion through default negation and choice heads are not.
160/// Arbitrary original constraints filter the result after all strata complete.
161/// With no failed constraint, the result is the unique answer set of this
162/// grammar. This certificate does not establish source grounding completeness
163/// or formal refinement of this Rust implementation.
164#[derive(Debug)]
165pub struct StratifiedPlan {
166    consequences: Interpretation,
167    failed_constraint: Option<usize>,
168    statistics: StratifiedPlanStatistics,
169}
170
171impl StratifiedPlan {
172    /// Certify all original roots and compute the stratified consequences once.
173    ///
174    /// # Errors
175    /// Returns class, resource, arithmetic or control refusal. No partial plan
176    /// or answer-set conclusion is returned on failure.
177    pub fn compile(
178        theory: &Theory,
179        limits: StratifiedPlanLimits,
180        cancellation: &Cancellation,
181    ) -> Result<Self, StratifiedError> {
182        Self::compile_accounted(theory, limits, cancellation).result
183    }
184
185    /// Compile with failed-prefix accounting. The iterative component passes
186    /// and queued positive closure take O(atoms + nodes + roots + incidences)
187    /// work and space. Shared bodies are not expanded per producer. Every vertex
188    /// becomes true at most once, and each such outgoing incidence is read once.
189    /// Final original evaluation shares the remaining work and storage allowance.
190    #[must_use]
191    pub fn compile_accounted(
192        theory: &Theory,
193        limits: StratifiedPlanLimits,
194        cancellation: &Cancellation,
195    ) -> StratifiedAttempt {
196        let mut budget = Budget {
197            limits,
198            cancellation,
199            statistics: StratifiedPlanStatistics::default(),
200            current_bytes: 0,
201        };
202        let result =
203            compile::build(theory, &mut budget).map(|(consequences, failed_constraint)| {
204                budget.statistics.retained_bytes = size_of::<Self>() as u128
205                    + consequences.words.capacity() as u128 * size_of::<u64>() as u128;
206                Self {
207                    consequences,
208                    failed_constraint,
209                    statistics: budget.statistics,
210                }
211            });
212        StratifiedAttempt {
213            result,
214            statistics: budget.statistics,
215        }
216    }
217
218    /// Exact original owner. Independently equal theories have different atom IDs.
219    #[must_use]
220    pub fn theory(&self) -> &Theory {
221        self.consequences.theory()
222    }
223
224    /// Completed stratum-by-stratum consequences. A failed constraint makes this
225    /// interpretation unsuitable as an answer of the complete original theory.
226    #[must_use]
227    pub const fn consequences(&self) -> &Interpretation {
228        &self.consequences
229    }
230
231    /// First failed original constraint in root order, after complete evaluation.
232    #[must_use]
233    pub const fn failed_constraint(&self) -> Option<usize> {
234        self.failed_constraint
235    }
236
237    /// Complete construction, evaluation and storage observations.
238    #[must_use]
239    pub const fn statistics(&self) -> StratifiedPlanStatistics {
240        self.statistics
241    }
242}

244/// A completed plan or refusal, with the same attempt's observed prefix.
245#[derive(Debug)]
246pub struct StratifiedAttempt {
247    /// Completed plan or typed refusal.
248    pub result: Result<StratifiedPlan, StratifiedError>,
249    /// Actual admitted work and named capacity, including failed attempts.
250    pub statistics: StratifiedPlanStatistics,
251}

253struct Budget<'a> {
254    limits: StratifiedPlanLimits,
255    cancellation: &'a Cancellation,
256    statistics: StratifiedPlanStatistics,
257    current_bytes: u128,
258}
259
260impl Budget<'_> {
261    fn tick(&mut self) -> Result<(), StratifiedError> {
262        self.cancellation.poll()?;
263        Self::ceiling(
264            StratifiedResource::Work,
265            u128::from(self.statistics.work) + 1,
266            u128::from(self.limits.max_work),
267        )?;
268        self.statistics.work += 1;
269        Ok(())
270    }
271
272    fn dependency(&mut self) -> Result<(), StratifiedError> {
273        let next = self
274            .statistics
275            .dependencies
276            .checked_add(1)
277            .ok_or(StratifiedError::Overflow)?;
278        Self::ceiling(
279            StratifiedResource::Dependencies,
280            next as u128,
281            self.limits.max_dependencies as u128,
282        )?;
283        self.statistics.dependencies = next;
284        Ok(())
285    }
286
287    fn observe_bytes(&mut self, bytes: u128) -> Result<(), StratifiedError> {
288        self.current_bytes = bytes;
289        self.statistics.peak_bytes = self.statistics.peak_bytes.max(bytes);
290        self.check_bytes(bytes)
291    }
292
293    fn check_bytes(&self, bytes: u128) -> Result<(), StratifiedError> {
294        Self::ceiling(
295            StratifiedResource::Bytes,
296            bytes,
297            self.limits.max_bytes as u128,
298        )
299    }
300
301    fn reserve<T>(&mut self, count: usize) -> Result<Vec<T>, StratifiedError> {
302        self.cancellation.poll()?;
303        let requested = self
304            .current_bytes
305            .checked_add(count as u128 * size_of::<T>() as u128)
306            .ok_or(StratifiedError::Overflow)?;
307        self.check_bytes(requested)?;
308        let mut values = Vec::new();
309        values
310            .try_reserve_exact(count)
311            .map_err(|_| Stop::Allocation)?;
312        let actual = self
313            .current_bytes
314            .checked_add(values.capacity() as u128 * size_of::<T>() as u128)
315            .ok_or(StratifiedError::Overflow)?;
316        self.observe_bytes(actual)?;
317        Ok(values)
318    }
319
320    fn filled<T: Copy>(&mut self, count: usize, value: T) -> Result<Vec<T>, StratifiedError> {
321        let mut values = self.reserve(count)?;
322        for _ in 0..count {
323            self.tick()?;
324            values.push(value);
325        }
326        Ok(values)
327    }
328
329    fn release<T>(&mut self, vector: Vec<T>) -> Result<(), StratifiedError> {
330        let bytes = vector.capacity() as u128 * size_of::<T>() as u128;
331        drop(vector);
332        self.current_bytes = self
333            .current_bytes
334            .checked_sub(bytes)
335            .ok_or(StratifiedError::Overflow)?;
336        Ok(())
337    }
338
339    fn ceiling(
340        resource: StratifiedResource,
341        observed: u128,
342        limit: u128,
343    ) -> Result<(), StratifiedError> {
344        if observed > limit {
345            return Err(StratifiedError::Limit {
346                resource,
347                observed,
348                limit,
349            });
350        }
351        Ok(())
352    }
