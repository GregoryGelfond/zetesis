1//! Iterative dependency components; negative edges require strict precedence.
2
3use super::{Budget, StratifiedError, compile::Graph};
4use zetesis_cpu::Stop;
5
6/// One explicit DFS frame; child progress is retained before descending.
7#[derive(Clone, Copy)]
8pub(super) struct Frame {
9    vertex: usize,
10    next: usize,
11}
12
13/// Vertices grouped by a topological ordering of strongly connected components.
14pub(super) struct Schedule {
15    pub vertices: Vec<usize>,
16    pub offsets: Vec<usize>,
17    pub component: Vec<usize>,
18}
19
20impl Schedule {
21    pub fn release(self, budget: &mut Budget<'_>) -> Result<(), StratifiedError> {
22        budget.release(self.vertices)?;
23        budget.release(self.offsets)?;
24        budget.release(self.component)
25    }
26}
27
28pub(super) fn compile(
29    graph: &Graph,
30    atoms: usize,
31    budget: &mut Budget<'_>,
32) -> Result<Schedule, StratifiedError> {
33    let finished = finishing_order(graph, budget)?;
34    let schedule = components(graph, &finished, budget)?;
35    budget.release(finished)?;
36    for source in 0..graph.vertices() {
37        budget.tick()?;
38        for edge in graph.outgoing(source) {
39            budget.tick()?;
40            let earlier = schedule.component[source];
41            let later = schedule.component[edge.target];
42            if earlier > later {
43                return Err(Stop::InvalidProgram.into());
44            }
45            if edge.negative && earlier == later {
46                return Err(StratifiedError::NegativeCycle {
47                    atom: source,
48                    body: edge.target - atoms,
49                });
50            }
51        }
52    }
53    Ok(schedule)
54}

56fn finishing_order(graph: &Graph, budget: &mut Budget<'_>) -> Result<Vec<usize>, StratifiedError> {
57    let mut seen = budget.filled(graph.vertices(), false)?;
58    let mut stack: Vec<Frame> = budget.reserve(graph.vertices())?;
59    let mut finished = budget.reserve(graph.vertices())?;
60    for start in 0..graph.vertices() {
61        budget.tick()?;
62        if seen[start] {
63            continue;
64        }
65        seen[start] = true;
66        stack.push(Frame {
67            vertex: start,
68            next: graph.offsets[start],
69        });
70        while let Some(frame) = stack.last().copied() {
71            budget.tick()?;
72            if frame.next == graph.offsets[frame.vertex + 1] {
73                stack.pop();
74                finished.push(frame.vertex);
75                continue;
76            }
77            let child = graph.edges[frame.next].target;
78            // The frame exists and advances before the child's frame is pushed.
79            stack.last_mut().ok_or(Stop::InvalidProgram)?.next += 1;
80            if !seen[child] {
81                seen[child] = true;
82                stack.push(Frame {
83                    vertex: child,
84                    next: graph.offsets[child],
85                });
86            }
87        }
88    }
89    budget.release(seen)?;
90    budget.release(stack)?;
91    Ok(finished)
92}

94fn components(
95    graph: &Graph,
96    finished: &[usize],
97    budget: &mut Budget<'_>,
98) -> Result<Schedule, StratifiedError> {
99    let mut assigned = budget.filled(graph.vertices(), false)?;
100    // A component coordinate is read only after its vertex has been assigned.
101    let mut component = budget.filled(graph.vertices(), 0_usize)?;
102    let mut vertices = budget.reserve(graph.vertices())?;
103    let capacity = graph
104        .vertices()
105        .checked_add(1)
106        .ok_or(StratifiedError::Overflow)?;
107    let mut offsets = budget.reserve(capacity)?;
108    offsets.push(0);
109    for &start in finished.iter().rev() {
110        budget.tick()?;
111        if assigned[start] {
112            continue;
113        }
114        let id = offsets.len() - 1;
115        let mut next = vertices.len();
116        assigned[start] = true;
117        component[start] = id;
118        vertices.push(start);
119        while let Some(&vertex) = vertices.get(next) {
120            budget.tick()?;
121            next += 1;
122            for &source in graph.incoming(vertex) {
123                budget.tick()?;
124                if !assigned[source] {
125                    assigned[source] = true;
126                    component[source] = id;
127                    vertices.push(source);
128                }
129            }
130        }
131        offsets.push(vertices.len());
132    }
133    budget.release(assigned)?;
134    Ok(Schedule {
135        vertices,
136        offsets,
137        component,
138    })
139}
