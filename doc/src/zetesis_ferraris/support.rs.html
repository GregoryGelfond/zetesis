9/// Bounds for constructing a candidate-only support restriction. Formula
10/// dimensions bound retained nodes, roots and atom-indexed scratch. Work counts
11/// all copied arena cells, node recounts, root visits, head traversal,
12/// construction and final validation.
13/// A root visit includes bounded exact atomic-choice recognition.
14#[derive(Clone, Copy, Debug)]
15pub struct SupportLimits {
16    /// Bounds on the resulting restriction, including copied original nodes.
17    pub admission: AdmissionLimits,
18    /// Cumulative construction operations. Zero is an actual ceiling.
19    pub max_work: u64,
20}
21
22/// A stopped restriction construction, never a logical rejection.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum SupportError {
25    /// Formula shape or allocation could not be admitted.
26    Admission(AdmissionError),
27    /// Cancellation, deadline or the construction work bound was reached.
28    Stopped(Stop),
29}
30
31impl Refusal for SupportError {
32    fn is_resource_refusal(&self) -> bool {
33        match self {
34            Self::Admission(error) => error.is_resource_refusal(),
35            Self::Stopped(stop) => stop.is_resource_refusal(),
36        }
37    }
38
39    fn from_stop(stop: Stop) -> Self {
40        Self::Stopped(stop)
41    }
42}
43
44impl std::fmt::Display for SupportError {
45    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        match self {
47            Self::Admission(error) => error.fmt(formatter),
48            Self::Stopped(error) => error.fmt(formatter),
49        }
50    }
51}
52impl std::error::Error for SupportError {}
53impl From<AdmissionError> for SupportError {
54    fn from(error: AdmissionError) -> Self {
55        Self::Admission(error)
56    }
57}
58impl From<Stop> for SupportError {
59    fn from(error: Stop) -> Self {
60        Self::Stopped(error)
61    }
62}

64/// A complete construction or conservative refusal, with work retained on every
65/// path. `None` means an asserted head lies outside the certified grammar, or
66/// no ordinary asserted head contains a disjunction or an ordinal witness cut. Choices alone
67/// do not trigger this optional construction.
68#[derive(Debug)]
69pub struct SupportAttempt {
70    /// Separate classical restriction over exactly the original atom indices.
71    /// Its identity differs from the original, which remains the reduct subject.
72    pub result: Result<Option<Theory>, SupportError>,
73    /// Operations consumed, including a shape refusal or failed construction.
74    pub work: u64,
75}
76
77/// Construct necessary supportedness for a complete mixed disjunctive theory.
78///
79/// Every asserted root must be a disjunction of atoms (possibly with falsum),
80/// including an ordinal cut's qualifying witnesses,
81/// an exact atomic choice `a or not a`, an implication with either consequent,
82/// a default negation, or falsum. Bodies may be arbitrary formulas. Choices
83/// accept either operand order and separate occurrences of the same semantic
84/// atom; richer or cross-atom alternatives decline the complete certificate.
85/// Repeated ordinary head atoms count once, including shared DAGs. A theory
86/// without an ordinary disjunction or ordinal witness cut in a head also declines; choices
87/// alone do not introduce this extra construction path.
88///
89/// For each atom `a`, require a rule whose body is true and whose other distinct
90/// heads are false whenever `a` is true, or an atomic choice for `a` whose body
91/// is true. An unconditional choice supplies true support; its chosen atom is
92/// not a positive body premise. This condition is necessary: otherwise
93/// removing `a` preserves every original root's frozen reduct, contradicting
94/// minimality. Independent rules can support several heads simultaneously.
95/// Neither ranks nor head exclusivity across the whole program are assumed.
96///
97/// The returned theory is only an outer-query restriction. It must not replace
98/// the original theory or be frozen in its place. Recognition and construction
99/// are iterative; scratch is linear in original nodes, operand occurrences,
100/// atoms and the largest distinct head. The complete original copy also retains
101/// every physical arena cell, including unused numeric storage. Work counts those
102/// copies, their node recount and the final admission's independent scans.
103#[must_use]
104pub fn support_restriction(
105    theory: &Theory,
106    limits: SupportLimits,
107    cancellation: &Cancellation,
108) -> SupportAttempt {
109    let metered = cancellation.metered(limits.max_work, |meter| {
110        Builder {
111            limits,
112            meter,
113            nodes: FormulaNodes::default(),
114        }
115        .build(theory)
116    });
117    SupportAttempt {
118        result: metered.result,
119        work: metered.charged,
120    }
121}

123/// One construction's limits, meter and the restriction under construction.
124struct Builder<'a, 'c> {
125    limits: SupportLimits,
126    meter: &'a mut Meter<'c>,
127    nodes: FormulaNodes,
128}
129
130impl Builder<'_, '_> {
131    fn tick(&mut self) -> Result<(), SupportError> {
132        self.meter
133            .charge()
134            .map_err(|error| error.into_stop().into())
135    }
136
137    fn push(&mut self, node: NodeView<'_>) -> Result<usize, SupportError> {
138        self.tick()?;
139        match node {
140            NodeView::And(operands) | NodeView::Or(operands) => {
141                for _ in operands {
142                    self.tick()?;
143                }
144            }
145            NodeView::Threshold(row) => {
146                for _ in row.conditions {
147                    self.tick()?;
148                }
149            }
150            NodeView::ExtremumCut(cut) => {
151                for _ in cut.row.conditions {
152                    self.tick()?;
153                }
154            }
155            NodeView::Implies(_, _) => {
156                self.tick()?;
157                self.tick()?;
158            }
159            NodeView::Atom(_) | NodeView::False => {}
160        }
161        let mut transaction = self.nodes.transaction();
162        let index = transaction.push(
163            node,
164            self.limits.admission.max_nodes,
165            self.limits.admission.max_operands,
166        )?;
167        transaction.commit();
168        Ok(index)
169    }
170
171    fn ordinary_operands(
172        &mut self,
173        operands: &[usize],
174        ordinary: &[bool],
175    ) -> Result<bool, SupportError> {
176        let mut all = true;
177        for &child in operands {
178            self.tick()?;
179            all &= ordinary[child];
180        }
181        Ok(all)
182    }
183
184    fn ordinary_cut(
185        &mut self,
186        cut: ExtremumCut<'_>,
187        ordinary: &[bool],
188    ) -> Result<bool, SupportError> {
189        if let Some(truth) = cut.endpoint_truth() {
190            return Ok(!truth);
191        }
192        let mut all = true;
193        for (&child, &rank) in cut.row.conditions.iter().zip(cut.row.ranks) {
194            self.tick()?;
195            if cut.qualifies(rank) {
196                all &= ordinary[child];
197            }
198        }
199        Ok(all)
200    }
201
202    fn applicable(&mut self, theory: &Theory) -> Result<bool, SupportError> {
203        self.meter.poll()?;
204        if theory.atom_count() > self.limits.admission.max_atoms
205            || theory.atom_count() > self.limits.admission.max_roots
206            || theory.nodes().len() > self.limits.admission.max_nodes
207            || theory.parts().occurrences() > self.limits.admission.max_operands
208            || theory.operands().len() > self.limits.admission.max_operands
209            || theory.parts().conditions().len() > self.limits.admission.max_operands
210            || theory.parts().numeric_values().len() > self.limits.admission.max_operands
211            || theory.parts().rows().len() > self.limits.admission.max_nodes
212            || theory.parts().thresholds().len() > self.limits.admission.max_nodes
213            || theory.parts().extremum_cuts().len() > self.limits.admission.max_nodes
214        {
215            return Err(AdmissionError::Limit.into());
216        }
217        let mut ordinary = reserve(theory.nodes().len())?;
218        for index in 0..theory.view().len() {
219            self.tick()?;
220            ordinary.push(match theory.view().node(index)? {
221                NodeView::Atom(_) | NodeView::False => true,
222                NodeView::Or(operands) => self.ordinary_operands(operands, &ordinary)?,
223                NodeView::ExtremumCut(cut) => self.ordinary_cut(cut, &ordinary)?,
224                _ => false,
225            });
226        }
227        // Inspect every root before constructing any restriction. A selected
228        // subset of producers cannot justify an atom's lack of other support.
229        let mut disjunctive = false;
230        for &root in theory.roots() {
231            self.tick()?;
232            let head = match theory.view().node(root)? {
233                NodeView::Implies(_, head) => head,
234                _ => root,
235            };
236            if !ordinary[head] && crate::atomic_choice::atom(theory, head).is_none() {
237                return Ok(false);
238            }
239            disjunctive |= ordinary[head]
240                && matches!(
241                    theory.view().node(head)?,
242                    NodeView::Or(_) | NodeView::ExtremumCut(_)
243                );
244        }
245        Ok(disjunctive)
246    }
247
248    fn build(&mut self, theory: &Theory) -> Result<Option<Theory>, SupportError> {
249        if !self.applicable(theory)? {
250            return Ok(None);
251        }
252        for _ in 0..theory.parts().copy_work() {
253            self.tick()?;
254        }
255        self.nodes = FormulaNodes::new(theory.parts().try_clone()?);
256        let falsum = self.push(NodeView::False)?;
257        let verum = self.push(NodeView::Implies(falsum, falsum))?;
258        let mut support = self.filled(theory.atom_count(), falsum)?;
259        let mut visited_nodes = self.filled(theory.nodes().len(), None)?;
260        let mut visited_atoms = self.filled(theory.atom_count(), None)?;
261        let mut stack = reserve(theory.nodes().len())?;
262        let mut heads = reserve(theory.atom_count())?;
263        let mut prefix = reserve(theory.atom_count())?;
264        for (ordinal, &root) in theory.roots().iter().enumerate() {
265            self.tick()?;
266            let (body, head) = match theory.view().node(root)? {
267                NodeView::Implies(body, head) => (body, head),
268                _ => (verum, root),
269            };
270            if let Some(atom) = crate::atomic_choice::atom(theory, head) {
271                // A selected choice head needs its original body's permission,
272                // independently of any other ordinary producer's true heads.
273                support[atom] = self.push(NodeView::Or(&[support[atom], body]))?;
274                continue;
275            }
276            heads.clear();
277            stack.clear();
278            stack.push(head);
279            while let Some(node) = stack.pop() {
280                self.tick()?;
281                if visited_nodes[node] == Some(ordinal) {
282                    continue;
283                }
284                visited_nodes[node] = Some(ordinal);
285                match theory.view().node(node)? {
286                    NodeView::Atom(atom) if visited_atoms[atom] != Some(ordinal) => {
287                        visited_atoms[atom] = Some(ordinal);
288                        heads.push(atom);
289                    }
290                    NodeView::Or(operands) => {
291                        // Each original node expands once per root; the pending
292                        // stack is bounded by admitted operand occurrences.
293                        stack
294                            .try_reserve(operands.len())
295                            .map_err(|_| AdmissionError::Allocation)?;
296                        for &child in operands.iter().rev() {
297                            self.tick()?;
298                            stack.push(child);
299                        }
300                    }
301                    NodeView::ExtremumCut(cut) if cut.endpoint_truth().is_none() => {
302                        stack
303                            .try_reserve(cut.row.conditions.len())
304                            .map_err(|_| AdmissionError::Allocation)?;
305                        for (&child, &rank) in cut.row.conditions.iter().zip(cut.row.ranks).rev() {
306                            self.tick()?;
307                            if cut.qualifies(rank) {
308                                stack.push(child);
309                            }
310                        }
311                    }
312                    _ => {}
313                }
314            }
315            prefix.clear();
316            let mut before = body;
317            for &atom in &heads {
318                self.tick()?;
319                prefix.push(before);
320                let atom_node = self.push(NodeView::Atom(atom))?;
321                let negative = self.push(NodeView::Implies(atom_node, falsum))?;
322                before = self.push(NodeView::And(&[before, negative]))?;
323            }
324            let mut after = verum;
325            for (&atom, &before) in heads.iter().zip(&prefix).rev() {
326                self.tick()?;
327                let witness = self.push(NodeView::And(&[before, after]))?;
328                support[atom] = self.push(NodeView::Or(&[support[atom], witness]))?;
329                let atom_node = self.push(NodeView::Atom(atom))?;
330                let negative = self.push(NodeView::Implies(atom_node, falsum))?;
331                after = self.push(NodeView::And(&[negative, after]))?;
332            }
333        }
334        let mut roots = reserve(theory.atom_count())?;
335        for (atom, body) in support.into_iter().enumerate() {
336            self.tick()?;
337            let atom = self.push(NodeView::Atom(atom))?;
338            roots.push(self.push(NodeView::Implies(atom, body))?);
339        }
340        let admission = std::mem::take(&mut self.nodes).prepare_admission(
341            theory.atom_count(),
342            roots,
343            self.limits.admission,
344        );
345        // Charge the exact retained validation route, including every numeric
346        // row and descriptor whenever this copy requires full admission.
347        for _ in 0..admission.work() {
348            self.tick()?;
349        }
350        admission.admit().map(Some).map_err(SupportError::from)
351    }
352
353    fn filled<T: Copy>(&mut self, count: usize, value: T) -> Result<Vec<T>, SupportError> {
354        let mut values = reserve(count)?;
355        for _ in 0..count {
356            self.tick()?;
357            values.push(value);
358        }
359        Ok(values)
360    }
361}
362
363fn reserve<T>(count: usize) -> Result<Vec<T>, SupportError> {
364    let mut vector = Vec::new();
365    vector
366        .try_reserve_exact(count)
367        .map_err(|_| AdmissionError::Allocation)?;
368    Ok(vector)
