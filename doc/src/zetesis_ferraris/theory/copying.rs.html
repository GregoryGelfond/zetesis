8pub(super) fn copy<T: Copy>(source: &[T]) -> Result<Vec<T>, AdmissionError> {
9    let mut result = Vec::new();
10    result
11        .try_reserve_exact(source.len())
12        .map_err(|_| AdmissionError::Allocation)?;
13    result.extend_from_slice(source);
14    Ok(result)
15}

21impl FormulaParts {
22    /// Copy all seven arenas fallibly, preserving sharing and absolute child IDs.
23    /// The copy is raw storage and carries no admission or original owner identity.
24    /// Work and temporary payload are linear in all stored cells, including unused
25    /// cells. [`Self::copy_work`] reports the complete logical copy/recount charge.
26    /// Actual destination capacities may differ from the source capacities.
27    ///
28    /// # Errors
29    /// Refuses failed reservation or overflow in declared logical occurrences.
30    pub fn try_clone(&self) -> Result<Self, AdmissionError> {
31        self.view().try_to_parts()
32    }
33
34    /// Complete logical work for [`Self::try_clone`]: one visit per copied cell
35    /// in all seven arenas, then one visit per node to recount child occurrences.
36    /// Shared numeric entries are copied once, irrespective of how many nodes
37    /// reference their row. Unreferenced arena cells still contribute.
38    ///
39    /// This allocation-free measurement is a full-attempt precharge; an early
40    /// refusal may perform less work. It excludes fixed reservation bookkeeping,
41    /// allocator work, hardware instructions and topology/numeric admission.
42    /// It neither checks dimensions nor measures bytes or retained capacity.
43    #[must_use]
44    pub fn copy_work(&self) -> u128 {
45        2 * self.nodes.len() as u128
46            + self.operands.len() as u128
47            + self.numeric.conditions.len() as u128
48            + self.numeric.values.len() as u128
49            + self.numeric.rows.len() as u128
50            + self.numeric.thresholds.len() as u128
51            + self.numeric.cuts.len() as u128
52    }
53
54    /// Bytes occupied by the lengths of all seven vector payloads.
55    /// Excludes vector headers, this owner, allocator overhead and roots.
56    #[must_use]
57    pub fn payload_bytes(&self) -> u128 {
58        bytes::<Node>(self.nodes.len())
59            + bytes::<usize>(self.operands.len())
60            + bytes::<usize>(self.numeric.conditions.len())
61            + bytes::<u64>(self.numeric.values.len())
62            + bytes::<NumericRow>(self.numeric.rows.len())
63            + bytes::<ThresholdDescriptor>(self.numeric.thresholds.len())
64            + bytes::<ExtremumCutDescriptor>(self.numeric.cuts.len())
65    }
66
67    /// Bytes retained by the actual capacities of all seven vector allocations.
68    /// Excludes vector headers, this owner, allocator overhead and roots.
69    #[must_use]
70    pub fn retained_bytes(&self) -> u128 {
71        bytes::<Node>(self.nodes.capacity())
72            + bytes::<usize>(self.operands.capacity())
73            + bytes::<usize>(self.numeric.conditions.capacity())
74            + bytes::<u64>(self.numeric.values.capacity())
75            + bytes::<NumericRow>(self.numeric.rows.capacity())
76            + bytes::<ThresholdDescriptor>(self.numeric.thresholds.capacity())
77            + bytes::<ExtremumCutDescriptor>(self.numeric.cuts.capacity())
78    }
79}

81impl FormulaView<'_> {
82    /// Fallibly copy the complete raw owner, preserving row sharing and child IDs.
83    /// This includes unreferenced cells and established numeric shape validation;
84    /// no topology frontier or admitted-theory identity is copied. The numeric
85    /// cache is safe because every node, row, descriptor and child ID is identical.
86    /// Work and payload are linear in all arena lengths.
87    ///
88    /// # Errors
89    /// Refuses failed reservations or overflow in declared logical occurrences.
90    pub(crate) fn try_to_parts(self) -> Result<FormulaParts, AdmissionError> {
91        let nodes = copy(self.nodes)?;
92        let operands = copy(self.operands)?;
93        let numeric = self.numeric.try_clone()?;
94        let occurrences = super::count_occurrences(&nodes, &numeric)?;
95        Ok(FormulaParts {
96            nodes,
97            operands,
98            numeric,
99            occurrences,
100        })
101    }
102
103    /// Copy every arena, remapping absolute child IDs through `children`.
104    ///
105    /// The node positions and atom IDs remain unchanged. All Boolean operand and
106    /// condition cells, including unused raw cells, require a map entry. Descriptor
107    /// and row indices retain their sharing. The result is raw remapping input:
108    /// remapped children need not precede these local node positions. In particular,
109    /// detached suffixes may map their external original child IDs into a new owner.
110    /// Work is linear in stored nodes and arena cells; one complete copy is retained.
111    ///
112    /// # Errors
113    /// Refuses a missing map entry, failed reservation or occurrence overflow.
114    pub fn try_remap(self, children: &[usize]) -> Result<FormulaParts, AdmissionError> {
115        let mut result = self.try_to_parts()?;
116        result.numeric.invalidate_validation();
117        let child = |id: usize| children.get(id).copied().ok_or(AdmissionError::Edge);
118        for node in &mut result.nodes {
119            node.0 = match node.0 {
120                Storage::Implies(left, right) => Storage::Implies(child(left)?, child(right)?),
121                Storage::AndPair([left, right]) => Storage::AndPair([child(left)?, child(right)?]),
122                Storage::OrPair([left, right]) => Storage::OrPair([child(left)?, child(right)?]),
123                other => other,
124            };
125        }
126        for id in result
127            .operands
128            .iter_mut()
129            .chain(&mut result.numeric.conditions)
130        {
131            *id = child(*id)?;
132        }
133        Ok(result)
134    }
135}

137impl FormulaTransaction<'_> {
138    /// Current appended payload across all seven arenas, excluding retained capacity.
139    /// Includes nodes, both numeric columns and row and numeric descriptors.
140    #[must_use]
141    pub fn appended_payload_bytes(&self) -> u128 {
142        bytes::<Node>(self.parts.nodes.len() - self.first)
143            + bytes::<usize>(self.parts.operands.len() - self.operand_start)
144            + bytes::<usize>(self.parts.numeric.conditions.len() - self.numeric_start.conditions)
145            + bytes::<u64>(self.parts.numeric.values.len() - self.numeric_start.values)
146            + bytes::<NumericRow>(self.parts.numeric.rows.len() - self.numeric_start.rows)
147            + bytes::<ThresholdDescriptor>(
148                self.parts.numeric.thresholds.len() - self.numeric_start.thresholds,
149            )
150            + bytes::<ExtremumCutDescriptor>(
151                self.parts.numeric.cuts.len() - self.numeric_start.cuts,
152            )
153    }
154
155    /// Physical appended cells across all seven arenas, for complete detachment work.
156    /// A shared row is counted once, unlike logical threshold child occurrences.
157    #[must_use]
158    pub fn appended_cells(&self) -> u128 {
159        (self.parts.nodes.len() - self.first) as u128
160            + (self.parts.operands.len() - self.operand_start) as u128
161            + (self.parts.numeric.conditions.len() - self.numeric_start.conditions) as u128
162            + (self.parts.numeric.values.len() - self.numeric_start.values) as u128
163            + (self.parts.numeric.rows.len() - self.numeric_start.rows) as u128
164            + (self.parts.numeric.thresholds.len() - self.numeric_start.thresholds) as u128
165            + (self.parts.numeric.cuts.len() - self.numeric_start.cuts) as u128
166    }
167}
