8fn sum(left: usize, right: usize, maximum: usize) -> Result<usize, AdmissionError> {
9    left.checked_add(right)
10        .filter(|&total| total <= maximum)
11        .ok_or(AdmissionError::Limit)
12}

14fn reserve<T>(target: &mut Vec<T>, additional: usize) -> Result<(), AdmissionError> {
15    target
16        .try_reserve(additional)
17        .map_err(|_| AdmissionError::Allocation)
18}

20fn reserve_numeric(
21    target: &mut super::super::numeric::NumericStorage,
22    source: &super::super::numeric::NumericStorage,
23) -> Result<(), AdmissionError> {
24    reserve(&mut target.conditions, source.conditions.len())?;
25    reserve(&mut target.values, source.values.len())?;
26    reserve(&mut target.rows, source.rows.len())?;
27    reserve(&mut target.thresholds, source.thresholds.len())?;
28    reserve(&mut target.cuts, source.cuts.len())
29}

31fn relocate(node: Node, operands: usize, thresholds: usize, cuts: usize) -> Node {
32    let span = |span: OperandSpan| OperandSpan {
33        start: span.start + operands,
34        length: span.length,
35    };
36    Node(match node.0 {
37        Storage::AndSpan(row) => Storage::AndSpan(span(row)),
38        Storage::OrSpan(row) => Storage::OrSpan(span(row)),
39        Storage::Threshold(descriptor) => Storage::Threshold(descriptor + thresholds),
40        Storage::ExtremumCut(descriptor) => Storage::ExtremumCut(descriptor + cuts),
41        other => other,
42    })
43}

45impl FormulaTransaction<'_> {
46    /// Append complete raw storage, preserving shared rows and absolute child IDs.
47    ///
48    /// Internal Boolean-span, numeric-row and numeric-descriptor offsets are
49    /// relocated into this owner. Each child must precede its destination node;
50    /// source node positions need not equal those absolute IDs. Atom admission stays
51    /// with Theory. All seven source arenas, including unused cells, are copied once.
52    /// Validate and reserve every destination before publishing any cell. On error
53    /// lengths/frontier are unchanged, but successfully reserved capacity remains.
54    /// Work covers source nodes, logical edges, numeric rows and their entries,
55    /// descriptors and copied cells; retained payload is all source cells. This
56    /// operation does not allocate a second complete source copy or root vector.
57    /// Numeric growth also validates an unchecked destination numeric prefix once,
58    /// scanning its nodes, rows and entries, and descriptors before appending.
59    /// Committed immutable prefixes reuse this cache; rollback restores it.
60    ///
61    /// # Errors
62    /// Refuses invalid source storage or destination topology, numeric shape or
63    /// total overflow, inclusive dimensions or failed reservations.
64    pub fn append(
65        &mut self,
66        source: FormulaView<'_>,
67        max_nodes: usize,
68        max_operands: usize,
69    ) -> Result<Range<usize>, AdmissionError> {
70        if !source.numeric.is_empty() {
71            self.parts.numeric.validate_for_append(&self.parts.nodes)?;
72        }
73        let first = self.parts.nodes.len();
74        let end = sum(first, source.len(), max_nodes)?;
75        sum(
76            self.parts.operands.len(),
77            source.operands.len(),
78            max_operands,
79        )?;
80        let numeric = &self.parts.numeric;
81        if numeric.conditions.len() != numeric.values.len() {
82            return Err(AdmissionError::NumericShape);
83        }
84        sum(
85            numeric.conditions.len(),
86            source.numeric.conditions.len(),
87            max_operands,
88        )?;
89        sum(
90            numeric.values.len(),
91            source.numeric.values.len(),
92            max_operands,
93        )?;
94        sum(numeric.rows.len(), source.numeric.rows.len(), max_nodes)?;
95        sum(
96            numeric.thresholds.len(),
97            source.numeric.thresholds.len(),
98            max_nodes,
99        )?;
100        sum(numeric.cuts.len(), source.numeric.cuts.len(), max_nodes)?;
101        let occurrences = sum(
102            self.parts.occurrences,
103            count_occurrences(source.nodes, source.numeric)?,
104            max_operands,
105        )?;
106        for index in 0..source.len() {
107            validate_children(first + index, source.node(index)?)?;
108        }
109        source.numeric.validate()?;
110        reserve(&mut self.parts.nodes, source.nodes.len())?;
111        reserve(&mut self.parts.operands, source.operands.len())?;
112        let numeric = &mut self.parts.numeric;
113        reserve_numeric(numeric, source.numeric)?;
114        let operands = self.parts.operands.len();
115        let conditions = numeric.conditions.len();
116        let rows = numeric.rows.len();
117        let thresholds = numeric.thresholds.len();
118        let cuts = numeric.cuts.len();
119        self.parts.nodes.extend(
120            source
121                .nodes
122                .iter()
123                .copied()
124                .map(|node| relocate(node, operands, thresholds, cuts)),
125        );
126        self.parts.operands.extend_from_slice(source.operands);
127        numeric
128            .conditions
129            .extend_from_slice(&source.numeric.conditions);
130        numeric.values.extend_from_slice(&source.numeric.values);
131        numeric
132            .rows
133            .extend(source.numeric.rows.iter().map(|row| NumericRow {
134                start: row.start + conditions,
135                length: row.length,
136                domain: row.domain,
137            }));
138        numeric
139            .thresholds
140            .extend(
141                source
142                    .numeric
143                    .thresholds
144                    .iter()
145                    .map(|threshold| ThresholdDescriptor {
146                        row: threshold.row + rows,
147                        bound: threshold.bound,
148                    }),
149            );
150        numeric
151            .cuts
152            .extend(source.numeric.cuts.iter().map(|cut| ExtremumCutDescriptor {
153                row: cut.row + rows,
154                rule: cut.rule,
155            }));
156        self.parts.occurrences = occurrences;
157        if *self.validated == first {
158            *self.validated = end;
159        }
160        Ok(first..end)
161    }
162}
