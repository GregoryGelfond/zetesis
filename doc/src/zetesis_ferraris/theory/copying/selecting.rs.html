6fn absent_map(length: usize) -> Result<Vec<Option<usize>>, AdmissionError> {
7    let mut map = Vec::new();
8    map.try_reserve_exact(length)
9        .map_err(|_| AdmissionError::Allocation)?;
10    map.resize(length, None);
11    Ok(map)
12}

24fn add(left: usize, right: usize) -> Result<usize, AdmissionError> {
25    left.checked_add(right).ok_or(AdmissionError::Limit)

28fn exact<T>(target: &mut Vec<T>, count: usize) -> Result<(), AdmissionError> {

30        .try_reserve_exact(count)
31        .map_err(|_| AdmissionError::Allocation)

34struct Selection<'a> {
35    source: FormulaView<'a>,
36    children: &'a [Option<usize>],
37    result: FormulaParts,
38    rows: Vec<Option<usize>>,
39    thresholds: Vec<Option<usize>>,
40    cuts: Vec<Option<usize>>,
41}
42
43impl<'a> Selection<'a> {
44    fn new(source: FormulaView<'a>, children: &'a [Option<usize>]) -> Result<Self, AdmissionError> {
45        if source.len() != children.len() {
46            return Err(AdmissionError::Edge);
47        }
48        let mut selection = Self {
49            source,
50            children,
51            result: FormulaParts::default(),
52            rows: absent_map(source.numeric.rows.len())?,
53            thresholds: absent_map(source.numeric.thresholds.len())?,
54            cuts: absent_map(source.numeric.cuts.len())?,
55        };
56        selection.reserve_output()?;
57        Ok(selection)
58    }
59
60    fn reserve_output(&mut self) -> Result<(), AdmissionError> {
61        let mut sizes = Sizes::default();
62        for (index, selected) in self.children.iter().enumerate() {
63            if selected.is_none() {
64                continue;
65            }
66            sizes.nodes = add(sizes.nodes, 1)?;
67            match self.source.node(index)? {
68                NodeView::And(row) | NodeView::Or(row) if row.len() >= 3 => {
69                    sizes.operands = add(sizes.operands, row.len())?;
70                }
71                _ => {}
72            }
73            let row = match self.source.nodes[index].0 {
74                Storage::Threshold(descriptor) => {
75                    if self.thresholds[descriptor].is_none() {
76                        self.thresholds[descriptor] = Some(0);
77                        sizes.thresholds = add(sizes.thresholds, 1)?;
78                    }
79                    Some(self.source.numeric.thresholds[descriptor].row)
80                }
81                Storage::ExtremumCut(descriptor) => {
82                    if self.cuts[descriptor].is_none() {
83                        self.cuts[descriptor] = Some(0);
84                        sizes.cuts = add(sizes.cuts, 1)?;
85                    }
86                    Some(self.source.numeric.cuts[descriptor].row)
87                }
88                _ => None,
89            };
90            if let Some(row) = row
91                && self.rows[row].is_none()
92            {
93                self.rows[row] = Some(0);
94                sizes.rows = add(sizes.rows, 1)?;
95                sizes.conditions = add(sizes.conditions, self.source.numeric.rows[row].length)?;
96            }
97        }
98        exact(&mut self.result.nodes, sizes.nodes)?;
99        exact(&mut self.result.operands, sizes.operands)?;
100        exact(&mut self.result.numeric.conditions, sizes.conditions)?;
101        exact(&mut self.result.numeric.values, sizes.conditions)?;
102        exact(&mut self.result.numeric.rows, sizes.rows)?;
103        exact(&mut self.result.numeric.thresholds, sizes.thresholds)?;
104        exact(&mut self.result.numeric.cuts, sizes.cuts)?;
105        self.rows.fill(None);
106        self.thresholds.fill(None);
107        self.cuts.fill(None);
108        Ok(())
109    }
110
111    fn child(&self, id: usize) -> Result<usize, AdmissionError> {
112        self.children
113            .get(id)
114            .copied()
115            .flatten()
116            .ok_or(AdmissionError::Edge)
117    }
118
119    fn group(&mut self, row: &[usize], conjunction: bool) -> Result<Node, AdmissionError> {
120        if let [left, right] = row {
121            let pair = [self.child(*left)?, self.child(*right)?];
122            return Ok(if conjunction {
123                Node::and_pair(pair)
124            } else {
125                Node::or_pair(pair)
126            });
127        }
128        let span = OperandSpan {
129            start: self.result.operands.len(),
130            length: row.len(),
131        };
132        for &id in row {
133            self.result.operands.push(self.child(id)?);
134        }
135        Ok(if conjunction {
136            Node::and_span(span)
137        } else {
138            Node::or_span(span)
139        })
140    }
141
142    fn threshold(&mut self, descriptor: usize) -> Result<Node, AdmissionError> {
143        let threshold = *self
144            .source
145            .numeric
146            .thresholds
147            .get(descriptor)
148            .ok_or(AdmissionError::Span)?;
149        if let Some(mapped) = self.thresholds[descriptor] {
150            return Ok(Node::threshold(mapped));
151        }
152        self.source.numeric.view(descriptor)?;
153        let row = self.row(threshold.row)?;
154        let mapped = self.result.numeric.thresholds.len();
155        self.result.numeric.thresholds.push(ThresholdDescriptor {
156            row,
157            bound: threshold.bound,
158        });
159        self.thresholds[descriptor] = Some(mapped);
160        Ok(Node::threshold(mapped))
161    }
162
163    fn row(&mut self, index: usize) -> Result<usize, AdmissionError> {
164        if let Some(mapped) = self.rows[index] {
165            return Ok(mapped);
166        }
167        let original = self.source.numeric.rows[index];
168        let (conditions, values) = self.source.numeric.columns(original)?;
169        let mapped = self.result.numeric.rows.len();
170        self.result.numeric.rows.push(NumericRow {
171            start: self.result.numeric.conditions.len(),
172            length: original.length,
173            domain: original.domain,
174        });
175        for &id in conditions {
176            self.result.numeric.conditions.push(self.child(id)?);
177        }
178        self.result.numeric.values.extend_from_slice(values);
179        self.rows[index] = Some(mapped);
180        Ok(mapped)
181    }
182
183    fn cut(&mut self, descriptor: usize) -> Result<Node, AdmissionError> {
184        if let Some(mapped) = self.cuts[descriptor] {
185            return Ok(Node::extremum_cut(mapped));
186        }
187        self.source.numeric.cut_view(descriptor)?;
188        let original = self.source.numeric.cuts[descriptor];
189        let row = self.row(original.row)?;
190        let mapped = self.result.numeric.cuts.len();
191        self.result.numeric.cuts.push(ExtremumCutDescriptor {
192            row,
193            rule: original.rule,
194        });
195        self.cuts[descriptor] = Some(mapped);
196        Ok(Node::extremum_cut(mapped))
197    }
198
199    fn append(&mut self, index: usize) -> Result<(), AdmissionError> {
200        let node = match self.source.nodes[index].0 {
201            Storage::Atom(atom) => Node::atom(atom),
202            Storage::False => Node::falsum(),
203            Storage::Implies(left, right) => Node::implies(self.child(left)?, self.child(right)?),
204            Storage::AndPair([left, right]) => {
205                Node::and_pair([self.child(left)?, self.child(right)?])
206            }
207            Storage::OrPair([left, right]) => {
208                Node::or_pair([self.child(left)?, self.child(right)?])
209            }
210            Storage::AndSpan(span) => {
211                self.group(super::super::read_span(self.source.operands, span)?, true)?
212            }
213            Storage::OrSpan(span) => {
214                self.group(super::super::read_span(self.source.operands, span)?, false)?
215            }
216            Storage::Threshold(descriptor) => self.threshold(descriptor)?,
217            Storage::ExtremumCut(descriptor) => self.cut(descriptor)?,
218        };
219        self.result.nodes.push(node);
220        Ok(())
221    }
222}

224impl FormulaView<'_> {
225    /// Copy selected nodes in original order, preserving shared numeric rows.
226    ///
227    /// `children` has one entry per source node. None omits that node; Some selects
228    /// it and supplies its absolute ID for remapping every selected child reference.
229    /// Output positions remain the selected source order, independently of the
230    /// supplied IDs, so the raw result may be appended into another owner's prefix.
231    /// Every selected child needs a selected map entry, including zero-weight contributions and constant ordinal cuts.
232    /// Unreferenced arena cells are discarded. A first pass counts selected cells,
233    /// then each destination reserves exactly once before the second pass copies.
234    /// Row and descriptor maps use O(source
235    /// rows + descriptors) temporary cells; work is O(source nodes + source rows +
236    /// source descriptors + selected Boolean edges + selected distinct row entries).
237    ///
238    /// # Errors
239    /// Refuses malformed selected storage, absent child map entries, allocation
240    /// failure or occurrence overflow. Atom-universe and topology admission remain
241    /// the resulting owner's responsibility.
242    pub fn try_select(self, children: &[Option<usize>]) -> Result<FormulaParts, AdmissionError> {
243        self.try_select_with_peak(children).map(|(parts, _)| parts)
244    }
245
246    /// Select and remap like [`Self::try_select`], measuring actual allocation peak.
247    ///
248    /// The second return value is the byte capacity of the output's seven arenas plus
249    /// all three internal row/descriptor maps while they coexist. It excludes the
250    /// borrowed source and child map, vector headers, stack values and allocator
251    /// overhead.
252    /// Exact reservations request compact storage; this receipt measures actual
253    /// capacities even if the allocator supplies more than requested. The maps are
254    /// released before returning, leaving only the returned formula's retained bytes.
255    ///
256    /// # Errors
257    /// Preserves the selection's malformed-storage, missing-child, allocation and
258    /// occurrence-overflow refusals. No partial output or receipt escapes on error.
259    pub fn try_select_with_peak(
260        self,
261        children: &[Option<usize>],
262    ) -> Result<(FormulaParts, u128), AdmissionError> {
263        let mut selection = Selection::new(self, children)?;
264        for (index, child) in children.iter().enumerate() {
265            if child.is_some() {
266                selection.append(index)?;
267            }
268        }
269        let peak = selection.result.retained_bytes()
270            + (selection.rows.capacity() as u128
271                + selection.thresholds.capacity() as u128
272                + selection.cuts.capacity() as u128)
273                * std::mem::size_of::<Option<usize>>() as u128;
274        let result = selection.result;
275        let parts = FormulaParts::from_numeric_raw(
276            result.nodes,
277            result.operands,
278            result.numeric.conditions,
279            result.numeric.values,
280            result.numeric.rows,
281            result.numeric.thresholds,
282            result.numeric.cuts,
283        )?;
284        Ok((parts, peak))
285    }
