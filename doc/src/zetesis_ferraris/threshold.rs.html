1//! Numeric threshold reduction over already-computed predecessor truth.
2
3use crate::Threshold;
4use zetesis_cpu::Stop;
5
6#[cfg(test)]
7mod tests;

9impl Threshold<'_> {
10    /// Test this nonnegative weighted threshold from predecessor truth.
11    ///
12    /// The callback reads each condition occurrence exactly once, in row order,
13    /// including repeated conditions and zero contributions. It owns truth
14    /// acquisition and its work/cancellation charges; this operation performs
15    /// no formula traversal or allocation and retains only a checked wide sum.
16    /// The caller applies the original-node truth gate for a frozen reduct.
17    ///
18    /// # Errors
19    /// Propagates callback errors. A mismatched row or an unrepresentable sum
20    /// returns [`Stop::InvalidProgram`]; admitted formula rows exclude both.
21    // Expose the callback to callers' hot loops so work and cancellation state
22    // need not spill across a separate reduction call.
23    #[inline]
24    pub fn evaluate<E: From<Stop>>(
25        self,
26        read: impl FnMut(usize) -> Result<bool, E>,
27    ) -> Result<bool, E> {
28        Ok(self.selected_sum(read)? >= self.bound)
29    }
30
31    /// Reduce every occurrence to a checked selected weight, independent of a
32    /// guard bound. The public threshold operation retains the same callback
33    /// order and error behavior; a pass-local row owner may reuse this result.
34    #[inline]
35    pub(crate) fn selected_sum<E: From<Stop>>(
36        self,
37        mut read: impl FnMut(usize) -> Result<bool, E>,
38    ) -> Result<i128, E> {
39        if self.conditions.len() != self.weights.len() {
40            return Err(Stop::InvalidProgram.into());
41        }
42        let mut total = 0_i128;
43        for (&condition, &weight) in self.conditions.iter().zip(self.weights) {
44            if read(condition)? {
45                total = total
46                    .checked_add(i128::from(weight))
47                    .ok_or(Stop::InvalidProgram)?;
48            }
49        }
50        Ok(total)
51    }
52}
