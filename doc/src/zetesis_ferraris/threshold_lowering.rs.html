1//! Checked physical Boolean expansion of nonnegative thresholds.
2//!
3//! This transport operation does not construct a second semantic theory. The
4//! caller retains the semantic node map and owns the two DP rows. For each
5//! nonzero contribution and target `t`, the recurrence is
6//! `next[t] = previous[t] OR (condition AND previous[t - weight])`, with
7//! saturating subtraction. Bounds requiring every positive contribution or any
8//! positive contribution instead use a linear connective fold with no DP rows.
9//! Repeated conditions retain independent coefficients.
10
11use crate::Threshold;
12
13/// A malformed numerical row or an unrepresentable physical expansion.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Error {
16    /// Columns, ranks or bounds are invalid, the total overflows, or scratch is too small.
17    Invalid,
18    /// Physical gates or scratch cannot be addressed by the host.
19    Capacity,
20}
21
22/// Exact unsimplified gate count and caller-owned temporary value slots.
23#[derive(Clone, Copy, Debug)]
24pub struct Shape {
25    /// Binary AND/OR gate attempts, before backend-specific simplification.
26    pub gates: usize,
27    /// Value slots required by the two recurrence rows.
28    pub scratch: usize,
29    expansion: Expansion,
30}
31
32#[derive(Clone, Copy, Debug)]
33enum Expansion {
34    Constant(bool),
35    All,
36    Any,
37    Dynamic { bound: usize },
38}
39
40impl Shape {
41    /// Inspect every contribution before any constant simplification.
42    ///
43    /// `visit` charges and polls once per input occurrence. No allocation occurs.
44    ///
45    /// # Errors
46    /// Returns the caller's interruption or a checked numerical/physical refusal.
47    pub fn new<E: From<Error>>(
48        threshold: Threshold<'_>,
49        mut visit: impl FnMut() -> Result<(), E>,
50    ) -> Result<Self, E> {
51        if threshold.conditions.len() != threshold.weights.len() {
52            return Err(Error::Invalid.into());
53        }
54        let mut total = 0_i128;
55        let mut nonzero = 0_usize;
56        let mut minimum = i128::MAX;
57        for &weight in threshold.weights {
58            visit()?;
59            total = total
60                .checked_add(i128::from(weight))
61                .ok_or(Error::Invalid)?;
62            if weight != 0 {
63                nonzero += 1;
64                minimum = minimum.min(i128::from(weight));
65            }
66        }
67        let constant = if threshold.bound <= 0 {
68            Some(true)
69        } else if threshold.bound > total {
70            Some(false)
71        } else {
72            None
73        };
74        if let Some(value) = constant {
75            return Ok(Self {
76                gates: 0,
77                scratch: 0,
78                expansion: Expansion::Constant(value),
79            });
80        }
81        // A nonconstant bound implies a positive total and at least one positive
82        // contribution. Dropping any such position loses at least minimum;
83        // selecting any one gains at least minimum. Equal conditions and zero
84        // positions do not change either implication.
85        let boundary = if threshold.bound > total - minimum {
86            Some(Expansion::All)
87        } else if threshold.bound <= minimum {
88            Some(Expansion::Any)
89        } else {
90            None
91        };
92        if let Some(expansion) = boundary {
93            return Ok(Self {
94                // Include the identity gate so even a single condition returns
95                // a newly appended last node in sinks that retain every gate.
96                gates: nonzero,
97                scratch: 0,
98                expansion,
99            });
100        }
101        let bound = usize::try_from(threshold.bound).map_err(|_| Error::Capacity)?;
102        let scratch = bound
103            .checked_add(1)
104            .and_then(|n| n.checked_mul(2))
105            .ok_or(Error::Capacity)?;
106        let gates = bound
107            .checked_mul(nonzero)
108            .and_then(|n| n.checked_mul(2))
109            .ok_or(Error::Capacity)?;
110        Ok(Self {
111            gates,
112            scratch,
113            expansion: Expansion::Dynamic { bound },
114        })
115    }
116
117    /// Whether the validated threshold is independent of all conditions.
118    #[must_use]
119    pub const fn constant(self) -> Option<bool> {
120        match self.expansion {
121            Expansion::Constant(value) => Some(value),
122            Expansion::All | Expansion::Any | Expansion::Dynamic { .. } => None,
123        }
124    }
125}

127/// Backend Boolean operations shared by threshold and ordinal-cut lowering.
128///
129/// Implementations preserve existing semantic atom identities. `gate` may
130/// simplify classical expressions; it must never replace the semantic owner.
131pub trait Sink {
132    /// Copyable physical output handle or constant.
133    type Value: Copy;
134    /// Caller-owned typed resource, control and shape failures.
135    type Error: From<Error>;
136    /// Charge one recurrence or input visit and poll cancellation.
137    ///
138    /// # Errors
139    /// Returns a work or control refusal.
140    fn tick(&mut self) -> Result<(), Self::Error>;
141    /// Materialize a Boolean constant.
142    ///
143    /// # Errors
144    /// Returns the backend's physical resource refusal.
145    fn constant(&mut self, value: bool) -> Result<Self::Value, Self::Error>;
146    /// Read the already translated semantic condition.
147    ///
148    /// # Errors
149    /// Returns a malformed map or physical resource refusal.
150    fn condition(&mut self, index: usize) -> Result<Self::Value, Self::Error>;
151    /// Compose an AND (`false`) or OR (`true`) gate.
152    ///
153    /// # Errors
154    /// Returns the backend's physical resource refusal.
155    fn gate(
156        &mut self,
157        disjunction: bool,
158        left: Self::Value,
159        right: Self::Value,
160    ) -> Result<Self::Value, Self::Error>;
161}

163/// Expand one threshold into physical Boolean operations using checked scratch.
164///
165/// Conditions must belong to an admitted predecessor map, including zero-weight
166/// positions: this numeric operation does not authenticate formula ownership.
167/// Boundary bounds use zero scratch and one AND/OR gate per positive occurrence;
168/// interior bounds use two O(bound) rows. Both attempt exactly `Shape::gates`
169/// gates. Each condition occurrence, initialization slot and DP cell consumes a
170/// `tick`, as does the checked shape scan. Every nonconstant expansion requests
171/// true and false constants before emitting gates, retaining the physical layout
172/// and last-node root contract. The caller retains the original-node truth gate
173/// when lowering a frozen query. All arithmetic is
174/// widened before any host-address conversion. No retained semantic nodes or
175/// allocations are created by this operation.
176///
177/// # Errors
178/// Returns input/physical shape, backend resource or interruption failures.
179pub fn lower<S: Sink>(
180    threshold: Threshold<'_>,
181    scratch: &mut [S::Value],
182    sink: &mut S,
183) -> Result<S::Value, S::Error> {
184    sink.tick()?;
185    let shape = Shape::new(threshold, || sink.tick())?;
186    if let Some(value) = shape.constant() {
187        return sink.constant(value);
188    }
189    if scratch.len() < shape.scratch {
190        return Err(Error::Invalid.into());
191    }
192    let truth = sink.constant(true)?;
193    let falsum = sink.constant(false)?;
194    let bound = match shape.expansion {
195        Expansion::All => return boundary(threshold, false, truth, sink),
196        Expansion::Any => return boundary(threshold, true, falsum, sink),
197        Expansion::Dynamic { bound } => bound,
198        Expansion::Constant(_) => unreachable!("constant expansion returned above"),
199    };
200    let width = bound + 1;
201    let (mut previous, mut current) = scratch[..shape.scratch].split_at_mut(width);
202    for target in 0..width {
203        sink.tick()?;
204        previous[target] = if target == 0 { truth } else { falsum };
205        current[target] = falsum;
206    }
207    for (&condition, &weight) in threshold.conditions.iter().zip(threshold.weights) {
208        sink.tick()?;
209        if weight == 0 {
210            continue;
211        }
212        let condition = sink.condition(condition)?;
213        // Only min(weight, bound) participates in a host index; a u64 weight
214        // never narrows through i32 or truncates on a narrower host.
215        let weight =
216            usize::try_from(u128::from(weight).min(bound as u128)).map_err(|_| Error::Capacity)?;
217        current[0] = truth;
218        for target in 1..width {
219            sink.tick()?;
220            let included = sink.gate(false, condition, previous[target.saturating_sub(weight)])?;
221            current[target] = sink.gate(true, previous[target], included)?;
222        }
223        std::mem::swap(&mut previous, &mut current);
224    }
225    Ok(previous[bound])
226}

228/// Every row position still receives its scan tick, including zero weights.
229/// Conditions keep their original order and repetitions; neither a decided
230/// result nor an identity gate bypasses later reads or their typed failures.
231fn boundary<S: Sink>(
232    threshold: Threshold<'_>,
233    disjunction: bool,
234    mut value: S::Value,
235    sink: &mut S,
236) -> Result<S::Value, S::Error> {
237    for (&condition, &weight) in threshold.conditions.iter().zip(threshold.weights) {
238        sink.tick()?;
239        if weight != 0 {
240            let condition = sink.condition(condition)?;
241            value = sink.gate(disjunction, value, condition)?;
242        }
243    }
244    Ok(value)
245}
