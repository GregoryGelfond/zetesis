26impl TightPlan {
27    /// Extract all original roots and derive a positive rank certificate.
28    ///
29    /// Accepted roots are atomic facts, ordinary atomic-head implications,
30    /// atomic choices `h ∨ not h` (optionally with a body), falsum and default
31    /// negations. Bodies admit atoms, falsum, conjunction, disjunction and
32    /// default negation; a negation freezes its whole interior. General
33    /// disjunctions, unnegated body implications and positive cycles fall back.
34    /// Every root is inspected, including constraints and candidate guards.
35    ///
36    /// # Errors
37    /// Returns explicit shape/cycle refusal, resource limits or shared control
38    /// and allocation failures. A failure never creates a partial certificate.
39    pub fn compile(
40        theory: &Theory,
41        limits: TightPlanLimits,
42        cancellation: &Cancellation,
43    ) -> Result<Self, TightError> {
44        Self::compile_accounted(theory, limits, cancellation).result
45    }
46
47    /// Derive the same certificate as [`Self::compile`], retaining charged work
48    /// on every result, including shape refusal, cancellation and work limits.
49    #[must_use]
50    pub fn compile_accounted(
51        theory: &Theory,
52        limits: TightPlanLimits,
53        cancellation: &Cancellation,
54    ) -> TightAttempt<Self> {
55        let mut work = Work {
56            used: 0,
57            max: limits.max_work,
58            cancellation: cancellation.polling(),
59        };
60        let result = build(theory, None, limits, &mut work);
61        TightAttempt {
62            result,
63            work: work.used,
64        }
65    }
66
67    /// Check an externally proposed atom rank against the same complete root
68    /// extraction. Ranks must be less than the atom count; gaps are permitted.
69    /// Source-analysis ranks are only proposals until this check succeeds.
70    ///
71    /// # Errors
72    /// Has the errors of [`Self::compile`], plus malformed or invalid ranks.
73    pub fn certify(
74        theory: &Theory,
75        ranks: &[usize],
76        limits: TightPlanLimits,
77        cancellation: &Cancellation,
78    ) -> Result<Self, TightError> {
79        let mut work = Work {
80            used: 0,
81            max: limits.max_work,
82            cancellation: cancellation.polling(),
83        };
84        build(theory, Some(ranks), limits, &mut work)
85    }
86}

88fn classify(theory: &Theory, work: &mut Work<'_>) -> Result<Vec<Body>, TightError> {
89    let mut classes = reserve(theory.nodes().len())?;
90    for index in 0..theory.view().len() {
91        classify_step(theory, index, &mut classes, work)?;
92    }
93    Ok(classes)
94}

96/// Charge and complete one classification before extending the valid prefix.
97fn classify_step(
98    theory: &Theory,
99    index: usize,
100    classes: &mut Vec<Body>,
101    work: &mut Work<'_>,
102) -> Result<(), TightError> {
103    work.tick()?;
104    let class = classify_node(theory, index, classes, work)?;
105    classes.push(class);
106    Ok(())
107}

109/// Complete one node's classification before publishing it to the prefix.
110/// Implications inspect their consequent; groups inspect every occurrence.
111fn classify_node(
112    theory: &Theory,
113    index: usize,
114    classes: &[Body],
115    work: &mut Work<'_>,
116) -> Result<Body, TightError> {
117    match theory
118        .view()
119        .node(index)
120        .map_err(|_| Stop::InvalidProgram)?
121    {
122        NodeView::Atom(_) => Ok(Body::Positive),
123        NodeView::False => Ok(Body::Frozen),
124        NodeView::Implies(_, consequent) => {
125            work.tick()?;
126            let node = theory
127                .view()
128                .node(consequent)
129                .map_err(|_| Stop::InvalidProgram)?;
130            Ok(if node.is_false() {
131                Body::Frozen
132            } else {
133                Body::Opaque
134            })
135        }
136        NodeView::And(operands) | NodeView::Or(operands) => {
137            classify_operands(operands, classes, work)
138        }
139    }
140}

142fn classify_operands(
143    operands: &[usize],
144    classes: &[Body],
145    work: &mut Work<'_>,
146) -> Result<Body, TightError> {
147    let mut class = Body::Frozen;
148    for &child in operands {
149        work.tick()?;
150        class = match (class, classes[child]) {
151            (Body::Opaque, _) | (_, Body::Opaque) => Body::Opaque,
152            (Body::Frozen, Body::Frozen) => Body::Frozen,
153            _ => Body::Positive,
154        };
155    }
156    Ok(class)
157}

159fn producer(
160    theory: &Theory,
161    root: usize,
162    classes: &[Body],
163) -> Result<Option<TightProducer>, TightError> {
164    let node = theory.view().node(root).map_err(|_| Stop::InvalidProgram)?;
165    let (body, head) = match node {
166        NodeView::False => return Ok(None),
167        NodeView::Implies(_, b)
168            if theory
169                .view()
170                .node(b)
171                .map_err(|_| Stop::InvalidProgram)?
172                .is_false() =>
173        {
174            return Ok(None);
175        }
176        NodeView::Implies(a, b) => (Some(a), b),
177        _ => (None, root),
178    };
179    let node = theory.view().node(head).map_err(|_| Stop::InvalidProgram)?;
180    let (head, kind) = match node {
181        NodeView::Atom(atom) => (atom, TightProducerKind::Normal),
182        _ => (
183            crate::atomic_choice::atom(theory, head).ok_or(TightError::UnsupportedRoot { root })?,
184            TightProducerKind::Choice,
185        ),
186    };
187    if let Some(body) = body
188        && classes[body] == Body::Opaque
189    {
190        return Err(TightError::UnsupportedBody { root, body });
191    }
192    Ok(Some(TightProducer {
193        head,
194        body,
195        root,
196        kind,
197    }))
198}

200fn extract(
201    theory: &Theory,
202    classes: &[Body],
203    limits: TightPlanLimits,
204    work: &mut Work<'_>,
205) -> Result<Vec<TightProducer>, TightError> {
206    let mut count = 0usize;
207    for &root in theory.roots() {
208        work.tick()?;
209        if producer(theory, root, classes)?.is_some() {
210            if count == limits.max_producers {
211                return Err(TightError::Limit(TightResource::Producers));
212            }
213            count += 1;
214        }
215    }
216    bytes(
217        theory.nodes().len() as u128 * size_of::<Body>() as u128
218            + count as u128 * size_of::<TightProducer>() as u128,
219        limits.max_bytes,
220    )?;
221    let mut producers = reserve(count)?;
222    for &root in theory.roots() {
223        work.tick()?;
224        if let Some(producer) = producer(theory, root, classes)? {
225            producers.push(producer);
226        }
227    }
228    Ok(producers)
229}

231fn each_dependency(
232    theory: &Theory,
233    classes: &[Body],
234    producers: &[TightProducer],
235    work: &mut Work<'_>,
236    mut visit: impl FnMut(usize, usize, bool) -> Result<(), TightError>,
237) -> Result<(), TightError> {
238    let offset = theory.atom_count();
239    for index in 0..theory.view().len() {
240        work.tick()?;
241        match theory
242            .view()
243            .node(index)
244            .map_err(|_| Stop::InvalidProgram)?
245        {
246            NodeView::Atom(atom) => visit(atom, offset + index, false)?,
247            NodeView::And(operands) | NodeView::Or(operands)
248                if classes[index] == Body::Positive =>
249            {
250                // Duplicate operands contribute duplicate edges and matching
251                // indegrees. Neither is silently deduplicated on its own.
252                for &child in operands {
253                    work.tick()?;
254                    if classes[child] == Body::Positive {
255                        visit(offset + child, offset + index, false)?;
256                    }
257                }
258            }
259            _ => {}
260        }
261    }
262    for producer in producers {
263        work.tick()?;
264        if let Some(body) = producer.body
265            && classes[body] == Body::Positive
266        {
267            visit(offset + body, producer.head, true)?;
268        }
269    }
270    Ok(())
271}

273fn build(
274    theory: &Theory,
275    proposed: Option<&[usize]>,
276    limits: TightPlanLimits,
277    work: &mut Work<'_>,
278) -> Result<TightPlan, TightError> {
279    work.cancellation.poll()?;
280    bytes(
281        theory.nodes().len() as u128 * size_of::<Body>() as u128,
282        limits.max_bytes,
283    )?;
284    let classes = classify(theory, work)?;
285    let producers = extract(theory, &classes, limits, work)?;
286    let mut dependencies = 0usize;
287    each_dependency(theory, &classes, &producers, work, |_, _, _| {
288        if dependencies == limits.max_dependencies {
289            return Err(TightError::Limit(TightResource::Dependencies));
290        }
291        dependencies += 1;
292        Ok(())
293    })?;
294    let vertices = theory
295        .atom_count()
296        .checked_add(theory.nodes().len())
297        .ok_or(TightError::Limit(TightResource::Bytes))?;
298    let resident_bytes = bytes(
299        producers.len() as u128 * size_of::<TightProducer>() as u128
300            + theory.atom_count() as u128 * size_of::<usize>() as u128,
301        limits.max_bytes,
302    )?;
303    let graph_bytes = if proposed.is_some() {
304        0
305    } else {
306        dependencies as u128 * size_of::<Edge>() as u128
307            + vertices as u128 * size_of::<usize>() as u128 * 4
308    };
309    let construction_bytes = bytes(
310        u128::from(resident_bytes)
311            + theory.nodes().len() as u128 * (size_of::<Body>() + size_of::<usize>()) as u128
312            + graph_bytes,
313        limits.max_bytes,
314    )?;
315    let ranks = if let Some(ranks) = proposed {
316        validate_ranks(theory, &classes, &producers, ranks, work)?;
317        let mut owned = reserve(ranks.len())?;
318        owned.extend_from_slice(ranks);
319        owned
320    } else {
321        derive_ranks(theory, &classes, &producers, vertices, dependencies, work)?
322    };
323    Ok(TightPlan {
324        theory: theory.clone(),
325        producers,
326        ranks,
327        statistics: TightPlanStatistics {
328            work: work.used,
329            dependencies,
330            construction_bytes,
331            resident_bytes,
332        },
333    })
334}

336fn derive_ranks(
337    theory: &Theory,
338    classes: &[Body],
339    producers: &[TightProducer],
340    vertices: usize,
341    dependency_count: usize,
342    work: &mut Work<'_>,
343) -> Result<Vec<usize>, TightError> {
344    let mut links = filled(vertices, END)?;
345    let mut indegrees = filled(vertices, 0usize)?;
346    let mut rank = filled(vertices, 0usize)?;
347    let mut queue = reserve(vertices)?;
348    let mut edges = reserve(dependency_count)?;
349    each_dependency(
350        theory,
351        classes,
352        producers,
353        work,
354        |source, target, strict| {
355            edges.push(Edge {
356                target,
357                next: links[source],
358                strict,
359            });
360            links[source] = edges.len() - 1;
361            indegrees[target] += 1;
362            Ok(())
363        },
364    )?;
365    for (index, &degree) in indegrees.iter().enumerate() {
366        work.tick()?;
367        if degree == 0 {
368            queue.push(index);
369        }
370    }
371    let mut cursor = 0;
372    while cursor < queue.len() {
373        work.tick()?;
374        let source = queue[cursor];
375        cursor += 1;
376        let mut edge_index = links[source];
377        while edge_index != END {
378            work.tick()?;
379            let edge = edges[edge_index];
380            rank[edge.target] = rank[edge.target].max(rank[source] + usize::from(edge.strict));
381            indegrees[edge.target] -= 1;
382            if indegrees[edge.target] == 0 {
383                queue.push(edge.target);
384            }
385            edge_index = edge.next;
386        }
387    }
388    if queue.len() != vertices {
389        for (atom, degree) in indegrees[..theory.atom_count()].iter().enumerate() {
390            work.tick()?;
391            if *degree != 0 {
392                return Err(TightError::PositiveCycle { atom });
393            }
394        }
395        // Formula edges are topological, so a remaining cycle necessarily
396        // passes through an atom. Keep a typed failure if that invariant fails.
397        return Err(zetesis_cpu::Stop::InvalidProgram.into());
398    }
399    let mut atoms = reserve(theory.atom_count())?;
400    atoms.extend_from_slice(&rank[..theory.atom_count()]);
401    validate_ranks(theory, classes, producers, &atoms, work)?;
402    Ok(atoms)
403}

405fn validate_ranks(
406    theory: &Theory,
407    classes: &[Body],
408    producers: &[TightProducer],
409    ranks: &[usize],
410    work: &mut Work<'_>,
411) -> Result<(), TightError> {
412    if ranks.len() != theory.atom_count() {
413        return Err(TightError::RankShape);
414    }
415    for &rank in ranks {
416        work.tick()?;
417        if rank >= ranks.len() {
418            return Err(TightError::RankShape);
419        }
420    }
421    let mut maximum = reserve::<usize>(theory.nodes().len())?;
422    for (index, &class) in classes.iter().enumerate() {
423        work.tick()?;
424        maximum.push(
425            match theory
426                .view()
427                .node(index)
428                .map_err(|_| Stop::InvalidProgram)?
429            {
430                NodeView::Atom(atom) => ranks[atom],
431                NodeView::And(operands) | NodeView::Or(operands) if class == Body::Positive => {
432                    maximum_operand(operands, &maximum, work)?
433                }
434                _ => 0,
435            },
436        );
437    }
438    for producer in producers {
439        work.tick()?;
440        if let Some(body) = producer.body
441            && classes[body] == Body::Positive
442            && maximum[body] >= ranks[producer.head]
443        {
444            return Err(TightError::RankOrder {
445                head: producer.head,
446            });
447        }
448    }
449    Ok(())
450}

452fn maximum_operand(
453    operands: &[usize],
454    maximum: &[usize],
455    work: &mut Work<'_>,
456) -> Result<usize, TightError> {
457    let mut value = 0;
458    for &child in operands {
459        work.tick()?;
460        value = value.max(maximum[child]);
461    }
462    Ok(value)
