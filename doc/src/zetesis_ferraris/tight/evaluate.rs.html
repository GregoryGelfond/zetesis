53/// Reusable truth and support storage for tight-certificate checks.
54///
55/// A complete check gives every formula node its truth in admitted order. Each
56/// conjunction or disjunction reads operands only through its first decisive
57/// value. Numeric nodes read and charge every physical occurrence once per
58/// authoritative row and pass, including repeats and zero weights, even for
59/// endpoint constants; dense measure growth and packed-validity resets are
60/// charged. Truth uses one byte per node, and the complete route's support one
61/// bit per atom, rounded to 32-bit words.
62///
63/// When the plan has its readers index ([`TightPlan::prepare_updates`]), a
64/// workspace keeps, after a check that completed, the complete evaluation of the
65/// candidate it checked together with a handle to that candidate's theory, and
66/// the next check of a candidate of the same theory updates it. A node's inputs
67/// are its atom's membership for an atom node, the truth of its row's
68/// conditions for a numeric guard, and its operands' truth otherwise. The update
69/// re-evaluates, in ascending node order, exactly the nodes with an input whose
70/// value changed, adjusts sums by each changed contribution and invalidates
71/// changed extrema (all other row measures survive),
72/// and maintains a four-byte support count per atom and the unsupported-atom and
73/// false-root sets, so its verdict and witness are those of a complete
74/// evaluation of the new candidate. A candidate that differs in more atoms than
75/// an eighth of the theory's atoms (and at least 64), a different theory, or a
76/// plan without the index evaluates completely. A check refused for a foreign
77/// theory, stopped at its first observation of control, or refused for storage
78/// at its start changes nothing. A check that ends later (refused for its work
79/// or for storage it grows, failing an allocation, or stopped at a later
80/// observation) leaves nothing the next check reads, so the next check
81/// evaluates completely. Each concurrent checker owns an exclusive workspace;
82/// immutable plans may be shared.
83#[derive(Debug, Default)]
84pub struct TightWorkspace {
85    values: Vec<u8>,
86    supported: Words<u32>,
87    numeric: NumericRowScratch,
88    retained: Retained,
89}

91/// One atom's support count: how many of its producers are bodyless or have a
92/// true body. Four bytes per atom; a plan admits the readers index only when no
93/// head can have more producers than this type holds
94/// ([`TightPlan::prepare_updates`]), so a count never passes it.
95pub(super) type SupportCount = u32;

97/// The complete evaluation a workspace keeps between checks of one theory, and
98/// the summaries an update maintains from it.
99///
100/// Each summary is a count over its inputs: a support count over its atom's
101/// producers, a missing flag over its atom's membership and support count, a
102/// false-root flag over its node's truth, and the two set counts over those
103/// flags. An update visits each changed input once, with its old and new value,
104/// and moves the count it feeds by one in the matching direction. Membership is
105/// applied before propagation, so a missing flag sees each of its two inputs
106/// change exactly when that input changes.
107#[derive(Debug, Default)]
108struct Retained {
109    /// Present exactly when node truth and the row measures are the complete
110    /// evaluation of `candidate` under this theory; holding the handle keeps the
111    /// theory's identity exact for as long as the evaluation is kept.
112    theory: Option<Theory>,
113    /// Packed words of the evaluated candidate.
114    candidate: Words<u64>,
115    /// Whether the support counts and the two sets summarize node truth.
116    summarized: bool,
117    /// For each atom, its producers that are bodyless or have a true body.
118    support: Vec<SupportCount>,
119    /// Present atoms without support, and how many.
120    missing: Words<u64>,
121    missing_count: usize,
122    /// Original root positions whose node is false, and how many.
123    false_roots: Words<u64>,
124    false_root_count: usize,
125    /// Update scratch: candidate words that changed, nodes awaiting
126    /// re-evaluation in ascending order, and their queued marks. The heap and the
127    /// marks are empty between completed updates.
128    changed: Vec<usize>,
129    pending: BinaryHeap<Reverse<usize>>,
130    queued: Words<u64>,
131}

133impl Retained {
134    fn bytes(&self) -> u128 {
135        (self.candidate.capacity()
136            + self.missing.capacity()
137            + self.false_roots.capacity()
138            + self.queued.capacity()) as u128
139            * size_of::<u64>() as u128
140            + self.changed.capacity() as u128 * size_of::<usize>() as u128
141            + self.support.capacity() as u128 * size_of::<SupportCount>() as u128
142            + self.pending.capacity() as u128 * size_of::<Reverse<usize>>() as u128
143    }
144}

157    /// Check original satisfaction and ranked support using retained storage.
158    ///
159    /// Identity failure precedes cancellation, which precedes storage admission.
160    /// A completed check's verdict and witness are always those of
161    /// [`TightPlan::check_accounted`]. Under a plan without the readers index,
162    /// work and error attribution match it as well. Under a plan with the index,
163    /// work and logical bytes are this workspace's own: an update charges its
164    /// update units, a complete evaluation that keeps its result also charges
165    /// the kept candidate and queued-mark words, retained storage counts against
166    /// the byte ceiling, and every check stays within
167    /// [`TightPlan::check_work_bound`]. Byte ceilings include existing capacity,
168    /// even for a smaller subsequent theory; checks do not shrink storage. Every
169    /// outcome retains its charged work and leaves actual capacity observable
170    /// through [`Self::retained_bytes`].
171    #[must_use]
172    pub fn check(
173        &mut self,
174        plan: &TightPlan,
175        candidate: &Interpretation,
176        limits: TightCheckLimits,
177        cancellation: &Cancellation,
178    ) -> TightAttempt<TightCheck> {
179        self.check_with(plan, candidate, limits, cancellation, true)
180    }

182    fn check_with(
183        &mut self,
184        plan: &TightPlan,
185        candidate: &Interpretation,
186        limits: TightCheckLimits,
187        cancellation: &Cancellation,
188        retain: bool,
189    ) -> TightAttempt<TightCheck> {
190        let metered = cancellation.metered(limits.max_work, |meter| {
191            self.evaluate(plan, candidate, limits, meter, retain)
192        });
193        TightAttempt {
194            result: metered.result,
195            work: metered.charged,
196        }
197    }

199    fn evaluate(
200        &mut self,
201        plan: &TightPlan,
202        candidate: &Interpretation,
203        limits: TightCheckLimits,
204        meter: &mut Meter<'_>,
205        retain: bool,
206    ) -> Result<TightCheck, TightError> {
207        if !plan.theory.same_instance(candidate.theory()) {
208            return Err(Stop::WrongProgram.into());
209        }
210        meter.poll()?;
211        // Storage admission precedes the take and either route's work, so a
212        // refusal here changes nothing and the retained evaluation stays valid.
213        // It counts what the workspace holds, each vector at no less than a
214        // complete check requires; each later growth passes through `grow`,
215        // which admits it before reserving it.
216        self.admit(plan, limits.max_bytes)?;
217        // Taken for the whole check: an interrupted update or evaluation leaves
218        // nothing a later check could read as a complete evaluation. The same
219        // theory instance also authenticates the retained summaries, which the
220        // last plan's producers computed: a plan's producers and readers are
221        // functions of its theory alone.
222        let previous = self.retained.theory.take();
223        if retain
224            && let Some(readers) = plan.readers.as_ref()
225            && previous
226                .as_ref()
227                .is_some_and(|theory| theory.same_instance(&plan.theory))
228            && let Some(check) = self.update(plan, readers, candidate, limits, meter)?
229        {
230            self.retained.theory = previous;
231            return Ok(check);
232        }
233        self.complete(
234            plan,
235            candidate,
236            limits,
237            meter,
238            retain && plan.readers.is_some(),
239        )
240    }

242    /// Evaluate `candidate` completely; with `keep`, retain the evaluation for
243    /// the next check's update. Precondition: the caller has admitted the
244    /// workspace's holdings ([`Self::admit`]), which counts the truth and support
245    /// vectors and the numeric scratch at a complete check's sizes, so the
246    /// reservations below grow nothing unadmitted.
247    fn complete(
248        &mut self,
249        plan: &TightPlan,
250        candidate: &Interpretation,
251        limits: TightCheckLimits,
252        meter: &mut Meter<'_>,
253        keep: bool,
254    ) -> Result<TightCheck, TightError> {
255        // Explicit byte cells give candidate tiles a fixed payload bound.
256        // Rust's Vec<bool> also stores byte-sized elements; it is not bit-packed.
257        self.values.clear();
258        self.values
259            .try_reserve_exact(plan.theory.nodes().len())
260            .map_err(|_| Stop::Allocation)?;
261        self.admit(plan, limits.max_bytes)?;
262        let base = self.bytes_beside_numeric(plan);
263        self.numeric.reserve_observed(&plan.theory, |actual| {
264            bytes(base + actual, limits.max_bytes).map(|_| ())
265        })?;
266        let logical_bytes = self.admit(plan, limits.max_bytes)?;
267        evaluate_truth(
268            &plan.theory,
269            candidate,
270            &mut self.values,
271            &mut self.numeric,
272            meter,
273        )?;
274        let logical_bytes = if keep {
275            self.keep(plan, candidate, limits, meter)?
276        } else {
277            logical_bytes
278        };
279        let (verdict, logical_bytes) =
280            if let Some(root) = false_root(&plan.theory, &self.values, meter)? {
281                (TightVerdict::NotModel { root }, logical_bytes)
282            } else {
283                self.supported.storage_mut().clear();
284                let words = plan.theory.atom_count().div_ceil(32);
285                self.supported
286                    .storage_mut()
287                    .try_reserve_exact(words)
288                    .map_err(|_| Stop::Allocation)?;
289                let logical_bytes = self.admit(plan, limits.max_bytes)?;
290                self.supported.storage_mut().resize(words, 0);
291                mark_support(&plan.producers, &self.values, &mut self.supported, meter)?;
292                let verdict = match unsupported_atom(candidate, &self.supported, meter)? {
293                    Some(unsupported_atom) => TightVerdict::Residual { unsupported_atom },
294                    None => TightVerdict::Stable,
295                };
296                (verdict, logical_bytes)
297            };
298        if keep {
299            self.retained.theory = Some(plan.theory.clone());
300        }
301        Ok(TightCheck {
302            verdict,
303            work: meter.charged(),
304            logical_bytes,
305        })
306    }

308    /// The bytes this workspace holds under `plan`, counting each vector at
309    /// the larger of its capacity and what a complete check requires, with
310    /// `more` bytes requested beyond them: the one measure every admission uses.
311    fn requested(&self, plan: &TightPlan, more: u128) -> u128 {
312        self.bytes_beside_numeric(plan) + self.numeric.required_bytes(&plan.theory) + more
313    }

315    /// Grow the retained evaluation past what the check's start admitted: admit
316    /// the workspace's measure with `more` bytes beside it, the most `reserve`
317    /// can add, against `limit`, and only then run `reserve`. Every such growth
318    /// passes through here, so a growth the byte limit refuses reserves nothing;
319    /// the work charged before it stays charged, as on every refusal. Storage
320    /// that grows only to a complete check's size needs no step of its own: the
321    /// measure already counts it at that size.
322    fn grow(
323        &mut self,
324        plan: &TightPlan,
325        limit: u64,
326        more: u128,
327        reserve: impl FnOnce(&mut Retained) -> Result<(), TryReserveError>,
328    ) -> Result<(), TightError> {
329        bytes(self.requested(plan, more), limit)?;
330        reserve(&mut self.retained).map_err(|_| Stop::Allocation.into())
331    }

333    /// Keep the evaluated candidate's words and an empty update scratch, one
334    /// charged operation per word; the summaries are derived when an update
335    /// first needs them.
336    fn keep(
337        &mut self,
338        plan: &TightPlan,
339        candidate: &Interpretation,
340        limits: TightCheckLimits,
341        meter: &mut Meter<'_>,
342    ) -> Result<u64, TightError> {
343        let words = candidate.words.len();
344        let marks = plan.theory.nodes().len().div_ceil(64);
345        self.retained.summarized = false;
346        self.retained.pending.clear();
347        self.retained.changed.clear();
348        self.retained.candidate.storage_mut().clear();
349        self.retained.queued.storage_mut().clear();
350        let more = (words.saturating_sub(self.retained.candidate.capacity())
351            + marks.saturating_sub(self.retained.queued.capacity())) as u128
352            * size_of::<u64>() as u128;
353        self.grow(plan, limits.max_bytes, more, |retained| {
354            retained.candidate.storage_mut().try_reserve_exact(words)?;
355            retained.queued.storage_mut().try_reserve_exact(marks)
356        })?;
357        let retained = &mut self.retained;
358        for &word in &candidate.words {
359            Tick::tick(meter)?;
360            retained.candidate.storage_mut().push(word);
361        }
362        for _ in 0..marks {
363            Tick::tick(meter)?;
364            retained.queued.storage_mut().push(0);
365        }
366        self.admit(plan, limits.max_bytes)
367    }

369    /// Derive support counts, unsupported present atoms and false root positions
370    /// from the retained complete evaluation, once per complete evaluation.
371    fn summarize(
372        &mut self,
373        plan: &TightPlan,
374        limits: TightCheckLimits,
375        meter: &mut Meter<'_>,
376    ) -> Result<(), TightError> {
377        let atoms = plan.theory.atom_count();
378        let roots = plan.theory.roots();
379        self.retained.support.clear();
380        self.retained.missing.storage_mut().clear();
381        self.retained.false_roots.storage_mut().clear();
382        let more = atoms.saturating_sub(self.retained.support.capacity()) as u128
383            * size_of::<SupportCount>() as u128
384            + (atoms
385                .div_ceil(64)
386                .saturating_sub(self.retained.missing.capacity())
387                + roots
388                    .len()
389                    .div_ceil(64)
390                    .saturating_sub(self.retained.false_roots.capacity())) as u128
391                * size_of::<u64>() as u128;
392        self.grow(plan, limits.max_bytes, more, |retained| {
393            retained.support.try_reserve_exact(atoms)?;
394            retained
395                .missing
396                .storage_mut()
397                .try_reserve_exact(atoms.div_ceil(64))?;
398            retained
399                .false_roots
400                .storage_mut()
401                .try_reserve_exact(roots.len().div_ceil(64))
402        })?;
403        let retained = &mut self.retained;
404        retained.support.resize(atoms, 0);
405        retained.missing.storage_mut().resize(atoms.div_ceil(64), 0);
406        retained
407            .false_roots
408            .storage_mut()
409            .resize(roots.len().div_ceil(64), 0);
410        for producer in &plan.producers {
411            Tick::tick(meter)?;
412            if producer.body.is_none_or(|body| self.values[body] != 0) {
413                let count = &mut retained.support[producer.head];
414                *count = count.checked_add(1).ok_or(Stop::InvalidProgram)?;
415            }
416        }
417        retained.false_root_count = 0;
418        for (position, &root) in roots.iter().enumerate() {
419            Tick::tick(meter)?;
420            if self.values[root] == 0 {
421                retained.false_roots.insert(position);
422                retained.false_root_count += 1;
423            }
424        }
425        retained.missing_count = 0;
426        for (word, &present) in retained.candidate.iter().enumerate() {
427            Tick::tick(meter)?;
428            for bit in Words::<u64>::bits(present) {
429                let atom = word * 64 + bit;
430                if retained.support[atom] == 0 {
431                    retained.missing.insert(atom);
432                    retained.missing_count += 1;
433                }
434            }
435        }
436        retained.summarized = true;
437        Ok(())
438    }

440    /// Bring the retained complete evaluation to `candidate`: re-evaluate, in
441    /// ascending node order, exactly the atom nodes of changed atoms and the
442    /// readers of changed values. Returns `None`, having changed nothing, when
443    /// more atoms changed than [`large_change`] admits. Precondition: the caller
444    /// has admitted the workspace's holdings ([`Self::admit`]); each growth here
445    /// passes through [`Self::grow`].
446    fn update(
447        &mut self,
448        plan: &TightPlan,
449        readers: &Readers,
450        candidate: &Interpretation,
451        limits: TightCheckLimits,
452        meter: &mut Meter<'_>,
453    ) -> Result<Option<TightCheck>, TightError> {
454        // First pass: find the changed words and count changed atoms, with no
455        // change to the retained state.
456        let threshold = large_change(plan.theory.atom_count());
457        let mut changed_atoms = 0_usize;
458        self.retained.changed.clear();
459        for (word, &present) in candidate.words.iter().enumerate() {
460            Tick::tick(meter)?;
461            let changed = self.retained.candidate[word] ^ present;
462            if changed != 0 {
463                changed_atoms += Words::<u64>::popcount(changed) as usize;
464                if changed_atoms > threshold {
465                    return Ok(None);
466                }
467                if self.retained.changed.len() == self.retained.changed.capacity() {
468                    let additional = self.retained.changed.capacity().max(8);
469                    self.grow(
470                        plan,
471                        limits.max_bytes,
472                        additional as u128 * size_of::<usize>() as u128,
473                        |retained| retained.changed.try_reserve_exact(additional),
474                    )?;
475                }
476                self.retained.changed.push(word);
477            }
478        }
479        if !self.retained.summarized {
480            self.summarize(plan, limits, meter)?;
481        }
482        // Second pass: apply the changed words.
483        for position in 0..self.retained.changed.len() {
484            Tick::tick(meter)?;
485            let index = self.retained.changed[position];
486            let present = candidate.words[index];
487            let changed = self.retained.candidate[index] ^ present;
488            self.retained.candidate[index] = present;
489            let mut changed = Words::<u64>::bits(changed);
490            while let Some(bit) = changed.next_with(|| Tick::tick(meter))? {
491                let atom = index * 64 + bit;
492                self.presence(atom, present & (1 << bit) != 0);
493                for &node in readers.atom_nodes(atom) {
494                    Tick::tick(meter)?;
495                    self.queue(node, plan, limits)?;
496                }
497            }
498        }
499        // Invariant: every node below the least queued node holds its truth under
500        // the new candidate, and a node is queued exactly when one of its inputs
501        // changed: an atom node's membership, a condition of a guard's row (the
502        // sum is adjusted or the extremum withdrawn before any guard runs), or an
503        // operand's truth. A reader has a larger index than everything it reads,
504        // so ascending pops re-evaluate each node from final inputs, and a popped
505        // node is never queued again: each is re-evaluated at most once.
506        while let Some(Reverse(node)) = self.retained.pending.pop() {
507            Tick::tick(meter)?;
508            self.retained.queued.remove(node);
509            let truth = evaluate_node(
510                &plan.theory,
511                node,
512                candidate,
513                &self.values,
514                &mut self.numeric,
515                meter,
516            )?;
517            if truth == (self.values[node] != 0) {
518                continue;
519            }
520            self.values[node] = u8::from(truth);
521            for &reader in readers.of(node) {
522                Tick::tick(meter)?;
523                let index = reader.index();
524                match reader.kind() {
525                    ReaderKind::Node => self.queue(index, plan, limits)?,
526                    ReaderKind::Row => {
527                        let (row, occurrence) = reader.row_occurrence();
528                        self.numeric
529                            .change_condition(&plan.theory, row, occurrence, truth)?;
530                        for &guard in readers.guards(row) {
531                            Tick::tick(meter)?;
532                            self.queue(guard, plan, limits)?;
533                        }
534                    }
535                    ReaderKind::Producer => self.support(plan.producers[index].head, truth)?,
536                    ReaderKind::Root => self.root(index, truth),
537                }
538            }
539        }
540        // Everything held was admitted, before the check's work or before its
541        // reservation. While each reservation's capacity is what was admitted
542        // for it, this only reports the bytes the check holds after its last
543        // reservation; an allocator may return more capacity than requested,
544        // and that excess is admitted here, against the same limit.
545        let logical_bytes = self.admit(plan, limits.max_bytes)?;
546        let retained = &self.retained;
547        let verdict = if retained.false_root_count > 0 {
548            TightVerdict::NotModel {
549                root: plan.theory.roots()[first_set(&retained.false_roots, meter)?],
550            }
551        } else if retained.missing_count > 0 {
552            TightVerdict::Residual {
553                unsupported_atom: first_set(&retained.missing, meter)?,
554            }
555        } else {
556            TightVerdict::Stable
557        };
558        Ok(Some(TightCheck {
559            verdict,
560            work: meter.charged(),
561            logical_bytes,
562        }))
563    }

565    /// Mark a node for re-evaluation once.
566    fn queue(
567        &mut self,
568        node: usize,
569        plan: &TightPlan,
570        limits: TightCheckLimits,
571    ) -> Result<(), TightError> {
572        let (word, bit) = Words::<u64>::locate(node);
573        if self.retained.queued[word] & bit != 0 {
574            return Ok(());
575        }
576        if self.retained.pending.len() == self.retained.pending.capacity() {
577            let additional = self.retained.pending.capacity().max(64);
578            self.grow(
579                plan,
580                limits.max_bytes,
581                additional as u128 * size_of::<Reverse<usize>>() as u128,
582                |retained| retained.pending.try_reserve_exact(additional),
583            )?;
584        }
585        self.retained.queued[word] |= bit;
586        self.retained.pending.push(Reverse(node));
587        Ok(())
588    }

590    /// An atom's membership changed; its support count did not.
591    fn presence(&mut self, atom: usize, present: bool) {
592        let retained = &mut self.retained;
593        let (word, bit) = Words::<u64>::locate(atom);
594        if present {
595            if retained.support[atom] == 0 {
596                retained.missing[word] |= bit;
597                retained.missing_count += 1;
598            }
599        } else if retained.missing[word] & bit != 0 {
600            retained.missing[word] &= !bit;
601            retained.missing_count -= 1;
602        }
603    }

605    /// A producer body changed truth; membership is already the new candidate's.
606    /// A plan with the index has at most `SupportCount::MAX` producers, so no
607    /// count can pass that bound; one that would is an invariant failure, as it
608    /// is when the counts are first derived. A falling count was raised by the same
609    /// producer, so it is at least one.
610    fn support(&mut self, head: usize, holds: bool) -> Result<(), TightError> {
611        let retained = &mut self.retained;
612        let (word, bit) = Words::<u64>::locate(head);
613        let present = retained.candidate[word] & bit != 0;
614        if holds {
615            let count = retained.support[head]
616                .checked_add(1)
617                .ok_or(Stop::InvalidProgram)?;
618            retained.support[head] = count;
619            if count == 1 && present {
620                retained.missing[word] &= !bit;
621                retained.missing_count -= 1;
622            }
623        } else {
624            retained.support[head] -= 1;
625            if retained.support[head] == 0 && present {
626                retained.missing[word] |= bit;
627                retained.missing_count += 1;
628            }
629        }
630        Ok(())
631    }
632
633    /// An original root position changed truth.
634    fn root(&mut self, position: usize, holds: bool) {
635        let retained = &mut self.retained;
636        let (word, bit) = Words::<u64>::locate(position);
637        if holds {
638            retained.false_roots[word] &= !bit;
639            retained.false_root_count -= 1;
640        } else {
641            retained.false_roots[word] |= bit;
642            retained.false_root_count += 1;
643        }
644    }
645
646    /// Admit the workspace's bytes under `plan` against `limit`, in one of two
647    /// roles. At a check's start it admits the holdings before any work. After a
648    /// reservation whose growth was admitted before it, it refuses only capacity
649    /// an allocator returned beyond the request, and otherwise reports the bytes
650    /// held.
651    fn admit(&self, plan: &TightPlan, limit: u64) -> Result<u64, TightError> {
652        bytes(self.requested(plan, 0), limit)
653    }
654
655    /// Every byte the workspace counts under `plan` except the numeric scratch:
656    /// the plan's resident bytes, the truth and support vectors at the larger of
657    /// capacity and a complete check's size, and the retained evaluation.
658    fn bytes_beside_numeric(&self, plan: &TightPlan) -> u128 {
659        u128::from(plan.statistics.resident_bytes)
660            + self.values.capacity().max(plan.theory.nodes().len()) as u128
661            + self
662                .supported
663                .capacity()
664                .max(plan.theory.atom_count().div_ceil(32)) as u128
665                * size_of::<u32>() as u128
666            + self.retained.bytes()
667    }
668}

670/// The truth of node `$index` from the truth of its operands, which precede it:
671/// the one rule a complete pass applies to every node and an update to each
672/// node it re-evaluates. A macro, so that both per-node loops receive the rule
673/// inline.
674macro_rules! node_truth {
675    ($theory:expr, $view:expr, $index:expr, $candidate:expr, $values:expr, $numeric:expr, $meter:expr) => {
676        match $view.node($index).map_err(|_| Stop::InvalidProgram)? {
677            NodeView::Atom(atom) => $candidate.contains(atom),
678            NodeView::False => false,
679            NodeView::And(operands) => evaluate_operands(operands, $values, true, $meter)?,
680            NodeView::Or(operands) => evaluate_operands(operands, $values, false, $meter)?,
681            NodeView::Threshold(threshold) => {
682                $numeric.threshold::<TightError>($theory, $index, threshold, |child| {
683                    Tick::tick($meter)?;
684                    Ok($values[child] != 0)
685                })?
686            }
687            NodeView::ExtremumCut(cut) => {
688                cut.rule.holds(
689                    $numeric.extremum::<TightError>($theory, $index, cut, |child| {
690                        Tick::tick($meter)?;
691                        Ok($values[child] != 0)
692                    })?,
693                )
694            }
695            NodeView::Implies(a, b) => {
696                Tick::tick($meter)?;
697                let left = $values[a];
698                Tick::tick($meter)?;
699                let right = $values[b];
700                left == 0 || right != 0
701            }
702        }
703    };
704}

706/// Write each formula truth in admitted topological order. The caller clears
707/// and reserves the target first; a stopped prefix is never membership evidence.
708/// The pass stays out of line so that its per-node loop, not its caller,
709/// receives the inlined operand reads.
710#[inline(never)]
711fn evaluate_truth(
712    theory: &Theory,
713    candidate: &Interpretation,
714    values: &mut Vec<u8>,
715    numeric: &mut NumericRowScratch,
716    meter: &mut Meter<'_>,
717) -> Result<(), TightError> {
718    numeric.reset(theory, || Tick::tick(meter))?;
719    let view = theory.view();
720    for index in 0..view.len() {
721        Tick::tick(meter)?;
722        let truth = node_truth!(theory, view, index, candidate, values, numeric, meter);
723        values.push(u8::from(truth));
724    }
725    Ok(())
726}
727
728/// One node's truth by [`node_truth`], for an update's re-evaluation.
729fn evaluate_node(
730    theory: &Theory,
731    index: usize,
732    candidate: &Interpretation,
733    values: &[u8],
734    numeric: &mut NumericRowScratch,
735    meter: &mut Meter<'_>,
736) -> Result<bool, TightError> {
737    let view = theory.view();
738    Ok(node_truth!(
739        theory, view, index, candidate, values, numeric, meter
740    ))
741}

743/// The first original root whose node is false, one charged test per root.
744/// Like the passes below, it stays out of line so that its loop receives the
745/// inlined charge.
746#[inline(never)]
747fn false_root(
748    theory: &Theory,
749    values: &[u8],
750    meter: &mut Meter<'_>,
751) -> Result<Option<usize>, TightError> {
752    for &root in theory.roots() {
753        Tick::tick(meter)?;
754        if values[root] == 0 {
755            return Ok(Some(root));
756        }
757    }
758    Ok(None)
759}

761/// Mark the head of every bodyless producer and of every producer whose body
762/// is true, one charged visit per producer.
763#[inline(never)]
764fn mark_support(
765    producers: &[TightProducer],
766    values: &[u8],
767    supported: &mut [u32],
768    meter: &mut Meter<'_>,
769) -> Result<(), TightError> {
770    let mut supported = Words::<u32, _>::new(supported);
771    for producer in producers {
772        Tick::tick(meter)?;
773        if producer.body.is_none_or(|body| values[body] != 0) {
774            supported.insert(producer.head);
775        }
776    }
777    Ok(())
778}

780/// The least present atom without support, one charged operation per 32-bit
781/// support word.
782#[inline(never)]
783fn unsupported_atom(
784    candidate: &Interpretation,
785    supported: &[u32],
786    meter: &mut Meter<'_>,
787) -> Result<Option<usize>, TightError> {
788    let supported = Words::<u32, _>::new(supported);
789    let mut present = candidate.words32();
790    for (word, &support) in supported.iter().enumerate() {
791        Tick::tick(meter)?;
792        let missing = present
793            .next()
794            .expect("support and candidate cover the same admitted atom universe")
795            & !support;
796        if missing != 0 {
797            return Ok(Some(word * 32 + Words::<u32>::first_bit(missing)));
798        }
799    }
800    Ok(None)
801}

803/// The most changed atoms an update takes on; a larger change evaluates
804/// completely. An execution choice: both routes return the same verdict.
805pub(super) fn large_change(atoms: usize) -> usize {
806    (atoms / 8).max(64)
807}

809/// The least set position of a packed set, one charged operation per word read.
810fn first_set(words: &[u64], meter: &mut Meter<'_>) -> Result<usize, TightError> {
811    Words::<u64, _>::new(words)
812        .next_from_with(0, || Tick::tick(meter))?
813        .ok_or_else(|| Stop::InvalidProgram.into())
814}

816#[inline]
817fn evaluate_operands(
818    operands: &[usize],
819    values: &[u8],
820    conjunction: bool,
821    meter: &mut Meter<'_>,
822) -> Result<bool, TightError> {
823    crate::operand_evaluation::reduce(operands, conjunction, |child| {
824        Tick::tick(meter)?;
825        Ok(values[child] != 0)
826    })
