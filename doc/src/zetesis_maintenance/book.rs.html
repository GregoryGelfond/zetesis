1//! A current Cargo artifact view for the manual's Rust examples.
2//!
3//! Incremental build directories may contain old libraries. Selection uses only
4//! one successful Cargo JSON message stream, never a directory glob or mtime.
5//! Publishing hard-links that set into a fresh directory without copying library
6//! payloads. The caller must keep the build artifacts immutable while rustdoc
7//! uses the view; this is not a concurrent filesystem snapshot or compiler seal.
8use crate::{Error, files, json, require};
9use std::{
10    collections::{BTreeMap, BTreeSet},
11    fs,
12    path::{Path, PathBuf},
13};
14
15/// Inclusive input and retained-path limits; these are not allocator RSS limits.
16#[derive(Clone, Copy, Debug)]
17pub struct Limits {
18    /// Serialized Cargo message bytes.
19    pub message_bytes: usize,
20    /// Distinct selected library paths.
21    pub artifacts: usize,
22}
23impl Default for Limits {
24    fn default() -> Self {
25        Self {
26            message_bytes: 16_777_216,
27            artifacts: 16_384,
28        }
29    }
30}

32/// Libraries named by one successful build, including reused fresh artifacts.
33#[derive(Debug)]
34pub struct Libraries {
35    paths: Vec<PathBuf>,
36}
37impl Libraries {
38    /// Ordered selected paths, before filesystem confinement is checked.
39    #[must_use]
40    pub fn paths(&self) -> &[PathBuf] {
41        &self.paths
42    }
43
44    /// Publish a fresh hard-linked view confined to `build_directory`.
45    ///
46    /// Validate every source and basename before creating `destination`. The
47    /// destination's parent must exist on the same filesystem as the build.
48    /// Filesystem failure during linking can leave a partial fresh view; never
49    /// pass that view to rustdoc. Existing destinations are never overwritten.
50    /// Work/storage are linear in path count after ordered basename lookup;
51    /// artifact contents are neither read nor copied.
52    ///
53    /// # Errors
54    /// Refuses missing/nonregular/symbolic-link sources, paths outside the build,
55    /// conflicting basenames, existing destinations and filesystem failures.
56    pub fn publish(&self, build_directory: &Path, destination: &Path) -> Result<(), Error> {
57        let root = build_directory
58            .canonicalize()
59            .map_err(|error| files::io(build_directory, error))?;
60        require(root.is_dir(), "book build root must be a directory")?;
61        let mut sources = BTreeMap::new();
62        for path in &self.paths {
63            let metadata = fs::symlink_metadata(path).map_err(|error| files::io(path, error))?;
64            require(metadata.is_file(), "book artifact must be a regular file")?;
65            let canonical = path
66                .canonicalize()
67                .map_err(|error| files::io(path, error))?;
68            require(
69                canonical.starts_with(&root),
70                "book artifact escapes the build directory",
71            )?;
72            let name = canonical
73                .file_name()
74                .ok_or_else(|| Error::Invalid("book artifact has no filename".into()))?
75                .to_owned();
76            require(
77                sources.insert(name, canonical).is_none(),
78                "conflicting book artifact basenames",
79            )?;
80        }
81        fs::create_dir(destination).map_err(|error| files::io(destination, error))?;
82        for (name, source) in sources {
83            let target = destination.join(name);
84            fs::hard_link(source, &target).map_err(|error| files::io(&target, error))?;
85        }
86        Ok(())
87    }
88}

90/// Select rlibs and procedural-macro libraries from a complete Cargo build.
91///
92/// Each `required` crate name must have a selected library. Other current
93/// libraries remain available for transitive rustdoc dependencies. Build scripts,
94/// executables and metadata-only files are not libraries in this view. A final
95/// successful `build-finished` event is required; it does not itself prove that
96/// a caller executed Cargo or that the listed bytes came from this source.
97/// Parsing is bounded by serialized bytes and selected paths; ordered sets add
98/// logarithmic lookup cost. JSON nesting retains `serde_json`'s recursion bound.
99///
100/// # Errors
101/// Refuses malformed/incomplete/failed streams, relative library paths, missing
102/// required crates and exceeded inclusive input/path limits. No files are written.
103pub fn select(messages: &[u8], required: &[&str], limits: Limits) -> Result<Libraries, Error> {
104    require(!required.is_empty(), "book view needs required crate names")?;
105    if messages.len() > limits.message_bytes {
106        return Err(Error::Limit {
107            resource: "Cargo message bytes",
108            limit: limits.message_bytes,
109        });
110    }
111    let mut paths = BTreeSet::new();
112    let mut crates = BTreeSet::new();
113    let mut finished = false;
114    for line in messages
115        .split(|byte| *byte == b'\n')
116        .filter(|line| !line.is_empty())
117    {
118        require(!finished, "Cargo messages follow build completion")?;
119        let message = json::parse(line)?;
120        match message["reason"].as_str() {
121            Some("build-finished") => {
122                require(message["success"] == true, "Cargo build did not succeed")?;
123                finished = true;
124            }
125            Some("compiler-artifact") => {
126                let name = json::string(&message["target"]["name"], "Cargo target name")?;
127                let kinds = message["target"]["kind"]
128                    .as_array()
129                    .ok_or_else(|| Error::Invalid("missing Cargo target kinds".into()))?;
130                let procedural = kinds.iter().any(|kind| kind == "proc-macro");
131                let filenames = message["filenames"]
132                    .as_array()
133                    .ok_or_else(|| Error::Invalid("missing Cargo artifact filenames".into()))?;
134                for filename in filenames {
135                    let path = Path::new(json::string(filename, "Cargo artifact filename")?);
136                    let suffix = path.extension().and_then(|extension| extension.to_str());
137                    if suffix == Some("rlib")
138                        || procedural && matches!(suffix, Some("so" | "dylib" | "dll"))
139                    {
140                        require(path.is_absolute(), "Cargo artifact path must be absolute")?;
141                        paths.insert(path.to_owned());
142                        if paths.len() > limits.artifacts {
143                            return Err(Error::Limit {
144                                resource: "book library paths",
145                                limit: limits.artifacts,
146                            });
147                        }
148                        crates.insert(name.to_owned());
149                    }
150                }
151            }
152            Some("compiler-message" | "build-script-executed") => {}
153            _ => return Err(Error::Invalid("unsupported Cargo message kind".into())),
154        }
155    }
156    require(finished, "Cargo build completion is missing")?;
157    for name in required {
158        require(
159            crates.contains(*name),
160            format!("missing book crate: {name}"),
161        )?;
162    }
163    Ok(Libraries {
164        paths: paths.into_iter().collect(),
165    })
166}
