1//! Finite selection and serial libtest protocol validation.
2use crate::{Error, require};
3use regex::Regex;
4use serde::Serialize;
5use std::{collections::BTreeSet, sync::LazyLock};
6use zetesis_backend::GpuApi;
7
8/// One reviewed physical target and its exact selected tests.
9#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
10pub struct Group {
11    /// Stable report directory label.
12    pub group: String,
13    /// Cargo library or integration-test target.
14    pub target_kind: String,
15    /// Serialized target identity, not an executable filename.
16    pub target: String,
17    /// Exact libtest names; successful zero-match runs cannot substitute.
18    pub tests: Vec<String>,
19    /// Independently specified count for this group.
20    pub expected_tests: usize,
21}
22/// A reviewed physical selection: the GPU API it qualifies and its groups.
23#[derive(Clone, Debug)]
24pub struct Selection {
25    /// The API whose reviewed table this is.
26    pub api: GpuApi,
27    /// The fourteen groups with their exact tests.
28    pub groups: Vec<Group>,
29}
30
31/// The reviewed selections: the Metal one and the Vulkan one, the same
32/// fourteen groups and counts, each naming the tests of its own API.
33const SELECTIONS: [(GpuApi, &str); 2] = [
34    (GpuApi::Metal, include_str!("physical-selection.txt")),
35    (
36        GpuApi::Vulkan,
37        include_str!("physical-selection-vulkan.txt"),
38    ),
39];
40
41/// Parse the maintained shell table, checking its finite inventory independently.
42/// The table must be one reviewed selection whole; one API's tests cannot
43/// stand in for the other's, and the selection says which API's it is.
44/// # Errors
45/// Refuses missing/extra groups, altered target/count identities or duplicate tests.
46pub fn selection(table: &str) -> Result<Selection, Error> {
47    const EXPECTED: [(&str, &str, usize); 14] = [
48        ("wgpu-lib", "lib", 15),
49        ("tight", "integration", 4),
50        ("formula", "integration", 2),
51        ("aggregate", "integration", 3),
52        ("lazy", "integration", 4),
53        ("cli-lazy", "integration", 5),
54        ("cli-formula", "integration", 3),
55        ("world-views", "integration", 4),
56        ("relation", "integration", 2),
57        ("context", "integration", 1),
58        ("solve-context", "lib", 3),
59        ("session-resources", "integration", 9),
60        ("language-consumers", "integration", 2),
61        ("static", "integration", 2),
62    ];
63    let api = SELECTIONS
64        .iter()
65        .find(|(_, selection)| table.trim() == selection.trim())
66        .map(|(api, _)| *api)
67        .ok_or_else(|| {
68            Error::Invalid(
69                "physical qualification requires one reviewed selection of 59 exact test identities".into(),
70            )
71        })?;
72    let rows: Vec<_> = table.lines().collect();
73    require(
74        rows.len() == EXPECTED.len(),
75        "physical qualification requires all fourteen groups and 59 named tests",
76    )?;
77    let mut groups = Vec::new();
78    let mut all_names = BTreeSet::new();
79    for (row, (name, target, count)) in rows.iter().zip(EXPECTED) {
80        let fields: Vec<_> = row.split('|').collect();
81        require(
82            fields.len() == 4,
83            "invalid physical qualification table row",
84        )?;
85        require(
86            fields[0] == name && fields[1] == target && fields[2].parse::<usize>() == Ok(count),
87            "changed physical qualification group identity",
88        )?;
89        let tests: Vec<String> = fields[3].split_whitespace().map(str::to_owned).collect();
90        require(
91            tests.len() == count && tests.iter().all(|test| all_names.insert(test.clone())),
92            "invalid physical qualification selection",
93        )?;
94        groups.push(Group {
95            group: name.into(),
96            target_kind: if target == "lib" { "lib" } else { "test" }.into(),
97            target: if target == "lib" {
98                "workspace libraries"
99            } else {
100                target
101            }
102            .into(),
103            tests,
104            expected_tests: count,
105        });
106    }
107    Ok(Selection { api, groups })
108}
109static RECORD: LazyLock<Regex> =
110    LazyLock::new(|| Regex::new(r"^test (\S+) \.\.\.[ \t]*(.*)$").unwrap());
111static SUMMARY: LazyLock<Regex> = LazyLock::new(|| {
112    Regex::new(r"^test result: ok\. ([0-9]+) passed; 0 failed; 0 ignored; 0 measured; [0-9]+ filtered out; finished in [^\r\n]+$").unwrap()
113});
114
115/// Validate exact individual outcomes and complete libtest summaries.
116/// Input must be serial `--nocapture` output: each record's final outcome is `ok`.
117/// Library selection may additionally contain zero-test workspace summaries.
118/// # Errors
119/// Refuses altered names/multiplicities, missing/failed outcomes, extra positive
120/// summaries, ignored tests and zero-match invocations. Exit status is checked
121/// independently by the invoking orchestration before this parser is called.
122pub fn physical_result(output: &str, group: &Group) -> Result<(), Error> {
123    super::input_bytes(output.len())?;
124    let mut selected = Vec::new();
125    let mut current_outcome: Option<&str> = None;
126    let mut counts = Vec::new();
127    for line in output.lines() {
128        if let Some(record) = RECORD.captures(line) {
129            if let Some(outcome) = current_outcome {
130                require(
131                    outcome == "ok",
132                    "physical test record lacks a passing outcome",
133                )?;
134            }
135            selected.push(record[1].to_owned());
136            current_outcome = Some(
137                record
138                    .get(2)
139                    .ok_or_else(|| Error::Invalid("missing physical outcome capture".into()))?
140                    .as_str()
141                    .trim(),
142            );
143        } else if line.starts_with("test result:") {
144            if let Some(outcome) = current_outcome.take() {
145                require(
146                    outcome == "ok",
147                    "physical test record lacks a passing outcome",
148                )?;
149            }
150            let summary = SUMMARY
151                .captures(line)
152                .ok_or_else(|| Error::Invalid("incomplete physical libtest summary".into()))?;
153            counts.push(
154                summary[1]
155                    .parse::<usize>()
156                    .map_err(|_| Error::Invalid("invalid physical summary count".into()))?,
157            );
158        } else if line.starts_with("test ") {
159            if let Some(outcome) = current_outcome.take() {
160                require(
161                    outcome == "ok",
162                    "physical test record lacks a passing outcome",
163                )?;
164            }
165        } else if current_outcome.is_some() && !line.trim().is_empty() {
166            current_outcome = Some(line.trim());
167        }
168    }
169    if let Some(outcome) = current_outcome {
170        require(
171            outcome == "ok",
172            "physical test record lacks a passing outcome",
173        )?;
174    }
175    selected.sort();
176    let mut expected = group.tests.clone();
177    expected.sort();
178    require(
179        selected == expected,
180        "physical qualification requires exactly the named passing tests",
181    )?;
182    let positive: Vec<_> = counts.iter().copied().filter(|count| *count > 0).collect();
183    // A `lib` or `integration` selection runs every workspace package's binary
184    // of that name, and the others report zero matches; the one positive
185    // summary, of the expected count, is the named tests' own binary.
186    let every_package = group.target_kind == "lib" || group.target == "integration";
187    require(
188        positive == [group.expected_tests] && (every_package || counts.len() == 1),
189        "physical qualification requires complete libtest summaries",
190    )
191}
