1//! Coverage metadata retains observed tool identities and disjoint populations.
2use super::{Floor, Group, Mode, SUPPORT_SOURCES, selection};
3use crate::{Error, files, require};
4use regex::Regex;
5use serde_json::{Value, json};
6use std::{path::Path, sync::LazyLock};

8static RUST_LLVM: LazyLock<Regex> =
9    LazyLock::new(|| Regex::new(r"(?m)^LLVM version: ([0-9]+\.[0-9]+\.[0-9]+)$").unwrap());
10static TOOL_LLVM: LazyLock<Regex> =
11    LazyLock::new(|| Regex::new(r"(?m)^[ \t]*LLVM version ([^\r\n]+)$").unwrap());
12/// Observed identity of one named LLVM executable.
13#[derive(Clone, Copy, Debug)]
14pub struct Tool<'a> {
15    /// Canonical executable path, resolved by the application.
16    pub path: &'a Path,
17    /// SHA-256 of the executable bytes actually read by the application.
18    pub sha256: &'a str,
19    /// Complete captured `--version` output.
20    pub version: &'a str,
21}
22/// Version observations; no tool is executed by metadata construction.
23#[derive(Clone, Copy, Debug)]
24pub struct Observation<'a> {
25    /// Pinned `rustc +1.97.1 -vV` output.
26    pub rustc: &'a str,
27    /// Complete observed `cargo +1.97.1 llvm-cov --version` output.
28    pub cargo_llvm_cov: &'a str,
29    /// Observed LLVM coverage executable.
30    pub llvm_cov: Tool<'a>,
31    /// Observed LLVM profile executable.
32    pub llvm_profdata: Tool<'a>,
33}
34/// A physical coverage stage: the backend it runs and that backend's reviewed
35/// selection, as the shell table the stage reads.
36#[derive(Clone, Copy, Debug)]
37pub struct Physical<'a> {
38    /// The backend whose physical tests the stage runs.
39    pub api: zetesis_backend::GpuApi,
40    /// The backend's reviewed selection.
41    pub table: &'a str,
42}
43/// Requested coverage policy and optional physical stage.
44#[derive(Clone, Copy, Debug)]
45pub struct Metadata<'a> {
46    /// Whether this run will gate or establish a baseline.
47    pub mode: Mode,
48    /// The exact committed file spelling after whitespace removal.
49    pub floor: &'a str,
50    /// The filename filter the reports and floors pass, which must be the
51    /// reviewed [`SUPPORT_SOURCES`].
52    pub filter: &'a str,
53    /// The physical stage, when physical tests are selected; absence means
54    /// portable only.
55    pub physical: Option<Physical<'a>>,
56    /// Versions and hashes observed at the command boundary.
57    pub observation: Observation<'a>,
58}
59fn tools(observation: Observation<'_>) -> Result<Value, Error> {
60    let captures = RUST_LLVM
61        .captures(observation.rustc)
62        .ok_or_else(|| Error::Invalid("pinned rustc did not report its LLVM version".into()))?;
63    let llvm = &captures[1];
64    let rust_build = format!("{llvm}-rust-1.97.1-stable");
65    let mut result = serde_json::Map::new();
66    for (name, tool) in [
67        ("LLVM_COV", observation.llvm_cov),
68        ("LLVM_PROFDATA", observation.llvm_profdata),
69    ] {
70        let matches: Vec<_> = TOOL_LLVM.captures_iter(tool.version).collect();
71        require(
72            matches.len() == 1 && matches[0][1].trim() == llvm
73                || matches.len() == 1 && matches[0][1].trim() == rust_build,
74            format!("{name} does not match rustc LLVM {llvm}"),
75        )?;
76        require(
77            tool.path.is_absolute(),
78            "LLVM tool identity requires an absolute path",
79        )?;
80        require(
81            tool.sha256.len() == 64
82                && tool
83                    .sha256
84                    .bytes()
85                    .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte)),
86            "invalid LLVM tool digest",
87        )?;
88        result.insert(
89            name.into(),
90            json!({"path":tool.path,"sha256":tool.sha256,"version":tool.version.trim()}),
91        );
92    }
93    Ok(Value::Object(result))
94}
95/// What a physical stage's 59 exact tests of `api` cover.
96fn physical_scope(api: zetesis_backend::GpuApi) -> String {
97    let (name, other) = match api {
98        zetesis_backend::GpuApi::Metal => ("Metal", "Vulkan"),
99        zetesis_backend::GpuApi::Vulkan => ("Vulkan", "Metal"),
100    };
101    format!(
102        "59 exact {name} tests: native aggregate reduction, static constructor and complete closure/reference checks, lazy transport and source closure, tight and formula oracles, ordinary lazy/formula CLI paths including automatic materialization and automatic CPU policy, complete-world-view collection, bounded relation equality filtering with 8/16/32-bit column transport, shared-context composition, contention and shared failure handling, caller-supplied session resources with policy and observer failures, observed complete collection on supplied contexts, explicit compiled formula profiles with independent oracle state and compilation identity, and combined language-consumer families with scored observations and optimum ties. Cooperative device control, interrupted preparation reuse and actual submission receipts are included, alongside completed-support table joins with actual GPU candidates and complete CPU/{name} answer families. Ordinary automatic membership sessions check complete tight families, the general device route for non-tight theories, and tight work refusal before dispatch. Terminal sessions check complete reconstruction from device-verified base answers. Unlisted tests and {other} are not selected."
103    )
104}
105/// Compose the established metadata schema from explicit validated observations.
106/// # Errors
107/// Refuses invalid policy, an altered physical selection or one of another
108/// backend, incompatible LLVM version strings or malformed executable identities.
109/// This does not attest tool execution.
110pub fn metadata(request: Metadata<'_>) -> Result<Value, Error> {
111    for value in [
112        request.floor,
113        request.observation.rustc,
114        request.observation.cargo_llvm_cov,
115        request.observation.llvm_cov.version,
116        request.observation.llvm_profdata.version,
117    ] {
118        super::input_bytes(value.len())?;
119    }
120    let cargo_version = request.observation.cargo_llvm_cov.trim();
121    require(
122        cargo_version == "cargo-llvm-cov 0.8.7",
123        "observed cargo-llvm-cov version does not match 0.8.7",
124    )?;
125    let version = cargo_version
126        .strip_prefix("cargo-llvm-cov ")
127        .ok_or_else(|| Error::Invalid("invalid cargo-llvm-cov banner".into()))?;
128    Floor::parse(request.floor)?.admit(request.mode)?;
129    require(
130        request.filter == SUPPORT_SOURCES,
131        "coverage reports leave out exactly the test-support crates' sources",
132    )?;
133    // A stage records the reviewed selection of the backend it runs.
134    let groups: Vec<Group> = match request.physical {
135        None => Vec::new(),
136        Some(physical) => {
137            let selection = selection(physical.table)?;
138            require(
139                selection.api == physical.api,
140                format!(
141                    "the {} stage records the {} reviewed selection only",
142                    physical.api.name(),
143                    physical.api.name()
144                ),
145            )?;
146            selection.groups
147        }
148    };
149    let tests: Vec<_> = groups.iter().flat_map(|group| group.tests.iter()).collect();
150    let api = request.physical.map(|physical| physical.api);
151    Ok(json!({
152        "mode":request.mode.label(),"committed_floor":request.floor,
153        "rustc":request.observation.rustc.trim(),"cargo_llvm_cov":version,"cargo_llvm_cov_observation":cargo_version,"llvm_tools":tools(request.observation)?,
154        "primary":"workspace --all-features","supplemental":"--package zetesis-cli --package zetesis-solve --package zetesis-engine --package zetesis --no-default-features",
155        "floor_profiles":["workspace","cli-cpu"],
156        "default_filename_filters":"cargo-llvm-cov 0.8.7 src/report.rs::ignore_filename_regex",
157        "project_added_filename_filters":[request.filter],"profiles_merged":false,
158        "profile_merge_scope":"profiles_merged describes floor profiles; raw execution profiles combine only within their own floor profile",
159        "workspace_execution":api.map_or_else(|| "portable".to_owned(), |api| format!("portable+{}", api.label())),
160        "workspace_stages":api.map_or_else(|| vec!["portable"], |api| vec!["portable", api.label()]),
161        "physical_test_groups":groups,"physical_tests":tests,"expected_physical_tests":tests.len(),
162        "physical_scope":api.map(physical_scope)
163    }))
164}
165/// Read one executable identity with the maintenance file ceiling.
166/// # Errors
167/// Refuses unavailable, nonregular or oversized executable files.
168pub fn executable_identity(path: &Path) -> Result<(std::path::PathBuf, String), Error> {
169    let canonical = path
170        .canonicalize()
171        .map_err(|error| files::io(path, error))?;
172    let parent = canonical
173        .parent()
174        .ok_or_else(|| Error::Invalid("executable has no parent".into()))?;
175    let name = canonical
176        .file_name()
177        .and_then(|value| value.to_str())
178        .ok_or_else(|| Error::Invalid("non-UTF-8 executable name".into()))?;
179    let mut tree = files::Tree::new(
180        parent,
181        files::Limits {
182            file_bytes: 268_435_456,
183            ..files::Limits::default()
184        },
185    )?;
186    let digest = files::digest(&tree.read(name)?);
187    Ok((canonical, digest))
188}
