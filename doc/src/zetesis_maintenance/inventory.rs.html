13/// Hash the declared Rust/WGSL workspace source boundary.
14/// The inventory covers root Cargo files/toolchain and recursive crate Rust,
15/// WGSL and Cargo manifests. The repository-level `target` directory is outside
16/// this traversal; matching files anywhere beneath `crates` are included.
17/// It does not seal tools, dependencies or executables.
18/// # Errors
19/// Refuses escaped/nonregular source paths, read/traversal limits and I/O errors.
20pub fn sources(root: &Path, limits: Limits) -> Result<BTreeMap<String, String>, Error> {
21    let mut tree = Tree::new(root, limits)?;
22    let mut names: BTreeSet<String> = ["Cargo.toml", "Cargo.lock", "rust-toolchain.toml"]
23        .map(str::to_owned)
24        .into();
25    names.extend(tree.inventory("crates", "")?.into_iter().filter(|name| {
26        let path = Path::new(name);
27        matches!(
28            path.extension().and_then(std::ffi::OsStr::to_str),
29            Some("rs" | "wgsl")
30        ) || path.file_name() == Some(std::ffi::OsStr::new("Cargo.toml"))
31    }));
32    names
33        .into_iter()
34        .map(|name| {
35            let digest = files::digest(&tree.read(&name)?);
36            Ok((name, digest))
37        })
38        .collect()
39}
40/// The maintained standalone packages, which the workspace does not build.
41const STANDALONE: [&str; 2] = ["validation/reference", "refinement/membership/rust"];
42/// Every maintained Rust source, relative to `root`, in order: the `src`,
43/// `tests`, `benches` and `examples` trees and the `build.rs` of each workspace
44/// package under `crates` and of each maintained standalone package, and the
45/// manual's shared examples. Build outputs, vendored imports and historical
46/// evidence lie outside these roots; a directory named `target` inside one is
47/// authored source.
48/// # Errors
49/// Refuses a package without a manifest, symbolic links, more than
50/// `limits.entries` directory entries in one tree, and I/O errors.
51pub fn authored(root: &Path, limits: Limits) -> Result<Vec<String>, Error> {
52    let tree = Tree::new(root, limits)?;
53    let mut packages = tree.directories("crates")?;
54    packages.extend(STANDALONE.map(str::to_owned));
55    let mut sources = BTreeSet::new();
56    for package in &packages {
57        let manifest = format!("{package}/Cargo.toml");
58        require(
59            tree.exists(&manifest)?,
60            format!("missing maintained package manifest: {package}"),
61        )?;
62        tree.member(&manifest)?;
63        for directory in ["src", "tests", "benches", "examples"] {
64            let directory = format!("{package}/{directory}");
65            if tree.exists(&directory)? {
66                sources.extend(tree.inventory(&directory, ".rs")?);
67            }
68        }
69        let build = format!("{package}/build.rs");
70        if tree.exists(&build)? {
71            tree.member(&build)?;
72            sources.insert(build);
73        }
74    }
75    sources.extend(tree.inventory("docs/book/examples", ".rs")?);
76    Ok(sources.into_iter().collect())
77}
78/// Read a small policy/record input through the same bounded file boundary.
79/// # Errors
80/// Refuses nonregular files, unavailable paths and inputs above `limit` bytes.
81pub fn read(path: &Path, limit: usize) -> Result<Vec<u8>, Error> {
82    let canonical = path
83        .canonicalize()
84        .map_err(|error| files::io(path, error))?;
85    let parent = canonical
86        .parent()
87        .ok_or_else(|| Error::Invalid("input has no parent".into()))?;
88    let name = canonical
89        .file_name()
90        .and_then(|value| value.to_str())
91        .ok_or_else(|| Error::Invalid("non-UTF-8 input name".into()))?;
92    Tree::new(
93        parent,
94        Limits {
95            file_bytes: limit,
96            total_bytes: limit,
97            ..Limits::default()
98        },
99    )?
100    .read(name)
101}
