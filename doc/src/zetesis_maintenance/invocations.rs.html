1//! Live documentation names only executables the workspace builds.
2//!
3//! Every command in a shell code block, and every tool table's first cell,
4//! whose first word is a zetesis executable must name a binary or package of
5//! the workspace, and `zetesis bench` is not a command. A document with
6//! [`RECORD`] on a line of its own is a dated measurement record: its commands
7//! keep the spellings of the binaries it records, so they are not checked.
8//! Prose is not checked; it may name an executable's former spelling on
9//! purpose. Code blocks are fenced with backticks or tildes alike.
10
11use std::collections::BTreeSet;
12
13use crate::markdown::{self, Line};
14use crate::{Error, require, workspace};
15
16/// Marks a dated measurement record, whose commands keep the spellings of the
17/// binaries it records. A record carries it on a line of its own, outside any
18/// code block.
19pub const RECORD: &str =
20    "<!-- A dated record: its commands keep the spellings of the binaries it records. -->";
21
22/// Whether `text` is a dated measurement record: a line of it outside every
23/// code block is exactly [`RECORD`]. A document that quotes the marker, as
24/// the contributor guide does in prose, or shows it in a code block, is not a
25/// record.
26#[must_use]
27pub fn is_record(text: &str) -> bool {
28    markdown::lines(text)
29        .any(|(_, line)| matches!(line, Line::Prose(prose) if prose.trim() == RECORD))
30}
31
32/// Code-block languages whose lines are shell commands.
33const SHELLS: [&str; 5] = ["sh", "shell", "bash", "zsh", "console"];
34
35/// Check each live document against the workspace's binaries and packages.
36///
37/// `documents` pairs each document's path, as reported, with its text;
38/// `metadata` is the output of `cargo metadata --no-deps --format-version 1`.
39///
40/// # Errors
41/// Returns [`Error::Invalid`] listing every offending line, and [`Error::Json`]
42/// for malformed metadata.
43pub fn check<'a>(
44    documents: impl IntoIterator<Item = (&'a str, &'a str)>,
45    metadata: &[u8],
46) -> Result<(), Error> {
47    let members = workspace::packages(metadata)?;
48    let known: BTreeSet<&str> = members
49        .iter()
50        .flat_map(|(name, package)| {
51            std::iter::once(name.as_str()).chain(package.binaries.iter().map(String::as_str))
52        })
53        .collect();
54    let mut stale = Vec::new();
55    for (path, text) in documents {
56        if is_record(text) {
57            continue;
58        }
59        for (line, command) in commands(text) {
60            if let Some(problem) = problem(command, &known) {
61                stale.push(format!("{path}:{line}: {problem}"));
62            }
63        }
64    }
65    require(
66        stale.is_empty(),
67        format!(
68            "live documentation names what the workspace does not build:\n{}",
69            stale.join("\n")
70        ),
71    )
72}

74/// Each command of a shell code block and each tool table's first cell, with
75/// its one-based line number.
76fn commands(text: &str) -> Vec<(usize, &str)> {
77    let mut commands = Vec::new();
78    let mut continued = false;
79    for (number, line) in markdown::lines(text) {
80        match line {
81            Line::Code { text, info } if shell(info) => {
82                // A line continuing the previous one carries arguments only.
83                let starts = !continued;
84                continued = text.trim_end().ends_with('\\');
85                let trimmed = text.trim_start();
86                if starts && !trimmed.starts_with('#') {
87                    commands.extend(
88                        trimmed
89                            .split([';', '|', '&'])
90                            .map(str::trim)
91                            .filter(|segment| !segment.is_empty())
92                            .map(|segment| (number, segment)),
93                    );
94                }
95            }
96            Line::Prose(text) => {
97                if let Some(cell) = table_cell(text.trim_start()) {
98                    commands.push((number, cell));
99                }
100            }
101            Line::Opening(_) | Line::Closing => continued = false,
102            Line::Code { .. } => {}
103        }
104    }
105    commands
106}

108/// Whether a code block's info string names a shell language.
109fn shell(info: &str) -> bool {
110    info.split([',', ' '])
111        .next()
112        .is_some_and(|language| SHELLS.contains(&language))
113}

115/// The code span filling a table row's first cell, as in "| `zetesis-bench` |".
116fn table_cell(line: &str) -> Option<&str> {
117    let cell = line.strip_prefix('|')?.split('|').next()?.trim();
118    cell.strip_prefix('`')?.strip_suffix('`')
119}

121/// Why a command names what the workspace does not build, if it does.
122fn problem(command: &str, known: &BTreeSet<&str>) -> Option<String> {
123    let mut words = command
124        .split_whitespace()
125        .skip_while(|word| *word == "$" || assignment(word));
126    let first = words.next()?;
127    let name = first.rsplit('/').next().unwrap_or(first);
128    if name != "zetesis" && !name.starts_with("zetesis-") {
129        return None;
130    }
131    if !known.contains(name) {
132        return Some(format!(
133            "`{name}` is not a binary or package of this workspace"
134        ));
135    }
136    // The CLI reserves `bench` only to refuse it with a pointer to the tool.
137    (name == "zetesis" && words.next() == Some("bench"))
138        .then(|| "`zetesis bench` is not a command; benchmarking is `zetesis-bench`".to_owned())
139}

141/// Whether `word` is a shell variable assignment such as `NAME=value`.
142fn assignment(word: &str) -> bool {
143    word.split_once('=').is_some_and(|(name, _)| {
144        !name.is_empty()
145            && name
146                .chars()
147                .all(|character| character == '_' || character.is_ascii_alphanumeric())
148    })
149}
