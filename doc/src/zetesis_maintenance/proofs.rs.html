1//! Retained Lean proof-record consistency under a deliberately restricted convention.
2//!
3//! The checker recognizes plain line-leading theorems, explicit namespaces and
4//! sections, and ASCII qualified names. Nested comments and strings are masked.
5//! It refuses other declaration forms instead of pretending to parse Lean.
6//! Actual pinned kernel checking must run independently before record validation.
7//! Hashes establish consistency, not command execution or compiler refinement.

35/// Counts derived from the actual record and source inventory.
36#[derive(Debug, PartialEq, Eq, Serialize)]
37pub struct Summary {
38    /// Exact unique theorem declarations and matching audit entries.
39    pub theorems: usize,
40    /// Recursively inventoried semantic Lean modules.
41    pub semantic_modules: usize,
42    /// Semantic modules plus the fixed configuration/source set.
43    pub source_files: usize,
44}
45/// Read and validate a retained record without executing commands or writing files.
46///
47/// The record name is relative to `root`. File reads and traversal use `limits`;
48/// retained storage is bounded by read bytes plus inventory and parsed JSON
49/// overhead. JSON nesting is limited by `serde_json`'s default recursion ceiling.
50/// Work is linear in source and record bytes, apart from ordered-set indexing.
51///
52/// # Errors
53/// Refuses malformed, stale, incomplete or unsupported records, escaped paths,
54/// exceeded read budgets and filesystem failures. None of these is a Lean verdict.
55pub fn verify(root: &Path, record_name: &str, limits: Limits) -> Result<Summary, Error> {
56    verify_record(root, record_name, limits, None)
57}
58
59/// Check a retained record against the actual output of a newly executed audit.
60///
61/// `live_audit` must contain complete stdout from the pinned strict Audit command;
62/// the caller separately requires its successful exit and empty stderr. This
63/// operation executes nothing and compares exact bytes as well as the existing
64/// source/index/axiom contracts. The borrowed output has the file-byte ceiling.
65///
66/// # Errors
67/// Refuses every invalid retained record, oversized live output, or a live audit
68/// differing from its retained counterpart. Equality does not attest execution.
69pub fn verify_with_audit(
70    root: &Path,
71    record_name: &str,
72    live_audit: &[u8],
73    limits: Limits,
74) -> Result<Summary, Error> {
75    if live_audit.len() > limits.file_bytes {
76        return Err(Error::Limit {
77            resource: "live audit bytes",
78            limit: limits.file_bytes,
79        });
80    }
81    verify_record(root, record_name, limits, Some(live_audit))
82}

84fn verify_record(
85    root: &Path,
86    record_name: &str,
87    limits: Limits,
88    live_audit: Option<&[u8]>,
89) -> Result<Summary, Error> {
90    let mut tree = Tree::new(root, limits)?;
91    let record = json::parse(&tree.read(record_name)?)?;
92    json::object(&record, "verification record")?;
93    require(
94        record["schema_version"].as_u64() == Some(1) && record["status"] == "PASS",
95        "unsupported or unsuccessful verification record",
96    )?;
97    let modules = tree.inventory("Zetesis", ".lean")?;
98    require(!modules.is_empty(), "missing semantic modules")?;
99    let mut sources = modules.clone();
100    sources.extend(FIXED.map(str::to_owned));
101    hashes(
102        &mut tree,
103        &record["source_sha256"],
104        Some(&sources),
105        "source",
106    )?;
107    hashes(
108        &mut tree,
109        &record["generated_artifact_sha256"],
110        Some(&ARTIFACTS.map(str::to_owned).into()),
111        "generated artifacts",
112    )?;
113    let mut entries = Vec::new();
114    for module in &modules {
115        entries.extend(source::declarations(module, &tree.text(module)?)?);
116    }
117    let names: BTreeSet<String> = entries.iter().map(|entry| entry.name.clone()).collect();
118    require(!entries.is_empty(), "empty source theorem inventory")?;
119    require(
120        names.len() == entries.len(),
121        "duplicate qualified source theorem",
122    )?;
123    index(&mut tree, &entries)?;
124    let axioms = audit::check(&mut tree, &names)?;
125    if let Some(live) = live_audit {
126        require(
127            tree.read("axiom-audit.txt")? == live,
128            "live audit differs from the retained kernel output",
129        )?;
130    }
131    counts(&record, &entries, modules.len())?;
132    require(
133        record["transitive_axioms"] == serde_json::to_value(axioms).map_err(Error::Json)?,
134        "transitive axiom inventory mismatch",
135    )?;
136    require(
137        record["project_axioms"]
138            .as_array()
139            .is_some_and(Vec::is_empty)
140            && record["proof_holes"].as_bool() == Some(false)
141            && record["native_evaluation_proof_shortcuts"].as_bool() == Some(false),
142        "unsupported proof assurance flags",
143    )?;
144    audit::commands(&mut tree, &record)?;
145    Ok(Summary {
146        theorems: entries.len(),
147        semantic_modules: modules.len(),
148        source_files: sources.len(),
149    })
150}
151fn hashes(
152    tree: &mut Tree,
153    value: &Value,
154    expected: Option<&BTreeSet<String>>,
155    label: &str,
156) -> Result<(), Error> {
157    let map = json::object(value, label)?;
158    if let Some(expected) = expected {
159        require(
160            map.keys().cloned().collect::<BTreeSet<_>>() == *expected,
161            format!("{label} inventory mismatch"),
162        )?;
163    }
164    for (name, value) in map {
165        let digest = json::string(value, "digest")?;
166        require(
167            digest.len() == 64
168                && digest
169                    .bytes()
170                    .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte)),
171            format!("invalid {label} digest: {name}"),
172        )?;
173        require(
174            files::digest(&tree.read(name)?) == digest,
175            format!("{label} hash mismatch: {name}"),
176        )?;
177    }
178    Ok(())
179}

188fn index(tree: &mut Tree, expected: &[source::Declaration]) -> Result<(), Error> {
189    let value = json::parse(&tree.read("theorems.json")?)?;
190    let recorded: Vec<RecordedDeclaration> = serde_json::from_value(value).map_err(Error::Json)?;
191    let entries: Vec<_> = recorded
192        .into_iter()
193        .map(|entry| source::Declaration {
194            name: entry.name,
195            file: entry.file,
196            line: entry.line,
197        })
198        .collect();
199    require(
200        entries.iter().all(|entry| entry.line > 0),
201        "invalid theorem index location",
202    )?;
203    let names: BTreeSet<&str> = entries.iter().map(|entry| entry.name.as_str()).collect();
204    require(names.len() == entries.len(), "duplicate theorem index name")?;
205    require(
206        entries.iter().collect::<BTreeSet<_>>() == expected.iter().collect(),
207        "theorem index names/files/lines differ from source declarations",
208    )
209}
210fn count(value: &Value, expected: usize, label: &str) -> Result<(), Error> {
211    require(
212        value.as_u64().and_then(|n| usize::try_from(n).ok()) == Some(expected),
213        format!("{label} count mismatch"),
214    )
215}
216fn counts(record: &Value, entries: &[source::Declaration], modules: usize) -> Result<(), Error> {
217    count(&record["semantic_modules"], modules, "semantic module")?;
218    count(
219        &record["theorems_audited"],
220        entries.len(),
221        "top-level theorem",
222    )?;
223    if let Some(value) = record.get("batch_accounting_laws") {
224        let expected = entries
225            .iter()
226            .filter(|entry| entry.file == "Zetesis/BatchAccounting.lean")
227            .count();
228        require(
229            expected > 0,
230            "batch accounting module theorem count mismatch",
231        )?;
232        count(value, expected, "batch accounting module theorem")?;
233    }
234    let consistency = &record["audit_consistency"];
235    json::object(consistency, "nested audit consistency")?;
236    for key in [
237        "source_theorem_declarations",
238        "indexed_theorems",
239        "printed_axiom_entries",
240        "unique_qualified_names",
241    ] {
242        count(&consistency[key], entries.len(), &format!("nested {key}"))?;
243    }
244    for key in ["source_index_line_check", "allowed_transitive_axioms_only"] {
245        require(
246            consistency[key] == "PASS",
247            format!("nested audit failure: {key}"),
248        )?;
249    }
250    Ok(())
