1//! Execute pinned Lean checks and publish their current record transactionally.
2//!
3//! The caller supplies actual tool directories and a fresh external evidence
4//! directory. Existing generated index/Audit views must match current sources.
5//! Captures have bounded polling/output and settle owned children before fallible
6//! retention. No success is inferred from a historical record or supplied count.
7//! Concurrent edits and other commands writing these artifacts are unsupported;
8//! the capture lock excludes other captures, not arbitrary filesystem writers.
9//! Kernel laws remain distinct from Rust/WGSL refinement and physical execution.
10

32type Result<T> = std::result::Result<T, Box<dyn Error>>;
33const INPUT_BYTES: usize = 256 * 1024 * 1024;

35fn require(condition: bool, message: &str) -> Result<()> {
36    if condition {
37        Ok(())
38    } else {
39        Err(message.into())
40    }
41}

43fn bytes(path: &Path) -> Result<Vec<u8>> {
44    Ok(inventory::read(path, INPUT_BYTES)?)
45}
46
47fn digest(path: &Path) -> Result<String> {
48    Ok(format!("{:x}", Sha256::digest(bytes(path)?)))
49}

51fn json_file(path: &Path, value: &Value) -> Result<()> {
52    let mut output = serde_json::to_vec_pretty(value)?;
53    output.push(b'\n');
54    fs::write(path, output)?;
55    Ok(())
56}

58fn copy_flat(source: &Path, destination: &Path) -> Result<()> {
59    copy_flat_with(source, destination, proofs::Limits::default())
60}
61fn copy_flat_with(source: &Path, destination: &Path, limits: proofs::Limits) -> Result<()> {
62    fs::create_dir(destination)?;
63    let mut tree = crate::files::Tree::new(source, limits)?;
64    for (index, entry) in fs::read_dir(source)?.enumerate() {
65        require(index < limits.entries, "retained-log entry limit exceeded")?;
66        let entry = entry?;
67        require(
68            entry.file_type()?.is_file(),
69            "unexpected non-file in retained logs",
70        )?;
71        let name = entry.file_name();
72        let name = name.to_str().ok_or("non-UTF-8 retained-log name")?;
73        fs::write(destination.join(name), tree.read(name)?)?;
74    }
75    Ok(())
76}

78/// Explicit effects and tool owners for a proof capture.
79#[derive(Clone, Copy, Debug)]
80pub struct Request<'a> {
81    /// Repository containing the `proofs` package.
82    pub repository: &'a Path,
83    /// Existing empty evidence directory, disjoint from the repository.
84    pub evidence: &'a Path,
85    /// Directory containing the actual pinned Lean and Lake executables.
86    pub lean_bin: &'a Path,
87    /// Directory containing the actual pinned Rust and Cargo executables.
88    pub rust_bin: &'a Path,
89    /// Actual maintenance executable to copy into the external evidence owner.
90    /// The immutable copy is used for checks even if Cargo replaces this path.
91    pub maintenance: &'a Path,
92    /// Per-command polling, output and initial cleanup limits.
93    pub command_limits: Limits,
94}

96/// Operation whose failure prevented a complete capture/publication.
97#[derive(Clone, Copy, Debug, PartialEq, Eq)]
98pub enum Phase {
99    /// Paths, current source/views or tool owners were not admitted.
100    Preflight,
101    /// Pinned Lean version/build/strict checks or audit failed.
102    Kernel,
103    /// Rust/maintenance identity or checker regressions failed.
104    Assurance,
105    /// Captured sources, tools or staged record did not agree.
106    Validation,
107    /// Artifact publication or recovery failed; inspect preserved evidence.
108    Publication,
109    /// The new record is committed, but owned scratch/lock cleanup failed.
110    Cleanup,
111}
112
113/// Capture failure with a typed operation boundary and original diagnostic chain.
114#[derive(Debug)]
115pub struct Failure {
116    phase: Phase,
117    cause: Box<dyn Error>,
118}
119impl Failure {
120    /// Operation whose contract was not completed.
121    #[must_use]
122    pub const fn phase(&self) -> Phase {
123        self.phase
124    }
125}
126impl std::fmt::Display for Failure {
127    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
128        write!(formatter, "proof capture {:?}: {}", self.phase, self.cause)
129    }
130}
131impl Error for Failure {
132    fn source(&self) -> Option<&(dyn Error + 'static)> {
133        Some(self.cause.as_ref())
134    }
135}
136fn phase<T>(which: Phase, work: impl FnOnce() -> Result<T>) -> std::result::Result<T, Failure> {
137    work().map_err(|cause| Failure {
138        phase: which,
139        cause,
140    })
141}

143fn freeze_maintenance(original: &Path, work: &Path) -> Result<PathBuf> {
144    let directory = work.join("tools");
145    fs::create_dir(&directory)?;
146    let frozen = directory.join("zetesis-maintenance");
147    let original_bytes = bytes(original)?;
148    let hash = format!("{:x}", Sha256::digest(&original_bytes));
149    fs::write(&frozen, original_bytes)?;
150    fs::set_permissions(&frozen, fs::metadata(original)?.permissions())?;
151    require(digest(&frozen)? == hash, "frozen maintenance copy differs")?;
152    json_file(
153        &directory.join("identity.json"),
154        &serde_json::json!({
155            "original_path": original, "original_sha256": hash,
156            "frozen_path": frozen, "frozen_sha256": hash,
157            "scope": "exact admitted executable bytes; permission-preserving copy, not a dependency closure"
158        }),
159    )?;
160    Ok(frozen)
161}

163struct Inputs {
164    repository: PathBuf,
165    work: PathBuf,
166    tools: Tools,
167    maintenance: PathBuf,
168    before: proofs::Inventory,
169    tool_hashes: BTreeMap<String, String>,
170    documents: BTreeMap<&'static str, String>,
171    lock: publication::Lock,
172}
173fn preflight(request: Request<'_>) -> Result<Inputs> {
174    let repository = request.repository.canonicalize()?;
175    let work = request.evidence.canonicalize()?;
176    require(
177        repository.is_dir() && work.is_dir(),
178        "repository/evidence must be directories",
179    )?;
180    require(
181        !work.starts_with(&repository) && !repository.starts_with(&work),
182        "evidence must be disjoint from the repository",
183    )?;
184    require(
185        fs::read_dir(&work)?.next().is_none(),
186        "evidence directory must be empty",
187    )?;
188    let proof_root = repository.join("proofs");
189    let lock = publication::Lock::acquire(&proof_root)?;
190    let before = source_preflight(&proof_root)?;
191    let documents = ["README.md", "theorems.json"]
192        .into_iter()
193        .map(|name| Ok((name, digest(&proof_root.join(name))?)))
194        .collect::<Result<BTreeMap<_, _>>>()?;
195    let tools = Tools::new(&request)?;
196    let original_maintenance = request.maintenance.canonicalize()?;
197    let maintenance = freeze_maintenance(&original_maintenance, &work)?;
198    let tool_hashes = tools.hashes(&maintenance)?;
199    json_file(
200        &work.join("inputs.json"),
201        &serde_json::json!({
202            "source_sha256": before.source_sha256(), "document_sha256": documents,
203            "tool_sha256": tool_hashes,
204            "system_executables": {
205                "/usr/bin/env": digest(Path::new("/usr/bin/env"))?,
206                "/bin/sh": digest(Path::new("/bin/sh"))?
207            }
208        }),
209    )?;
210    Ok(Inputs {
211        repository,
212        work,
213        tools,
214        maintenance,
215        before,
216        tool_hashes,
217        documents,
218        lock,
219    })
220}

222/// Execute current pinned checks and publish only their validated record.
223///
224/// The index and Audit must already equal `inventory`'s deterministic views.
225/// Before execution, archive original records in `evidence/before`; failed
226/// commands retain raw receipts under `evidence/commands` and record views under
227/// `evidence/stage`. The process owner settles
228/// children before fallible retention. A staged record must verify against the
229/// unchanged source/tool identities before publication. Publication uses local
230/// renames with rollback; concurrent readers can observe a temporary mismatch,
231/// so this is an exclusive multi-file transaction, not a lock-free snapshot.
232/// Failed recovery retains the original archive and reports it explicitly.
233///
234/// This operation writes records and Lean/Cargo build artifacts and runs the
235/// recorded commands. It never imports coverage or changes semantic sources.
236/// Byte/traversal ceilings apply to each bounded read pass; they are not a total
237/// process-memory or filesystem snapshot guarantee. Theorem counts are derived.
238///
239/// # Errors
240/// Refuses bad/freshness-overlapping paths, stale views, wrong tools, failed or
241/// interrupted commands, changed inputs, invalid audit/records and publication
242/// failures. No failure establishes a proof result or implementation refinement.
243pub fn capture(request: Request<'_>) -> std::result::Result<proofs::Summary, Failure> {
244    let Inputs {
245        repository,
246        work,
247        tools,
248        maintenance,
249        before,
250        tool_hashes,
251        documents,
252        lock,
253    } = phase(Phase::Preflight, || preflight(request))?;
254    let proof_root = repository.join("proofs");
255    let (archive, mut runner) = phase(Phase::Preflight, || {
256        prepare_record_storage(
257            repository,
258            &work,
259            &proof_root,
260            command_environment(&tools)?,
261            request.command_limits,
262        )
263    })?;
264    let kernel = phase(Phase::Kernel, || {
265        execute_kernel(&mut runner, &before, &tools)
266    })?;
267    phase(Phase::Assurance, || {
268        check_assurance_tools(&mut runner, &tools, &maintenance)
269    })?;
270    let (after, summary) = phase(Phase::Validation, || {
271        require(
272            tools.hashes(&maintenance)? == tool_hashes,
273            "capture tools changed during execution",
274        )?;
275        for (name, hash) in &documents {
276            require(
277                digest(&proof_root.join(name))? == *hash,
278                "proof documentation/views changed during capture",
279            )?;
280        }
281        let after = retain_current_sources(&mut runner, &proof_root, &before)?;
282        let record = construct_record(&runner, &after, &kernel, &tools, &maintenance)?;
283        check_staged_record(&runner, &record, after.declarations().len())?;
284        let summary = proofs::verify_with_audit(
285            &runner.stage,
286            "verification.json",
287            &kernel.audit,
288            proofs::Limits::default(),
289        )?;
290        Ok((after, summary))
291    })?;
292    let published = phase(Phase::Publication, || {
293        publication::publish(&runner.stage, &proof_root, &archive, || {
294            proofs::verify_with_audit(
295                &proof_root,
296                "verification.json",
297                &kernel.audit,
298                proofs::Limits::default(),
299            )?;
300            require(
301                proofs::inventory(&proof_root, proofs::Limits::default())?.source_sha256()
302                    == after.source_sha256(),
303                "proof sources changed before publication",
304            )?;
305            Ok(())
306        })
307    })?;
308    phase(Phase::Cleanup, || published.cleanup())?;
309    phase(Phase::Cleanup, || lock.release())?;
310    Ok(summary)
311}

317    #[test]
318    fn archive_copy_admits_inclusive_read_limits() {
319        let directory = tempfile::tempdir().unwrap();
320        let source = directory.path().join("source");
321        fs::create_dir(&source).unwrap();
322        fs::write(source.join("one"), "abc").unwrap();
323        fs::write(source.join("two"), "def").unwrap();
324        let exact = proofs::Limits {
325            file_bytes: 3,
326            total_bytes: 6,
327            entries: 2,
328        };
329        copy_flat_with(&source, &directory.path().join("exact"), exact).unwrap();
330        for (label, limits) in [
331            (
332                "file",
333                proofs::Limits {
334                    file_bytes: 2,
335                    ..exact
336                },
337            ),
338            (
339                "total",
340                proofs::Limits {
341                    total_bytes: 5,
342                    ..exact
343                },
344            ),
345            (
346                "entries",
347                proofs::Limits {
348                    entries: 1,
349                    ..exact
350                },
351            ),
352        ] {
353            assert!(copy_flat_with(&source, &directory.path().join(label), limits).is_err());
354            assert_eq!(fs::read(source.join("one")).unwrap(), b"abc");
355            assert_eq!(fs::read(source.join("two")).unwrap(), b"def");
356        }
357    }

359    #[test]

363        fs::create_dir_all(source.join("nested")).unwrap();
364        assert!(copy_flat(&source, &directory.path().join("copy")).is_err());
365    }
