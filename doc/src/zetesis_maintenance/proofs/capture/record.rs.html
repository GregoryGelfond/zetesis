15pub(super) fn source_preflight(proof_root: &Path) -> Result<proofs::Inventory> {
16    let before = proofs::inventory(proof_root, proofs::Limits::default())?;
17    require(
18        bytes(&proof_root.join("lean-toolchain"))? == b"leanprover/lean4:v4.33.1\n",
19        "capture requires the declared Lean 4.33.1 toolchain",
20    )?;
21    require(
22        before.theorem_index()? == bytes(&proof_root.join("theorems.json"))?,
23        "rendered theorem index differs",
24    )?;
25    require(
26        before.audit_source().as_bytes() == bytes(&proof_root.join("Audit.lean"))?,
27        "rendered Audit source differs",
28    )?;
29    let manifest: Value = serde_json::from_slice(&bytes(&proof_root.join("lake-manifest.json"))?)?;
30    require(
31        manifest["packages"] == json!([]),
32        "unexpected external Lean packages",
33    )?;
34
35    Ok(before)
36}

38pub(super) fn prepare_record_storage(
39    repository: PathBuf,
40    work: &Path,
41    proof_root: &Path,
42    environment: BTreeMap<String, OsString>,
43    command_limits: Limits,
44) -> Result<(PathBuf, Runner)> {
45    let archive = work.join("before");
46    fs::create_dir(&archive)?;
47    fs::write(
48        archive.join("verification.json"),
49        bytes(&proof_root.join("verification.json"))?,
50    )?;
51    fs::write(
52        archive.join("axiom-audit.txt"),
53        bytes(&proof_root.join("axiom-audit.txt"))?,
54    )?;
55    copy_flat(
56        &proof_root.join("verification/current"),
57        &archive.join("current"),
58    )?;
59    let stage = work.join("stage");
60    fs::create_dir(&stage)?;
61    fs::create_dir_all(stage.join("verification/current"))?;
62    let evidence = work.join("commands");
63    fs::create_dir(&evidence)?;
64    let runner = Runner {
65        repository,
66        stage,
67        evidence,
68        commands: Vec::new(),
69        logs: BTreeMap::new(),
70        environment,
71        command_limits,
72    };
73
74    Ok((archive, runner))
75}

77pub(super) fn retain_current_sources(
78    runner: &mut Runner,
79    proof_root: &Path,
80    before: &proofs::Inventory,
81) -> Result<proofs::Inventory> {
82    let after = proofs::inventory(proof_root, proofs::Limits::default())?;
83    require(
84        before.source_sha256() == after.source_sha256(),
85        "proof sources changed during execution",
86    )?;
87    let inventory_path = runner.stage.join("verification/current/inventory.json");
88    json_file(&inventory_path, &serde_json::to_value(&after)?)?;
89    runner.logs.insert(
90        "verification/current/inventory.json".into(),
91        digest(&inventory_path)?,
92    );
93    for name in after.source_sha256().keys() {
94        let destination = runner.stage.join(name);
95        fs::create_dir_all(destination.parent().ok_or("source path has no parent")?)?;
96        fs::write(destination, bytes(&proof_root.join(name))?)?;
97    }
98    for name in ["README.md", "theorems.json"] {
99        fs::write(runner.stage.join(name), bytes(&proof_root.join(name))?)?;
100    }
101    Ok(after)
102}

104pub(super) fn construct_record(
105    runner: &Runner,
106    after: &proofs::Inventory,
107    kernel: &Kernel,
108    tools: &Tools,
109    maintenance: &Path,
110) -> Result<Value> {
111    let artifacts: BTreeMap<String, String> = ["README.md", "theorems.json", "axiom-audit.txt"]
112        .into_iter()
113        .map(|name| Ok((name.into(), digest(&runner.stage.join(name))?)))
114        .collect::<Result<_>>()?;
115    let audit_text = std::str::from_utf8(&kernel.audit)?;
116    let axiom_brackets = Regex::new(r"\[([^\]]*)\]")?;
117    let axioms: BTreeSet<&str> = axiom_brackets
118        .captures_iter(audit_text)
119        .flat_map(|capture| capture.get(1).unwrap().as_str().split(',').map(str::trim))
120        .filter(|value| !value.is_empty())
121        .collect();
122    require(
123        axioms
124            .iter()
125            .all(|name| ["Classical.choice", "Quot.sound", "propext"].contains(name)),
126        "unexpected transitive axiom",
127    )?;
128    let count = after.declarations().len();
129    let record = json!({
130        "schema_version": 1,
131        "project": "zetesis",
132        "status": "PASS",
133        "lean_version": kernel.identity.version,
134        "lean_commit": kernel.identity.commit,
135        "platform": kernel.identity.platform,
136        "external_packages": [],
137        "semantic_modules": after.modules().len(),
138        "theorems_audited": count,
139        "path_convention": {
140            "command_cwd": "Paths are relative to the repository root; proofs denotes its Lean package directory.",
141            "record_members": "Source, generated-artifact and log paths are relative to proofs/.",
142            "toolchain": "Command executable names denote the explicitly selected pinned tools; executable hashes are recorded separately.",
143            "capture": "Each command declares its log layout. stdout_then_stderr retains raw stdout followed by raw stderr. json_stdout_and_stderr stores those exact UTF-8 streams reversibly, with the raw concatenation digest. Separate byte lengths are recorded. Unix timestamps are seconds since 1970-01-01 UTC."
144        },
145        "tool_sha256": tools.hashes(maintenance)?,
146        "rust_toolchain": "1.97.1",
147        "assurance_tool": concat!("zetesis-maintenance ", env!("CARGO_PKG_VERSION"), ": retained record consistency; not Lean kernel execution"),
148        "command_environment": { "CARGO_BUILD_JOBS": "1", "CARGO_NET_OFFLINE": "true", "RUSTUP_TOOLCHAIN": "1.97.1" },
149        "capture_limits": { "polling_deadline_seconds": runner.command_limits.timeout.as_secs_f64(), "retained_output_bytes": runner.command_limits.max_output_bytes, "cleanup_seconds": runner.command_limits.cleanup_timeout.as_secs_f64(), "cleanup_retry_seconds": 2 },
150        "commands": runner.commands,
151        "transitive_axioms": axioms,
152        "project_axioms": [],
153        "proof_holes": false,
154        "native_evaluation_proof_shortcuts": false,
155        "source_sha256": after.source_sha256(),
156        "generated_artifact_sha256": artifacts,
157        "verification_log_sha256": runner.logs,
158        "inventory_observation": {
159            "producer": "zetesis_maintenance::proofs::inventory",
160            "path": "verification/current/inventory.json",
161            "scope": "Observed declarations and source hashes; not a kernel verdict"
162        },
163        "audit_consistency": {
164            "source_theorem_declarations": count,
165            "indexed_theorems": count,
166            "printed_axiom_entries": count,
167            "unique_qualified_names": count,
168            "source_index_line_check": "PASS",
169            "allowed_transitive_axioms_only": "PASS"
170        },
171        "refinement_boundary": "Kernel-checked mathematical definitions and laws. Concrete parsing, source lowering, grounding, Rust representations, resource behavior, scheduling and WGSL execution remain separate unproved implementation correspondences. Rust record checks establish retained evidence consistency; they do not replace Lean kernel execution."
172    });
173    Ok(record)
174}

176pub(super) fn check_staged_record(runner: &Runner, record: &Value, count: usize) -> Result<()> {
177    json_file(&runner.stage.join("verification.json"), record)?;
178    let summary = proofs::verify(
179        &runner.stage,
180        "verification.json",
181        proofs::Limits::default(),
182    )?;
183    require(summary.theorems == count, "stage consistency count differs")?;
184
185    Ok(())
186}
