17/// An observed proof source inventory, not a proof-verification result.
18/// Files must remain stable while read; this value is not a filesystem snapshot.
19#[derive(Debug, Serialize)]
20pub struct Inventory {
21    modules: BTreeSet<String>,
22    declarations: Vec<Declaration>,
23    source_sha256: BTreeMap<String, String>,
24}
25impl Inventory {
26    /// Semantic module paths in deterministic lexicographic order.
27    #[must_use]
28    pub fn modules(&self) -> &BTreeSet<String> {
29        &self.modules
30    }
31    /// Declarations in module order, then source-line order.
32    #[must_use]
33    pub fn declarations(&self) -> &[Declaration] {
34        &self.declarations
35    }
36    /// Hashes of observed semantic modules and fixed source/configuration files.
37    /// These describe current bytes, before any rendered view is published.
38    #[must_use]
39    pub fn source_sha256(&self) -> &BTreeMap<String, String> {
40        &self.source_sha256
41    }
42    /// Render the exact declaration index, without writing it or certifying proofs.
43    /// # Errors
44    /// Refuses a serialization error.
45    pub fn theorem_index(&self) -> Result<Vec<u8>, Error> {
46        let mut output = serde_json::to_vec_pretty(&self.declarations).map_err(Error::Json)?;
47        output.push(b'\n');
48        Ok(output)
49    }
50    /// Render an umbrella import and one axiom request per observed theorem.
51    /// These requests must subsequently be executed by the pinned Lean kernel.
52    #[must_use]
53    pub fn audit_source(&self) -> String {
54        let mut output = String::from("import Zetesis\n\n");
55        for declaration in &self.declarations {
56            output.push_str("#print axioms ");
57            output.push_str(declaration.name());
58            output.push('\n');
59        }
60        output
61    }
62}
63/// Inspect proof sources and produce deterministic index/audit views.
64///
65/// The restricted scanner is shared with record verification. Every source is
66/// read within the supplied limits; symbolic links in the semantic inventory are
67/// refused. Retained bytes and ordered indexes bound work and storage. No command
68/// runs, file is written, record status is invented or kernel acceptance inferred.
69///
70/// # Errors
71/// Refuses unsupported declarations, duplicate qualified names, empty inventory,
72/// missing fixed source files, exceeded limits and filesystem failures.
73pub fn inventory(root: &Path, limits: Limits) -> Result<Inventory, Error> {
74    let mut tree = Tree::new(root, limits)?;
75    let modules = tree.inventory("Zetesis", ".lean")?;
76    require(!modules.is_empty(), "missing semantic modules")?;
77    let mut declarations = Vec::new();
78    let mut source_sha256 = BTreeMap::new();
79    for module in &modules {
80        let bytes = tree.read(module)?;
81        let text = std::str::from_utf8(&bytes)
82            .map_err(|_| Error::Invalid(format!("invalid UTF-8: {module}")))?;
83        declarations.extend(source::declarations(module, text)?);
84        source_sha256.insert(module.clone(), files::digest(&bytes));
85    }
86    let names: BTreeSet<_> = declarations.iter().map(Declaration::name).collect();
87    require(!declarations.is_empty(), "empty source theorem inventory")?;
88    require(
89        names.len() == declarations.len(),
90        "duplicate qualified source theorem",
91    )?;
92    for file in FIXED {
93        source_sha256.insert(file.into(), files::digest(&tree.read(file)?));
94    }
95    Ok(Inventory {
96        modules,
97        declarations,
98        source_sha256,
99    })
100}
