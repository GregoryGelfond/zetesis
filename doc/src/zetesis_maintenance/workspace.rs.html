1//! The workspace's packages and binaries, as `cargo metadata` reports them.
2
3use std::collections::{BTreeMap, BTreeSet};
4use std::path::Path;
5
6use crate::Error;
7
8/// A workspace package as Cargo reports it.
9pub(crate) struct Package {
10    /// The name of the directory holding its manifest.
11    pub(crate) directory: String,
12    /// The names of its binary targets.
13    pub(crate) binaries: BTreeSet<String>,
14}
15
16/// Every package in `metadata`, the output of `cargo metadata --no-deps
17/// --format-version 1`, by name.
18pub(crate) fn packages(metadata: &[u8]) -> Result<BTreeMap<String, Package>, Error> {
19    let metadata: serde_json::Value = serde_json::from_slice(metadata).map_err(Error::Json)?;
20    let packages = metadata["packages"]
21        .as_array()
22        .ok_or_else(|| Error::Invalid("Cargo metadata lists no packages".into()))?;
23    let mut workspace = BTreeMap::new();
24    for package in packages {
25        let (Some(name), Some(manifest)) =
26            (package["name"].as_str(), package["manifest_path"].as_str())
27        else {
28            return Err(Error::Invalid(
29                "a Cargo metadata package lacks its name or manifest".into(),
30            ));
31        };
32        let directory = Path::new(manifest)
33            .parent()
34            .and_then(Path::file_name)
35            .and_then(|directory| directory.to_str())
36            .ok_or_else(|| Error::Invalid(format!("{name}'s manifest has no directory")))?;
37        let binaries = package["targets"]
38            .as_array()
39            .map_or(&[][..], Vec::as_slice)
40            .iter()
41            .filter(|target| {
42                target["kind"]
43                    .as_array()
44                    .is_some_and(|kinds| kinds.iter().any(|kind| kind == "bin"))
45            })
46            .filter_map(|target| target["name"].as_str().map(str::to_owned))
47            .collect();
48        workspace.insert(
49            name.to_owned(),
50            Package {
51                directory: directory.to_owned(),
52                binaries,
53            },
54        );
55    }
56    Ok(workspace)
57}
