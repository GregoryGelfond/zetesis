12fn variable(name: &str, default: &str) -> String {
13    env::var(name).unwrap_or_else(|_| default.into())
14}
15fn value<'a>(arguments: &'a [String], flag: &str) -> Option<&'a str> {
16    arguments
17        .iter()
18        .position(|arg| arg == flag)
19        .and_then(|index| arguments.get(index + 1))
20        .map(String::as_str)
21}
22fn has(arguments: &[String], flag: &str) -> bool {
23    arguments.iter().any(|arg| arg == flag)
24}
25fn fail(reason: &str) -> Result<(), String> {

28fn trace(role: &str, arguments: &[String]) {
29    if let Ok(path) = env::var("CHECK_TEST_TRACE") {
30        writeln!(
31            OpenOptions::new()
32                .append(true)
33                .create(true)
34                .open(path)
35                .unwrap(),
36            "{role} {}",
37            arguments.join(" ")
38        )
39        .unwrap();
40    }
41}
42fn report_arguments(arguments: &[String]) -> Result<(), String> {
43    if arguments.get(2).map(String::as_str) != Some("report") {
44        return Ok(());
45    }
46    let mut remaining = &arguments[3..];
47    while let Some((argument, rest)) = remaining.split_first() {
48        if ["--locked", "--json", "--html"].contains(&argument.as_str()) {
49            remaining = rest;
50        } else if [
51            "--package",
52            "--output-path",
53            "--output-dir",
54            "--fail-under-lines",
55            "--ignore-filename-regex",
56        ]
57        .contains(&argument.as_str())
58            && !rest.is_empty()
59        {
60            remaining = &rest[1..];
61        } else {
62            return fail(&format!("invalid mock report option: {argument}"));
63        }
64    }
65    Ok(())
66}
67fn clean_arguments(arguments: &[String]) -> Result<(), String> {
68    // cargo-llvm-cov refuses `--no-clean` beside `--no-report` or `--no-run`,
69    // which already keep earlier profiles and build artifacts. Options after
70    // `--` belong to the test binary.
71    let own = arguments
72        .iter()
73        .position(|arg| arg == "--")
74        .map_or(arguments, |end| &arguments[..end]);
75    for flag in ["--no-report", "--no-run"] {
76        if has(own, flag) && has(own, "--no-clean") {
77            return fail(&format!("{flag} may not be used together with --no-clean"));
78        }
79    }
80    Ok(())
81}
82fn physical(arguments: &[String]) -> Result<(), String> {
83    let target = if has(arguments, "--lib") {
84        "lib"
85    } else {
86        value(arguments, "--test").ok_or("missing mock test target")?
87    };
88    let position = arguments
89        .iter()
90        .position(|arg| arg == "--exact")
91        .ok_or("missing exact filter")?;
92    let filters = &arguments[position + 1..];
93    let fields: Vec<_> = [
94        include_str!("../../src/coverage/physical-selection.txt"),
95        include_str!("../../src/coverage/physical-selection-vulkan.txt"),
96    ]
97    .into_iter()
98    .flat_map(str::lines)
99    .map(|line| line.split('|').collect::<Vec<_>>())
100    .find(|fields| {
101        fields[1] == target
102            && fields[3]
103                .split_whitespace()
104                .any(|name| filters.iter().any(|filter| filter == name))
105    })
106    .ok_or("unexpected physical target")?;
107    let group = fields[0];
108    let mut mode = variable("COVERAGE_TEST_PHYSICAL", "passed");
109    if variable("COVERAGE_TEST_PHYSICAL_GROUP", "wgpu-lib") != group {
110        mode = "passed".into();
111    }
112    if mode == "failed" {
113        return fail("simulated physical test failure");
114    }
115    let mut selected: Vec<String> = fields[3]
116        .split_whitespace()
117        .filter(|name| filters.iter().any(|arg| arg == name))
118        .map(str::to_owned)
119        .collect();
120    match mode.as_str() {
121        "missing" => {
122            selected.pop();
123        }
124        "empty" => selected.clear(),
125        "wrong-name" => {
126            *selected.last_mut().ok_or("no selected tests")? = "unrelated_physical_test".into();
127        }
128        "duplicate-name" => {
129            let first = selected.first().ok_or("no selected tests")?.clone();
130            *selected.last_mut().unwrap() = first;
131        }
132        _ => {}
133    }
134    for (index, name) in selected.iter().enumerate() {
135        println!("test {name} ... adapter=qualified-fixture");
136        if mode != "missing-outcome" || index + 1 != selected.len() {
137            println!(
138                "{}",
139                if mode == "failed-outcome" && index + 1 == selected.len() {
140                    "FAILED"
141                } else {
142                    "ok"
143                }
144            );
145        }
146    }
147    if target == "lib" {
148        println!(
149            "test result: ok. 0 passed; 0 failed; 0 ignored; 0 measured; 10 filtered out; finished in 0.00s"
150        );
151    }
152    if mode != "unreported" {
153        println!(
154            "test result: ok. {} passed; 0 failed; {} ignored; 0 measured; 1 filtered out; finished in 0.01s",
155            selected.len(),
156            usize::from(mode == "ignored")
157        );
158    }
159    if mode == "extra-summary" {
160        println!(
161            "test result: ok. 1 passed; 0 failed; 0 ignored; 0 measured; 0 filtered out; finished in 0.01s"
162        );
163    }
164    if mode == "failed-summary" {
165        println!(
166            "test result: FAILED. 0 passed; 1 failed; 0 ignored; 0 measured; 0 filtered out; finished in 0.01s"
167        );
168    }
169    Ok(())
170}
171fn book_artifacts() -> Result<(), String> {
172    let directory = env::current_dir()
173        .map_err(|error| error.to_string())?
174        .join("target/debug/deps");
175    fs::create_dir_all(&directory).map_err(|error| error.to_string())?;
176    for name in [
177        "zetesis",
178        "zetesis_cli",
179        "zetesis_solve",
180        "zetesis_validation",
181    ] {
182        let file = directory.join(format!("lib{name}-current.rlib"));
183        fs::write(&file, b"current").map_err(|error| error.to_string())?;
184        println!(
185            "{}",
186            json!({"reason":"compiler-artifact","target":{"name":name,"kind":["lib"]},"filenames":[file],"fresh":true})
187        );
188    }
189    println!("{}", json!({"reason":"build-finished","success":true}));
190    Ok(())
191}

193fn cargo(arguments: &[String]) -> Result<(), String> {
194    if env::var_os("CHECK_TEST_TRACE").is_some() {
195        let failed = variable("CHECK_TEST_ORACLE_FAILURE", "");
196        // A campaign names its tests as a target (`--test name`) or as a module
197        // of a crate's one integration binary (the filter `name::`).
198        let selects = |name: &str| {
199            arguments
200                .windows(2)
201                .any(|pair| pair[0] == "--test" && pair[1] == name)
202                || arguments
203                    .iter()
204                    .any(|argument| argument.strip_suffix("::") == Some(name))
205        };
206        if arguments.first().is_some_and(|argument| argument == "test")
207            && failed
208                .split(',')
209                .any(|name| !name.is_empty() && selects(name))
210        {
211            return fail("simulated oracle campaign failure");
212        }
213        if has(arguments, "--message-format=json-render-diagnostics") {
214            book_artifacts()?;
215        }
216        return Ok(());
217    }
218    if arguments
219        .get(..2)
220        .is_some_and(|args| args == ["+1.97.1", "test"])
221        && has(arguments, "--ignored")
222    {
223        // The hardware gate's exact test runs, without instrumentation.
224        return physical(arguments);
225    }
226    if arguments == ["+1.97.1", "llvm-cov", "--version"] {
227        println!(
228            "{}",
229            variable("COVERAGE_TEST_CARGO_VERSION", "cargo-llvm-cov 0.8.7")
230        );
231        return Ok(());
232    }
233    if arguments
234        .get(..2)
235        .is_none_or(|args| args != ["+1.97.1", "llvm-cov"])
236    {
237        return fail("unexpected mock Cargo invocation");
238    }
239    report_arguments(arguments)?;
240    clean_arguments(arguments)?;
241    let directory = env::var("CARGO_LLVM_COV_TARGET_DIR").map_err(|error| error.to_string())?;
242    let profile = Path::new(&directory).file_name().unwrap().to_str().unwrap();
243    let log = env::var("COVERAGE_TEST_LOG").map_err(|error| error.to_string())?;
244    writeln!(
245        OpenOptions::new()
246            .create(true)
247            .append(true)
248            .open(log)
249            .unwrap(),
250        "{}",
251        json!({"args":arguments,"profile":profile})
252    )
253    .unwrap();
254    let phase = if has(arguments, "--fail-under-lines") {
255        "gate"
256    } else if has(arguments, "--html") {
257        "html"
258    } else if has(arguments, "--json") {
259        "json"
260    } else if has(arguments, "clean") {
261        "clean"
262    } else {
263        "test"
264    };
265    let failure = variable("COVERAGE_TEST_FAIL", "");
266    if failure == phase || failure == format!("{profile}:{phase}") {
267        return fail(&format!("simulated failure {phase}"));
268    }
269    if has(arguments, "--ignored") {
270        physical(arguments)?;
271    }
272    if let Some(path) = value(arguments, "--output-path") {
273        fs::write(path, "{}\n").unwrap();
274    }
275    if let Some(path) = value(arguments, "--output-dir") {
276        let path = Path::new(path).join("html");
277        fs::create_dir_all(&path).unwrap();
278        fs::write(path.join("index.html"), "mock HTML").unwrap();
279    }
280    Ok(())
281}
282/// Whether `CHECK_TEST_FAILURE` asks this call of `role` to fail: a Lean build
283/// or audit, the proof record, or the check of the oracle campaigns' runs.
284fn simulated_failure(role: &str, arguments: &[String]) -> bool {
285    let failure = variable("CHECK_TEST_FAILURE", "");
286    let command = arguments.first().map(String::as_str);
287    (role == "lake"
288        && ((arguments == ["build"] && failure == "build")
289            || (has(arguments, "Audit.lean") && failure == "audit")))
290        || (role == "maintenance" && command == Some("proof-record") && failure == "record")
291        || (role == "maintenance" && command == Some("oracle-runs") && failure == "oracle-runs")
292}
293fn execute(role: &str, arguments: &[String]) -> Result<(), String> {
294    trace(role, arguments);
295    if let Some(expected) = env::var_os("CHECK_TEST_EXPECT_DIRECTORY")
296        && env::current_dir().map_err(|error| error.to_string())? != Path::new(&expected)
297    {
298        return fail("tool started outside the expected repository");
299    }
300    if let Ok(expected) = env::var("CHECK_TEST_BOOK_TOOLCHAIN")
301        && ((role == "mdbook" && has(arguments, "test"))
302            || (role == "cargo" && has(arguments, "--message-format=json-render-diagnostics")))
303        && env::var("RUSTUP_TOOLCHAIN").ok().as_deref() != Some(expected.as_str())
304    {
305        return fail("temporary book compiler selected a different toolchain");
306    }
307    if role == "lake"
308        && has(arguments, "Audit.lean")
309        && env::var_os("CHECK_TEST_AUDIT_STDERR").is_some()
310    {
311        eprintln!("synthetic unexpected Audit stderr");
312    }
313    if simulated_failure(role, arguments) {
314        return fail("simulated check failure");
315    }
316    match role {
317        "cargo" => cargo(arguments),
318        "rustup" if arguments == ["show", "active-toolchain"] => {
319            println!(
320                "{} (overridden by repository toolchain)",
321                variable("CHECK_TEST_BOOK_TOOLCHAIN", "1.97.1-fake-host")
322            );
323            Ok(())
324        }
325        "rustc" if arguments == ["+1.97.1", "--print", "sysroot"] => {
326            println!("{}", env::var("COVERAGE_TEST_SYSROOT").unwrap());
327            Ok(())
328        }
329        "rustc" if arguments == ["+1.97.1", "-vV"] => {
330            println!("rustc 1.97.1\nhost: fake-host\nLLVM version: 22.1.6");
331            Ok(())
332        }
333        "llvm" => {
334            println!(
335                "LLVM (http://llvm.org/):\n  LLVM version {}\n  Optimized build.",
336                variable("COVERAGE_TEST_LLVM_VERSION", "22.1.6")
337            );
338            Ok(())
339        }
340        "clingo" if arguments == ["--version"] => {
341            if variable("CHECK_TEST_CLINGO_FAILURE", "") == "version" {
342                return fail("simulated clingo version failure");
343            }
344            println!(
345                "clingo version {}",
346                variable("CHECK_TEST_CLINGO_VERSION", "5.8.2")
347            );
348            Ok(())
349        }
350        "mdbook" => {
351            if arguments == ["--version"] {
352                println!("mdbook v0.5.4");
353            }
354            if has(arguments, "test") && env::var_os("CHECK_TEST_BOOK_ARTIFACTS").is_some() {
355                let path =
356                    Path::new(value(arguments, "--library-path").ok_or("missing book view")?);
357                let entries: std::collections::BTreeSet<_> = fs::read_dir(path)
358                    .map_err(|error| error.to_string())?
359                    .map(|entry| {
360                        entry
361                            .map(|entry| entry.file_name())
362                            .map_err(|error| error.to_string())
363                    })
364                    .collect::<Result<_, _>>()?;
365                let expected = [
366                    "libzetesis-current.rlib",
367                    "libzetesis_cli-current.rlib",
368                    "libzetesis_solve-current.rlib",
369                    "libzetesis_validation-current.rlib",
370                ]
371                .map(std::ffi::OsString::from)
372                .into_iter()
373                .collect();
374                if entries != expected {
375                    return fail("rustdoc received stale or missing libraries");
376                }
377            }
378            Ok(())
379        }
380        "maintenance" | "lake" | "coverage.sh" | "validate.sh" => Ok(()),
381        _ => fail("unexpected mock tool invocation"),
382    }
383}
384fn main() -> ExitCode {
385    let arguments: Vec<_> = env::args().skip(1).collect();
386    let executable = env::args().next().unwrap();
387    let name = Path::new(&executable)
388        .file_name()
389        .and_then(|name| name.to_str())
390        .unwrap();
391    let result = if ["lake", "lean", "rustc", "cargo", "zetesis-maintenance"].contains(&name) {
392        capture_fixture::run(name, &arguments)
393    } else {
394        arguments.split_first().map_or_else(
395            || fail("missing tool role"),
396            |(role, arguments)| execute(role, arguments),
397        )
398    };
399    match result {
400        Ok(()) => ExitCode::SUCCESS,
401        Err(reason) => {
402            eprintln!("{reason}");
403            ExitCode::from(if env::var_os("CHECK_TEST_TRACE").is_some() {
404                23
405            } else {
406                37
407            })
408        }
409    }
