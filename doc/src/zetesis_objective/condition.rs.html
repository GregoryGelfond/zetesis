12/// One operation in an acyclic query over a complete model.
13/// Every operand index must precede the operation that references it. The
14/// default atom operand is an owned construction description; admitted reads
15/// borrow an `AtomRef` and canonical storage retains only an occurrence index.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum ConditionNode<A = Atom> {
18    /// Constant logical truth.
19    Boolean(bool),
20    /// Membership of a complete typed atom, including its classical sign.
21    Atom(A),
22    /// Logical negation of an earlier result.
23    Not(usize),
24    /// Conjunction of earlier results.
25    And(usize, usize),
26    /// Disjunction of earlier results.
27    Or(usize, usize),
28}
29
30/// Atom operands are positions in the exact catalog supplied to construction.
31pub type ConditionIndex = ConditionNode<usize>;
32/// A logical operation borrowing its complete atom identity.
33pub type ConditionNodeRef<'a> = ConditionNode<AtomRef<'a>>;
34
35impl<A> ConditionNode<A> {
36    fn as_ref(&self) -> ConditionNode<&A> {
37        match self {
38            Self::Boolean(value) => ConditionNode::Boolean(*value),
39            Self::Atom(atom) => ConditionNode::Atom(atom),
40            Self::Not(operand) => ConditionNode::Not(*operand),
41            Self::And(left, right) => ConditionNode::And(*left, *right),
42            Self::Or(left, right) => ConditionNode::Or(*left, *right),
43        }
44    }
45    fn try_map<B, E>(self, atom: impl FnOnce(A) -> Result<B, E>) -> Result<ConditionNode<B>, E> {
46        Ok(match self {
47            Self::Boolean(value) => ConditionNode::Boolean(value),
48            Self::Atom(value) => ConditionNode::Atom(atom(value)?),
49            Self::Not(operand) => ConditionNode::Not(operand),
50            Self::And(left, right) => ConditionNode::And(left, right),
51            Self::Or(left, right) => ConditionNode::Or(left, right),
52        })
53    }
54}

56/// Invalid catalog coordinates, backward references or named storage.
57#[derive(Clone, Debug, PartialEq, Eq)]
58pub enum ConditionError {
59    /// An atom occurrence is outside the supplied catalog.
60    AtomReference {
61        /// Operation containing the occurrence.
62        node: usize,
63        /// Supplied occurrence position.
64        atom: usize,
65        /// Catalog occurrence count.
66        atoms: usize,
67    },
68    /// An operation references itself or a later operation.
69    Reference {
70        /// Operation containing the reference.
71        node: usize,
72        /// Operand that must precede it.
73        operand: usize,
74    },
75    /// Canonical or node storage could not be admitted.
76    Storage(CatalogError),
77}
78impl fmt::Display for ConditionError {
79    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
80        match self {
81            Self::AtomReference { node, atom, atoms } => write!(
82                f,
83                "condition node {node} references atom {atom} outside catalog of {atoms} atoms"
84            ),
85            Self::Reference { node, operand } => write!(
86                f,
87                "condition node {node} references nonpreceding operand {operand}"
88            ),
89            Self::Storage(error) => error.fmt(f),
90        }
91    }
92}
93impl std::error::Error for ConditionError {
94    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
95        match self {
96            Self::Storage(error) => Some(error),
97            _ => None,
98        }
99    }
100}

102/// Checked condition construction preserves the caller's exact stop cause.
103#[derive(Debug)]
104pub enum ConditionFailure<E> {
105    /// Invalid coordinates or unavailable storage.
106    Condition(ConditionError),
107    /// The caller refused before the next operation.
108    Stopped(E),
109}
110impl<E: fmt::Display> fmt::Display for ConditionFailure<E> {
111    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
112        match self {
113            Self::Condition(error) => error.fmt(f),
114            Self::Stopped(error) => error.fmt(f),
115        }
116    }
117}
118impl<E: std::error::Error + 'static> std::error::Error for ConditionFailure<E> {
119    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
120        match self {
121            Self::Condition(error) => Some(error),
122            Self::Stopped(error) => Some(error),
123        }
124    }
125}

137/// A closed model query whose final node is its result; an empty query is true.
138/// `new` retains construction input until `ObjectiveProgram` admission consumes
139/// it. Every admitted condition stores ID-only nodes over shared canonical
140/// payload. Catalog-backed construction preserves the supplied authority.
141/// Cloning an admitted condition shares all its nodes and payload in O(1).
142#[derive(Clone)]
143pub struct Condition(Source);
144impl Default for Condition {
145    fn default() -> Self {
146        Self::new(Vec::new())
147    }
148}
149impl Condition {
150    /// Assemble owned input for subsequent objective-program admission.
151    /// This operation does not establish valid references or execution storage.
152    #[must_use]
153    pub fn new(nodes: Vec<ConditionNode>) -> Self {
154        Self(Source::Ingress(nodes))
155    }
156
157    /// Validate ID-only nodes over the supplied exact atom occurrence catalog.
158    /// No payload is copied. `max_bytes` includes the complete retained catalog,
159    /// node capacity and shared condition envelope; allocator bookkeeping and
160    /// Arc counters are excluded. Caller input buffers are already allocated.
161    /// Work callbacks precede each node, reference and catalog-measure step.
162    ///
163    /// # Errors
164    /// Returns invalid atom/backward coordinates, storage refusal, or the exact
165    /// caller stop. No partial condition escapes; the supplied values are consumed.
166    pub fn from_catalog_with<E>(
167        atoms: AtomCatalog,
168        nodes: Vec<ConditionIndex>,
169        max_bytes: usize,
170        mut before: impl FnMut() -> Result<(), E>,
171    ) -> Result<Self, ConditionFailure<E>> {
172        for (index, node) in nodes.iter().enumerate() {
173            before().map_err(ConditionFailure::Stopped)?;
174            let operands = match node {
175                ConditionNode::Boolean(_) => 0,
176                ConditionNode::Atom(_) | ConditionNode::Not(_) => 1,
177                ConditionNode::And(..) | ConditionNode::Or(..) => 2,
178            };
179            for _ in 0..operands {
180                before().map_err(ConditionFailure::Stopped)?;
181            }
182            check_node(node.as_ref(), index, |id| {
183                if *id < atoms.atoms().len() {
184                    Ok(())
185                } else {
186                    Err(ConditionError::AtomReference {
187                        node: index,
188                        atom: *id,
189                        atoms: atoms.atoms().len(),
190                    })
191                }
192            })
193            .map_err(ConditionError::from)
194            .map_err(ConditionFailure::Condition)?;
195        }
196        let bytes = atoms
197            .storage_with(&mut before)
198            .map_err(ConditionFailure::Stopped)?
199            .bytes
200            + node_bytes(nodes.capacity());
201        storage_bound(bytes, max_bytes).map_err(ConditionFailure::Condition)?;
202        before().map_err(ConditionFailure::Stopped)?;
203        Ok(Self(Source::Canonical(Arc::new(Canonical {
204            atoms,
205            nodes,
206        }))))
207    }
208
209    /// Borrow logical operations without materializing canonical atoms.
210    #[must_use]
211    pub fn nodes(&self) -> ConditionNodes<'_> {
212        ConditionNodes(&self.0)
213    }
214
215    pub(crate) fn validate(
216        &self,
217        limits: AdmissionLimits,
218        template: usize,
219    ) -> Result<(), AdmissionError> {
220        super::program::check_bound(
221            AdmissionResource::ConditionNodes,
222            self.nodes().len(),
223            limits.max_condition_nodes,
224            Some(template),
225        )?;
226        for (index, node) in self.nodes().iter().enumerate() {
227            check_node(node, index, |atom| {
228                super::program::check_bound(
229                    AdmissionResource::PredicateArity,
230                    atom.values().len(),
231                    limits.max_predicate_arity,
232                    Some(template),
233                )
234            })
235            .map_err(|error| match error {
236                NodeError::Atom(error) => error,
237                NodeError::Reference { node, operand } => AdmissionError::ConditionReference {
238                    template,
239                    node,
240                    operand,
241                },
242            })?;
243        }
244        Ok(())
245    }
246
247    /// Owned atom descriptions still awaiting the objective's shared vocabulary.
248    pub(crate) fn ingress_atoms(&self) -> impl Iterator<Item = AtomRef<'_>> {
249        let nodes = match &self.0 {
250            Source::Ingress(nodes) => nodes.as_slice(),
251            Source::Canonical(_) => &[],
252        };
253        nodes.iter().filter_map(|node| match node {
254            ConditionNode::Atom(atom) => Some(atom.into()),
255            _ => None,
256        })
257    }
258
259    pub(crate) fn canonical_catalog(&self) -> Option<&AtomCatalog> {
260        match &self.0 {
261            Source::Canonical(data) => Some(&data.atoms),
262            Source::Ingress(_) => None,
263        }
264    }
265
266    /// Named storage of canonical condition nodes, excluding their atom catalog.
267    /// For a catalog-backed condition this is its shared condition envelope and
268    /// actual node-buffer capacity, counted once across clones. Allocator
269    /// bookkeeping and Arc counters are excluded. Owned construction input reports
270    /// the proposed canonical envelope and one cell per node; this does not measure
271    /// its still-owned atom payload or spare ingress capacity.
272    #[must_use]
273    pub fn node_storage_bytes(&self) -> u128 {
274        match &self.0 {
275            Source::Canonical(data) => node_bytes(data.nodes.capacity()),
276            Source::Ingress(nodes) => node_bytes(nodes.len()),
277        }
278    }
279
280    pub(crate) fn same_node_owner(&self, other: &Self) -> bool {
281        matches!((&self.0, &other.0), (Source::Canonical(left), Source::Canonical(right)) if Arc::ptr_eq(left, right))
282    }

284    /// Import rows into the objective's shared tuple authority. The supplied
285    /// owner allowance excludes all other live objective metadata; this operation
286    /// additionally admits its own node buffer before allocation and row import.
287    pub(crate) fn prepare_in(
288        self,
289        owner: &mut AtomInterner,
290        limits: InternLimits,
291        node_limit: usize,
292    ) -> Result<PendingCondition, ConditionError> {
293        match self.0 {
294            Source::Canonical(data) => {
295                storage_bound(node_bytes(data.nodes.capacity()), node_limit)?;
296                Ok(PendingCondition::Existing(Self(Source::Canonical(data))))
297            }
298            Source::Ingress(nodes) => {
299                storage_bound(node_bytes(nodes.len()), node_limit)?;
300                storage_bound_u128(
301                    owner.storage_bytes() + node_bytes(nodes.len()),
302                    limits.max_bytes,
303                )?;
304                let mut indices = Vec::new();
305                indices
306                    .try_reserve_exact(nodes.len())
307                    .map_err(|_| ConditionError::Storage(CatalogError::Allocation))?;
308                let extra = node_bytes(indices.capacity());
309                storage_bound(extra, node_limit)?;
310                storage_bound_u128(owner.storage_bytes() + extra, limits.max_bytes)?;
311                let limits = InternLimits {
312                    max_bytes: limits.max_bytes - extra,
313                    ..limits
314                };
315                for node in nodes {
316                    indices.push(node.try_map(|atom| {
317                        owner
318                            .entry_atom_with(&atom, limits, || Ok::<_, Infallible>(()))
319                            .map_err(intern_error)?
320                            .insert_with(limits, || Ok::<_, Infallible>(()))
321                            .map_err(intern_error)
322                    })?);
323                }
324                Ok(PendingCondition::Nodes(indices))
325            }
326        }
327    }
328}

330/// Private admission state: coordinates address the single writer used by
331/// `prepare_in`, and finish receives that writer's discovery-order publication.
332pub(crate) enum PendingCondition {
333    Existing(Condition),
334    Nodes(Vec<ConditionIndex>),
335}
336impl PendingCondition {
337    pub(crate) fn node_storage_bytes(&self) -> u128 {
338        match self {
339            Self::Existing(condition) => condition.node_storage_bytes(),
340            Self::Nodes(nodes) => node_bytes(nodes.capacity()),
341        }
342    }
343    pub(crate) fn finish(self, atoms: &AtomCatalog) -> Result<Condition, ConditionError> {
344        match self {
345            Self::Existing(condition) => Ok(condition),
346            Self::Nodes(nodes) => {
347                for (index, node) in nodes.iter().enumerate() {
348                    check_node(node.as_ref(), index, |id| {
349                        if *id < atoms.atoms().len() {
350                            Ok(())
351                        } else {
352                            Err(ConditionError::AtomReference {
353                                node: index,
354                                atom: *id,
355                                atoms: atoms.atoms().len(),
356                            })
357                        }
358                    })
359                    .map_err(ConditionError::from)?;
360                }
361                Ok(Condition(Source::Canonical(Arc::new(Canonical {
362                    atoms: atoms.clone(),
363                    nodes,
364                }))))
365            }
366        }
367    }
368}

370// One reference check serves both owned input admission and catalog construction.
371enum NodeError<E> {
372    Atom(E),
373    Reference { node: usize, operand: usize },
374}
375fn check_node<A, E>(
376    node: ConditionNode<A>,
377    index: usize,
378    atom: impl FnOnce(A) -> Result<(), E>,
379) -> Result<(), NodeError<E>> {
380    let reference = |operand| {
381        if operand < index {
382            Ok(())
383        } else {
384            Err(NodeError::Reference {
385                node: index,
386                operand,
387            })
388        }
389    };
390    match node {
391        ConditionNode::Boolean(_) => Ok(()),
392        ConditionNode::Atom(value) => atom(value).map_err(NodeError::Atom),
393        ConditionNode::Not(operand) => reference(operand),
394        ConditionNode::And(left, right) | ConditionNode::Or(left, right) => {
395            reference(left)?;
396            reference(right)
397        }
398    }
399}
400impl<E> From<NodeError<E>> for ConditionError
401where
402    E: Into<ConditionError>,
403{
404    fn from(error: NodeError<E>) -> Self {
405        match error {
406            NodeError::Atom(error) => error.into(),
407            NodeError::Reference { node, operand } => Self::Reference { node, operand },
408        }
409    }
410}
411
412fn node_bytes(capacity: usize) -> u128 {
413    size_of::<Canonical>() as u128 + capacity as u128 * size_of::<ConditionIndex>() as u128
414}
415fn storage_bound(required: u128, limit: usize) -> Result<(), ConditionError> {
416    if required > limit as u128 {
417        Err(ConditionError::Storage(CatalogError::Storage {
418            required,
419            limit,
420        }))
421    } else {
422        Ok(())
423    }
424}
425
426fn storage_bound_u128(required: u128, limit: u128) -> Result<(), ConditionError> {
427    if required > limit {
428        Err(ConditionError::Storage(CatalogError::Storage {
429            required,
430            limit: usize::try_from(limit).unwrap_or(usize::MAX),
431        }))
432    } else {
433        Ok(())
434    }
435}
436
437fn intern_error(error: InternFailure<Infallible>) -> ConditionError {
438    ConditionError::Storage(match error {
439        InternFailure::Catalog(error) => error,
440        InternFailure::Allocation(_) => CatalogError::Allocation,
441        InternFailure::Bytes { required, limit } => CatalogError::Storage {
442            required,
443            limit: usize::try_from(limit).unwrap_or(usize::MAX),
444        },
445        InternFailure::Overflow | InternFailure::Atoms { .. } => CatalogError::Overflow,
446        InternFailure::Stopped(never) => match never {},
447    })
448}

450/// A borrowed logical node sequence, preserving the original operation order.
451#[derive(Clone, Copy)]
452pub struct ConditionNodes<'a>(&'a Source);
453impl<'a> ConditionNodes<'a> {
454    /// Number of query operations.
455    #[must_use]
456    pub fn len(self) -> usize {
457        match self.0 {
458            Source::Ingress(nodes) => nodes.len(),
459            Source::Canonical(data) => data.nodes.len(),
460        }
461    }
462    /// Whether the query is the empty, true query.
463    #[must_use]
464    pub fn is_empty(self) -> bool {
465        self.len() == 0
466    }
467    /// Borrow one operation, or return absence outside the sequence.
468    ///
469    /// # Panics
470    /// Panics if a canonical condition refers outside its retained atom catalog.
471    /// Condition publication validates every occurrence before exposing a view.
472    #[must_use]
473    pub fn at(self, index: usize) -> Option<ConditionNodeRef<'a>> {
474        let result = match self.0 {
475            Source::Ingress(nodes) => nodes
476                .get(index)?
477                .as_ref()
478                .try_map(|atom| Ok::<_, Infallible>(atom.into())),
479            Source::Canonical(data) => data.nodes.get(index)?.as_ref().try_map(|id| {
480                Ok::<_, Infallible>(
481                    data.atoms
482                        .atoms()
483                        .at(*id)
484                        .expect("validated condition occurrence"),
485                )
486            }),
487        };
488        Some(match result {
489            Ok(node) => node,
490            Err(never) => match never {},
491        })
492    }
493    /// Traverse the sequence; cloning the cursor retains only its borrow.
494    #[must_use]
495    pub fn iter(self) -> ConditionIter<'a> {
496        ConditionIter {
497            nodes: self,
498            index: 0,
499        }
500    }
501}
502impl<'a> IntoIterator for ConditionNodes<'a> {
503    type Item = ConditionNodeRef<'a>;
504    type IntoIter = ConditionIter<'a>;
505    fn into_iter(self) -> Self::IntoIter {
506        self.iter()
507    }
508}
509/// Exact forward traversal of borrowed condition operations.
510#[derive(Clone)]
511pub struct ConditionIter<'a> {
512    nodes: ConditionNodes<'a>,
513    index: usize,
514}
515impl<'a> Iterator for ConditionIter<'a> {
516    type Item = ConditionNodeRef<'a>;
517    fn next(&mut self) -> Option<Self::Item> {
518        let node = self.nodes.at(self.index)?;
519        self.index += 1;
520        Some(node)
521    }
522    fn size_hint(&self) -> (usize, Option<usize>) {
523        let remaining = self.nodes.len() - self.index;
524        (remaining, Some(remaining))
525    }
526}
527impl ExactSizeIterator for ConditionIter<'_> {}
528impl FusedIterator for ConditionIter<'_> {}
529impl fmt::Debug for Condition {
530    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
531        f.debug_list().entries(self.nodes()).finish()
532    }
533}
534impl PartialEq for Condition {
535    fn eq(&self, other: &Self) -> bool {
536        self.nodes().len() == other.nodes().len() && self.nodes().iter().eq(other.nodes())
537    }
538}
