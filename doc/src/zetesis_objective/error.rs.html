3/// Per-model ceilings, independent of source admission and stable-model search.
4#[derive(Clone, Copy, Debug)]
5pub struct Limits {
6    /// Charged join, equality, tuple, key-comparison, shift and accounting work.
7    pub max_work: u64,
8    /// Complete positive joins visited, including bindings rejected by filters.
9    pub max_bindings: u64,
10    /// Distinct global contribution keys retained.
11    pub max_keys: usize,
12    /// Canonical encoded payload bytes of distinct keys; not allocator overhead.
13    pub max_key_bytes: usize,
14}

16    fn default() -> Self {
17        Self {
18            max_work: 10_000_000,
19            max_bindings: 1_000_000,
20            max_keys: 1_000_000,
21            max_key_bytes: 67_108_864,
22        }
23    }
24}

43/// An interrupted evaluation, never a proof about optimum or satisfiability.
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum Stop {
46    /// Shared cancellation was observed.
47    Cancelled,
48    /// The absolute deadline expired.
49    Deadline,
50    /// The charged operation ceiling was reached.
51    WorkLimit,
52    /// The complete-binding ceiling was reached.
53    BindingLimit,
54    /// Adding another unique key would exceed the count ceiling.
55    KeyLimit,
56    /// Adding another unique key would exceed the payload-byte ceiling.
57    KeyBytesLimit,
58    /// Fallible storage reservation failed.
59    Allocation,
60    /// Size or accounting arithmetic cannot be represented on this host.
61    ArithmeticOverflow,
62    /// An unexpected shared-control refusal, retained without reclassification.
63    Control(zetesis_cpu::Stop),
64}
65impl fmt::Display for Stop {
66    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
67        match self {
68            Self::Cancelled => f.write_str("objective evaluation cancelled"),
69            Self::Deadline => f.write_str("objective evaluation deadline expired"),
70            Self::WorkLimit => f.write_str("objective work limit reached"),
71            Self::BindingLimit => f.write_str("objective binding limit reached"),
72            Self::KeyLimit => f.write_str("objective contribution-key limit reached"),
73            Self::KeyBytesLimit => f.write_str("objective key-byte limit reached"),
74            Self::Allocation => f.write_str("objective storage reservation failed"),
75            Self::ArithmeticOverflow => f.write_str("objective accounting arithmetic overflow"),
76            Self::Control(reason) => reason.fmt(f),
77        }
78    }
79}

82/// The typed reason an evaluator cannot return a complete score.
83#[derive(Clone, Copy, Debug, PartialEq, Eq)]
84pub enum ErrorKind {
85    /// A control, work, storage or representation bound stopped evaluation.
86    Stopped(Stop),
87    /// A negated numeric weight is outside the signed 32-bit contribution range.
88    WeightNormalizationOverflow,
89    /// The exact total at one priority is outside the signed 64-bit score range.
90    CostOverflow {
91        /// The affected priority.
92        priority: i32,
93    },
94    /// Admitted safety was violated internally while resolving a local variable.
95    UnboundVariable {
96        /// The unbound local variable index.
97        variable: usize,
98    },
99}

101/// An evaluation failure with source-template correlation and partial statistics.
102/// No partial objective score is exposed as a successful result.
103#[derive(Clone, Copy, Debug, PartialEq, Eq)]
104pub struct Error {
105    pub(crate) kind: ErrorKind,
106    pub(crate) template: Option<usize>,
107    pub(crate) statistics: Statistics,
108}
109impl Error {
110    /// The structured refusal or stopped resource.
111    #[must_use]
112    pub const fn kind(&self) -> ErrorKind {
113        self.kind
114    }
115    /// Original template index for the caller's separate source-origin catalog.
116    /// Final cross-template cost overflow has no single originating template.
117    #[must_use]
118    pub const fn template_index(&self) -> Option<usize> {
119        self.template
120    }
121    /// Work completed before the failure; these counts imply no complete score.
122    #[must_use]
123    pub const fn statistics(&self) -> Statistics {
124        self.statistics
125    }
126}
127impl fmt::Display for Error {
128    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
129        match self.kind {
130            ErrorKind::Stopped(reason) => reason.fmt(f)?,
131            ErrorKind::WeightNormalizationOverflow => {
132                f.write_str("objective numeric weight negation exceeds signed 32-bit range")?;
133            }
134            ErrorKind::CostOverflow { priority } => write!(
135                f,
136                "objective cost at priority {priority} exceeds signed 64-bit range"
137            )?,
138            ErrorKind::UnboundVariable { variable } => {
139                write!(f, "objective evaluation lacks local variable {variable}")?;
140            }
141        }
142        if let Some(template) = self.template {
143            write!(f, " (template {template})")?;
144        }
145        Ok(())
146    }
147}
148impl std::error::Error for Error {
149    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
150        match &self.kind {
151            ErrorKind::Stopped(reason) => Some(reason),
152            _ => None,
153        }
154    }
