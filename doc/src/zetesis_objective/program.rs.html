16/// Numeric normalization applied before constructing a global contribution key.
17/// All scores remain minimization costs, including negated maximize weights.
18#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
19pub enum WeightPolarity {
20    /// Retain the resolved numeric weight (minimize and weak constraints).
21    #[default]
22    AsWritten,
23    /// Negate the resolved numeric weight using checked i32 arithmetic.
24    Negated,
25}
26impl WeightPolarity {
27    /// Normalize a numeric contribution. `None` means negating `i32::MIN`
28    /// cannot be represented; it never means an ignored nonnumeric term.
29    #[must_use]
30    pub const fn normalize(self, weight: i32) -> Option<i32> {
31        match self {
32            Self::AsWritten => Some(weight),
33            Self::Negated => weight.checked_neg(),
34        }
35    }
36}

38/// A lifted minimization element with positive relational bindings and an
39/// optional closed model query. Variable IDs are local; comparisons and closed
40/// conditions never establish variable safety.
41#[derive(Clone, Debug, PartialEq, Eq)]
42pub struct ObjectiveTemplate {
43    weight: Term,
44    polarity: WeightPolarity,
45    priority: i32,
46    tuple: Vec<Term>,
47    positive: Vec<AtomPattern>,
48    filters: Vec<Filter>,
49    condition: Condition,
50}
51impl ObjectiveTemplate {
52    /// Assemble a template. Program admission checks dimensions and safety.
53    /// An omitted source priority is represented by the caller as zero.
54    #[must_use]
55    pub fn new(
56        weight: Term,
57        priority: i32,
58        tuple: Vec<Term>,
59        positive: Vec<AtomPattern>,
60        filters: Vec<Filter>,
61    ) -> Self {
62        Self {
63            weight,
64            polarity: WeightPolarity::AsWritten,
65            priority,
66            tuple,
67            positive,
68            filters,
69            condition: Condition::default(),
70        }
71    }
72    /// Set the numeric normalization while retaining the lifted source term.
73    /// Normalization precedes global key deduplication in every evaluator.
74    #[must_use]
75    pub fn with_weight_polarity(mut self, polarity: WeightPolarity) -> Self {
76        self.polarity = polarity;
77        self
78    }
79    /// Require this closed query in addition to the positive bindings and
80    /// scalar filters. It reads the original supplied model and has no effect
81    /// on this element's admitted priority slot or complete contribution key.
82    #[must_use]
83    pub fn with_condition(mut self, condition: Condition) -> Self {
84        self.condition = condition;
85        self
86    }
87    /// Closed model query; an empty query is true.
88    #[must_use]
89    pub const fn condition(&self) -> &Condition {
90        &self.condition
91    }
92    /// Numeric normalization applied to the resolved source weight.
93    #[must_use]
94    pub const fn weight_polarity(&self) -> WeightPolarity {
95        self.polarity
96    }
97    /// Original scalar weight, required numeric only for an active contribution.
98    /// Apply [`Self::weight_polarity`] before interpreting a numeric key or cost.
99    #[must_use]
100    pub const fn weight(&self) -> &Term {
101        &self.weight
102    }
103    /// Fixed priority; larger priorities are compared first.
104    #[must_use]
105    pub const fn priority(&self) -> i32 {
106        self.priority
107    }
108    /// Explicit tuple following the weight and priority.
109    #[must_use]
110    pub fn tuple(&self) -> &[Term] {
111        &self.tuple
112    }
113    /// Ordinary positive conditions joined against supplied model atoms.
114    #[must_use]
115    pub fn positive(&self) -> &[AtomPattern] {
116        &self.positive
117    }
118    /// Exact equality or inequality comparisons on bound scalar values.
119    #[must_use]
120    pub fn filters(&self) -> &[Filter] {
121        &self.filters
122    }
123    /// Validate scoped data fields before a source priority has been resolved.
124    /// Returns the number of densely numbered, positively bound variables.
125    /// Numeric priority values do not affect these shape and safety obligations.
126    /// `index` identifies the caller's original element in indexed errors.
127    ///
128    /// A source frontend may call this before evaluating its priority expression.
129    /// That frontend owns expression safety, numeric evaluation and the complete
130    /// priority carrier; every expression input must have an independent binder.
131    /// The fields are borrowed and remain caller-owned. After resolving a numeric
132    /// priority, construct a template and admit it through [`ObjectiveProgram::new`].
133    ///
134    /// # Errors
135    /// Refuses shape limits, unsafe or sparse variable IDs, overflow or allocation.
136    pub fn validate_fields(
137        weight: &Term,
138        tuple: &[Term],
139        positive: &[AtomPattern],
140        filters: &[Filter],
141        limits: AdmissionLimits,
142        index: usize,
143    ) -> Result<usize, AdmissionError> {
144        admit_fields(weight, tuple, positive, filters, limits, index)
145    }
146
147    /// Check tuple width, positive body count, filter count and predicate arity.
148    /// This does not establish variable safety or dense IDs. A source frontend
149    /// with additional binding forms must establish their scope independently;
150    /// resolved templates still require [`ObjectiveProgram::new`].
151    ///
152    /// # Errors
153    /// Refuses a configured shape ceiling before source values are evaluated.
154    pub fn validate_shape(
155        tuple_width: usize,
156        positive: &[AtomPattern],
157        filter_count: usize,
158        limits: AdmissionLimits,
159        index: usize,
160    ) -> Result<(), AdmissionError> {
161        validate_shape(
162            tuple_width,
163            positive.iter().map(PatternRef::from),
164            filter_count,
165            limits,
166            index,
167        )
168    }
169
170    /// Validate the relational scope and output width before evaluating source
171    /// expressions. The source frontend must independently establish that every
172    /// expression input has a binder in `positive`; expressions never add binders.
173    /// This shares shape, dense-variable and filter checks with `validate_fields`.
174    /// Resolved output values must still pass through [`ObjectiveProgram::new`].
175    ///
176    /// # Errors
177    /// Refuses shape limits, unsafe or sparse variable IDs, overflow or allocation.
178    pub fn validate_scope(
179        tuple_width: usize,
180        positive: &[AtomPattern],
181        filters: &[Filter],
182        limits: AdmissionLimits,
183        index: usize,
184    ) -> Result<usize, AdmissionError> {
185        admit_scope(
186            tuple_width,
187            positive
188                .iter()
189                .flat_map(AtomPattern::terms)
190                .chain(filters.iter().flat_map(|filter| {
191                    let (left, right) = filter.terms();
192                    [left, right]
193                }))
194                .map(TemplateTerm::from),
195            positive.iter().map(PatternRef::from),
196            filters.len(),
197            limits,
198            index,
199        )
200    }
201}

203fn terms<'a>(
204    weight: &'a Term,
205    tuple: &'a [Term],
206    positive: &'a [AtomPattern],
207    filters: &'a [Filter],
208) -> impl Iterator<Item = &'a Term> + Clone {
209    std::iter::once(weight)

211        .chain(positive.iter().flat_map(AtomPattern::terms))

316/// An objective program cannot enter the evaluator unless these checks pass.
317#[derive(Clone, Debug, PartialEq, Eq)]
318pub enum AdmissionError {
319    /// An objective occurrence names no row of the supplied catalog.
320    CatalogRow {
321        /// Original objective occurrence.
322        template: usize,
323        /// Requested catalog row.
324        row: usize,
325    },
326    /// A referenced catalog row has no scalar weight field.
327    MissingWeight {
328        /// Original objective occurrence.
329        template: usize,
330        /// Referenced catalog row.
331        row: usize,
332    },
333    /// The catalog entry point cannot import owned condition descriptions.
334    ConditionNotCanonical {
335        /// Original objective occurrence.
336        template: usize,
337    },
338    /// Combined canonical vocabulary or retained metadata could not be admitted.
339    Storage {
340        /// Original template where known.
341        template: Option<usize>,
342        /// Exact canonical/storage refusal.
343        error: zetesis_core::catalog::Error,
344    },
345    /// A closed condition's canonical representation could not be admitted.
346    ConditionStorage {
347        /// Original input template.
348        template: usize,
349        /// Exact coordinate or canonical storage cause.
350        error: crate::ConditionError,
351    },
352    /// A configured shape ceiling was exceeded.
353    Limit {
354        /// Refused dimension.
355        resource: AdmissionResource,
356        /// Original input template, or `None` for the program-wide limit.
357        template: Option<usize>,
358        /// Observed or proposed count.
359        actual: usize,
360        /// Inclusive ceiling.
361        limit: usize,
362    },
363    /// A weight/tuple/filter variable has no positive binding condition.
364    UnsafeVariable {
365        /// Original input template.
366        template: usize,
367        /// Unbound local variable.
368        variable: usize,
369    },
370    /// Local variable indices have a gap below a used index.
371    NonDenseVariable {
372        /// Original input template.
373        template: usize,
374        /// Missing local index.
375        variable: usize,
376    },
377    /// A variable index or count cannot be incremented on this host.
378    Overflow {
379        /// Original input template, if known.
380        template: Option<usize>,
381    },
382    /// Fallible admission storage reservation failed.
383    Allocation,
384    /// The internal priority registry lacks a previously collected template.
385    MissingPriority {
386        /// Original input template whose priority could not be resolved.
387        template: usize,
388    },
389    /// A closed model query references its own or a later operation.
390    ConditionReference {
391        /// Original input template.
392        template: usize,
393        /// Operation containing the invalid reference.
394        node: usize,
395        /// Operand that must precede the operation.
396        operand: usize,
397    },
398}
399impl AdmissionError {
400    /// Resolve a refusal into the caller's parallel source-origin catalog.
401    #[must_use]
402    pub const fn template_index(&self) -> Option<usize> {
403        match *self {
404            Self::Storage { template, .. }
405            | Self::Limit { template, .. }
406            | Self::Overflow { template } => template,
407            Self::CatalogRow { template, .. }
408            | Self::MissingWeight { template, .. }
409            | Self::ConditionNotCanonical { template }
410            | Self::UnsafeVariable { template, .. }
411            | Self::NonDenseVariable { template, .. }
412            | Self::MissingPriority { template }
413            | Self::ConditionStorage { template, .. }
414            | Self::ConditionReference { template, .. } => Some(template),
415            Self::Allocation => None,
416        }
417    }
418}
419impl fmt::Display for AdmissionError {
420    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
421        match self {
422            Self::CatalogRow { template, row } => write!(
423                f,
424                "objective template {template} references missing catalog row {row}"
425            ),
426            Self::MissingWeight { template, row } => write!(
427                f,
428                "objective template {template} catalog row {row} has no weight field"
429            ),
430            Self::ConditionNotCanonical { template } => write!(
431                f,
432                "objective template {template} requires a canonical condition for catalog admission"
433            ),
434            Self::Storage { template, error } => {
435                write!(f, "objective storage (template {template:?}): {error}")
436            }
437            Self::ConditionStorage { template, error } => {
438                write!(f, "objective template {template} condition: {error}")
439            }
440            Self::Limit {
441                resource,
442                template,
443                actual,
444                limit,
445            } => write!(
446                f,
447                "objective {resource:?} count {actual} exceeds {limit} (template {template:?})"
448            ),
449            Self::UnsafeVariable { template, variable } => write!(
450                f,
451                "objective template {template} variable {variable} has no positive binding"
452            ),
453            Self::NonDenseVariable { template, variable } => write!(
454                f,
455                "objective template {template} is missing local variable {variable}"
456            ),
457            Self::Overflow { template } => write!(
458                f,
459                "objective admission arithmetic overflow (template {template:?})"
460            ),
461            Self::Allocation => f.write_str("objective admission storage reservation failed"),
462            Self::MissingPriority { template } => {
463                write!(f, "objective priority registry lacks template {template}")
464            }
465            Self::ConditionReference {
466                template,
467                node,
468                operand,
469            } => write!(
470                f,
471                "objective template {template} condition node {node} references nonpreceding operand {operand}"
472            ),
473        }
474    }

495struct AdmittedTemplate {
496    row: usize,
497    polarity: WeightPolarity,
498    condition: Condition,
499    variables: usize,

502impl Data {
503    fn row(&self, template: &AdmittedTemplate) -> TemplateRow<'_> {
504        self.catalog
505            .as_ref()
506            .and_then(|catalog| catalog.at(template.row))
507            .expect("objective metadata names its complete canonical component row")
508    }
509    fn fields(&self, template: &AdmittedTemplate) -> (TemplateTerm<'_>, PatternTerms<'_>) {
510        let fields = self.row(template).terms();
511        (
512            fields
513                .at(0)
514                .expect("objective admission stores a weight before its tuple"),
515            fields
516                .slice(1..fields.len())
517                .expect("the remaining scalar fields are the objective tuple"),
518        )
519    }
520    fn priority(&self, template: &AdmittedTemplate) -> i32 {
521        self.priorities[template.slot]
522    }

525/// An immutable collection of objective metadata over canonical template storage.
526/// It supplies costs, never logical support. Equal complete contribution keys
527/// from separate templates/directives are globally coalesced during evaluation.
528#[derive(Clone, Debug)]
529pub struct ObjectiveProgram(Arc<Data>);
530impl ObjectiveProgram {
531    /// Consume original-order descriptions and admit one canonical vocabulary.
532    /// Owned condition input shares that vocabulary; supplied canonical condition
533    /// catalogs retain their original authority. Source presence and complete
534    /// priority discovery remain the frontend's responsibility.
535    ///
536    /// # Errors
537    /// Refuses shape/safety, invalid condition references, combined named storage,
538    /// overflow or fallible reservation. No partial `ObjectiveProgram` escapes.
539    pub fn new(
540        templates: Vec<ObjectiveTemplate>,
541        limits: AdmissionLimits,
542    ) -> Result<Self, AdmissionError> {
543        admission::admit(templates, limits).map(|data| Self(Arc::new(data)))
544    }
545    /// Admit objective metadata over existing canonical components without
546    /// importing or copying their payload. Row occurrences remain in descriptor
547    /// order and may repeat. Each row stores its weight before its tuple fields.
548    /// Positive patterns alone establish dense variable safety; filters and
549    /// closed conditions do not. Empty conditions mean true; nonempty conditions
550    /// must already have a canonical atom catalog.
551    ///
552    /// The byte ceiling covers the retained catalog, occurrence/priority buffers,
553    /// unique condition allocations and simultaneous safety scratch. Exact shared
554    /// snapshots are counted once; partially shared prefixes are conservative.
555    /// Caller descriptor storage and allocator/Arc bookkeeping are excluded.
556    ///
557    /// # Errors
558    /// Refuses missing rows or weights, owned nonempty conditions, shape/safety,
559    /// condition references, named storage, overflow or fallible reservation.
560    pub fn from_catalog(
561        catalog: TemplateCatalog,
562        rows: Vec<ObjectiveElement>,
563        limits: AdmissionLimits,
564    ) -> Result<Self, AdmissionError> {
565        admission::from_catalog(catalog, rows, limits).map(|data| Self(Arc::new(data)))
566    }
567
568    /// An absent objective, distinct from a present empty or zero-cost objective.
569    #[must_use]
570    pub fn none() -> Self {
571        Self(Arc::new(Data {
572            catalog: None,
573            templates: Vec::new(),
574            priorities: Vec::new(),
575            present: false,
576            storage_bytes: size_of::<Data>() as u128,
577        }))
578    }
579    /// Whether an objective was admitted, independently of current cost.
580    #[must_use]
581    pub fn is_present(&self) -> bool {
582        self.0.present
583    }
584    /// Original template occurrences as borrowed canonical views.
585    #[must_use]
586    pub fn templates(&self) -> ObjectiveTemplates<'_> {
587        ObjectiveTemplates(&self.0)
588    }
589    /// Fixed distinct descending priority slots, including inactive templates.
590    #[must_use]
591    pub fn priorities(&self) -> &[i32] {
592        &self.0.priorities
593    }
594    /// Named retained canonical and metadata storage. Shared clones name the
595    /// same allocations. Independent source catalogs can share segments beyond
596    /// their exact occurrence owner and are conservatively counted separately.
597    #[must_use]
598    pub fn storage_bytes(&self) -> u128 {
599        self.0.storage_bytes
600    }
601    pub(crate) fn variables(&self, index: usize) -> usize {
602        self.0.templates[index].variables
603    }
604    pub(crate) fn slot(&self, index: usize) -> usize {
605        self.0.templates[index].slot
606    }
607}

609/// One admitted objective element, borrowing all typed terms and patterns.
610#[derive(Clone, Copy, Debug)]
611pub struct ObjectiveTemplateRef<'a> {
612    data: &'a Data,
613    template: &'a AdmittedTemplate,
614}
615impl PartialEq for ObjectiveTemplateRef<'_> {
616    fn eq(&self, other: &Self) -> bool {
617        self.weight() == other.weight()
618            && self.weight_polarity() == other.weight_polarity()
619            && self.priority() == other.priority()
620            && self.tuple().iter().eq(other.tuple())
621            && self.positive().iter().eq(other.positive())
622            && self.filters().iter().eq(other.filters())
623            && self.condition() == other.condition()
624    }
625}
626impl Eq for ObjectiveTemplateRef<'_> {}
627impl<'a> ObjectiveTemplateRef<'a> {
628    /// Check a borrowed source scope's shape without copying logical terms.
629    /// This checks tuple width, body size, filters and predicate arity; the
630    /// frontend remains responsible for the safety of additional binding forms.
631    /// # Errors
632    /// Refuses a configured shape ceiling.
633    pub fn validate_shape(
634        tuple_width: usize,
635        positive: impl Iterator<Item = PatternRef<'a>> + Clone,
636        filter_count: usize,
637        limits: AdmissionLimits,
638        index: usize,
639    ) -> Result<(), AdmissionError> {
640        validate_shape(tuple_width, positive, filter_count, limits, index)
641    }
642
643    /// Validate borrowed positive patterns and filters before source expression
644    /// evaluation. Constants stay in the caller's canonical vocabulary.
645    /// The caller establishes that expression inputs have independent binders.
646    /// # Errors
647    /// Refuses shape limits, unsafe or sparse variable IDs, storage or allocation.
648    pub fn validate_scope(
649        tuple_width: usize,
650        positive: impl Iterator<Item = PatternRef<'a>> + Clone,
651        filters: impl Iterator<Item = FilterRef<'a>> + Clone,
652        limits: AdmissionLimits,
653        index: usize,
654    ) -> Result<usize, AdmissionError> {
655        let filter_count = filters.clone().count();
656        let terms = positive
657            .clone()
658            .flat_map(|pattern| pattern.terms().iter())
659            .chain(filters.flat_map(|filter| {
660                let (left, right) = filter.terms();
661                [left, right]
662            }));
663        admit_scope(tuple_width, terms, positive, filter_count, limits, index)
664    }
665
666    /// Original scalar weight; numeric normalization follows substitution.
667    #[must_use]
668    pub fn weight(self) -> TemplateTerm<'a> {
669        self.data.fields(self.template).0
670    }
671    /// Checked normalization applied before contribution-key deduplication.
672    #[must_use]
673    pub fn weight_polarity(self) -> WeightPolarity {
674        self.template.polarity
675    }
676    /// Admitted priority, retained even when this element has no active binding.
677    #[must_use]
678    pub fn priority(self) -> i32 {
679        self.data.priority(self.template)
680    }
681    /// Original tuple fields after the weight, including repeated occurrences.
682    #[must_use]
683    pub fn tuple(self) -> PatternTerms<'a> {
684        self.data.fields(self.template).1
685    }
686    /// Positive relational conditions, which alone establish variable safety.
687    #[must_use]
688    pub fn positive(self) -> Patterns<'a> {
689        self.data.row(self.template).patterns()
690    }
691    /// Exact comparison conditions; they do not establish variable safety.
692    #[must_use]
693    pub fn filters(self) -> Filters<'a> {
694        self.data.row(self.template).filters()
695    }
696    /// Closed query against the supplied original model; empty means true.
697    #[must_use]
698    pub fn condition(self) -> &'a Condition {
699        &self.template.condition
700    }
701}

711    }
712    /// Whether the objective has no element rows.
713    #[must_use]
714    pub fn is_empty(self) -> bool {
715        self.0.templates.is_empty()
716    }
717    /// Borrow one original occurrence or return None outside the view.
718    #[must_use]
719    pub fn at(self, index: usize) -> Option<ObjectiveTemplateRef<'a>> {
720        self.0
721            .templates
722            .get(index)
723            .map(|template| ObjectiveTemplateRef {
724                data: self.0,
725                template,
726            })
727    }
728    /// Same checked occurrence read as at.
729    #[must_use]
730    pub fn get(self, index: usize) -> Option<ObjectiveTemplateRef<'a>> {
731        self.at(index)
732    }
733    /// Exact double-ended traversal borrowing this owner.
734    #[must_use]
735    pub fn iter(self) -> ObjectiveTemplateIter<'a> {
736        ObjectiveTemplateIter {
737            view: self,
738            range: 0..self.len(),
739        }
740    }
741}
742/// Exact double-ended cursor over admitted objective occurrences.
743#[derive(Clone, Debug)]
744pub struct ObjectiveTemplateIter<'a> {
745    view: ObjectiveTemplates<'a>,
746    range: Range<usize>,
747}
748impl<'a> Iterator for ObjectiveTemplateIter<'a> {
749    type Item = ObjectiveTemplateRef<'a>;
750    fn next(&mut self) -> Option<Self::Item> {
751        self.range.next().and_then(|index| self.view.at(index))
752    }
753    fn size_hint(&self) -> (usize, Option<usize>) {
754        self.range.size_hint()
755    }
756}
757impl DoubleEndedIterator for ObjectiveTemplateIter<'_> {
758    fn next_back(&mut self) -> Option<Self::Item> {
759        self.range.next_back().and_then(|index| self.view.at(index))
760    }
761}
762impl ExactSizeIterator for ObjectiveTemplateIter<'_> {}
763impl FusedIterator for ObjectiveTemplateIter<'_> {}
764impl<'a> IntoIterator for ObjectiveTemplates<'a> {
765    type Item = ObjectiveTemplateRef<'a>;
766    type IntoIter = ObjectiveTemplateIter<'a>;
767    fn into_iter(self) -> Self::IntoIter {
768        self.iter()
769    }
770}

772fn reserved<T>(count: usize) -> Result<Vec<T>, AdmissionError> {
773    let mut result = Vec::new();
774    result
775        .try_reserve_exact(count)
776        .map_err(|_| AdmissionError::Allocation)?;
777    Ok(result)
778}

780pub(crate) fn check_bound(
781    resource: AdmissionResource,
782    actual: usize,
783    limit: usize,
784    template: Option<usize>,
785) -> Result<(), AdmissionError> {
786    if actual > limit {
787        Err(AdmissionError::Limit {
788            resource,
789            template,
790            actual,
791            limit,
792        })
793    } else {
794        Ok(())
795    }
796}

814fn admit_fields(
815    weight: &Term,
816    tuple: &[Term],
817    positive: &[AtomPattern],
818    filters: &[Filter],
819    limits: AdmissionLimits,
820    index: usize,
821) -> Result<usize, AdmissionError> {
822    admit_scope(
823        tuple.len(),
824        terms(weight, tuple, positive, filters).map(TemplateTerm::from),
825        positive.iter().map(PatternRef::from),
826        filters.len(),
827        limits,
828        index,
829    )
830}

832fn admit_scope<'a>(
833    tuple_width: usize,
834    terms: impl Iterator<Item = TemplateTerm<'a>> + Clone,
835    positive: impl Iterator<Item = PatternRef<'a>> + Clone,
836    filter_count: usize,
837    limits: AdmissionLimits,
838    index: usize,
839) -> Result<usize, AdmissionError> {
840    validate_shape(tuple_width, positive.clone(), filter_count, limits, index)?;
841    let mut count = 0;
842    for term in terms.clone() {
843        if let TemplateTerm::Variable(variable) = term {
844            let proposed = variable.checked_add(1).ok_or(AdmissionError::Overflow {
845                template: Some(index),
846            })?;
847            check_bound(
848                AdmissionResource::Variables,
849                proposed,
850                limits.max_variables_per_template,
851                Some(index),
852            )?;
853            count = count.max(proposed);
854        }
855    }
856    let headers = 2 * size_of::<Vec<bool>>() as u128;
857    admission::check_storage(headers + 2 * count as u128, limits.max_bytes, Some(index))?;
858    let mut used = reserved(count)?;
859    admission::check_storage(
860        headers + used.capacity() as u128 + count as u128,
861        limits.max_bytes,
862        Some(index),
863    )?;
864    used.resize(count, false);
865    let mut bound = reserved(count)?;
866    admission::check_storage(
867        headers + used.capacity() as u128 + bound.capacity() as u128,
868        limits.max_bytes,
869        Some(index),
870    )?;
871    bound.resize(count, false);
872    for term in terms {
873        if let TemplateTerm::Variable(variable) = term {
874            used[variable] = true;
875        }
876    }
877    for term in positive.flat_map(PatternRef::terms) {
878        if let TemplateTerm::Variable(variable) = term {
879            bound[variable] = true;
880        }
881    }
882    for variable in 0..count {
883        if !used[variable] {
884            return Err(AdmissionError::NonDenseVariable {
885                template: index,
886                variable,
887            });
888        }
889        if !bound[variable] {
890            return Err(AdmissionError::UnsafeVariable {
891                template: index,
892                variable,
893            });
894        }
895    }
896    Ok(count)
897}
898
899fn validate_shape<'a>(
900    tuple_width: usize,
901    positive: impl Iterator<Item = PatternRef<'a>> + Clone,
902    filter_count: usize,
903    limits: AdmissionLimits,
904    index: usize,
905) -> Result<(), AdmissionError> {
906    for (resource, actual, limit) in [
907        (
908            AdmissionResource::TupleWidth,
909            tuple_width,
910            limits.max_tuple_width,
911        ),
912        (
913            AdmissionResource::PositiveBody,
914            positive.clone().count(),
915            limits.max_positive_body,
916        ),
917        (AdmissionResource::Filters, filter_count, limits.max_filters),
918    ] {
919        check_bound(resource, actual, limit, Some(index))?;
920    }
921    for pattern in positive {
922        check_bound(
923            AdmissionResource::PredicateArity,
924            pattern.terms().len(),
925            limits.max_predicate_arity,
926            Some(index),
927        )?;
928    }
929    Ok(())
930}
