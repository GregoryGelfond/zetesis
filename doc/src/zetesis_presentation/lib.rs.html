1//! Human presentation with explicit capabilities and no environment discovery.
2//!
3//! Logical results and measurements belong to their domain libraries. This
4//! module renders caller-supplied text without interpreting solver semantics.
5//! Tables retain all cell content and use a vertical layout when the columns do
6//! not fit. Callers bound the number and size of rows before constructing a view.

17/// Styling policy. Process adapters resolve terminal and environment evidence.
18/// Generic library writers leave automatic styling disabled.
19#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
20#[cfg_attr(feature = "clap", derive(clap::ValueEnum))]
21pub enum ColorMode {
22    /// Style supported terminals; generic writers remain plain.
23    #[default]
24    Auto,
25    /// Emit ANSI color and emphasis.
26    Always,
27    /// Emit plain text.
28    Never,
29}
30
31/// Meaning of a styled fragment, independent of its content.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum Role {
34    /// A table heading or metadata label, in blue.
35    Label,
36    /// Auxiliary values and units, in italic gray.
37    Metadata,
38    /// A total or terminal result, in bold italic gray.
39    Conclusion,
40    /// Explicitly requested table data, in upright gray.
41    Data,
42    /// A table total, in bold upright gray.
43    Total,
44    /// An optimization label or value, in italic green.
45    Objective,
46    /// A recoverable diagnostic, in yellow.
47    Warning,
48    /// A failed operation, in red.
49    Error,
50}
51
52impl ColorMode {
53    /// Resolve explicit capabilities. An explicit override takes precedence over
54    /// environment conventions. This function performs no I/O.
55    #[must_use]
56    pub const fn resolve(self, terminal: bool, disabled: bool) -> Self {
57        match self {
58            Self::Auto if terminal && !disabled => Self::Always,
59            Self::Auto => Self::Never,
60            mode => mode,
61        }
62    }
63
64    /// Machine-readable output has no terminal styling.
65    #[must_use]
66    pub const fn human(self, json: bool) -> Self {
67        if json { Self::Never } else { self }
68    }
69
70    /// Write a styled fragment. Content is caller-owned and is not interpreted.
71    ///
72    /// # Errors
73    /// Returns the writer's error, including a partial write.
74    pub fn styled(
75        self,
76        output: &mut impl Write,
77        role: Role,
78        value: fmt::Arguments<'_>,
79    ) -> io::Result<()> {
80        if self == Self::Always {
81            write!(output, "{}{value}{RESET}", role.prefix())
82        } else {
83            output.write_fmt(value)
84        }
85    }
86
87    /// Write a numbered answer heading.
88    ///
89    /// # Errors
90    /// Returns the writer's error.
91    pub fn answer(self, output: &mut impl Write, number: usize) -> io::Result<()> {
92        if self == Self::Always {
93            writeln!(output, "\u{1b}[1;36mAnswer:\u{1b}[22;36m {number}{RESET}")
94        } else {
95            writeln!(output, "Answer: {number}")
96        }
97    }
98
99    /// Begin an optimization line; finish it with `Self::objective_end`.
100    ///
101    /// # Errors
102    /// Returns the writer's error.
103    pub fn objective(self, output: &mut impl Write) -> io::Result<()> {
104        if self == Self::Always {
105            write!(output, "{}Optimization:", Role::Objective.prefix())
106        } else {
107            write!(output, "Optimization:")
108        }
109    }
110
111    /// End a previously begun optimization line.
112    ///
113    /// # Errors
114    /// Returns the writer's error.
115    pub fn objective_end(self, output: &mut impl Write) -> io::Result<()> {
116        if self == Self::Always {
117            write!(output, "{RESET}")?;
118        }
119        writeln!(output)
120    }
121
122    /// Write an untagged terminal result.
123    ///
124    /// # Errors
125    /// Returns the writer's error.
126    pub fn status(self, output: &mut impl Write, text: &str) -> io::Result<()> {
127        self.styled(output, Role::Conclusion, format_args!("{text}"))?;
128        writeln!(output)
129    }
130
131    /// Write a blue label followed by italic gray metadata.
132    ///
133    /// # Errors
134    /// Returns the writer's error.
135    pub fn metadata(
136        self,
137        output: &mut impl Write,
138        label: &str,
139        value: fmt::Arguments<'_>,
140    ) -> io::Result<()> {
141        if self == Self::Always {
142            writeln!(
143                output,
144                "{}{label}:{} {value}{RESET}",
145                Role::Label.prefix(),
146                Role::Metadata.prefix()
147            )
148        } else {
149            writeln!(output, "{label}: {value}")
150        }
151    }
152}
153
154impl Role {
155    const fn prefix(self) -> &'static str {
156        match self {
157            Self::Label => "\u{1b}[34m",
158            Self::Metadata => "\u{1b}[3;90m",
159            Self::Conclusion => "\u{1b}[1;3;90m",
160            Self::Data => "\u{1b}[90m",
161            Self::Total => "\u{1b}[1;90m",
162            Self::Objective => "\u{1b}[3;32m",
163            Self::Warning => "\u{1b}[33m",
164            Self::Error => "\u{1b}[31m",
165        }
166    }
167}
168
169const RESET: &str = "\u{1b}[0m";
