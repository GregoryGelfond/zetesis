1//! Bounded lifted reference: source templates compose binding, exact filtering,
2//! frozen-seed gating, projection, union, and least closure. No carrier expansion
3//! or ground-rule table occurs in `check`. Independent grounding is test-only.
4use crate::Result;
5use std::collections::{BTreeMap, BTreeSet};

7/// Closed-domain scalar identity; text and symbols remain distinct.
8#[derive(Clone, Debug, Eq, PartialEq, Ord, PartialOrd)]
9pub enum Value {
10    /// Signed finite integer.
11    Int(i32),
12    /// Text value whose bytes determine identity.
13    Text(String),
14    /// Symbol value whose bytes determine identity.
15    Symbol(String),
16}
17/// A template variable or a closed-domain constant.
18#[derive(Clone, Debug, Eq, PartialEq)]
19pub enum Term {
20    /// Variable identifier local to the template.
21    Var(usize),
22    /// Fixed scalar value.
23    Const(Value),
24}
25/// Predicate application whose arguments may contain template variables.
26#[derive(Clone, Debug, Eq, PartialEq)]
27pub struct AtomPattern {
28    /// Index in the program's predicate catalog.
29    pub predicate: usize,
30    /// Ordered predicate arguments.
31    pub terms: Vec<Term>,
32}
33/// Ground predicate application with exact logical scalar identity.
34#[derive(Clone, Debug, Eq, PartialEq, Ord, PartialOrd)]
35pub struct Atom {
36    /// Index in the program's predicate catalog.
37    pub predicate: usize,
38    /// Ordered ground argument values.
39    pub tuple: Vec<Value>,
40}
41/// Equality or disequality applied once both terms are bound.
42#[derive(Clone, Debug, Eq, PartialEq)]
43pub enum Filter {
44    /// Require identical logical values.
45    Eq(Term, Term),
46    /// Require different logical values.
47    Neq(Term, Term),
48}
49/// Safe lifted rule with positive joins and frozen reduct gates.
50#[derive(Clone, Debug)]
51pub struct Template {
52    /// Derived head, or absence for an integrity constraint.
53    pub head: Option<AtomPattern>,
54    /// Positive body applications supplying all variable bindings.
55    pub positive: Vec<AtomPattern>,
56    /// Applications required present in the frozen seed.
57    pub gate_true: Vec<AtomPattern>,
58    /// Applications required absent from the frozen seed.
59    pub gate_false: Vec<AtomPattern>,
60    /// Logical comparisons applied during binding.
61    pub filters: Vec<Filter>,
62}
63/// Predicate identity within the finite source signature.
64#[derive(Clone, Debug)]
65pub struct Predicate {
66    /// Predicate name; identity includes arity.
67    pub name: String,
68    /// Number of ordered arguments.
69    pub arity: usize,
70}
71/// Validated templates over a finite closed domain and predicate signature.
72#[derive(Clone, Debug)]
73pub struct LiftedProgram {
74    domain: BTreeSet<Value>,
75    predicates: Vec<Predicate>,
76    templates: Vec<Template>,
77    gate_predicates: BTreeSet<usize>,
78}
79/// Independent execution ceilings; zero never means unlimited.
80#[derive(Clone, Copy, Debug)]
81pub struct Limits {
82    /// Maximum synchronous closure passes, including the final quiescent pass.
83    pub max_rounds: usize,
84    /// Maximum charged binding entries and tuple probes.
85    pub max_work: u64,
86    /// Maximum distinct derived atoms retained across relations and the delta.
87    pub max_atoms: usize,
88}
89impl Default for Limits {
90    fn default() -> Self {
91        Self {
92            max_rounds: 1000,
93            max_work: 1_000_000,
94            max_atoms: 100_000,
95        }
96    }
97}
98/// Actual inference work, distinct from the symbolic carrier size.
99#[derive(Clone, Debug, Default)]
100pub struct LiftedStats {
101    /// Synchronous closure passes executed.
102    pub rounds: usize,
103    /// Relation tuples inspected during positive joins.
104    pub tuple_probes: u64,
105    /// Complete bindings published after their filters and gates held.
106    pub complete_bindings: u64,
107    /// Partial or complete bindings rejected by a bound filter or gate.
108    pub guard_prunes: u64,
109    /// Charged binding entries plus tuple probes.
110    pub work: u64,
111    /// Materialized symbolic-carrier tuples; always zero for this source engine.
112    pub carrier_tuples_enumerated: u64,
113    /// Retained ground-rule rows; always zero for this source engine.
114    pub ground_rows_materialized: u64,
115}
116/// Complete lifted reduct result with its inference accounting.
117#[derive(Clone, Debug)]
118pub struct LiftedCheck {
119    /// Exact least closure of all enabled positive source instances.
120    pub closure: BTreeSet<Atom>,
121    /// Whether closure agrees with the seed and violates no active constraint.
122    pub stable: bool,
123    /// Whether a complete enabled constraint binding was found.
124    pub constraint_violated: bool,
125    /// Work performed by this complete check.
126    pub stats: LiftedStats,
127}
128type Binding = BTreeMap<usize, Value>;
129type Relations = Vec<BTreeSet<Vec<Value>>>;

131/// Reference-only capability limit bounding the recursive join stack.
132pub const MAX_POSITIVE_BODY: usize = 64;
133
134fn term_value<'a>(t: &'a Term, b: &'a Binding) -> Option<&'a Value> {
135    match t {
136        Term::Const(v) => Some(v),
137        Term::Var(i) => b.get(i),
138    }
139}
140fn instantiate(p: &AtomPattern, b: &Binding) -> Option<Atom> {
141    Some(Atom {
142        predicate: p.predicate,
143        tuple: p
144            .terms
145            .iter()
146            .map(|t| term_value(t, b).cloned())
147            .collect::<Option<_>>()?,
148    })
149}
150fn filter_truth(f: &Filter, b: &Binding) -> Option<bool> {
151    let (a, c, equal) = match f {
152        Filter::Eq(a, c) => (a, c, true),
153        Filter::Neq(a, c) => (a, c, false),
154    };
155    Some((term_value(a, b)? == term_value(c, b)?) == equal)
156}
157fn charge(stats: &mut LiftedStats, limits: Limits) -> Result<()> {
158    stats.work = stats
159        .work
160        .checked_add(1)
161        .ok_or("incomplete: work counter overflow")?;
162    if stats.work > limits.max_work {
163        return Err("incomplete: lifted work budget".into());
164    }
165    Ok(())
166}

168impl LiftedProgram {
169    /// Validate and own source templates over the supplied closed domain.
170    ///
171    /// # Errors
172    /// Rejects duplicate predicate identities, excessive positive-body length,
173    /// unsafe variables, constants outside the domain, and invalid arities or
174    /// predicate indices. Domain duplicates are coalesced by logical identity.
175    pub fn new(
176        domain: Vec<Value>,
177        predicates: Vec<Predicate>,
178        templates: Vec<Template>,
179    ) -> Result<Self> {
180        let domain: BTreeSet<_> = domain.into_iter().collect();
181        let mut identities = BTreeSet::new();
182        for p in &predicates {
183            if !identities.insert((&p.name, p.arity)) {
184                return Err("duplicate predicate identity".into());
185            }
186        }
187        let mut gate_predicates = BTreeSet::new();
188        for r in &templates {
189            if r.positive.len() > MAX_POSITIVE_BODY {
190                return Err("reference capability: positive body exceeds 64 atoms".into());
191            }
192            let bound: BTreeSet<_> = r
193                .positive
194                .iter()
195                .flat_map(|p| &p.terms)
196                .filter_map(|t| if let Term::Var(v) = t { Some(*v) } else { None })
197                .collect();
198            let validate_term = |t: &Term| -> Result<()> {
199                match t {
200                    Term::Var(v) if !bound.contains(v) => {
201                        Err("unsafe variable: no positive body binding".into())
202                    }
203                    Term::Const(v) if !domain.contains(v) => {
204                        Err("constant outside closed domain".into())
205                    }
206                    _ => Ok(()),
207                }
208            };
209            for pattern in r
210                .head
211                .iter()
212                .chain(&r.positive)
213                .chain(&r.gate_true)
214                .chain(&r.gate_false)
215            {
216                if predicates
217                    .get(pattern.predicate)
218                    .is_none_or(|p| p.arity != pattern.terms.len())
219                {
220                    return Err("invalid predicate or pattern arity".into());
221                }
222                for t in &pattern.terms {
223                    validate_term(t)?;
224                }
225            }
226            for f in &r.filters {
227                let (a, b) = match f {
228                    Filter::Eq(a, b) | Filter::Neq(a, b) => (a, b),
229                };
230                validate_term(a)?;
231                validate_term(b)?;
232            }
233            for p in r.gate_true.iter().chain(&r.gate_false) {
234                gate_predicates.insert(p.predicate);
235            }
236        }
237        Ok(Self {
238            domain,
239            predicates,
240            templates,
241            gate_predicates,
242        })
243    }
244    fn validate_seed(&self, seed: &BTreeSet<Atom>) -> Result<()> {
245        for a in seed {
246            if !self.gate_predicates.contains(&a.predicate)
247                || self
248                    .predicates
249                    .get(a.predicate)
250                    .is_none_or(|p| p.arity != a.tuple.len())
251                || a.tuple.iter().any(|v| !self.domain.contains(v))
252            {
253                return Err("sparse seed atom outside symbolic gate carrier".into());
254            }
255        }
256        Ok(())
257    }
258    /// A frozen finite true set is total on S: missing carrier tuples are false.
259    /// Errors, including budget exhaustion, return no accepted/rejected value.
260    ///
261    /// # Errors
262    /// Rejects an out-of-carrier seed, exhausted round/work/atom limits, counter
263    /// overflow, or an internal head whose variables were not bound.
264    pub fn check(&self, seed: &BTreeSet<Atom>, limits: Limits) -> Result<LiftedCheck> {
265        self.validate_seed(seed)?;
266        let mut relations: Relations = vec![BTreeSet::new(); self.predicates.len()];
267        let mut stats = LiftedStats::default();
268        let mut violated = false;
269        let mut atom_count: usize = 0;
270        loop {
271            if stats.rounds == limits.max_rounds {
272                return Err("incomplete: lifted round budget".into());
273            }
274            stats.rounds += 1;
275            let mut delta = BTreeSet::new();
276            for rule in &self.templates {
277                let mut publish = |b: &Binding| -> Result<()> {
278                    if let Some(h) = &rule.head {
279                        let a = instantiate(h, b).ok_or("internal unbound head")?;
280                        if !relations[a.predicate].contains(&a.tuple) && !delta.contains(&a) {
281                            if atom_count
282                                .checked_add(delta.len())
283                                .and_then(|n| n.checked_add(1))
284                                .is_none_or(|n| n > limits.max_atoms)
285                            {
286                                return Err("incomplete: lifted atom budget".into());
287                            }
288                            delta.insert(a);
289                        }
290                    } else {
291                        violated = true;
292                    }
293                    Ok(())
294                };
295                Join {
296                    relations: &relations,
297                    seed,
298                    limits,
299                }
300                .bind(rule, 0, &Binding::new(), &mut stats, &mut publish)?;
301            }
302            if delta.is_empty() {
303                break;
304            }
305            atom_count += delta.len();
306            for a in delta {
307                relations[a.predicate].insert(a.tuple);
308            }
309        }
310        let closure: BTreeSet<_> = relations
311            .into_iter()
312            .enumerate()
313            .flat_map(|(predicate, tuples)| {
314                tuples
315                    .into_iter()
316                    .map(move |tuple| Atom { predicate, tuple })
317            })
318            .collect();
319        let projected: BTreeSet<_> = closure
320            .iter()
321            .filter(|a| self.gate_predicates.contains(&a.predicate))
322            .cloned()
323            .collect();
324        Ok(LiftedCheck {
325            stable: &projected == seed && !violated,
326            closure,
327            constraint_violated: violated,
328            stats,
329        })
330    }
331}

333/// Immutable relation snapshot and frozen seed for one synchronous source join.
334struct Join<'a> {
335    relations: &'a Relations,
336    seed: &'a BTreeSet<Atom>,
337    limits: Limits,
338}
339impl Join<'_> {
340    /// Filter and gate as soon as arguments are bound, then publish complete rows.
341    /// Recursion is bounded by the admitted positive-body length.
342    fn bind(
343        &self,
344        rule: &Template,
345        depth: usize,
346        binding: &Binding,
347        stats: &mut LiftedStats,
348        publish: &mut impl FnMut(&Binding) -> Result<()>,
349    ) -> Result<()> {
350        charge(stats, self.limits)?;
351        let blocked = rule
352            .filters
353            .iter()
354            .any(|f| filter_truth(f, binding) == Some(false))
355            || rule
356                .gate_true
357                .iter()
358                .any(|p| instantiate(p, binding).is_some_and(|a| !self.seed.contains(&a)))
359            || rule
360                .gate_false
361                .iter()
362                .any(|p| instantiate(p, binding).is_some_and(|a| self.seed.contains(&a)));
363        if blocked {
364            stats.guard_prunes += 1;
365            return Ok(());
366        }
367        if depth == rule.positive.len() {
368            stats.complete_bindings += 1;
369            return publish(binding);
370        }
371        let pattern = &rule.positive[depth];
372        for tuple in &self.relations[pattern.predicate] {
373            charge(stats, self.limits)?;
374            stats.tuple_probes += 1;
375            let mut extended = binding.clone();
376            let compatible = pattern
377                .terms
378                .iter()
379                .zip(tuple)
380                .all(|(term, value)| match term {
381                    Term::Const(v) => v == value,
382                    Term::Var(i) => {
383                        if let Some(v) = extended.get(i) {
384                            v == value
385                        } else {
386                            extended.insert(*i, value.clone());
387                            true
388                        }
389                    }
390                });
391            if compatible {
392                self.bind(rule, depth + 1, &extended, stats, publish)?;
393            }
394        }
395        Ok(())
396    }
397}

399fn pattern(predicate: usize, terms: Vec<Term>) -> AtomPattern {

402fn head_rule(head: AtomPattern, positive: Vec<AtomPattern>) -> Template {
403    Template {
404        head: Some(head),
405        positive,
406        gate_true: vec![],
407        gate_false: vec![],
408        filters: vec![],
409    }

411fn predicates(entries: &[(&str, usize)]) -> Vec<Predicate> {
412    entries
413        .iter()
414        .map(|(name, arity)| Predicate {
415            name: (*name).into(),
416            arity: *arity,
417        })
418        .collect()
419}
420fn integer_domain(size: usize) -> Result<Vec<Value>> {
421    (0..size)
422        .map(|value| {
423            i32::try_from(value)
424                .map(Value::Int)
425                .map_err(|_| "fixture domain exceeds integer carrier".into())
426        })
427        .collect()
428}
429fn diagonal_fixture(n: usize) -> Result<LiftedProgram> {
430    let domain = integer_domain(n)?;
431    let mut rules: Vec<_> = domain
432        .iter()
433        .map(|v| head_rule(pattern(0, vec![Term::Const(v.clone())]), vec![]))
434        .collect();
435    let x = Term::Var(0);
436    let dom = pattern(0, vec![x.clone()]);
437    let pick = pattern(1, vec![x.clone(), x.clone()]);
438    let mut choice = head_rule(pick.clone(), vec![dom.clone()]);
439    choice.gate_true.push(pick);
440    let mut seen = head_rule(pattern(2, vec![x.clone(), x.clone()]), vec![dom]);
441    seen.gate_false.push(pattern(3, vec![x.clone(), x]));
442    rules.extend([choice, seen]);
443    LiftedProgram::new(
444        domain,
445        predicates(&[("dom", 1), ("pick", 2), ("seen", 2), ("ban", 2)]),
446        rules,
447    )
448}

450fn pair_fixture(n: usize) -> Result<LiftedProgram> {
451    let domain = integer_domain(n)?;
452    let mut rules: Vec<_> = domain
453        .iter()
454        .map(|v| head_rule(pattern(0, vec![Term::Const(v.clone())]), vec![]))
455        .collect();
456    let x = Term::Var(0);
457    let y = Term::Var(1);
458    let dom_x = pattern(0, vec![x.clone()]);
459    let dom_y = pattern(0, vec![y.clone()]);
460    let pick_x = pattern(1, vec![x.clone()]);
461    let pick_y = pattern(1, vec![y.clone()]);
462    let mut choice = head_rule(pick_x.clone(), vec![dom_x.clone()]);
463    choice.gate_true.push(pick_x.clone());
464    let mut pair = head_rule(pattern(2, vec![x, y]), vec![dom_x, dom_y]);
465    pair.gate_true = vec![pick_x, pick_y];
466    rules.extend([choice, pair]);
467    LiftedProgram::new(
468        domain,
469        predicates(&[("dom", 1), ("pick", 1), ("pair", 2)]),
470        rules,
471    )
472}

474/// One executed sparse fixture with its symbolic size and exact closure record.
475#[derive(Clone, Debug)]
476pub struct SparseFixtureReport {
477    /// Stable fixture name used in the published report.
478    pub fixture: &'static str,
479    /// Number of distinct scalar values in its authored domain.
480    pub domain_size: usize,
481    /// Number of possible ground atoms in the symbolic seed carrier.
482    pub symbolic_seed_atoms: usize,
483    /// Number of true atoms supplied in the sparse seed.
484    pub true_seed_atoms: usize,
485    /// Complete domain substitutions for the observed source rule before gates.
486    pub target_rule_source_substitutions: usize,
487    /// Completed reduct inference and measured work counts.
488    pub check: LiftedCheck,
489}
490/// Executed by the CLI; these assert mechanism/work counts, not timing speedups.
491///
492/// # Errors
493/// Returns an error if fixture construction, bounded inference, exact model
494/// identity, or the declared sparse-work invariants fail.
495pub fn run_sparse_fixtures() -> Result<Vec<SparseFixtureReport>> {
496    let mut reports = Vec::new();
497    for n in [32, 128] {
498        let p = diagonal_fixture(n)?;
499        let check = p.check(&BTreeSet::new(), Limits::default())?;
500        let expected: BTreeSet<_> = integer_domain(n)?
501            .into_iter()
502            .flat_map(|v| {
503                [
504                    Atom {
505                        predicate: 0,
506                        tuple: vec![v.clone()],
507                    },
508                    Atom {
509                        predicate: 2,
510                        tuple: vec![v.clone(), v],
511                    },
512                ]
513            })
514            .collect();
515        if !check.stable
516            || check.closure != expected
517            || check.stats.carrier_tuples_enumerated != 0
518            || check.stats.ground_rows_materialized != 0
519            || check.stats.tuple_probes > (8 * n) as u64
520            || check.stats.complete_bindings > (8 * n) as u64
521        {
522            return Err(format!("sparse lifted fixture failed at n={n}"));
523        }
524        reports.push(SparseFixtureReport {
525            fixture: "diagonal_carrier_laziness",
526            domain_size: n,
527            symbolic_seed_atoms: 2 * n * n,
528            true_seed_atoms: 0,
529            target_rule_source_substitutions: n,
530            check,
531        });
532        let p = pair_fixture(n)?;
533        let pick = Atom {
534            predicate: 1,
535            tuple: vec![Value::Int(0)],
536        };
537        let seed = [pick.clone()].into_iter().collect();
538        let check = p.check(&seed, Limits::default())?;
539        let mut expected: BTreeSet<_> = integer_domain(n)?
540            .into_iter()
541            .map(|value| Atom {
542                predicate: 0,
543                tuple: vec![value],
544            })
545            .collect();
546        expected.insert(pick);
547        expected.insert(Atom {
548            predicate: 2,
549            tuple: vec![Value::Int(0), Value::Int(0)],
550        });
551        if !check.stable
552            || check.closure != expected
553            || check.stats.carrier_tuples_enumerated != 0
554            || check.stats.ground_rows_materialized != 0
555            || check.stats.tuple_probes > (12 * n) as u64
556            || check.stats.complete_bindings > (8 * n) as u64
557        {
558            return Err(format!("candidate-gated pair fixture failed at n={n}"));
559        }
560        reports.push(SparseFixtureReport {
561            fixture: "candidate_gated_pair_join",
562            domain_size: n,
563            symbolic_seed_atoms: n,
564            true_seed_atoms: 1,
565            target_rule_source_substitutions: n * n,
566            check,
567        });
568    }
569    Ok(reports)
570}

576    // This complete Cartesian expansion is deliberately outside the lazy engine.
577    fn products(domain: &[Value], width: usize) -> Vec<Vec<Value>> {
578        if width == 0 {
579            return vec![vec![]];
580        }
581        products(domain, width - 1)
582            .into_iter()
583            .flat_map(|prefix| {
584                domain.iter().map(move |v| {
585                    let mut tuple = prefix.clone();
586                    tuple.push(v.clone());
587                    tuple
588                })
589            })
590            .collect()
591    }
592    fn ground(p: &LiftedProgram) -> (Program, Vec<Atom>) {
593        let domain: Vec<_> = p.domain.iter().cloned().collect();
594        let atoms: Vec<_> = p
595            .predicates
596            .iter()
597            .enumerate()
598            .flat_map(|(predicate, pred)| {
599                products(&domain, pred.arity)
600                    .into_iter()
601                    .map(move |tuple| Atom { predicate, tuple })
602            })
603            .collect();
604        let index: BTreeMap<_, _> = atoms
605            .iter()
606            .cloned()
607            .enumerate()
608            .map(|(i, a)| (a, i))
609            .collect();
610        let mut rules = vec![];
611        for t in &p.templates {
612            let vars: Vec<_> = t
613                .positive
614                .iter()
615                .flat_map(|p| &p.terms)
616                .filter_map(|v| if let Term::Var(i) = v { Some(*i) } else { None })
617                .collect::<BTreeSet<_>>()
618                .into_iter()
619                .collect();
620            for values in products(&domain, vars.len()) {
621                let b: Binding = vars.iter().copied().zip(values).collect();
622                // Independently substitute, then compare concrete typed values.
623                let value = |term: &Term| match term {
624                    Term::Const(v) => v.clone(),
625                    Term::Var(i) => b[i].clone(),
626                };
627                if !t.filters.iter().all(|f| match f {
628                    Filter::Eq(a, c) => value(a) == value(c),
629                    Filter::Neq(a, c) => value(a) != value(c),
630                }) {
631                    continue;
632                }
633                let atom = |p: &AtomPattern| Atom {
634                    predicate: p.predicate,
635                    tuple: p.terms.iter().map(&value).collect(),
636                };
637                let mask = |patterns: &[AtomPattern]| {
638                    patterns
639                        .iter()
640                        .fold(0, |m, pat| m | (1u64 << index[&atom(pat)]))
641                };
642                rules.push(Rule {
643                    head: t.head.as_ref().map(|h| index[&atom(h)]),
644                    choice: false,
645                    positive: mask(&t.positive),
646                    negative: mask(&t.gate_false),
647                    double_negative: mask(&t.gate_true),
648                });
649            }
650        }
651        (Program::new(atoms.len(), rules).unwrap(), atoms)
652    }
653    fn compare(p: &LiftedProgram) {
654        let (grounded, atoms) = ground(p);
655        assert!(
656            atoms.len() <= 8,
657            "keep full-model reference genuinely small"
658        );
659        let symbolic_seed_carrier = atoms.iter().enumerate().fold(0, |m, (i, a)| {
660            m | if p.gate_predicates.contains(&a.predicate) {
661                1u64 << i
662            } else {
663                0
664            }
665        });
666        let mut models = vec![];
667        for seed in subsets(symbolic_seed_carrier) {
668            let sparse = atoms
669                .iter()
670                .enumerate()
671                .filter(|(i, _)| seed & (1u64 << i) != 0)
672                .map(|(_, a)| a.clone())
673                .collect();
674            let checked = p.check(&sparse, Limits::default()).unwrap();
675            let model = atoms.iter().enumerate().fold(0, |m, (i, a)| {
676                m | if checked.closure.contains(a) {
677                    1u64 << i
678                } else {
679                    0
680                }
681            });
682            // S may conservatively include gates removed by false filters;
683            // those extra seed bits do not influence the grounded reduct.
684            let expected = grounded.check_seed(seed & grounded.seed_carrier()).unwrap();
685            assert_eq!(model, expected.closure);
686            assert_eq!(checked.constraint_violated, expected.constraint_violated);
687            assert_eq!(
688                checked.stable,
689                model & symbolic_seed_carrier == seed && !expected.constraint_violated
690            );
691            if checked.stable {
692                models.push(model);
693            }
694        }
695        models.sort_unstable();
696        assert_eq!(models, grounded.direct_full_candidate_models());
697    }
