1//! Test support over `zetesis-themelios`: admission of a source under default
2//! or given limits, the native reference the zetesis workspace's tests compare
3//! with clingo (every answer set of a small program, found by checking each
4//! candidate against the reduct), preparation and hybrid grounding,
5//! source-bundle fixtures, and, by the resource a refusal reports, the limits
6//! an admission takes and those of three other operations whose tests span
7//! crates ([`resource_limits`] states which).
8//!
9//! [`admit`] admits a source under given formula limits, the rest at their
10//! defaults, and [`formula`] under the default limits throughout. [`exhaustive`]
11//! enumerates every subset of an
12//! admitted source's atoms, keeping those the reduct check
13//! (`zetesis_ferraris::check`) accepts, each with its objective's costs; atoms
14//! are spelled as clingo spells them ([`canonical`]). [`exhaustive_models`] and
15//! [`exhaustive_bundle_models`] retain typed atoms from the same subset reference.
16//! [`prepare`] prepares a
17//! source under the default options and limits, [`prepared`] when preparation
18//! must accept it, and [`hybrid`] grounds it for hybrid execution;
19//! [`BundleFixture`] writes a source bundle into a temporary directory for the
20//! tests that load or run one. [`resource_limits`] sets those limits by the
21//! resource a refusal reports, and reads the formula and expansion limits'
22//! defaults by resource. The crate stands above `zetesis-themelios`, so only
23//! the tests that need these references compile it. It is not published or
24//! installed.
25

39/// An atom as clingo spells it: a leading `-` for strong negation, and
40/// arguments in clingo's notation, strings quoted.
41///
42/// # Panics
43/// Panics if a string argument cannot be quoted as JSON, which never happens.
44#[must_use]
45pub fn canonical<'a>(atom: impl Into<zetesis_core::catalog::AtomRef<'a>>) -> String {
46    let atom = atom.into();
47    let sign = if atom.predicate().sign() == Sign::Negative {
48        "-"
49    } else {
50        ""
51    };
52    let name = format!("{sign}{}", atom.predicate().name());
53    let arguments: Vec<_> = atom
54        .values()
55        .iter()
56        .map(|value| match value.descriptor() {
57            zetesis_core::ValueNodeRef::Infimum => "#inf".into(),
58            zetesis_core::ValueNodeRef::Supremum => "#sup".into(),
59            zetesis_core::ValueNodeRef::Function { .. }
60            | zetesis_core::ValueNodeRef::Tuple { .. } => value.to_string(),
61            zetesis_core::ValueNodeRef::Number(value) => value.to_string(),
62            zetesis_core::ValueNodeRef::Symbol(value) => value.to_owned(),
63            zetesis_core::ValueNodeRef::String(value) => serde_json::to_string(value).unwrap(),
64        })
65        .collect();
66    if arguments.is_empty() {
67        name
68    } else {
69        format!("{name}({})", arguments.join(","))
70    }
71}

73/// `source` admitted as a formula program under the default options and
74/// expansion limits, and `limits`.
75///
76/// # Errors
77/// Returns the admission's refusal.
78pub fn admit(source: &str, limits: &FormulaLimits) -> Result<AdmittedFormula, FormulaFailure> {
79    admit_formula(
80        source.into(),
81        AdmissionOptions::default(),
82        ExpansionLimits::default(),
83        *limits,
84    )
85}

87/// `source` admitted as a formula program under the default options and
88/// limits.
89///
90/// # Panics
91/// Panics, naming `source` and the refusal, if the admission refuses it.
92#[must_use]
93pub fn formula(source: &str) -> AdmittedFormula {
94    admit(source, &FormulaLimits::default()).unwrap_or_else(|error| panic!("{source}: {error}"))
95}

119/// `source` prepared under the default options and limits and grounded for
120/// hybrid execution.
121///
122/// # Panics
123/// Panics if preparation refuses `source` or hybrid grounding refuses it.
124#[must_use]
125pub fn hybrid(source: &str) -> HybridFormula {
126    prepared(source)
127        .ground_hybrid()
128        .unwrap_or_else(|error| panic!("{source}: {error}"))
129}

131/// Every answer set of `input`, each with its objective's costs: every subset
132/// of its atoms the reduct check accepts.
133///
134/// # Panics
135/// Panics if `input` has more than twelve atoms, since the enumeration visits
136/// every subset, or if a candidate's check or evaluation is refused.
137#[must_use]
138pub fn exhaustive(input: &AdmittedFormula) -> Records {
139    let cancellation = Cancellation::default();
140    let mut records = Records::new();
141    for model in exhaustive_models(input) {
142        let evaluation = zetesis_objective::evaluate(
143            input.objectives(),
144            &model,
145            zetesis_objective::Limits::default(),
146            &cancellation,
147        )
148        .unwrap();
149        let score = evaluation.score();
150        let costs = score
151            .is_present()
152            .then(|| score.costs().iter().map(|&(_, value)| value).collect());
153        assert!(records.insert((model.atoms().iter().map(canonical).collect(), costs)));
154    }
155    records
156}

158/// Every answer set of `input`, retaining its catalog and typed atom values.
159///
160/// Enumerates every subset independently of the production candidate walk,
161/// accepting exactly the subsets admitted by the Ferraris reduct check. This
162/// shares the membership checker, not an independent membership oracle.
163///
164/// # Panics
165/// Panics if `input` has more than twelve atoms, a membership check refuses,
166/// or an accepted interpretation cannot be read from the original catalog.
167#[must_use]
168pub fn exhaustive_models(input: &AdmittedFormula) -> Vec<Model> {
169    models(input.theory(), input.atom_catalog())
170}

172/// Every answer set of a formula bundle, preserving its typed atom catalog.
173///
174/// Uses the same exhaustive subset reference as [`exhaustive_models`].
175///
176/// # Panics
177/// Panics under the same bounds and membership failures as [`exhaustive_models`].
178#[must_use]
179pub fn exhaustive_bundle_models(input: &AdmittedFormulaBundle) -> Vec<Model> {
180    models(input.theory(), input.atom_catalog())
181}

183fn models(theory: &zetesis_ferraris::Theory, catalog: &zetesis_core::AtomCatalog) -> Vec<Model> {
184    let count = catalog.atoms().len();
185    assert!(count <= 12, "small independent subset enumeration");
186    let cancellation = Cancellation::default();
187    let mut models = Vec::new();
188    for bits in 0..(1_usize << count) {
189        let candidate =
190            Interpretation::new(theory, (0..count).filter(|&atom| bits & (1 << atom) != 0))
191                .unwrap();
192        if check(theory, &candidate, Limits::default(), &cancellation)
193            .unwrap()
194            .accepted()
195        {
196            models.push(Model::from_positions(catalog, candidate.atoms()).unwrap());
197        }
198    }
199    models
200}

202#[cfg(test)]
203mod tests {
204    use super::*;

206    #[test]
207    fn a_choice_has_the_empty_and_the_full_answer_set() {
208        let admitted = admit("{a}.", &FormulaLimits::default()).unwrap();
209        let expected: Records = [([].into(), None), (["a".to_owned()].into(), None)].into();
210        assert_eq!(exhaustive(&admitted), expected);
211    }
212
213    #[test]
214    fn a_formula_is_its_source_admitted_under_the_default_limits() {
215        let admitted = admit("{a}. b :- a.", &FormulaLimits::default()).unwrap();
216        assert_eq!(exhaustive(&formula("{a}. b :- a.")), exhaustive(&admitted));
217    }
218
219    #[test]
220    fn an_objective_gives_every_answer_set_its_costs() {
221        let admitted = admit("{a}. #minimize{1:a}.", &FormulaLimits::default()).unwrap();
222        let expected: Records = [
223            ([].into(), Some(vec![0])),
224            (["a".to_owned()].into(), Some(vec![1])),
225        ]
226        .into();
227        assert_eq!(exhaustive(&admitted), expected);
228    }

230    #[test]
231    fn a_hybrid_grounding_keeps_the_prepared_atoms() {

241
242    #[test]
243    fn normal_rules_preserve_their_complete_answer_family() {
244        let admitted = formula("a :- not b. b :- not a. c :- a.");
245        let mut answer_sets: Vec<Vec<String>> = exhaustive_models(&admitted)
246            .iter()
247            .map(|model| model.atoms().iter().map(canonical).collect())
248            .collect();
249        for answer_set in &mut answer_sets {
250            answer_set.sort();
251        }
252        answer_sets.sort();
253        assert_eq!(answer_sets, [vec!["a", "c"], vec!["b"]]);
254    }
