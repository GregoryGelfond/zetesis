1//! Subject-bound native membership records, separate from public verdict data.

3use zetesis_ferraris::{Interpretation, Theory};

5use crate::{Cancellation, Check, Incomplete, Limits, StableModels};

7/// A native membership attempt retaining its exact candidate and theory.
8/// Constructed only by [`check_interpretation`]; an inconclusive attempt remains
9/// distinguishable through [`Self::verdict`]. This runtime record is not a formal
10/// proof object and makes no assertion about enumeration coverage.
11///
12/// A caller-supplied verdict cannot be attached to a candidate:
13/// ```compile_fail
14/// use zetesis_ferraris::Interpretation;
15/// use zetesis_sat::{Check, CheckedInterpretation};
16/// fn attach(candidate: Interpretation) -> CheckedInterpretation {
17///     CheckedInterpretation { candidate, check: Check::Stable }
18/// }
19/// ```
20#[derive(Clone, Debug)]
21pub struct CheckedInterpretation {
22    candidate: Interpretation,
23    check: Check,
24}
25
26impl CheckedInterpretation {
27    /// The checked subject, borrowed without allocation or evaluation.
28    #[must_use]
29    pub const fn candidate(&self) -> &Interpretation {
30        &self.candidate
31    }
32
33    /// The native result, including the reason for any inconclusive attempt.
34    #[must_use]
35    pub const fn verdict(&self) -> &Check {
36        &self.check
37    }
38
39    /// Whether the native operation accepted this exact candidate.
40    #[must_use]
41    pub fn accepted(&self) -> bool {
42        self.check.accepted()
43    }
44
45    /// Retain the accepted candidate as a stable interpretation. No checking,
46    /// copying of packed words, or allocation is repeated.
47    ///
48    /// # Errors
49    /// Returns this entire decision for a rejected or inconclusive candidate.
50    pub fn into_stable_interpretation(self) -> Result<StableInterpretation, Self> {
51        if self.accepted() {
52            Ok(StableInterpretation {
53                interpretation: self.candidate,
54            })
55        } else {
56            Err(self)
57        }
58    }
59}

61/// An interpretation accepted by native membership checking or scalar search.
62/// The immutable interpretation retains the exact theory instance. Membership
63/// makes no assertion about enumeration coverage or an objective optimum.
64/// Cloning copies the packed words and shares the theory handle.
65///
66/// Public verdict data cannot construct this record:
67/// ```compile_fail
68/// use zetesis_ferraris::Interpretation;
69/// use zetesis_sat::StableInterpretation;
70/// fn forge(interpretation: Interpretation) -> StableInterpretation {
71///     StableInterpretation { interpretation }
72/// }
73/// ```
74#[derive(Clone, Debug)]
75pub struct StableInterpretation {
76    interpretation: Interpretation,
77}
78
79impl StableInterpretation {
80    /// Exact checked theory, borrowed in constant time.
81    #[must_use]
82    pub fn theory(&self) -> &Theory {
83        self.interpretation.theory()
84    }
85
86    /// Accepted subject, borrowed without allocation or evaluation.
87    #[must_use]
88    pub const fn interpretation(&self) -> &Interpretation {
89        &self.interpretation
90    }
91
92    /// Transfer the raw interpretation, preserving its theory identity and words.
93    #[must_use]
94    pub fn into_interpretation(self) -> Interpretation {
95        self.interpretation
96    }
97}
98
99/// Check and retain an owned candidate against its own immutable theory.
100/// Uses exactly [`crate::check`]'s original satisfaction and proper-subset
101/// search. Search can be exponential; all native bounds and stops apply. The
102/// candidate words are moved, not cloned. Caller-supplied verdicts do not enter
103/// this operation. An inconclusive result retains the candidate and stop reason.
104#[must_use]
105pub fn check_interpretation(
106    candidate: Interpretation,
107    limits: Limits,
108    cancellation: &Cancellation,
109) -> CheckedInterpretation {
110    let check = crate::check(candidate.theory(), &candidate, limits, cancellation);
111    CheckedInterpretation { candidate, check }
112}
113
114impl StableModels {
115    /// Advance scalar enumeration and retain each native accepted interpretation
116    /// with its theory identity. This runs exactly one ordinary iterator step,
117    /// including its native checking, work, and error behavior; no second check or
118    /// word copy is performed. It never converts a supplied batch verdict into
119    /// an accepted record. Pending batch proposals cause the existing scalar
120    /// [`Incomplete::PendingBatch`] refusal.
121    ///
122    /// A returned interpretation establishes membership only. Consult
123    /// [`Self::exhausted`] for completion, whose scope includes any accumulated
124    /// candidate restrictions and relies on the contract of any earlier trusted
125    /// batch calls. An error is emitted once and subsequent calls return `None`.
126    pub fn next_verified(&mut self) -> Option<Result<StableInterpretation, Incomplete>> {
127        self.next()
128            .map(|result| result.map(|interpretation| StableInterpretation { interpretation }))
129    }
130}
