6/// Why no logical decision is available. Every variant leaves coverage open.
7#[derive(Clone, Copy, Debug, PartialEq, Eq)]
8pub enum Incomplete {
9    /// Shared cancellation was observed.
10    Cancelled,
11    /// The absolute deadline expired.
12    Deadline,
13    /// Charged SAT work reached its inclusive ceiling.
14    WorkLimit,
15    /// Starting another decision would exceed the ceiling.
16    DecisionLimit,
17    /// Checking another classical candidate would exceed the ceiling.
18    CandidateLimit,
19    /// Pending candidate storage would exceed the caller's byte ceiling.
20    PendingBytes,
21    /// Concurrent completion workspaces and ordered results exceed logical scratch.
22    CompletionScratch,
23    /// Named prepared-reduct or query storage exceeds its explicit byte ceiling.
24    ReductStorage {
25        /// Required named vector/map capacity, excluding shared source payload.
26        required: u128,
27        /// Inclusive ceiling for this preparation or query owner.
28        limit: u128,
29    },
30    /// A retained batch exceeds the current call's candidate ceiling.
31    BatchCandidateLimit,
32    /// Scalar iteration cannot discard an outstanding candidate batch.
33    PendingBatch,
34    /// Fallible storage reservation failed.
35    Allocation,
36    /// A finite encoding or index exceeded admission or representation limits.
37    Admission(AdmissionError),
38    /// Exact exclusion history exceeded its own inclusive resource ceiling.
39    ProjectionLimit {
40        /// Refused history dimension.
41        resource: ProjectionResource,
42        /// Proposed count or named byte capacity, before publishing a key.
43        required: u128,
44        /// Inclusive configured ceiling.
45        limit: u128,
46    },
47    /// A candidate belongs to a different admitted theory instance.
48    WrongTheory,
49    /// A candidate-only restriction declares a different semantic atom count.
50    RestrictionUniverse {
51        /// Original semantic atom count.
52        expected: usize,
53        /// Restriction's declared atom count.
54        actual: usize,
55    },
56    /// A fused iterator cannot restart after exhaustion or terminal failure.
57    ClosedEnumerator,
58    /// A certificate cannot be configured after candidate generation has begun.
59    LateCertificate,
60    /// A caller-owned candidate-region filter retained an external failure.
61    /// Its adapter must supply the typed receipt; this never means exhaustion.
62    RegionFilter,
63    /// Original-region filtering is unavailable for the selected proposer.
64    RegionFilterUnsupported,
65    /// A region filter cannot be installed after candidate traversal began.
66    LateRegionFilter,
67    /// An installed original-region filter cannot be replaced.
68    RegionFilterAlreadySet,
69    /// Optional candidate certification stopped without establishing membership.
70    Certificate(crate::CertificateError),
71    /// Independent formula evaluation stopped without validating a witness.
72    Verification(Stop),
73    /// An internal SAT witness failed independent semantic validation.
74    InvalidWitness,
75    /// A worker unwound before it could account for its remaining search.
76    WorkerPanicked,
77    /// An accounting counter cannot be incremented without overflow.
78    CounterOverflow,
79}
80impl From<AdmissionError> for Incomplete {
81    fn from(error: AdmissionError) -> Self {
82        Self::Admission(error)
83    }
84}
85impl From<Stop> for Incomplete {
86    fn from(error: Stop) -> Self {
87        match error {
88            Stop::Cancelled => Self::Cancelled,
89            Stop::Deadline => Self::Deadline,
90            Stop::Allocation => Self::Allocation,
91            Stop::WrongProgram => Self::WrongTheory,
92            _ => Self::Verification(error),
93        }
94    }
95}
96impl fmt::Display for Incomplete {
97    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
98        match self {
99            Self::Cancelled => f.write_str("SAT operation cancelled"),
100            Self::Deadline => f.write_str("SAT operation deadline expired"),
101            Self::WorkLimit => f.write_str("SAT work limit reached"),
102            Self::DecisionLimit => f.write_str("SAT decision limit reached"),
103            Self::CandidateLimit => f.write_str("stable candidate limit reached"),
104            Self::CompletionScratch => f.write_str("completion logical scratch byte limit reached"),
105            Self::ReductStorage { required, limit } => write!(
106                f,
107                "prepared reduct storage requires {required} bytes; limit is {limit}"
108            ),
109            Self::PendingBytes => f.write_str("pending candidate byte limit reached"),
110            Self::BatchCandidateLimit => f.write_str("pending batch candidate limit reached"),
111            Self::PendingBatch => f.write_str("an unresolved candidate batch remains"),
112            Self::Allocation => f.write_str("SAT storage reservation failed"),
113            Self::Admission(error) => error.fmt(f),
114            Self::ProjectionLimit {
115                resource,
116                required,
117                limit,
118            } => write!(
119                f,
120                "projection history {resource:?} requires {required}; limit is {limit}"
121            ),
122            Self::WrongTheory => f.write_str("candidate belongs to a different theory"),
123            Self::RestrictionUniverse { expected, actual } => write!(
124                f,
125                "candidate restriction declares {actual} atoms; expected {expected}"
126            ),
127            Self::ClosedEnumerator => f.write_str("candidate enumeration is already closed"),
128            Self::LateCertificate => {
129                f.write_str("certificate configuration requires an unstarted candidate stream")
130            }
131            Self::RegionFilter => {
132                f.write_str("candidate-region filter failed; inspect its receipt")
133            }
134            Self::RegionFilterUnsupported => {
135                f.write_str("candidate-region filtering requires region search")
136            }
137            Self::LateRegionFilter => {
138                f.write_str("candidate-region filtering requires an unstarted stream")
139            }
140            Self::RegionFilterAlreadySet => {
141                f.write_str("candidate-region filter is already installed")
142            }
143            Self::Certificate(error) => write!(f, "certified membership: {error}"),
144            Self::Verification(error) => write!(f, "independent verification: {error}"),
145            Self::InvalidWitness => f.write_str("SAT witness failed independent verification"),
146            Self::WorkerPanicked => f.write_str("SAT worker panicked before completing its search"),
147            Self::CounterOverflow => f.write_str("SAT accounting counter overflow"),
148        }
149    }
150}
151impl std::error::Error for Incomplete {
152    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
153        match self {
154            Self::Admission(error) => Some(error),
155            Self::Verification(error) => Some(error),
156            Self::Certificate(error) => Some(error),
157            _ => None,
158        }
159    }
