43/// Whole-operation ceilings for a membership check or stable-model enumeration.
44#[derive(Clone, Copy, Debug)]
45pub struct Limits {
46    /// Original candidate CNF, including submitted candidate restrictions.
47    /// Exact candidate exclusions have their own `projections` population.
48    pub admission: AdmissionLimits,
49    /// Immutable prepared reduct CNF, or the explicit fresh-check reduct CNF.
50    /// These dimensions are independent of candidate/restriction admission.
51    pub reduct_admission: AdmissionLimits,
52    /// Distinct exclusion keys, logical trie nodes and named retained capacity.
53    pub projections: ProjectionLimits,
54    /// Cumulative encoding, certificate and search work/decisions across a run.
55    pub search: SearchLimits,
56    /// Maximum classical candidates checked during enumeration; not model count.
57    pub max_candidates: u64,
58    /// Per-call work ceiling for independent original/reduct formula evaluation.
59    pub max_verification_work: u64,
60    /// Named immutable reduct preparation and retained per-query capacity,
61    /// each bounded independently. Aggregate completion also admits the shared
62    /// prepared owner once alongside all worker/transient/result storage.
63    pub max_reduct_bytes: u64,
64}
65impl Default for Limits {
66    fn default() -> Self {
67        Self {
68            admission: AdmissionLimits::default(),
69            reduct_admission: AdmissionLimits::default(),
70            projections: ProjectionLimits::default(),
71            search: SearchLimits::default(),
72            max_candidates: 1_000_000,
73            max_verification_work: 100_000_000,
74            max_reduct_bytes: crate::ReductPreparationLimits::DEFAULT_BYTES,
75        }
76    }

79/// Low-level membership verdict data. Native [`check`] returns a verdict for its
80/// supplied subject; the enum itself carries no binding to that subject or its
81/// theory and can also be constructed by callers. For a record that retains the
82/// subject of an actual native check, use [`crate::check_interpretation`].
83#[derive(Clone, Debug)]
84pub enum Check {
85    /// Original model whose proper-subset frozen-reduct query was proved UNSAT.
86    Stable,
87    /// The candidate does not satisfy the original theory.
88    NotModel,
89    /// Independently validated proper-subset model of the frozen reduct.
90    NonMinimal(Interpretation),
91    /// No semantic membership decision can be made within available resources.
92    Inconclusive(Incomplete),
93}
94impl Check {
95    /// Whether this verdict is the `Stable` variant. This inspects data only;
96    /// it does not run a check or authenticate a caller-constructed verdict.
97    #[must_use]
98    pub fn accepted(&self) -> bool {
99        matches!(self, Self::Stable)
100    }
101}

154/// the default method, [`SearchMethod::default`]: a region tree narrowed by

157/// `max_candidates` applies only to [`StableModels`].

163    cancellation: &Cancellation,

166        theory,
167        candidate,
168        SearchMethod::default(),
169        limits,
170        cancellation,
171    )

174/// Check stability by the chosen method: the proper-subset query of the
175/// frozen reduct as a region tree, or as a clause query on the kernel. The
176/// verdict is the same either way; a countermodel is validated independently
177/// of the method that proposed it.
178#[must_use]
179pub fn check_with(
180    theory: &Theory,
181    candidate: &Interpretation,
182    method: SearchMethod,
183    limits: Limits,
184    cancellation: &Cancellation,
185) -> Check {
186    let mut budget = Budget {
187        quota: crate::search::LocalQuota,
188        limits: limits.search,
189        cancellation,
190        statistics: SearchStatistics::default(),
191    };
192    let mut statistics = Statistics::default();
193    match fresh_membership(
194        theory,
195        candidate,
196        method,
197        limits,
198        &mut budget,
199        &mut statistics,
200        &mut reduct_query::Workspace::default(),
201    ) {
202        Ok(result) => result,
203        Err(error) => Check::Inconclusive(error),
204    }
205}

207fn fresh_membership(
208    theory: &Theory,
209    candidate: &Interpretation,
210    method: SearchMethod,
211    limits: Limits,
212    budget: &mut Budget<'_, impl Quota>,
213    statistics: &mut Statistics,
214    workspace: &mut reduct_query::Workspace,
215) -> Result<Check, Incomplete> {
216    let original = original_model(theory, candidate, limits, budget, statistics);
217    if !original? {
218        return Ok(Check::NotModel);
219    }
220    let started = timing::start(statistics.phase_timings.as_ref());
221    let result = match method {
222        SearchMethod::Clauses => {
223            reduct_membership(theory, candidate, limits, budget, statistics, workspace)
224        }
225        SearchMethod::Regions => {
226            // The one standalone holder: its index is built here, inside the
227            // reduct phase, and charged to the reduct's region counts.
228            let mut original = OriginalIndex::new(theory);
229            original
230                .ensure(|work| {
231                    budget.charge(work)?;
232                    statistics.reduct.regions.work = statistics
233                        .reduct
234                        .regions
235                        .work
236                        .checked_add(work)
237                        .ok_or(Incomplete::CounterOverflow)?;
238                    Ok(())
239                })
240                .and_then(|index| {
241                    crate::prepared_reduct::State::new(method).check(
242                        theory,
243                        Some(index),
244                        candidate,
245                        limits,
246                        budget,
247                        statistics,
248                    )
249                })
250        }
251    };
252    timing::finish(&mut statistics.phase_timings, Phase::Reduct, started);
253    result
254}

256pub(crate) fn original_model(
257    theory: &Theory,
258    candidate: &Interpretation,
259    limits: Limits,
260    budget: &Budget<'_, impl Quota>,
261    statistics: &mut Statistics,
262) -> Result<bool, Incomplete> {
263    let started = timing::start(statistics.phase_timings.as_ref());
264    let original = (|| {
265        budget.cancellation.poll()?;
266        if !theory.same_instance(candidate.theory()) {
267            return Err(Incomplete::WrongTheory);
268        }
269        models(theory, candidate, verification(limits), budget.cancellation)
270            .map_err(Incomplete::from)
271    })();
272    timing::finish(
273        &mut statistics.phase_timings,
274        Phase::OriginalValidation,
275        started,
276    );
277    original
278}

280fn reduct_membership(
281    theory: &Theory,
282    candidate: &Interpretation,
283    limits: Limits,
284    budget: &mut Budget<'_, impl Quota>,
285    statistics: &mut Statistics,
286    workspace: &mut reduct_query::Workspace,
287) -> Result<Check, Incomplete> {
288    let (reduct, search) = workspace.encode(theory, candidate, limits.reduct_admission, budget)?;
289    increment(&mut statistics.countermodel_queries)?;
290    let result = search.query(reduct, budget);
291    checked_reduct_result(theory, candidate, limits, budget, statistics, result)
292}

294pub(crate) fn checked_reduct_result(
295    theory: &Theory,
296    candidate: &Interpretation,
297    limits: Limits,
298    budget: &mut Budget<'_, impl Quota>,
299    statistics: &mut Statistics,
300    result: Solve,
301) -> Result<Check, Incomplete> {
302    match result {
303        Solve::Unsat => Ok(Check::Stable),
304        Solve::Inconclusive(error) => Err(error),
305        Solve::Sat(assignment) => {
306            let subset = encoding::interpretation(theory, &assignment, budget)?;
307            checked_countermodel(theory, candidate, subset, limits, budget, statistics)
308        }
309    }
310}

312/// Validate a proposed countermodel independently: it must be a proper
313/// subset of the candidate and model the candidate's frozen reduct. Either
314/// proposer, the clause query or the region query, is held to this.
315pub(crate) fn checked_countermodel(
316    theory: &Theory,
317    candidate: &Interpretation,
318    subset: Interpretation,
319    limits: Limits,
320    budget: &mut Budget<'_, impl Quota>,
321    statistics: &mut Statistics,
322) -> Result<Check, Incomplete> {
323    let mut proper = false;
324    for atom in 0..theory.atom_count() {
325        budget.tick()?;
326        if subset.contains(atom) && !candidate.contains(atom) {
327            return Err(Incomplete::InvalidWitness);
328        }
329        proper |= candidate.contains(atom) && !subset.contains(atom);
330    }
331    if !proper
332        || !models_reduct(
333            theory,
334            candidate,
335            &subset,
336            verification(limits),
337            budget.cancellation,
338        )?
339    {
340        return Err(Incomplete::InvalidWitness);
341    }
342    increment(&mut statistics.countermodels)?;
343    Ok(Check::NonMinimal(subset))
344}

572    /// Append a classical candidate-only constraint. The clauses proposer adds
573    /// it to the CNF and restarts the outer cursor; original clauses and
574    /// earlier restrictions stay in the CNF, and the separate exact projection
575    /// index retains every earlier exclusion without turning it into watched
576    /// CNF storage. The regions proposer narrows the regions still to visit by
577    /// it and continues. A positive cursor tests the least interpretation
578    /// against these same restrictions. Original theory and reduct acceptance stay unchanged.
579    ///
580    /// The restriction must use the original semantic atom count and index
581    /// meanings. Its separate immutable instance is expected. Restrictions
582    /// accumulate by conjunction; this method cannot widen a candidate region.
583    /// Exhaustion afterwards proves coverage only of that constrained region.
584    /// An optimizer must separately justify that omitted candidates are dominated.
585    ///
586    /// # Errors
587    /// Refuses a different atom count, a closed iterator, pending blocking
588    /// failure, capacity, work, cancellation or allocation. Failed encoding
589    /// restores the previous CNF and cursor; charged work remains cumulative.
590    pub fn restrict_candidates(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
591        if self.terminal {
592            return Err(Incomplete::ClosedEnumerator);
593        }
594        if let Some(error) = self.pending_error {
595            return Err(error);
596        }
597        if restriction.atom_count() != self.theory.atom_count() {
598            return Err(Incomplete::RestrictionUniverse {
599                expected: self.theory.atom_count(),
600                actual: restriction.atom_count(),
601            });
602        }
603        let count = self
604            .statistics
605            .candidate_restrictions
606            .checked_add(1)
607            .ok_or(Incomplete::CounterOverflow)?;
608        let mut budget = Budget {
609            quota: crate::search::LocalQuota,
610            limits: self.limits.search,
611            cancellation: &self.cancellation,
612            statistics: self.statistics.search,
613        };
614        let result = self.proposer.restrict(restriction, &mut budget);
615        self.statistics.search = budget.statistics;
616        if result.is_ok() {
617            self.statistics.candidate_restrictions = count;
618        }
619        result
620    }

622    /// Install a successively stronger candidate-only bound.
623    ///
624    /// # Caller obligation
625    /// Each bound must use the original atom count and semantic index meanings.
626    /// Every classical interpretation satisfying both the original theory and
627    /// this bound must satisfy the previous bound, if any. This implication
628    /// relative to the unchanged original theory is **not checked**. Violating
629    /// it can omit models because regions pruned earlier are not reopened. The
630    /// first bound has no implication obligation. Bounds never support original
631    /// atoms and never change the original theory or its reduct.
632    ///
633    /// Region search retains only the latest bound separately from permanent
634    /// [`Self::restrict_candidates`] constraints. Active worker snapshots may
635    /// finish under an older bound, and already pending candidates are retained;
636    /// their original membership checks still apply. An optimizer must compare
637    /// every returned model with its current incumbent. Exhaustion covers the
638    /// remaining constrained family, not the original unrestricted world view.
639    /// The optional clauses method appends each bound instead of retiring it;
640    /// the implication obligation makes that conjunction equivalent in the
641    /// presence of the original theory.
642    ///
643    /// # Errors
644    /// Refuses a different atom count, a closed iterator, pending blocking error,
645    /// generation overflow, resource exhaustion or cancellation. Failed setup
646    /// leaves the active bound and generation unchanged; admitted work stays
647    /// charged. Region knowledge retains its existing infallible allocation
648    /// contract; this operation does not make all search allocation fallible.
649    pub fn tighten_candidate_bound(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
650        if self.terminal {
651            return Err(Incomplete::ClosedEnumerator);
652        }
653        if let Some(error) = self.pending_error {
654            return Err(error);
655        }
656        if restriction.atom_count() != self.theory.atom_count() {
657            return Err(Incomplete::RestrictionUniverse {
658                expected: self.theory.atom_count(),
659                actual: restriction.atom_count(),
660            });
661        }
662        let generation = self
663            .bound_generation
664            .checked_add(1)
665            .ok_or(Incomplete::CounterOverflow)?;
666        let count = self
667            .statistics
668            .candidate_restrictions
669            .checked_add(1)
670            .ok_or(Incomplete::CounterOverflow)?;
671        let mut budget = Budget {
672            quota: crate::search::LocalQuota,
673            limits: self.limits.search,
674            cancellation: &self.cancellation,
675            statistics: self.statistics.search,
676        };
677        let result = self.proposer.tighten(restriction, generation, &mut budget);
678        self.statistics.search = budget.statistics;
679        if result.is_ok() {
680            self.bound_generation = generation;
681            self.statistics.candidate_restrictions = count;
682        }
683        result
684    }
685
686    /// Install a caller-owned original-program condition before traversal.
687    /// Certificates may already be prepared. The filter can narrow or refute
688    /// candidate regions; it never participates in frozen-reduct queries.
689    /// Exhaustion then covers the family allowed by this filter as well as
690    /// explicit candidate restrictions. See [`crate::RegionFilter`] for the
691    /// caller's soundness, resource and failure-receipt obligations.
692    ///
693    /// # Errors
694    /// Refuses a closed or started stream, replacement, or clause search.
695    pub fn set_region_filter(
696        &mut self,
697        filter: std::sync::Arc<dyn crate::RegionFilter>,
698    ) -> Result<(), Incomplete> {
699        if self.terminal {
700            return Err(Incomplete::ClosedEnumerator);
701        }
702        let statistics = self.statistics();
703        if self.pending_error.is_some()
704            || !self.batch.pending.is_empty()
705            || statistics.candidates != 0
706            || statistics.candidate_queries != 0
707            || statistics
708                .regions
709                .is_some_and(|regions| regions.counts.regions != 0)
710        {
711            return Err(Incomplete::LateRegionFilter);
712        }
713        if self.proposer.filter().is_some() {
714            return Err(Incomplete::RegionFilterAlreadySet);
715        }
716        let filter = crate::region_filter::Filter::new(filter);
717        match &mut self.proposer {
718            Proposer::Clauses(_) => return Err(Incomplete::RegionFilterUnsupported),
719            Proposer::Regions(regions) => regions.set_filter(filter),
720            Proposer::Proposals(proposals) => proposals.set_filter(filter),
721            Proposer::Parallel(parallel) => parallel.set_filter(filter)?,
722        }
723        Ok(())
724    }
725
726    /// Fuse future pulls and join candidate workers without cancelling the
727    /// caller's shared token. Statistics retain every joined worker's prefix;
728    /// unresolved batch entries remain visible in their pending receipt.
729    /// Queued worker models are discarded, not counted as delivered answers.
730    /// An early stop cannot establish exhaustion; prior exhaustion is retained.
731    ///
732    /// # Errors
733    /// Reports the first worker stop or a failure to merge joined receipts.
734    /// All workers are joined even on failure; the stream remains fused.
735    pub fn stop(&mut self) -> Result<(), Incomplete> {
736        self.terminal = true;
737        let production = self.proposer.finish_production();
738        if let Proposer::Parallel(parallel) = &mut self.proposer {
739            let result = parallel.stop();
740            let joined = parallel.search_statistics();
741            // Before the first pull, certificate preparation can have charged
742            // the coordinator while the workers still hold construction totals.
743            self.statistics.search.work = self.statistics.search.work.max(joined.work);
744            self.statistics.search.decisions =
745                self.statistics.search.decisions.max(joined.decisions);
746            return result;
747        }
748        production
749    }
750
751    /// True only after successful coverage of every unreturned answer set
752    /// satisfying all successful candidate restrictions and the original-region
753    /// filter, if either was configured. A complete positive certificate can
754    /// establish this without refuting larger classical models individually.
755    #[must_use]
756    pub const fn exhausted(&self) -> bool {
757        self.exhausted
758    }
759    /// Cumulative work, including an incomplete terminal attempt.
760    #[must_use]
761    pub fn statistics(&self) -> Statistics {
762        let mut statistics = match &self.proposer {
763            Proposer::Clauses(clauses) => Statistics {
764                projections: clauses.cursor.projection_statistics(),
765                ..self.statistics
766            },
767            Proposer::Regions(regions) => Statistics {
768                regions: Some(regions.statistics()),
769                ..self.statistics
770            },
771            Proposer::Proposals(proposals) => Statistics {
772                regions: Some(proposals.statistics()),
773                ..self.statistics
774            },
775            Proposer::Parallel(parallel) if self.determined_candidates.is_none() => {
776                let merged = parallel.merged();
777                let merged = &merged;
778                let mut certified = self.statistics.certified;
779                if let (Some(into), Some(from)) = (certified.as_mut(), merged.certified.as_ref()) {
780                    into.checks = from.checks;
781                    into.stable = from.stable;
782                    into.refuted = from.refuted;
783                    into.failed = from.failed;
784                    into.checking_work = from.checking_work;
785                    into.tight_check_peak_bytes = from.tight_check_peak_bytes;
786                    into.positive_check_peak_bytes = from.positive_check_peak_bytes;
787                    into.stratified_check_peak_bytes = from.stratified_check_peak_bytes;
788                }
789                Statistics {
790                    regions: Some(parallel.statistics()),
791                    candidates: merged.candidates,
792                    countermodel_queries: merged.countermodel_queries,
793                    countermodels: merged.countermodels,
794                    certified,
795                    // The workers' sums, and the coordinator's own phases:
796                    // the index build before the workers launched.
797                    phase_timings: crate::timing::combined(
798                        merged.phase_timings,
799                        self.statistics.phase_timings,
800                    ),
801                    reduct: crate::ReductStatistics {
802                        original_work: merged.reduct.original_work,
803                        regions: merged.reduct.regions,
804                        ..self.statistics.reduct
805                    },
806                    ..self.statistics
807                }
808            }
809            Proposer::Parallel(parallel) => Statistics {
810                regions: Some(parallel.statistics()),
811                ..self.statistics
812            },
813        };
814        statistics.region_filter = self
815            .proposer
816            .filter()
817            .map(crate::region_filter::Filter::statistics);
818        statistics
819    }

821    fn advance(&mut self) -> Result<Option<Interpretation>, Incomplete> {
822        let mut budget = Budget {
823            quota: crate::search::LocalQuota,
824            limits: self.limits.search,
825            cancellation: &self.cancellation,
826            statistics: self.statistics.search,
827        };
828        let result = advance(
829            Membership {
830                theory: &self.theory,
831                limits: self.limits,
832                certificate: self
833                    .certificate
834                    .as_ref()
835                    .and_then(certified::Certificate::cpu),
836                certificate_workspace: &mut self.certificate_workspace,
837                reduct: &mut self.reduct,
838                index: &mut self.index,
839                determined_candidates: self.determined_candidates.as_mut(),
840            },
841            &mut self.proposer,
842            &mut budget,
843            &mut self.statistics,
844            &mut self.pending_error,
845        );
846        self.statistics.search = budget.statistics;
847        result
848    }
849}

851impl Iterator for StableModels {
852    type Item = Result<Interpretation, Incomplete>;
853    fn next(&mut self) -> Option<Self::Item> {
854        if self.terminal {
855            return None;
856        }
857        if !self.batch.pending.is_empty() {
858            self.terminal = true;
859            let _ = self.proposer.finish_production();
860            return Some(Err(Incomplete::PendingBatch));
861        }
862        if let Some(error) = self.pending_error.take() {
863            self.terminal = true;
864            let _ = self.proposer.finish_production();
865            return Some(Err(error));
866        }
867        match self.advance() {
868            Ok(Some(model)) => Some(Ok(model)),
869            Ok(None) => {
870                self.terminal = true;
871                if let Err(error) = self.proposer.finish_production() {
872                    return Some(Err(error));
873                }
874                self.exhausted = true;
875                None
876            }
877            Err(error) => {
878                // Terminal evidence includes every worker's settled prefix.
879                // Cleanup cannot replace the interruption that stopped this pull.
880                let _ = self.stop();
881                Some(Err(error))
882            }
883        }
884    }
885}

888/// What the enumeration does with a candidate's membership verdict: the
889/// candidate is an answer set; it is refuted, by a proper-subset model of
890/// its reduct or by the support law, and the search goes on; or the verdict
891/// is one no proposed candidate can have, every candidate being a classical
892/// model, and the enumeration stops on an invalid witness.
893#[derive(Clone, Copy, Debug, PartialEq, Eq)]
894pub(crate) enum Decision {
895    Stable,
896    Refuted,
897    Invalid,
898}
899impl From<Check> for Decision {
900    fn from(check: Check) -> Self {
901        match check {
902            Check::Stable => Self::Stable,
903            Check::NonMinimal(_) => Self::Refuted,
904            Check::NotModel | Check::Inconclusive(_) => Self::Invalid,
905        }
906    }
907}

909struct Membership<'a> {
910    theory: &'a Theory,
911    limits: Limits,
912    /// The certificate as the enumeration owns it, so that workers can
913    /// share it; the coordinator reads through it.
914    certificate: Option<&'a std::sync::Arc<certified::Certification>>,
915    certificate_workspace: &'a mut certified::Workspace,
916    reduct: &'a mut crate::prepared_reduct::State,
917    index: &'a mut OriginalIndex,
918    determined_candidates: Option<&'a mut certified::DeterminedCandidates>,
919}

921/// The component that proposes classical candidates: it realizes the
922/// [`SearchMethod`] the enumeration was asked for, from a clause form of the
923/// theory by the retained cursor or from regions narrowed by the theory's
924/// readings, and under several workers the regions method walked in
925/// parallel, which no method names.
926#[derive(Debug)]
927enum Proposer {
928    Clauses(Box<ClauseProposer>),
929    Regions(Box<RegionSearch>),
930    /// Several workers walk the region tree and decide the leaves themselves.
931    Parallel(Box<ParallelRegions>),
932    /// Bounded parallel production, before any membership operation.
933    Proposals(Box<RegionProposals>),
934}

936/// Where the next proposal comes from: the positive cursor, which walks no
937/// region, or the proposer's own walk with the original index it reads
938/// (`None` under the clause kernel, which reads none).
939enum Walk<'a> {
940    Determined(&'a mut certified::DeterminedCandidates),
941    Index(Option<&'a std::sync::Arc<IndexedTheory>>),
942}

944/// What a proposer hands the enumeration.
945enum Proposal {
946    /// A classical model the reduct has yet to decide.
947    Candidate(Interpretation),
948    /// A stable model a worker has already decided.
949    Stable(Interpretation),
950}

953#[derive(Debug)]

956    cursor: Cursor,

959impl Proposer {
960    fn filter(&self) -> Option<&crate::region_filter::Filter> {
961        match self {
962            Self::Clauses(_) => None,
963            Self::Regions(regions) => regions.filter(),
964            Self::Proposals(proposals) => proposals.filter(),
965            Self::Parallel(parallel) => parallel.filter(),
966        }
967    }
968
969    /// Serial and joined producers are idle between pulls. Release their
970    /// borrowed scratch when the stream closes, preserving candidate receipts.
971    fn finish_production(&mut self) -> Result<(), Incomplete> {
972        match self {
973            Self::Regions(regions) => regions.finish(),
974            Self::Proposals(proposals) => proposals.finish(),
975            Self::Clauses(_) | Self::Parallel(_) => Ok(()),
976        }
977    }
978
979    /// Record the original index's work, charged when a region walk first
980    /// needed it, once in this proposer's region counts. The clause kernel
981    /// walks no region and indexes no original theory.
982    fn record_index_work(&mut self, work: u64) -> Result<(), Incomplete> {
983        let counts = match self {
984            Self::Clauses(_) => return Err(Incomplete::InvalidWitness),
985            Self::Regions(regions) => regions.counts_mut(),
986            Self::Parallel(parallel) => parallel.counts_mut(),
987            Self::Proposals(proposals) => proposals.counts_mut(),
988        };
989        counts.work = counts
990            .work
991            .checked_add(work)
992            .ok_or(Incomplete::CounterOverflow)?;
993        Ok(())
994    }
995
996    /// The next classical candidate, from the positive cursor when one is
997    /// active and else from this proposer's own walk, or `None` when the
998    /// candidate space is covered. A proposal is refused, not returned, once
999    /// the candidate ceiling is reached; the caller admits it. A region walk
1000    /// reads the index `walk` carries, which [`walk_index`] built; its
1001    /// absence there is refused as an invalid witness.
1002    fn propose(
1003        &mut self,
1004        theory: &Theory,
1005        walk: Walk<'_>,
1006        limits: Limits,
1007        certificate: Option<&std::sync::Arc<certified::Certification>>,
1008        budget: &mut Budget<'_>,
1009        statistics: &mut Statistics,
1010    ) -> Result<Option<Proposal>, Incomplete> {
1011        let index = match walk {
1012            Walk::Determined(candidates) => {
1013                return candidates
1014                    .propose(self, theory, limits, budget, statistics)
1015                    .map(|candidate| candidate.map(Proposal::Candidate));
1016            }
1017            Walk::Index(index) => index,
1018        };
1019        let proposal = match self {
1020            Self::Clauses(clauses) => {
1021                increment(&mut statistics.candidate_queries)?;
1022                match clauses.cursor.query(&clauses.cnf, budget) {
1023                    Solve::Sat(assignment) => {
1024                        if statistics.candidates >= limits.max_candidates {
1025                            return Err(Incomplete::CandidateLimit);
1026                        }
1027                        Some(encoding::interpretation(theory, &assignment, budget)?)
1028                    }
1029                    Solve::Unsat => None,
1030                    Solve::Inconclusive(error) => return Err(error),
1031                }
1032            }
1033            Self::Regions(regions) => {
1034                let index = index.ok_or(Incomplete::InvalidWitness)?;
1035                let proposal =
1036                    regions.propose(theory, index, budget, &mut statistics.phase_timings)?;
1037                if proposal.is_some() && statistics.candidates >= limits.max_candidates {
1038                    return Err(Incomplete::CandidateLimit);
1039                }
1040                proposal
1041            }
1042            Self::Parallel(parallel) => {
1043                let index = index.ok_or(Incomplete::InvalidWitness)?;
1044                let timed = statistics.phase_timings.is_some();
1045                return Ok(parallel
1046                    .propose(index, certificate, timed, budget)?
1047                    .map(Proposal::Stable));
1048            }
1049            Self::Proposals(proposals) => {
1050                let subject = index.ok_or(Incomplete::InvalidWitness)?.subject(theory)?;
1051                let mut output = crate::search::storage(1)?;
1052                let produced = proposals.fill(
1053                    subject,
1054                    1,
1055                    limits.max_candidates.saturating_sub(statistics.candidates),
1056                    budget,
1057                    &mut output,
1058                    statistics.phase_timings.is_some(),
1059                );
1060                if let Some(timings) = statistics.phase_timings.as_mut() {
1061                    timings
1062                        .original_validation
1063                        .merge(produced.original_validation);
1064                }
1065                // With one reserved slot no second producer can stop after
1066                // the first emits. Batched calls retain that richer outcome.
1067                if let Some(error) = produced.stopped {
1068                    return Err(error);
1069                }
1070                output.pop()
1071            }
1072        };
1073        Ok(proposal.map(Proposal::Candidate))
1074    }
1075
1076    /// Exclude a proposed candidate from every later proposal. Regions need
1077    /// nothing: a leaf is never visited twice.
1078    fn exclude(
1079        &mut self,
1080        candidate: &Interpretation,
1081        budget: &mut Budget<'_>,
1082    ) -> Result<(), Incomplete> {
1083        match self {
1084            Self::Clauses(clauses) => clauses.cursor.exclude(&clauses.cnf, candidate, budget),
1085            Self::Regions(_) | Self::Parallel(_) | Self::Proposals(_) => Ok(()),
1086        }
1087    }
1088
1089    fn tighten(
1090        &mut self,
1091        restriction: &Theory,
1092        generation: u64,
1093        budget: &mut Budget<'_>,
1094    ) -> Result<(), Incomplete> {
1095        match self {
1096            Self::Clauses(_) => self.restrict(restriction, budget),
1097            Self::Regions(regions) => regions.tighten(restriction, generation, budget),
1098            Self::Parallel(parallel) => parallel.tighten(restriction, generation, budget),
1099            Self::Proposals(proposals) => proposals.tighten(restriction, generation, budget),
1100        }
1101    }
1102
1103    /// Restrict every later proposal to the classical models of `restriction`.
1104    fn restrict(
1105        &mut self,
1106        restriction: &Theory,
1107        budget: &mut Budget<'_>,
1108    ) -> Result<(), Incomplete> {
1109        match self {
1110            Self::Clauses(clauses) => {
1111                encoding::restrict(&mut clauses.cnf, restriction, budget)?;
1112                clauses.cursor.restart();
1113                Ok(())
1114            }
1115            Self::Regions(regions) => regions.restrict(restriction, budget),
1116            Self::Parallel(parallel) => parallel.restrict(restriction, budget),
1117            Self::Proposals(proposals) => proposals.restrict(restriction, budget),
1118        }
1119    }
1120}

1122fn advance(
1123    membership_input: Membership<'_>,
1124    proposer: &mut Proposer,
1125    budget: &mut Budget<'_>,
1126    statistics: &mut Statistics,
1127    pending_error: &mut Option<Incomplete>,
1128) -> Result<Option<Interpretation>, Incomplete> {
1129    let Membership {
1130        theory,
1131        limits,
1132        certificate,
1133        certificate_workspace,
1134        reduct,
1135        index,
1136        mut determined_candidates,
1137    } = membership_input;
1138    let index = if determined_candidates.is_none() {
1139        walk_index(index, proposer, budget, &mut statistics.phase_timings)?
1140    } else {
1141        None
1142    };
1143    loop {
1144        // The parallel walk's workers time their own phases; the wait for
1145        // their models is not a phase.
1146        let started = match proposer {
1147            Proposer::Parallel(_) if determined_candidates.is_none() => None,
1148            _ => timing::start(statistics.phase_timings.as_ref()),
1149        };
1150        let walk = match determined_candidates.as_deref_mut() {
1151            Some(candidates) => Walk::Determined(candidates),
1152            None => Walk::Index(index),
1153        };
1154        let proposal = proposer.propose(theory, walk, limits, certificate, budget, statistics);
1155        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
1156        let candidate = match proposal? {
1157            None => return Ok(None),
1158            Some(Proposal::Stable(model)) => {
1159                // Decided by a worker; the coordinator only counts it.
1160                increment(&mut statistics.stable_models)?;
1161                return Ok(Some(model));
1162            }
1163            Some(Proposal::Candidate(candidate)) => candidate,
1164        };
1165        increment(&mut statistics.candidates)?;
1166        let decision: Decision = if let Some(certificate) = certificate {
1167            certified::classify(
1168                certificate,
1169                certificate_workspace,
1170                &candidate,
1171                limits,
1172                budget.cancellation,
1173                statistics,
1174                &mut budget.statistics,
1175            )?
1176            .into()
1177        } else {
1178            reduct
1179                .check(
1180                    theory,
1181                    index.map(std::sync::Arc::as_ref),
1182                    &candidate,
1183                    limits,
1184                    budget,
1185                    statistics,
1186                )?
1187                .into()
1188        };
1189        if decision == Decision::Invalid {
1190            return Err(Incomplete::InvalidWitness);
1191        }
1192        let started = timing::start(statistics.phase_timings.as_ref());
1193        let blocking = proposer.exclude(&candidate, budget);
1194        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
1195        if decision == Decision::Stable {
1196            increment(&mut statistics.stable_models)?;
1197            *pending_error = blocking.err();
1198            return Ok(Some(candidate));
1199        }
1200        blocking?;
1201    }
1202}

1204/// The original index a region walk reads, built and charged before the
1205/// walk's first step: this is the one build site of every enumeration route,
1206/// scalar, parallel and batched, and it runs on the coordinator before any
1207/// worker starts. The clause kernel walks no region and gets `None`.
1208///
1209/// The build is charged to `budget` first, one unit per node and operand
1210/// occurrence, and recorded once in the proposer's region counts in the same
1211/// step, so search work and
1212/// `regions.work` agree on every exit, a build that fails after its charge
1213/// was admitted included. Its time, when timed, is one call of the candidate
1214/// phase; a built index costs neither work nor a timed call.
1215fn walk_index<'i>(
1216    original: &'i mut OriginalIndex,
1217    proposer: &mut Proposer,
1218    budget: &mut Budget<'_>,
1219    timings: &mut Option<crate::SearchPhaseTimings>,
1220) -> Result<Option<&'i std::sync::Arc<IndexedTheory>>, Incomplete> {
1221    if matches!(proposer, Proposer::Clauses(_)) {
1222        return Ok(None);
1223    }
1224    if original.get().is_some() {
1225        return Ok(original.get());
1226    }
1227    let started = timing::start(timings.as_ref());
1228    let built = original.ensure(|work| {
1229        budget.charge(work)?;
1230        proposer.record_index_work(work)
1231    });
1232    timing::finish(timings, Phase::Candidates, started);
1233    built.map(Some)
1234}
