38/// Whole-operation ceilings for a membership check or stable-model enumeration.
39#[derive(Clone, Copy, Debug)]
40pub struct Limits {
41    /// Original candidate CNF, including submitted candidate restrictions.
42    /// Exact candidate exclusions have their own `projections` population.
43    pub admission: AdmissionLimits,
44    /// Immutable prepared reduct CNF, or the explicit fresh-check reduct CNF.
45    /// These dimensions are independent of candidate/restriction admission.
46    pub reduct_admission: AdmissionLimits,
47    /// Distinct exclusion keys, logical trie nodes and named retained capacity.
48    pub projections: ProjectionLimits,
49    /// Cumulative encoding, certificate and search work/decisions across a run.
50    pub search: SearchLimits,
51    /// Maximum classical candidates checked during enumeration; not model count.
52    pub max_candidates: u64,
53    /// Per-call work ceiling for independent original/reduct formula evaluation.
54    pub max_verification_work: u64,
55    /// Named immutable reduct preparation and retained per-query capacity,
56    /// each bounded independently. Aggregate completion also admits the shared
57    /// prepared owner once alongside all worker/transient/result storage.
58    pub max_reduct_bytes: u64,
59}

67            max_candidates: 1_000_000,

70        }

74/// Low-level membership verdict data. Native [`check`] returns a verdict for its
75/// supplied subject; the enum itself carries no binding to that subject or its
76/// theory and can also be constructed by callers. For a record that retains the
77/// subject of an actual native check, use [`crate::check_interpretation`].
78#[derive(Clone, Debug)]
79pub enum Check {
80    /// Original model whose proper-subset frozen-reduct query was proved UNSAT.
81    Stable,
82    /// The candidate does not satisfy the original theory.
83    NotModel,
84    /// Independently validated proper-subset model of the frozen reduct.
85    NonMinimal(Interpretation),
86    /// No semantic membership decision can be made within available resources.
87    Inconclusive(Incomplete),
88}
89impl Check {
90    /// Whether this verdict is the `Stable` variant. This inspects data only;
91    /// it does not run a check or authenticate a caller-constructed verdict.
92    #[must_use]
93    pub fn accepted(&self) -> bool {
94        matches!(self, Self::Stable)
95    }
96}

156    cancellation: &Cancellation,

159        theory,
160        candidate,
161        SearchMethod::default(),
162        limits,
163        cancellation,
164    )

167/// Check stability by the chosen method: the proper-subset query of the
168/// frozen reduct as a region tree, or as a clause query on the kernel. The
169/// verdict is the same either way; a countermodel is validated independently
170/// of the method that proposed it.
171#[must_use]
172pub fn check_with(
173    theory: &Theory,
174    candidate: &Interpretation,
175    method: SearchMethod,
176    limits: Limits,
177    cancellation: &Cancellation,
178) -> Check {
179    let mut budget = Budget {
180        quota: crate::search::LocalQuota,
181        limits: limits.search,
182        cancellation,
183        statistics: SearchStatistics::default(),
184    };
185    let mut statistics = Statistics::default();
186    match fresh_membership(
187        theory,
188        candidate,
189        method,
190        limits,
191        &mut budget,
192        &mut statistics,
193        &mut reduct_query::Workspace::default(),
194    ) {
195        Ok(result) => result,
196        Err(error) => Check::Inconclusive(error),
197    }
198}

200fn fresh_membership(
201    theory: &Theory,
202    candidate: &Interpretation,
203    method: SearchMethod,
204    limits: Limits,
205    budget: &mut Budget<'_, impl Quota>,
206    statistics: &mut Statistics,
207    workspace: &mut reduct_query::Workspace,
208) -> Result<Check, Incomplete> {
209    let original = original_model(theory, candidate, limits, budget, statistics);
210    if !original? {
211        return Ok(Check::NotModel);
212    }
213    let started = timing::start(statistics.phase_timings.as_ref());
214    let result = match method {
215        SearchMethod::Clauses => {
216            reduct_membership(theory, candidate, limits, budget, statistics, workspace)
217        }
218        SearchMethod::Regions => {
219            let mut state = crate::prepared_reduct::State::new(method);
220            state.check(theory, candidate, limits, budget, statistics)
221        }
222    };
223    timing::finish(&mut statistics.phase_timings, Phase::Reduct, started);
224    result
225}

227pub(crate) fn original_model(
228    theory: &Theory,
229    candidate: &Interpretation,
230    limits: Limits,
231    budget: &Budget<'_, impl Quota>,
232    statistics: &mut Statistics,
233) -> Result<bool, Incomplete> {
234    let started = timing::start(statistics.phase_timings.as_ref());
235    let original = (|| {
236        budget.cancellation.poll()?;
237        if !theory.same_instance(candidate.theory()) {
238            return Err(Incomplete::WrongTheory);
239        }
240        models(theory, candidate, verification(limits), budget.cancellation)
241            .map_err(Incomplete::from)
242    })();
243    timing::finish(
244        &mut statistics.phase_timings,
245        Phase::OriginalValidation,
246        started,
247    );
248    original
249}

251fn reduct_membership(
252    theory: &Theory,
253    candidate: &Interpretation,
254    limits: Limits,
255    budget: &mut Budget<'_, impl Quota>,
256    statistics: &mut Statistics,
257    workspace: &mut reduct_query::Workspace,
258) -> Result<Check, Incomplete> {
259    let (reduct, search) = workspace.encode(theory, candidate, limits.reduct_admission, budget)?;
260    increment(&mut statistics.countermodel_queries)?;
261    let result = search.query(reduct, budget);
262    checked_reduct_result(theory, candidate, limits, budget, statistics, result)
263}

265pub(crate) fn checked_reduct_result(
266    theory: &Theory,
267    candidate: &Interpretation,
268    limits: Limits,
269    budget: &mut Budget<'_, impl Quota>,
270    statistics: &mut Statistics,
271    result: Solve,
272) -> Result<Check, Incomplete> {
273    match result {
274        Solve::Unsat => Ok(Check::Stable),
275        Solve::Inconclusive(error) => Err(error),
276        Solve::Sat(assignment) => {
277            let subset = encoding::interpretation(theory, &assignment, budget)?;
278            checked_countermodel(theory, candidate, subset, limits, budget, statistics)
279        }
280    }
281}

283/// Validate a proposed countermodel independently: it must be a proper
284/// subset of the candidate and model the candidate's frozen reduct. Either
285/// proposer, the clause query or the region query, is held to this.
286pub(crate) fn checked_countermodel(
287    theory: &Theory,
288    candidate: &Interpretation,
289    subset: Interpretation,
290    limits: Limits,
291    budget: &mut Budget<'_, impl Quota>,
292    statistics: &mut Statistics,
293) -> Result<Check, Incomplete> {
294    let mut proper = false;
295    for atom in 0..theory.atom_count() {
296        budget.tick()?;
297        if subset.contains(atom) && !candidate.contains(atom) {
298            return Err(Incomplete::InvalidWitness);
299        }
300        proper |= candidate.contains(atom) && !subset.contains(atom);
301    }
302    if !proper
303        || !models_reduct(
304            theory,
305            candidate,
306            &subset,
307            verification(limits),
308            budget.cancellation,
309        )?
310    {
311        return Err(Incomplete::InvalidWitness);
312    }
313    increment(&mut statistics.countermodels)?;
314    Ok(Check::NonMinimal(subset))
315}

317/// Native all-model search over classical candidates with exact semantic blocking.
318/// SAT assignments include Tseitin variables, but returned interpretations and
319/// exact exclusions contain only the original theory's atom universe.
320///
321/// Work and candidate limits are cumulative. An incomplete result is emitted
322/// once and terminates the iterator without marking it exhausted. If blocking
323/// storage fails after a stable model has been proved, that model is returned
324/// first and the pending failure is returned on the next call.
325#[derive(Debug)]
326pub struct StableModels {
327    theory: Theory,
328    proposer: Proposer,
329    limits: Limits,
330    cancellation: Cancellation,
331    statistics: Statistics,
332    terminal: bool,
333    exhausted: bool,
334    pending_error: Option<Incomplete>,

457    /// Enumerate by the chosen method. Under [`SearchMethod::Regions`] no
458    /// clause form of the theory is built: the theory's producers are
459    /// extracted for the support cut and the root region is opened, both
460    /// charged as search work, and the reduct's proper-subset query is a
461    /// region tree too. The verdict on every candidate is the same either
462    /// way.
463    ///
464    /// # Errors
465    /// Refuses admission, work limits, cancellation or allocation.
466    pub fn with_method(
467        theory: &Theory,
468        method: SearchMethod,
469        limits: Limits,
470        cancellation: Cancellation,
471    ) -> Result<Self, Incomplete> {
472        let mut budget = Budget {
473            quota: crate::search::LocalQuota,
474            limits: limits.search,
475            cancellation: &cancellation,
476            statistics: SearchStatistics::default(),
477        };
478        let (proposer, support, reduct) = match method {
479            SearchMethod::Clauses => {
480                let mut cnf = encoding::encode(theory, None, limits.admission, &mut budget)?;
481                let support = candidate_support::restrict(&mut cnf, theory, limits, &mut budget)?;
482                let cursor = Cursor::projected(theory.atom_count(), limits.projections)?;
483                (
484                    Proposer::Clauses(Box::new(ClauseProposer { cnf, cursor })),
485                    Some(support),
486                    crate::prepared_reduct::State::new(method),
487                )
488            }
489            SearchMethod::Regions => {
490                let regions = RegionSearch::new(theory, &mut budget)?;
491                let reduct = crate::prepared_reduct::State::with_index(std::sync::Arc::clone(
492                    regions.index(),
493                ));
494                (Proposer::Regions(Box::new(regions)), None, reduct)
495            }
496        };
497        let statistics = Statistics {
498            search: budget.statistics,
499            support,
500            ..Default::default()
501        };
502        Ok(Self {
503            theory: theory.clone(),
504            proposer,
505            limits,
506            cancellation,
507            statistics,
508            terminal: false,
509            exhausted: false,
510            pending_error: None,
511            batch: batch::State::default(),
512            certificate: None,
513            reduct,
514        })
515    }

529    /// Append a classical candidate-only constraint. The clauses proposer adds
530    /// it to the CNF and restarts the outer cursor; original clauses and
531    /// earlier restrictions stay in the CNF, and the separate exact projection
532    /// index retains every earlier exclusion without turning it into watched
533    /// CNF storage. The regions proposer narrows the regions still to visit by
534    /// it and continues. Original theory and reduct acceptance stay unchanged.
535    ///
536    /// The restriction must use the original semantic atom count and index
537    /// meanings. Its separate immutable instance is expected. Restrictions
538    /// accumulate by conjunction; this method cannot widen a candidate region.
539    /// Exhaustion afterwards proves coverage only of that constrained region.
540    /// An optimizer must separately justify that omitted candidates are dominated.
541    ///
542    /// # Errors
543    /// Refuses a different atom count, a closed iterator, pending blocking
544    /// failure, capacity, work, cancellation or allocation. Failed encoding
545    /// restores the previous CNF and cursor; charged work remains cumulative.
546    pub fn restrict_candidates(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
547        if self.terminal {
548            return Err(Incomplete::ClosedEnumerator);
549        }
550        if let Some(error) = self.pending_error {
551            return Err(error);
552        }
553        if restriction.atom_count() != self.theory.atom_count() {
554            return Err(Incomplete::RestrictionUniverse {
555                expected: self.theory.atom_count(),
556                actual: restriction.atom_count(),
557            });
558        }
559        let count = self
560            .statistics
561            .candidate_restrictions
562            .checked_add(1)
563            .ok_or(Incomplete::CounterOverflow)?;
564        let mut budget = Budget {
565            quota: crate::search::LocalQuota,
566            limits: self.limits.search,
567            cancellation: &self.cancellation,
568            statistics: self.statistics.search,
569        };
570        let result = self.proposer.restrict(restriction, &mut budget);
571        self.statistics.search = budget.statistics;
572        if result.is_ok() {
573            self.statistics.candidate_restrictions = count;
574        }
575        result
576    }

578    /// Install a caller-owned original-program condition before traversal.
579    /// Certificates may already be prepared. The filter can only refute
580    /// candidate regions; it never participates in frozen-reduct queries.
581    /// Exhaustion then covers the family allowed by this filter as well as
582    /// explicit candidate restrictions. See [`crate::RegionFilter`] for the
583    /// caller's soundness, resource and failure-receipt obligations.
584    ///
585    /// # Errors
586    /// Refuses a closed or started stream, replacement, or clause search.
587    pub fn set_region_filter(
588        &mut self,
589        filter: std::sync::Arc<dyn crate::RegionFilter>,
590    ) -> Result<(), Incomplete> {
591        if self.terminal {
592            return Err(Incomplete::ClosedEnumerator);
593        }
594        let statistics = self.statistics();
595        if self.pending_error.is_some()
596            || !self.batch.pending.is_empty()
597            || statistics.candidates != 0
598            || statistics.candidate_queries != 0
599            || statistics
600                .regions
601                .is_some_and(|regions| regions.counts.regions != 0)
602        {
603            return Err(Incomplete::LateRegionFilter);
604        }
605        if self.proposer.filter().is_some() {
606            return Err(Incomplete::RegionFilterAlreadySet);
607        }
608        let filter = crate::region_filter::Filter::new(filter);
609        match &mut self.proposer {
610            Proposer::Clauses(_) => return Err(Incomplete::RegionFilterUnsupported),
611            Proposer::Regions(regions) => regions.filter = Some(filter),
612            Proposer::Proposals(proposals) => proposals.filter = Some(filter),
613            Proposer::Parallel(parallel) => parallel.set_filter(filter)?,
614        }
615        Ok(())
616    }

618    /// Fuse future pulls and join candidate workers without cancelling the
619    /// caller's shared token. Statistics retain every joined worker's prefix;
620    /// unresolved batch entries remain visible in their pending receipt.
621    /// Queued worker models are discarded, not counted as delivered answers.
622    /// An early stop cannot establish exhaustion; prior exhaustion is retained.
623    ///
624    /// # Errors
625    /// Reports the first worker stop or a failure to merge joined receipts.
626    /// All workers are joined even on failure; the stream remains fused.
627    pub fn stop(&mut self) -> Result<(), Incomplete> {
628        self.terminal = true;
629        if let Proposer::Parallel(parallel) = &mut self.proposer {
630            let result = parallel.stop();
631            let joined = parallel.search_statistics();
632            // Before the first pull, certificate preparation can have charged
633            // the coordinator while the workers still hold construction totals.
634            self.statistics.search.work = self.statistics.search.work.max(joined.work);
635            self.statistics.search.decisions =
636                self.statistics.search.decisions.max(joined.decisions);
637            return result;
638        }
639        Ok(())
640    }

649    /// Cumulative work, including an incomplete terminal attempt.
650    #[must_use]
651    pub fn statistics(&self) -> Statistics {
652        let mut statistics = match &self.proposer {
653            Proposer::Clauses(clauses) => Statistics {
654                projections: clauses.cursor.projection_statistics(),
655                ..self.statistics
656            },
657            Proposer::Regions(regions) => Statistics {
658                regions: Some(regions.statistics()),
659                ..self.statistics
660            },
661            Proposer::Proposals(proposals) => Statistics {
662                regions: Some(proposals.statistics()),
663                ..self.statistics
664            },
665            Proposer::Parallel(parallel) => {
666                let merged = parallel.merged();
667                let merged = &merged;
668                let mut certified = self.statistics.certified;
669                if let (Some(into), Some(from)) = (certified.as_mut(), merged.certified.as_ref()) {
670                    into.checks = from.checks;
671                    into.stable = from.stable;
672                    into.refuted = from.refuted;
673                    into.failed = from.failed;
674                    into.checking_work = from.checking_work;
675                    into.positive_check_peak_bytes = from.positive_check_peak_bytes;
676                }
677                Statistics {
678                    regions: Some(parallel.statistics()),
679                    candidates: merged.candidates,
680                    countermodel_queries: merged.countermodel_queries,
681                    countermodels: merged.countermodels,
682                    certified,
683                    phase_timings: merged.phase_timings.or(self.statistics.phase_timings),
684                    reduct: crate::ReductStatistics {
685                        original_work: merged.reduct.original_work,
686                        regions: merged.reduct.regions,
687                        ..self.statistics.reduct
688                    },
689                    ..self.statistics
690                }
691            }
692        };
693        statistics.region_filter = self
694            .proposer
695            .filter()
696            .map(crate::region_filter::Filter::statistics);
697        statistics
698    }
699
700    fn advance(&mut self) -> Result<Option<Interpretation>, Incomplete> {
701        let mut budget = Budget {
702            quota: crate::search::LocalQuota,
703            limits: self.limits.search,
704            cancellation: &self.cancellation,
705            statistics: self.statistics.search,
706        };
707        let result = advance(
708            Membership {
709                theory: &self.theory,
710                limits: self.limits,
711                certificate: self
712                    .certificate
713                    .as_ref()
714                    .and_then(certified::Certificate::cpu),
715                reduct: &mut self.reduct,
716            },
717            &mut self.proposer,
718            &mut budget,
719            &mut self.statistics,
720            &mut self.pending_error,
721        );
722        self.statistics.search = budget.statistics;
723        result
724    }
725}
726
727impl Iterator for StableModels {
728    type Item = Result<Interpretation, Incomplete>;
729    fn next(&mut self) -> Option<Self::Item> {
730        if self.terminal {
731            return None;
732        }
733        if !self.batch.pending.is_empty() {
734            self.terminal = true;
735            return Some(Err(Incomplete::PendingBatch));
736        }
737        if let Some(error) = self.pending_error.take() {
738            self.terminal = true;
739            return Some(Err(error));
740        }
741        match self.advance() {
742            Ok(Some(model)) => Some(Ok(model)),
743            Ok(None) => {
744                self.terminal = true;
745                self.exhausted = true;
746                None
747            }
748            Err(error) => {
749                self.terminal = true;
750                Some(Err(error))
751            }
752        }
753    }
754}
755impl std::iter::FusedIterator for StableModels {}

757/// What the enumeration does with a candidate's membership verdict: the
758/// candidate is an answer set; it is refuted, by a proper-subset model of
759/// its reduct or by the support law, and the search goes on; or the verdict
760/// is one no proposed candidate can have, every candidate being a classical
761/// model, and the enumeration stops on an invalid witness.
762#[derive(Clone, Copy, Debug, PartialEq, Eq)]
763pub(crate) enum Decision {
764    Stable,
765    Refuted,
766    Invalid,
767}
768impl From<Check> for Decision {
769    fn from(check: Check) -> Self {
770        match check {
771            Check::Stable => Self::Stable,
772            Check::NonMinimal(_) => Self::Refuted,
773            Check::NotModel | Check::Inconclusive(_) => Self::Invalid,
774        }
775    }
776}
777
778struct Membership<'a> {
779    theory: &'a Theory,
780    limits: Limits,
781    /// The certificate as the enumeration owns it, so that workers can
782    /// share it; the coordinator reads through it.
783    certificate: Option<&'a std::sync::Arc<certified::Certification>>,
784    reduct: &'a mut crate::prepared_reduct::State,
785}
786
787/// The component that proposes classical candidates: it realizes the
788/// [`SearchMethod`] the enumeration was asked for, from a clause form of the
789/// theory by the retained cursor or from regions narrowed by the theory's
790/// readings, and under several workers the regions method walked in
791/// parallel, which no method names.
792#[derive(Debug)]
793enum Proposer {
794    Clauses(Box<ClauseProposer>),
795    Regions(Box<RegionSearch>),
796    /// Several workers walk the region tree and decide the leaves themselves.
797    Parallel(Box<ParallelRegions>),
798    /// Bounded parallel production, before any membership operation.
799    Proposals(Box<RegionProposals>),
800}
801
802/// What a proposer hands the enumeration.
803enum Proposal {
804    /// A classical model the reduct has yet to decide.
805    Candidate(Interpretation),
806    /// A stable model a worker has already decided.
807    Stable(Interpretation),
808}
809
810/// The clause form of the theory and the retained cursor over it.
811#[derive(Debug)]
812struct ClauseProposer {
813    cnf: Cnf,
814    cursor: Cursor,
815}
816
817impl Proposer {
818    fn filter(&self) -> Option<&crate::region_filter::Filter> {
819        match self {
820            Self::Clauses(_) => None,
821            Self::Regions(regions) => regions.filter.as_ref(),
822            Self::Proposals(proposals) => proposals.filter.as_ref(),
823            Self::Parallel(parallel) => parallel.filter(),
824        }
825    }
826
827    /// The next classical candidate, or `None` when the proposer has
828    /// covered the candidate space. A proposal is refused, not returned,
829    /// once the candidate ceiling is reached; the caller admits it.
830    fn propose(
831        &mut self,
832        theory: &Theory,
833        limits: Limits,
834        certificate: Option<&std::sync::Arc<certified::Certification>>,
835        budget: &mut Budget<'_>,
836        statistics: &mut Statistics,
837    ) -> Result<Option<Proposal>, Incomplete> {
838        let proposal = match self {
839            Self::Clauses(clauses) => {
840                increment(&mut statistics.candidate_queries)?;
841                match clauses.cursor.query(&clauses.cnf, budget) {
842                    Solve::Sat(assignment) => {
843                        if statistics.candidates >= limits.max_candidates {
844                            return Err(Incomplete::CandidateLimit);
845                        }
846                        Some(encoding::interpretation(theory, &assignment, budget)?)
847                    }
848                    Solve::Unsat => None,
849                    Solve::Inconclusive(error) => return Err(error),
850                }
851            }
852            Self::Regions(regions) => {
853                let proposal = regions.propose(theory, budget, &mut statistics.phase_timings)?;
854                if proposal.is_some() && statistics.candidates >= limits.max_candidates {
855                    return Err(Incomplete::CandidateLimit);
856                }
857                proposal
858            }
859            Self::Parallel(parallel) => {
860                let timed = statistics.phase_timings.is_some();
861                return Ok(parallel
862                    .propose(certificate, timed, budget)?
863                    .map(Proposal::Stable));
864            }
865            Self::Proposals(proposals) => {
866                let mut output = crate::search::storage(1)?;
867                let produced = proposals.fill(
868                    theory,
869                    1,
870                    limits.max_candidates.saturating_sub(statistics.candidates),
871                    budget,
872                    &mut output,
873                    statistics.phase_timings.is_some(),
874                );
875                if let Some(timings) = statistics.phase_timings.as_mut() {
876                    timings
877                        .original_validation
878                        .merge(produced.original_validation);
879                }
880                // With one reserved slot no second producer can stop after
881                // the first emits. Batched calls retain that richer outcome.
882                if let Some(error) = produced.stopped {
883                    return Err(error);
884                }
885                output.pop()
886            }
887        };
888        Ok(proposal.map(Proposal::Candidate))
889    }
890
891    /// Exclude a proposed candidate from every later proposal. Regions need
892    /// nothing: a leaf is never visited twice.
893    fn exclude(
894        &mut self,
895        candidate: &Interpretation,
896        budget: &mut Budget<'_>,
897    ) -> Result<(), Incomplete> {
898        match self {
899            Self::Clauses(clauses) => clauses.cursor.exclude(&clauses.cnf, candidate, budget),
900            Self::Regions(_) | Self::Parallel(_) | Self::Proposals(_) => Ok(()),
901        }
902    }
903
904    /// Restrict every later proposal to the classical models of `restriction`.
905    fn restrict(
906        &mut self,
907        restriction: &Theory,
908        budget: &mut Budget<'_>,
909    ) -> Result<(), Incomplete> {
910        match self {
911            Self::Clauses(clauses) => {
912                encoding::restrict(&mut clauses.cnf, restriction, budget)?;
913                clauses.cursor.restart();
914                Ok(())
915            }
916            Self::Regions(regions) => regions.restrict(restriction, budget),
917            Self::Parallel(parallel) => parallel.restrict(restriction, budget),
918            Self::Proposals(proposals) => proposals.restrict(restriction, budget),
919        }
920    }
921}
922
923fn advance(
924    membership_input: Membership<'_>,
925    proposer: &mut Proposer,
926    budget: &mut Budget<'_>,
927    statistics: &mut Statistics,
928    pending_error: &mut Option<Incomplete>,
929) -> Result<Option<Interpretation>, Incomplete> {
930    let Membership {
931        theory,
932        limits,
933        certificate,
934        reduct,
935    } = membership_input;
936    loop {
937        // The parallel walk's workers time their own phases; the wait for
938        // their models is not a phase.
939        let started = match proposer {
940            Proposer::Parallel(_) => None,
941            _ => timing::start(statistics.phase_timings.as_ref()),
942        };
943        let proposal = proposer.propose(theory, limits, certificate, budget, statistics);
944        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
945        let candidate = match proposal? {
946            None => return Ok(None),
947            Some(Proposal::Stable(model)) => {
948                // Decided by a worker; the coordinator only counts it.
949                increment(&mut statistics.stable_models)?;
950                return Ok(Some(model));
951            }
952            Some(Proposal::Candidate(candidate)) => candidate,
953        };
954        increment(&mut statistics.candidates)?;
955        let decision: Decision = if let Some(certificate) = certificate {
956            certified::classify(
957                certificate,
958                &candidate,
959                limits,
960                budget.cancellation,
961                statistics,
962                &mut budget.statistics,
963            )?
964            .into()
965        } else {
966            reduct
967                .check(theory, &candidate, limits, budget, statistics)?
968                .into()
969        };
970        if decision == Decision::Invalid {
971            return Err(Incomplete::InvalidWitness);
972        }
973        let started = timing::start(statistics.phase_timings.as_ref());
974        let blocking = proposer.exclude(&candidate, budget);
975        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
976        if decision == Decision::Stable {
977            increment(&mut statistics.stable_models)?;
978            *pending_error = blocking.err();
979            return Ok(Some(candidate));
980        }
981        blocking?;
982    }
983}
