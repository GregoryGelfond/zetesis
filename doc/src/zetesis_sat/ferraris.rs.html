39/// Whole-operation ceilings for a membership check or stable-model enumeration.
40#[derive(Clone, Copy, Debug)]
41pub struct Limits {
42    /// Original candidate CNF, including submitted candidate restrictions.
43    /// Exact candidate exclusions have their own `projections` population.
44    pub admission: AdmissionLimits,
45    /// Immutable prepared reduct CNF, or the explicit fresh-check reduct CNF.
46    /// These dimensions are independent of candidate/restriction admission.
47    pub reduct_admission: AdmissionLimits,
48    /// Distinct exclusion keys, logical trie nodes and named retained capacity.
49    pub projections: ProjectionLimits,
50    /// Cumulative encoding, certificate and search work/decisions across a run.
51    pub search: SearchLimits,
52    /// Maximum classical candidates checked during enumeration; not model count.
53    pub max_candidates: u64,
54    /// Per-call work ceiling for independent original/reduct formula evaluation.
55    pub max_verification_work: u64,
56    /// Named immutable reduct preparation and retained per-query capacity,
57    /// each bounded independently. Aggregate completion also admits the shared
58    /// prepared owner once alongside all worker/transient/result storage.
59    pub max_reduct_bytes: u64,
60}
61impl Default for Limits {
62    fn default() -> Self {
63        Self {
64            admission: AdmissionLimits::default(),
65            reduct_admission: AdmissionLimits::default(),
66            projections: ProjectionLimits::default(),
67            search: SearchLimits::default(),
68            max_candidates: 1_000_000,
69            max_verification_work: 100_000_000,
70            max_reduct_bytes: crate::ReductPreparationLimits::DEFAULT_BYTES,
71        }
72    }

75/// Low-level membership verdict data. Native [`check`] returns a verdict for its
76/// supplied subject; the enum itself carries no binding to that subject or its
77/// theory and can also be constructed by callers. For a record that retains the
78/// subject of an actual native check, use [`crate::check_interpretation`].
79#[derive(Clone, Debug)]
80pub enum Check {
81    /// Original model whose proper-subset frozen-reduct query was proved UNSAT.
82    Stable,
83    /// The candidate does not satisfy the original theory.
84    NotModel,
85    /// Independently validated proper-subset model of the frozen reduct.
86    NonMinimal(Interpretation),
87    /// No semantic membership decision can be made within available resources.
88    Inconclusive(Incomplete),
89}
90impl Check {
91    /// Whether this verdict is the `Stable` variant. This inspects data only;
92    /// it does not run a check or authenticate a caller-constructed verdict.
93    #[must_use]
94    pub fn accepted(&self) -> bool {
95        matches!(self, Self::Stable)
96    }
97}

164        limits,

169/// Check stability by the chosen method: the proper-subset query of the
170/// frozen reduct as a region tree, or as a clause query on the kernel. The
171/// verdict is the same either way; a countermodel is validated independently
172/// of the method that proposed it.
173#[must_use]
174pub fn check_with(
175    theory: &Theory,
176    candidate: &Interpretation,
177    method: SearchMethod,
178    limits: Limits,
179    cancellation: &Cancellation,
180) -> Check {
181    let mut budget = Budget {
182        quota: crate::search::LocalQuota,
183        limits: limits.search,
184        cancellation,
185        statistics: SearchStatistics::default(),
186    };
187    let mut statistics = Statistics::default();
188    match fresh_membership(
189        theory,
190        candidate,
191        method,
192        limits,
193        &mut budget,
194        &mut statistics,
195        &mut reduct_query::Workspace::default(),
196    ) {
197        Ok(result) => result,
198        Err(error) => Check::Inconclusive(error),
199    }
200}

202fn fresh_membership(
203    theory: &Theory,
204    candidate: &Interpretation,
205    method: SearchMethod,
206    limits: Limits,
207    budget: &mut Budget<'_, impl Quota>,
208    statistics: &mut Statistics,
209    workspace: &mut reduct_query::Workspace,
210) -> Result<Check, Incomplete> {
211    let original = original_model(theory, candidate, limits, budget, statistics);
212    if !original? {
213        return Ok(Check::NotModel);
214    }
215    let started = timing::start(statistics.phase_timings.as_ref());
216    let result = match method {
217        SearchMethod::Clauses => {
218            reduct_membership(theory, candidate, limits, budget, statistics, workspace)
219        }
220        SearchMethod::Regions => {
221            let mut state = crate::prepared_reduct::State::new(method);
222            state.check(theory, candidate, limits, budget, statistics)
223        }
224    };
225    timing::finish(&mut statistics.phase_timings, Phase::Reduct, started);
226    result
227}

229pub(crate) fn original_model(
230    theory: &Theory,
231    candidate: &Interpretation,
232    limits: Limits,
233    budget: &Budget<'_, impl Quota>,
234    statistics: &mut Statistics,
235) -> Result<bool, Incomplete> {
236    let started = timing::start(statistics.phase_timings.as_ref());
237    let original = (|| {
238        budget.cancellation.poll()?;
239        if !theory.same_instance(candidate.theory()) {
240            return Err(Incomplete::WrongTheory);
241        }
242        models(theory, candidate, verification(limits), budget.cancellation)
243            .map_err(Incomplete::from)
244    })();
245    timing::finish(
246        &mut statistics.phase_timings,
247        Phase::OriginalValidation,
248        started,
249    );
250    original
251}

253fn reduct_membership(
254    theory: &Theory,
255    candidate: &Interpretation,
256    limits: Limits,
257    budget: &mut Budget<'_, impl Quota>,
258    statistics: &mut Statistics,
259    workspace: &mut reduct_query::Workspace,
260) -> Result<Check, Incomplete> {
261    let (reduct, search) = workspace.encode(theory, candidate, limits.reduct_admission, budget)?;
262    increment(&mut statistics.countermodel_queries)?;
263    let result = search.query(reduct, budget);
264    checked_reduct_result(theory, candidate, limits, budget, statistics, result)
265}

267pub(crate) fn checked_reduct_result(
268    theory: &Theory,
269    candidate: &Interpretation,
270    limits: Limits,
271    budget: &mut Budget<'_, impl Quota>,
272    statistics: &mut Statistics,
273    result: Solve,
274) -> Result<Check, Incomplete> {
275    match result {
276        Solve::Unsat => Ok(Check::Stable),
277        Solve::Inconclusive(error) => Err(error),
278        Solve::Sat(assignment) => {
279            let subset = encoding::interpretation(theory, &assignment, budget)?;
280            checked_countermodel(theory, candidate, subset, limits, budget, statistics)
281        }
282    }
283}

285/// Validate a proposed countermodel independently: it must be a proper
286/// subset of the candidate and model the candidate's frozen reduct. Either
287/// proposer, the clause query or the region query, is held to this.
288pub(crate) fn checked_countermodel(
289    theory: &Theory,
290    candidate: &Interpretation,
291    subset: Interpretation,
292    limits: Limits,
293    budget: &mut Budget<'_, impl Quota>,
294    statistics: &mut Statistics,
295) -> Result<Check, Incomplete> {
296    let mut proper = false;
297    for atom in 0..theory.atom_count() {
298        budget.tick()?;
299        if subset.contains(atom) && !candidate.contains(atom) {
300            return Err(Incomplete::InvalidWitness);
301        }
302        proper |= candidate.contains(atom) && !subset.contains(atom);
303    }
304    if !proper
305        || !models_reduct(
306            theory,
307            candidate,
308            &subset,
309            verification(limits),
310            budget.cancellation,
311        )?
312    {
313        return Err(Incomplete::InvalidWitness);
314    }
315    increment(&mut statistics.countermodels)?;
316    Ok(Check::NonMinimal(subset))
317}

319/// Native answer-set enumeration over classical candidates with exact semantic blocking.
320/// A complete positive certificate can restrict proposals to the unique possible
321/// answer; larger classical models need not be enumerated or refuted individually.
322/// SAT assignments include Tseitin variables, but returned interpretations and
323/// exact exclusions contain only the original theory's atom universe.
324///
325/// Work and candidate limits are cumulative. An incomplete result is emitted
326/// once and terminates the iterator without marking it exhausted. If blocking
327/// storage fails after a stable model has been proved, that model is returned
328/// first and the pending failure is returned on the next call.
329#[derive(Debug)]
330pub struct StableModels {
331    theory: Theory,
332    proposer: Proposer,
333    limits: Limits,
334    cancellation: Cancellation,
335    statistics: Statistics,
336    terminal: bool,
337    exhausted: bool,

339    batch: batch::State,
340    certificate: Option<certified::Certificate>,
341    positive_candidates: Option<certified::PositiveCandidates>,
342    bound_generation: u64,
343    reduct: crate::prepared_reduct::State,
344}

467    /// Enumerate by the chosen method. Under [`SearchMethod::Regions`] no
468    /// clause form of the theory is built: the theory's producers are
469    /// extracted for the support cut and the root region is opened, both
470    /// charged as search work, and the reduct's proper-subset query is a

476    pub fn with_method(
477        theory: &Theory,
478        method: SearchMethod,
479        limits: Limits,
480        cancellation: Cancellation,
481    ) -> Result<Self, Incomplete> {
482        let mut budget = Budget {
483            quota: crate::search::LocalQuota,
484            limits: limits.search,
485            cancellation: &cancellation,
486            statistics: SearchStatistics::default(),
487        };
488        let (proposer, support, reduct) = match method {
489            SearchMethod::Clauses => {

492                let cursor = Cursor::projected(theory.atom_count(), limits.projections)?;
493                (
494                    Proposer::Clauses(Box::new(ClauseProposer { cnf, cursor })),
495                    Some(support),
496                    crate::prepared_reduct::State::new(method),
497                )
498            }
499            SearchMethod::Regions => {
500                let regions = RegionSearch::new(theory, &mut budget)?;
501                let reduct = crate::prepared_reduct::State::with_index(std::sync::Arc::clone(
502                    regions.index(),
503                ));
504                (Proposer::Regions(Box::new(regions)), None, reduct)
505            }
506        };
507        let statistics = Statistics {
508            search: budget.statistics,
509            support,
510            ..Default::default()
511        };
512        Ok(Self {
513            theory: theory.clone(),
514            proposer,
515            limits,

517            statistics,

541    /// Append a classical candidate-only constraint. The clauses proposer adds
542    /// it to the CNF and restarts the outer cursor; original clauses and
543    /// earlier restrictions stay in the CNF, and the separate exact projection
544    /// index retains every earlier exclusion without turning it into watched
545    /// CNF storage. The regions proposer narrows the regions still to visit by
546    /// it and continues. A positive cursor tests the least interpretation
547    /// against these same restrictions. Original theory and reduct acceptance stay unchanged.
548    ///
549    /// The restriction must use the original semantic atom count and index
550    /// meanings. Its separate immutable instance is expected. Restrictions
551    /// accumulate by conjunction; this method cannot widen a candidate region.
552    /// Exhaustion afterwards proves coverage only of that constrained region.
553    /// An optimizer must separately justify that omitted candidates are dominated.
554    ///
555    /// # Errors
556    /// Refuses a different atom count, a closed iterator, pending blocking
557    /// failure, capacity, work, cancellation or allocation. Failed encoding
558    /// restores the previous CNF and cursor; charged work remains cumulative.
559    pub fn restrict_candidates(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
560        if self.terminal {
561            return Err(Incomplete::ClosedEnumerator);
562        }
563        if let Some(error) = self.pending_error {
564            return Err(error);
565        }
566        if restriction.atom_count() != self.theory.atom_count() {
567            return Err(Incomplete::RestrictionUniverse {
568                expected: self.theory.atom_count(),
569                actual: restriction.atom_count(),
570            });
571        }
572        let count = self
573            .statistics
574            .candidate_restrictions
575            .checked_add(1)
576            .ok_or(Incomplete::CounterOverflow)?;
577        let mut budget = Budget {
578            quota: crate::search::LocalQuota,
579            limits: self.limits.search,
580            cancellation: &self.cancellation,
581            statistics: self.statistics.search,
582        };
583        let result = self.proposer.restrict(restriction, &mut budget);
584        self.statistics.search = budget.statistics;
585        if result.is_ok() {
586            self.statistics.candidate_restrictions = count;
587        }
588        result
589    }

591    /// Install a successively stronger candidate-only bound.
592    ///
593    /// # Caller obligation
594    /// Each bound must use the original atom count and semantic index meanings.
595    /// Every classical model of this bound must satisfy the previous bound,
596    /// if any. This logical implication is **not checked**. Violating it can
597    /// omit models because regions pruned earlier are not reopened. The first
598    /// bound has no implication obligation. Bounds never support original atoms
599    /// and never change the original theory or its reduct.
600    ///
601    /// Region search retains only the latest bound separately from permanent
602    /// [`Self::restrict_candidates`] constraints. Active worker snapshots may
603    /// finish under an older bound, and already pending candidates are retained;
604    /// their original membership checks still apply. An optimizer must compare
605    /// every returned model with its current incumbent. Exhaustion covers the
606    /// remaining constrained family, not the original unrestricted world view.
607    /// The optional clauses method appends each bound instead of retiring it;
608    /// the implication obligation makes that conjunction equivalent.
609    ///
610    /// # Errors
611    /// Refuses a different atom count, a closed iterator, pending blocking error,
612    /// generation overflow, resource exhaustion or cancellation. Failed setup
613    /// leaves the active bound and generation unchanged; admitted work stays
614    /// charged. Region knowledge retains its existing infallible allocation
615    /// contract; this operation does not make all search allocation fallible.
616    pub fn tighten_candidate_bound(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
617        if self.terminal {
618            return Err(Incomplete::ClosedEnumerator);
619        }
620        if let Some(error) = self.pending_error {
621            return Err(error);
622        }
623        if restriction.atom_count() != self.theory.atom_count() {
624            return Err(Incomplete::RestrictionUniverse {
625                expected: self.theory.atom_count(),
626                actual: restriction.atom_count(),
627            });
628        }
629        let generation = self
630            .bound_generation
631            .checked_add(1)
632            .ok_or(Incomplete::CounterOverflow)?;
633        let count = self
634            .statistics
635            .candidate_restrictions
636            .checked_add(1)
637            .ok_or(Incomplete::CounterOverflow)?;
638        let mut budget = Budget {
639            quota: crate::search::LocalQuota,
640            limits: self.limits.search,
641            cancellation: &self.cancellation,
642            statistics: self.statistics.search,
643        };
644        let result = self.proposer.tighten(restriction, generation, &mut budget);
645        self.statistics.search = budget.statistics;
646        if result.is_ok() {
647            self.bound_generation = generation;
648            self.statistics.candidate_restrictions = count;
649        }
650        result
651    }

653    /// Install a caller-owned original-program condition before traversal.
654    /// Certificates may already be prepared. The filter can only refute
655    /// candidate regions; it never participates in frozen-reduct queries.
656    /// Exhaustion then covers the family allowed by this filter as well as
657    /// explicit candidate restrictions. See [`crate::RegionFilter`] for the
658    /// caller's soundness, resource and failure-receipt obligations.
659    ///
660    /// # Errors
661    /// Refuses a closed or started stream, replacement, or clause search.
662    pub fn set_region_filter(
663        &mut self,
664        filter: std::sync::Arc<dyn crate::RegionFilter>,
665    ) -> Result<(), Incomplete> {
666        if self.terminal {
667            return Err(Incomplete::ClosedEnumerator);
668        }
669        let statistics = self.statistics();
670        if self.pending_error.is_some()
671            || !self.batch.pending.is_empty()
672            || statistics.candidates != 0
673            || statistics.candidate_queries != 0
674            || statistics
675                .regions
676                .is_some_and(|regions| regions.counts.regions != 0)
677        {
678            return Err(Incomplete::LateRegionFilter);
679        }
680        if self.proposer.filter().is_some() {
681            return Err(Incomplete::RegionFilterAlreadySet);
682        }
683        let filter = crate::region_filter::Filter::new(filter);
684        match &mut self.proposer {
685            Proposer::Clauses(_) => return Err(Incomplete::RegionFilterUnsupported),
686            Proposer::Regions(regions) => regions.filter = Some(filter),
687            Proposer::Proposals(proposals) => proposals.filter = Some(filter),
688            Proposer::Parallel(parallel) => parallel.set_filter(filter)?,
689        }
690        Ok(())
691    }
692
693    /// Fuse future pulls and join candidate workers without cancelling the
694    /// caller's shared token. Statistics retain every joined worker's prefix;
695    /// unresolved batch entries remain visible in their pending receipt.
696    /// Queued worker models are discarded, not counted as delivered answers.
697    /// An early stop cannot establish exhaustion; prior exhaustion is retained.
698    ///
699    /// # Errors
700    /// Reports the first worker stop or a failure to merge joined receipts.
701    /// All workers are joined even on failure; the stream remains fused.
702    pub fn stop(&mut self) -> Result<(), Incomplete> {
703        self.terminal = true;
704        if let Proposer::Parallel(parallel) = &mut self.proposer {
705            let result = parallel.stop();
706            let joined = parallel.search_statistics();
707            // Before the first pull, certificate preparation can have charged
708            // the coordinator while the workers still hold construction totals.
709            self.statistics.search.work = self.statistics.search.work.max(joined.work);
710            self.statistics.search.decisions =
711                self.statistics.search.decisions.max(joined.decisions);
712            return result;
713        }
714        Ok(())
715    }
716
717    /// True only after successful coverage of every unreturned answer set
718    /// satisfying all successful candidate restrictions and the original-region
719    /// filter, if either was configured. A complete positive certificate can
720    /// establish this without refuting larger classical models individually.
721    #[must_use]
722    pub const fn exhausted(&self) -> bool {
723        self.exhausted
724    }
725    /// Cumulative work, including an incomplete terminal attempt.
726    #[must_use]
727    pub fn statistics(&self) -> Statistics {
728        let mut statistics = match &self.proposer {
729            Proposer::Clauses(clauses) => Statistics {
730                projections: clauses.cursor.projection_statistics(),
731                ..self.statistics
732            },
733            Proposer::Regions(regions) => Statistics {
734                regions: Some(regions.statistics()),
735                ..self.statistics
736            },
737            Proposer::Proposals(proposals) => Statistics {
738                regions: Some(proposals.statistics()),
739                ..self.statistics
740            },
741            Proposer::Parallel(parallel) if self.positive_candidates.is_none() => {
742                let merged = parallel.merged();
743                let merged = &merged;
744                let mut certified = self.statistics.certified;
745                if let (Some(into), Some(from)) = (certified.as_mut(), merged.certified.as_ref()) {
746                    into.checks = from.checks;
747                    into.stable = from.stable;
748                    into.refuted = from.refuted;
749                    into.failed = from.failed;
750                    into.checking_work = from.checking_work;
751                    into.positive_check_peak_bytes = from.positive_check_peak_bytes;
752                }
753                Statistics {
754                    regions: Some(parallel.statistics()),
755                    candidates: merged.candidates,
756                    countermodel_queries: merged.countermodel_queries,
757                    countermodels: merged.countermodels,
758                    certified,
759                    phase_timings: merged.phase_timings.or(self.statistics.phase_timings),
760                    reduct: crate::ReductStatistics {
761                        original_work: merged.reduct.original_work,
762                        regions: merged.reduct.regions,
763                        ..self.statistics.reduct
764                    },
765                    ..self.statistics
766                }
767            }
768            Proposer::Parallel(parallel) => Statistics {
769                regions: Some(parallel.statistics()),
770                ..self.statistics
771            },
772        };
773        statistics.region_filter = self
774            .proposer
775            .filter()
776            .map(crate::region_filter::Filter::statistics);
777        statistics
778    }
779
780    fn advance(&mut self) -> Result<Option<Interpretation>, Incomplete> {
781        let mut budget = Budget {
782            quota: crate::search::LocalQuota,
783            limits: self.limits.search,
784            cancellation: &self.cancellation,
785            statistics: self.statistics.search,
786        };
787        let result = advance(
788            Membership {
789                theory: &self.theory,
790                limits: self.limits,
791                certificate: self
792                    .certificate
793                    .as_ref()
794                    .and_then(certified::Certificate::cpu),
795                reduct: &mut self.reduct,
796                positive_candidates: self.positive_candidates.as_mut(),
797            },
798            &mut self.proposer,
799            &mut budget,
800            &mut self.statistics,
801            &mut self.pending_error,
802        );
803        self.statistics.search = budget.statistics;
804        result
805    }
806}

808impl Iterator for StableModels {
809    type Item = Result<Interpretation, Incomplete>;
810    fn next(&mut self) -> Option<Self::Item> {
811        if self.terminal {
812            return None;
813        }
814        if !self.batch.pending.is_empty() {
815            self.terminal = true;
816            return Some(Err(Incomplete::PendingBatch));
817        }
818        if let Some(error) = self.pending_error.take() {
819            self.terminal = true;
820            return Some(Err(error));
821        }
822        match self.advance() {
823            Ok(Some(model)) => Some(Ok(model)),
824            Ok(None) => {
825                self.terminal = true;
826                self.exhausted = true;
827                None
828            }
829            Err(error) => {
830                self.terminal = true;
831                Some(Err(error))
832            }
833        }
834    }
835}
836impl std::iter::FusedIterator for StableModels {}

838/// What the enumeration does with a candidate's membership verdict: the
839/// candidate is an answer set; it is refuted, by a proper-subset model of
840/// its reduct or by the support law, and the search goes on; or the verdict
841/// is one no proposed candidate can have, every candidate being a classical
842/// model, and the enumeration stops on an invalid witness.
843#[derive(Clone, Copy, Debug, PartialEq, Eq)]
844pub(crate) enum Decision {
845    Stable,
846    Refuted,
847    Invalid,
848}
849impl From<Check> for Decision {
850    fn from(check: Check) -> Self {
851        match check {
852            Check::Stable => Self::Stable,
853            Check::NonMinimal(_) => Self::Refuted,
854            Check::NotModel | Check::Inconclusive(_) => Self::Invalid,
855        }
856    }
857}
858
859struct Membership<'a> {
860    theory: &'a Theory,
861    limits: Limits,
862    /// The certificate as the enumeration owns it, so that workers can
863    /// share it; the coordinator reads through it.
864    certificate: Option<&'a std::sync::Arc<certified::Certification>>,
865    reduct: &'a mut crate::prepared_reduct::State,
866    positive_candidates: Option<&'a mut certified::PositiveCandidates>,
867}
868
869/// The component that proposes classical candidates: it realizes the
870/// [`SearchMethod`] the enumeration was asked for, from a clause form of the
871/// theory by the retained cursor or from regions narrowed by the theory's
872/// readings, and under several workers the regions method walked in
873/// parallel, which no method names.
874#[derive(Debug)]
875enum Proposer {
876    Clauses(Box<ClauseProposer>),
877    Regions(Box<RegionSearch>),
878    /// Several workers walk the region tree and decide the leaves themselves.
879    Parallel(Box<ParallelRegions>),
880    /// Bounded parallel production, before any membership operation.
881    Proposals(Box<RegionProposals>),
882}
883
884/// What a proposer hands the enumeration.
885enum Proposal {
886    /// A classical model the reduct has yet to decide.
887    Candidate(Interpretation),
888    /// A stable model a worker has already decided.
889    Stable(Interpretation),
890}
891
892/// The clause form of the theory and the retained cursor over it.
893#[derive(Debug)]
894struct ClauseProposer {
895    cnf: Cnf,
896    cursor: Cursor,
897}
898
899impl Proposer {
900    fn filter(&self) -> Option<&crate::region_filter::Filter> {
901        match self {
902            Self::Clauses(_) => None,
903            Self::Regions(regions) => regions.filter.as_ref(),
904            Self::Proposals(proposals) => proposals.filter.as_ref(),
905            Self::Parallel(parallel) => parallel.filter(),
906        }
907    }
908
909    /// The next classical candidate, or `None` when the proposer has
910    /// covered the candidate space. A proposal is refused, not returned,
911    /// once the candidate ceiling is reached; the caller admits it.
912    fn propose(
913        &mut self,
914        theory: &Theory,
915        limits: Limits,
916        certificate: Option<&std::sync::Arc<certified::Certification>>,
917        positive_candidates: Option<&mut certified::PositiveCandidates>,
918        budget: &mut Budget<'_>,
919        statistics: &mut Statistics,
920    ) -> Result<Option<Proposal>, Incomplete> {
921        if let Some(candidates) = positive_candidates {
922            return candidates
923                .propose(self, theory, limits, budget, statistics)
924                .map(|candidate| candidate.map(Proposal::Candidate));
925        }
926        let proposal = match self {
927            Self::Clauses(clauses) => {
928                increment(&mut statistics.candidate_queries)?;
929                match clauses.cursor.query(&clauses.cnf, budget) {
930                    Solve::Sat(assignment) => {
931                        if statistics.candidates >= limits.max_candidates {
932                            return Err(Incomplete::CandidateLimit);
933                        }
934                        Some(encoding::interpretation(theory, &assignment, budget)?)
935                    }
936                    Solve::Unsat => None,
937                    Solve::Inconclusive(error) => return Err(error),
938                }
939            }
940            Self::Regions(regions) => {
941                let proposal = regions.propose(theory, budget, &mut statistics.phase_timings)?;
942                if proposal.is_some() && statistics.candidates >= limits.max_candidates {
943                    return Err(Incomplete::CandidateLimit);
944                }
945                proposal
946            }
947            Self::Parallel(parallel) => {
948                let timed = statistics.phase_timings.is_some();
949                return Ok(parallel
950                    .propose(certificate, timed, budget)?
951                    .map(Proposal::Stable));
952            }
953            Self::Proposals(proposals) => {
954                let mut output = crate::search::storage(1)?;
955                let produced = proposals.fill(
956                    theory,
957                    1,
958                    limits.max_candidates.saturating_sub(statistics.candidates),
959                    budget,
960                    &mut output,
961                    statistics.phase_timings.is_some(),
962                );
963                if let Some(timings) = statistics.phase_timings.as_mut() {
964                    timings
965                        .original_validation
966                        .merge(produced.original_validation);
967                }
968                // With one reserved slot no second producer can stop after
969                // the first emits. Batched calls retain that richer outcome.
970                if let Some(error) = produced.stopped {
971                    return Err(error);
972                }
973                output.pop()
974            }
975        };
976        Ok(proposal.map(Proposal::Candidate))
977    }
978
979    /// Exclude a proposed candidate from every later proposal. Regions need
980    /// nothing: a leaf is never visited twice.
981    fn exclude(
982        &mut self,
983        candidate: &Interpretation,
984        budget: &mut Budget<'_>,
985    ) -> Result<(), Incomplete> {
986        match self {
987            Self::Clauses(clauses) => clauses.cursor.exclude(&clauses.cnf, candidate, budget),
988            Self::Regions(_) | Self::Parallel(_) | Self::Proposals(_) => Ok(()),
989        }
990    }
991
992    fn tighten(
993        &mut self,
994        restriction: &Theory,
995        generation: u64,
996        budget: &mut Budget<'_>,
997    ) -> Result<(), Incomplete> {
998        match self {
999            Self::Clauses(_) => self.restrict(restriction, budget),
1000            Self::Regions(regions) => regions.tighten(restriction, generation, budget),
1001            Self::Parallel(parallel) => parallel.tighten(restriction, generation, budget),
1002            Self::Proposals(proposals) => proposals.tighten(restriction, generation, budget),
1003        }
1004    }
1005
1006    /// Restrict every later proposal to the classical models of `restriction`.
1007    fn restrict(
1008        &mut self,
1009        restriction: &Theory,
1010        budget: &mut Budget<'_>,
1011    ) -> Result<(), Incomplete> {
1012        match self {
1013            Self::Clauses(clauses) => {
1014                encoding::restrict(&mut clauses.cnf, restriction, budget)?;
1015                clauses.cursor.restart();
1016                Ok(())
1017            }
1018            Self::Regions(regions) => regions.restrict(restriction, budget),
1019            Self::Parallel(parallel) => parallel.restrict(restriction, budget),
1020            Self::Proposals(proposals) => proposals.restrict(restriction, budget),
1021        }
1022    }
1023}

1025fn advance(
1026    membership_input: Membership<'_>,
1027    proposer: &mut Proposer,
1028    budget: &mut Budget<'_>,
1029    statistics: &mut Statistics,
1030    pending_error: &mut Option<Incomplete>,
1031) -> Result<Option<Interpretation>, Incomplete> {
1032    let Membership {
1033        theory,
1034        limits,
1035        certificate,
1036        reduct,
1037        mut positive_candidates,
1038    } = membership_input;
1039    loop {
1040        // The parallel walk's workers time their own phases; the wait for
1041        // their models is not a phase.
1042        let started = match proposer {
1043            Proposer::Parallel(_) if positive_candidates.is_none() => None,
1044            _ => timing::start(statistics.phase_timings.as_ref()),
1045        };
1046        let proposal = proposer.propose(
1047            theory,
1048            limits,
1049            certificate,
1050            positive_candidates.as_deref_mut(),
1051            budget,
1052            statistics,
1053        );
1054        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
1055        let candidate = match proposal? {
1056            None => return Ok(None),
1057            Some(Proposal::Stable(model)) => {
1058                // Decided by a worker; the coordinator only counts it.
1059                increment(&mut statistics.stable_models)?;
1060                return Ok(Some(model));
1061            }
1062            Some(Proposal::Candidate(candidate)) => candidate,
1063        };
1064        increment(&mut statistics.candidates)?;
1065        let decision: Decision = if let Some(certificate) = certificate {
1066            certified::classify(
1067                certificate,
1068                &candidate,
1069                limits,
1070                budget.cancellation,
1071                statistics,
1072                &mut budget.statistics,
1073            )?
1074            .into()
1075        } else {
1076            reduct
1077                .check(theory, &candidate, limits, budget, statistics)?
1078                .into()
1079        };
1080        if decision == Decision::Invalid {
1081            return Err(Incomplete::InvalidWitness);
1082        }
1083        let started = timing::start(statistics.phase_timings.as_ref());
1084        let blocking = proposer.exclude(&candidate, budget);
1085        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
1086        if decision == Decision::Stable {
1087            increment(&mut statistics.stable_models)?;
1088            *pending_error = blocking.err();
1089            return Ok(Some(candidate));
1090        }
1091        blocking?;
1092    }
1093}
