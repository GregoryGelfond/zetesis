13/// Verdict data supplied through the trusted batch-checker protocol. The caller
14/// must supply a sound decision about the exact original candidate and its frozen
15/// reduct, as specified by [`StableModels::next_batch`]. Constructing a variant
16/// does not check that obligation or bind the data to a subject.
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub enum BatchVerdict {
19    /// The original candidate satisfies the theory and no proper subset models its reduct.
20    NoProperSubset,
21    /// Exact native membership must finish this candidate.
22    Residual,
23    /// The candidate is not an answer set and needs no query. This requires
24    /// sound nonminimality evidence: for example a checked proper-subset reduct
25    /// model, or a present atom lacking a true producer under a complete tight
26    /// plan, whose removal models the reduct by the support law.
27    Refuted,
28    /// Original satisfaction failed, contradicting this model-only producer.
29    NotModel,
30}
31
32/// Bounds on explicitly retained, not-yet-committed candidate interpretations.
33#[derive(Clone, Copy, Debug)]
34pub struct BatchLimits {
35    /// Maximum candidates proposed for one checker call.
36    pub max_candidates: NonZeroUsize,
37    /// Logical payload allowance for pending/result vectors and interpretation words.
38    /// Shared theory storage, allocator overhead and checker-private storage are excluded.
39    pub max_pending_bytes: u64,
40}
41
42/// Proposal/check/commit counters, including unresolved work after a failure.
43#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
44pub struct BatchStatistics {
45    /// Calls made to the injected checker, including failed calls.
46    pub checker_calls: u64,
47    /// Exact completion attempts entered, including scratch refusals.
48    pub completion_calls: u64,
49    /// Candidates with completed membership and committed results.
50    pub committed: u64,
51    /// Committed candidates decided by the injected no-proper-subset verdict.
52    pub propagated: u64,
53    /// Committed candidates completed by native exact residual search.
54    pub residuals: u64,
55    /// Proposed candidates whose results are still retained. A failed last exact
56    /// block is a delayed terminal error and cannot establish exhaustive coverage.
57    pub pending: usize,
58}

60/// An unresolved batch; no failure asserts exhausted candidate coverage.
61#[derive(Debug)]
62pub enum BatchError<E> {
63    /// Native proposal, validation, completion or accounting stopped.
64    Search(Incomplete),
65    /// The injected checker failed; the same pending candidates remain retryable.
66    Checker(E),
67    /// Current-call limits cannot admit the retained batch. Its proposals remain
68    /// owned and retryable under sufficient limits, without calling the checker.
69    Limits(Incomplete),
70    /// The checker did not return exactly one ordered verdict per pending candidate.
71    Shape {
72        /// Pending candidate count.
73        expected: usize,
74        /// Returned verdict count.
75        actual: usize,
76    },
77}
78impl<E: fmt::Display> fmt::Display for BatchError<E> {
79    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
80        match self {
81            Self::Search(error) | Self::Limits(error) => error.fmt(f),
82            Self::Checker(error) => error.fmt(f),
83            Self::Shape { expected, actual } => write!(
84                f,
85                "candidate checker returned {actual} results for {expected} candidates"
86            ),
87        }
88    }
89}
90impl<E: std::error::Error + 'static> std::error::Error for BatchError<E> {
91    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
92        match self {
93            Self::Search(error) | Self::Limits(error) => Some(error),
94            Self::Checker(error) => Some(error),
95            Self::Shape { .. } => None,
96        }
97    }
98}

118    /// Propose a bounded batch with the retained native cursor, check its original
119    /// candidates, then return its stable members in proposal order. An empty
120    /// result alone is not exhaustion; consult [`Self::exhausted`].
121    ///
122    /// The checker must return ordered, sound verdicts for this exact immutable
123    /// theory. In particular, `NoProperSubset` must establish original satisfaction
124    /// and refute every proper-subset reduct model. This is an explicit semantic
125    /// trust boundary, not verification of an arbitrary supplied implementation.
126    /// Every residual receives the existing exact native membership check under
127    /// cumulative search limits. Original satisfaction is independently checked
128    /// before proposal, and objective restrictions never enter this theory.
129    ///
130    /// Proposals are exactly blocked before the next is generated and remain
131    /// owned here until the entire batch is classified. A later proposal/block
132    /// failure is reported after this batch; failed checking returns no partial
133    /// batch. Returned models remain valid across subsequent restrictions, even
134    /// when they no longer meet an improved objective bound.
135    ///
136    /// # Errors
137    /// Native failures terminate without exhaustion. Checker/shape/retry-limit failures keep
138    /// pending candidates retryable. Scalar iteration with a pending batch is
139    /// refused. Allocation, work and cancellation never discard pending coverage.
140    pub fn next_batch<E>(
141        &mut self,
142        limits: BatchLimits,
143        checker: impl FnOnce(&Theory, &[Interpretation]) -> Result<Vec<BatchVerdict>, E>,
144    ) -> Result<Vec<Interpretation>, BatchError<E>> {
145        self.next_batch_with_completion(limits, &mut CompletionExecutor::default(), checker)
146    }
147
148    /// The same proposal/check/commit contract as [`Self::next_batch`], with a
149    /// reusable exact completion executor. The checker runs on the calling
150    /// thread; only residual membership jobs may run concurrently.
151    ///
152    /// All workers share this enumeration's remaining work and decision quota.
153    /// They join before any model is committed. A failed job retains the entire
154    /// pending batch and terminates search; already committed older batches
155    /// remain valid. Completed models preserve candidate order. Parallel
156    /// scheduling can change attempted work when a limit interrupts completion.
157    /// Worker timing sums are available from the executor, separately from the
158    /// scalar search phase recorder.
159    ///
160    /// # Errors
161    /// Has the same typed failures and retry rules as [`Self::next_batch`].
162    pub fn next_batch_with_completion<E>(
163        &mut self,
164        limits: BatchLimits,
165        completion: &mut CompletionExecutor,
166        checker: impl FnOnce(&Theory, &[Interpretation]) -> Result<Vec<BatchVerdict>, E>,
167    ) -> Result<Vec<Interpretation>, BatchError<E>> {
168        if self.exhausted {
169            return Ok(Vec::new());
170        }
171        if self.terminal {
172            return Err(BatchError::Search(Incomplete::ClosedEnumerator));
173        }
174        let result = self.batch_step(limits, completion, checker);
175        if matches!(result, Err(BatchError::Search(_))) {
176            self.terminal = true;
177            let _ = self.proposer.finish_production();
178        }
179        result
180    }
181
182    fn batch_step<E>(
183        &mut self,
184        limits: BatchLimits,
185        completion: &mut CompletionExecutor,
186        checker: impl FnOnce(&Theory, &[Interpretation]) -> Result<Vec<BatchVerdict>, E>,
187    ) -> Result<Vec<Interpretation>, BatchError<E>> {
188        self.cancellation
189            .poll()
190            .map_err(|e| BatchError::Search(e.into()))?;
191        if self.batch.pending.is_empty() {
192            if let Some(error) = self.pending_error.take() {
193                return Err(BatchError::Search(error));
194            }
195            self.propose(limits).map_err(BatchError::Search)?;
196        }
197        if self.batch.pending.is_empty() {
198            if let Some(error) = self.pending_error.take() {
199                return Err(BatchError::Search(error));
200            }
201            self.exhausted = self.batch.exhausted;
202            self.terminal = self.exhausted;
203            return Ok(Vec::new());
204        }
205        if self.batch.pending.len() > limits.max_candidates.get() {
206            return Err(BatchError::Limits(Incomplete::BatchCandidateLimit));
207        }
208        // Retained vector capacity can exceed the number of rows in a partial
209        // batch. Do not silently waive its payload on a smaller retry request.
210        if self.pending_row_bytes() * self.batch.pending.capacity() as u128
211            > u128::from(limits.max_pending_bytes)
212        {
213            return Err(BatchError::Limits(Incomplete::PendingBytes));
214        }
215        increment(&mut self.batch.statistics.checker_calls).map_err(BatchError::Search)?;
216        let mut verdicts =
217            checker(&self.theory, &self.batch.pending).map_err(BatchError::Checker)?;
218        if verdicts.len() != self.batch.pending.len() {
219            return Err(BatchError::Shape {
220                expected: self.batch.pending.len(),
221                actual: verdicts.len(),
222            });
223        }
224        self.certify_pending(&mut verdicts)
225            .map_err(BatchError::Search)?;
226        self.commit(&verdicts, completion).map_err(|error| {
227            if error == Incomplete::CompletionScratch {
228                BatchError::Limits(error)
229            } else {
230                BatchError::Search(error)
231            }
232        })
233    }

235    fn certify_pending(&mut self, verdicts: &mut [BatchVerdict]) -> Result<(), Incomplete> {
236        let Some(certificate) = self
237            .certificate
238            .as_ref()
239            .and_then(super::certified::Certificate::cpu)
240        else {
241            return Ok(());
242        };
243        let mut search = self.statistics.search;
244        let result = (|| {
245            for (candidate, verdict) in self.batch.pending.iter().zip(verdicts) {
246                if *verdict != BatchVerdict::Residual {
247                    continue;
248                }
249                *verdict = match super::certified::classify(
250                    certificate,
251                    &mut self.certificate_workspace,
252                    candidate,
253                    self.limits,
254                    &self.cancellation,
255                    &mut self.statistics,
256                    &mut search,
257                )? {
258                    super::certified::Verdict::Stable => BatchVerdict::NoProperSubset,
259                    super::certified::Verdict::NotModel => BatchVerdict::NotModel,
260                    super::certified::Verdict::Unsupported
261                    | super::certified::Verdict::NonMinimal => BatchVerdict::Refuted,
262                };
263            }
264            Ok(())
265        })();
266        self.statistics.search = search;
267        result
268    }

270    fn propose(&mut self, limits: BatchLimits) -> Result<(), Incomplete> {
271        if self.batch.exhausted {
272            return Ok(());
273        }
274        if self.pending_row_bytes() * limits.max_candidates.get() as u128
275            > u128::from(limits.max_pending_bytes)
276        {
277            return Err(Incomplete::PendingBytes);
278        }
279        self.batch
280            .pending
281            .try_reserve_exact(limits.max_candidates.get())
282            .map_err(|_| Incomplete::Allocation)?;
283        let index = self.walk_index()?;
284        let mut budget = Budget {
285            quota: crate::search::LocalQuota,
286            limits: self.limits.search,
287            cancellation: &self.cancellation,
288            statistics: self.statistics.search,
289        };
290        if self.determined_candidates.is_none()
291            && let Some(index) = &index
292            && let super::Proposer::Proposals(proposals) = &mut self.proposer
293        {
294            let subject = match index.subject(&self.theory) {
295                Ok(subject) => subject,
296                Err(error) => {
297                    self.statistics.search = budget.statistics;
298                    return Err(error);
299                }
300            };
301            let started = timing::start(self.statistics.phase_timings.as_ref());
302            let produced = proposals.fill(
303                subject,
304                limits.max_candidates.get(),
305                self.limits
306                    .max_candidates
307                    .saturating_sub(self.statistics.candidates),
308                &mut budget,
309                &mut self.batch.pending,
310                self.statistics.phase_timings.is_some(),
311            );
312            if let Some(timings) = self.statistics.phase_timings.as_mut() {
313                timings
314                    .original_validation
315                    .merge(produced.original_validation);
316            }
317            timing::finish(
318                &mut self.statistics.phase_timings,
319                Phase::Candidates,
320                started,
321            );
322            self.statistics.search = budget.statistics;
323            self.admit_produced(&produced);
324            return Ok(());
325        }
326        while self.batch.pending.len() < limits.max_candidates.get() {
327            let walk = match self.determined_candidates.as_mut() {
328                Some(candidates) => super::Walk::Determined(candidates),
329                None => super::Walk::Index(index.as_ref()),
330            };
331            match proposal(
332                &self.theory,
333                &mut self.proposer,
334                walk,
335                self.limits,
336                &mut budget,
337                &mut self.statistics,
338            ) {
339                Ok(Some(candidate)) => {
340                    // The interpretation is retained even if its exact block cannot fit.
341                    let started = timing::start(self.statistics.phase_timings.as_ref());
342                    let block = self.proposer.exclude(&candidate, &mut budget);
343                    timing::finish(
344                        &mut self.statistics.phase_timings,
345                        Phase::Candidates,
346                        started,
347                    );
348                    self.batch.pending.push(candidate);
349                    if let Err(error) = block {
350                        self.pending_error = Some(error);
351                        break;
352                    }
353                }
354                Ok(None) => {
355                    self.batch.exhausted = true;
356                    break;
357                }
358                Err(error) => {
359                    self.pending_error = Some(error);
360                    break;
361                }
362            }
363        }
364        self.statistics.search = budget.statistics;
365        Ok(())
366    }

368    /// The original index the batch's walk reads, built at the walk's one
369    /// build site, as in scalar iteration; the positive cursor walks no
370    /// region and builds no index. A refused build charge stops the batch.
371    fn walk_index(&mut self) -> Result<Option<Arc<IndexedTheory>>, Incomplete> {
372        if self.determined_candidates.is_some() {
373            return Ok(None);
374        }
375        let mut budget = Budget {
376            quota: crate::search::LocalQuota,
377            limits: self.limits.search,
378            cancellation: &self.cancellation,
379            statistics: self.statistics.search,
380        };
381        let index = super::walk_index(
382            &mut self.index,
383            &mut self.proposer,
384            &mut budget,
385            &mut self.statistics.phase_timings,
386        )
387        .map(|index| index.map(Arc::clone));
388        self.statistics.search = budget.statistics;
389        index
390    }

392    /// Admit a joined production round's pending candidates. No producer
393    /// decides membership. Every completed classical leaf crosses the same
394    /// independent original-satisfaction boundary as scalar proposals before
395    /// the external batch checker receives it.
396    fn admit_produced(&mut self, produced: &super::region_proposals::Produced) {
397        self.batch.exhausted = produced.exhausted;
398        self.pending_error = produced.stopped;
399        for (validated, candidate) in self.batch.pending.iter().enumerate() {
400            let result = validate_proposal(
401                &self.theory,
402                candidate,
403                self.limits,
404                &self.cancellation,
405                &mut self.statistics,
406            )
407            .and_then(|()| increment(&mut self.statistics.candidates));
408            if let Err(error) = result {
409                // Only the validated prefix crossed the proposal boundary.
410                // The stopped suffix establishes no coverage, even when
411                // production had reached the end of its region frontier.
412                self.batch.pending.truncate(validated);
413                self.batch.exhausted = false;
414                self.pending_error = Some(error);
415                break;
416            }
417        }
418    }

420    fn pending_row_bytes(&self) -> u128 {
421        (std::mem::size_of::<Interpretation>() as u128) * 2
422            + (self.theory.atom_count().div_ceil(64) as u128) * 8
423            + std::mem::size_of::<BatchVerdict>() as u128
424            + 1
425    }

427    fn commit(
428        &mut self,
429        verdicts: &[BatchVerdict],
430        completion: &mut CompletionExecutor,
431    ) -> Result<Vec<Interpretation>, Incomplete> {
432        // Completion owns every resident query workspace in this attempt.
433        // Scalar scratch carries no semantic state and must not remain as an
434        // additional unaccounted workspace when switching execution modes.
435        self.reduct.workspace = crate::ReductWorkspace::default();
436        let mut accepted = Vec::new();
437        let mut budget = Budget {
438            quota: crate::search::LocalQuota,
439            limits: self.limits.search,
440            cancellation: &self.cancellation,
441            statistics: self.statistics.search,
442        };
443        increment(&mut self.batch.statistics.completion_calls)?;
444        let result = completion.complete(
445            super::completion::Input {
446                theory: &self.theory,
447                candidates: &self.batch.pending,
448                verdicts,
449                limits: self.limits,
450                prepared: None,
451                // A regions residual exists only after a walk, which built
452                // the index; completion borrows it and never builds one.
453                query: self.index.get().map(|index| super::ReductQuery::new(index)),
454            },
455            &mut budget,
456            &mut self.statistics,
457            &mut accepted,
458            &mut self.reduct,
459        );
460        self.statistics.search = budget.statistics;
461        result?;
462        // The executor preflight included these output slots before any query.
463        let mut output = crate::search::storage(verdicts.len())?;
464        let mut stats = self.batch.statistics;
465        let mut stable = self.statistics.stable_models;
466        for (verdict, accepted) in verdicts.iter().zip(&accepted) {
467            increment(&mut stats.committed)?;
468            increment(if *verdict == BatchVerdict::Residual {
469                &mut stats.residuals
470            } else {
471                &mut stats.propagated
472            })?;
473            if *accepted {
474                increment(&mut stable)?;
475            }
476        }
477        for (candidate, accepted) in std::mem::take(&mut self.batch.pending)
478            .into_iter()
479            .zip(accepted)
480        {
481            if accepted {
482                output.push(candidate);
483            }
484        }
485        self.batch.statistics = stats;
486        self.statistics.stable_models = stable;
487        Ok(output)
488    }
489}

491fn proposal(
492    theory: &Theory,
493    proposer: &mut super::Proposer,
494    walk: super::Walk<'_>,
495    limits: super::Limits,
496    budget: &mut Budget<'_>,
497    statistics: &mut super::Statistics,
498) -> Result<Option<Interpretation>, Incomplete> {
499    let started = timing::start(statistics.phase_timings.as_ref());
500    let proposal = proposer.propose(theory, walk, limits, None, budget, statistics);
501    timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
502    // The batched protocol checks its proposals itself; a worker-decided
503    // model is refused here rather than checked twice.
504    let candidate = match proposal? {
505        None => return Ok(None),
506        Some(super::Proposal::Candidate(candidate)) => candidate,
507        Some(super::Proposal::Stable(_)) => return Err(Incomplete::InvalidWitness),
508    };
509    // Proposals cross to an external checker, so they are validated here
510    // under their own limit and phase before that boundary, whatever the
511    // proposer's construction promises.
512    validate_proposal(theory, &candidate, limits, budget.cancellation, statistics)?;
513    increment(&mut statistics.candidates)?;
514    Ok(Some(candidate))
515}
516
517fn validate_proposal(
518    theory: &Theory,
519    candidate: &Interpretation,
520    limits: super::Limits,
521    cancellation: &crate::Cancellation,
522    statistics: &mut super::Statistics,
523) -> Result<(), Incomplete> {
524    let started = timing::start(statistics.phase_timings.as_ref());
525    let original = models(theory, candidate, verification(limits), cancellation);
526    timing::finish(
527        &mut statistics.phase_timings,
528        Phase::OriginalValidation,
529        started,
530    );
531    if !original? {
532        return Err(Incomplete::InvalidWitness);
533    }
534    Ok(())
