65impl Certification {
66    fn determined(&self) -> Option<determined::Determined<'_>> {
67        match self {
68            Self::Tight { .. } => None,
69            Self::Positive { plan, .. } => Some(determined::Determined {
70                interpretation: plan.least_consequences(),
71                failed_constraint: plan.failed_constraint(),
72                retained_bytes: plan.statistics().retained_bytes,
73            }),
74            Self::Stratified { plan, .. } => Some(determined::Determined {
75                interpretation: plan.consequences(),
76                failed_constraint: plan.failed_constraint(),
77                retained_bytes: plan.statistics().retained_bytes,
78            }),
79        }
80    }

82    /// A complete check's work bound in the primitive's charged units:
83    /// tight checking visits every node, operand occurrence, root, producer
84    /// and atom once; deterministic checking compares atoms, then visits nodes,
85    /// operand occurrences and roots. Early
86    /// refusals or refutations can consume less, never more.
87    pub(super) fn checking_work_bound(&self) -> Result<u64, Incomplete> {
88        let (theory, producers) = match self {
89            Self::Tight { plan, .. } => (plan.theory(), plan.producers().len()),
90            Self::Positive { plan, .. } => (plan.theory(), 0),
91            Self::Stratified { plan, .. } => (plan.theory(), 0),
92        };
93        [
94            theory.atom_count(),
95            theory.nodes().len(),
96            theory.parts().occurrences(),
97            theory.roots().len(),
98            producers,
99        ]
100        .into_iter()
101        .try_fold(0u64, |sum, count| {
102            sum.checked_add(u64::try_from(count).map_err(|_| Incomplete::CounterOverflow)?)
103                .ok_or(Incomplete::CounterOverflow)
104        })
105    }
106}

115/// Only checked evidence from the actual selected plan produces these values.
116pub(super) enum Verdict {
117    Stable,
118    NotModel,
119    /// A present atom has no producer with a true body under the complete
120    /// tight plan. By `TightPlans.stable_supported` the candidate is not an
121    /// answer set: the candidate without that atom models its reduct.
122    Unsupported,
123    /// An original model differs from a completely determined answer.
124    /// The complete producer certificate establishes that it is not stable.
125    NonMinimal,
126}

128impl From<Verdict> for crate::ferraris::Decision {
129    /// What the enumeration does with a certificate's verdict: a refutation
130    /// by support or direct consequences is a refutation, and a candidate that is not a
131    /// model is invalid, since every proposed candidate is one.
132    fn from(verdict: Verdict) -> Self {
133        match verdict {
134            Verdict::Stable => Self::Stable,
135            Verdict::NotModel => Self::Invalid,
136            Verdict::Unsupported | Verdict::NonMinimal => Self::Refuted,
137        }
138    }

141impl StableModels {
142    /// Prepare a complete tight certificate for a membership route outside the
143    /// CPU checker, such as the GPU tight route.
144    ///
145    /// Uses the same accounted construction as CPU checking, but installs no
146    /// CPU membership policy. The immutable plan and its first preparation
147    /// attempt remain owned by this enumeration. Repeated preparation retains
148    /// that attempt, including refusal, without replenishing work. A CPU policy
149    /// selected earlier remains selected; this operation never replaces it.
150    ///
151    /// # Errors
152    /// Refuses late or closed configuration, cancellation and cumulative work
153    /// exhaustion. Optional shape and storage refusals return `None` with their
154    /// construction receipts retained in the enumeration's statistics.
155    pub fn prepare_tight_certificate(
156        &mut self,
157        limits: TightPlanLimits,
158    ) -> Result<Option<Arc<TightPlan>>, Incomplete> {
159        self.configure_certificates(
160            CertificateLimits {
161                tight: limits,
162                ..Default::default()
163            },
164            &[Kind::Tight],
165            Use::External,
166        )?;
167        Ok(self.prepared_tight_certificate())
168    }

170    /// Shared complete-original-theory tight certificate, if preparation chose
171    /// one. Retaining this handle neither checks a candidate nor selects an
172    /// execution backend. Its construction work belongs to this enumeration.
173    #[must_use]
174    pub fn prepared_tight_certificate(&self) -> Option<Arc<TightPlan>> {
175        match self.certificate.as_ref()?.plan.as_ref() {
176            Certification::Tight { plan, .. } => Some(Arc::clone(plan)),
177            Certification::Positive { .. } | Certification::Stratified { .. } => None,
178        }
179    }

181    /// Try only a tight normal/choice certificate for the original theory.
182    /// This compatibility operation never selects the positive-cycle algorithm.
183    /// Repeated configuration retains the first attempt, including refusal.
184    ///
185    /// # Errors
186    /// Refuses late/closed configuration, shared control/allocation failures or
187    /// cumulative work exhaustion. Failed setup never establishes coverage.
188    pub fn enable_certified_checking(
189        &mut self,
190        limits: TightPlanLimits,
191    ) -> Result<bool, Incomplete> {
192        self.configure_certificates(
193            CertificateLimits {
194                tight: limits,
195                ..Default::default()
196            },
197            &[Kind::Tight],
198            Use::Cpu,
199        )

202    /// Try complete-original-theory class certificates in the requested order.
203    /// The order is only a scheduling hint; each plan checks every original root.
204    /// For clause search, a deterministic plan installs exact original-atom units for
205    /// its determined interpretation, or an empty clause for a violated constraint.
206    /// Units are transactional and consume the existing candidate CNF admission;
207    /// no restriction formula DAG is copied. Region modes propose only the determined
208    /// interpretation, if it satisfies the original constraints and accumulated
209    /// candidate conditions. Independent membership checks retain the original owner.
210    /// Original theory and reduct semantics are unchanged.
211    ///
212    /// Optional shape/storage refusals leave the next plan and general reduct
213    /// checking available. All preparation, failed attempts, unit construction
214    /// and candidate checking consume the same cumulative search work. Invoke
215    /// before enumeration; repeated calls never replace the first configuration
216    /// or replenish a budget. Candidate restrictions retain the selected owner.
217    /// Injected GPU checkers should use an enumeration without this CPU policy.
218    ///
219    /// # Errors
220    /// Refuses late/closed configuration, shared control/allocation failures or
221    /// cumulative work exhaustion. Partially appended units roll back on failure.
222    pub fn enable_class_checking(
223        &mut self,
224        limits: CertificateLimits,
225        order: CertificateOrder,
226    ) -> Result<bool, Incomplete> {
227        let kinds: &[Kind] = match order {
228            CertificateOrder::TightFirst => &[Kind::Tight, Kind::Positive],
229            CertificateOrder::PositiveFirst => &[Kind::Positive, Kind::Tight],
230            CertificateOrder::StratifiedFirst => &[Kind::Stratified, Kind::Tight, Kind::Positive],
231        };
232        self.configure_certificates(limits, kinds, Use::Cpu)
233    }

235    fn configure_certificates(
236        &mut self,
237        limits: CertificateLimits,
238        kinds: &[Kind],
239        usage: Use,
240    ) -> Result<bool, Incomplete> {
241        if self.terminal {
242            return Err(Incomplete::ClosedEnumerator);
243        }
244        if self.statistics.certified.is_some() {
245            let started =
246                self.statistics.candidate_queries != 0 || self.statistics().candidates != 0;
247            if let Some(certificate) = self.certificate.as_mut()
248                && usage == Use::Cpu
249                && certificate.usage == Use::External
250            {
251                if started {
252                    return Err(Incomplete::LateCertificate);
253                }
254                certificate.usage = Use::Cpu;
255            }
256            return Ok(self.certificate.is_some());
257        }
258        if self.statistics.candidate_queries != 0 || self.statistics().candidates != 0 {
259            return Err(Incomplete::LateCertificate);
260        }
261        let mut stats = CertifiedStatistics::default();
262        let result = (|| {
263            for kind in kinds {
264                let prepared = match kind {
265                    Kind::Tight => self.prepare_tight(limits.tight, &mut stats),
266                    Kind::Positive => self.prepare_positive(limits.positive, &mut stats),
267                    Kind::Stratified => self.prepare_stratified(limits.stratified, &mut stats),
268                }?;
269                if let Some(plan) = prepared {
270                    stats.plan = Some(match &plan {
271                        Certification::Tight { plan, .. } => {
272                            CertificatePlanStatistics::Tight(plan.statistics())
273                        }
274                        Certification::Positive { plan, .. } => {
275                            CertificatePlanStatistics::Positive(plan.statistics())
276                        }
277                        Certification::Stratified { plan, .. } => {
278                            CertificatePlanStatistics::Stratified(plan.statistics())
279                        }
280                    });
281                    stats.refusal = None;
282                    let plan = Arc::new(plan);
283                    if usage == Use::Cpu
284                        && plan.determined().is_some()
285                        && !matches!(self.proposer, super::Proposer::Clauses(_))
286                    {
287                        self.determined_candidates =
288                            Some(DeterminedCandidates::new(Arc::clone(&plan)));
289                    }
290                    self.certificate = Some(Certificate { plan, usage });
291                    return Ok(true);
292                }
293            }
294            Ok(false)
295        })();
296        self.statistics.certified = Some(stats);
297        if result.is_err() {
298            self.terminal = true;
299        }
300        result
301    }

310    fn prepare_tight(
311        &mut self,
312        mut limits: TightPlanLimits,
313        stats: &mut CertifiedStatistics,
314    ) -> Result<Option<Certification>, Incomplete> {
315        limits.max_work = limits.max_work.min(self.remaining_certificate_work());
316        let attempt = TightPlan::compile_accounted(&self.theory, limits, &self.cancellation);
317        self.statistics.search.work += attempt.work;
318        stats.construction_work += attempt.work;
319        match attempt.result {
320            Ok(plan) => Ok(Some(Certification::Tight {
321                plan: Arc::new(plan),
322                max_bytes: limits.max_bytes,
323            })),
324            Err(error) => {
325                stats.tight_refusal = Some(error);
326                stats.refusal = Some(CertificateError::Tight(error));
327                match error {
328                    TightError::Stopped(stop) => Err(stop.into()),
329                    TightError::Limit(TightResource::Work)
330                        if self.remaining_certificate_work() == 0 =>
331                    {
332                        Err(Incomplete::WorkLimit)
333                    }
334                    _ => Ok(None),
335                }
336            }
337        }
338    }

340    fn prepare_positive(
341        &mut self,
342        mut limits: PositivePlanLimits,
343        stats: &mut CertifiedStatistics,
344    ) -> Result<Option<Certification>, Incomplete> {
345        limits.max_work = limits.max_work.min(self.remaining_certificate_work());
346        let attempt = PositivePlan::compile_accounted(&self.theory, limits, &self.cancellation);
347        self.statistics.search.work += attempt.statistics.work;
348        stats.construction_work += attempt.statistics.work;
349        stats.positive_attempt = Some(attempt.statistics);
350        let plan = match attempt.result {
351            Ok(plan) => plan,
352            Err(error) => {
353                stats.positive_refusal = Some(error);
354                stats.refusal = Some(CertificateError::Positive(error));
355                return match error {
356                    PositiveError::InvalidClosure { .. } => Err(Incomplete::InvalidWitness),
357                    PositiveError::Stopped(stop) => Err(stop.into()),
358                    PositiveError::Limit {
359                        resource: PositiveResource::Work,
360                        ..
361                    } if self.remaining_certificate_work() == 0 => Err(Incomplete::WorkLimit),
362                    _ => Ok(None),
363                };
364            }
365        };
366        self.prepare_determined(
367            Certification::Positive {
368                plan,
369                max_bytes: limits.max_bytes,
370            },
371            stats,
372        )
373    }

410    fn prepare_determined(
411        &mut self,
412        certificate: Certification,
413        stats: &mut CertifiedStatistics,
414    ) -> Result<Option<Certification>, Incomplete> {
415        let plan = certificate.determined().ok_or(Incomplete::InvalidWitness)?;
416        let mut budget = Budget {
417            quota: LocalQuota,
418            limits: self.limits.search,
419            cancellation: &self.cancellation,
420            statistics: self.statistics.search,
421        };
422        // Clauses retain their transactional units. Region modes activate the
423        // singleton cursor only after this complete plan has been installed.
424        let result = match &mut self.proposer {
425            super::Proposer::Clauses(clauses) => {
426                determined::restrict(plan, &mut clauses.cnf, &mut budget)
427            }
428            super::Proposer::Regions(_)
429            | super::Proposer::Parallel(_)
430            | super::Proposer::Proposals(_) => Ok(0),
431        };
432        stats.restriction_work = stats
433            .restriction_work
434            .checked_add(budget.statistics.work - self.statistics.search.work)
435            .ok_or(Incomplete::CounterOverflow)?;
436        self.statistics.search = budget.statistics;
437        match result {
438            Ok(clauses) => {
439                stats.restriction_clauses = clauses;
440                Ok(Some(certificate))
441            }
442            Err(Incomplete::Admission(
443                error @ (AdmissionError::Limit { .. } | AdmissionError::Overflow),
444            )) => {
445                stats.restriction_refusal = Some(error);
446                stats.refusal = Some(CertificateError::Restriction(error));
447                Ok(None)
448            }
449            Err(error) => Err(error),
450        }
451    }

454/// Classify a candidate and record the certificate's actual checking work.
455/// The coordinator owns its local allowance; a parallel caller reserves a
456/// bounded allowance first and settles the returned work before publication.
457pub(super) fn classify(
458    certificate: &Certification,
459    workspace: &mut Workspace,
460    candidate: &Interpretation,
461    limits: Limits,
462    cancellation: &Cancellation,
463    statistics: &mut Statistics,
464    search: &mut SearchStatistics,
465) -> Result<Verdict, Incomplete> {
466    let started = timing::start(statistics.phase_timings.as_ref());
467    let result = evaluate(
468        certificate,
469        workspace,
470        candidate,
471        limits,
472        cancellation,
473        statistics,
474        search,
475    );
476    timing::finish(&mut statistics.phase_timings, Phase::Certified, started);
477    result
478}

480fn evaluate(
481    certificate: &Certification,
482    workspace: &mut Workspace,
483    candidate: &Interpretation,
484    limits: Limits,
485    cancellation: &Cancellation,
486    statistics: &mut Statistics,
487    search: &mut SearchStatistics,
488) -> Result<Verdict, Incomplete> {
489    let stats = statistics
490        .certified
491        .as_mut()
492        .ok_or(Incomplete::InvalidWitness)?;
493    increment(&mut stats.checks)?;
494    let before = search.work;
495    let result = match certificate {
496        Certification::Tight { plan, max_bytes } => {
497            let result = check_tight(
498                plan,
499                &mut workspace.tight,
500                candidate,
501                *max_bytes,
502                limits,
503                cancellation,
504                search,
505            );
506            let retained =
507                u128::from(plan.statistics().resident_bytes) + workspace.tight.retained_bytes();
508            stats.tight_check_peak_bytes =
509                Some(stats.tight_check_peak_bytes.unwrap_or(0).max(retained));
510            result
511        }
512        Certification::Positive { plan, max_bytes } => {
513            let (result, peak) = determined::check(
514                determined::Determined {
515                    interpretation: plan.least_consequences(),
516                    failed_constraint: plan.failed_constraint(),
517                    retained_bytes: plan.statistics().retained_bytes,
518                },
519                candidate,
520                *max_bytes,
521                limits,
522                cancellation,
523                search,
524            );
525            stats.positive_check_peak_bytes =
526                Some(stats.positive_check_peak_bytes.unwrap_or(0).max(peak));
527            result
528        }
529        Certification::Stratified { plan, max_bytes } => {
530            let (result, peak) = determined::check(
531                determined::Determined {
532                    interpretation: plan.consequences(),
533                    failed_constraint: plan.failed_constraint(),
534                    retained_bytes: plan.statistics().retained_bytes,
535                },
536                candidate,
537                *max_bytes,
538                limits,
539                cancellation,
540                search,
541            );
542            stats.stratified_check_peak_bytes =
543                Some(stats.stratified_check_peak_bytes.unwrap_or(0).max(peak));
544            result
545        }
546    };
547    stats.checking_work = stats
548        .checking_work
549        .checked_add(search.work - before)
550        .ok_or(Incomplete::CounterOverflow)?;
551    match &result {
552        Ok(Verdict::Stable) => increment(&mut stats.stable)?,
553        Ok(Verdict::Unsupported | Verdict::NonMinimal) => increment(&mut stats.refuted)?,
554        Ok(Verdict::NotModel) => {}
555        Err(_) => increment(&mut stats.failed)?,
556    }
557    result
558}

560fn check_tight(
561    plan: &TightPlan,
562    workspace: &mut TightWorkspace,
563    candidate: &Interpretation,
564    max_bytes: u64,
565    limits: Limits,
566    cancellation: &Cancellation,
567    search: &mut SearchStatistics,
568) -> Result<Verdict, Incomplete> {
569    let remaining = limits.search.max_work.saturating_sub(search.work);
570    let attempt = workspace.check(
571        plan,
572        candidate,
573        TightCheckLimits {
574            max_bytes,
575            max_work: remaining.min(limits.max_verification_work),
576        },
577        cancellation,
578    );
579    search.work += attempt.work;
580    match attempt.result {
581        Ok(check) => Ok(match check.verdict {
582            TightVerdict::Stable => Verdict::Stable,
583            TightVerdict::NotModel { .. } => Verdict::NotModel,
584            TightVerdict::Residual { .. } => Verdict::Unsupported,
585        }),
586        Err(TightError::Stopped(stop)) => Err(stop.into()),
587        Err(TightError::Limit(TightResource::Work))
588            if remaining <= limits.max_verification_work =>
589        {
590            Err(Incomplete::WorkLimit)
591        }
592        Err(error) => Err(Incomplete::Certificate(CertificateError::Tight(error))),
593    }
594}
