10/// Only original semantic atom IDs enter these units. No new formula owner or
11/// auxiliary gates are constructed. The original CNF remains unchanged on error;
12/// admitted work and allocated capacity remain visible after rollback.
13pub(super) fn restrict(
14    plan: &PositivePlan,
15    cnf: &mut Cnf,
16    budget: &mut Budget<'_>,
17) -> Result<usize, Incomplete> {
18    let checkpoint = cnf.checkpoint();
19    let result = (|| {
20        if plan.failed_constraint().is_some() {
21            budget.tick()?;
22            crate::encoding::clause(cnf, [], budget)?;
23            return Ok(1);
24        }
25        let least = plan.least_consequences();
26        for atom in 0..plan.theory().atom_count() {
27            // One charged identity/membership read and one charged literal write.
28            budget.tick()?;
29            crate::encoding::clause(cnf, [Literal::new(atom, least.contains(atom))], budget)?;
30        }
31        Ok(plan.theory().atom_count())
32    })();
33    if result.is_err() {
34        cnf.rollback(checkpoint);
35    }
36    result
37}

39pub(super) fn check(
40    plan: &PositivePlan,
41    candidate: &Interpretation,
42    max_bytes: usize,
43    limits: Limits,
44    cancellation: &Cancellation,
45    search: &mut SearchStatistics,
46) -> (Result<Verdict, Incomplete>, u128) {
47    let remaining = limits.search.max_work.saturating_sub(search.work);
48    let mut budget = Budget {
49        quota: LocalQuota,
50        limits: crate::SearchLimits {
51            max_work: search
52                .work
53                .saturating_add(remaining.min(limits.max_verification_work)),
54            ..limits.search
55        },
56        cancellation,
57        statistics: *search,
58    };
59    let mut peak = plan.statistics().retained_bytes;
60    let result = validate(plan, candidate, max_bytes, &mut budget, &mut peak);
61    *search = budget.statistics;
62    let result = match result {
63        Err(Incomplete::WorkLimit) if limits.max_verification_work < remaining => {
64            Err(Incomplete::Verification(zetesis_cpu::Stop::WorkLimit))
65        }
66        other => other,
67    };
68    (result, peak)
69}

71fn validate(
72    plan: &PositivePlan,
73    candidate: &Interpretation,
74    max_bytes: usize,
75    budget: &mut Budget<'_>,
76    peak: &mut u128,
77) -> Result<Verdict, Incomplete> {
78    budget.cancellation.poll()?;
79    if !plan.theory().same_instance(candidate.theory()) {
80        return Err(Incomplete::WrongTheory);
81    }
82    // Regions can propose larger supported models of positive cycles.
83    // Compare with the least consequences, then independently establish
84    // original satisfaction before distinguishing stability from nonminimality.
85    let mut is_least = true;
86    for atom in 0..plan.theory().atom_count() {
87        budget.tick()?;
88        is_least &= candidate.contains(atom) == plan.least_consequences().contains(atom);
89    }
90    let available = (max_bytes as u128)
91        .checked_sub(plan.statistics().retained_bytes)
92        .ok_or(Incomplete::InvalidWitness)?;
93    let mut evaluation = EvaluationWorkspace::default();
94    let attempt = evaluation.evaluate(
95        candidate,
96        EvaluationLimits {
97            max_work: budget
98                .limits
99                .max_work
100                .saturating_sub(budget.statistics.work),
101            max_bytes: usize::try_from(available).map_err(|_| Incomplete::CounterOverflow)?,
102        },
103        budget.cancellation,
104    );
105    budget.statistics.work += attempt.work;
106    *peak = (*peak).max(plan.statistics().retained_bytes + attempt.retained_bytes);
107    match attempt.result {
108        Ok(truth) if !truth.is_model() => Ok(Verdict::NotModel),
109        Ok(_) if is_least => Ok(Verdict::Stable),
110        // Every original model contains the producers' least consequences.
111        // Constraints satisfied by the candidate have tautological reducts,
112        // even if those constraints fail in the least interpretation itself.
113        // Thus a different original model has this proper-subset countermodel.
114        Ok(_) => Ok(Verdict::NonMinimal),
115        Err(EvaluationError::Stopped(zetesis_cpu::Stop::WorkLimit)) => Err(Incomplete::WorkLimit),
116        Err(EvaluationError::Stopped(stop)) => Err(stop.into()),
117        Err(error @ EvaluationError::Storage { .. }) => {
118            Err(Incomplete::Certificate(CertificateError::Evaluation(error)))
119        }
120    }
121}
