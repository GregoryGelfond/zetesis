19/// Which complete-original-theory classifier to try first.
20/// A source analysis hint may choose order, but cannot authenticate a plan.
21#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
22pub enum CertificateOrder {
23    /// Try ranked normal/choice support before positive least consequences.

26    /// Try positive least consequences before ranked normal/choice support.
27    PositiveFirst,

30/// The actual selected plan; neither variant describes the other algorithm.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum CertificatePlanStatistics {
33    /// Complete ranked-support plan.
34    Tight(TightPlanStatistics),
35    /// Complete positive least-consequence plan.
36    Positive(PositivePlanStatistics),

39/// Optional construction refusal or failed certified membership operation.
40/// None of these errors establishes inconsistency or exhaustive coverage.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum CertificateError {
43    /// Ranked-support classification, construction or checking refusal.
44    Tight(TightError),
45    /// Positive classification or least-consequence construction refusal.
46    Positive(PositiveError),
47    /// The exact least-model units could not fit candidate CNF admission.
48    /// No part of the attempted restriction remains in the CNF.
49    Restriction(AdmissionError),
50    /// Independent original-theory evaluation failed. A storage limit here is
51    /// the remaining workspace allowance after the retained positive plan.
52    Evaluation(EvaluationError),
53}
54
55impl fmt::Display for CertificateError {
56    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
57        match self {
58            Self::Tight(error) => error.fmt(formatter),
59            Self::Positive(error) => error.fmt(formatter),
60            Self::Restriction(error) => write!(formatter, "least-model restriction: {error}"),
61            Self::Evaluation(error) => error.fmt(formatter),
62        }
63    }
64}
65
66impl std::error::Error for CertificateError {
67    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
68        Some(match self {
69            Self::Tight(error) => error,
70            Self::Positive(error) => error,
71            Self::Restriction(error) => error,
72            Self::Evaluation(error) => error,
73        })
74    }
75}

77/// Attempted certificate construction and candidate checking, before batch commit.
78/// Only the enumeration's `stable_models` counter records publication/commit.
79#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
80pub struct CertifiedStatistics {
81    /// Successfully selected complete-theory certificate, if eligible.
82    pub plan: Option<CertificatePlanStatistics>,
83    /// Last optional refusal when no plan was selected; absent on success.
84    pub refusal: Option<CertificateError>,
85    /// Actual ranked-plan refusal, retained even if the positive plan succeeds.
86    pub tight_refusal: Option<TightError>,
87    /// Actual positive-plan refusal, retained even if the ranked plan succeeds.
88    pub positive_refusal: Option<PositiveError>,
89    /// Actual positive construction receipt, including a refused work/storage prefix.
90    /// A successful primitive may still be followed by a restriction refusal.
91    pub positive_attempt: Option<PositivePlanStatistics>,
92    /// Exact unit-restriction admission refusal, even if another plan succeeds.
93    pub restriction_refusal: Option<AdmissionError>,
94    /// Sum of actual plan-construction work, including every refused attempt.
95    pub construction_work: u64,
96    /// Exact least-model restriction work, including rolled-back attempts.
97    /// Included in cumulative search work, separate from plan construction.
98    pub restriction_work: u64,
99    /// Successfully committed original-atom units, or one empty candidate contradiction.
100    /// Zero after a failed transactional restriction attempt.
101    pub restriction_clauses: usize,
102    /// Candidate checks entered, including interrupted attempts.
103    pub checks: u64,
104    /// Completed certificate proofs of stability, before publication or commit.
105    pub stable: u64,
106    /// Candidates a complete certificate refuted: an unsupported present atom
107    /// under the tight plan, or an original model different from the positive
108    /// producers' least consequences, without a reduct query. Every candidate a
109    /// selected certificate checks is decided: stable, refuted or not a model.
110    pub refuted: u64,
111    /// Interrupted checks; no membership verdict was produced.
112    pub failed: u64,
113    /// Candidate-check work, including interrupted attempts.
114    pub checking_work: u64,
115    /// Maximum actual retained positive plan plus local evaluation workspace
116    /// capacity during an entered positive check, including failures. Absent
117    /// when that route was never entered; excludes proposed/refused growth,
118    /// shared theory, allocator overhead and other stack state.
119    pub positive_check_peak_bytes: Option<u128>,
120}
