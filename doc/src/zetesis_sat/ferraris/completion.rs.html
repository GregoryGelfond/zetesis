18/// Accounting for the last entered completion attempt, including failed jobs.
19/// Worker intervals may overlap; their sums are not coordinator wall time.
20#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
21pub struct CompletionStatistics {
22    /// Configured maximum simultaneous exact queries (one for scalar execution).
23    pub workers: usize,
24    /// Maximum query concurrency selected at preflight (zero without residuals).
25    /// Candidate work starts only after the retained storage is also admitted.
26    pub effective_workers: usize,
27    /// Minimum requested conservative envelope selected before allocation.
28    pub requested_scratch_bytes: u64,
29    /// Shared prepared owner plus actual worker-vector capacities and
30    /// conservative transient/result storage. May exceed the ceiling on a post-
31    /// reservation scratch refusal; no candidate work has then started.
32    /// Allocator/table control overhead, stacks and device memory are excluded.
33    /// Saturates at `u64::MAX` if the envelope is unrepresentable. Such an
34    /// attempt fails; an earlier reservation error retains precedence.
35    pub peak_scratch_bytes: u64,
36    /// Candidate slots actually entered, including supplied certificates.
37    pub candidates: usize,
38    /// Entered slots that requested exact native membership.
39    pub residuals: usize,
40    /// Slots whose completed result was stable or a validated countermodel.
41    /// These results are not published unless the entire batch commits.
42    pub completed: usize,
43    /// Slots that returned a typed incomplete result or an invalid certificate.
44    pub failed: usize,
45    /// Residual slots completed locally, whether or not the batch committed.
46    pub residual_completed: usize,
47    /// Residual slots returning incomplete or invalid-witness results.
48    pub residual_failed: usize,
49    /// Coordinator elapsed time, present only when search timing was enabled.
50    pub elapsed: Option<Duration>,
51    /// Sum of original-validation worker intervals, when timing was enabled.
52    pub worker_original_validation: Option<PhaseMeasurement>,
53    /// Sum of frozen-reduct worker intervals, when timing was enabled.
54    pub worker_reduct: Option<PhaseMeasurement>,
55}

57/// A reusable exact membership executor for original candidate batches.
58///
59/// One worker uses the existing scalar path without creating a thread pool.
60/// Multiple workers own independent query state and share the solve's cumulative
61/// work and decision ceilings. All jobs join before accounting or publication.
62/// Completed output order is independent of scheduling; a resource-limited run
63/// need not perform the same subset of work under different schedules.
64/// Shared work permits are granted to each query in bounded groups; unused
65/// permits return on every query exit and joined accounting records only work
66/// actually charged. A temporarily empty pool waits for outstanding grants,
67/// polling cooperative control between bounded timed waits. Decision permits
68/// remain individually shared. Neither policy gives a hard response deadline.
69///
70/// Completion admits all ordered result slots and a conservative query envelope
71/// before allocating scratch. Concurrency shrinks to fit that requested
72/// envelope; actual retained capacities are checked before any candidate work.
73/// Allocation slack can still refuse a batch, which remains pending.
74/// Slots reserved but unused remain charged until the joined attempt returns.
75/// Worker/result scratch is released between calls; the enumeration retains
76/// its shared prepared owner. This is not an RSS, allocator-overhead,
77/// thread-stack, original-theory, candidate-cursor or GPU-memory limit.
78#[derive(Debug)]
79pub struct CompletionExecutor {
80    pool: Option<rayon::ThreadPool>,
81    last: Option<CompletionStatistics>,
82    max_scratch_bytes: u64,
83}
84
85impl Default for CompletionExecutor {
86    fn default() -> Self {
87        Self {
88            pool: None,
89            last: None,
90            max_scratch_bytes: Self::DEFAULT_SCRATCH_BYTES,
91        }
92    }
93}

95impl CompletionExecutor {
96    /// Default aggregate logical scratch ceiling for bounded completion batches.
97    pub const DEFAULT_SCRATCH_BYTES: u64 = 256 * 1024 * 1024;
98
99    /// Requested parametric storage estimated from the original theory.
100    ///
101    /// Treats every node as an implication, bounded by `reduct_admission`.
102    /// Includes one shared owner, one query and the ordered batch results;
103    /// allocator slack is excluded. [`PreparedReduct::scratch_requirements`]
104    /// instead uses the actual compiled shape and shared owner capacity.
105    ///
106    /// # Errors
107    /// Refuses unrepresentable shape or byte arithmetic before allocation.
108    pub fn scratch_requirements(
109        theory: &Theory,
110        limits: Limits,
111        candidates: usize,
112    ) -> Result<CompletionScratch, Incomplete> {
113        scratch::requirements(theory, limits, candidates)
114    }
115
116    /// The separately configured logical completion-scratch ceiling.
117    #[must_use]
118    pub const fn scratch_limit(&self) -> u64 {
119        self.max_scratch_bytes
120    }
121
122    /// Build a dedicated pool once, or select the scalar path for one worker.
123    ///
124    /// # Errors
125    /// Returns the pool builder's error if worker threads cannot be created.
126    pub fn new(workers: NonZeroUsize) -> Result<Self, rayon::ThreadPoolBuildError> {
127        Self::with_scratch_limit(workers, Self::DEFAULT_SCRATCH_BYTES)
128    }
129
130    /// Build once with a separately bounded aggregate logical scratch allowance.
131    /// A zero allowance admits no nonempty completion batch.
132    ///
133    /// # Errors
134    /// Returns the pool builder's error if worker threads cannot be created.
135    pub fn with_scratch_limit(
136        workers: NonZeroUsize,
137        max_scratch_bytes: u64,
138    ) -> Result<Self, rayon::ThreadPoolBuildError> {
139        Ok(Self {
140            pool: if workers.get() == 1 {
141                None
142            } else {
143                Some(
144                    rayon::ThreadPoolBuilder::new()
145                        .num_threads(workers.get())
146                        .thread_name(|index| format!("zetesis-reduct-{index}"))
147                        .build()?,
148                )
149            },
150            last: None,
151            max_scratch_bytes,
152        })
153    }
154
155    /// Maximum simultaneous exact queries; the outer candidate cursor is serial.
156    #[must_use]
157    pub fn workers(&self) -> usize {
158        self.pool
159            .as_ref()
160            .map_or(1, rayon::ThreadPool::current_num_threads)
161    }
162
163    /// The last entered executor attempt. Proposal/checker failures leave it
164    /// unchanged. Scratch preflight or initial allocation failure records zero
165    /// entered slots. A later output reservation failure retains the completed
166    /// attempt's accounting without committing its results. `None` means this
167    /// executor has never been entered.
168    #[must_use]
169    pub const fn last_statistics(&self) -> Option<CompletionStatistics> {
170        self.last
171    }

173    pub(super) fn complete(
174        &mut self,
175        input: Input<'_>,
176        budget: &mut Budget<'_>,
177        statistics: &mut Statistics,
178        accepted: &mut Vec<bool>,
179        reduct: &mut crate::prepared_reduct::State,
180    ) -> Result<(), Incomplete> {
181        self.complete_with(
182            input,
183            budget,
184            statistics,
185            accepted,
186            reduct,
187            ReductWorkspace::reserve,
188        )
189    }

191    fn complete_with(
192        &mut self,
193        input: Input<'_>,
194        budget: &mut Budget<'_>,
195        statistics: &mut Statistics,
196        accepted: &mut Vec<bool>,
197        reduct: &mut crate::prepared_reduct::State,
198        mut reserve: impl FnMut(
199            &mut ReductWorkspace,
200            &PreparedReduct,
201            u64,
202            &Cancellation,
203        ) -> Result<(), Incomplete>,
204    ) -> Result<(), Incomplete> {
205        self.last = None;
206        let timed = statistics.phase_timings.is_some();
207        let mut progress = CompletionStatistics {
208            workers: self.workers(),
209            worker_original_validation: timed.then(PhaseMeasurement::default),
210            worker_reduct: timed.then(PhaseMeasurement::default),
211            ..CompletionStatistics::default()
212        };
213        let started = timed.then(Instant::now);
214        let result = (|| {
215            let result_slots = scratch::results(input.candidates.len())?;
216            result_slots.admit(0, self.max_scratch_bytes)?;
217            let residuals = input
218                .verdicts
219                .iter()
220                .filter(|v| **v == BatchVerdict::Residual)
221                .count();
222            if residuals != 0 {
223                reduct.ensure(input.theory, input.limits, budget, statistics)?;
224            }
225            let input = Input {
226                prepared: reduct.prepared(),
227                ..input
228            };
229            let requirements = input.prepared.map_or(Ok(result_slots), |owner| {
230                owner.scratch_requirements(input.candidates.len())
231            })?;
232            progress.peak_scratch_bytes = requirements.shared_bytes;
233            let admission =
234                requirements.admit(self.workers().min(residuals), self.max_scratch_bytes)?;
235            progress.effective_workers = admission.0;
236            progress.requested_scratch_bytes = admission.1;
237            progress.peak_scratch_bytes = requirements.shared_bytes + requirements.result_bytes;
238            accepted
239                .try_reserve_exact(input.candidates.len())
240                .map_err(|_| Incomplete::Allocation)?;
241            let mut workspaces = storage(admission.0)?;
242            let transient = input.prepared.map_or(0, scratch::transient);
243            let mut peak = u128::from(requirements.shared_bytes)
244                + u128::from(requirements.result_bytes)
245                + workspaces.capacity() as u128 * std::mem::size_of::<ReductWorkspace>() as u128;
246            record_peak(&mut progress, peak)?;
247            for _ in 0..admission.0 {
248                budget.cancellation.poll()?;
249                let mut workspace = ReductWorkspace::default();
250                // The region query has no prepared owner to reserve against;
251                // its worker workspace holds the evaluation alone.
252                let reservation = match input.prepared {
253                    Some(owner) => reserve(
254                        &mut workspace,
255                        owner,
256                        input.limits.max_reduct_bytes,
257                        budget.cancellation,
258                    ),
259                    None => Ok(()),
260                };
261                // Every header was counted with the outer allocation. Even a
262                // failed reservation can retain newly allocated query vectors.
263                peak += workspace.retained_bytes() - std::mem::size_of::<ReductWorkspace>() as u128
264                    + transient;
265                record_reservation(&mut progress, peak, reservation)?;
266                if progress.peak_scratch_bytes > self.max_scratch_bytes {
267                    return Err(Incomplete::CompletionScratch);
268                }
269                workspaces.push(workspace);
270            }
271            if let Some(pool) = &self.pool
272                && !workspaces.is_empty()
273            {
274                parallel(
275                    pool,
276                    input,
277                    budget,
278                    statistics,
279                    accepted,
280                    &mut progress,
281                    &mut workspaces,
282                )
283            } else {
284                scalar(
285                    input,
286                    budget,
287                    statistics,
288                    accepted,
289                    &mut progress,
290                    &mut workspaces,
291                )
292            }
293        })();
294        progress.elapsed = started.map(|start| start.elapsed());
295        self.last = Some(progress);
296        result
297    }
298}

300fn record_peak(progress: &mut CompletionStatistics, peak: u128) -> Result<(), Incomplete> {
301    let measured = u64::try_from(peak).map_err(|_| Incomplete::CounterOverflow);
302    progress.peak_scratch_bytes = measured.unwrap_or(u64::MAX);
303    measured.map(|_| ())
304}

306fn record_reservation(
307    progress: &mut CompletionStatistics,
308    peak: u128,
309    reservation: Result<(), Incomplete>,
310) -> Result<(), Incomplete> {
311    let measured = record_peak(progress, peak);
312    reservation?;
313    measured
314}

337fn classify(
338    input: Input<'_>,
339    candidate: &Interpretation,
340    verdict: BatchVerdict,
341    budget: &mut Budget<'_, impl Quota>,
342    statistics: &mut Statistics,
343    workspace: &mut ReductWorkspace,
344) -> Result<bool, Incomplete> {
345    budget.cancellation.poll()?;
346    match verdict {
347        BatchVerdict::NoProperSubset => Ok(true),
348        BatchVerdict::Refuted => Ok(false),
349        BatchVerdict::NotModel => Err(Incomplete::InvalidWitness),
350        BatchVerdict::Residual => {
351            let verdict = match (input.prepared, input.query) {
352                (Some(prepared), _) => {
353                    prepared.check_with(candidate, workspace, input.limits, budget, statistics)?
354                }
355                (None, Some(query)) => {
356                    let result = (|| {
357                        let (truth, scratch) = workspace.evaluate(
358                            candidate,
359                            input.limits,
360                            budget.cancellation,
361                            statistics,
362                        )?;
363                        if !truth.is_model() {
364                            return Err(Incomplete::InvalidWitness);
365                        }
366                        increment(&mut statistics.countermodel_queries)?;
367                        query.check(
368                            zetesis_ferraris::FrozenSubject::new(input.theory, truth.truth()),
369                            candidate,
370                            input.limits,
371                            budget,
372                            statistics,
373                            scratch,
374                        )
375                    })();
376                    workspace.finish_check(result, input.limits.max_reduct_bytes, statistics)?
377                }
378                (None, None) => return Err(Incomplete::InvalidWitness),
379            };
380            match Decision::from(verdict) {
381                Decision::Stable => Ok(true),
382                Decision::Refuted => Ok(false),
383                Decision::Invalid => Err(Incomplete::InvalidWitness),
384            }
385        }
386    }
387}

389fn scalar(
390    input: Input<'_>,
391    budget: &mut Budget<'_>,
392    statistics: &mut Statistics,
393    accepted: &mut Vec<bool>,
394    progress: &mut CompletionStatistics,
395    workspaces: &mut [ReductWorkspace],
396) -> Result<(), Incomplete> {
397    progress.worker_original_validation = None;
398    progress.worker_reduct = None;
399    let mut bounded = Budget {
400        quota: BoundedQuota(LocalQuota),
401        limits: budget.limits,
402        cancellation: budget.cancellation,
403        statistics: budget.statistics,
404    };
405    let mut unused = ReductWorkspace::default();
406    let workspace = workspaces.first_mut().unwrap_or(&mut unused);
407    let result = (|| {
408        for (candidate, &verdict) in input.candidates.iter().zip(input.verdicts) {
409            // Retain the existing scalar counters, work order and failure boundary.
410            let result = classify(
411                input,
412                candidate,
413                verdict,
414                &mut bounded,
415                statistics,
416                workspace,
417            );
418            entered(progress, verdict, &result);
419            accepted.push(result?);
420        }
421        Ok(())
422    })();
423    budget.statistics = bounded.statistics;
424    result
425}

427fn parallel(
428    pool: &rayon::ThreadPool,
429    input: Input<'_>,
430    budget: &mut Budget<'_>,
431    statistics: &mut Statistics,
432    accepted: &mut Vec<bool>,
433    progress: &mut CompletionStatistics,
434    workspaces: &mut [ReductWorkspace],
435) -> Result<(), Incomplete> {
436    let mut outcomes = storage(input.candidates.len())?;
437    outcomes.resize_with(input.candidates.len(), || None);
438    let shared = SharedBudget::new(budget.limits, budget.statistics);
439    let cancellation = budget.cancellation;
440    let timed = statistics.phase_timings.is_some();
441    let chunk_size = outcomes.len().div_ceil(progress.effective_workers.max(1));
442    pool.install(|| {
443        outcomes
444            .par_chunks_mut(chunk_size)
445            .zip(workspaces.par_iter_mut())
446            .enumerate()
447            .for_each(|(chunk, (slots, workspace))| {
448                for (offset, slot) in slots.iter_mut().enumerate() {
449                    *slot = Some(run(
450                        input,
451                        chunk * chunk_size + offset,
452                        &shared,
453                        cancellation,
454                        timed,
455                        workspace,
456                    ));
457                }
458            });
459    });
460    // Work is reserved before the operation. Read after join, including every
461    // failed job, before choosing an error in original candidate order.
462    shared.record(&mut budget.statistics);
463    let mut failure = None;
464    for (outcome, &verdict) in outcomes.into_iter().zip(input.verdicts) {
465        let outcome = outcome.expect("joined indexed execution fills every owned slot");
466        entered(progress, verdict, &outcome.result);
467        let merged = merge(statistics, budget, &outcome.statistics);
468        merge_timing(progress, outcome.statistics.phase_timings);
469        if failure.is_none() {
470            failure = merged
471                .err()
472                .or_else(|| outcome.result.as_ref().err().copied());
473        }
474        if let Ok(value) = outcome.result {
475            accepted.push(value);
476        }
477    }
478    failure.map_or(Ok(()), Err)
479}

481fn run(
482    input: Input<'_>,
483    index: usize,
484    shared: &SharedBudget,
485    cancellation: &Cancellation,
486    timed: bool,
487    workspace: &mut ReductWorkspace,
488) -> Outcome {
489    let mut budget = Budget {
490        quota: BoundedQuota(shared.lease(cancellation)),
491        limits: input.limits.search,
492        cancellation,
493        statistics: crate::SearchStatistics::default(),
494    };
495    let mut statistics = Statistics {
496        phase_timings: timed.then(SearchPhaseTimings::default),
497        ..Statistics::default()
498    };
499    let result = classify(
500        input,
501        &input.candidates[index],
502        input.verdicts[index],
503        &mut budget,
504        &mut statistics,
505        workspace,
506    );
507    statistics.search = budget.statistics;
508    Outcome { result, statistics }
509}

511fn entered(
512    progress: &mut CompletionStatistics,
513    verdict: BatchVerdict,
514    result: &Result<bool, Incomplete>,
515) {
516    progress.candidates += 1;
517    progress.residuals += usize::from(verdict == BatchVerdict::Residual);
518    progress.completed += usize::from(result.is_ok());
519    progress.failed += usize::from(result.is_err());
520    progress.residual_completed += usize::from(verdict == BatchVerdict::Residual && result.is_ok());
521    progress.residual_failed += usize::from(verdict == BatchVerdict::Residual && result.is_err());
522}

524fn merge(
525    statistics: &mut Statistics,
526    budget: &mut Budget<'_>,
527    worker: &Statistics,
528) -> Result<(), Incomplete> {
529    add(
530        &mut statistics.countermodel_queries,
531        worker.countermodel_queries,
532    )?;
533    add(&mut statistics.countermodels, worker.countermodels)?;
534    add(
535        &mut budget.statistics.propagations,
536        worker.search.propagations,
537    )?;
538    add(&mut budget.statistics.conflicts, worker.search.conflicts)?;
539    add(
540        &mut statistics.reduct.original_work,
541        worker.reduct.original_work,
542    )?;
543    add(
544        &mut statistics.reduct.parameter_work,
545        worker.reduct.parameter_work,
546    )?;
547    statistics.reduct.regions.add(worker.reduct.regions)?;
548    statistics.reduct.peak_workspace_bytes = statistics
549        .reduct
550        .peak_workspace_bytes
551        .max(worker.reduct.peak_workspace_bytes);
552    Ok(())
553}
554
555fn add(counter: &mut u64, value: u64) -> Result<(), Incomplete> {
556    *counter = counter
557        .checked_add(value)
558        .ok_or(Incomplete::CounterOverflow)?;
559    Ok(())
560}

562fn merge_timing(progress: &mut CompletionStatistics, worker: Option<SearchPhaseTimings>) {
563    if let Some(worker) = worker {
564        for (target, source) in [
565            (
566                &mut progress.worker_original_validation,
567                worker.original_validation,
568            ),
569            (&mut progress.worker_reduct, worker.reduct),
570        ] {
571            if let Some(target) = target {
572                match (
573                    target.calls.checked_add(source.calls),
574                    target.elapsed.checked_add(source.elapsed),
575                ) {
576                    (Some(calls), Some(elapsed)) if !target.overflowed && !source.overflowed => {
577                        target.calls = calls;
578                        target.elapsed = elapsed;
579                    }
580                    _ => target.overflowed = true,
581                }
582            }
583        }
584    }
585}
