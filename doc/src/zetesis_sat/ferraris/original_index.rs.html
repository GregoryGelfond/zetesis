1//! The original theory's narrowing index: one owner, built when a walk first
2//! needs it.
3//!
4//! An enumeration or a standalone membership check owns one
5//! [`OriginalIndex`]. Its candidate walk and every proper-subset query of its
6//! reduct read the same immutable [`IndexedTheory`], so the index is built at
7//! most once per owner, and only by [`OriginalIndex::ensure`]: no other code
8//! can construct an `IndexedTheory`. A run decided without a region walk, by a
9//! positive certificate or by the clause kernel, builds none.
10
11use std::sync::Arc;
12
13use zetesis_ferraris::{Narrower, Theory};
14
15use super::regions::stopped;
16use crate::Incomplete;
17
18/// One immutable original-theory index. Construction binds the index to the
19/// exact admitted instance; equal independently admitted DAGs are not its
20/// subject. Candidate and reduct traversals share this owner, never their
21/// mutable knowledge.
22#[derive(Debug)]
23pub(crate) struct IndexedTheory {
24    theory: Theory,
25    narrower: Narrower,
26}
27
28impl IndexedTheory {
29    pub(crate) fn theory(&self) -> &Theory {
30        &self.theory
31    }
32
33    pub(crate) fn narrower(&self) -> &Narrower {
34        &self.narrower
35    }
36
37    /// Authenticate before borrowing the indexed subject for any traversal.
38    pub(crate) fn subject(&self, theory: &Theory) -> Result<(&Theory, &Narrower), Incomplete> {
39        if self.theory.same_instance(theory) {
40            Ok((&self.theory, &self.narrower))
41        } else {
42            Err(Incomplete::WrongTheory)
43        }
44    }
45}
46
47/// The admitted theory and, once a walk has needed it, its index.
48#[derive(Debug)]
49pub(crate) struct OriginalIndex {
50    theory: Theory,
51    index: Option<Arc<IndexedTheory>>,
52}
53
54impl OriginalIndex {
55    /// An owner of the theory's index, not yet built.
56    pub(crate) fn new(theory: &Theory) -> Self {
57        Self {
58            theory: theory.clone(),
59            index: None,
60        }
61    }
62
63    /// The index, if a walk has built it; never builds.
64    pub(crate) fn get(&self) -> Option<&Arc<IndexedTheory>> {
65        self.index.as_ref()
66    }
67
68    /// The index, built on the first call that finds none.
69    ///
70    /// Before building, `charge` receives the index's documented cost: one
71    /// work unit per node and operand occurrence of the theory, which equals the built narrower's
72    /// [`Narrower::work`] and is known before building. A refused charge
73    /// builds nothing and leaves the owner unbuilt. An admitted charge stays
74    /// admitted even if the build then fails, as a failed narrowing keeps
75    /// its admitted prefix: `charge` is where the caller records it, in the
76    /// budget and in the receipt that counts index work, so the two agree on
77    /// every exit. A built index is returned without calling `charge` again.
78    ///
79    /// # Errors
80    /// The charge's refusal, or the build's allocation refusal.
81    pub(crate) fn ensure(
82        &mut self,
83        charge: impl FnOnce(u64) -> Result<(), Incomplete>,
84    ) -> Result<&Arc<IndexedTheory>, Incomplete> {
85        let index = if let Some(index) = self.index.take() {
86            index
87        } else {
88            let work = u64::try_from(
89                self.theory.nodes().len() as u128 + self.theory.parts().occurrences() as u128,
90            )
91            .map_err(|_| Incomplete::CounterOverflow)?;
92            charge(work)?;
93            let narrower = build(&self.theory)?;
94            debug_assert_eq!(narrower.work(), work, "the charge is the index's work");
95            Arc::new(IndexedTheory {
96                theory: self.theory.clone(),
97                narrower,
98            })
99        };
100        Ok(self.index.insert(index))
101    }
102}
103
104/// The one construction site of an original-theory narrower.
105fn build(theory: &Theory) -> Result<Narrower, Incomplete> {
106    #[cfg(test)]
107    hooks::observe(theory)?;
108    Narrower::try_new(theory).map_err(stopped)
109}
110
111/// Test-only observation of the construction site: a count of builds per
112/// admitted instance, and an injected build failure. Neither exists in a
113/// non-test build.
114#[cfg(test)]
115pub(crate) mod hooks {
116    use std::cell::Cell;
117    use std::sync::{Mutex, PoisonError};
118
119    use zetesis_ferraris::Theory;
120
121    use crate::Incomplete;
122
123    /// Every instance whose index construction was entered, once per entry.
124    /// Instances are compared by identity, so concurrently running tests,
125    /// each with its own admitted theory, never count each other's builds.
126    static BUILDS: Mutex<Vec<Theory>> = Mutex::new(Vec::new());
127
128    thread_local! {
129        /// Fail the next construction entered on this thread.
130        static FAIL: Cell<bool> = const { Cell::new(false) };
131    }
132
133    pub(super) fn observe(theory: &Theory) -> Result<(), Incomplete> {
134        BUILDS
135            .lock()
136            .unwrap_or_else(PoisonError::into_inner)
137            .push(theory.clone());
138        if FAIL.with(Cell::take) {
139            Err(Incomplete::Allocation)
140        } else {
141            Ok(())
142        }
143    }
144
145    /// Index constructions entered for this exact instance, on any thread.
146    pub(crate) fn builds(theory: &Theory) -> usize {
147        BUILDS
148            .lock()
149            .unwrap_or_else(PoisonError::into_inner)
150            .iter()
151            .filter(|built| built.same_instance(theory))
152            .count()
153    }
154
155    /// Make the next construction entered on this thread refuse its
156    /// allocation, as `Narrower::try_new` refuses one.
157    pub(crate) fn fail_next_build() {
158        FAIL.with(|fail| fail.set(true));
159    }
160}
