1//! Parallel production of classical candidates, with membership left to the caller.
2//!
3//! The frontier contains disjoint regions of the original candidate space. A
4//! worker removes one region, applies the same readings as the scalar traversal,
5//! then refutes it, returns its two children, or emits its classical leaf. No
6//! worker checks answer-set membership. A bounded round reserves one output slot
7//! per active region, so completed candidates plus active regions never exceed
8//! the batch allowance. A split or refutation returns that reservation.
9//!
10//! All workers join before the next stage runs. Their work leases then settle
11//! into the coordinator's cumulative budget, including failed attempts. Retained
12//! frontier regions supply the next round; a stop never establishes coverage.
13//! Candidate order depends on scheduling, while identity and coverage do not.
14

31/// Idle producers poll control while another producer owns the last region.
32const CONTROL_WAIT: Duration = Duration::from_millis(1);
33
34type PendingRegion = (Region, CandidateKnowledge);
35
36/// The same LIFO frontier, with incremental ownership observations. A region's
37/// payload changes only while a worker owns it outside this frontier. Count it
38/// on insertion/removal, visiting its restriction knowledges but no other entry.
39/// Actual disjoint allocations and their headers fit comfortably in u128 on the
40/// supported 32/64-bit hosts; no search counter or resource limit is changed.
41struct Frontier {
42    entries: Vec<PendingRegion>,
43    payload_bytes: u128,
44    statistics: RegionFrontierStatistics,
45}
46
47impl Default for Frontier {
48    fn default() -> Self {
49        let mut frontier = Self {
50            entries: Vec::new(),
51            payload_bytes: 0,
52            statistics: RegionFrontierStatistics::default(),
53        };
54        frontier.record();
55        frontier
56    }
57}
58
59impl Frontier {
60    fn new(entry: PendingRegion) -> Result<Self, Incomplete> {
61        let mut frontier = Self {
62            entries: crate::search::storage(1)?,
63            ..Self::default()
64        };
65        frontier.push(entry);
66        Ok(frontier)
67    }
68
69    fn len(&self) -> usize {
70        self.entries.len()
71    }
72
73    fn is_empty(&self) -> bool {
74        self.entries.is_empty()
75    }
76
77    fn try_reserve(&mut self, additional: usize) -> Result<(), std::collections::TryReserveError> {
78        let result = self.entries.try_reserve(additional);
79        self.record();
80        result
81    }
82
83    fn push(&mut self, entry: PendingRegion) {
84        self.payload_bytes += Self::payload(&entry);
85        self.entries.push(entry);
86        self.record();
87    }
88
89    fn pop(&mut self) -> Option<PendingRegion> {
90        let entry = self.entries.pop()?;
91        self.payload_bytes -= Self::payload(&entry);
92        self.record();
93        Some(entry)
94    }
95
96    /// Entry headers already occupy slots in the outer vector. Count only
97    /// their owned allocations here, including unused knowledge slots.
98    fn payload((region, knowledge): &PendingRegion) -> u128 {
99        region.retained_bytes() - size_of::<Region>() as u128 + knowledge.allocated_bytes()
100    }
101
102    fn record(&mut self) {
103        let statistics = &mut self.statistics;
104        statistics.regions = self.entries.len();
105        statistics.capacity = self.entries.capacity();
106        statistics.retained_bytes = size_of::<Vec<PendingRegion>>() as u128
107            + self.entries.capacity() as u128 * size_of::<PendingRegion>() as u128
108            + self.payload_bytes;
109        statistics.peak_regions = statistics.peak_regions.max(statistics.regions);
110        statistics.peak_capacity = statistics.peak_capacity.max(statistics.capacity);
111        statistics.peak_retained_bytes = statistics
112            .peak_retained_bytes
113            .max(statistics.retained_bytes);
114    }
115}

138/// Joined production retains its completed candidates even if another region
139/// stopped. Only an empty frontier with no stop establishes exhaustion.
140pub(crate) struct Produced {
141    pub(crate) exhausted: bool,
142    pub(crate) stopped: Option<Incomplete>,
143    pub(crate) original_validation: crate::PhaseMeasurement,
144}

146impl RegionProposals {
147    pub(crate) fn new(
148        theory: &Theory,
149        workers: NonZeroUsize,
150        budget: &mut Budget<'_>,
151    ) -> Result<Self, Incomplete> {
152        let regions::Opened {
153            producers,
154            index,
155            statistics,
156        } = regions::open(theory, budget)?;
157        let pool = rayon::ThreadPoolBuilder::new()
158            .num_threads(workers.get())
159            .thread_name(|index| format!("zetesis-proposal-{index}"))
160            .build()
161            .map_err(|_| Incomplete::Allocation)?;
162        let pending = Frontier::new((
163            Region::all_open(theory.atom_count()),
164            CandidateKnowledge::new(index.narrower().knowledge()),
165        ))?;
166        Ok(Self {
167            producers,
168            index,
169            restrictions: Conditions::default(),
170            pending,
171            pool,
172            statistics,
173            filter: None,
174        })
175    }
176
177    pub(crate) fn index(&self) -> &Arc<IndexedTheory> {
178        &self.index
179    }
180
181    pub(crate) const fn statistics(&self) -> RegionSearchStatistics {
182        RegionSearchStatistics {
183            frontier: Some(self.pending.statistics),
184            ..self.statistics
185        }
186    }
187
188    pub(crate) fn restrict(
189        &mut self,
190        restriction: &Theory,
191        budget: &mut Budget<'_>,
192    ) -> Result<(), Incomplete> {
193        self.restrictions
194            .permanent
195            .try_reserve(1)
196            .map_err(|_| Incomplete::Allocation)?;
197        let narrower = Narrower::try_new(restriction).map_err(super::regions::stopped)?;
198        budget.charge(narrower.work())?;
199        self.statistics.counts.work = self
200            .statistics
201            .counts
202            .work
203            .checked_add(narrower.work())
204            .ok_or(Incomplete::CounterOverflow)?;
205        self.restrictions
206            .permanent
207            .push((restriction.clone(), narrower));
208        Ok(())
209    }
210
211    /// Commit only after preparation and every charge succeeds.
212    pub(crate) fn tighten(
213        &mut self,
214        restriction: &Theory,
215        generation: u64,
216        budget: &mut Budget<'_>,
217    ) -> Result<(), Incomplete> {
218        budget.cancellation.poll()?;
219        let bound = Bound::prepare(restriction, generation)?;
220        budget.charge(bound.work())?;
221        let work = self
222            .statistics
223            .counts
224            .work
225            .checked_add(bound.work())
226            .ok_or(Incomplete::CounterOverflow)?;
227        self.statistics.counts.work = work;
228        budget.cancellation.poll()?;
229        self.restrictions.bound = Some(bound);
230        Ok(())
231    }

249    /// Fill pre-admitted output slots without evaluating a reduct. The output
250    /// is empty on entry; its allocation remains the caller's batch ownership.
251    /// A zero remaining candidate allowance still permits coverage work until
252    /// either the frontier is refuted or another classical leaf is encountered.
253    pub(crate) fn fill(
254        &mut self,
255        theory: &Theory,
256        maximum: usize,
257        remaining: u64,
258        budget: &mut Budget<'_>,
259        output: &mut Vec<Interpretation>,
260        timed: bool,
261    ) -> Produced {
262        debug_assert!(output.is_empty());
263        debug_assert!(maximum > 0 && output.capacity() >= maximum);
264        let (theory, narrower) = match self.index.subject(theory) {
265            Ok(subject) => subject,
266            Err(error) => {
267                return Produced {
268                    exhausted: false,
269                    stopped: Some(error),
270                    original_validation: crate::PhaseMeasurement::default(),
271                };
272            }
273        };
274        let allowance = SharedBudget::new(budget.limits, budget.statistics);
275        let round = Round {
276            theory,
277            filter: self.filter.as_ref(),
278            timed,
279            producers: self.producers.as_ref(),
280            narrower,
281            restrictions: &self.restrictions,
282            allowance: &allowance,
283            limits: budget.limits,
284            cancellation: budget.cancellation,
285            maximum: maximum.min(usize::try_from(remaining).unwrap_or(usize::MAX).max(1)),
286            remaining,
287            changed: Condvar::new(),
288            state: Mutex::new(State {
289                pending: std::mem::take(&mut self.pending),
290                output: std::mem::take(output),
291                active: 0,
292                counts: RegionCounts::default(),
293                stopped: None,
294                original_validation: crate::PhaseMeasurement::default(),
295            }),
296        };
297        self.pool.install(|| {
298            (0..self.pool.current_num_threads())
299                .into_par_iter()
300                .for_each(|_| {
301                    if std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| round.work()))
302                        .is_err()
303                    {
304                        round.stop(Incomplete::WorkerPanicked);
305                    }
306                });
307        });
308        allowance.record(&mut budget.statistics);
309        let mut state = round
310            .state
311            .into_inner()
312            .unwrap_or_else(PoisonError::into_inner);
313        if let Err(error) = self.statistics.counts.add(state.counts) {
314            state.stopped.get_or_insert(error);
315        }
316        self.pending = state.pending;
317        *output = state.output;
318        Produced {
319            exhausted: state.stopped.is_none() && self.pending.is_empty(),
320            stopped: state.stopped,
321            original_validation: state.original_validation,
322        }
323    }
324}

339struct Round<'a> {
340    theory: &'a Theory,
341    filter: Option<&'a crate::region_filter::Filter>,
342    timed: bool,
343    producers: Option<&'a Producers>,
344    narrower: &'a Narrower,
345    restrictions: &'a Conditions<(Theory, Narrower)>,
346    allowance: &'a SharedBudget,
347    limits: crate::SearchLimits,
348    cancellation: &'a Cancellation,
349    maximum: usize,
350    remaining: u64,
351    changed: Condvar,
352    state: Mutex<State>,
353}
354
355enum Step {
356    Refuted,
357    Split(PendingRegion, PendingRegion),
358    Candidate(Interpretation),
359}
360
361impl Round<'_> {
362    fn lock(&self) -> MutexGuard<'_, State> {
363        self.state.lock().unwrap_or_else(PoisonError::into_inner)
364    }
365
366    fn stop(&self, error: Incomplete) {
367        self.lock().stopped.get_or_insert(error);
368        self.changed.notify_all();
369    }
370
371    fn take(&self) -> Option<PendingRegion> {
372        let mut state = self.lock();
373        loop {
374            if state.stopped.is_some() || state.output.len() == self.maximum {
375                return None;
376            }
377            if let Err(error) = self.cancellation.poll() {
378                state.stopped.get_or_insert(error.into());
379                self.changed.notify_all();
380                return None;
381            }
382            if state.active + state.output.len() < self.maximum {
383                if let Some(entry) = state.pending.pop() {
384                    state.active += 1;
385                    return Some(entry);
386                }
387                if state.active == 0 {
388                    return None;
389                }
390            }
391            state = self
392                .changed
393                .wait_timeout(state, CONTROL_WAIT)
394                .unwrap_or_else(PoisonError::into_inner)
395                .0;
396        }
397    }
398
399    fn work(&self) {
400        let mut filter = None;
401        let mut timings = self.timed.then(crate::SearchPhaseTimings::default);
402        while let Some(mut entry) = self.take() {
403            let mut counts = RegionCounts {
404                regions: 1,
405                ..Default::default()
406            };
407            // Settle permits before touching the frontier mutex or waiting;
408            // an idle producer must not strand another worker's allowance.
409            let result = {
410                let mut budget = Budget {
411                    quota: self.allowance.lease(self.cancellation),
412                    limits: self.limits,
413                    cancellation: self.cancellation,
414                    statistics: SearchStatistics::default(),
415                };
416                self.step(
417                    &mut entry,
418                    &mut budget,
419                    &mut counts,
420                    &mut filter,
421                    &mut timings,
422                )
423            };
424            let mut state = self.lock();
425            state.active -= 1;
426            if let Err(error) = state.counts.add(counts) {
427                state.stopped.get_or_insert(error);
428            }
429            match result {
430                Ok(Step::Refuted) => {}
431                Ok(Step::Candidate(candidate)) => state.output.push(candidate),
432                Ok(Step::Split(cut, held)) => {
433                    if state.pending.try_reserve(2).is_ok() {
434                        state.pending.push(held);
435                        state.pending.push(cut);
436                    } else {
437                        state.stopped.get_or_insert(Incomplete::Allocation);
438                    }
439                }
440                Err(error) => {
441                    // This region remains unresolved. Retain it when storage
442                    // permits; the explicit stop is authoritative either way.
443                    if state.pending.try_reserve(1).is_ok() {
444                        state.pending.push(entry);
445                    }
446                    state.stopped.get_or_insert(error);
447                }
448            }
449            self.changed.notify_all();
450        }
451        if let Some(timings) = timings {
452            self.lock()
453                .original_validation
454                .merge(timings.original_validation);
455        }
456    }
457
458    fn step<'a>(
459        &'a self,
460        (region, knowledge): &mut PendingRegion,
461        budget: &mut Budget<'a, WorkLease<'a>>,
462        counts: &mut RegionCounts,
463        filter: &mut Option<crate::region_filter::Worker<'a>>,
464        timings: &mut Option<crate::SearchPhaseTimings>,
465    ) -> Result<Step, Incomplete> {
466        if regions::narrow(
467            (self.theory, self.narrower),
468            self.producers,
469            self.restrictions,
470            region,
471            knowledge,
472            budget,
473            counts,
474        )? == Narrowing::Refuted
475            || match self.filter {
476                Some(factory) => {
477                    factory.check(filter, self.theory, region, self.cancellation, timings)?
478                        == crate::RegionFeasibility::Refuted
479                }
480                None => false,
481            }
482        {
483            counts.refuted = 1;
484            return Ok(Step::Refuted);
485        }
486        if let Some(atom) = region.split_atom() {
487            budget.decide()?;
488            // Take the region out to split it by value; the emptied placeholder
489            // (no atoms, no allocation) is discarded with this pending entry.
490            let (cut, held) = std::mem::replace(region, Region::all_open(0)).split(atom);
491            let cut = (cut, knowledge.clone());
492            let held = (held, std::mem::take(knowledge));
493            return Ok(Step::Split(cut, held));
494        }
495        if self.remaining == 0 {
496            return Err(Incomplete::CandidateLimit);
497        }
498        counts.leaves = 1;
499        regions::leaf_interpretation(self.theory, region).map(Step::Candidate)
500    }
