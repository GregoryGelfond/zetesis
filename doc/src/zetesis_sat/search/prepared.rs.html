1//! Retained watch topology and unconditional consequences of one immutable CNF.
2
3use std::mem::size_of;
4
5use super::{Budget, Quota, Solve, Workspace, increment, search_seeded};
6use crate::{Assignment, Cnf, Incomplete, Literal};
7
8/// The caller binds this workspace to an exact immutable CNF owner and must
9/// invalidate it before changing owners. Shape equality is insufficient.
10/// `indexed` certifies that every nonunit clause has two distinct positions and
11/// exactly one list node for each position. The separate base capability records
12/// only unit consequences of that immutable CNF, before any query parameters.
13#[derive(Debug, Default)]
14pub(crate) struct PreparedWorkspace {
15    workspace: Workspace,
16    indexed: bool,
17    base: BaseClosure,
18}
19
20#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
21enum BaseClosure {
22    /// No completed unconditional propagation result may be reused.
23    #[default]
24    Unprepared,
25    /// All assignments in this prefix follow from the immutable CNF alone.
26    /// Quiescence does not assert that every clause is satisfied.
27    Closed { trail_len: usize },
28    /// The immutable CNF's unit propagation contradicts, without parameters.
29    Refuted,
30}
31
32impl PreparedWorkspace {
33    pub(crate) fn invalidate(&mut self) {
34        self.indexed = false;
35        self.base = BaseClosure::Unprepared;
36    }
37
38    pub(crate) fn reserve(&mut self, variables: usize, clauses: usize) -> Result<(), Incomplete> {
39        self.workspace.reserve(variables, clauses)
40    }
41
42    pub(crate) fn retained_bytes(&self) -> u128 {
43        (size_of::<Self>() - size_of::<Workspace>()) as u128 + self.workspace.retained_bytes()
44    }
45
46    pub(crate) fn required_bytes(&self, variables: usize, clauses: usize) -> u128 {
47        (size_of::<Self>() - size_of::<Workspace>()) as u128
48            + self.workspace.required_bytes(variables, clauses)
49    }
50
51    /// Unit indices and the empty-clause fact come from the same immutable CNF.
52    /// A failed initial index is rebuilt on retry. Once complete, watch moves
53    /// preserve the registry even if inspection or later search is interrupted:
54    /// every fallible replacement inspection precedes its no-fail link mutation.
55    pub(crate) fn query(
56        &mut self,
57        cnf: &Cnf,
58        units: &[usize],
59        has_empty_clause: bool,
60        assumptions: &[Literal],
61        budget: &mut Budget<'_, impl Quota>,
62    ) -> Solve {
63        match self.run(cnf, units, has_empty_clause, assumptions, budget) {
64            Ok(Some(assignment)) => Solve::Sat(assignment),
65            Ok(None) => Solve::Unsat,
66            Err(error) => Solve::Inconclusive(error),
67        }
68    }
69
70    fn run(
71        &mut self,
72        cnf: &Cnf,
73        units: &[usize],
74        has_empty_clause: bool,
75        assumptions: &[Literal],
76        budget: &mut Budget<'_, impl Quota>,
77    ) -> Result<Option<Assignment>, Incomplete> {
78        self.reset(cnf, budget)?;
79        if !self.prepare_base(cnf, units, has_empty_clause, budget)? {
80            // Each query encountering a retained contradiction records that
81            // conflict outcome; work/propagations count only executed steps.
82            increment(&mut budget.statistics.conflicts)?;
83            return Ok(None);
84        }
85        search_seeded(&mut self.workspace.0, cnf, assumptions, budget)
86    }
87
88    fn prepare_base(
89        &mut self,
90        cnf: &Cnf,
91        units: &[usize],
92        has_empty_clause: bool,
93        budget: &mut Budget<'_, impl Quota>,
94    ) -> Result<bool, Incomplete> {
95        match self.base {
96            BaseClosure::Closed { .. } => return Ok(true),
97            BaseClosure::Refuted => return Ok(false),
98            BaseClosure::Unprepared => (),
99        }
100        if has_empty_clause {
101            self.base = BaseClosure::Refuted;
102            return Ok(false);
103        }
104        let state = &mut self.workspace.0;
105        for &clause in units {
106            budget.tick()?;
107            let literal = cnf.clause_at(clause).at(0);
108            if state.value(literal).is_none() {
109                increment(&mut budget.statistics.propagations)?;
110            }
111            if !state.assign(literal) {
112                self.base = BaseClosure::Refuted;
113                return Ok(false);
114            }
115        }
116        // No candidate parameter has been assigned. A stopped propagation keeps
117        // Unprepared; reset must undo its whole unfinished trail before retry.
118        if !state.propagate(cnf, budget)? {
119            self.base = BaseClosure::Refuted;
120            return Ok(false);
121        }
122        self.base = BaseClosure::Closed {
123            trail_len: state.trail.len(),
124        };
125        Ok(true)
126    }
127
128    fn reset(&mut self, cnf: &Cnf, budget: &mut Budget<'_, impl Quota>) -> Result<(), Incomplete> {
129        let state = &mut self.workspace.0;
130        if !self.indexed {
131            // reset may stop while filling any array; none of that partial
132            // topology is reusable. Publish readiness only after the full scan.
133            state.reset(cnf, budget)?;
134            for clause in 0..state.base_clauses {
135                budget.tick()?;
136                if cnf.clause_at(clause).len() >= 2 {
137                    state.index_clause(cnf, clause)?;
138                }
139            }
140            self.indexed = true;
141            return Ok(());
142        }
143        budget.tick()?;
144        let base_len = match self.base {
145            BaseClosure::Closed { trail_len } => trail_len,
146            BaseClosure::Unprepared => 0,
147            BaseClosure::Refuted => return Ok(()),
148        };
149        // Every Some value has exactly one live trail entry. Admission precedes
150        // each pop/clear pair, preserving unfinished suffix undo on a stop.
151        // The completed base prefix is never a candidate's truth assumption.
152        while state.trail.len() > base_len {
153            budget.tick()?;
154            let literal = state.trail.pop().ok_or(Incomplete::InvalidWitness)?;
155            state.values[literal.variable()] = None;
156        }
157        state.decisions.clear();
158        state.order.clear();
159        state.ranks.clear();
160        state.ordering.clear();
161        // At the base fixpoint, a base-false watch is protected by a base-true
162        // other watch. That true watch never moves; later moved watches were
163        // nonfalse under a stronger query assignment and remain nonfalse after
164        // suffix undo. Thus no completed base event needs to be replayed.
165        state.propagation_head = base_len;
166        state.next_position = 0;
167        Ok(())
168    }
169}
