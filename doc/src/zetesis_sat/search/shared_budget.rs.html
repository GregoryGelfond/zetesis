1//! Scoped work permits for joined queries and parallel region walks.
2//!
3//! Committed work + available permits + outstanding grants equals the initial
4//! allowance. A lease consumes its grant locally, then commits used permits and
5//! returns unused ones when it settles: before its holder waits, delivers or
6//! reserves, and on every exit. A settled lease refills on its next use.
7//! Outstanding grants are not reported as spent work. Decision reservations
8//! remain individually shared and nonblocking.
9
10use std::cell::Cell;
11use std::sync::atomic::{AtomicU64, Ordering};
12use std::sync::{Condvar, Mutex, MutexGuard};
13use std::time::Duration;
14
15use crate::{Cancellation, Incomplete, SearchLimits, SearchStatistics};
16
17/// A scheduling policy, not a measured crossover: an incremental grant holds
18/// at most this many permits. A region walker keeps its lease across its
19/// consecutive regions, which commonly use a few hundred permits each, so one
20/// grant serves many regions and the shared ledger lock is taken about twice
21/// per grant rather than twice per region. Accounted kernels reserve their
22/// complete bounded allowance separately. No grant changes the charged
23/// operation sequence, so the work limit still bounds the identical total; a
24/// larger grant only coarsens how the shared allowance is parcelled out among
25/// workers and how often a lease rechecks cooperative control.
26const WORK_QUANTUM: u64 = 16384;
27/// A waiting query rechecks cooperative control at least once per timed wait.
28/// Lock acquisition and OS scheduling do not provide a hard response deadline.
29const CONTROL_WAIT: Duration = Duration::from_millis(1);
30
31pub(crate) struct SharedBudget {
32    limits: SearchLimits,
33    work: Mutex<Work>,
34    returned: Condvar,
35    decisions: AtomicU64,
36}
37
38struct Work {
39    spent: u64,
40    available: u64,
41    outstanding: u64,
42}
43
44impl Work {
45    /// None means another lease can still return permits, not exhaustion.
46    fn grant(&mut self) -> Result<Option<u64>, Incomplete> {
47        if self.available != 0 {
48            let grant = self.available.min(WORK_QUANTUM);
49            self.available -= grant;
50            self.outstanding += grant;
51            Ok(Some(grant))
52        } else if self.outstanding == 0 {
53            Err(Incomplete::WorkLimit)
54        } else {
55            Ok(None)
56        }
57    }
58
59    /// An accounted kernel needs its allowance before entering. Wait for
60    /// outstanding grants when they can still cover a shortfall; otherwise
61    /// the remaining allowance bounds a possibly incomplete kernel attempt.
62    fn reserve(&mut self, wanted: u64) -> Option<u64> {
63        if self.available < wanted && self.outstanding != 0 {
64            return None;
65        }
66        let granted = wanted.min(self.available);
67        self.available -= granted;
68        self.outstanding += granted;
69        Some(granted)
70    }
71}
72
73impl SharedBudget {
74    pub(crate) fn new(limits: SearchLimits, spent: SearchStatistics) -> Self {
75        Self {
76            limits,
77            work: Mutex::new(Work {
78                spent: spent.work,
79                available: limits.max_work.saturating_sub(spent.work),
80                outstanding: 0,
81            }),
82            returned: Condvar::new(),
83            decisions: AtomicU64::new(spent.decisions),
84        }
85    }
86
87    pub(crate) fn lease<'a>(&'a self, cancellation: &'a Cancellation) -> WorkLease<'a> {
88        WorkLease {
89            shared: self,
90            cancellation,
91            granted: Cell::new(0),
92            remaining: Cell::new(0),
93        }
94    }
95
96    fn lock(&self) -> MutexGuard<'_, Work> {
97        // Only infallible conservation updates run while held: no query work,
98        // allocation, user callback or control poll can unwind through this lock.
99        self.work
100            .lock()
101            .unwrap_or_else(std::sync::PoisonError::into_inner)
102    }
103
104    fn decide(&self) -> Result<(), Incomplete> {
105        self.decisions
106            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |spent| {
107                (spent < self.limits.max_decisions).then(|| spent + 1)
108            })
109            .map(|_| ())
110            .map_err(|_| Incomplete::DecisionLimit)
111    }
112
113    /// Commit `work` permits at once, for work done outside any lease. The
114    /// permits are taken from those available; when they do not cover it
115    /// the charge is refused as exhaustion without waiting for outstanding
116    /// grants or kernel reservations to return, so held allowances can cause
117    /// an early refusal.
118    pub(crate) fn charge(&self, work: u64) -> Result<(), Incomplete> {
119        let mut permits = self.lock();
120        if permits.available < work {
121            return Err(Incomplete::WorkLimit);
122        }
123        permits.available -= work;
124        permits.spent += work;
125        Ok(())
126    }
127
128    /// The work committed so far, while leases may still hold permits: a
129    /// lower bound on the work spent, exact once every lease has dropped.
130    pub(crate) fn snapshot(&self, statistics: &mut SearchStatistics) {
131        let work = self.lock();
132        statistics.work = work.spent;
133        statistics.decisions = self.decisions.load(Ordering::Relaxed);
134    }
135
136    /// Every lease has dropped and all workers have joined before this read,
137    /// so the snapshot is exact.
138    pub(crate) fn record(&self, statistics: &mut SearchStatistics) {
139        debug_assert_eq!(self.lock().outstanding, 0);
140        self.snapshot(statistics);
141    }
142}

144/// A query owns at most one grant and never waits while retaining a permit.
145/// The Cells are local to that query; the lease is deliberately not Sync.
146pub(crate) struct WorkLease<'a> {
147    shared: &'a SharedBudget,
148    cancellation: &'a Cancellation,
149    granted: Cell<u64>,
150    remaining: Cell<u64>,
151}
152
153impl WorkLease<'_> {
154    /// Reserve an accounted kernel's finite work bound before it runs.
155    /// Settle this lease first, so waiting cannot retain the permits needed
156    /// by another reservation. Returned attempts settle their reported work;
157    /// an unwinding attempt conservatively consumes its whole reservation.
158    pub(crate) fn reserve(&mut self, wanted: u64) -> Result<WorkReservation<'_>, Incomplete> {
159        self.settle();
160        loop {
161            self.cancellation.poll()?;
162            let mut work = self.shared.lock();
163            if let Some(granted) = work.reserve(wanted) {
164                return Ok(WorkReservation {
165                    shared: self.shared,
166                    granted,
167                    used: granted,
168                });
169            }
170            drop(
171                self.shared
172                    .returned
173                    .wait_timeout(work, CONTROL_WAIT)
174                    .unwrap_or_else(std::sync::PoisonError::into_inner),
175            );
176        }
177    }
178
179    pub(crate) fn tick(&self) -> Result<(), Incomplete> {
180        if self.remaining.get() == 0 {
181            self.refill()?;
182        }
183        self.remaining.set(self.remaining.get() - 1);
184        Ok(())
185    }
186
187    pub(crate) fn decide(&self) -> Result<(), Incomplete> {
188        self.shared.decide()
189    }
190
191    /// Consume `amount` permits, refilling grant by grant; a refill that
192    /// finds the allowance exhausted fails after the permits already taken.
193    pub(crate) fn take(&self, mut amount: u64) -> Result<(), Incomplete> {
194        while amount > 0 {
195            if self.remaining.get() == 0 {
196                self.refill()?;
197            }
198            let taken = self.remaining.get().min(amount);
199            self.remaining.set(self.remaining.get() - taken);
200            amount -= taken;
201        }
202        Ok(())
203    }
204
205    fn refill(&self) -> Result<(), Incomplete> {
206        debug_assert_eq!(self.remaining.get(), 0);
207        loop {
208            if let Err(error) = self.cancellation.poll() {
209                // A refused renewal still commits the consumed grant before
210                // returning, without acquiring another allowance.
211                self.settle();
212                return Err(error.into());
213            }
214            let mut work = self.shared.lock();
215            // The old grant is wholly consumed. Settle it and acquire its
216            // replacement under one lock; no unused permits become available.
217            let consumed = self.granted.replace(0);
218            work.outstanding -= consumed;
219            work.spent += consumed;
220            match work.grant() {
221                Ok(Some(grant)) => {
222                    self.granted.set(grant);
223                    self.remaining.set(grant);
224                    return Ok(());
225                }
226                Err(error) => {
227                    // Settling the last outstanding grant establishes true
228                    // exhaustion. Peers must wake even without a permit return.
229                    drop(work);
230                    self.shared.returned.notify_all();
231                    return Err(error);
232                }
233                Ok(None) => {}
234            }
235            // Another query can still return unused permits. The timed wait
236            // releases this sole shared lock, and the next iteration polls.
237            // This lease retains no grant and has enabled no waiting peer.
238            drop(
239                self.shared
240                    .returned
241                    .wait_timeout(work, CONTROL_WAIT)
242                    .unwrap_or_else(std::sync::PoisonError::into_inner),
243            );
244        }
245    }
246
247    /// Commit the consumed part of the current grant and return its unused
248    /// part; the lease stays usable and refills on its next use. A holder calls
249    /// this before any wait, delivery or reservation, so a peer never waits for
250    /// permits that this lease is not using.
251    pub(crate) fn settle(&self) {
252        let granted = self.granted.replace(0);
253        if granted == 0 {
254            return;
255        }
256        let unused = self.remaining.replace(0);
257        {
258            let mut work = self.shared.lock();
259            work.outstanding -= granted;
260            work.spent += granted - unused;
261            work.available += unused;
262        }
263        self.shared.returned.notify_all();
264    }
265}
266
267impl Drop for WorkLease<'_> {
268    fn drop(&mut self) {
269        self.settle();
270    }
271}

273/// An accounted operation's pre-reserved allowance. A returned attempt records
274/// its actual work and refunds the unused part. Without a returned receipt,
275/// including unwinding, the whole grant is charged; the caller must report
276/// that failure without establishing enumeration coverage.
277pub(crate) struct WorkReservation<'a> {
278    shared: &'a SharedBudget,
279    granted: u64,
280    used: u64,
281}
282
283impl WorkReservation<'_> {
284    pub(crate) fn allowance(&self) -> u64 {
285        self.granted
286    }
287
288    pub(crate) fn finish(mut self, used: u64) -> Result<(), Incomplete> {
289        if used > self.granted {
290            return Err(Incomplete::InvalidWitness);
291        }
292        self.used = used;
293        Ok(())
294    }
295}
296
297impl Drop for WorkReservation<'_> {
298    fn drop(&mut self) {
299        {
300            let mut work = self.shared.lock();
301            work.outstanding -= self.granted;
302            work.spent += self.used;
303            work.available += self.granted - self.used;
304        }
305        self.shared.returned.notify_all();
306    }
307}
