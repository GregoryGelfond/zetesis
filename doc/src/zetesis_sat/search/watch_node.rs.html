1//! Optional watch links retain an index without a separate presence word.
2
3use std::num::NonZeroUsize;
4
5use crate::AdmissionError;
6
7/// A node in the fixed two-per-clause intrusive watch registry.
8///
9/// The stored successor represents a zero-based index; absence remains `None`.
10/// The transparent nonzero representation gives `Option<WatchNode>` the layout
11/// of one `usize`. Construction checks the sole excluded index, `usize::MAX`.
12/// CNF admission already excludes it by bounding twice the clause count.
13/// Moving a link copies this identity without allocation or re-encoding.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15#[repr(transparent)]
16pub(super) struct WatchNode(NonZeroUsize);
17
18impl WatchNode {
19    pub(super) fn new(index: usize) -> Result<Self, AdmissionError> {
20        index
21            .checked_add(1)
22            .and_then(NonZeroUsize::new)
23            .map(Self)
24            .ok_or(AdmissionError::Overflow)
25    }
26
27    pub(super) const fn index(self) -> usize {
28        self.0.get() - 1
29    }
30}

32#[cfg(test)]
33mod tests {

35    use proptest::prelude::*;
36
37    proptest! {
38        #[test]
39        fn watch_identity_round_trips(index in 0..usize::MAX) {
40            prop_assert_eq!(WatchNode::new(index).unwrap().index(), index);
41        }
42    }
43
44    #[test]
45    fn last_representable_identity_round_trips() {
46        assert_eq!(
47            WatchNode::new(usize::MAX - 1).unwrap().index(),
48            usize::MAX - 1
49        );
50    }
51
52    #[test]
53    fn unrepresentable_identity_is_refused() {
54        assert_eq!(WatchNode::new(usize::MAX), Err(AdmissionError::Overflow));
55    }

58    fn optional_identity_occupies_one_index_word() {
59        assert_eq!(size_of::<Option<WatchNode>>(), size_of::<usize>());
