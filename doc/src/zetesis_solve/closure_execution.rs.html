1//! Ordinary independent CPU closure accounting, summed over completed checks.
2
3use zetesis_cpu::{StaticStatistics, Statistics};
4
5/// Cumulative counters of the independent CPU closure routes, lazy and eager.
6///
7/// Each completed check returns its own counters and this receipt sums them.
8/// A stopped check returns none, so its partial work is counted nowhere: the
9/// sums describe completed checks only, and `stopped_checks` says how many
10/// checks are missing from them. Units follow the route's `max_work` ceiling
11/// and differ between the routes. These are work counts, not process memory.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub struct ClosureExecutionStatistics {
14    /// The route that ran, whose units the counters use, carrying the
15    /// counters only it produces.
16    pub route: ClosureRoute,
17    /// Checks that returned a complete closure.
18    pub completed_checks: u64,
19    /// Checks stopped by a resource ceiling, cancellation or deadline.
20    pub stopped_checks: u64,
21    /// Lazy source rounds or eager rule passes, including each check's final
22    /// unchanged one.
23    pub rounds: u64,
24    /// Charged operations, in the route's units.
25    pub work: u64,
26    /// Atoms in the completed closures, summed over checks.
27    pub derived_atoms: u64,
28}
29
30/// The independent CPU closure route a receipt sums, with the counters only
31/// it produces.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum ClosureRoute {
34    /// Source joins without a complete ground rule store, counted in join
35    /// and copy units, with the joins' own counters.
36    Lazy(ClosureJoinStatistics),
37    /// Rule passes over a materialized static program, counted in eager
38    /// scan units.
39    Eager,
40}
41
42impl ClosureRoute {
43    /// Stable spelling for execution reports.
44    #[must_use]
45    pub const fn label(self) -> &'static str {
46        match self {
47            Self::Lazy(_) => "lazy",
48            Self::Eager => "eager",
49        }
50    }
51}
52
53/// Counters of the lazy route's source joins, summed over completed checks.
54#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
55pub struct ClosureJoinStatistics {
56    /// Work spent on retained typed catalogs; included in `work`.
57    pub catalog_work: u64,
58    /// Fully matched bindings visited by the round schedules.
59    pub bindings: u64,
60    /// Source rows offered to the whole-row matcher; bounded by `work`.
61    pub tuple_probes: u64,
62    /// Derived heads recorded as a bit of a dense relation, for which no atom
63    /// was built before the model was assembled.
64    pub dense_heads: u64,
65    /// Blocks of body rows joined into a head's pending marks a word at a
66    /// time, in place of binding their rows one by one.
67    pub block_steps: u64,
68    /// Largest named closure envelope any completed check admitted or
69    /// reserved; not an observed allocation peak and not RSS.
70    pub peak_closure_bytes: usize,
71}
72
73impl ClosureExecutionStatistics {
74    /// The empty receipt of the route about to run.
75    pub(crate) const fn new(route: ClosureRoute) -> Self {
76        Self {
77            route,
78            completed_checks: 0,
79            stopped_checks: 0,
80            rounds: 0,
81            work: 0,
82            derived_atoms: 0,
83        }
84    }
85
86    pub(crate) fn stopped(&mut self) -> Result<(), crate::SolveError> {
87        self.stopped_checks = add(self.stopped_checks, 1)?;
88        Ok(())
89    }
90
91    /// Sum one completed lazy check. Every field is checked before any is
92    /// replaced, so an overflow leaves the receipt as it was.
93    pub(crate) fn completed_lazy(&mut self, check: &Statistics) -> Result<(), crate::SolveError> {
94        let ClosureRoute::Lazy(joins) = self.route else {
95            unreachable!("a lazy check is summed into a lazy receipt")
96        };
97        let next = Self {
98            completed_checks: add(self.completed_checks, 1)?,
99            rounds: add(self.rounds, check.rounds)?,
100            work: add(self.work, check.work)?,
101            derived_atoms: add(self.derived_atoms, count(check.derived_atoms)?)?,
102            route: ClosureRoute::Lazy(ClosureJoinStatistics {
103                catalog_work: add(joins.catalog_work, check.catalog_work)?,
104                bindings: add(joins.bindings, check.bindings)?,
105                tuple_probes: add(joins.tuple_probes, check.tuple_probes)?,
106                dense_heads: add(joins.dense_heads, check.dense_heads)?,
107                block_steps: add(joins.block_steps, check.block_steps)?,
108                peak_closure_bytes: joins.peak_closure_bytes.max(check.peak_closure_bytes),
109            }),
110            ..self.clone()
111        };
112        *self = next;
113        Ok(())
114    }
115
116    /// Sum one completed eager check, with the same overflow discipline.
117    pub(crate) fn completed_eager(
118        &mut self,
119        check: &StaticStatistics,
120    ) -> Result<(), crate::SolveError> {
121        if self.route != ClosureRoute::Eager {
122            unreachable!("an eager check is summed into an eager receipt")
123        }
124        let next = Self {
125            completed_checks: add(self.completed_checks, 1)?,
126            rounds: add(self.rounds, check.passes)?,
127            work: add(self.work, check.work)?,
128            derived_atoms: add(self.derived_atoms, count(check.derived_atoms)?)?,
129            ..self.clone()
130        };
131        *self = next;
132        Ok(())
133    }
134}
135
136fn add(left: u64, right: u64) -> Result<u64, crate::SolveError> {
137    left.checked_add(right)
138        .ok_or(crate::SolveError::ClosureStatisticsOverflow)
139}
140
141fn count(atoms: usize) -> Result<u64, crate::SolveError> {
142    u64::try_from(atoms).map_err(|_| crate::SolveError::ClosureStatisticsOverflow)
143}
