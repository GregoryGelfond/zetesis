11pub(crate) struct Input<'a> {
12    pub(crate) theory: &'a Theory,
13    pub(crate) atoms: &'a AtomCatalog,
14    pub(crate) gate_atoms: usize,
15    /// Written constraints asked as the one atom their key admits.

22    /// Source analysis chooses attempt order; each plan checks the whole theory.

26pub(crate) fn certificate_order(
27    analysis: &zetesis_themelios::analysis::Analysis,
28    basis: zetesis_themelios::AnalysisBasis,
29) -> zetesis_sat::CertificateOrder {
30    use zetesis_themelios::analysis::classify::{HornKind, Normality, Stratification};
31    let classes = analysis.classes();
32    if basis == zetesis_themelios::AnalysisBasis::NormalizedProgram
33        && matches!(classes.horn(), HornKind::Horn)
34    {
35        zetesis_sat::CertificateOrder::PositiveFirst
36    } else if basis == zetesis_themelios::AnalysisBasis::NormalizedProgram
37        && matches!(classes.normality(), Normality::Normal)
38        && matches!(classes.stratification(), Stratification::Stratified)
39    {
40        zetesis_sat::CertificateOrder::StratifiedFirst
41    } else {
42        zetesis_sat::CertificateOrder::TightFirst
43    }
44}

46/// An optional setup interruption is retained by the same search report. A
47/// diagnostic failure propagates separately and never erases the owned stream.
48pub(crate) fn prepare_certificate(
49    models: &mut zetesis_sat::StableModels,
50    order: zetesis_sat::CertificateOrder,
51    options: &SolveConfig,
52    diagnostics: &mut impl ExecutionSink,
53    phases: &Recorder,
54) -> Result<Option<zetesis_sat::Incomplete>, SolveError> {
55    if options.oracle != crate::Oracle::Auto {
56        return Ok(None);
57    }
58    let limits = certificate_limits(options);
59    let eligibility = phases.measure(SolvePhase::CertificateSetup, || {
60        if options.backend.is_gpu() {
61            // Device execution currently implements tight support. Preparation
62            // authenticates the complete theory and charges the same owner, but
63            // does not install CPU membership checks behind a device request.
64            return models
65                .prepare_tight_certificate(limits.tight)
66                .map(|plan| plan.is_some());
67        }
68        models.enable_class_checking(limits, order)
69    });
70    match eligibility {
71        Ok(true) => match models.statistics().certified.and_then(|stats| stats.plan) {
72            Some(zetesis_sat::CertificatePlanStatistics::Tight(_)) => {
73                diagnostics.record(Event::TightMembership)?;
74            }
75            Some(zetesis_sat::CertificatePlanStatistics::Positive(_)) => {
76                diagnostics.record(Event::PositiveMembership)?;
77            }
78            Some(zetesis_sat::CertificatePlanStatistics::Stratified(_)) => {
79                diagnostics.record(Event::StratifiedMembership)?;
80            }
81            None => return Ok(Some(zetesis_sat::Incomplete::InvalidWitness)),
82        },
83        Ok(false) => diagnostics.record(Event::GeneralMembership(
84            models
85                .statistics()
86                .certified
87                .and_then(|s| s.refusal)
88                .expect("refused certificate records its reason"),
89        ))?,
90        Err(error) => return Ok(Some(error)),
91    }
92    Ok(None)
93}

95/// Certificate populations use checked representation ceilings. Each compiler
96/// counts the actual program and checks its storage before allocating; these
97/// ceilings are not reservation sizes. Preparation consumes the enumeration's
98/// remaining search work, so ordinary execution adds no independent work cap.
99fn certificate_limits(options: &SolveConfig) -> zetesis_sat::CertificateLimits {
100    zetesis_sat::CertificateLimits {
101        tight: zetesis_ferraris::TightPlanLimits {
102            max_producers: usize::MAX,
103            max_dependencies: usize::MAX,
104            max_bytes: options.max_completion_scratch_bytes,
105            max_work: options.max_search_work,
106        },
107        positive: zetesis_ferraris::PositivePlanLimits {
108            max_dependencies: usize::MAX,
109            max_bytes: usize::try_from(options.max_completion_scratch_bytes).unwrap_or(usize::MAX),
110            max_work: options.max_search_work,
111        },
112        stratified: zetesis_ferraris::StratifiedPlanLimits {
113            max_dependencies: usize::MAX,
114            max_bytes: usize::try_from(options.max_completion_scratch_bytes).unwrap_or(usize::MAX),
115            max_work: options.max_search_work,
116        },
117    }
118}

120pub(crate) fn search_limits(options: &SolveConfig) -> zetesis_sat::Limits {
121    zetesis_sat::Limits {
122        admission: cnf_admission(options.max_candidate_bytes as u128),
123        reduct_admission: cnf_admission(u128::from(options.max_reduct_bytes)),
124        search: zetesis_sat::SearchLimits {
125            max_work: options.max_search_work,
126            max_decisions: options.max_search_decisions,
127        },
128        projections: zetesis_sat::ProjectionLimits {
129            max_entries: options.max_projection_entries,
130            max_nodes: options.max_projection_nodes,
131            max_bytes: options.max_projection_bytes,
132        },
133        max_candidates: options.max_candidates,
134        max_reduct_bytes: options.max_reduct_bytes,
135        max_verification_work: options.max_work,
136    }
137}

139// These independent logical populations derive from their existing named owner
140// allowance. They do not estimate total RSS. The extent cap keeps doubling a
141// literal population and the variable/sign index within checked host indices.
142fn cnf_admission(bytes: u128) -> zetesis_sat::AdmissionLimits {
143    let extent = usize::try_from(bytes.min(isize::MAX as u128)).unwrap_or(isize::MAX as usize);
144    zetesis_sat::AdmissionLimits {
145        max_variables: extent / 32,
146        max_clauses: extent / (4 * size_of::<usize>()),
147        max_literals: extent / size_of::<zetesis_sat::Literal>(),
148    }
149}
