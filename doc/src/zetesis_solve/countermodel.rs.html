11pub(crate) struct Input<'a> {
12    pub(crate) theory: &'a Theory,
13    pub(crate) atoms: &'a AtomCatalog,
14    pub(crate) gate_atoms: usize,
15    /// Written constraints asked as the one atom their key admits.
16    pub(crate) keyed_constraints: usize,
17    /// How the key analysis that asked them ended.
18    pub(crate) key_analysis: zetesis_themelios::KeyAnalysis,
19    pub(crate) objectives: &'a zetesis_objective::ObjectiveProgram,
20    /// Source analysis chooses attempt order; each plan checks the whole theory.
21    pub(crate) certificate_order: zetesis_sat::CertificateOrder,
22}

24pub(crate) fn certificate_order(
25    analysis: &zetesis_themelios::analysis::Analysis,
26    basis: zetesis_themelios::AnalysisBasis,
27) -> zetesis_sat::CertificateOrder {
28    if basis == zetesis_themelios::AnalysisBasis::NormalizedProgram
29        && matches!(
30            analysis.classes().horn(),
31            zetesis_themelios::analysis::classify::HornKind::Horn
32        )
33    {
34        zetesis_sat::CertificateOrder::PositiveFirst
35    } else {
36        zetesis_sat::CertificateOrder::TightFirst
37    }
38}

40/// An optional setup interruption is retained by the same search report. A
41/// diagnostic failure propagates separately and never erases the owned stream.
42pub(crate) fn prepare_certificate(
43    models: &mut zetesis_sat::StableModels,
44    order: zetesis_sat::CertificateOrder,
45    options: &SolveConfig,
46    diagnostics: &mut impl ExecutionSink,
47    phases: &Recorder,
48) -> Result<Option<zetesis_sat::Incomplete>, SolveError> {
49    if options.oracle != crate::Oracle::Auto {
50        return Ok(None);
51    }
52    let eligibility = phases.measure(SolvePhase::CertificateSetup, || {
53        if options.backend.is_gpu() {
54            // Device execution currently implements tight support. Preparation
55            // authenticates the complete theory and charges the same owner, but
56            // does not install CPU membership checks behind a device request.
57            return models
58                .prepare_tight_certificate(zetesis_ferraris::TightPlanLimits {
59                    max_bytes: options.max_completion_scratch_bytes,
60                    ..Default::default()
61                })
62                .map(|plan| plan.is_some());
63        }
64        models.enable_class_checking(
65            zetesis_sat::CertificateLimits {
66                tight: zetesis_ferraris::TightPlanLimits {
67                    max_bytes: options.max_completion_scratch_bytes,
68                    ..Default::default()
69                },
70                positive: zetesis_ferraris::PositivePlanLimits {
71                    max_bytes: usize::try_from(options.max_completion_scratch_bytes)
72                        .unwrap_or(usize::MAX),
73                    ..Default::default()
74                },
75            },
76            order,
77        )
78    });
79    match eligibility {
80        Ok(true) => match models.statistics().certified.and_then(|stats| stats.plan) {
81            Some(zetesis_sat::CertificatePlanStatistics::Tight(_)) => {
82                diagnostics.record(Event::TightMembership)?;
83            }
84            Some(zetesis_sat::CertificatePlanStatistics::Positive(_)) => {
85                diagnostics.record(Event::PositiveMembership)?;
86            }
87            None => return Ok(Some(zetesis_sat::Incomplete::InvalidWitness)),
88        },
89        Ok(false) => diagnostics.record(Event::GeneralMembership(
90            models
91                .statistics()
92                .certified
93                .and_then(|s| s.refusal)
94                .expect("refused certificate records its reason"),
95        ))?,
96        Err(error) => return Ok(Some(error)),
97    }
98    Ok(None)
99}

101pub(crate) fn search_limits(options: &SolveConfig) -> zetesis_sat::Limits {
102    zetesis_sat::Limits {
103        search: zetesis_sat::SearchLimits {
104            max_work: options.max_search_work,
105            max_decisions: options.max_search_decisions,
106        },
107        projections: zetesis_sat::ProjectionLimits {
108            max_entries: options.max_projection_entries,
109            max_nodes: options.max_projection_nodes,
110            max_bytes: options.max_projection_bytes,
111        },
112        max_candidates: options.max_candidates,
113        max_reduct_bytes: options.max_reduct_bytes,
114        max_verification_work: options.max_work,
115        ..Default::default()
116    }
117}
