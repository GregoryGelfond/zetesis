7/// The integrity constraints a hybrid base streams.

9pub struct StreamedConstraints {
10    /// Lowered constraint templates, not a count of original declarations.
11    pub templates: usize,
12    /// Possible instances visited during complete source admission.
13    pub instances: u64,

16/// One observation at an execution boundary of the current solve.
17///
18/// Values borrow the active call; the solver retains no observation queue. These
19/// facts describe attempted preparation or execution choices, not completed
20/// membership, publication or coverage. An observer may copy what it needs under
21/// its own storage policy. Its work and storage are outside solver resource limits.
22#[derive(Debug)]
23pub enum ExecutionObservation<'a> {
24    /// The original source is represented by a checked base theory and terminal
25    /// definitions. Formula populations and membership events that follow refer
26    /// to that base; host reconstruction precedes every original answer.
27    TerminalDefinitions {
28        /// Requested schedule: automatic for an eager base, lazy for a hybrid one.
29        requested: Grounder,
30        /// How the base was grounded.
31        base: zetesis_themelios::BaseKind,
32        /// Admitted deferred rule occurrences, not a count of ground answers.
33        deferred_templates: usize,
34        /// A hybrid base's streamed constraints; absent for an eager base.
35        streamed: Option<StreamedConstraints>,
36    },
37    /// The complete producer core is retained; eligible source constraints are
38    /// checked on the host before any original-program answer is accepted.
39    HybridGrounding {
40        /// Requested schedule; effective execution combines eager and lazy work.
41        requested: Grounder,
42        /// Lowered constraint templates, not a count of original declarations.
43        streamed_templates: usize,
44        /// Possible instances visited during complete source admission.
45        streamed_instances: u64,
46    },
47    /// An admitted static relational representation is available.
48    StaticGrounding {
49        /// Requested materialization policy.
50        requested: Grounder,
51        /// Admitted atom count.
52        atoms: usize,
53        /// Admitted ground rule count.
54        rules: usize,
55        /// Configured lowering limits; a device can impose a smaller atom ceiling.
56        limits: zetesis_core::StaticLimits,
57    },
58    /// Host source joins will supply CPU reduct closure.
59    LazyGrounding {
60        /// Requested materialization policy.
61        requested: Grounder,
62    },
63    /// CPU closure executor was prepared.
64    CpuClosure {
65        /// Effective materialization; always Eager or Lazy.
66        grounder: Grounder,
67        /// Source round policy.
68        batching: SourceBatching,
69        /// Requested worker population.
70        workers: NonZeroUsize,
71    },
72    /// CPU formula membership execution was selected.
73    CpuFormula {
74        /// Requested membership policy.
75        oracle: Oracle,
76        /// Requested materialization policy.
77        grounder: Grounder,
78        /// How candidates are proposed and the reduct queried.
79        search: SearchMethod,
80    },
81    /// Several workers walk the region tree at once and decide its leaves.
82    ParallelRegions {
83        /// Requested worker population.
84        workers: NonZeroUsize,
85    },
86    /// Bounded workers propose formula leaves, then join before an independent
87    /// membership executor checks the batch. This is not CPU membership.
88    ParallelProposals {
89        /// Requested candidate-production population.
90        workers: NonZeroUsize,
91    },
92    /// Bounded parallel residual completion is requested.
93    ExactCompletion {
94        /// Requested worker population.
95        workers: NonZeroUsize,
96        /// Logical scratch ceiling, not process RSS.
97        max_scratch_bytes: u64,
98    },
99    /// Host source joins will supply device consequence rounds.
100    #[cfg(feature = "gpu")]
101    LazyDeviceGrounding {
102        /// Requested materialization policy.
103        requested: Grounder,
104    },
105    /// Device closure execution was prepared.
106    #[cfg(feature = "gpu")]
107    DeviceClosure {
108        /// Observed adapter, borrowed from this executor.
109        adapter: zetesis_wgpu::AdapterMetadata<'a>,
110        /// Static atom/rule counts, or absence for lazy source rounds.
111        static_counts: Option<(usize, usize)>,
112    },
113    /// A device propagator is ready; exact CPU residual completion is requested.
114    #[cfg(feature = "gpu")]
115    DeviceFormula {
116        /// Observed adapter, borrowed from this executor.
117        adapter: zetesis_wgpu::AdapterMetadata<'a>,
118        /// Exact gate evaluator compiled into this executor. This records the
119        /// selected implementation, not a device submission or completed check.
120        projection: zetesis_wgpu::GateProjection,
121        /// Requested materialization policy.
122        grounder: Grounder,
123        /// How candidates are proposed and the reduct queried.
124        search: SearchMethod,
125        /// Candidate batch ceiling.
126        batch_size: NonZeroUsize,
127        /// Requested CPU completion population.
128        completion_workers: NonZeroUsize,
129    },
130    /// A complete tight certificate selects original truth and producer support
131    /// checking on this device; no general propagator is constructed.
132    #[cfg(feature = "gpu")]
133    DeviceTight {
134        /// Observed adapter of the actual tight-support executor.
135        adapter: zetesis_wgpu::AdapterMetadata<'a>,
136        /// Requested materialization policy.
137        grounder: Grounder,
138        /// Candidate proposal method; membership uses the tight certificate.
139        search: SearchMethod,
140        /// Candidate batch ceiling.
141        batch_size: NonZeroUsize,
142    },
143    /// Formula search is about to be prepared over this admitted population.
144    Formula {
145        /// Dense atom count.
146        atoms: usize,
147        /// DAG node count.
148        nodes: usize,
149        /// Logical child occurrences, including repeated children. Inline pairs
150        /// and implications each contribute two; this is not arena length.
151        operands: usize,
152        /// Asserted roots of this membership subproblem. Under a preceding
153        /// `TerminalDefinitions` event these are base roots, not the full source.
154        roots: usize,
155        /// Written constraints over a keyed value asked as the one atom their
156        /// key admits before grounding; zero when no constraint had the form.
157        keyed_constraints: usize,
158    },
159    /// The key analysis behind the asked constraints stopped at its work
160    /// ceiling; every constraint not yet asked was grounded as written.
161    KeyAnalysisStopped(zetesis_themelios::DomainStop),
162    /// A checked tight certificate enables specialized membership checking.
163    TightMembership,
164    /// Complete positive atomic-head classification and least consequences enable
165    /// the unique-answer restriction; original constraints remain authoritative.
166    PositiveMembership,
167    /// General reduct checking remains after an optional certificate refusal.
168    GeneralMembership(zetesis_sat::CertificateError),
169    /// Optional objective-plan preparation was refused; exact search remains.
170    ObjectiveUnavailable(ObjectiveBoundError),
171    /// Optional objective-bound construction was refused; exact search remains.
172    ObjectiveBoundStopped(ObjectiveBoundError),
173    /// A bound failed the original-theory ownership check.
174    ObjectiveTheoryMismatch,
175    /// Candidate restriction was refused; prior restrictions remain sound.
176    ObjectiveRestrictionStopped(zetesis_sat::Incomplete),
177    /// A dominance bound was installed after retaining a verified incumbent.
178    ObjectiveBound {
179        /// Cumulative installed restriction count.
180        restrictions: u64,
181        /// Incumbent priority/cost pairs.
182        costs: &'a [(i32, i64)],
183        /// Cumulative charged construction work.
184        work: u64,
185    },
186}

188/// Synchronous observations of one session's execution boundaries.
189///
190/// Each call is scoped to the session being initialized or pulled. The solver
191/// borrows this observer only for that operation, does not retain events and
192/// never treats an observer's error as a device failure or semantic evidence.
193/// Failure stops the operation before its next execution step; a failed pull
194/// leaves the session terminal and preserves its already checked prefix. The
195/// callback controls its own effects, cancellation and resource use. Panics are
196/// not caught. Returning an error cannot undo effects the callback already made.
197pub trait ExecutionObserver {
198    /// External observation failure, retained as the source of the solve fault.
199    type Error: Error + Send + Sync + 'static;
200
201    /// Consume an observation before the next execution step.
202    ///
203    /// # Errors
204    /// Return an external failure to stop the current operation.
205    fn observe(&mut self, observation: ExecutionObservation<'_>) -> Result<(), Self::Error>;
206}
207
208pub(crate) trait ExecutionSink {
209    fn record(&mut self, observation: ExecutionObservation<'_>) -> Result<(), SolveError>;
210}
211
212pub(crate) struct Ignore;
213impl ExecutionSink for Ignore {
214    fn record(&mut self, _: ExecutionObservation<'_>) -> Result<(), SolveError> {
215        Ok(())
216    }
217}
218
219pub(crate) struct Observer<'a, O>(pub(crate) &'a mut O);
220impl<O: ExecutionObserver> ExecutionSink for Observer<'_, O> {
221    fn record(&mut self, observation: ExecutionObservation<'_>) -> Result<(), SolveError> {
222        self.0
223            .observe(observation)
224            .map_err(|error| SolveError::ExecutionObservation(Box::new(error)))
225    }
