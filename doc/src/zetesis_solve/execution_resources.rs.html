1//! Explicit device ownership reused across independent ordinary solves.
2
3/// Caller-owned execution infrastructure, independent of any semantic subject.
4///
5/// The default retains no device and preserves ordinary per-session discovery.
6/// With the `gpu` feature, `with_gpu` retains the supplied context and compiles
7/// primitive pipelines per session. `with_formula_profile` also retains one
8/// explicit formula compilation for repeated formula sessions. Every session
9/// still owns fresh resident subjects, epochs, search state, budgets, pending
10/// answers, counters and incumbents. CPU worker pools remain session-owned.
11///
12/// Resources do not select a solving policy: the CPU backend ignores the
13/// supplied device. A GPU backend uses this exact context or fails; it never
14/// discovers a replacement. Resource ownership does not override the session's
15/// selected backend.
16///
17/// Context operations are serialized through nonblocking leases. Busy refuses
18/// an overlapping operation without poisoning the owner; device invalidation is
19/// shared by every client. Per-primitive limits do not bound all simultaneously
20/// live sessions or the context's driver storage. Cloning only retains ownership;
21/// it does not copy device contents or provide concurrent GPU scheduling.
22#[derive(Clone, Default)]
23pub struct ExecutionResources {
24    #[cfg(feature = "gpu")]
25    gpu: Option<GpuResources>,
26}
27
28// The profile determines its context: callers cannot pair a compilation with
29// another context, even one reporting identical adapter metadata.
30#[cfg(feature = "gpu")]
31#[derive(Clone)]
32enum GpuResources {
33    Context(zetesis_wgpu::GpuContext),
34    Formula(zetesis_wgpu::GpuFormulaProfile),
35}
36
37impl ExecutionResources {
38    /// Retain an existing device for later ordinary-session GPU attempts.
39    ///
40    /// This performs one shared-owner clone, with no discovery, health check,
41    /// pipeline construction or submission. The supplied handle need not outlive
42    /// this owner. Actual setup checks the requested hardware policy and primitive
43    /// capability; it can refuse a foreign, busy or invalidated context.
44    #[cfg(feature = "gpu")]
45    #[must_use]
46    pub fn with_gpu(context: &zetesis_wgpu::GpuContext) -> Self {
47        Self {
48            gpu: Some(GpuResources::Context(context.clone())),
49        }
50    }
51
52    /// Retain a compiled formula pipeline for later ordinary formula sessions.
53    ///
54    /// The profile's exact context also serves other GPU primitives. This makes
55    /// one shared-owner clone with no discovery, health check, compilation or
56    /// submission. General formula sessions reuse this pipeline; a tight-support
57    /// session compiles its own primitive on the profile's exact context.
58    /// Each formula session validates policy, Busy, health and
59    /// granted capabilities before starting independent residency and search.
60    /// CPU sessions ignore these resources.
61    #[cfg(feature = "gpu")]
62    #[must_use]
63    pub fn with_formula_profile(profile: &zetesis_wgpu::GpuFormulaProfile) -> Self {
64        Self {
65            gpu: Some(GpuResources::Formula(profile.clone())),
66        }
67    }
68
69    /// The exact retained formula compilation, if explicitly supplied.
70    ///
71    /// Context-only resources do not implicitly compile or cache a profile.
72    #[cfg(feature = "gpu")]
73    #[must_use]
74    pub const fn formula_profile(&self) -> Option<&zetesis_wgpu::GpuFormulaProfile> {
75        match self.gpu.as_ref() {
76            Some(GpuResources::Formula(profile)) => Some(profile),
77            _ => None,
78        }
79    }
80
81    /// The exact retained context, if the caller supplied one.
82    ///
83    /// Metadata and identity inspection do not establish device health. The
84    /// returned borrow cannot outlive this owner; clone the context to retain it.
85    #[cfg(feature = "gpu")]
86    #[must_use]
87    pub fn gpu_context(&self) -> Option<&zetesis_wgpu::GpuContext> {
88        match self.gpu.as_ref() {
89            Some(GpuResources::Context(context)) => Some(context),
90            Some(GpuResources::Formula(profile)) => Some(profile.context()),
91            None => None,
92        }
93    }
94
95    #[cfg(feature = "gpu")]
96    pub(crate) fn gpu_for(
97        &self,
98        backend: crate::Backend,
99    ) -> Result<Option<&zetesis_wgpu::GpuContext>, zetesis_wgpu::GpuError> {
100        if let Some(context) = self.gpu_context() {
101            context.check_selection(
102                zetesis_wgpu::GpuOptions::default(),
103                crate::engine::selection(backend),
104            )?;
105        }
106        Ok(self.gpu_context())
107    }
108}
