17/// Effective per-candidate device propagation limits; CPU search has separate units.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub struct FormulaDeviceLimits {
20    /// Charged original-truth, initialization and propagation work.
21    pub work_per_candidate: u32,
22    /// Maximum propagation sweeps after original-truth setup.
23    pub rounds_per_candidate: u32,
24}
25
26impl From<&SolveConfig> for FormulaDeviceLimits {
27    fn from(options: &SolveConfig) -> Self {
28        Self {
29            work_per_candidate: options.gpu_formula_work,
30            rounds_per_candidate: options.gpu_formula_rounds,
31        }
32    }
33}

35/// Decoded general-device propagation outcomes that still require exact
36/// membership. These precede CPU completion and commit: a later failure does
37/// not erase them, and an unreturned device batch contributes no decoded reason.
38#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
39pub struct FormulaResidualStatistics {
40    /// A complete no-change sweep left the proper-subset query unresolved.
41    pub fixed_point: u64,
42    /// The configured full-sweep ceiling was reached.
43    pub round_limit: u64,
44    /// The next full sweep could not fit its device work allowance.
45    pub work_limit: u64,
46}

48/// Actual formula GPU work and explicitly retained candidate results.
49#[derive(Clone, Debug, Default)]
50pub struct FormulaExecutionStatistics {
51    /// Physical adapter name and native API selected during this invocation.
52    pub adapter: String,
53    /// Bounded batch completion accounting; scalar cursor execution has no record.
54    pub completion: CompletionAccounting,
55    /// Effective general propagation limits, absent for CPU and tight support.
56    pub gpu_limits: Option<FormulaDeviceLimits>,
57    /// Decoded residual reasons for the general GPU primitive. Absent for CPU
58    /// and tight-support execution; distinct from committed `cpu_residuals`.
59    pub gpu_residuals: Option<FormulaResidualStatistics>,
60    /// Complete tight-support scan work ceiling. Present only for that device
61    /// primitive; propagation sweep limits do not apply to a support scan.
62    pub tight_work_per_candidate: Option<u64>,
63    /// Scheduled full-scan work, including unreturned submitted tight batches.
64    /// Propagation has candidate-dependent work and leaves this absent.
65    pub gpu_scheduled_work: Option<u64>,
66    /// Batches actually submitted, including submissions without a decoded result.
67    pub gpu_submitted_batches: u64,
68    /// Candidate worlds actually submitted, including unreturned results.
69    pub gpu_submitted_candidates: u64,
70    /// Successfully dispatched and decoded GPU batches.
71    pub gpu_batches: u64,
72    /// Candidate worlds returned by successful GPU dispatches.
73    pub gpu_candidates: u64,
74    /// Charged primitive work from successfully decoded results; not time or
75    /// GPU instruction count. Work in an interrupted unreturned batch is unknown.
76    pub gpu_work: u64,
77    /// Sum of completed per-candidate propagation sweeps in decoded results.
78    /// Sweeps in an interrupted unreturned batch are unknown.
79    pub gpu_rounds: u64,
80    /// Committed candidates completed by exact native CPU residual search.
81    pub cpu_residuals: u64,
82    /// Successfully committed candidates decided by the selected GPU primitive.
83    pub gpu_decided: u64,
84    /// Proposed candidates whose membership has not been committed.
85    pub pending_candidates: usize,
86    /// Verified stable interpretations still queued before scoring/output.
87    pub queued_models: usize,
88    /// Conservative peak authored allocation accounting reported by the device API.
89    pub peak_accounted_bytes: u64,
90}

92pub(crate) enum Failure {
93    Search(Incomplete),
94    Run(SolveError),

97impl From<crate::ExecutorError> for Failure {
98    fn from(error: crate::ExecutorError) -> Self {
99        Self::Run(match error {
100            crate::ExecutorError::Shape { expected, actual } => {
101                SolveError::FormulaBatchShape { expected, actual }
102            }
103            crate::ExecutorError::ForeignBatch => {
104                SolveError::Executor(crate::ExecutorError::ForeignBatch)
105            }
106        })
107    }

110/// A stream of original, reduct-checked semantic models, before display or scoring.
111pub(crate) trait MembershipExecution {
112    /// Return `None` only after `models.exhausted()` establishes complete
113    /// candidate coverage. A cooperative stop is an explicit `Failure::Search`;
114    /// an execution failure must not masquerade as end of enumeration.
115    fn next(
116        &mut self,
117        models: &mut StableModels,
118        options: &SolveConfig,
119        cancellation: &zetesis_cpu::Cancellation,
120        phases: &Recorder,
121    ) -> Option<Result<Interpretation, Failure>>;
122    fn statistics(&self, models: &StableModels) -> Option<FormulaExecutionStatistics>;
123}

125pub(crate) enum Execution {
126    Cpu,
127    Batched {
128        queue: Box<crate::formula_queue::BatchQueue>,
129    },
130    #[cfg(feature = "gpu")]
131    Hybrid {
132        oracle: Box<zetesis_wgpu::GpuFormulaOracle>,
133        queue: Box<crate::formula_queue::BatchQueue>,
134        statistics: Box<FormulaExecutionStatistics>,
135    },
136    #[cfg(feature = "gpu")]
137    Tight {
138        oracle: Box<zetesis_wgpu::GpuTightOracle>,
139        plan: Arc<zetesis_ferraris::TightPlan>,
140        queue: Box<crate::formula_queue::BatchQueue>,
141        statistics: Box<FormulaExecutionStatistics>,
142    },
143}

145impl Execution {
146    pub(crate) fn with_resources(
147        options: &SolveConfig,
148        resources: &ExecutionResources,
149        tight_plan: Option<Arc<zetesis_ferraris::TightPlan>>,
150        observations: &mut impl ExecutionSink,
151    ) -> Result<Self, SolveError> {
152        if options.backend == Backend::Cpu {
153            observations.record(Event::CpuFormula {
154                oracle: options.oracle,
155                grounder: options.grounder,
156                search: options.search,
157            })?;
158            if let Some(workers) = options.region_workers() {
159                // The workers decide their leaves themselves; the batched
160                // protocol would check them again.
161                observations.record(Event::ParallelRegions { workers })?;
162                return Ok(Self::Cpu);
163            }
164            if options.completion_workers.get() > 1 {
165                observations.record(Event::ExactCompletion {
166                    workers: options.completion_workers,
167                    max_scratch_bytes: options.max_completion_scratch_bytes,
168                })?;
169                return Ok(Self::Batched {
170                    queue: Box::new(crate::formula_queue::BatchQueue::new(options)?),
171                });
172            }
173            return Ok(Self::Cpu);
174        }
175        Self::gpu(options, resources, tight_plan, observations)
176    }
177
178    #[cfg(not(feature = "gpu"))]
179    fn gpu(
180        _: &SolveConfig,
181        _: &ExecutionResources,
182        _: Option<Arc<zetesis_ferraris::TightPlan>>,
183        _: &mut impl ExecutionSink,
184    ) -> Result<Self, SolveError> {
185        Err(SolveError::BackendUnavailable)
186    }
187
188    #[cfg(feature = "gpu")]
189    fn gpu(
190        options: &SolveConfig,
191        resources: &ExecutionResources,
192        tight_plan: Option<Arc<zetesis_ferraris::TightPlan>>,
193        observations: &mut impl ExecutionSink,
194    ) -> Result<Self, SolveError> {
195        let context = resources
196            .gpu_for(options.backend)
197            .map_err(SolveError::Gpu)?;
198        if let Some(plan) = tight_plan {
199            return Self::tight(options, context, plan, observations);
200        }
201        let oracle = match (resources.formula_profile(), context) {
202            (Some(profile), _) => zetesis_wgpu::GpuFormulaOracle::from_profile(profile),
203            (None, Some(context)) => zetesis_wgpu::GpuFormulaOracle::from_context(context),
204            (None, None) => zetesis_wgpu::GpuFormulaOracle::new_selected(
205                zetesis_wgpu::GpuOptions::default(),
206                crate::engine::selection(options.backend),
207            ),
208        }
209        .map_err(SolveError::Gpu)?;
210        let adapter = format!(
211            "{}, {}; vendor=0x{:04x}",
212            oracle.info().name(),
213            oracle.info().backend(),
214            oracle.info().vendor_id()
215        );
216        observations.record(Event::DeviceFormula {
217            adapter: oracle.info().metadata(),
218            projection: oracle.projection(),
219            grounder: options.grounder,
220            search: options.search,
221            batch_size: options.batch_size,
222            completion_workers: options.completion_workers,
223        })?;
224        Ok(Self::Hybrid {
225            oracle: Box::new(oracle),
226            queue: Box::new(crate::formula_queue::BatchQueue::new(options)?),
227            statistics: Box::new(FormulaExecutionStatistics {
228                adapter,
229                gpu_limits: Some(options.into()),
230                gpu_residuals: Some(FormulaResidualStatistics::default()),
231                ..Default::default()
232            }),
233        })
234    }
235}

237impl MembershipExecution for Execution {
238    fn statistics(&self, models: &StableModels) -> Option<FormulaExecutionStatistics> {
239        match self {
240            Self::Cpu => None,
241            Self::Batched { queue } => Some(batch_statistics(
242                queue,
243                models,
244                FormulaExecutionStatistics::default(),
245            )),
246            #[cfg(feature = "gpu")]
247            Self::Hybrid {
248                queue, statistics, ..
249            }
250            | Self::Tight {
251                queue, statistics, ..
252            } => {
253                let batch = models.batch_statistics();
254                Some(FormulaExecutionStatistics {
255                    cpu_residuals: batch.residuals,
256                    gpu_decided: batch.propagated,
257                    pending_candidates: batch.pending,
258                    queued_models: queue.len(),
259                    completion: queue.accounting(),
260                    ..statistics.as_ref().clone()
261                })
262            }
263        }
264    }
265
266    fn next(
267        &mut self,
268        models: &mut StableModels,
269        options: &SolveConfig,
270        cancellation: &zetesis_cpu::Cancellation,
271        phases: &Recorder,
272    ) -> Option<Result<Interpretation, Failure>> {
273        match self {
274            Self::Cpu => {
275                let _ = options;
276                let _ = cancellation;
277                let _ = phases;
278                models.next().map(|result| result.map_err(Failure::Search))
279            }
280            Self::Batched { queue } => queue.next(models, options, cancellation, |batch| {
281                let candidates = batch.candidates();
282                let mut verdicts = Vec::new();
283                verdicts
284                    .try_reserve_exact(candidates.len())
285                    .map_err(|_| Failure::Search(Incomplete::Allocation))?;
286                verdicts.resize(candidates.len(), zetesis_sat::BatchVerdict::Residual);
287                batch.finish(verdicts).map_err(Failure::from)
288            }),
289            #[cfg(feature = "gpu")]
290            Self::Hybrid {
291                oracle,
292                queue,
293                statistics,
294            } => queue.next(models, options, cancellation, |batch| {
295                let verdicts = propagate(
296                    oracle,
297                    statistics,
298                    batch.theory(),
299                    batch.candidates(),
300                    options,
301                    cancellation,
302                    phases,
303                )?;
304                batch.finish(verdicts).map_err(Failure::from)
305            }),
306            #[cfg(feature = "gpu")]
307            Self::Tight {
308                oracle,
309                plan,
310                queue,
311                statistics,
312            } => queue.next(models, options, cancellation, |batch| {
313                let verdicts = super::formula_tight::check(
314                    oracle,
315                    plan,
316                    statistics,
317                    batch.candidates(),
318                    options,
319                    cancellation,
320                    phases,
321                )?;
322                batch.finish(verdicts).map_err(Failure::from)
323            }),
324        }
325    }
326}

328#[cfg(feature = "gpu")]
329fn propagate(
330    oracle: &mut zetesis_wgpu::GpuFormulaOracle,
331    statistics: &mut FormulaExecutionStatistics,
332    theory: &zetesis_ferraris::Theory,
333    candidates: &[Interpretation],
334    options: &SolveConfig,
335    cancellation: &zetesis_cpu::Cancellation,
336    phases: &Recorder,
337) -> Result<Vec<zetesis_sat::BatchVerdict>, Failure> {
338    let limits = device_limits(options);
339    // The device API returns one ordered verdict per candidate. Reserve its
340    // host conversion before dispatch so allocation failure cannot hide a
341    // successfully decoded batch from the execution counters.
342    let mut verdicts = Vec::new();
343    verdicts
344        .try_reserve_exact(candidates.len())
345        .map_err(|_| Failure::Search(Incomplete::Allocation))?;
346    let result = phases.measure(SolvePhase::GpuHostOracle, || {
347        oracle.propagate_batch_with_cancellation(theory, candidates, limits, cancellation)
348    });
349    // Submission is observable even when mapping or decoding fails. Preserve
350    // that operation's original error over a secondary accounting overflow.
351    let submitted = record_submission(statistics, oracle.last_submission_candidates());
352    let checks = result.map_err(|error| match error.interruption() {
353        Some(stop) => Failure::Search(stop.into()),
354        None => Failure::Run(SolveError::Gpu(error)),
355    })?;
356    submitted?;
357    add(&mut statistics.gpu_batches, 1)?;
358    add(
359        &mut statistics.gpu_candidates,
360        u64::try_from(checks.len()).map_err(|_| Failure::Search(Incomplete::CounterOverflow))?,
361    )?;
362    if let Some(batch) = oracle.last_batch_stats() {
363        statistics.peak_accounted_bytes =
364            statistics.peak_accounted_bytes.max(batch.accounted_bytes);
365    }
366    for check in checks {
367        add(&mut statistics.gpu_work, u64::from(check.statistics().work))?;
368        add(
369            &mut statistics.gpu_rounds,
370            u64::from(check.statistics().rounds),
371        )?;
372        verdicts.push(decoded_verdict(
373            statistics.gpu_residuals.get_or_insert_default(),
374            check.verdict(),
375        )?);
376    }
377    Ok(verdicts)
378}

380#[cfg(feature = "gpu")]
381fn decoded_verdict(
382    residuals: &mut FormulaResidualStatistics,
383    verdict: zetesis_wgpu::FormulaVerdict,
384) -> Result<zetesis_sat::BatchVerdict, Failure> {
385    use zetesis_sat::BatchVerdict;
386    use zetesis_wgpu::{FormulaVerdict, ResidualReason};
387    Ok(match verdict {
388        FormulaVerdict::NotModel => BatchVerdict::NotModel,
389        FormulaVerdict::NoProperSubset => BatchVerdict::NoProperSubset,
390        FormulaVerdict::Residual(reason) => {
391            let count = match reason {
392                ResidualReason::FixedPoint => &mut residuals.fixed_point,
393                ResidualReason::RoundLimit => &mut residuals.round_limit,
394                ResidualReason::WorkLimit => &mut residuals.work_limit,
395            };
396            add(count, 1)?;
397            BatchVerdict::Residual
398        }
399    })
400}

413#[cfg(feature = "gpu")]
414pub(super) fn record_submission(
415    statistics: &mut FormulaExecutionStatistics,
416    candidates: Option<usize>,
417) -> Result<(), Failure> {
418    let Some(candidates) = candidates else {
419        return Ok(());
420    };
421    let candidates =
422        u64::try_from(candidates).map_err(|_| Failure::Search(Incomplete::CounterOverflow))?;
423    // Commit both counters together, retaining the previous coherent receipt
424    // when their representation is exhausted.
425    let batches = statistics.gpu_submitted_batches.checked_add(1);
426    let candidates = statistics.gpu_submitted_candidates.checked_add(candidates);
427    let (Some(batches), Some(candidates)) = (batches, candidates) else {
428        return Err(Failure::Search(Incomplete::CounterOverflow));
429    };
430    statistics.gpu_submitted_batches = batches;
431    statistics.gpu_submitted_candidates = candidates;
432    Ok(())
433}

435#[cfg(feature = "gpu")]
436pub(super) fn add(counter: &mut u64, value: u64) -> Result<(), Failure> {
437    *counter = counter
438        .checked_add(value)
439        .ok_or(Failure::Search(Incomplete::CounterOverflow))?;
440    Ok(())
441}

443fn batch_statistics(
444    queue: &crate::formula_queue::BatchQueue,
445    models: &StableModels,
446    mut statistics: FormulaExecutionStatistics,
447) -> FormulaExecutionStatistics {
448    let batch = models.batch_statistics();
449    statistics.cpu_residuals = batch.residuals;
450    statistics.pending_candidates = batch.pending;
451    statistics.queued_models = queue.len();
452    statistics.completion = queue.accounting();
453    statistics
454}

456impl<E: MembershipExecution + ?Sized> MembershipExecution for &mut E {
457    fn next(
458        &mut self,
459        models: &mut StableModels,
460        config: &SolveConfig,
461        cancellation: &zetesis_cpu::Cancellation,
462        phases: &Recorder,
463    ) -> Option<Result<Interpretation, Failure>> {
464        (**self).next(models, config, cancellation, phases)

466    fn statistics(&self, models: &StableModels) -> Option<FormulaExecutionStatistics> {
467        (**self).statistics(models)
468    }
