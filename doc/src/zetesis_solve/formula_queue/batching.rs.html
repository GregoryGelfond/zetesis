1//! Candidate slates follow completed work, bound feedback and the two existing
2//! capacity owners.
3//!
4//! A slate is the candidates the stream proposes together for one membership
5//! check and commit; `batch_size` is its initial and reset size. Under the
6//! adaptive policy, a commit only records its evidence: the slate's selected
7//! count, whether its proposals filled it, and the candidate bound's
8//! generation. The next selection decides, after the committed slate's models
9//! have been scored, because the queue hands every model of a slate out before
10//! it selects again. A bound that moved since the commit was installed by that
11//! slate's answers. The session's last selection follows its last non-empty
12//! slate, consumes that slate's commit like any other, and precedes the empty
13//! proposal that establishes exhaustion.
14//!
15//! A slate proposed ahead, while its predecessor is checked, takes the size
16//! the selection after that predecessor's commit would take: the doubling, the
17//! cap and the floor read the predecessor's proposal record (its selected
18//! count, whether the stream filled it, and the device prefixes its
19//! submissions admitted), and the pending-memory capacity left beside it bounds
20//! the count. An enumeration declares no candidate bounds, so no reset can
21//! intervene. A slate handed out ahead or as a retry keeps the count recorded
22//! at its proposal.
23
24use std::num::NonZeroUsize;
25
26use zetesis_stable::StableModels;
27
28use crate::{GpuBatching, SolveConfig};
29
30/// The slate-size policy of one GPU formula queue, with the statistics owed
31/// under either policy.
32#[derive(Default)]
33pub(super) struct Batching {
34    policy: Policy,
35    /// The oldest pending slate's proposal record, while a slate is pending.
36    in_flight: Option<Proposal>,
37    /// The proposal record of the slate proposed ahead of it.
38    ahead: Option<Proposal>,
39    /// Slates committed with at least one candidate. The stream's checked
40    /// committed-candidate count bounds it, so it cannot wrap.
41    slates: u64,
42    /// The largest committed slate's candidate count.
43    largest: Option<NonZeroUsize>,
44    /// Selections that returned to the batch size because the bound moved
45    /// since the previous commit. Each consumes one recorded commit, and the
46    /// stream's checked checker-call count bounds the commits, so it cannot
47    /// wrap.
48    resets: u64,
49}
50
51#[derive(Default)]
52enum Policy {
53    /// Every slate is the batch size.
54    #[default]
55    Fixed,
56    /// The current target, never below the batch size; the smallest shortened
57    /// device prefix observed; and the last commit no selection has consumed.
58    Adaptive {
59        target: NonZeroUsize,
60        prefix_limit: NonZeroUsize,
61        last: Option<Commit>,
62    },
63}
64
65/// Evidence one complete, successful commit leaves for the next selection.
66/// What a pending slate's proposal established: the count selected for it, the
67/// candidates the stream supplied, and the shortest device prefix its
68/// submissions admitted below what they were offered.
69#[derive(Clone, Copy)]
70struct Proposal {
71    selected: NonZeroUsize,
72    proposed: usize,
73    prefix: Option<NonZeroUsize>,
74}
75
76#[derive(Clone, Copy)]
77struct Commit {
78    /// The count selected for the committed slate.
79    selected: NonZeroUsize,
80    /// Whether the slate's proposals filled that count.
81    filled: bool,
82    /// The candidate bound's generation at the commit.
83    generation: u64,
84}
85
86impl Batching {
87    pub(super) fn new(options: &SolveConfig) -> Self {
88        Self {
89            policy: match options.gpu_batching {
90                GpuBatching::Fixed => Policy::Fixed,
91                GpuBatching::Adaptive => Policy::Adaptive {
92                    target: options.batch_size,
93                    prefix_limit: NonZeroUsize::MAX,
94                    last: None,
95                },
96            },
97            ..Self::default()
98        }
99    }
100
101    /// The count the next slate requests. A retry of pending proposals keeps
102    /// the target and changes nothing. Otherwise the last commit, if any, is
103    /// consumed: a bound generation that moved since it resets the target to
104    /// the batch size; an unchanged one doubles a filled slate's count, up to
105    /// the device prefix but never below the batch size; a short slate keeps
106    /// the target. The pending-memory capacity bounds the count selected.
107    pub(super) fn select(&mut self, models: &StableModels, options: &SolveConfig) -> NonZeroUsize {
108        self.select_within(models, options, options.max_batch_bytes)
109    }
110
111    pub(super) fn select_within(
112        &mut self,
113        models: &StableModels,
114        options: &SolveConfig,
115        max_pending_bytes: u64,
116    ) -> NonZeroUsize {
117        let Policy::Adaptive {
118            target,
119            prefix_limit,
120            last,
121        } = &mut self.policy
122        else {
123            return options.batch_size;
124        };
125        // Retried proposals keep their original count. The normal batch
126        // admission still checks changed limits and retained capacity.
127        if models.batch_statistics().pending != 0 {
128            return *target;
129        }
130        let base = options.batch_size;
131        let cap = (*prefix_limit).max(base);
132        if let Some(commit) = last.take() {
133            *target = if models.bound_generation() != commit.generation {
134                self.resets += 1;
135                base
136            } else if commit.filled {
137                commit
138                    .selected
139                    .checked_mul(NonZeroUsize::new(2).unwrap())
140                    .unwrap_or(NonZeroUsize::MAX)
141                    .max(base)
142                    .min(cap)
143            } else {
144                (*target).min(cap)
145            };
146        }
147        let memory = NonZeroUsize::new(models.joint_batch_capacity(max_pending_bytes))
148            .unwrap_or(NonZeroUsize::MIN);
149        // If even one cannot fit, ordinary admission reports its typed
150        // refusal; a sizing observation cannot claim exhaustion.
151        (*target).min(memory)
152    }
153
154    /// Record the slate the stream just proposed under `selected` with
155    /// `proposed` candidates as the oldest pending slate.
156    pub(super) fn proposed(&mut self, selected: NonZeroUsize, proposed: usize) {
157        self.in_flight = Some(Proposal {
158            selected,
159            proposed,
160            prefix: None,
161        });
162    }
163
164    /// Record the slate proposed ahead of the oldest pending one.
165    pub(super) fn proposed_ahead(&mut self, selected: NonZeroUsize, proposed: usize) {
166        self.ahead = Some(Proposal {
167            selected,
168            proposed,
169            prefix: None,
170        });
171    }
172
173    /// The slate proposed ahead became the oldest pending slate.
174    pub(super) fn promoted(&mut self) {
175        self.in_flight = self.ahead.take();
176    }
177
178    /// A submission of the oldest pending slate admitted `admitted` of the
179    /// `offered` candidates left in it.
180    pub(super) fn admitted(&mut self, offered: usize, admitted: usize) {
181        if admitted < offered
182            && let (Some(proposal), Some(admitted)) =
183                (self.in_flight.as_mut(), NonZeroUsize::new(admitted))
184        {
185            proposal.prefix = Some(
186                proposal
187                    .prefix
188                    .map_or(admitted, |prior| prior.min(admitted)),
189            );
190        }
191    }
192
193    /// The count a slate proposed ahead of the oldest pending one requests: the
194    /// selection after that slate's commit would take it from its proposal
195    /// record, and the pending-memory capacity beside it bounds it. The target
196    /// becomes that selection, as after the commit.
197    pub(super) fn select_lookahead(
198        &mut self,
199        models: &StableModels,
200        options: &SolveConfig,
201        max_pending_bytes: u64,
202    ) -> NonZeroUsize {
203        let memory = NonZeroUsize::new(models.lookahead_capacity(max_pending_bytes))
204            .unwrap_or(NonZeroUsize::MIN);
205        let Policy::Adaptive {
206            target,
207            prefix_limit,
208            ..
209        } = &mut self.policy
210        else {
211            return options.batch_size.min(memory);
212        };
213        let Some(current) = self.in_flight else {
214            return (*target).min(memory);
215        };
216        let base = options.batch_size;
217        let observed = current
218            .prefix
219            .map_or(*prefix_limit, |prefix| (*prefix_limit).min(prefix));
220        let cap = observed.max(base);
221        *target = if current.proposed == current.selected.get() {
222            current
223                .selected
224                .checked_mul(NonZeroUsize::new(2).unwrap())
225                .unwrap_or(NonZeroUsize::MAX)
226                .max(base)
227                .min(cap)
228        } else {
229            current.selected.min(cap)
230        };
231        (*target).min(memory)
232    }
233
234    /// Commit the oldest pending slate's proposal record at bound
235    /// `generation`, as [`Self::committed`] records a closure-form slate.
236    pub(super) fn committed_in_flight(&mut self, generation: u64) {
237        if let Some(proposal) = self.in_flight.take() {
238            self.committed(
239                proposal.selected,
240                proposal.proposed,
241                proposal.prefix,
242                generation,
243            );
244        }
245    }
246
247    /// Record a slate of `selected` candidates, `proposed` of which the stream
248    /// supplied, committed at bound `generation`, with any shortened device
249    /// prefix the checker observed. Called only after complete membership and
250    /// commit, never after a partial device prefix or a failed completion. The
251    /// empty proposal that establishes exhaustion commits no slate.
252    pub(super) fn committed(
253        &mut self,
254        selected: NonZeroUsize,
255        proposed: usize,
256        observed_prefix: Option<NonZeroUsize>,
257        generation: u64,
258    ) {
259        if let Some(proposed) = NonZeroUsize::new(proposed) {
260            self.slates += 1;
261            self.largest = Some(
262                self.largest
263                    .map_or(proposed, |largest| largest.max(proposed)),
264            );
265        }
266        let Policy::Adaptive {
267            prefix_limit, last, ..
268        } = &mut self.policy
269        else {
270            return;
271        };
272        if let Some(observed) = observed_prefix {
273            *prefix_limit = (*prefix_limit).min(observed);
274        }
275        *last = Some(Commit {
276            selected,
277            filled: proposed == selected.get(),
278            generation,
279        });
280    }
281
282    /// Slates committed with at least one candidate.
283    pub(super) const fn slates(&self) -> u64 {
284        self.slates
285    }
286
287    /// The largest committed slate's candidate count; absent before the first.
288    pub(super) const fn largest(&self) -> Option<NonZeroUsize> {
289        self.largest
290    }
291
292    /// Selections that returned to the batch size on bound feedback.
293    pub(super) const fn resets(&self) -> u64 {
294        self.resets
295    }
296}
