13/// The reserved verdict slots are also present in the pending-row account.
14/// Only excess slot capacity and separately owned regional storage add to it.
15struct RegionalSlate {
16    roots: Vec<(usize, Region)>,
17    authorized: Words<u64>,
18    conclusions: Vec<BatchVerdict>,
19    root_payload: u128,
20    extra: u128,
21}
22
23pub(crate) struct CheckedRows {
24    pub(super) verdicts: Vec<BatchVerdict>,
25    regional: Option<RegionalSlate>,
26    pub(crate) count: usize,
27}

29fn space(ceiling: u64, terms: &[u128]) -> Result<u64, Failure> {
30    let remaining = terms
31        .iter()
32        .try_fold(u128::from(ceiling), |left, term| left.checked_sub(*term))
33        .ok_or(Failure::Search(Incomplete::PendingBytes))?;
34    Ok(u64::try_from(remaining).expect("the remaining u64 allowance fits"))
35}
36
37pub(super) fn after_pending(available: u64, pending: u128) -> Result<u64, Failure> {
38    space(available, &[pending])
39}
40
41fn reserve<T>(values: &mut Vec<T>, n: usize) -> Result<u128, Failure> {
42    values
43        .try_reserve_exact(n)
44        .map_err(|_| Failure::Search(Incomplete::Allocation))?;
45    Ok((values.capacity() - n) as u128 * size_of::<T>() as u128)
46}

48impl CheckedRows {
49    pub(super) fn new(count: usize, regional: Option<(u64, u128, u64)>) -> Result<Self, Failure> {
50        let mut verdicts = Vec::new();
51        let regional = if let Some((limit, pending, resident)) = regional {
52            let words = count.div_ceil(64);
53            let requested = (size_of::<Self>() - size_of::<Vec<BatchVerdict>>()) as u128
54                + count as u128
55                    * (size_of::<(usize, Region)>() + size_of::<BatchVerdict>()) as u128
56                + words as u128 * size_of::<u64>() as u128;
57            let room = space(limit, &[pending, u128::from(resident), requested])?;
58            let mut excess = reserve(&mut verdicts, count)?;
59            space(room, &[excess])?;
60            let mut roots = Vec::new();
61            excess += reserve(&mut roots, count)?;
62            space(room, &[excess])?;
63            let mut authorized = Vec::new();
64            excess += reserve(&mut authorized, words)?;
65            space(room, &[excess])?;
66            authorized.resize(words, 0);
67            let mut conclusions = Vec::new();
68            excess += reserve(&mut conclusions, count)?;
69            space(room, &[excess])?;
70            Some(RegionalSlate {
71                roots,
72                authorized: authorized.into(),
73                conclusions,
74                root_payload: 0,
75                extra: requested + excess,
76            })
77        } else {
78            reserve(&mut verdicts, count)?;
79            None
80        };
81        Ok(Self {
82            verdicts,
83            regional,
84            count,
85        })
86    }

88    pub(crate) fn classified(&mut self, verdict: BatchVerdict) -> Result<(), Failure> {
89        if self.verdicts.len() >= self.count {
90            return Err(Failure::Run(SolveError::FormulaBatchShape {
91                expected: self.count,
92                actual: self.verdicts.len().saturating_add(1),
93            }));
94        }
95        self.verdicts.push(verdict);
96        Ok(())
97    }
98
99    /// Called only in order after the GPU's subject-bound receipt authenticates.
100    pub(crate) fn projected(&mut self, root: Region) -> Result<(), Failure> {
101        let slot = self.verdicts.len();
102        let regional = self
103            .regional
104            .as_mut()
105            .filter(|_| slot < self.count)
106            .ok_or(Failure::Run(SolveError::FormulaBatchShape {
107                expected: self.count,
108                actual: slot.saturating_add(1),
109            }))?;
110        let payload = root.retained_bytes() - size_of::<Region>() as u128;
111        regional.root_payload = regional
112            .root_payload
113            .checked_add(payload)
114            .ok_or(Failure::Search(Incomplete::CounterOverflow))?;
115        regional.authorized.insert(slot);
116        regional.roots.push((slot, root));
117        self.verdicts.push(BatchVerdict::Residual);
118        Ok(())
119    }

121    /// Before a submission, or before its overlap, subtract only owners which
122    /// the device plan does not already count. Lookahead admits its own P.
123    pub(super) fn remaining(
124        &self,
125        ceiling: u64,
126        pending: u128,
127        device: u64,
128    ) -> Result<u64, Failure> {
129        match &self.regional {
130            Some(r) => space(
131                ceiling,
132                &[pending, r.extra, r.root_payload, u128::from(device)],
133            ),
134            None => Ok(ceiling),
135        }
136    }

138    pub(super) fn finish(
139        self,
140        models: &mut StableModels,
141        ticket: SlateTicket,
142        completion: &mut CompletionExecutor,
143    ) -> Result<Vec<Interpretation>, Failure> {
144        let Some(mut state) = self.regional else {
145            return models
146                .commit_slate(ticket, self.verdicts, completion)
147                .map_err(Failure::from);
148        };
149        let pass = models
150            .begin_regional_pass(ticket, self.verdicts, state.roots, completion)
151            .map_err(Failure::from)?;
152        for (slot, outcome) in pass.outcomes() {
153            if slot >= self.count || !state.authorized.contains(slot) {
154                return Err(Failure::Search(Incomplete::InvalidWitness));
155            }
156            let whole = match outcome {
157                RegionalOutcome::ValidatedCountermodel => BatchVerdict::Refuted,
158                RegionalOutcome::NoWitnessInSuppliedRegion => BatchVerdict::NoProperSubset,
159            };
160            state.conclusions.push(whole);
161        }
162        pass.finish(state.conclusions).map_err(Failure::from)
163    }
164}
