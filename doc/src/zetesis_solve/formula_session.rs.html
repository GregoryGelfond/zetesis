19pub(crate) struct FormulaSession<'a, E> {
20    input: Input<'a>,
21    selection: AnswerSelection,
22    execution: Option<E>,
23    models: Option<StableModels>,
24    model_order: Option<ModelOrder<'a>>,
25    construction: Option<crate::model_construction::Account>,
26    bounds: Option<Bounds>,
27    incumbents: Incumbents,
28    ready: std::vec::IntoIter<Model>,
29    yielded: usize,
30    final_outcome: Option<SemanticOutcome>,
31    pending_error: Option<SolveError>,
32    // This session's last imported cumulative prefix, independently of every
33    // other session sharing the host recorder. Repeated outcomes import no work.
34    imported_timings: Cell<zetesis_sat::SearchPhaseTimings>,
35}

37impl<'a, E: MembershipExecution> FormulaSession<'a, E> {
38    /// Install an original-only source restriction before the first proposal.
39    /// An already stopped initialization has no traversal left to configure.
40    pub(crate) fn set_region_filter(
41        &mut self,
42        filter: std::sync::Arc<dyn zetesis_sat::RegionFilter>,
43        phases: &Recorder,
44    ) {
45        if let Some(models) = self.models.as_mut()
46            && let Err(error) = models.set_region_filter(filter)
47        {
48            self.complete(
49                SearchState::Interrupted(Interruption::Countermodel(error)),
50                phases,
51            );
52        }
53    }
54
55    /// Stop candidate production and join native workers without cancelling the
56    /// caller's shared token. Refresh receipts after the workers have settled.
57    pub(crate) fn stop(&mut self, phases: &Recorder) -> Result<(), zetesis_sat::Incomplete> {
58        let result = self.models.as_mut().map_or(Ok(()), StableModels::stop);
59        if let Some(previous) = self.final_outcome.take() {
60            let mut outcome = self.snapshot(phases);
61            outcome.search_state = previous.search_state;
62            self.final_outcome = Some(outcome);
63        }
64        result
65    }

68    pub(crate) fn with_selection(
69        input: Input<'a>,
70        execution: E,
71        config: &SolveConfig,
72        observations: &mut impl ExecutionSink,
73        cancellation: &Cancellation,
74        phases: &Recorder,
75        selection: AnswerSelection,
76    ) -> Self {
77        let mut session = Self::uninitialized(input, selection);
78        session.execution = Some(execution);
79        if let Err(error) = session.initialize(config, observations, cancellation, phases) {
80            session.fail(error, phases);
81        }
82        session
83    }

85    fn uninitialized(input: Input<'a>, selection: AnswerSelection) -> Self {
86        Self {
87            input,
88            selection: if input.objectives.is_present() {
89                selection
90            } else {
91                AnswerSelection::All
92            },
93            execution: None,
94            models: None,
95            model_order: None,
96            construction: None,
97            bounds: None,

101            final_outcome: None,
102            pending_error: None,
103            imported_timings: Cell::new(zetesis_sat::SearchPhaseTimings::default()),
104        }

107    fn initialize(
108        &mut self,
109        config: &SolveConfig,
110        observations: &mut impl ExecutionSink,
111        cancellation: &Cancellation,
112        phases: &Recorder,
113    ) -> Result<(), SolveError> {
114        observations.record(Event::Formula {
115            atoms: self.input.theory.atom_count(),
116            nodes: self.input.theory.nodes().len(),
117            roots: self.input.theory.roots().len(),
118            keyed_constraints: self.input.keyed_constraints,
119        })?;
120        if let zetesis_themelios::KeyAnalysis::Stopped(stop) = self.input.key_analysis {
121            observations.record(Event::KeyAnalysisStopped(stop))?;
122        }
123        let account = self
124            .construction
125            .insert(crate::model_construction::Account::default());
126        let prepared = phases.measure(SolvePhase::ModelConstruction, || {
127            account.prepare(self.input.atoms, config, cancellation)
128        });
129        match prepared {
130            Ok(order) => self.model_order = Some(order),
131            Err(crate::model_construction::Failure::Interrupted(stop)) => {
132                self.complete(
133                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
134                    phases,
135                );
136                return Ok(());
137            }
138            Err(crate::model_construction::Failure::Run(error)) => return Err(error),
139        }
140        let models = phases.measure(SolvePhase::CandidateSetup, || {
141            // CPU workers decide their own leaves. Device producers return
142            // unchecked leaves to the bounded batch protocol and join before
143            // membership execution starts.
144            if let Some(workers) = config.region_workers() {
145                StableModels::with_region_workers(
146                    self.input.theory,
147                    workers,
148                    crate::countermodel::search_limits(config),
149                    cancellation.clone(),
150                )
151            } else if config.search == crate::SearchMethod::Regions && config.workers.get() > 1 {
152                StableModels::with_region_producers(
153                    self.input.theory,
154                    config.workers,
155                    crate::countermodel::search_limits(config),
156                    cancellation.clone(),
157                )
158            } else {
159                StableModels::with_method(
160                    self.input.theory,
161                    config.search,
162                    crate::countermodel::search_limits(config),
163                    cancellation.clone(),
164                )
165            }
166        });
167        let mut models = match models {
168            Ok(models) => models,
169            Err(error) => {
170                self.complete(
171                    SearchState::Interrupted(Interruption::Countermodel(error)),
172                    phases,
173                );
174                return Ok(());
175            }
176        };
177        if config.stats {
178            models.enable_phase_timing();
179        }
180        // Ownership is established before any fallible certificate/diagnostic
181        // operation, so its attempted work remains available on failure.
182        self.models = Some(models);
183        if config.region_workers().is_none()
184            && config.search == crate::SearchMethod::Regions
185            && config.workers.get() > 1
186        {
187            observations.record(Event::ParallelProposals {
188                workers: config.workers,
189            })?;
190        }
191        let models = self.models.as_mut().expect("candidate stream installed");
192        if let Some(error) = crate::countermodel::prepare_certificate(
193            models,
194            self.input.certificate_order,
195            config,
196            observations,
197            phases,
198        )? {
199            self.complete(
200                SearchState::Interrupted(Interruption::Countermodel(error)),
201                phases,
202            );
203            return Ok(());
204        }
205        if self.selection == AnswerSelection::Optimal {
206            self.bounds = Some(
207                if self.input.objectives.is_present() && config.max_objective_bound_work != 0 {
208                    phases.measure(SolvePhase::ObjectiveFeedback, || {
209                        Bounds::new(self.input, config, observations, cancellation)
210                    })?
211                } else {
212                    Bounds::new(self.input, config, observations, cancellation)?
213                },
214            );
215        }
216        Ok(())
217    }

219    pub(crate) fn next(
220        &mut self,
221        config: &SolveConfig,
222        observations: &mut impl ExecutionSink,
223        cancellation: &Cancellation,
224        phases: &Recorder,
225    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
226        let next = self.next_result(config, observations, cancellation, phases);
227        self.import_timings(
228            phases,
229            self.models
230                .as_ref()
231                .and_then(|models| models.statistics().phase_timings),
232        );
233        next
234    }

236    fn next_result(
237        &mut self,
238        config: &SolveConfig,
239        observations: &mut impl ExecutionSink,
240        cancellation: &Cancellation,
241        phases: &Recorder,
242    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
243        if let Some(error) = self.pending_error.take() {
244            return Some(Err(error));
245        }
246        if self.final_outcome.is_some() {
247            return self.next_retained().map(Ok);
248        }
249        if self.selection == AnswerSelection::All
250            && config.models != 0
251            && self.yielded >= config.models
252        {
253            self.complete(SearchState::RequestedModels, phases);
254            return None;
255        }
256        // The retained candidate stream owns cumulative work and exact blocks;
257        // pulls never restart it. Objective improvements narrow only candidates.
258        loop {
259            let models = self
260                .models
261                .as_mut()
262                .expect("unfinished session has a candidate stream");
263            let next = self
264                .execution
265                .as_mut()
266                .expect("unfinished session has a membership executor")
267                .next(models, config, cancellation, phases);
268            let interpretation = match next {
269                Some(Ok(model)) => model,
270                Some(Err(Failure::Search(error))) => {
271                    self.complete(
272                        SearchState::Interrupted(Interruption::Countermodel(error)),
273                        phases,
274                    );
275                    return self.next_retained().map(Ok);
276                }
277                Some(Err(Failure::Run(error))) => {
278                    self.fail(error, phases);
279                    return self.pending_error.take().map(Err);
280                }
281                None => return self.finish_candidate_stream(phases),
282            };
283            let model =
284                match self.construct_model(interpretation.atoms(), config, cancellation, phases) {
285                    Ok(model) => model,
286                    Err(failure) => return self.finish_construction_failure(failure, phases),
287                };
288            if !self.input.objectives.is_present() {
289                self.yielded += 1;
290                return Some(Ok((model, None)));
291            }
292            if self.selection == AnswerSelection::All {
293                let score = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
294                    self.incumbents
295                        .evaluate(self.input.objectives, &model, config, cancellation)
296                });
297                return match score {
298                    Ok(score) => {
299                        self.yielded += 1;
300                        Some(Ok((model, Some(score))))
301                    }
302                    Err(reason) => {
303                        self.complete(SearchState::Interrupted(reason), phases);
304                        None
305                    }
306                };
307            }
308            let scored = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
309                self.incumbents
310                    .consider(self.input.objectives, model, config, cancellation)
311            });
312            match scored {
313                Ok(true) => {
314                    let _feedback = (config.max_objective_bound_work != 0)
315                        .then(|| phases.start(SolvePhase::ObjectiveFeedback));
316                    let improved = self
317                        .bounds
318                        .as_mut()
319                        .expect("initialized objective plan")
320                        .improve(
321                            self.incumbents
322                                .score()
323                                .expect("retained improvement has a score"),
324                            self.models.as_mut().expect("owned candidate stream"),
325                            config,
326                            observations,
327                            cancellation,
328                        );
329                    if let Err(error) = improved {
330                        self.fail(error, phases);
331                        return self.pending_error.take().map(Err);
332                    }
333                }
334                Ok(false) => {}
335                Err(reason) => {
336                    self.complete(SearchState::Interrupted(reason), phases);
337                    return self.next_retained().map(Ok);
338                }
339            }
340        }
341    }

343    fn construct_model(
344        &mut self,
345        positions: impl IntoIterator<Item = usize>,
346        config: &SolveConfig,
347        cancellation: &Cancellation,
348        phases: &Recorder,
349    ) -> Result<Model, crate::model_construction::Failure> {
350        phases.measure(SolvePhase::ModelConstruction, || {
351            self.construction
352                .as_mut()
353                .expect("initialized construction account")
354                .select(
355                    self.model_order.as_ref().expect("prepared model order"),
356                    positions,
357                    config,
358                    cancellation,
359                )
360        })
361    }
362
363    fn finish_construction_failure(
364        &mut self,
365        failure: crate::model_construction::Failure,
366        phases: &Recorder,
367    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
368        // Settle workers before snapshotting their checked prefix, preserving
369        // the construction failure over any secondary cleanup failure.
370        let _ = self.models.as_mut().expect("owned candidate stream").stop();
371        match failure {
372            crate::model_construction::Failure::Interrupted(stop) => {
373                self.complete(
374                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
375                    phases,
376                );
377                self.next_retained().map(Ok)
378            }
379            crate::model_construction::Failure::Run(error) => {
380                self.fail(error, phases);
381                self.pending_error.take().map(Err)
382            }
383        }
384    }
385
386    fn finish_candidate_stream(
387        &mut self,
388        phases: &Recorder,
389    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
390        if !self
391            .models
392            .as_ref()
393            .expect("unfinished session has a candidate stream")
394            .exhausted()
395        {
396            self.fail(SolveError::CandidateStreamNotExhausted, phases);
397            return self.pending_error.take().map(Err);
398        }
399        self.complete(SearchState::Exhausted, phases);
400        self.next_retained().map(Ok)
401    }
402
403    fn next_retained(&mut self) -> Option<(Model, Option<Score>)> {
404        self.ready.next().map(|model| {
405            let score = self.incumbents.score().cloned();
406            (model, score)
407        })
408    }
409
410    fn complete(&mut self, search_state: SearchState, phases: &Recorder) {
411        let mut outcome = self.snapshot(phases);
412        outcome.search_state = Some(search_state);
413        self.final_outcome = Some(outcome);
414        self.ready = self.incumbents.take_models();
415    }
416
417    fn fail(&mut self, error: SolveError, phases: &Recorder) {
418        self.final_outcome = Some(self.snapshot(phases));
419        self.pending_error = Some(error);
420    }
421
422    pub(crate) fn outcome(&self, phases: &Recorder) -> SemanticOutcome {
423        self.final_outcome
424            .clone()
425            .unwrap_or_else(|| self.snapshot(phases))
426    }
427
428    pub(crate) const fn finished(&self) -> bool {
429        self.final_outcome.is_some()
430    }
431
432    fn snapshot(&self, phases: &Recorder) -> SemanticOutcome {
433        let statistics = self.models.as_ref().map(StableModels::statistics);
434        self.import_timings(
435            phases,
436            statistics.and_then(|statistics| statistics.phase_timings),
437        );
438        SemanticOutcome {
439            projection: None,
440            subject: Some(crate::Subject::Theory(self.input.theory.clone())),
441            selection: Some(self.selection),
442            verified: statistics.map_or(0, |s| s.stable_models),
443            scored: self.incumbents.scored(),
444            retained: self.incumbents.retained(),
445            search_state: None,
446            optimization: self.incumbents.metadata().cloned(),
447            checked: statistics.map_or(0, |s| s.candidates),
448            gate_atoms: self.input.gate_atoms,
449            candidate_statistics: None,
450            countermodel_statistics: statistics,
451            lazy_execution: None,
452            shared_execution: None,
453            closure_execution: None,
454            query_execution: None,
455            hybrid_execution: None,
456            terminal_execution: None,
457            model_construction: self
458                .construction
459                .as_ref()
460                .map(crate::model_construction::Account::statistics),
461            formula_execution: self
462                .models
463                .as_ref()
464                .and_then(|models| self.execution.as_ref()?.statistics(models)),
465        }
466    }
467
468    fn import_timings(&self, phases: &Recorder, timing: Option<zetesis_sat::SearchPhaseTimings>) {
469        if let Some(timing) = timing {
470            phases.search(self.imported_timings.replace(timing), timing);
471        }
472    }

478impl<'a> FormulaSession<'a, crate::formula_execution::Execution> {
479    /// Own semantic preparation before choosing its execution route. A stopped
480    /// preparation retains its receipts without constructing a device pipeline.
481    /// Route construction keeps the ordinary session's fallible start door.
482    pub(crate) fn with_resources(
483        input: Input<'a>,
484        config: &SolveConfig,
485        resources: &crate::ExecutionResources,
486        observations: &mut impl ExecutionSink,
487        cancellation: &Cancellation,
488        phases: &Recorder,
489        selection: AnswerSelection,
490    ) -> Result<Self, SolveError> {
491        let mut session = Self::uninitialized(input, selection);
492        if let Err(error) = session.initialize(config, observations, cancellation, phases) {
493            session.fail(error, phases);
494        }
495        if session.final_outcome.is_none() {
496            let plan = session
497                .models
498                .as_ref()
499                .and_then(StableModels::prepared_tight_certificate);
500            session.execution = Some(phases.measure(SolvePhase::ExecutionSetup, || {
501                crate::formula_execution::Execution::with_resources(
502                    config,
503                    resources,
504                    plan,
505                    observations,
506                )
507            })?);
508        }
509        Ok(session)
510    }
