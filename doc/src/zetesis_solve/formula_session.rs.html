19pub(crate) struct FormulaSession<'a, E> {
20    input: Input<'a>,
21    selection: AnswerSelection,
22    execution: Option<E>,
23    models: Option<StableModels>,
24    model_order: Option<ModelOrder<'a>>,
25    construction: Option<crate::model_construction::Account>,
26    objective_plan: Option<Preparation>,
27    bounds: Bounds,
28    incumbents: Incumbents,
29    ready: std::vec::IntoIter<Model>,
30    yielded: usize,
31    final_outcome: Option<SemanticOutcome>,
32    pending_error: Option<SolveError>,
33    // This session's last imported cumulative prefix, independently of every
34    // other session sharing the host recorder. Repeated outcomes import no work.
35    imported_timings: Cell<zetesis_sat::SearchPhaseTimings>,
36}

38impl<'a, E: MembershipExecution> FormulaSession<'a, E> {
39    /// Install an original-only source restriction before the first proposal.
40    /// An already stopped initialization has no traversal left to configure.
41    pub(crate) fn set_region_filter(
42        &mut self,
43        filter: std::sync::Arc<dyn zetesis_sat::RegionFilter>,
44        phases: &Recorder,
45    ) {
46        if let Some(models) = self.models.as_mut()
47            && let Err(error) = models.set_region_filter(filter)
48        {
49            self.complete(
50                SearchState::Interrupted(Interruption::Countermodel(error)),
51                phases,
52            );
53        }
54    }

56    /// Stop candidate production and join native workers without cancelling the
57    /// caller's shared token. Refresh receipts after the workers have settled.
58    pub(crate) fn stop(&mut self, phases: &Recorder) -> Result<(), zetesis_sat::Incomplete> {
59        let result = self.models.as_mut().map_or(Ok(()), StableModels::stop);
60        if let Some(previous) = self.final_outcome.take() {
61            let mut outcome = self.snapshot(phases);
62            outcome.search_state = previous.search_state;
63            self.final_outcome = Some(outcome);
64        }
65        result
66    }

69    pub(crate) fn with_selection(
70        input: Input<'a>,
71        execution: E,
72        config: &SolveConfig,
73        observations: &mut impl ExecutionSink,
74        cancellation: &Cancellation,
75        phases: &Recorder,
76        selection: AnswerSelection,
77    ) -> Self {
78        let mut session = Self::uninitialized(input, selection);
79        session.execution = Some(execution);
80        if let Err(error) = session.initialize(config, observations, cancellation, phases) {
81            session.fail(error, phases);
82        }
83        session
84    }

87        Self {
88            input,
89            selection: if input.objectives.is_present() {
90                selection
91            } else {
92                AnswerSelection::All
93            },
94            execution: None,
95            models: None,
96            model_order: None,
97            construction: None,
98            objective_plan: None,

103            final_outcome: None,
104            pending_error: None,
105            imported_timings: Cell::new(zetesis_sat::SearchPhaseTimings::default()),
106        }

109    fn initialize(
110        &mut self,
111        config: &SolveConfig,
112        observations: &mut impl ExecutionSink,
113        cancellation: &Cancellation,
114        phases: &Recorder,
115    ) -> Result<(), SolveError> {
116        observations.record(Event::Formula {
117            atoms: self.input.theory.atom_count(),
118            nodes: self.input.theory.nodes().len(),
119            operands: self.input.theory.parts().occurrences(),
120            roots: self.input.theory.roots().len(),
121            keyed_constraints: self.input.keyed_constraints,
122        })?;
123        if let zetesis_themelios::KeyAnalysis::Stopped(stop) = self.input.key_analysis {
124            observations.record(Event::KeyAnalysisStopped(stop))?;
125        }
126        let account = self
127            .construction
128            .insert(crate::model_construction::Account::default());
129        let prepared = phases.measure(SolvePhase::ModelConstruction, || {
130            account.prepare(self.input.atoms, config, cancellation)
131        });
132        match prepared {
133            Ok(order) => self.model_order = Some(order),
134            Err(crate::model_construction::Failure::Interrupted(stop)) => {
135                self.complete(
136                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
137                    phases,
138                );
139                return Ok(());
140            }
141            Err(crate::model_construction::Failure::Run(error)) => return Err(error),
142        }
143        let models = phases.measure(SolvePhase::CandidateSetup, || {
144            // CPU workers decide their own leaves. Device producers return
145            // unchecked leaves to the bounded batch protocol and join before
146            // membership execution starts.
147            if let Some(workers) = config.region_workers() {
148                StableModels::with_region_workers(
149                    self.input.theory,
150                    workers,
151                    crate::countermodel::search_limits(config),
152                    cancellation.clone(),
153                )
154            } else if config.search == crate::SearchMethod::Regions && config.workers.get() > 1 {
155                StableModels::with_region_producers(
156                    self.input.theory,
157                    config.workers,
158                    crate::countermodel::search_limits(config),
159                    cancellation.clone(),
160                )
161            } else {
162                StableModels::with_method(
163                    self.input.theory,
164                    config.search,
165                    crate::countermodel::search_limits(config),
166                    cancellation.clone(),
167                )
168            }
169        });
170        let mut models = match models {
171            Ok(models) => models,
172            Err(error) => {
173                self.complete(
174                    SearchState::Interrupted(Interruption::Countermodel(error)),
175                    phases,
176                );
177                return Ok(());
178            }
179        };
180        if config.stats {
181            models.enable_phase_timing();
182        }
183        // Ownership is established before any fallible certificate/diagnostic
184        // operation, so its attempted work remains available on failure.
185        self.models = Some(models);
186        if config.region_workers().is_none()
187            && config.search == crate::SearchMethod::Regions
188            && config.workers.get() > 1
189        {
190            observations.record(Event::ParallelProposals {
191                workers: config.workers,
192            })?;
193        }
194        let models = self.models.as_mut().expect("candidate stream installed");
195        if let Some(error) = crate::countermodel::prepare_certificate(
196            models,
197            self.input.certificate_order,
198            config,
199            observations,
200            phases,
201        )? {
202            self.complete(
203                SearchState::Interrupted(Interruption::Countermodel(error)),
204                phases,
205            );
206            return Ok(());
207        }
208        self.prepare_objectives(config, observations, cancellation, phases)
209    }

211    /// Retain and charge preparation before exposing its optional refusal.
212    fn prepare_objectives(
213        &mut self,
214        config: &SolveConfig,
215        observations: &mut impl ExecutionSink,
216        cancellation: &Cancellation,
217        phases: &Recorder,
218    ) -> Result<(), SolveError> {
219        if self.input.objectives.is_present() {
220            self.objective_plan =
221                Some(phases.measure(SolvePhase::ObjectiveScoringRetention, || {
222                    Preparation::new(self.input, config, cancellation)
223                }));
224            let preparation = self
225                .objective_plan
226                .as_ref()
227                .expect("prepared objective attempt");
228            if let Err(reason) = self.incumbents.prepare(preparation.work()) {
229                self.complete(SearchState::Interrupted(reason), phases);
230                return Ok(());
231            }
232            preparation.observe(observations)?;
233        }
234        self.bounds = Bounds::new(config);
235        Ok(())
236    }

238    pub(crate) fn next(
239        &mut self,
240        config: &SolveConfig,
241        observations: &mut impl ExecutionSink,
242        cancellation: &Cancellation,
243        phases: &Recorder,
244    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
245        let next = self.next_result(config, observations, cancellation, phases);
246        self.import_timings(
247            phases,
248            self.models
249                .as_ref()
250                .and_then(|models| models.statistics().phase_timings),
251        );
252        next
253    }

255    fn next_result(
256        &mut self,
257        config: &SolveConfig,
258        observations: &mut impl ExecutionSink,
259        cancellation: &Cancellation,
260        phases: &Recorder,
261    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
262        if let Some(error) = self.pending_error.take() {
263            return Some(Err(error));
264        }
265        if self.final_outcome.is_some() {
266            return self.next_retained().map(Ok);
267        }
268        if self.selection == AnswerSelection::All
269            && config.models != 0
270            && self.yielded >= config.models
271        {
272            self.complete(SearchState::RequestedModels, phases);
273            return None;
274        }
275        // The retained candidate stream owns cumulative work and exact blocks;
276        // pulls never restart it. Objective improvements narrow only candidates.
277        loop {
278            let models = self
279                .models
280                .as_mut()
281                .expect("unfinished session has a candidate stream");
282            let next = self
283                .execution
284                .as_mut()
285                .expect("unfinished session has a membership executor")
286                .next(models, config, cancellation, phases);
287            let interpretation = match next {
288                Some(Ok(model)) => model,
289                Some(Err(Failure::Search(error))) => {
290                    self.complete(
291                        SearchState::Interrupted(Interruption::Countermodel(error)),
292                        phases,
293                    );
294                    return self.next_retained().map(Ok);
295                }
296                Some(Err(Failure::Run(error))) => {
297                    self.fail(error, phases);
298                    return self.pending_error.take().map(Err);
299                }
300                None => return self.finish_candidate_stream(phases),
301            };
302            let model =
303                match self.construct_model(interpretation.atoms(), config, cancellation, phases) {
304                    Ok(model) => model,
305                    Err(failure) => return self.finish_construction_failure(failure, phases),
306                };
307            if !self.input.objectives.is_present() {
308                self.yielded += 1;
309                return Some(Ok((model, None)));
310            }
311            if self.selection == AnswerSelection::All {
312                return self.score_model(model, &interpretation, config, cancellation, phases);
313            }
314            let scored = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
315                self.incumbents.consider(
316                    self.input.objectives,
317                    model,
318                    config,
319                    cancellation,
320                    self.objective_plan
321                        .as_ref()
322                        .and_then(Preparation::plan)
323                        .map(|plan| (plan, &interpretation)),
324                )
325            });
326            match scored {
327                Ok(true) => {
328                    let _feedback = (config.max_objective_bound_work != 0)
329                        .then(|| phases.start(SolvePhase::ObjectiveFeedback));
330                    let improved = self.bounds.improve(
331                        self.objective_plan.as_ref().and_then(Preparation::plan),
332                        self.incumbents
333                            .score()
334                            .expect("retained improvement has a score"),
335                        self.models.as_mut().expect("owned candidate stream"),
336                        config,
337                        observations,
338                        cancellation,
339                    );
340                    if let Err(error) = improved {
341                        self.fail(error, phases);
342                        return self.pending_error.take().map(Err);
343                    }
344                }
345                Ok(false) => {}
346                Err(reason) => {
347                    self.complete(SearchState::Interrupted(reason), phases);
348                    return self.next_retained().map(Ok);
349                }
350            }
351        }
352    }

354    /// Score one verified model without selecting or retaining an incumbent.
355    fn score_model(
356        &mut self,
357        model: Model,
358        interpretation: &zetesis_ferraris::Interpretation,
359        config: &SolveConfig,
360        cancellation: &Cancellation,
361        phases: &Recorder,
362    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
363        let score = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
364            self.incumbents.evaluate(
365                self.input.objectives,
366                &model,
367                config,
368                cancellation,
369                self.objective_plan
370                    .as_ref()
371                    .and_then(Preparation::plan)
372                    .map(|plan| (plan, interpretation)),
373            )
374        });
375        match score {
376            Ok(score) => {
377                self.yielded += 1;
378                Some(Ok((model, Some(score))))
379            }
380            Err(reason) => {
381                self.complete(SearchState::Interrupted(reason), phases);
382                None
383            }
384        }
385    }

387    fn construct_model(
388        &mut self,
389        positions: impl IntoIterator<Item = usize>,
390        config: &SolveConfig,
391        cancellation: &Cancellation,
392        phases: &Recorder,
393    ) -> Result<Model, crate::model_construction::Failure> {
394        phases.measure(SolvePhase::ModelConstruction, || {
395            self.construction
396                .as_mut()
397                .expect("initialized construction account")
398                .select(
399                    self.model_order.as_ref().expect("prepared model order"),
400                    positions,
401                    config,
402                    cancellation,
403                )
404        })
405    }

407    fn finish_construction_failure(
408        &mut self,
409        failure: crate::model_construction::Failure,
410        phases: &Recorder,
411    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
412        // Settle workers before snapshotting their checked prefix, preserving
413        // the construction failure over any secondary cleanup failure.
414        let _ = self.models.as_mut().expect("owned candidate stream").stop();
415        match failure {
416            crate::model_construction::Failure::Interrupted(stop) => {
417                self.complete(
418                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
419                    phases,
420                );
421                self.next_retained().map(Ok)
422            }
423            crate::model_construction::Failure::Run(error) => {
424                self.fail(error, phases);
425                self.pending_error.take().map(Err)
426            }
427        }
428    }
429
430    fn finish_candidate_stream(
431        &mut self,
432        phases: &Recorder,
433    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
434        if !self
435            .models
436            .as_ref()
437            .expect("unfinished session has a candidate stream")
438            .exhausted()
439        {
440            self.fail(SolveError::CandidateStreamNotExhausted, phases);
441            return self.pending_error.take().map(Err);
442        }
443        self.complete(SearchState::Exhausted, phases);
444        self.next_retained().map(Ok)
445    }
446
447    fn next_retained(&mut self) -> Option<(Model, Option<Score>)> {
448        self.ready.next().map(|model| {
449            let score = self.incumbents.score().cloned();
450            (model, score)
451        })
452    }
453
454    fn complete(&mut self, search_state: SearchState, phases: &Recorder) {
455        let mut outcome = self.snapshot(phases);
456        outcome.search_state = Some(search_state);
457        self.final_outcome = Some(outcome);
458        self.ready = self.incumbents.take_models();
459    }
460
461    fn fail(&mut self, error: SolveError, phases: &Recorder) {
462        self.final_outcome = Some(self.snapshot(phases));
463        self.pending_error = Some(error);
464    }
465
466    pub(crate) fn outcome(&self, phases: &Recorder) -> SemanticOutcome {
467        self.final_outcome
468            .clone()
469            .unwrap_or_else(|| self.snapshot(phases))
470    }
471
472    pub(crate) const fn finished(&self) -> bool {
473        self.final_outcome.is_some()
474    }
475
476    fn snapshot(&self, phases: &Recorder) -> SemanticOutcome {
477        let statistics = self.models.as_ref().map(StableModels::statistics);
478        self.import_timings(
479            phases,
480            statistics.and_then(|statistics| statistics.phase_timings),
481        );
482        SemanticOutcome {
483            projection: None,
484            subject: Some(crate::Subject::Theory(self.input.theory.clone())),
485            selection: Some(self.selection),
486            verified: statistics.map_or(0, |s| s.stable_models),
487            scored: self.incumbents.scored(),
488            retained: self.incumbents.retained(),
489            search_state: None,
490            optimization: self.incumbents.metadata().cloned(),
491            objective_work: self.incumbents.work(),
492            checked: statistics.map_or(0, |s| s.candidates),
493            gate_atoms: self.input.gate_atoms,
494            candidate_statistics: None,
495            countermodel_statistics: statistics,
496            lazy_execution: None,
497            shared_execution: None,
498            closure_execution: None,
499            query_execution: None,
500            hybrid_execution: None,
501            terminal_execution: None,
502            model_construction: self
503                .construction
504                .as_ref()
505                .map(crate::model_construction::Account::statistics),
506            formula_execution: self
507                .models
508                .as_ref()
509                .and_then(|models| self.execution.as_ref()?.statistics(models)),
510        }
511    }
512
513    fn import_timings(&self, phases: &Recorder, timing: Option<zetesis_sat::SearchPhaseTimings>) {
514        if let Some(timing) = timing {
515            phases.search(self.imported_timings.replace(timing), timing);
516        }
517    }

523impl<'a> FormulaSession<'a, crate::formula_execution::Execution> {
524    /// Own semantic preparation before choosing its execution route. A stopped
525    /// preparation retains its receipts without constructing a device pipeline.
526    /// Route construction keeps the ordinary session's fallible start door.
527    pub(crate) fn with_resources(
528        input: Input<'a>,
529        config: &SolveConfig,
530        resources: &crate::ExecutionResources,
531        observations: &mut impl ExecutionSink,
532        cancellation: &Cancellation,
533        phases: &Recorder,
534        selection: AnswerSelection,
535    ) -> Result<Self, SolveError> {
536        let mut session = Self::uninitialized(input, selection);
537        if let Err(error) = session.initialize(config, observations, cancellation, phases) {
538            session.fail(error, phases);
539        }
540        if session.final_outcome.is_none() {
541            let plan = session
542                .models
543                .as_ref()
544                .and_then(StableModels::prepared_tight_certificate);
545            session.execution = Some(phases.measure(SolvePhase::ExecutionSetup, || {
546                crate::formula_execution::Execution::with_resources(
547                    config,
548                    resources,
549                    plan,
550                    observations,
551                )
552            })?);
553        }
554        Ok(session)
555    }
