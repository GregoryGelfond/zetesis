25pub(crate) struct FormulaSession<'a, E> {
26    input: Input<'a>,
27    selection: AnswerSelection,
28    execution: Option<E>,
29    models: Option<StableModels>,
30    /// The latest construction prefix until a complete stream owns it. On a
31    /// failed tie-pass construction this follows the retained earlier stream.
32    candidate_setup: Option<zetesis_stable::Statistics>,
33    /// The learned rows this solve's streams share, unless sharing is off.
34    learned: Option<Arc<LearnedStore>>,
35    model_order: Option<ModelOrder<'a>>,
36    construction: Option<crate::model_construction::Account>,
37    objective_plan: Option<Preparation>,
38    bounds: Bounds,
39    incumbents: Incumbents,
40    ready: std::vec::IntoIter<Model>,
41    yielded: usize,
42    final_outcome: Option<SemanticOutcome>,
43    pending_error: Option<SolveError>,
44    // This session's last imported cumulative prefix, independently of every
45    // other session sharing the host recorder. Repeated outcomes import no work.
46    imported_timings: Cell<zetesis_stable::SearchPhaseTimings>,
47    // The streamed constraints an owning session installed, which a tie
48    // pass's fresh stream installs again.
49    streamed_constraints: Option<std::sync::Arc<zetesis_stable::StreamedConstraints>>,
50    // While a tie pass runs: how many of the retained optima the strict
51    // search found, which the pass meets again and must not retain twice.
52    tie_pass: Option<usize>,
53    // The strict search exhausted at the optimum before its tie pass: a stop
54    // of that pass leaves the optimal family incomplete, not the optimum
55    // unproved.
56    optimum_established: bool,
57}
58
59/// What one pull returns: the next answer with its score, a fault, or the end
60/// of the answers.
61type Next = Option<Result<(Model, Option<Score>), SolveError>>;
62
63/// How the start of a tie pass ended.
64enum TiePass {
65    /// No tie pass is owed: the stream's end finishes the session.
66    Skipped,
67    /// A fresh stream bounded to cost at most the optimum replaced the
68    /// exhausted one.
69    Started,
70    /// The pass could not start; the proved optimum stands, and the family
71    /// is reported without a claim of completeness.
72    Stopped(zetesis_stable::Incomplete),
73    /// An execution or observation fault.
74    Failed(SolveError),
75}

78    /// Check the owner's streamed constraints over candidate regions, from the
79    /// first proposal on. An already stopped initialization has no traversal
80    /// left to configure.
81    pub(crate) fn set_streamed_constraints(
82        &mut self,
83        constraints: std::sync::Arc<zetesis_stable::StreamedConstraints>,
84        phases: &Recorder,
85    ) {
86        self.streamed_constraints = Some(constraints.clone());
87        if let Some(models) = self.models.as_mut()
88            && let Err(error) = models.set_streamed_constraints(constraints)
89        {
90            self.complete(
91                SearchState::Interrupted(Interruption::Countermodel(error)),
92                phases,
93            );
94        }
95    }

97    /// Stop candidate production and join native workers without cancelling the
98    /// caller's shared token. A device execution then reads its last batch's
99    /// compute-pass interval, unless the session's control, which it kept from
100    /// construction, has fired. Every completion, failure and stop of this
101    /// session or of an owning hybrid or terminal-definition session passes
102    /// here. Refresh receipts after the workers and the execution have settled.
103    pub(crate) fn stop(&mut self, phases: &Recorder) -> Result<(), zetesis_stable::Incomplete> {
104        let result = self.models.as_mut().map_or(Ok(()), StableModels::stop);
105        if let Some(execution) = self.execution.as_mut() {
106            execution.settle(phases);
107        }
108        if let Some(previous) = self.final_outcome.take() {
109            let mut outcome = self.snapshot(phases);
110            outcome.search_state = previous.search_state;
111            // Completion already moved these models into the delivery queue.
112            // Joining workers refreshes search receipts, not retained evidence.
113            outcome.retained = previous.retained;
114            self.final_outcome = Some(outcome);
115        }
116        result
117    }

174    /// The candidate stream this session's configuration selects over its
175    /// theory, within `limits`: native region workers, device region
176    /// producers, or one walk.
177    fn candidate_stream(
178        &self,
179        config: &SolveConfig,
180        limits: zetesis_stable::Limits,
181        cancellation: &Cancellation,
182        phases: &Recorder,
183    ) -> zetesis_stable::ConstructionAttempt {
184        phases.measure(SolvePhase::CandidateSetup, || {
185            // CPU workers decide their own leaves. Device producers return
186            // unchecked leaves to the bounded batch protocol and join before
187            // membership execution starts.
188            if let Some(workers) = config.region_workers() {
189                StableModels::prepare_with_region_workers(
190                    self.input.theory,
191                    workers,
192                    limits,
193                    cancellation.clone(),
194                )
195            } else if config.workers.get() > 1 {
196                StableModels::prepare_with_region_producers(
197                    self.input.theory,
198                    config.workers,
199                    limits,
200                    cancellation.clone(),
201                )
202            } else {
203                StableModels::prepare(self.input.theory, limits, cancellation.clone())
204            }
205        })
206    }

208    fn initialize(
209        &mut self,
210        config: &SolveConfig,
211        observations: &mut impl ExecutionSink,
212        cancellation: &Cancellation,
213        phases: &Recorder,
214    ) -> Result<(), SolveError> {
215        observations.record(Event::Formula {
216            atoms: self.input.theory.atom_count(),
217            nodes: self.input.theory.nodes().len(),
218            operands: self.input.theory.parts().occurrences(),
219            roots: self.input.theory.roots().len(),
220            keyed_constraints: self.input.keyed_constraints,
221        })?;
222        if let zetesis_themelios::KeyAnalysis::Stopped(stop) = self.input.key_analysis {
223            observations.record(Event::KeyAnalysisStopped(stop))?;
224        }
225        let account = self
226            .construction
227            .insert(crate::model_construction::Account::default());
228        let prepared = phases.measure(SolvePhase::ModelConstruction, || {
229            account.prepare(self.input.atoms, config, cancellation)
230        });
231        match prepared {
232            Ok(order) => self.model_order = Some(order),
233            Err(crate::model_construction::Failure::Interrupted(stop)) => {
234                self.complete(
235                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
236                    phases,
237                );
238                return Ok(());
239            }
240            Err(crate::model_construction::Failure::Run(error)) => return Err(error),
241        }
242        let limits = crate::countermodel::search_limits(config);
243        let models = self.candidate_stream(config, limits, cancellation, phases);
244        self.candidate_setup = Some(models.statistics);
245        let mut models = match models.result {
246            Ok(models) => models,
247            Err(error) => {
248                self.complete(
249                    SearchState::Interrupted(Interruption::Countermodel(error)),
250                    phases,
251                );
252                return Ok(());
253            }
254        };
255        models.set_query_division(config.query_division);
256        if config.stats {
257            models.enable_phase_timing();
258        }
259        // Ownership is established before any fallible certificate/diagnostic
260        // operation, so its attempted work remains available on failure.
261        self.models = Some(models);
262        self.candidate_setup = None;
263        // One store for this solve call: every stream the session builds
264        // shares it, and so does the membership execution.
265        if config.share_learned_rows {
266            let store = Arc::new(LearnedStore::new());
267            let models = self.models.as_mut().expect("candidate stream installed");
268            if let Err(error) = models.set_learned_store(Arc::clone(&store)) {
269                self.complete(
270                    SearchState::Interrupted(Interruption::Countermodel(error)),
271                    phases,
272                );
273                return Ok(());
274            }
275            self.learned = Some(store);
276        }
277        if config.region_workers().is_none() && config.workers.get() > 1 {
278            observations.record(Event::ParallelProposals {
279                workers: config.workers,
280            })?;
281        }
282        let models = self.models.as_mut().expect("candidate stream installed");
283        if let Some(error) = crate::countermodel::prepare_certificate(
284            models,
285            self.input.certificate_order,
286            config,
287            observations,
288            phases,
289        )? {
290            self.complete(
291                SearchState::Interrupted(Interruption::Countermodel(error)),
292                phases,
293            );
294            return Ok(());
295        }
296        self.prepare_objectives(config, observations, cancellation, phases)
297    }

299    /// Retain and charge preparation before exposing its optional refusal.
300    fn prepare_objectives(
301        &mut self,
302        config: &SolveConfig,
303        observations: &mut impl ExecutionSink,
304        cancellation: &Cancellation,
305        phases: &Recorder,
306    ) -> Result<(), SolveError> {
307        if self.input.objectives.is_present() {
308            self.objective_plan =
309                Some(phases.measure(SolvePhase::ObjectiveScoringRetention, || {
310                    let mut preparation = Preparation::new(self.input, config, cancellation);
311                    if self.selection == AnswerSelection::Optimal {
312                        preparation.prepare_choices(
313                            self.input.required_choices,
314                            config,
315                            cancellation,
316                        );
317                    }
318                    preparation
319                }));
320            let preparation = self
321                .objective_plan
322                .as_ref()
323                .expect("prepared objective attempt");
324            if let Err(reason) = self.incumbents.prepare(preparation.work()) {
325                self.complete(SearchState::Interrupted(reason), phases);
326                return Ok(());
327            }
328            preparation.observe(observations)?;
329        }
330        self.bounds = Bounds::new(config);
331        // An optimizing search tightens the candidate bound with each better
332        // answer, so its producers share one frontier: production then stays
333        // where the latest splits are, and older regions wait for the tighter
334        // bounds that can refute them near the root.
335        if self.selection == AnswerSelection::Optimal
336            && self.bounds.is_enabled()
337            && self
338                .objective_plan
339                .as_ref()
340                .and_then(Preparation::plan)
341                .is_some()
342            && let Some(models) = self.models.as_mut()
343            && let Err(error) = models.expect_candidate_bounds()
344        {
345            self.complete(
346                SearchState::Interrupted(Interruption::Countermodel(error)),
347                phases,
348            );
349        }
350        Ok(())
351    }

365    /// Qualify each newly constructed model before scoring, retention or bound
366    /// feedback. Ordinary formula execution uses a statically specialized
367    /// identity callback. Retained answers bypass this callback.
368    ///
369    /// A callback failure returns before classifying completion: its owner must
370    /// settle shared failures and workers, then call [`Self::conclude`]. It must
371    /// not resume candidate production after that failure.
372    pub(crate) fn next_with_acceptance(
373        &mut self,
374        config: &SolveConfig,
375        observations: &mut impl ExecutionSink,
376        cancellation: &Cancellation,
377        phases: &Recorder,
378        accept: &mut impl FnMut(&Model) -> Result<bool, SolveError>,
379    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
380        let next = self.next_result(config, observations, cancellation, phases, accept);
381        self.import_timings(
382            phases,
383            self.models
384                .as_ref()
385                .and_then(|models| models.statistics().phase_timings),
386        );
387        next
388    }

390    fn next_result(
391        &mut self,
392        config: &SolveConfig,
393        observations: &mut impl ExecutionSink,
394        cancellation: &Cancellation,
395        phases: &Recorder,
396        accept: &mut impl FnMut(&Model) -> Result<bool, SolveError>,
397    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
398        if let Some(error) = self.pending_error.take() {
399            return Some(Err(error));
400        }
401        if self.final_outcome.is_some() {
402            return self.next_retained().map(Ok);
403        }
404        if self.selection == AnswerSelection::All
405            && config.models != 0
406            && self.yielded >= config.models
407        {
408            self.complete(SearchState::RequestedModels, phases);
409            return None;
410        }
411        // The retained candidate stream owns cumulative work and exact blocks;
412        // pulls never restart it. Objective improvements narrow only candidates.
413        loop {
414            let models = self
415                .models
416                .as_mut()
417                .expect("unfinished session has a candidate stream");
418            let execution = self
419                .execution
420                .as_mut()
421                .expect("unfinished session has a membership executor");
422            // An enumeration owes the answers it has not yet yielded; a
423            // request of zero is unbounded.
424            let enumerates = self.selection == AnswerSelection::All;
425            execution.demand(
426                enumerates,
427                (enumerates && config.models != 0)
428                    .then(|| config.models.saturating_sub(self.yielded)),
429            );
430            let next = execution.next(models, config, cancellation, phases);
431            let interpretation = match next {
432                Some(Ok(model)) => model,
433                Some(Err(Failure::Search(error))) => {
434                    self.complete(
435                        SearchState::Interrupted(Interruption::Countermodel(error)),
436                        phases,
437                    );
438                    return self.next_retained().map(Ok);
439                }
440                Some(Err(Failure::Run(error))) => {
441                    self.fail(error, phases);
442                    return self.pending_error.take().map(Err);
443                }
444                None => match self.end_of_stream(config, observations, cancellation, phases) {
445                    ControlFlow::Continue(()) => continue,
446                    ControlFlow::Break(result) => return result,
447                },
448            };
449            let model =
450                match self.construct_model(interpretation.atoms(), config, cancellation, phases) {
451                    Ok(model) => model,
452                    Err(failure) => return self.finish_construction_failure(failure, phases),
453                };
454            match accept(&model) {
455                Ok(true) => {}
456                Ok(false) => continue,
457                Err(error) => return Some(Err(error)),
458            }
459            if !self.input.objectives.is_present() {
460                self.yielded += 1;
461                return Some(Ok((model, None)));
462            }
463            if self.selection == AnswerSelection::All {
464                return self.score_model(model, &interpretation, config, cancellation, phases);
465            }
466            if let ControlFlow::Break(result) = self.select_optimal(
467                model,
468                &interpretation,
469                config,
470                observations,
471                cancellation,
472                phases,
473            ) {
474                return result;
475            }
476        }
477    }

479    /// Score and retain one verified model of an optimizing search, and bound
480    /// the candidates by it when it improves the incumbent. Continue to search
481    /// on, or break with what the pull returns.
482    fn select_optimal(
483        &mut self,
484        model: Model,
485        interpretation: &zetesis_ferraris::Interpretation,
486        config: &SolveConfig,
487        observations: &mut impl ExecutionSink,
488        cancellation: &Cancellation,
489        phases: &Recorder,
490    ) -> ControlFlow<Next> {
491        if let Some(revisit) = self.tie_pass
492            && self.incumbents.holds_among(revisit, &model)
493        {
494            // The strict search already retained this optimum.
495            return ControlFlow::Continue(());
496        }
497        let scored = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
498            self.incumbents.consider(
499                self.input.objectives,
500                model,
501                config,
502                cancellation,
503                self.objective_plan
504                    .as_ref()
505                    .and_then(Preparation::plan)
506                    .map(|plan| (plan, interpretation)),
507            )
508        });
509        match scored {
510            Ok(true) if self.tie_pass.is_some() => {
511                // The strict search's exhaustion proved that nothing is
512                // cheaper, so this cannot occur; if it did, the pass, under
513                // a bound it never tightens, still meets every model at
514                // the better cost, and the retained models are its own.
515                self.tie_pass = Some(0);
516            }
517            Ok(true) => {
518                let _feedback = (config.max_objective_bound_work != 0)
519                    .then(|| phases.start(SolvePhase::ObjectiveFeedback));
520                let improved = self.bounds.improve(
521                    self.objective_plan.as_ref().and_then(Preparation::plan),
522                    self.incumbents
523                        .score()
524                        .expect("retained improvement has a score"),
525                    self.models.as_mut().expect("owned candidate stream"),
526                    config,
527                    observations,
528                    cancellation,
529                );
530                if let Err(error) = improved {
531                    self.fail(error, phases);
532                    return ControlFlow::Break(self.pending_error.take().map(Err));
533                }
534            }
535            Ok(false) => {}
536            Err(reason) => {
537                self.complete(SearchState::Interrupted(reason), phases);
538                return ControlFlow::Break(self.next_retained().map(Ok));
539            }
540        }
541        ControlFlow::Continue(())
542    }

544    /// At the end of a candidate stream, begin the tie pass when one is owed
545    /// and continue to search it, or break with what the finished search
546    /// returns.
547    fn end_of_stream(
548        &mut self,
549        config: &SolveConfig,
550        observations: &mut impl ExecutionSink,
551        cancellation: &Cancellation,
552        phases: &Recorder,
553    ) -> ControlFlow<Next> {
554        match self.begin_tie_pass(config, observations, cancellation, phases) {
555            TiePass::Started => ControlFlow::Continue(()),
556            TiePass::Skipped => ControlFlow::Break(self.finish_candidate_stream(phases)),
557            TiePass::Stopped(error) => {
558                self.complete(
559                    SearchState::Interrupted(Interruption::Countermodel(error)),
560                    phases,
561                );
562                ControlFlow::Break(self.next_retained().map(Ok))
563            }
564            TiePass::Failed(error) => {
565                self.fail(error, phases);
566                ControlFlow::Break(self.pending_error.take().map(Err))
567            }
568        }
569    }

571    /// Score one verified model without selecting or retaining an incumbent.
572    fn score_model(
573        &mut self,
574        model: Model,
575        interpretation: &zetesis_ferraris::Interpretation,
576        config: &SolveConfig,
577        cancellation: &Cancellation,
578        phases: &Recorder,
579    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
580        let score = phases.measure(SolvePhase::ObjectiveScoringRetention, || {
581            self.incumbents.evaluate(
582                self.input.objectives,
583                &model,
584                config,
585                cancellation,
586                self.objective_plan
587                    .as_ref()
588                    .and_then(Preparation::plan)
589                    .map(|plan| (plan, interpretation)),
590            )
591        });
592        match score {
593            Ok(score) => {
594                self.yielded += 1;
595                Some(Ok((model, Some(score))))
596            }
597            Err(reason) => {
598                self.complete(SearchState::Interrupted(reason), phases);
599                None
600            }
601        }
602    }

624    fn finish_construction_failure(
625        &mut self,
626        failure: crate::model_construction::Failure,
627        phases: &Recorder,
628    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
629        // Settle workers before snapshotting their checked prefix, preserving
630        // the construction failure over any secondary cleanup failure.
631        let _ = self.models.as_mut().expect("owned candidate stream").stop();
632        match failure {
633            crate::model_construction::Failure::Interrupted(stop) => {
634                self.complete(
635                    SearchState::Interrupted(Interruption::ModelConstruction(stop)),
636                    phases,
637                );
638                self.next_retained().map(Ok)
639            }
640            crate::model_construction::Failure::Run(error) => {
641                self.fail(error, phases);
642                self.pending_error.take().map(Err)
643            }
644        }
645    }

647    /// At the end of a candidate stream under strict bounds, start the tie
648    /// pass when one is owed: more than one optimal answer is requested, and
649    /// the last installed bound, strict at the optimum, pruned its ties. The
650    /// exhausted stream proved that no model is cheaper than the optimum; a
651    /// fresh stream over the same theory, with the same certificate and
652    /// streamed constraints and bounded to cost at most the optimum, then
653    /// enumerates every optimal model, each once. It reports its receipts after
654    /// the exhausted stream's, and only its exhaustion completes the optimal
655    /// family; a stop from here on leaves that family incomplete and the
656    /// optimum proved. Its one bound never tightens, so the fresh stream
657    /// declares no candidate bounds: its producers walk apart, as an
658    /// enumeration's do.
659    fn begin_tie_pass(
660        &mut self,
661        config: &SolveConfig,
662        observations: &mut impl ExecutionSink,
663        cancellation: &Cancellation,
664        phases: &Recorder,
665    ) -> TiePass {
666        if self.tie_pass.is_some()
667            || self.selection != AnswerSelection::Optimal
668            || config.models == 1
669        {
670            return TiePass::Skipped;
671        }
672        let Some(optimum) = self.incumbents.score().cloned() else {
673            return TiePass::Skipped;
674        };
675        if self
676            .bounds
677            .pruned_ties_at()
678            .is_none_or(|pruned| pruned.compare_costs(&optimum) != Ordering::Equal)
679        {
680            return TiePass::Skipped;
681        }
682        let earlier = self
683            .models
684            .as_mut()
685            .expect("unfinished session has a candidate stream");
686        if !earlier.exhausted() {
687            // Finishing the stream reports that it did not exhaust.
688            return TiePass::Skipped;
689        }
690        // The exhausted strict search proved that no model is cheaper than
691        // the optimum; whatever stops the tie pass, that proof stands.
692        self.optimum_established = true;
693        if let Err(error) = earlier.stop() {
694            return TiePass::Stopped(error);
695        }
696        // The run's cumulative limits cover both passes together.
697        let limits =
698            crate::countermodel::search_limits(config).remaining_after(&earlier.statistics());
699        let attempt = self.candidate_stream(config, limits, cancellation, phases);
700        self.candidate_setup = Some(attempt.statistics);
701        let mut fresh = match attempt.result {
702            Ok(fresh) => fresh,
703            Err(error) => return TiePass::Stopped(error),
704        };
705        if config.stats {
706            fresh.enable_phase_timing();
707        }
708        if let Err(error) = fresh.follow(self.models.as_ref().expect("exhausted stream")) {
709            return TiePass::Stopped(error);
710        }
711        // The fresh stream reports the exhausted one's receipts with its own,
712        // so it owns them before any fallible step, whose attempted work then
713        // remains in the session's receipts.
714        let fresh = self.models.insert(fresh);
715        self.candidate_setup = None;
716        // The same certificate as the strict search's, already reported.
717        match crate::countermodel::prepare_certificate(
718            fresh,
719            self.input.certificate_order,
720            config,
721            &mut crate::execution_observation::Ignore,
722            phases,
723        ) {
724            Ok(None) => {}
725            Ok(Some(error)) => return TiePass::Stopped(error),
726            Err(error) => return TiePass::Failed(error),
727        }
728        if let Some(constraints) = self.streamed_constraints.clone()
729            && let Err(error) = fresh.set_streamed_constraints(constraints)
730        {
731            return TiePass::Stopped(error);
732        }
733        // The tie pass is another stream of this solve call, so it shares the
734        // call's learned store: the strict pass's rows seed its walk.
735        if let Some(store) = self.learned.clone()
736            && let Err(error) = fresh.set_learned_store(store)
737        {
738            return TiePass::Stopped(error);
739        }
740        if let Err(error) = observations.record(Event::ObjectiveTiePass {
741            costs: optimum.costs(),
742        }) {
743            return TiePass::Failed(error);
744        }
745        if let Err(error) = self.bounds.bound_ties(
746            self.objective_plan.as_ref().and_then(Preparation::plan),
747            &optimum,
748            fresh,
749            config,
750            observations,
751            cancellation,
752        ) {
753            return TiePass::Failed(error);
754        }
755        self.tie_pass = Some(self.incumbents.begin_tie_pass());
756        TiePass::Started
757    }
758
759    fn finish_candidate_stream(
760        &mut self,
761        phases: &Recorder,
762    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
763        if !self
764            .models
765            .as_ref()
766            .expect("unfinished session has a candidate stream")
767            .exhausted()
768        {
769            self.fail(SolveError::CandidateStreamNotExhausted, phases);
770            return self.pending_error.take().map(Err);
771        }
772        self.complete(SearchState::Exhausted, phases);
773        self.next_retained().map(Ok)
774    }
775
776    pub(crate) fn next_retained(&mut self) -> Option<(Model, Option<Score>)> {
777        self.ready.next().map(|model| {
778            let score = self.incumbents.score().cloned();
779            (model, score)
780        })
781    }
782
783    /// Apply an enclosing acceptance stage's settled classification. The caller
784    /// has joined workers and resolved any shared original-source failure first.
785    /// Reclassification preserves an already draining queue; an unclassified
786    /// fault retains its evidence but publishes no more answers.
787    pub(crate) fn conclude(&mut self, state: Option<SearchState>, phases: &Recorder) {
788        if let Some(outcome) = &mut self.final_outcome {
789            outcome.search_state = state;
790        } else if let Some(state) = state {
791            self.complete(state, phases);
792        } else {
793            self.final_outcome = Some(self.snapshot(phases));
794        }
795        if state.is_none() {
796            self.ready = Vec::new().into_iter();
797        }
798    }
799
800    fn complete(&mut self, search_state: SearchState, phases: &Recorder) {
801        let cleanup = self.stop(phases);
802        let mut outcome = self.snapshot(phases);
803        outcome.search_state = crate::completion::after_cleanup(Some(search_state), cleanup);
804        self.final_outcome = Some(outcome);
805        self.ready = self.incumbents.take_models();
806    }
807
808    fn fail(&mut self, error: SolveError, phases: &Recorder) {
809        // Preserve the primary fault, but settle worker receipts before
810        // publishing its checked prefix. Live progress never performs this join.
811        let _ = self.stop(phases);
812        self.final_outcome = Some(self.snapshot(phases));
813        self.pending_error = Some(error);
814    }
815
816    pub(crate) fn outcome(&self, phases: &Recorder) -> SemanticOutcome {
817        self.final_outcome
818            .clone()
819            .unwrap_or_else(|| self.snapshot(phases))
820    }
821
822    pub(crate) const fn finished(&self) -> bool {
823        self.final_outcome.is_some()
824    }
825
826    fn snapshot(&self, phases: &Recorder) -> SemanticOutcome {
827        let statistics = match (
828            self.models.as_ref().map(StableModels::statistics),
829            self.candidate_setup,
830        ) {
831            (Some(earlier), Some(setup)) => Some(earlier.followed_by(&setup)),
832            (earlier, setup) => setup.or(earlier),
833        };
834        self.import_timings(
835            phases,
836            statistics.and_then(|statistics| statistics.phase_timings),
837        );
838        SemanticOutcome {
839            projection: None,
840            original: Some(crate::Original::Theory(self.input.theory.clone())),
841            selection: Some(self.selection),
842            verified: statistics.map_or(0, |s| s.stable_models),
843            scored: self.incumbents.scored(),
844            retained: self.incumbents.retained(),
845            search_state: None,
846            optimum_established: self.optimum_established,
847            optimization: self.incumbents.metadata().map(|best| Optimization {
848                // A strict bound at the optimum pruned its ties, and no tie
849                // pass began to count them again.
850                ties_pruned: self.tie_pass.is_none()
851                    && self
852                        .bounds
853                        .pruned_ties_at()
854                        .is_some_and(|pruned| pruned.compare_costs(&best.score) == Ordering::Equal),
855                ..best.clone()
856            }),
857            objective_work: self.incumbents.work(),
858            checked: statistics.map_or(0, |s| s.candidates),
859            countermodel_statistics: statistics,
860            hybrid_execution: None,
861            terminal_execution: None,
862            source_evaluation: None,
863            model_construction: self
864                .construction
865                .as_ref()
866                .map(crate::model_construction::Account::statistics),
867            formula_execution: self
868                .models
869                .as_ref()
870                .and_then(|models| self.execution.as_ref()?.statistics(models)),
871        }
872    }
873
874    fn import_timings(
875        &self,
876        phases: &Recorder,
877        timing: Option<zetesis_stable::SearchPhaseTimings>,
878    ) {
879        if let Some(timing) = timing {
880            phases.search(self.imported_timings.replace(timing), timing);
881        }
882    }

906    /// Own semantic preparation before choosing its execution route. A stopped
907    /// preparation retains its receipts without constructing a device pipeline.
908    /// Route construction keeps the ordinary session's fallible start door.
909    pub(crate) fn with_resources(
910        input: Input<'a>,
911        config: &SolveConfig,
912        resources: &crate::ExecutionResources,
913        observations: &mut impl ExecutionSink,
914        cancellation: &Cancellation,
915        phases: &Recorder,
916        selection: AnswerSelection,
917    ) -> Result<Self, SolveError> {
918        let mut session = Self::uninitialized(input, selection);
919        if let Err(error) = session.initialize(config, observations, cancellation, phases) {
920            session.fail(error, phases);
921        }
922        if session.final_outcome.is_none() {
923            let plan = session
924                .models
925                .as_ref()
926                .and_then(StableModels::prepared_tight_certificate);
927            session.execution = Some(phases.measure(SolvePhase::ExecutionSetup, || {
928                crate::formula_execution::Execution::with_resources(
929                    config,
930                    resources,
931                    plan,
932                    cancellation,
933                    observations,
934                )
935            })?);
936            session.hand_learned_store();
937        }
938        Ok(session)
939    }
