1//! Shared source-evaluation and materialization policy for ordinary solves.
2//!
3//! `Eager` instantiates every rule before solving. `Lazy` streams eligible
4//! integrity constraints over an instantiated producer core and defers
5//! certified terminal definitions to per-answer reconstruction. `Auto`
6//! first evaluates a certified acyclic normal source directly on the CPU;
7//! otherwise it instantiates an eager base and defers certified terminal definitions. The
8//! choice of relational source joins for programs in the relational profile is
9//! made before formula admission and is not part of this mapping.
10
11use zetesis_themelios::{
12    AdmittedFormula, AdmittedFormulaBundle, FormulaBundleFailure, FormulaFailure,
13    FormulaMaterialization, GroundingObserver, HybridFormula, PreparedFormula,
14    PreparedFormulaBundle, TerminalFormula,
15};

17use crate::{Grounder, PreparedInput, SolveConfig};

19/// A complete semantic input selected for an ordinary solve. The source variant
20/// retains the original program and its checked interpretation; it does not
21/// manufacture a formula theory containing only the computed answer.
22#[derive(Debug)]
23pub enum PreparedExecution {
24    /// Exact CPU source evaluation without producer formula construction.
25    Source(zetesis_themelios::AcyclicSource),
26    /// The materialization selected by an explicit or inapplicable source policy.
27    Grounded(Box<GroundedFormula>),
28}
29
30impl PreparedExecution {
31    /// Borrow the coherent owner for an ordinary session.
32    #[must_use]
33    pub fn input(&self) -> PreparedInput<'_> {
34        match self {
35            Self::Source(owner) => PreparedInput::source(owner),
36            Self::Grounded(owner) => owner.input(),
37        }
38    }
39
40    /// Original declarations and observation policy.
41    #[must_use]
42    pub fn metadata(&self) -> &zetesis_themelios::SourceMetadata {
43        match self {
44            Self::Source(owner) => owner.metadata(),
45            Self::Grounded(owner) => owner.metadata(),
46        }
47    }
48}
49
50pub(crate) fn source_compatible(config: &SolveConfig) -> bool {
51    config.backend == crate::Backend::Cpu
52        && config.oracle == crate::Oracle::Auto
53        && config.grounder == Grounder::Auto
54        && config.source_batching == crate::SourceBatching::Independent
55}

57/// Prepare one source or typed program for an ordinary solve. Automatic CPU
58/// execution first attempts complete signed-acyclic source certification.
59/// Explicit grounding, device and oracle requests retain their materialization
60/// policy. A class mismatch resumes the same preparation with its work retained.
61///
62/// # Errors
63/// Returns the selected admission failure. Cancellation, exhausted allowances
64/// and invalid source never cause a fresh attempt under another policy.
65pub fn prepare_formula_execution(
66    prepared: PreparedFormula,
67    config: &SolveConfig,
68    observer: Option<&dyn GroundingObserver>,
69) -> Result<PreparedExecution, FormulaFailure> {
70    let prepared = if source_compatible(config) {
71        match prepared.try_source_with_observer(observer)? {
72            zetesis_themelios::SourceAdmission::Admitted(owner) => {
73                return Ok(PreparedExecution::Source(owner));
74            }
75            zetesis_themelios::SourceAdmission::NotApplicable(prepared) => prepared,
76        }
77    } else {
78        prepared
79    };
80    ground_formula(prepared, config.grounder, observer)
81        .map(|owner| PreparedExecution::Grounded(Box::new(owner)))
82}

84/// Prepare an original include bundle with the same policy as a single input.
85/// No source text is rendered or reparsed at this boundary.
86///
87/// # Errors
88/// Returns the selected admission failure with the original source bundle.
89pub fn prepare_bundle_execution(
90    prepared: PreparedFormulaBundle,
91    config: &SolveConfig,
92    observer: Option<&dyn GroundingObserver>,
93) -> Result<PreparedExecution, FormulaBundleFailure> {
94    let prepared = if source_compatible(config) {
95        match prepared.try_source_with_observer(observer)? {
96            zetesis_themelios::SourceAdmission::Admitted(owner) => {
97                return Ok(PreparedExecution::Source(owner));
98            }
99            zetesis_themelios::SourceAdmission::NotApplicable(prepared) => prepared,
100        }
101    } else {
102        prepared
103    };
104    ground_bundle(prepared, config.grounder, observer)
105        .map(|owner| PreparedExecution::Grounded(Box::new(owner)))
106}

108/// A formula program materialized for one requested grounder.
109#[derive(Debug)]
110pub enum GroundedFormula {
111    /// Every rule of a single source or logical program instantiated.
112    Source(AdmittedFormula),
113    /// Every rule of an include bundle instantiated.
114    Bundle(AdmittedFormulaBundle),
115    /// A producer core with streamed constraints; nothing deferred.
116    Hybrid(HybridFormula),
117    /// A base, eager or hybrid, with deferred terminal definitions.
118    Terminal(TerminalFormula),
119}
120
121impl GroundedFormula {
122    /// Borrow the materialization as a session input.
123    #[must_use]
124    pub fn input(&self) -> PreparedInput<'_> {
125        match self {
126            Self::Source(owner) => PreparedInput::formula(owner),
127            Self::Bundle(owner) => PreparedInput::formula_bundle(owner),
128            Self::Hybrid(owner) => PreparedInput::hybrid(owner),
129            Self::Terminal(owner) => PreparedInput::terminal(owner),
130        }
131    }
132
133    /// The original declarations and observation policy.
134    #[must_use]
135    pub fn metadata(&self) -> &zetesis_themelios::SourceMetadata {
136        match self {
137            Self::Source(owner) => owner.metadata(),
138            Self::Bundle(owner) => owner.metadata(),
139            Self::Hybrid(owner) => owner.metadata(),
140            Self::Terminal(owner) => owner.metadata(),
141        }
142    }
143}

145/// Materialize a prepared single-source or logical program for `grounder`.
146///
147/// # Errors
148/// Returns the refusal of the materialization the grounder selects; see
149/// `PreparedFormula::ground`, `ground_lazy` and `ground_adaptive`.
150pub fn ground_formula(
151    prepared: PreparedFormula,
152    grounder: Grounder,
153    observer: Option<&dyn GroundingObserver>,
154) -> Result<GroundedFormula, FormulaFailure> {
155    match grounder {
156        Grounder::Eager => prepared
157            .ground_with_observer(observer)
158            .map(GroundedFormula::Source),
159        Grounder::Lazy => prepared
160            .ground_lazy_with_observer(observer)
161            .map(|materialized| match materialized {
162                FormulaMaterialization::Complete(owner) => GroundedFormula::Hybrid(owner),
163                FormulaMaterialization::Terminal(owner) => GroundedFormula::Terminal(owner),
164            }),
165        Grounder::Auto => prepared
166            .ground_adaptive_with_observer(observer)
167            .map(|materialized| match materialized {
168                FormulaMaterialization::Complete(owner) => GroundedFormula::Source(owner),
169                FormulaMaterialization::Terminal(owner) => GroundedFormula::Terminal(owner),
170            }),
171    }
172}

174/// Materialize a prepared include bundle for `grounder`.
175///
176/// # Errors
177/// Returns the refusal of the materialization the grounder selects, with the
178/// original bundle.
179pub fn ground_bundle(
180    prepared: PreparedFormulaBundle,
181    grounder: Grounder,
182    observer: Option<&dyn GroundingObserver>,
183) -> Result<GroundedFormula, FormulaBundleFailure> {
184    match grounder {
185        Grounder::Eager => prepared
186            .ground_with_observer(observer)
187            .map(GroundedFormula::Bundle),
188        Grounder::Lazy => prepared
189            .ground_lazy_with_observer(observer)
190            .map(|materialized| match materialized {
191                FormulaMaterialization::Complete(owner) => GroundedFormula::Hybrid(owner),
192                FormulaMaterialization::Terminal(owner) => GroundedFormula::Terminal(owner),
193            }),
194        Grounder::Auto => prepared
195            .ground_adaptive_with_observer(observer)
196            .map(|materialized| match materialized {
197                FormulaMaterialization::Complete(owner) => GroundedFormula::Bundle(owner),
198                FormulaMaterialization::Terminal(owner) => GroundedFormula::Terminal(owner),
199            }),
200    }
