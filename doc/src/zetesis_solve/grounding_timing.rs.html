1//! Fixed-space aggregation of optional formula-grounding observations.
2
3use std::sync::{Mutex, MutexGuard};
4use std::time::{Duration, Instant};
5
6use zetesis_themelios::{GroundingOutcome, GroundingPhase, GroundingWork};
7
8/// Host intervals and work for all attempted occurrences of one grounding phase.
9/// Counts include failed and unwinding attempts; they are not semantic verdicts.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub struct GroundingMeasurement {
12    /// Sum of observed host intervals, or `None` on duration overflow.
13    pub elapsed: Option<Duration>,
14    /// Combined frontend work counts and maximum support-capacity peak.
15    pub work: GroundingWork,
16    outcomes: [Option<u64>; GroundingOutcome::ALL.len()],
17}
18
19impl GroundingMeasurement {
20    /// Number of phase attempts ending with this outcome, or `None` on overflow.
21    #[must_use]
22    pub fn count(&self, outcome: GroundingOutcome) -> Option<u64> {
23        self.outcomes[outcome_index(outcome)]
24    }
25
26    fn record(&mut self, elapsed: Duration, outcome: GroundingOutcome, work: &GroundingWork) {
27        self.elapsed = self.elapsed.and_then(|value| value.checked_add(elapsed));
28        let count = &mut self.outcomes[outcome_index(outcome)];
29        *count = count.and_then(|value| value.checked_add(1));
30        self.work = self.work.checked_sum(*work);
31    }
32}
33
34impl Default for GroundingMeasurement {
35    fn default() -> Self {
36        Self {
37            elapsed: Some(Duration::ZERO),
38            work: GroundingWork::default(),
39            outcomes: [Some(0); GroundingOutcome::ALL.len()],
40        }
41    }
42}
43
44/// Optional eager-formula attribution aggregated by phase in constant space.
45/// Rule locations are available to custom frontend observers but are not retained
46/// here. Durations include counter/timer overhead and are not GPU kernel times.
47/// This view does not attribute relational eager or interleaved lazy grounding.
48/// An unrequested frontend domain-analysis phase is absent. When requested by a
49/// library preparation, its actual work is aggregated through the same observer;
50/// a completed optional attempt need not have reached an analysis fixed point.
51#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
52pub struct GroundingTimings {
53    invalid: bool,
54    measurements: [Option<GroundingMeasurement>; GroundingPhase::ALL.len()],
55}
56
57impl GroundingTimings {
58    /// Recorded attempts of a phase; `None` means no completed callback pair or
59    /// unavailable bookkeeping after poisoning. It is not zero work.
60    #[must_use]
61    pub fn get(&self, phase: GroundingPhase) -> Option<&GroundingMeasurement> {
62        (!self.invalid)
63            .then(|| self.measurements[phase_index(phase)].as_ref())
64            .flatten()
65    }
66}
67
68fn phase_index(phase: GroundingPhase) -> usize {
69    GroundingPhase::ALL
70        .iter()
71        .position(|&entry| entry == phase)
72        .expect("frontend phase catalog covers its variants")
73}
74
75fn outcome_index(outcome: GroundingOutcome) -> usize {
76    GroundingOutcome::ALL
77        .iter()
78        .position(|&entry| entry == outcome)
79        .expect("frontend outcome catalog covers its variants")
80}
81
82#[derive(Default)]
83pub(crate) struct Recorder {
84    timings: Mutex<GroundingTimings>,
85}
86
87// A callback observer owns its own active interval. Concurrent observers never
88// replace another observer's entry, and no lock spans frontend grounding work.
89#[must_use = "retain the attempt until its matching phase exit"]
90pub(crate) struct Attempt {
91    phase: GroundingPhase,
92    started: Instant,
93}
94
95impl Attempt {
96    pub(crate) fn start(phase: GroundingPhase) -> Self {
97        Self {
98            phase,
99            started: Instant::now(),
100        }
101    }
102
103    fn finish(self, phase: GroundingPhase) -> Duration {
104        assert_eq!(phase, self.phase, "phase exit identifies its entry");
105        self.started.elapsed()
106    }
107}
108
109impl Recorder {
110    pub(crate) fn exit(
111        &self,
112        attempt: Attempt,
113        phase: GroundingPhase,
114        outcome: GroundingOutcome,
115        work: &GroundingWork,
116    ) {
117        let elapsed = attempt.finish(phase);
118        self.lock_timings().measurements[phase_index(phase)]
119            .get_or_insert_default()
120            .record(elapsed, outcome, work);
121    }
122
123    fn lock_timings(&self) -> MutexGuard<'_, GroundingTimings> {
124        self.timings.lock().unwrap_or_else(|poisoned| {
125            let mut timings = poisoned.into_inner();
126            timings.invalid = true;
127            timings
128        })
129    }
130
131    pub(crate) fn snapshot(&self) -> GroundingTimings {
132        *self.lock_timings()
133    }
134}
