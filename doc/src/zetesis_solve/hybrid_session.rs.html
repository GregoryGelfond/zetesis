1//! Source restrictions on candidate regions and complete answer checking.
2//!
3//! The retained core uses the ordinary formula session. Its answers
4//! become answers of the original subject only after all streamed constraints
5//! are satisfied. The append-constraints law justifies this composition; source
6//! instance coverage and completed admission remain separate prerequisites.
7

20/// Consumed core answers and their streamed-constraint decisions.
21///
22/// `core_answers = accepted + rejected + pending`. Core answers still buffered
23/// by the inner enumerator are excluded. Only `accepted` establishes membership
24/// in the original program; core membership alone is insufficient.
25#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
26pub struct HybridExecutionStatistics {
27    /// Retained-core answers consumed by the source constraint checker.
28    pub core_answers: u64,
29    /// Answers satisfying every required streamed constraint instance.
30    pub accepted: u64,
31    /// Core answers rejected by an observed constraint violation.
32    pub rejected: u64,
33    /// Consumed core answers whose constraint check did not finish.
34    pub pending: u64,
35    /// Cumulative source checking work, including region checks, preparation and
36    /// interrupted attempts. Live worker counters become final after joining.
37    pub constraints: zetesis_themelios::ConstraintCheckStatistics,
38}

43pub(crate) struct HybridSession<'a> {
44    owner: &'a HybridFormula,
45    core: FormulaSession<'a, Execution>,
46    core_config: SolveConfig,
47    checker: ConstraintChecker<'a>,
48    allowance: ConstraintAllowance,
49    regions: Option<Arc<crate::hybrid_regions::Constraints>>,
50    statistics: HybridExecutionStatistics,
51    final_outcome: Option<SemanticOutcome>,
52}
53
54impl<'a> HybridSession<'a> {
55    pub(crate) fn new(
56        owner: &'a HybridFormula,
57        config: &SolveConfig,
58        resources: &crate::ExecutionResources,
59        observations: &mut impl ExecutionSink,
60        cancellation: &Cancellation,
61        phases: &Recorder,
62        selection: AnswerSelection,
63    ) -> Result<Self, SolveError> {
64        phases.lazy_grounding();
65        observations.record(crate::ExecutionObservation::HybridGrounding {
66            requested: config.grounder,
67            streamed_templates: owner.streamed_templates(),
68            streamed_instances: owner.streamed_instances(),
69        })?;
70        let allowance = ConstraintAllowance::new(config.constraints);
71        let checker = owner
72            .checker_with_allowance(&allowance, cancellation)
73            .map_err(SolveError::Constraint)?;
74        let statistics = HybridExecutionStatistics {
75            constraints: checker.statistics(),
76            ..HybridExecutionStatistics::default()
77        };
78        let core_config = SolveConfig {
79            models: 0,
80            grounder: crate::Grounder::Eager,
81            ..*config
82        };
83        let input = crate::countermodel::Input {
84            theory: owner.core_theory(),
85            atoms: owner.atom_catalog(),
86            gate_atoms: 0,
87            keyed_constraints: owner.keyed_constraints(),
88            key_analysis: owner.key_analysis(),
89            objectives: owner.objectives(),
90            certificate_order: crate::countermodel::certificate_order(
91                owner.source_analysis(),
92                owner.analysis_basis(),
93            ),
94        };
95        let mut core = FormulaSession::with_resources(
96            input,
97            &core_config,
98            resources,
99            observations,
100            cancellation,
101            phases,
102            selection,
103        )?;
104        let regions = (owner.streamed_templates() != 0
105            && config.search == crate::SearchMethod::Regions)
106            .then(|| {
107                let regions = Arc::new(crate::hybrid_regions::Constraints::new(
108                    owner.clone(),
109                    allowance.clone(),
110                ));
111                core.set_region_filter(regions.clone(), phases);
112                regions
113            });
114        Ok(Self {
115            owner,
116            core,
117            core_config,
118            checker,
119            allowance,
120            regions,
121            statistics,
122            final_outcome: None,
123        })
124    }
125
126    pub(crate) fn next(
127        &mut self,
128        config: &SolveConfig,
129        observations: &mut impl ExecutionSink,
130        cancellation: &Cancellation,
131        phases: &Recorder,
132    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
133        if self.final_outcome.is_some() {
134            return None;
135        }
136        if config.models != 0 && self.statistics.accepted >= config.models as u64 {
137            return self.finish(Some(SearchState::RequestedModels), None, phases);
138        }
139        loop {
140            let model = match self
141                .core
142                .next(&self.core_config, observations, cancellation, phases)
143            {
144                Some(Ok((model, _))) => model,
145                Some(Err(error)) => {
146                    return self.finish(None, Some(error), phases);
147                }
148                None => return self.finish(self.core.outcome(phases).search_state, None, phases),
149            };
150            let Some(count) = self.statistics.core_answers.checked_add(1) else {
151                return self.finish(None, Some(SolveError::HybridStatisticsOverflow), phases);
152            };
153            self.statistics.core_answers = count;
154            self.statistics.pending = 1;
155            let checked = phases.measure(SolvePhase::OriginalValidation, || {
156                self.checker.check(&model, cancellation)
157            });
158            self.statistics.constraints = self.allowance.statistics();
159            match checked {
160                Ok(ConstraintVerdict::Satisfied) => {
161                    // The completed count is bounded by the checked core count.
162                    self.statistics.accepted += 1;
163                    self.statistics.pending = 0;
164                    return Some(Ok((model, None)));
165                }
166                Ok(ConstraintVerdict::Violated { .. }) => {
167                    self.statistics.rejected += 1;
168                    self.statistics.pending = 0;
169                }
170                Err(error) => {
171                    return self.finish(None, Some(SolveError::Constraint(error)), phases);
172                }
173            }
174        }
175    }
176
177    fn snapshot(&self, phases: &Recorder) -> SemanticOutcome {
178        let mut outcome = self.core.outcome(phases);
179        outcome.subject = Some(crate::Subject::Hybrid(self.owner.clone()));
180        outcome.verified = self.statistics.accepted;
181        // Core exhaustion cannot establish completion of an unfinished source check.
182        outcome.search_state = None;
183        outcome.hybrid_execution = Some(HybridExecutionStatistics {
184            constraints: self.allowance.statistics(),
185            ..self.statistics
186        });
187        outcome
188    }

190    /// Record a final-check failure before stopping workers, then resolve every
191    /// terminal path once. A worker may already have recorded a source fault;
192    /// their shared slot preserves the first recording, not a wall-clock order.
193    fn finish(
194        &mut self,
195        state: Option<SearchState>,
196        error: Option<SolveError>,
197        phases: &Recorder,
198    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
199        let mut error = match (error, &self.regions) {
200            (Some(SolveError::Constraint(error)), Some(regions)) => {
201                regions.record_failure(error);
202                None
203            }
204            (error, _) => error,
205        };
206        let stopped = self.core.stop(phases);
207        let source_failure = self.regions.as_ref().and_then(|r| r.take_failure());
208        let source_state = matches!(
209            state,
210            Some(SearchState::Interrupted(Interruption::Countermodel(
211                zetesis_sat::Incomplete::RegionFilter
212            )))
213        );
214        // A source marker still needs its precise recorded cause. Any other
215        // finalized interruption precedes faults discovered while joining;
216        // taking the slot above still settles the source's retained receipt.
217        if !matches!(state, Some(SearchState::Interrupted(_))) || source_state {
218            error = error.or_else(|| source_failure.map(SolveError::Constraint));
219            if (stopped == Err(zetesis_sat::Incomplete::RegionFilter) || source_state)
220                && error.is_none()
221            {
222                error = Some(SolveError::ConstraintFailureMissing);
223            }
224        }
225        let (state, result) = match error {
226            Some(SolveError::Constraint(error)) if error.stop().is_some() => (
227                Some(SearchState::Interrupted(Interruption::Constraint(
228                    error.stop().expect("checked source stop"),
229                ))),
230                None,
231            ),
232            Some(error) => (None, Some(Err(error))),
233            None => (crate::completion::after_cleanup(state, stopped), None),
234        };
235        let mut outcome = self.snapshot(phases);
236        outcome.search_state = state;
237        self.final_outcome = Some(outcome);
238        result
239    }
240
241    pub(crate) fn outcome(&self, phases: &Recorder) -> SemanticOutcome {
242        self.final_outcome
243            .clone()
244            .unwrap_or_else(|| self.snapshot(phases))
245    }
246
247    pub(crate) const fn finished(&self) -> bool {
248        self.final_outcome.is_some()
249    }
