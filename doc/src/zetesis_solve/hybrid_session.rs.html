1//! Source restrictions on candidate regions and complete answer checking.
2//!
3//! The retained core uses the ordinary formula session. Its answers
4//! become answers of the original input only after all streamed constraints
5//! are satisfied. Only those complete original answers enter scoring, incumbent
6//! retention and objective-bound feedback. The retained formula core uses the
7//! selected backend;
8//! streamed constraints are checked on the host before scoring or publication.
9//! Terminal-definition objectives remain excluded.
10//! The append-constraints law justifies this composition; source
11//! instance coverage and completed admission remain separate prerequisites.
12

26/// Consumed core answers and their streamed-constraint decisions.
27///
28/// `core_answers = accepted + rejected + pending`. Core answers still buffered
29/// by the inner enumerator are excluded. Only `accepted` establishes membership
30/// in the original program; core membership alone is insufficient.
31#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
32pub struct HybridExecutionStatistics {
33    /// Retained-core answers consumed by the source constraint checker.
34    pub core_answers: u64,
35    /// Answers satisfying every required streamed constraint instance.
36    pub accepted: u64,
37    /// Core answers rejected by an observed constraint violation.
38    pub rejected: u64,
39    /// Consumed core answers whose constraint check did not finish.
40    pub pending: u64,
41    /// Cumulative source checking work, including region checks, preparation and
42    /// interrupted attempts. Live worker counters become final after joining.
43    pub constraints: zetesis_themelios::ConstraintCheckStatistics,
44    /// The eligible constraints streamed, and those grounded in the core.
45    pub streamed: crate::StreamedConstraints,
46}

58pub(crate) struct HybridSession<'a> {
59    /// The original input an outcome reports: the hybrid owner, or the terminal owner
60    /// whose base this core is.
61    original: crate::Original,
62    core: FormulaSession<'a, Execution>,
63    checker: ConstraintChecker<'a>,
64    allowance: ConstraintAllowance,
65    regions: Option<Arc<zetesis_stable::StreamedConstraints>>,
66    statistics: HybridExecutionStatistics,
67    final_outcome: Option<SemanticOutcome>,
68}

70/// Establish original membership for one fresh core answer. A failed check
71/// leaves it pending; successful acceptance precedes every objective operation.
72fn accept(
73    checker: &mut ConstraintChecker<'_>,
74    statistics: &mut HybridExecutionStatistics,
75    model: &Model,
76    cancellation: &Cancellation,
77    phases: &Recorder,
78) -> Result<bool, SolveError> {
79    let count = CheckedSum
80        .combine(Some(statistics.core_answers), Some(1))
81        .ok_or(SolveError::HybridStatisticsOverflow)?;
82    statistics.core_answers = count;
83    statistics.pending = 1;
84    let verdict = phases.measure(SolvePhase::OriginalValidation, || {
85        checker.check(model, cancellation)
86    });
87    match verdict.map_err(SolveError::Constraint)? {
88        ConstraintVerdict::Satisfied => {
89            // A refused score still leaves this original answer counted.
90            statistics.accepted += 1;
91            statistics.pending = 0;
92            Ok(true)
93        }
94        ConstraintVerdict::Violated { .. } => {
95            statistics.rejected += 1;
96            statistics.pending = 0;
97            Ok(false)
98        }
99    }
100}

113    /// A hybrid owner's session: the route is recorded as lazy grounding and
114    /// observed as hybrid grounding before the core session starts. A terminal
115    /// owner's hybrid base is observed by its terminal session instead.
116    pub(crate) fn observed(
117        input: HybridInput<'a>,
118        config: &SolveConfig,
119        resources: &crate::ExecutionResources,
120        observations: &mut impl ExecutionSink,
121        cancellation: &Cancellation,
122        phases: &Recorder,
123        selection: AnswerSelection,
124    ) -> Result<Self, SolveError> {
125        phases.lazy_grounding();
126        observations.record(crate::ExecutionObservation::HybridGrounding {
127            requested: config.grounder,
128            streamed: crate::StreamedConstraints::of(input.core),
129        })?;
130        Self::new(
131            input,
132            config,
133            resources,
134            observations,
135            cancellation,
136            phases,
137            selection,
138        )
139    }

141    pub(crate) fn new(
142        input: HybridInput<'a>,
143        config: &SolveConfig,
144        resources: &crate::ExecutionResources,
145        observations: &mut impl ExecutionSink,
146        cancellation: &Cancellation,
147        phases: &Recorder,
148        selection: AnswerSelection,
149    ) -> Result<Self, SolveError> {
150        let HybridInput {
151            core: owner,
152            original,
153        } = input;
154        let allowance = ConstraintAllowance::new(config.constraints);
155        let checker = owner
156            .checker_with_allowance(&allowance, cancellation)
157            .map_err(SolveError::Constraint)?;
158        let statistics = HybridExecutionStatistics {
159            streamed: crate::StreamedConstraints::of(owner),
160            ..HybridExecutionStatistics::default()
161        };
162        let core_config = SolveConfig {
163            grounder: crate::Grounder::Eager,
164            ..*config
165        };
166        let input = crate::countermodel::Input {
167            theory: owner.core_theory(),
168            atoms: owner.atom_catalog(),
169            keyed_constraints: owner.keyed_constraints(),
170            key_analysis: owner.key_analysis(),
171            objectives: owner.objectives(),
172            required_choices: owner.required_choices(),
173            certificate_order: crate::countermodel::certificate_order(
174                owner.source_classes(),
175                owner.analysis_basis(),
176            ),
177        };
178        let mut core = FormulaSession::with_resources(
179            input,
180            &core_config,
181            resources,
182            observations,
183            cancellation,
184            phases,
185            selection,
186        )?;
187        let regions = (owner.streamed_templates() != 0).then(|| {
188            let regions = Arc::new(zetesis_stable::StreamedConstraints::new(
189                owner.clone(),
190                allowance.clone(),
191            ));
192            core.set_streamed_constraints(regions.clone(), phases);
193            regions
194        });
195        Ok(Self {
196            original,
197            core,
198            checker,
199            allowance,
200            regions,
201            statistics,
202            final_outcome: None,
203        })
204    }

206    pub(crate) fn next(
207        &mut self,
208        config: &SolveConfig,
209        observations: &mut impl ExecutionSink,
210        cancellation: &Cancellation,
211        phases: &Recorder,
212    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
213        if self.final_outcome.is_some() {
214            return self.core.next_retained().map(Ok);
215        }
216        let Self {
217            core,
218            checker,
219            statistics,
220            ..
221        } = self;
222        let next =
223            core.next_with_acceptance(config, observations, cancellation, phases, &mut |model| {
224                accept(checker, statistics, model, cancellation, phases)
225            });
226        match next {
227            Some(Err(error)) => {
228                let state = self.core.outcome(phases).search_state;
229                match self.finish(state, Some(error), phases) {
230                    Ok(()) => self.core.next_retained().map(Ok),
231                    Err(error) => Some(Err(error)),
232                }
233            }
234            next => {
235                if self.core.finished() {
236                    let state = self.core.outcome(phases).search_state;
237                    if let Err(error) = self.finish(state, None, phases) {
238                        return Some(Err(error));
239                    }
240                }
241                next
242            }
243        }
244    }
245
246    fn snapshot(&self, phases: &Recorder) -> SemanticOutcome {
247        let mut outcome = self.core.outcome(phases);
248        outcome.original = Some(self.original.clone());
249        outcome.verified = self.statistics.accepted;
250        // Core exhaustion cannot establish completion of an unfinished source check.
251        outcome.search_state = None;
252        outcome.hybrid_execution = Some(HybridExecutionStatistics {
253            constraints: self.allowance.statistics(),
254            ..self.statistics
255        });
256        outcome
257    }

259    /// Record a final-check failure before stopping workers, then resolve every
260    /// terminal path once. A worker may already have recorded a source fault;
261    /// their shared slot preserves the first recording, not a wall-clock order.
262    fn finish(
263        &mut self,
264        state: Option<SearchState>,
265        error: Option<SolveError>,
266        phases: &Recorder,
267    ) -> Result<(), SolveError> {
268        let mut error = match (error, &self.regions) {
269            (Some(SolveError::Constraint(error)), Some(regions)) => {
270                regions.record_failure(error);
271                None
272            }
273            (error, _) => error,
274        };
275        let stopped = self.core.stop(phases);
276        let source_failure = self.regions.as_ref().and_then(|r| r.take_failure());
277        let source_state = matches!(
278            state,
279            Some(SearchState::Interrupted(Interruption::Countermodel(
280                zetesis_stable::Incomplete::RegionFilter
281            )))
282        );
283        // A source marker still needs its precise recorded cause. Any other
284        // finalized interruption precedes faults discovered while joining;
285        // taking the slot above still settles the source's retained receipt.
286        if !matches!(state, Some(SearchState::Interrupted(_))) || source_state {
287            error = error.or_else(|| source_failure.map(SolveError::Constraint));
288            if (stopped == Err(zetesis_stable::Incomplete::RegionFilter) || source_state)
289                && error.is_none()
290            {
291                error = Some(SolveError::ConstraintFailureMissing);
292            }
293        }
294        let (state, result) = match error {
295            Some(SolveError::Constraint(error)) if error.stop().is_some() => (
296                Some(SearchState::Interrupted(Interruption::Constraint(
297                    error.stop().expect("checked source stop"),
298                ))),
299                Ok(()),
300            ),
301            Some(error) => (None, Err(error)),
302            None => (crate::completion::after_cleanup(state, stopped), Ok(())),
303        };
304        self.core.conclude(state, phases);
305        let mut outcome = self.snapshot(phases);
306        outcome.search_state = state;
307        self.final_outcome = Some(outcome);
308        result
309    }
310
311    pub(crate) fn outcome(&self, phases: &Recorder) -> SemanticOutcome {
312        self.final_outcome
313            .clone()
314            .unwrap_or_else(|| self.snapshot(phases))
315    }
316
317    pub(crate) const fn finished(&self) -> bool {
318        self.final_outcome.is_some()
319    }
320
321    /// Settle this session for an enclosing one that stops for `state`: resolve
322    /// it once (workers stopped, the region-failure slot taken) unless it has
323    /// already finished, and return its final search state, or the constraint
324    /// failure that cleanup established.
325    pub(crate) fn conclude(
326        &mut self,
327        state: Option<SearchState>,
328        phases: &Recorder,
329    ) -> Result<Option<SearchState>, SolveError> {
330        if let Some(outcome) = &self.final_outcome {
331            return Ok(outcome.search_state);
332        }
333        self.finish(state, None, phases)?;
334        Ok(self
335            .final_outcome
336            .as_ref()
337            .and_then(|outcome| outcome.search_state))
338    }
