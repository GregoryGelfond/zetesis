1//! Feature-independent views of submitted per-binding transport observations.
2
3/// Requests associated with one buffer binding in submitted lazy chunks.
4/// Allocation and reuse sum to the observed dispatch count for this binding.
5#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
6pub struct LazyBufferUsage {
7    /// Submitted chunks requesting a new buffer for this binding.
8    pub allocations: u64,
9    /// Submitted chunks retaining this binding's existing buffer.
10    pub reuses: u64,
11}
12
13/// Per-binding observations, separate from whole-set replacement causes.
14/// Counts describe authored buffer requests, not driver allocations or RSS.
15/// Views remain available without the optional GPU dependency.
16#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
17pub struct LazyTransportUsage {
18    /// Dispatch dimensions.
19    pub uniform: LazyBufferUsage,
20    /// Source-row offsets.
21    pub offsets: LazyBufferUsage,
22    /// Source-instance records.
23    pub records: LazyBufferUsage,
24    /// Immutable candidate snapshots.
25    pub snapshots: LazyBufferUsage,
26    /// Frozen candidate seeds.
27    pub seeds: LazyBufferUsage,
28    /// Exact-sized device output.
29    pub output: LazyBufferUsage,
30    /// Exact-sized host readback.
31    pub readback: LazyBufferUsage,
32    /// Submissions releasing input slack to fit the prospective byte ceiling.
33    pub budget_releases: u64,
34    /// Submissions releasing input slack after prospective accounting overflow.
35    pub accounting_overflow_releases: u64,
36}
37
38#[cfg(feature = "gpu")]
39impl LazyBufferUsage {
40    fn record(self, observed: zetesis_wgpu::LazyBufferUsage) -> Option<Self> {
41        Some(Self {
42            allocations: self.allocations.checked_add(observed.allocations)?,
43            reuses: self.reuses.checked_add(observed.reuses)?,
44        })
45    }
46}
47
48#[cfg(feature = "gpu")]
49impl LazyTransportUsage {
50    pub(super) fn record(self, observed: zetesis_wgpu::LazyTransportUsage) -> Option<Self> {
51        Some(Self {
52            uniform: self.uniform.record(observed.uniform)?,
53            offsets: self.offsets.record(observed.offsets)?,
54            records: self.records.record(observed.records)?,
55            snapshots: self.snapshots.record(observed.snapshots)?,
56            seeds: self.seeds.record(observed.seeds)?,
57            output: self.output.record(observed.output)?,
58            readback: self.readback.record(observed.readback)?,
59            budget_releases: self.budget_releases.checked_add(observed.budget_releases)?,
60            accounting_overflow_releases: self
61                .accounting_overflow_releases
62                .checked_add(observed.accounting_overflow_releases)?,
63        })
64    }
65}
