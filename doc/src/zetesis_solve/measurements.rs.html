3use std::sync::Arc;
4
5use crate::phase_timing::{Recorder, Span};
6use crate::{PhaseTimings, SolvePhase, SolveStage};
7use zetesis_telemetry::StageSpan;

9/// Caller-owned host measurements shared across operations and threads.
10///
11/// Cloning shares one recorder and its original start time. It does not copy or
12/// restart measurements. The last owner releases the recorder; guards and
13/// grounding observers borrow an owner and cannot outlive it. Independent calls
14/// to [`Self::new`] or [`Self::stages_only`] create independent scopes. Clones
15/// are `Send` and `Sync`; brief internal locks protect bookkeeping without
16/// enclosing application work, frontend calls or observer callbacks.
17///
18/// Each owner retains a fixed set of phase, stage and grounding slots. Creating
19/// an owner allocates one fixed-size shared record; cloning and all recording
20/// operations take constant space. Snapshots and recording visit only those
21/// fixed slots; concurrent bookkeeping may briefly wait for a lock. Enabled
22/// stages read the host clock. Detailed measurements additionally time phase
23/// attempts and collect frontend work. A stages-only scope leaves those
24/// detailed measurements absent. Disabled operations read no clocks, and
25/// snapshots remain absent.
26///
27/// Callers define the measured scope and may record arbitrary attempted work.
28/// Neither a measurement nor its label establishes admission, membership,
29/// completion, publication or recorder provenance. Measurements do not impose
30/// resource limits or poll cancellation. Nested stage guards must be dropped in
31/// stack order on their creating thread across every clone. Overlapping stage
32/// scopes from different threads, transferred live stage guards, invalid
33/// nesting or poisoned bookkeeping make attribution unavailable without
34/// changing the application result. Independent phase durations sum attempted
35/// intervals and can overlap across threads.
36#[derive(Clone)]
37pub struct SolveMeasurements {
38    recorder: Arc<Recorder>,
39}
40
41impl SolveMeasurements {
42    /// Create one fixed-size recorder with optional detailed host measurements.
43    /// `true` enables stages, phase attempts and frontend work attribution;
44    /// `false` disables all measurement clocks and returns no snapshots.
45    #[must_use]
46    pub fn new(enabled: bool) -> Self {
47        Self {
48            recorder: Arc::new(Recorder::new(enabled)),
49        }
50    }
51
52    /// Record exclusive host stages without detailed phase or grounding work.
53    ///
54    /// Snapshots retain their stage intervals and driver elapsed time, while
55    /// every detailed phase and grounding measurement remains absent. Passing
56    /// this scope to a session disables its optional detailed statistics.
57    /// Admission and observation phase guards still enter their coarse stages;
58    /// other phase guards read no clocks. Each scope has independent ownership.
59    #[must_use]
60    pub fn stages_only() -> Self {
61        Self {
62            recorder: Arc::new(Recorder::stages_only()),
63        }
64    }
65
66    /// Whether this scope records host-clock measurements.
67    /// This inspection reads no clock.
68    #[must_use]
69    pub fn is_enabled(&self) -> bool {
70        self.recorder().enabled()
71    }
72
73    /// Whether detailed phase clocks and frontend work attribution are enabled.
74    /// A stages-only scope returns `false`. This inspection reads no clock.
75    #[must_use]
76    pub fn details_enabled(&self) -> bool {
77        self.recorder().details_enabled()
78    }
79
80    /// Enter one phase attempt, recording its duration when the guard is dropped.
81    ///
82    /// Phase intervals are nonexclusive. Admission and observation phases also
83    /// enter their corresponding exclusive host stage, including in a stages-only
84    /// scope. Other phase guards read no clock when detailed timing is disabled.
85    /// Keep those stage guards in stack order; overlapping phase intervals
86    /// cannot be added as a total.
87    pub fn enter(&self, phase: SolvePhase) -> MeasurementSpan<'_> {
88        MeasurementSpan {
89            _span: self.recorder().start(phase),
90        }
91    }
92
93    /// Measure an attempted operation and return its value unchanged.
94    /// Returned errors and unwinding still close the phase attempt.
95    pub fn measure<T>(&self, phase: SolvePhase, action: impl FnOnce() -> T) -> T {
96        let _span = self.enter(phase);
97        action()
98    }
99
100    /// Enter an exclusive host stage, suspending its parent until guard drop.
101    /// Enter and drop guards in stack order on their creating thread across all
102    /// clones. Cross-thread overlap or transferring a live guard makes the
103    /// exclusive partition unavailable.
104    pub fn stage(&self, stage: SolveStage) -> StageSpan<'_> {
105        self.recorder().stage(stage)
106    }
107
108    /// Borrow optional attribution for themelios's non-reentrant grounding calls.
109    ///
110    /// Pass this observer to the frontend's grounding-observer API. Each observer
111    /// owns its active callback token; phase callbacks must be paired in order
112    /// without nesting on that observer. Separate observers can record concurrent
113    /// attempts, whose durations and work are summed. The frontend's callbacks
114    /// establish measurement boundaries, not evidence that admission or
115    /// grounding succeeded.
116    /// A stages-only observer requests no detailed counters and ignores phase
117    /// callbacks, but still records the coarse grounding interval and route.
118    /// A disabled scope returns `None` without reading a clock.
119    #[must_use]
120    pub fn grounding_observer(&self) -> Option<impl zetesis_themelios::GroundingObserver + '_> {
121        self.recorder().grounding_observer()
122    }
123
124    /// Copy the currently available measurements without resetting this scope.
125    ///
126    /// Active exclusive stages contribute their elapsed prefix. Phase and
127    /// grounding measurements include only closed attempts. The returned value
128    /// can be inspected or edited without changing the shared recorder.
129    /// A stages-only snapshot has no detailed phase or grounding measurements.
130    #[must_use]
131    pub fn snapshot(&self) -> Option<PhaseTimings> {
132        self.recorder().snapshot()
133    }
134
135    pub(crate) fn recorder(&self) -> &Recorder {
136        &self.recorder
137    }
138}
139
140/// Borrowed phase attempt closed on return, explicit drop or unwinding.
141/// Retain this guard for the full attempted interval. It carries no semantic
142/// evidence and cannot outlive the measurement owner it borrows.
143#[must_use = "retain the guard until the measured phase ends"]
144pub struct MeasurementSpan<'a> {
145    _span: Span<'a>,
146}
