12/// Why checked model construction stopped without publishing its interpretation.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum ModelConstructionStop {
15    /// Caller cancellation or deadline, checked before the next operation.
16    Control(Stop),
17    /// The next operation would exceed this session's cumulative allowance.
18    Work {
19        /// Proposed cumulative work, including the refused next operation.
20        observed: u128,
21        /// Inclusive session work ceiling.
22        limit: u64,
23    },
24    /// Preparing the order or a model would exceed live metadata capacity.
25    Bytes {
26        /// Proposed or observed named capacity, independently of actual peak.
27        required: u128,
28        /// Inclusive named metadata allowance.
29        limit: usize,
30    },
31}
32
33impl fmt::Display for ModelConstructionStop {
34    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
35        match self {
36            Self::Control(stop) => stop.fmt(formatter),
37            Self::Work { observed, limit } => write!(
38                formatter,
39                "model construction work {observed} exceeds allowance {limit}"
40            ),
41            Self::Bytes { required, limit } => write!(
42                formatter,
43                "model construction requires {required} bytes, allowance is {limit}"
44            ),
45        }
46    }
47}

49/// One formula session's order preparation and selected-model construction.
50/// Hybrid and terminal inputs report their base models here; original-answer
51/// qualification retains its separate receipt. This record is available without
52/// timings. No field establishes membership, output delivery or exhaustion.
53#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
54pub struct ModelConstructionStatistics {
55    /// Accepted cumulative preparation and construction operations.
56    pub work: u64,
57    /// Retained prepared-order metadata; zero if preparation did not complete.
58    pub prepared_bytes: u128,
59    /// Largest actual named metadata peak, including refused attempts.
60    /// Proposed reservations and already published models are excluded.
61    pub peak_bytes: u128,
62    /// Complete models constructed after membership, before scoring or output.
63    pub constructed: u64,
64}
65
66#[derive(Default)]
67pub(crate) struct Account {
68    statistics: ModelConstructionStatistics,
69}
70
71pub(crate) enum Failure {
72    Interrupted(ModelConstructionStop),
73    Run(SolveError),
74}
75
76impl Account {
77    pub(crate) const fn statistics(&self) -> ModelConstructionStatistics {
78        self.statistics
79    }
80
81    pub(crate) fn prepare<'a>(
82        &mut self,
83        catalog: &'a AtomCatalog,
84        config: &SolveConfig,
85        cancellation: &Cancellation,
86    ) -> Result<ModelOrder<'a>, Failure> {
87        let order = ModelOrder::prepare_with(catalog, config.max_model_bytes, || {
88            self.permit(config, cancellation)
89        })
90        .map_err(|error| self.failure(error))?;
91        self.statistics.prepared_bytes = order.retained_bytes();
92        self.statistics.peak_bytes = self
93            .statistics
94            .peak_bytes
95            .max(order.preparation_peak_bytes());
96        Ok(order)
97    }
98
99    pub(crate) fn select(
100        &mut self,
101        order: &ModelOrder<'_>,
102        positions: impl IntoIterator<Item = usize>,
103        config: &SolveConfig,
104        cancellation: &Cancellation,
105    ) -> Result<Model, Failure> {
106        // Admit the completed counter before the operation can publish a model.
107        let constructed = self
108            .statistics
109            .constructed
110            .checked_add(1)
111            .ok_or(Failure::Run(SolveError::ModelStatisticsOverflow))?;
112        let publication = order
113            .select_with(positions, config.max_model_bytes, || {
114                self.permit(config, cancellation)
115            })
116            .map_err(|error| self.failure(error))?;
117        let (model, peak) = publication.into_parts();
118        self.statistics.peak_bytes = self.statistics.peak_bytes.max(peak);
119        self.statistics.constructed = constructed;
120        Ok(model)
121    }
122
123    fn permit(
124        &mut self,
125        config: &SolveConfig,
126        cancellation: &Cancellation,
127    ) -> Result<(), ModelConstructionStop> {
128        cancellation
129            .poll()
130            .map_err(ModelConstructionStop::Control)?;
131        let observed = u128::from(self.statistics.work) + 1;
132        if observed > u128::from(config.max_model_work) {
133            return Err(ModelConstructionStop::Work {
134                observed,
135                limit: config.max_model_work,
136            });
137        }
138        // The inclusive u64 allowance has admitted this addition.
139        self.statistics.work += 1;
140        Ok(())
141    }
142
143    fn failure(&mut self, error: ModelPublicationFailure<ModelConstructionStop>) -> Failure {
144        let (failure, peak) = error.into_parts();
145        self.statistics.peak_bytes = self.statistics.peak_bytes.max(peak);
146        match failure {
147            ModelFailure::Stopped(stop) => Failure::Interrupted(stop),
148            ModelFailure::Model(ModelError::Bytes { required, limit }) => {
149                Failure::Interrupted(ModelConstructionStop::Bytes { required, limit })
150            }
151            ModelFailure::Model(error) => Failure::Run(SolveError::Model(error)),
152        }
153    }
154}
