14/// A single preparation shared by score reads and optional bound generation.
15/// Its owner and attempted work survive every pruning refusal.
16pub(crate) struct Preparation {
17    plan: Option<ObjectivePlan>,
18    work: u64,
19    refusal: Option<ObjectiveBoundError>,
20    choice_refusal: Option<ObjectiveBoundError>,
21}
22
23impl Preparation {
24    pub(crate) fn new(
25        input: Input<'_>,
26        options: &SolveConfig,
27        cancellation: &Cancellation,
28    ) -> Self {
29        let mut state = Self {
30            plan: None,
31            work: 0,
32            refusal: None,
33            choice_refusal: None,
34        };
35        if !input.objectives.is_present() || options.max_objective_work == 0 {
36            return state;
37        }
38        let limits = plan_limits(options, 0);
39        match ObjectivePlan::new(
40            input.theory,
41            input.atoms.atoms(),
42            input.objectives,
43            limits,
44            cancellation,
45        ) {
46            Ok(plan) => {
47                state.work = plan.statistics().work;
48                state.plan = Some(plan);
49            }
50            Err(error) => {
51                state.work = error.statistics().work;
52                state.refusal = Some(error);
53            }
54        }
55        state
56    }
57
58    /// Optional necessary bounds supplement the exact plan. A refusal retains
59    /// that plan and charges the attempted work before any diagnostic is sent.
60    pub(crate) fn prepare_choices(
61        &mut self,
62        choices: Option<&zetesis_themelios::RequiredChoices>,
63        options: &SolveConfig,
64        cancellation: &Cancellation,
65    ) {
66        if options.max_objective_bound_work == 0 {
67            return;
68        }
69        let (Some(plan), Some(choices)) = (self.plan.as_mut(), choices) else {
70            return;
71        };
72        let result = plan.prepare_choice_bounds(
73            choices,
74            plan_limits(options, self.work),
75            options.max_candidate_bytes as u128,
76            cancellation,
77        );
78        match result {
79            Ok(statistics) => self.work += statistics.work,
80            Err(error) => {
81                self.work += error.statistics().work;
82                self.choice_refusal = Some(error);
83            }
84        }
85    }
86
87    pub(crate) const fn work(&self) -> u64 {
88        self.work
89    }
90    pub(crate) fn plan(&self) -> Option<&ObjectivePlan> {
91        self.plan.as_ref()
92    }
93
94    /// Observe only after the session owns and accounts for this preparation.
95    pub(crate) fn observe(&self, observations: &mut impl ExecutionSink) -> Result<(), SolveError> {
96        if let Some(error) = self.refusal {
97            observations.record(Event::ObjectiveUnavailable(error))?;
98        }
99        if let Some(error) = self.choice_refusal {
100            observations.record(Event::ChoiceObjectiveBoundUnavailable(error))?;
101        }
102        Ok(())
103    }
104}
105
106fn plan_limits(options: &SolveConfig, spent: u64) -> ObjectivePlanLimits {
107    let defaults = ObjectivePlanLimits::default();
108    ObjectivePlanLimits {
109        max_work: options
110            .max_objective_work
111            .saturating_sub(spent)
112            .min(defaults.max_work),
113        max_bindings: options.max_objective_bindings,
114        max_keys: options.max_objective_keys,
115        max_key_bytes: options.max_objective_key_bytes,
116        ..defaults
117    }
118}

120#[derive(Default)]
121pub(crate) struct Bounds {
122    enabled: bool,
123    work: u64,
124}
125
126impl Bounds {
127    pub(crate) fn new(options: &SolveConfig) -> Self {
128        Self {
129            enabled: options.max_objective_bound_work != 0,
130            work: 0,
131        }
132    }
133
134    /// Call only after a strictly better, fully retained reduct-verified model.
135    /// Bound generation borrows the same immutable plan used by score reads.
136    pub(crate) fn improve(
137        &mut self,
138        plan: Option<&ObjectivePlan>,
139        score: &Score,
140        models: &mut StableModels,
141        options: &SolveConfig,
142        observations: &mut impl ExecutionSink,
143        cancellation: &Cancellation,
144    ) -> Result<(), SolveError> {
145        let Some(plan) = plan.filter(|_| self.enabled) else {
146            return Ok(());
147        };
148        let limits = ObjectiveBoundLimits {
149            max_work: options.max_objective_bound_work.saturating_sub(self.work),
150            ..Default::default()
151        };
152        let bound = match plan.bound(score, limits, cancellation) {
153            Ok(bound) => {
154                self.work += bound.statistics().work;
155                bound
156            }
157            Err(error) => {
158                self.work += error.statistics().work;
159                self.enabled = false;
160                observations.record(Event::ObjectiveBoundStopped(error))?;
161                return Ok(());
162            }
163        };
164        if let Some(error) = bound.choice_failure() {
165            observations.record(Event::ChoiceObjectiveBoundUnavailable(error))?;
166        }
167        if !bound.original().same_instance(models.theory()) {
168            self.enabled = false;
169            observations.record(Event::ObjectiveTheoryMismatch)?;
170            return Ok(());
171        }
172        if let Err(error) = models.tighten_candidate_bound(bound.theory()) {
173            // Replacement is transactional: the older bound remains on failure.
174            // Later scoring still borrows the unchanged immutable preparation.
175            self.enabled = false;
176            observations.record(Event::ObjectiveRestrictionStopped(error))?;
177        } else {
178            observations.record(Event::ObjectiveBound {
179                restrictions: models.statistics().candidate_restrictions,
180                costs: score.costs(),
181                work: self.work,
182            })?;
183        }
184        Ok(())
185    }
186}
