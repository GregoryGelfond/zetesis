11/// Independent ceilings for distinct projected answer identities.
12/// These do not replace candidate exclusion, membership or objective budgets.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub struct ProjectionLimits {
15    /// Maximum distinct projection keys; duplicate answers require no new key.
16    pub max_keys: usize,
17    /// History header, index capacity and packed key words, including conservative
18    /// old/new growth overlap and the reusable current-key buffer. Excludes the
19    /// borrowed domain, answer, allocator metadata and search; this is not RSS.
20    pub max_bytes: usize,
21    /// Cumulative key-word resets, domain probes, hashing/comparison/copy words,
22    /// and index initialization/probe steps, including failed attempts. A domain
23    /// probe includes the model's typed lookup, not one machine instruction.
24    /// Hash collisions consume this same finite allowance.
25    pub max_work: u64,
26}
27
28impl Default for ProjectionLimits {
29    fn default() -> Self {
30        Self {
31            max_keys: 1_000_000,
32            max_bytes: 128 * 1024 * 1024,
33            max_work: 100_000_000,
34        }
35    }
36}

38/// A projected-enumeration limit, independent of original answer identity.
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40pub enum ProjectionResource {
41    /// Distinct completed keys.
42    Keys,
43    /// Named history and construction capacity.
44    Bytes,
45    /// Cumulative key construction and lookup work.
46    Work,
47}
48
49/// Projection failed without publishing a partial or unexamined key.
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub enum ProjectionError {
52    /// A projected door requires completed source projection declarations.
53    MissingDeclaration,
54    /// An inclusive projection ceiling was exceeded.
55    Limit {
56        /// Resource whose admission failed.
57        resource: ProjectionResource,
58        /// Required amount, before publication.
59        observed: u128,
60        /// Inclusive configured ceiling.
61        limit: u128,
62    },
63    /// Checked capacity or counter arithmetic overflowed.
64    Overflow,
65    /// A fallible history reservation failed.
66    Allocation,
67    /// Cooperative cancellation or deadline reached during projection.
68    Control(Stop),
69}
70
71impl fmt::Display for ProjectionError {
72    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
73        match self {
74            Self::MissingDeclaration => {
75                f.write_str("projected enumeration requires a completed #project declaration")
76            }
77            Self::Limit {
78                resource,
79                observed,
80                limit,
81            } => write!(
82                f,
83                "answer projection {resource:?} limit: {observed} exceeds {limit}"
84            ),
85            Self::Overflow => f.write_str("answer projection accounting overflow"),
86            Self::Allocation => f.write_str("answer projection storage could not be reserved"),
87            Self::Control(stop) => stop.fmt(f),
88        }
89    }
90}

94/// Evidence from projected enumeration, separate from full-model search counts.
95#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
96pub struct ProjectionStatistics {
97    /// Distinct keys retained and returned as full answer-set representatives.
98    pub representatives: usize,
99    /// Fully keyed answers equal to a previously returned representative.
100    pub duplicates: u64,
101    /// Charged work, including attempts that failed before key publication.
102    pub work: u64,
103    /// Current named retained history capacity, excluding borrowed owners.
104    pub retained_bytes: u128,
105    /// Greatest observed named capacity, including conservative old/new growth
106    /// overlap. Refused proposals that allocated nothing do not increase it.
107    pub peak_bytes: u128,
108    /// Every selected answer was consumed and original search was exhaustive.
109    /// Requested representative limits and failures leave this false even when
110    /// optimization independently established the best score.
111    pub complete: bool,
112}

114pub(crate) struct Projection<'a> {
115    domain: &'a PreparedProjection,
116    limits: ProjectionLimits,
117    history: index::History,
118    current: Vec<u64>,
119    stats: ProjectionStatistics,
120}
121
122impl<'a> Projection<'a> {
123    pub(crate) fn new(
124        domain: &'a PreparedProjection,
125        limits: ProjectionLimits,
126    ) -> Result<Self, ProjectionError> {
127        if !domain.is_explicit() {
128            return Err(ProjectionError::MissingDeclaration);
129        }
130        let words = domain.atoms().len().div_ceil(64);
131        let mut result = Self {
132            domain,
133            limits,
134            history: index::History::new(words),
135            current: Vec::new(),
136            stats: ProjectionStatistics::default(),
137        };
138        ceiling(
139            ProjectionResource::Bytes,
140            result.base_bytes() + bytes::<u64>(words)?,
141            limits.max_bytes as u128,
142        )?;
143        result
144            .current
145            .try_reserve_exact(words)
146            .map_err(|_| ProjectionError::Allocation)?;
147        result.stats.retained_bytes = result.base_bytes();
148        result.stats.peak_bytes = result.stats.retained_bytes;
149        ceiling(
150            ProjectionResource::Bytes,
151            result.stats.retained_bytes,
152            limits.max_bytes as u128,
153        )?;
154        result.current.resize(words, 0);
155        Ok(result)
156    }
157
158    pub(crate) fn statistics(&self) -> ProjectionStatistics {
159        self.stats
160    }
161
162    pub(crate) fn finish(&mut self, complete: bool) {
163        self.stats.complete = complete;
164    }
165
166    /// Complete the key before indexing it. History has one packed copy per key;
167    /// hash slots contain only its local index. Neither hash nor local index is
168    /// an answer identity, and collision equality always compares every word.
169    pub(crate) fn insert(
170        &mut self,
171        model: &Model,
172        cancellation: &Cancellation,
173    ) -> Result<bool, ProjectionError> {
174        let base = self.base_bytes();
175        let mut work = Work {
176            limits: self.limits,
177            stats: &mut self.stats,
178            cancellation,
179        };
180        for word in &mut self.current {
181            work.charge(1)?;
182            *word = 0;
183        }
184        for (index, atom) in self.domain.atoms().iter().enumerate() {
185            work.charge(1)?;
186            if model.contains(atom) {
187                self.current[index / 64] |= 1_u64 << (index % 64);
188            }
189        }
190        let inserted = self.history.insert(&self.current, base, &mut work)?;
191        if inserted {
192            work.stats.representatives = self.history.len();
193        } else {
194            work.stats.duplicates = work
195                .stats
196                .duplicates
197                .checked_add(1)
198                .ok_or(ProjectionError::Overflow)?;
199        }
200        Ok(inserted)
201    }
202
203    // All vector headers, including History's, are already within Self.
204    fn base_bytes(&self) -> u128 {
205        size_of::<Self>() as u128 + self.current.capacity() as u128 * size_of::<u64>() as u128
206    }
207}

209struct Work<'a> {
210    limits: ProjectionLimits,
211    stats: &'a mut ProjectionStatistics,
212    cancellation: &'a Cancellation,
213}
214
215impl Work<'_> {
216    fn charge(&mut self, amount: u64) -> Result<(), ProjectionError> {
217        self.cancellation.poll().map_err(ProjectionError::Control)?;
218        let observed = self
219            .stats
220            .work
221            .checked_add(amount)
222            .ok_or(ProjectionError::Overflow)?;
223        ceiling(
224            ProjectionResource::Work,
225            u128::from(observed),
226            u128::from(self.limits.max_work),
227        )?;
228        self.stats.work = observed;
229        Ok(())
230    }
231
232    fn admit_bytes(&self, required: u128) -> Result<(), ProjectionError> {
233        ceiling(
234            ProjectionResource::Bytes,
235            required,
236            self.limits.max_bytes as u128,
237        )
238    }
239
240    /// Observe actual retained capacity plus a currently allocated replacement.
241    /// A refused proposal calls `admit_bytes` alone. Allocator slack is observed
242    /// before refusal; failed temporary construction drops that replacement.
243    fn observe(&mut self, retained: u128, temporary: u128) -> Result<(), ProjectionError> {
244        self.stats.retained_bytes = retained;
245        let live = retained
246            .checked_add(temporary)
247            .ok_or(ProjectionError::Overflow)?;
248        self.stats.peak_bytes = self.stats.peak_bytes.max(live);
249        self.admit_bytes(live)
250    }
251}
252
253fn bytes<T>(count: usize) -> Result<u128, ProjectionError> {
254    (count as u128)
255        .checked_mul(size_of::<T>() as u128)
256        .ok_or(ProjectionError::Overflow)
257}
258
259fn ceiling(
260    resource: ProjectionResource,
261    observed: u128,
262    limit: u128,
263) -> Result<(), ProjectionError> {
264    if observed > limit {
265        Err(ProjectionError::Limit {
266            resource,
267            observed,
268            limit,
269        })
270    } else {
271        Ok(())
272    }
