1//! Append-only packed identities with a bounded, indices-only hash table.
2//!
3//! Slots are empty or the positive successor of a key's insertion position.
4//! The table stays at most half full. Hashing chooses a starting slot only:
5//! every occupied candidate is compared against the authoritative packed key.
6//! Capacity grows geometrically; collision chains have an explicit work bound,
7//! not a worst-case constant-time promise. Hash values are not serialized.
8//!
9//! Growth copies into a fallibly reserved replacement before swapping buffers.
10//! A stopped copy/rehash preserves every previous key and slot. New key words
11//! remain unpublished until complete, and are truncated on any failure. Only
12//! equivalent capacity changes may survive a failed insertion.
13
14use std::hash::{DefaultHasher, Hasher};
15use std::num::NonZeroUsize;
16
17use super::{ProjectionError, ProjectionResource, Work, bytes, ceiling};
18
19type Slot = Option<NonZeroUsize>;
20
21pub(super) struct History {
22    width: usize,
23    entries: usize,
24    keys: Vec<u64>,
25    slots: Vec<Slot>,
26}
27
28impl History {
29    pub(super) const fn new(width: usize) -> Self {
30        Self {
31            width,
32            entries: 0,
33            keys: Vec::new(),
34            slots: Vec::new(),
35        }
36    }
37
38    pub(super) const fn len(&self) -> usize {
39        self.entries
40    }
41
42    fn key(&self, index: usize) -> &[u64] {
43        let start = index * self.width;
44        &self.keys[start..start + self.width]
45    }
46
47    fn retained(&self, base: u128) -> Result<u128, ProjectionError> {
48        base.checked_add(bytes::<u64>(self.keys.capacity())?)
49            .and_then(|sum| {
50                sum.checked_add(self.slots.capacity() as u128 * size_of::<Slot>() as u128)
51            })
52            .ok_or(ProjectionError::Overflow)
53    }
54
55    /// Duplicate lookup precedes entry admission, including the empty key.
56    pub(super) fn insert(
57        &mut self,
58        key: &[u64],
59        base: u128,
60        work: &mut Work<'_>,
61    ) -> Result<bool, ProjectionError> {
62        assert_eq!(key.len(), self.width, "fixed projection domain");
63        work.charge(1)?;
64        if self.width == 0 && self.entries != 0 {
65            return Ok(false);
66        }
67        let hash = hash(key, work)?;
68        if !self.slots.is_empty() && self.contains(key, hash, work)? {
69            return Ok(false);
70        }
71        let count = self
72            .entries
73            .checked_add(1)
74            .ok_or(ProjectionError::Overflow)?;
75        ceiling(
76            ProjectionResource::Keys,
77            count as u128,
78            work.limits.max_keys as u128,
79        )?;
80        if self.width == 0 {
81            self.entries = count;
82            return Ok(true);
83        }
84        let words = count
85            .checked_mul(self.width)
86            .ok_or(ProjectionError::Overflow)?;
87        self.reserve_keys(words, base, work)?;
88        self.reserve_slots(count, base, work)?;
89        let slot = empty_slot(&self.slots, hash, work)?;
90        let start = self.keys.len();
91        let result = self.append(key, work);
92        if let Err(error) = result {
93            self.keys.truncate(start);
94            return Err(error);
95        }
96        // Both dimensions and the positive successor were admitted above.
97        // No callback, allocation or fallible operation follows publication.
98        self.slots[slot] = NonZeroUsize::new(count);
99        self.entries = count;
100        Ok(true)
101    }
102
103    fn append(&mut self, key: &[u64], work: &mut Work<'_>) -> Result<(), ProjectionError> {
104        for word in key {
105            work.charge(1)?;
106            self.keys.push(*word);
107        }
108        work.charge(1)
109    }
110
111    fn contains(
112        &self,
113        key: &[u64],
114        hash: u64,
115        work: &mut Work<'_>,
116    ) -> Result<bool, ProjectionError> {
117        let mut slot = first_slot(hash, self.slots.len())?;
118        loop {
119            work.charge(1)?;
120            let Some(index) = self.slots[slot] else {
121                return Ok(false);
122            };
123            if equal(self.key(index.get() - 1), key, work)? {
124                return Ok(true);
125            }
126            slot = (slot + 1) & (self.slots.len() - 1);
127        }
128    }
129
130    fn reserve_keys(
131        &mut self,
132        required: usize,
133        base: u128,
134        work: &mut Work<'_>,
135    ) -> Result<(), ProjectionError> {
136        if required <= self.keys.capacity() {
137            return Ok(());
138        }
139        let capacity = capacity(required)?;
140        let retained = self.retained(base)?;
141        work.admit_bytes(
142            retained
143                .checked_add(bytes::<u64>(capacity)?)
144                .ok_or(ProjectionError::Overflow)?,
145        )?;
146        let mut replacement = Vec::new();
147        replacement
148            .try_reserve_exact(capacity)
149            .map_err(|_| ProjectionError::Allocation)?;
150        work.observe(retained, bytes::<u64>(replacement.capacity())?)?;
151        for word in &self.keys {
152            work.charge(1)?;
153            replacement.push(*word);
154        }
155        self.keys = replacement;
156        work.observe(self.retained(base)?, 0)
157    }
158
159    fn reserve_slots(
160        &mut self,
161        count: usize,
162        base: u128,
163        work: &mut Work<'_>,
164    ) -> Result<(), ProjectionError> {
165        let required = count.checked_mul(2).ok_or(ProjectionError::Overflow)?;
166        if required <= self.slots.len() {
167            return Ok(());
168        }
169        let capacity = capacity(required)?;
170        let retained = self.retained(base)?;
171        work.admit_bytes(
172            retained
173                .checked_add(bytes::<Slot>(capacity)?)
174                .ok_or(ProjectionError::Overflow)?,
175        )?;
176        let mut replacement = Vec::new();
177        replacement
178            .try_reserve_exact(capacity)
179            .map_err(|_| ProjectionError::Allocation)?;
180        work.observe(retained, bytes::<Slot>(replacement.capacity())?)?;
181        for _ in 0..capacity {
182            work.charge(1)?;
183            replacement.push(None);
184        }
185        for index in 0..self.entries {
186            let hash = hash(self.key(index), work)?;
187            let slot = empty_slot(&replacement, hash, work)?;
188            replacement[slot] = NonZeroUsize::new(index + 1);
189        }
190        self.slots = replacement;
191        work.observe(self.retained(base)?, 0)
192    }
193}
194
195fn capacity(required: usize) -> Result<usize, ProjectionError> {
196    required
197        .max(4)
198        .checked_next_power_of_two()
199        .ok_or(ProjectionError::Overflow)
200}
201
202fn hash(key: &[u64], work: &mut Work<'_>) -> Result<u64, ProjectionError> {
203    // This standard-library hasher is deterministic within the build. Neither
204    // its algorithm nor its result is a persistent identity or public format.
205    let mut hasher = DefaultHasher::new();
206    for word in key {
207        work.charge(1)?;
208        hasher.write_u64(*word);
209    }
210    Ok(hasher.finish())
211}
212
213fn first_slot(hash: u64, length: usize) -> Result<usize, ProjectionError> {
214    let mask = u64::try_from(length - 1).map_err(|_| ProjectionError::Overflow)?;
215    usize::try_from(hash & mask).map_err(|_| ProjectionError::Overflow)
216}
217
218fn empty_slot(slots: &[Slot], hash: u64, work: &mut Work<'_>) -> Result<usize, ProjectionError> {
219    let mut slot = first_slot(hash, slots.len())?;
220    loop {
221        work.charge(1)?;
222        if slots[slot].is_none() {
223            return Ok(slot);
224        }
225        slot = (slot + 1) & (slots.len() - 1);
226    }
227}
228
229fn equal(left: &[u64], right: &[u64], work: &mut Work<'_>) -> Result<bool, ProjectionError> {
230    for (left, right) in left.iter().zip(right) {
231        work.charge(1)?;
232        if left != right {
233            return Ok(false);
234        }
235    }
236    Ok(true)
237}
