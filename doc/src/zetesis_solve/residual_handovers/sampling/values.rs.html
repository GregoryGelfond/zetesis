1//! Whole-population reductions and terminal sample ranks.
2
3use super::{ResidualQuantiles, ResidualSampleMetric};

5/// Whether the member belongs to this metric, independently of whether its
6/// numerical value was successfully observed.
7#[derive(Clone, Copy, Debug)]
8pub(super) enum Value {
9    Outside,
10    UnknownEligibility,
11    Missing,
12    Known(u128),
13}
14
15impl Value {
16    pub(super) fn optional<T: Into<u128>>(value: Option<T>) -> Self {
17        value.map_or(Self::Missing, |v| Self::Known(v.into()))
18    }
19
20    pub(super) fn measured(self) -> Option<u128> {
21        if let Self::Known(v) = self {
22            Some(v)
23        } else {
24            None
25        }
26    }
27}
28
29#[derive(Clone, Copy)]
30pub(super) struct Totals {
31    eligible: Option<u64>,
32    known: Option<u64>,
33    sum: Option<u128>,
34    maximum: Option<u128>,
35}
36
37impl Default for Totals {
38    fn default() -> Self {
39        Self {
40            eligible: Some(0),
41            known: Some(0),
42            sum: Some(0),
43            maximum: None,
44        }
45    }
46}
47
48impl Totals {
49    pub(super) fn observe(&mut self, value: Value) {
50        match value {
51            Value::Outside => return,
52            Value::UnknownEligibility => {
53                self.eligible = None;
54                return;
55            }
56            Value::Missing => {}
57            Value::Known(v) => {
58                self.known = self.known.and_then(|n| n.checked_add(1));
59                self.sum = self.sum.and_then(|sum| sum.checked_add(v));
60                self.maximum = self.maximum.max(Some(v));
61            }
62        }
63        self.eligible = self.eligible.and_then(|n| n.checked_add(1));
64    }
65
66    /// Selection mutates the terminal sample's order only. Each comparison
67    /// projects a single field of an existing record. No scratch allocation
68    /// is needed; the retained set stays identical for subsequent metrics.
69    pub(super) fn finish<T>(
70        &self,
71        records: &mut [T],
72        value: impl Fn(&T) -> Option<u128>,
73    ) -> ResidualSampleMetric {
74        let retained = if self.known == Some(0) {
75            0
76        } else {
77            records.iter().filter(|r| value(r).is_some()).count()
78        };
79        let complete = self.known.is_some() && self.eligible == self.known;
80        let mut metric = ResidualSampleMetric {
81            eligible: self.eligible,
82            known: self.known,
83            retained,
84            sum: complete.then_some(self.sum).flatten(),
85            maximum: complete.then_some(self.maximum).flatten(),
86            ..ResidualSampleMetric::default()
87        };
88        if complete && retained != 0 {
89            let missing = records.len() - retained;
90            let mut rank = |index: usize| {
91                let (_, selected, _) = records.select_nth_unstable_by_key(missing + index, &value);
92                value(selected)
93            };
94            let middle = rank((retained - 1) / 2);
95            metric.median = if retained.is_multiple_of(2) {
96                middle
97                    .zip(rank(retained / 2))
98                    .map(|(low, high)| low.midpoint(high))
99            } else {
100                middle
101            };
102            // ceil(9 n / 10) - 1, without multiplying a sample size.
103            metric.p90 = rank(retained - retained / 10 - 1);
104            metric.quantiles = if self.known == Some(retained as u64) {
105                ResidualQuantiles::Exact
106            } else {
107                ResidualQuantiles::Estimated
108            };
109        }
110        metric
111    }
112}
113
114impl Totals {
115    pub(super) fn observed_maximum(&self) -> Option<u128> {
116        self.maximum
117    }
118
119    pub(super) fn missing_members(&mut self, count: Option<u64>) {
120        self.eligible = self.eligible.and_then(|n| n.checked_add(count?));
121    }
122}
