1//! Ordinary execution resources shared by source, session and output consumers.
2//!
3//! Work, visited substitutions, rounds and copied-byte traffic are receipts,
4//! bounded only by their counter representation. Cancellation and an optional
5//! deadline remain the control boundary. Explicit low-level limit types still
6//! support finite operation allowances. Optional analysis and device dispatch
7//! retain bounded effort. Optional analyses fail open; mandatory device setup
8//! can still refuse a dispatch that exceeds its internal capacity.
9//!
10//! Memory supplies conservative, independent named capacity shares: support and
11//! formula storage use quarters, canonical/source owners eighths, and ancillary
12//! metadata/output owners sixteenths or smaller. Population limits use logical
13//! cell envelopes within those shares. These owners can overlap; their sum is
14//! not the allowance, and neither the allowance nor these counts guarantee RSS.
15//! Allocator overhead, source-library allocations and thread stacks remain outside
16//! the named accounting. Existing checked depth limits protect recursive ingress.
17

28/// Ordinary named-memory allowance and CPU worker capacity.
29///
30/// No selected work ceiling is stored here. Pass the same cancellation token
31/// through preparation and the session to impose an optional deadline.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub struct Resources {
34    memory: usize,
35    workers: NonZeroUsize,
36}
37
38impl Resources {
39    /// Reference memory for ordinary library defaults: two gibibytes.
40    pub const REFERENCE_MEMORY: u64 = 2 * 1024 * 1024 * 1024;
41    /// Reference policy with four CPU workers. Frontends can select the host's
42    /// parallelism and construct this same policy with [`Self::new`].
43    pub const DEFAULT: Self = Self {
44        memory: if usize::BITS > 32 {
45            2 * 1024 * 1024 * 1024
46        } else {
47            isize::MAX as usize
48        },
49        workers: NonZeroUsize::new(4).unwrap(),
50    };
51
52    /// Derive every ordinary capacity from one allowance. Clamp to the maximum
53    /// signed allocation extent supported by this host; zero remains zero.
54    #[must_use]
55    pub fn new(memory: u64, workers: NonZeroUsize) -> Self {
56        Self {
57            memory: usize::try_from(memory)
58                .unwrap_or(usize::MAX)
59                .min(isize::MAX as usize),
60            workers,
61        }
62    }
63
64    const fn bytes(self, denominator: usize) -> usize {
65        self.memory / denominator
66    }
67
68    const fn cells(self, denominator: usize, bytes: usize) -> usize {
69        self.bytes(denominator) / bytes
70    }
71
72    fn cells32(self, denominator: usize, bytes: usize) -> u32 {
73        u32::try_from(self.cells(denominator, bytes)).unwrap_or(u32::MAX)
74    }
