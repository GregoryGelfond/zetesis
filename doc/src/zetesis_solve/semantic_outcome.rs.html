1//! Established semantic evidence, independent of external publication.

3use crate::{Completion, Interruption, Optimization, SearchState};

5/// The answer family requested by a semantic session, independently of whether
6/// it was completely searched or delivered. Optional projected enumeration
7/// selects representatives only after establishing this original family choice.
8#[derive(Clone, Copy, Debug, PartialEq, Eq)]
9pub enum AnswerSelection {
10    /// All full answer sets of the original program, regardless of objectives.
11    All,
12    /// Objective-selected answers. Interrupted incumbents need not be optimal;
13    /// even a proved optimum need not have every tie retained or yielded.
14    Optimal,
15}
16
17/// Immutable semantic accounting when a session stops or finishes its search.
18/// The driver retains this evidence as its semantic authority; compatibility
19/// reports derive from it together with separate publication and timing state.
20/// A verified model need not have been scored, retained, yielded or published.
21/// Neither missing coverage nor a zero publication count establishes UNSAT.
22#[derive(Clone, Debug)]
23pub struct SemanticOutcome {
24    pub(crate) projection: Option<crate::ProjectionStatistics>,
25    pub(crate) subject: Option<crate::Subject>,
26    pub(crate) selection: Option<AnswerSelection>,
27    pub(crate) verified: u64,
28    pub(crate) scored: u64,
29    pub(crate) retained: usize,
30    pub(crate) search_state: Option<SearchState>,
31    pub(crate) optimization: Option<Optimization>,
32    pub(crate) objective_work: u64,
33    pub(crate) checked: u64,
34    pub(crate) gate_atoms: usize,
35    pub(crate) candidate_statistics: Option<zetesis_cpu::CandidateStatistics>,
36    pub(crate) countermodel_statistics: Option<zetesis_sat::Statistics>,
37    pub(crate) formula_execution: Option<crate::FormulaExecutionStatistics>,
38    pub(crate) lazy_execution: Option<crate::LazyExecutionStatistics>,
39    pub(crate) shared_execution: Option<crate::SharedExecutionStatistics>,
40    pub(crate) closure_execution: Option<crate::ClosureExecutionStatistics>,
41    pub(crate) query_execution: Option<crate::QueryExecutionObservation>,
42    pub(crate) hybrid_execution: Option<crate::HybridExecutionStatistics>,
43    pub(crate) terminal_execution: Option<crate::TerminalExecutionStatistics>,
44    pub(crate) source_evaluation: Option<zetesis_themelios::SourceEvaluationStatistics>,
45    pub(crate) model_construction: Option<crate::ModelConstructionStatistics>,
46}

192    /// Prepared independent CPU ownership receipts, including a snapshot fault

198    /// Completed exact stable-model memberships of the original subject,
199    /// including still-queued results. A terminal-definition input counts only
200    /// completed full reconstruction, never queued or consumed base answers.
201    #[must_use]
202    pub const fn verified_models(&self) -> u64 {
203        self.verified
204    }
205
206    /// Exact size of the requested full answer-set family after exhaustion.
207    /// Optimal selection counts every optimal tie, including unretained ties;
208    /// all-answer selection counts full identities, even with projection.
209    /// Incomplete or absent coverage returns None, never a partial total.
210    /// Engine counters are checked or bounded by checked candidate counts.
211    #[must_use]
212    pub const fn complete_answer_count(&self) -> Option<u64> {
213        if !matches!(self.search_state, Some(SearchState::Exhausted)) {
214            return None;
215        }
216        match self.selection {
217            Some(AnswerSelection::All) => Some(self.verified),
218            Some(AnswerSelection::Optimal) => match &self.optimization {
219                Some(best) => Some(best.tied_models),
220                None if self.verified == 0 => Some(0),
221                None => None,
222            },
223            None => None,
224        }
225    }
226
227    /// Verified models whose objective evaluation completed.
228    #[must_use]
229    pub const fn scored_models(&self) -> u64 {
230        self.scored
231    }
232
233    /// Cumulative accepted objective preparation and scoring work, including
234    /// refused prefixes. Available before any incumbent and in unrestricted
235    /// enumeration; zero when no objective work has been accepted. Optional
236    /// candidate-bound construction uses a separate account.
237    #[must_use]
238    pub const fn objective_work(&self) -> u64 {
239        self.objective_work
240    }
241
242    /// Incumbent models retained when search stopped, before any delivery.

246        self.retained
247    }
248
249    /// Established search status, including a stop pending behind checked answers.
250    /// Absent when no stopping classification has been established.
251    #[must_use]
252    pub const fn search_state(&self) -> Option<SearchState> {
253        self.search_state
254    }
255
256    /// Established search coverage; absent after an unrelated failure or when the
257    /// consumer stops pulling before the engine establishes a stopping classification.
258    #[must_use]
259    pub const fn completion(&self) -> Option<Completion> {
260        match self.search_state {
261            Some(state) => state.completion(),
262            None => None,
263        }
264    }
265
266    /// Established logical stop, independent of a later delivery failure.
267    #[must_use]
268    pub const fn interruption(&self) -> Option<Interruption> {
269        match self.search_state {
270            Some(state) => state.interruption(),
271            None => None,
272        }
273    }
274
275    /// Retained objective evidence; a score alone does not establish an optimum.
276    #[must_use]
277    pub const fn incumbent(&self) -> Option<&Optimization> {
278        self.optimization.as_ref()
279    }
280
281    /// Complete relevant search established the retained incumbent's optimum.
282    #[must_use]
283    pub const fn optimum_proved(&self) -> bool {
284        matches!(self.selection, Some(AnswerSelection::Optimal))
285            && matches!(self.search_state, Some(SearchState::Exhausted))
286            && self.optimization.is_some()
287    }
288
289    /// Complete search found no stable model, independently of output delivery.
290    #[must_use]
291    pub const fn unsatisfiable(&self) -> bool {
292        matches!(self.search_state, Some(SearchState::Exhausted)) && self.verified == 0
293    }

296#[cfg(test)]
297mod count_tests {
298    use super::*;
299
300    #[test]
301    fn complete_count_needs_exhaustion_and_a_selected_family() {
302        let mut outcome = SemanticOutcome::interrupted_before_start(Interruption::Preparation(
303            zetesis_cpu::Stop::Cancelled,
304        ));
305        outcome.verified = 7;
306        outcome.selection = Some(AnswerSelection::All);
307        assert_eq!(outcome.complete_answer_count(), None);
308        outcome.search_state = Some(SearchState::RequestedModels);
309        assert_eq!(outcome.complete_answer_count(), None);
310        outcome.search_state = Some(SearchState::Exhausted);
311        assert_eq!(outcome.complete_answer_count(), Some(7));
312        outcome.selection = None;
313        assert_eq!(outcome.complete_answer_count(), None);
314    }
315
316    #[test]
317    fn an_absent_incumbent_only_counts_as_zero_for_empty_complete_search() {
318        let mut outcome = SemanticOutcome::interrupted_before_start(Interruption::Preparation(
319            zetesis_cpu::Stop::Cancelled,
320        ));
321        outcome.selection = Some(AnswerSelection::Optimal);
322        outcome.search_state = Some(SearchState::Exhausted);
323        assert_eq!(outcome.complete_answer_count(), Some(0));
324        outcome.verified = 1;
325        assert_eq!(outcome.complete_answer_count(), None);
326    }
