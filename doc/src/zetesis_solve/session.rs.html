1//! Writer-free ordinary solves over coherent, already admitted inputs.
2//!
3//! These sessions use the same retained engine loops as the command adapter.
4//! They neither parse source nor render model text. A caller owns admission and
5//! can reuse its immutable owner and explicit device resources across sessions;
6//! each session owns fresh search budgets, worker pools, pending results and
7//! incumbent storage.

33/// The complete semantic representation supplied to an ordinary session.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum PreparedProfile {
36    /// Finite relational normal program; closure execution chooses lazy or eager.
37    Relational,
38    /// Coherently indexed finite formulas and their objective/observation programs.
39    Formula,
40    /// Complete producer theory with admitted constraints checked from source.
41    Hybrid,
42    /// A checked base theory plus terminal definitions of the original source.
43    TerminalDefinitions,
44    /// A complete ground graph retaining its original relational program identity.
45    Ground,
46}
47
48#[derive(Clone, Copy)]
49enum Prepared<'a> {
50    Relational(&'a Program),
51    Formula(crate::countermodel::Input<'a>),
52    Hybrid(&'a zetesis_themelios::HybridFormula),
53    TerminalDefinitions(&'a zetesis_themelios::TerminalFormula),
54    Ground(&'a Arc<GroundProgram>),
55}

57/// A coherent borrowed owner, never an independently supplied theory/atom table.
58/// Constructors do no admission, grounding, or source replay. Explicit oracle
59/// requests incompatible with this representation are rejected at session setup.
60#[derive(Clone, Copy)]
61pub struct PreparedInput<'a> {
62    input: Prepared<'a>,
63    metadata: Option<&'a SourceMetadata>,
64    projection: Option<&'a zetesis_themelios::PreparedProjection>,
65}
66impl<'a> PreparedInput<'a> {
67    /// Borrow the original source plan with its checked base and terminal
68    /// definitions. Each verified base answer is reconstructed before a full
69    /// answer can be published. An eager base runs under automatic grounding;
70    /// a hybrid base (from lazy materialization) runs under automatic or lazy
71    /// grounding, on the CPU backend only, its constraints checked before
72    /// reconstruction. An eager request never acquires terminal meaning. Explicit projection and
73    /// objectives are outside this profile. No admission or execution occurs in
74    /// this borrow.
75    #[must_use]
76    pub fn terminal(owner: &'a zetesis_themelios::TerminalFormula) -> Self {
77        Self {
78            input: Prepared::TerminalDefinitions(owner),
79            metadata: Some(owner.metadata()),
80            projection: None,
81        }
82    }
83
84    /// Borrow a coherent producer theory and its admitted streamed constraints.
85    /// No grounding or solving occurs here. The CPU backend supports lazy and
86    /// automatic grounding. Objectives are scored only after complete original
87    /// constraint acceptance. Device checking is outside this hybrid profile.
88    #[must_use]
89    pub fn hybrid(owner: &'a zetesis_themelios::HybridFormula) -> Self {
90        Self {
91            input: Prepared::Hybrid(owner),
92            metadata: Some(owner.metadata()),
93            projection: Some(owner.projection()),
94        }
95    }

97    /// Reuse an admitted native relational program without source metadata.
98    /// This borrows the program directly: it performs no parsing, carrier
99    /// expansion or grounding. Lazy and eager execution use the same relational
100    /// session branch as source-admitted programs.
101    #[must_use]
102    pub const fn program(program: &'a Program) -> Self {
103        Self {
104            input: Prepared::Relational(program),
105            metadata: None,
106            projection: None,
107        }
108    }
109    /// Borrow a canonical relational preparation with its original display
110    /// metadata. The owner keeps the program and metadata together; this borrow
111    /// performs no parsing, expansion or grounding. Lazy and eager sessions use
112    /// the same admitted relational program.
113    #[must_use]
114    pub fn relational(owner: &'a zetesis_themelios::PreparedRelational) -> Self {
115        Self {
116            input: Prepared::Relational(owner.program()),
117            metadata: Some(owner.metadata()),
118            projection: None,
119        }
120    }
121    /// Reuse an admitted normal program and its source metadata.
122    #[must_use]
123    pub fn admitted(owner: &'a Admitted) -> Self {
124        Self {
125            input: Prepared::Relational(owner.program()),
126            metadata: Some(owner.metadata()),
127            projection: None,
128        }
129    }
130    /// Reuse an admitted original-source bundle without loading its files again.
131    #[must_use]
132    pub fn bundle(owner: &'a AdmittedBundle) -> Self {
133        Self {
134            input: Prepared::Relational(owner.program()),
135            metadata: Some(owner.metadata()),
136            projection: None,
137        }
138    }
139    /// Reuse a formula owner, preserving its exact atom indexing and objectives.
140    #[must_use]
141    pub fn formula(owner: &'a AdmittedFormula) -> Self {

143            input: Prepared::Formula(crate::countermodel::Input {
144                theory: owner.theory(),
145                atoms: owner.atom_catalog(),
146                objectives: owner.objectives(),
147                required_choices: owner.required_choices(),

157            projection: Some(owner.projection()),

160    /// Reuse a formula bundle with its original metadata and objective program.
161    #[must_use]
162    pub fn formula_bundle(owner: &'a AdmittedFormulaBundle) -> Self {
163        Self {
164            input: Prepared::Formula(crate::countermodel::Input {
165                theory: owner.theory(),
166                atoms: owner.atom_catalog(),
167                objectives: owner.objectives(),
168                required_choices: owner.required_choices(),
169                gate_atoms: 0,
170                keyed_constraints: owner.keyed_constraints(),
171                key_analysis: owner.key_analysis(),
172                certificate_order: crate::countermodel::certificate_order(

174                    owner.analysis_basis(),
175                ),
176            }),
177            metadata: Some(owner.metadata()),
178            projection: Some(owner.projection()),

181    /// Reuse the supplied complete graph. Auto chooses eager execution; explicit
182    /// lazy or countermodel requests are refused. No second graph is compiled.
183    /// Ground-only models have no source-derived observation metadata.
184    #[must_use]
185    pub const fn ground(owner: &'a Arc<GroundProgram>) -> Self {
186        Self {
187            input: Prepared::Ground(owner),
188            metadata: None,
189            projection: None,
190        }
191    }
192    /// Representation used for strategy compatibility checks.
193    #[must_use]
194    pub const fn profile(self) -> PreparedProfile {
195        match self.input {
196            Prepared::Relational(_) => PreparedProfile::Relational,
197            Prepared::Formula(_) => PreparedProfile::Formula,
198            Prepared::Hybrid(_) => PreparedProfile::Hybrid,
199            Prepared::TerminalDefinitions(_) => PreparedProfile::TerminalDefinitions,
200            Prepared::Ground(_) => PreparedProfile::Ground,
201        }
202    }
203    /// Validated source views, when the prepared owner retained them.
204    #[must_use]
205    pub const fn metadata(self) -> Option<&'a SourceMetadata> {
206        self.metadata
207    }

209    /// Fixed source projection domain compiled with this coherent input.
210    /// An absent or implicit domain leaves ordinary full enumeration unchanged.
211    #[must_use]
212    pub const fn projection(self) -> Option<&'a zetesis_themelios::PreparedProjection> {
213        self.projection
214    }
215    pub(crate) fn subject(self) -> Subject {
216        match self.input {
217            Prepared::Relational(program) => Subject::Program(program.clone()),
218            Prepared::Formula(input) => Subject::Theory(input.theory.clone()),
219            Prepared::Hybrid(owner) => Subject::Hybrid(owner.clone()),
220            Prepared::TerminalDefinitions(owner) => Subject::TerminalDefinitions(owner.clone()),
221            Prepared::Ground(ground) => Subject::Program(ground.program().clone()),
222        }
223    }
224    fn selection(self, requested: AnswerSelection) -> AnswerSelection {
225        match self.input {
226            Prepared::Formula(input) if input.objectives.is_present() => requested,
227            Prepared::Hybrid(owner) if owner.objectives().is_present() => requested,
228            _ => AnswerSelection::All,
229        }
230    }
231    fn configure(self, mut config: SolveConfig) -> Result<SolveConfig, SolveError> {
232        if matches!(self.input, Prepared::Hybrid(_)) {
233            config.validate_hybrid()?;
234            return Ok(config);
235        }
236        // A hybrid terminal base runs as the hybrid route does: an eager
237        // request contradicts its materialization and is refused, automatic
238        // and lazy requests run it, under the hybrid oracle and batching
239        // restrictions.
240        if let Prepared::TerminalDefinitions(owner) = self.input
241            && owner.base_kind() == zetesis_themelios::BaseKind::Hybrid
242        {
243            config.validate_hybrid().map_err(|error| match error {
244                SolveError::PreparedInput {
245                    oracle, grounder, ..
246                } => SolveError::PreparedInput {
247                    profile: self.profile(),
248                    oracle,
249                    grounder,
250                },
251                other => other,
252            })?;
253            return Ok(config);
254        }
255        if !matches!(self.input, Prepared::Relational(_))
256            && config.source_batching != crate::SourceBatching::Independent
257        {
258            return Err(SolveError::UnsupportedSourceBatching);
259        }
260        let compatible = match self.input {
261            Prepared::Relational(_) => config.oracle != Oracle::Countermodel,
262            Prepared::Formula(_) => {
263                config.oracle != Oracle::Closure && config.grounder != Grounder::Lazy
264            }
265            Prepared::TerminalDefinitions(_) => {
266                config.oracle != Oracle::Closure && config.grounder == Grounder::Auto
267            }
268            Prepared::Hybrid(_) => unreachable!("hybrid policy validated above"),
269            Prepared::Ground(_) => {
270                config.oracle != Oracle::Countermodel && config.grounder != Grounder::Lazy
271            }
272        };
273        if !compatible {
274            return Err(SolveError::PreparedInput {
275                profile: self.profile(),
276                oracle: config.oracle,
277                grounder: config.grounder,
278            });
279        }
280        if matches!(self.input, Prepared::Ground(_)) {
281            config.grounder = Grounder::Eager;
282        }
283        crate::engine::validate_combination(&config)?;
284        Ok(config)
285    }
286}

288/// Immutable identity of the original semantic subject checked by a session.
289/// Handles are cheap shared owners. Equality of printed atoms is not identity;
290/// use the contained program/theory's `same_instance` method.
291#[derive(Clone, Debug)]
292pub enum Subject {
293    /// Original finite relational program, also retained by a ground graph.
294    Program(Program),
295    /// Original indexed finite formula theory, before candidate restrictions.
296    Theory(Theory),
297    /// Original producer core together with its admitted source constraints.
298    Hybrid(zetesis_themelios::HybridFormula),
299    /// Original source with a checked base/terminal-definition correspondence.
300    /// The contained base theory alone is not this semantic subject.
301    TerminalDefinitions(zetesis_themelios::TerminalFormula),
302}
303impl Subject {
304    /// Whether both handles retain the same original immutable semantic instance.
305    /// Structural equality of independently admitted programs is insufficient.
306    #[must_use]
307    pub fn same_instance(&self, other: &Self) -> bool {
308        match (self, other) {
309            (Self::Program(left), Self::Program(right)) => left.same_instance(right),
310            (Self::Theory(left), Self::Theory(right)) => left.same_instance(right),
311            (Self::Hybrid(left), Self::Hybrid(right)) => left.same_instance(right),
312            (Self::TerminalDefinitions(left), Self::TerminalDefinitions(right)) => {
313                left.same_instance(right)
314            }
315            _ => false,
316        }
317    }
318}

320/// A full stable interpretation produced by the session's membership engine.
321/// Private construction preserves its subject association after detachment.
322/// This is semantic evidence, independently of display selection or publication.
323/// It records completed membership under the selected implementation, not a Lean
324/// proof or enumeration coverage. Cloning shares the subject, atom catalog and
325/// selected interpretation; the optional score's cost vector is cloned. Retained
326/// interpretations keep the entire shared catalog alive, including unselected atoms.
327///
328/// ```compile_fail
329/// use zetesis_solve::{AnswerSet, Subject};
330/// use zetesis_core::Model;
331/// fn forge(subject: Subject, interpretation: Model) -> AnswerSet {
332///     AnswerSet { subject, interpretation, score: None }
333/// }
334/// ```
335#[derive(Clone, Debug)]
336pub struct AnswerSet {
337    subject: Subject,
338    interpretation: Model,
339    score: Option<Score>,
340}
341impl AnswerSet {
342    /// Original immutable subject used by the membership engine.
343    #[must_use]
344    pub const fn subject(&self) -> &Subject {
345        &self.subject
346    }
347    /// Complete interpretation, including atoms hidden by source observations.
348    #[must_use]
349    pub const fn interpretation(&self) -> &Model {
350        &self.interpretation
351    }
352    /// Completely evaluated cost, if this solve has an objective. Absence is
353    /// distinct from an active objective with zero cost or no priority slots.
354    /// Optimality is determined by the session outcome, never by a score alone.
355    #[must_use]
356    pub const fn score(&self) -> Option<&Score> {
357        self.score.as_ref()
358    }
359    /// Discard the subject association explicitly for raw-model interoperability.
360    #[must_use]
361    pub fn into_interpretation(self) -> Model {
362        self.interpretation
363    }
364}

377/// An ordinary solve request, before validation or execution begins.
378///
379/// The input borrows one coherent semantic owner. Resource handles may be shared;
380/// candidates, budgets, worker pools and outcomes are created by each start.
381/// Modifiers perform no device discovery, grounding, callbacks or control polls.
382/// Dropping an unstarted request performs no solve work.
383pub struct SessionBuilder<'a> {
384    input: PreparedInput<'a>,
385    config: SolveConfig,
386    cancellation: Cancellation,
387    selection: AnswerSelection,
388    resources: ExecutionResources,
389    measurements: Option<crate::SolveMeasurements>,
390    projection: Option<crate::ProjectionLimits>,
391}
392
393impl<'a> SessionBuilder<'a> {
394    /// Return one full answer-set representative per source `#project` key.
395    /// Objective selection happens first. This changes enumeration identity,
396    /// not membership, scoring or the atoms retained in each representative.
397    /// `config.models` counts representatives; zero requests all classes.
398    /// Construction only records the request. Starting requires a completed
399    /// explicit projection domain; history limits are independent of search.
400    #[must_use]
401    pub const fn projected(mut self, limits: crate::ProjectionLimits) -> Self {
402        self.projection = Some(limits);
403        self
404    }
405
406    pub(crate) const fn full_identity(mut self) -> Self {
407        self.projection = None;
408        self
409    }
410    /// Choose the answer family. The default is optimal ties when an objective
411    /// is present; inputs without objectives always enumerate their full family.
412    #[must_use]
413    pub const fn selection(mut self, selection: AnswerSelection) -> Self {
414        self.selection = selection;
415        self
416    }
417
418    /// Share the caller's execution resource handles with this request.
419    ///
420    /// Cloning the handles does not copy device allocations or logical state.
421    /// The CPU backend ignores them: supplying a device does not force its use.
422    /// A GPU backend must match the supplied context's adapter instead of
423    /// discovering another one.
424    #[must_use]
425    pub fn resources(mut self, resources: &ExecutionResources) -> Self {
426        self.resources = resources.clone();
427        self
428    }
429
430    /// Enumerate and retain the complete original answer-set family using this
431    /// request's input, execution resources, configuration and control.
432    ///
433    /// Collection always uses [`AnswerSelection::All`], overriding any earlier
434    /// [`Self::selection`] choice. Objectives annotate every answer, including
435    /// nonoptimal answers. The request is consumed before any search begins;
436    /// an already started or partially consumed [`Session`] cannot be collected
437    /// through this operation. Use `config.models = 0` to request exhaustion.
438    ///
439    /// Work and retained space follow [`WorldView::collect`]. Members move into
440    /// one bounded collection without repeated membership checks or full-model
441    /// clones. The resource handles do not retain the collected family.
442    ///
443    /// # Errors
444    /// Preserves [`WorldView::collect`]'s typed setup, execution, incomplete
445    /// coverage and storage failures, including the checked prefix and original
446    /// subject. Supplied device failures retain their ordinary session boundary.
447    pub fn collect(self, limits: WorldViewLimits) -> Result<WorldView, WorldViewFailure> {
448        WorldView::collect_request(self, limits, &mut Ignore)
449    }
450
451    /// Collect the complete original family with synchronous observations of
452    /// both session preparation and every subsequent pull.
453    ///
454    /// This has [`Self::collect`]'s unrestricted selection and storage contract.
455    /// The observer is borrowed for the whole operation and no event queue is
456    /// retained. Its effects and storage remain outside solver limits.
457    ///
458    /// # Errors
459    /// Returns [`Self::collect`]'s failures. An observer refusal retains its
460    /// original external cause and the checked prefix, stops subsequent work
461    /// and callbacks, and cannot establish complete coverage or trigger device
462    /// fallback. Formula preparation failures deferred to the first pull retain
463    /// [`Self::start_observed`]'s failure semantics.
464    pub fn collect_observed(
465        self,
466        limits: WorldViewLimits,
467        observer: &mut impl ExecutionObserver,
468    ) -> Result<WorldView, WorldViewFailure> {
469        WorldView::collect_request(self, limits, &mut Observer(observer))
470    }
471
472    /// Share a caller-owned host measurement scope with this solve.
473    ///
474    /// Its detailed-measurement setting replaces config.stats. A stages-only
475    /// scope records coarse intervals without enabling detailed solver statistics.
476    /// Admission and publication can record into the same scope; each session
477    /// retains its own search state.
478    /// This performs no solve work, device discovery or clock reads.
479    #[must_use]
480    pub fn measurements(mut self, measurements: &crate::SolveMeasurements) -> Self {
481        self.config.stats = measurements.details_enabled();
482        self.measurements = Some(measurements.clone());
483        self
484    }
485
486    /// Validate the request and start a fresh session without observations.
487    ///
488    /// # Errors
489    /// Returns the setup failures of [`Session::new`]. Supplied device resources
490    /// additionally retain their explicit adapter, contention and health errors.
491    pub fn start(self) -> Result<Session<'a>, SolveFailure> {
492        self.start_with(&mut Ignore)
493    }
494
495    /// Start with typed preparation observations, borrowing the observer only
496    /// for this call. Later observations require [`Session::next_observed`].
497    ///
498    /// # Errors
499    /// Preserves [`Session::new_observed`]'s setup and callback failure behavior.
500    pub fn start_observed(
501        self,
502        observer: &mut impl ExecutionObserver,
503    ) -> Result<Session<'a>, SolveFailure> {
504        self.start_with(&mut Observer(observer))
505    }
506
507    pub(crate) fn subject(&self) -> Subject {
508        self.input.subject()
509    }
510
511    pub(crate) fn start_with(
512        mut self,
513        observations: &mut impl ExecutionSink,
514    ) -> Result<Session<'a>, SolveFailure> {
515        let phases = self
516            .measurements
517            .unwrap_or_else(|| crate::SolveMeasurements::new(self.config.stats));
518        let projected_limit = self.config.models;
519        let projection = self
520            .projection
521            .map(|limits| {
522                let _solving = phases.stage(crate::SolveStage::Solving);
523                self.cancellation
524                    .poll()
525                    .map_err(crate::ProjectionError::Control)?;
526                let domain = self
527                    .input
528                    .projection()
529                    .ok_or(crate::ProjectionError::MissingDeclaration)?;
530                crate::projection::Projection::new(domain, limits)
531            })
532            .transpose()
533            .map_err(|error| {
534                let mut failure = SolveFailure::from(SolveError::Projection(error));
535                failure.subject = Some(self.input.subject());
536                failure.phase_timings = phases.snapshot().map(Box::new);
537                failure
538            })?;
539        if projection.is_some() {
540            self.config.models = 0;
541        }
542        let result = Session::initialize(
543            self.input,
544            self.config,
545            &self.cancellation,
546            phases.recorder(),
547            self.input.selection(self.selection),
548            &self.resources,
549            observations,
550        );
551        match result {
552            Ok((state, config)) => Ok(Session {
553                state,
554                config,
555                cancellation: self.cancellation,
556                phases,
557                subject: self.input.subject(),
558                projection,
559                projected_limit,
560                projection_done: false,
561            }),
562            Err(error) => {
563                let mut failure = SolveFailure::from(error);
564                failure.subject = Some(self.input.subject());
565                failure.phase_timings = phases.snapshot().map(Box::new);
566                Err(failure)
567            }
568        }
569    }
570}

572/// A pull-based ordinary solve, independent of argument parsing and writers.
573/// [`Self::new`] selects objective ties after search; [`Self::enumerate`] streams
574/// the unrestricted original family. `models` limits yielded answers, or
575/// retained objective ties in a selected solve. Budgets persist across pulls.
576/// Stopping pulls early establishes no additional coverage. The caller decides
577/// how to store or publish each full interpretation; no delivery is inferred.
578///
579/// Enabled elapsed time spans the measurement owner's lifetime through each
580/// snapshot. Time between pulls is unattributed unless the caller records another
581/// stage. A scope injected with [`SessionBuilder::measurements`] includes all work
582/// recorded by its owners, including other sessions. This session's solving spans
583/// cover setup and active pulls. Disabled instrumentation reads no clocks;
584/// deadline control remains independent.
585///
586/// ```
587/// use zetesis_solve::{Backend, PreparedInput, Session, SolveConfig};
588/// use zetesis_cpu::Cancellation;
589/// use zetesis_themelios::{admit, AdmissionOptions};
590///
591/// let admitted = admit("a.".into(), AdmissionOptions::default())?;
592/// let config = SolveConfig { backend: Backend::Cpu, models: 0, ..Default::default() };
593/// let mut session = Session::new(PreparedInput::admitted(&admitted), config, Cancellation::default())?;
594/// let model = session.next().unwrap()?;
595/// assert_eq!(model.interpretation().atoms().len(), 1);
596/// assert!(session.next().is_none());
597/// assert_eq!(session.outcome().unwrap().verified_models(), 1);
598/// # Ok::<(), Box<dyn std::error::Error>>(())
599/// ```
600pub struct Session<'a> {
601    state: State<'a>,
602    config: SolveConfig,
603    cancellation: Cancellation,
604    phases: crate::SolveMeasurements,
605    subject: Subject,
606    projection: Option<crate::projection::Projection<'a>>,
607    projected_limit: usize,
608    projection_done: bool,
609}

611impl<'a> Session<'a> {
612    /// Compose answer selection, execution resources and preparation
613    /// observations before starting an ordinary solve. Construction performs no
614    /// validation or execution. Existing convenience constructors use this same
615    /// request with independently owned execution resources.
616    #[must_use]
617    pub fn builder(
618        input: PreparedInput<'a>,
619        config: SolveConfig,
620        cancellation: Cancellation,
621    ) -> SessionBuilder<'a> {
622        SessionBuilder {
623            input,
624            config,
625            cancellation,
626            selection: AnswerSelection::Optimal,
627            resources: ExecutionResources::default(),
628            measurements: None,
629            projection: None,
630        }
631    }
632
633    /// Start a new ordinary solve over a coherent prepared owner.
634    ///
635    /// # Errors
636    /// Returns typed strategy/backend/setup failures with attempted timing.
637    /// Search/control limits remain typed semantic interruptions in the outcome.
638    /// After strategy validation, every prepared profile polls control before
639    /// allocating worker pools or beginning static compilation. Once bounded
640    /// static compilation starts, it is not preemptible; cancellation is observed
641    /// at the next cooperative poll. Prepared graphs avoid compilation.
642    pub fn new(
643        input: PreparedInput<'a>,
644        config: SolveConfig,
645        cancellation: Cancellation,
646    ) -> Result<Self, SolveFailure> {
647        Self::builder(input, config, cancellation).start()
648    }
649    /// Stream the original program's answer sets, including nonoptimal answers.
650    ///
651    /// Objective scores are evaluated but never restrict candidates or select
652    /// incumbents. Every yielded answer carries completed membership and its
653    /// full interpretation. `config.models` remains a yield limit; zero requests
654    /// exhaustive enumeration. Formula search work/decisions and objective work
655    /// are cumulative. Per-model objective binding/key ceilings and per-candidate
656    /// closure work retain [`SolveConfig`]'s resource contracts. Storage is the
657    /// engine's bounded batches and one yielded
658    /// answer, with no incumbent retention. The caller controls any collection.
659    ///
660    /// # Errors
661    /// Returns the setup failures of [`Self::new`]. During iteration, scoring or
662    /// search stops preserve verified accounting without claiming completeness;
663    /// an answer whose scoring stopped is not yielded as a fully scored answer.
664    pub fn enumerate(
665        input: PreparedInput<'a>,
666        config: SolveConfig,
667        cancellation: Cancellation,
668    ) -> Result<Self, SolveFailure> {
669        Self::builder(input, config, cancellation)
670            .selection(AnswerSelection::All)
671            .start()
672    }
673    /// Start an ordinary solve with synchronous typed execution observations.
674    ///
675    /// The observer is borrowed only during preparation. Use
676    /// [`Self::next_observed`] for subsequent pulls; ordinary iteration discards
677    /// later observations. Events belong to this prepared input, not any other
678    /// session the observer may also serve. No event queue is retained.
679    ///
680    /// # Errors
681    /// Execution-setup failures return immediately, retaining the known subject.
682    /// Formula initialization owns its terminal failure: an observer failure at

685    /// Both paths preserve the original cause and establish no coverage.

689        cancellation: Cancellation,

712    fn initialize(
713        input: PreparedInput<'a>,
714        config: SolveConfig,
715        cancellation: &Cancellation,
716        phases: &Recorder,
717        selection: AnswerSelection,
718        resources: &ExecutionResources,
719        observations: &mut impl ExecutionSink,
720    ) -> Result<(State<'a>, SolveConfig), SolveError> {
721        let config = input.configure(config)?;
722        if let Prepared::TerminalDefinitions(owner) = input.input {
723            phases.terminal_grounding(owner.base_kind());
724        }
725        let _solving = phases.stage(crate::SolveStage::Solving);
726        if let Err(stop) = cancellation.poll() {
727            let interruption = Interruption::Preparation(stop);
728            return Ok((
729                State::Stopped(Box::new(SemanticOutcome {
730                    projection: None,
731                    subject: Some(input.subject()),
732                    selection: Some(selection),
733                    verified: 0,
734                    scored: 0,
735                    retained: 0,
736                    search_state: Some(crate::SearchState::Interrupted(interruption)),
737                    optimization: None,
738                    objective_work: 0,
739                    checked: 0,
740                    gate_atoms: 0,
741                    candidate_statistics: None,
742                    countermodel_statistics: None,
743                    formula_execution: None,
744                    lazy_execution: None,
745                    shared_execution: None,
746                    closure_execution: None,
747                    query_execution: None,
748                    hybrid_execution: None,
749                    terminal_execution: None,
750                    model_construction: None,
751                })),
752                config,
753            ));
754        }
755        let state = match input.input {
756            Prepared::Relational(program) => {
757                State::Closure(Box::new(ClosureSession::with_resources(
758                    program,
759                    None,
760                    &config,
761                    resources,
762                    observations,
763                    cancellation,
764                    phases,
765                )?))
766            }
767            Prepared::Ground(ground) => State::Closure(Box::new(ClosureSession::with_resources(
768                ground.program(),
769                Some(Arc::clone(ground)),
770                &config,
771                resources,
772                observations,
773                cancellation,
774                phases,
775            )?)),
776            Prepared::Formula(input) => State::Formula(Box::new(FormulaSession::with_resources(
777                input,
778                &config,
779                resources,
780                observations,
781                cancellation,
782                phases,
783                selection,
784            )?)),
785            Prepared::Hybrid(owner) => State::Hybrid(Box::new(HybridSession::observed(
786                crate::hybrid_session::HybridInput {
787                    core: owner.core(),
788                    subject: crate::Subject::Hybrid(owner.clone()),
789                },
790                &config,
791                resources,
792                observations,
793                cancellation,
794                phases,
795                selection,
796            )?)),
797            Prepared::TerminalDefinitions(owner) => {
798                State::TerminalDefinitions(Box::new(TerminalSession::new(
799                    owner,
800                    &config,
801                    resources,
802                    observations,
803                    cancellation,
804                    phases,
805                )?))
806            }
807        };
808        Ok((state, config))
809    }
810    /// Final semantic evidence once search finishes, stops, or fails. An
811    /// optimized outcome can be available while retained ties await delivery.
812    #[must_use]
813    pub fn outcome(&self) -> Option<SemanticOutcome> {
814        if self.projection_done {
815            return Some(self.progress());
816        }
817        let mut outcome = match &self.state {
818            State::Closure(state) => state.terminal().then(|| state.outcome()),
819            State::Formula(state) => state
820                .finished()
821                .then(|| state.outcome(self.phases.recorder())),
822            State::Hybrid(state) => state
823                .finished()
824                .then(|| state.outcome(self.phases.recorder())),
825            State::TerminalDefinitions(state) => state
826                .finished()
827                .then(|| state.outcome(self.phases.recorder())),
828            State::Stopped(outcome) => Some((**outcome).clone()),
829        };
830        if let Some(outcome) = &mut outcome {
831            outcome.projection = self
832                .projection
833                .as_ref()
834                .map(crate::projection::Projection::statistics);
835        }
836        outcome
837    }
838    /// End an unfinished session and retain its settled evidence. Native workers
839    /// are joined without cancelling the caller's token. Coverage stays unavailable
840    /// unless the retained engine established a terminal state or cleanup stopped.
841    #[must_use]
842    pub fn stop(mut self) -> SemanticOutcome {
843        let phases = self.phases.recorder();
844        match &mut self.state {
845            State::Formula(state) => {
846                let previous = state.outcome(phases).search_state();
847                let cleanup = state.stop(phases);
848                state.conclude(crate::completion::after_cleanup(previous, cleanup), phases);
849            }
850            State::Hybrid(state) => {
851                let previous = state.outcome(phases).search_state();
852                // A source fault leaves its settled evidence unclassified;
853                // consuming stop has no further answer or error to publish.
854                let _ = state.conclude(previous, phases);
855            }
856            State::TerminalDefinitions(state) => state.stop(phases),
857            State::Closure(_) | State::Stopped(_) => {}
858        }
859        self.progress()
860    }
861    /// Attempted host phases from this session's measurement owner; absence means
862    /// instrumentation was disabled. An injected owner includes all recorded
863    /// caller work and other sessions sharing that scope.
864    #[must_use]
865    pub fn phase_timings(&self) -> Option<PhaseTimings> {
866        self.phases.snapshot()
867    }
868    /// Current checked evidence, including results buffered before delivery.
869    ///
870    /// A snapshot does not stop search. Missing completion remains absent; only
871    /// the recorded coverage determines whether enumeration is complete.
872    #[must_use]
873    pub fn progress(&self) -> SemanticOutcome {
874        let mut outcome = match &self.state {
875            State::Closure(state) => state.outcome(),
876            State::Formula(state) => state.outcome(self.phases.recorder()),
877            State::Hybrid(state) => state.outcome(self.phases.recorder()),
878            State::TerminalDefinitions(state) => state.outcome(self.phases.recorder()),
879            State::Stopped(outcome) => (**outcome).clone(),
880        };
881        outcome.projection = self
882            .projection
883            .as_ref()
884            .map(crate::projection::Projection::statistics);
885        if self.projection_done
886            && self.projected_limit != 0
887            && outcome
888                .projection
889                .is_some_and(|stats| stats.representatives >= self.projected_limit)
890            && outcome.search_state.is_none()
891        {
892            outcome.search_state = Some(crate::SearchState::RequestedModels);
893        }
894        outcome
895    }
896    /// Pull the next answer using synchronous typed execution observations.
897    ///
898    /// The observer is borrowed only for this pull. Events describe this session's
899    /// attempted work and cannot establish answer-set membership or coverage.
900    /// An observer error stops this session: later pulls return None, with its
901    /// checked prefix retained in the failure and [`Self::outcome`]. Successful
902    /// callbacks do not imply that subsequent work or answer publication succeeds.
903    ///
904    /// # Errors
905    /// Returns ordinary solve faults or the observer's external error, wrapped
906    /// separately from device faults so it cannot trigger fallback or retry.
907    pub fn next_observed(
908        &mut self,
909        observer: &mut impl ExecutionObserver,
910    ) -> Option<Result<AnswerSet, SolveFailure>> {
911        self.pull(&mut Observer(observer))
912    }

914    pub(crate) fn pull(
915        &mut self,
916        observations: &mut impl ExecutionSink,
917    ) -> Option<Result<AnswerSet, SolveFailure>> {
918        if self.projection_done {
919            return None;
920        }
921        loop {
922            let next = self.pull_original(observations);
923            let Some(projection) = self.projection.as_mut() else {
924                return next;
925            };
926            match next {
927                Some(Ok(answer)) => {
928                    let solving = self.phases.stage(crate::SolveStage::Solving);
929                    let inserted = projection.insert(answer.interpretation(), &self.cancellation);
930                    drop(solving);
931                    match inserted {
932                        Ok(false) => {}
933                        Ok(true) => {
934                            self.projection_done = self.projected_limit != 0
935                                && projection.statistics().representatives >= self.projected_limit;
936                            return Some(Ok(answer));
937                        }
938                        Err(error) => {
939                            self.projection_done = true;
940                            let mut failure = SolveFailure::from(SolveError::Projection(error));
941                            failure.subject = Some(self.subject.clone());
942                            failure.semantic = Some(Box::new(self.progress()));
943                            failure.phase_timings = self.phase_timings().map(Box::new);
944                            return Some(Err(failure));
945                        }
946                    }
947                }
948                Some(Err(error)) => {
949                    self.projection_done = true;
950                    return Some(Err(error));
951                }
952                None => {
953                    self.projection_done = true;
954                    let complete =
955                        self.progress().completion() == Some(crate::Completion::Exhausted);
956                    self.projection
957                        .as_mut()
958                        .expect("projected stream")
959                        .finish(complete);
960                    return None;
961                }
962            }
963        }
964    }
965
966    fn pull_original(
967        &mut self,
968        observations: &mut impl ExecutionSink,
969    ) -> Option<Result<AnswerSet, SolveFailure>> {
970        let solving = self.phases.stage(crate::SolveStage::Solving);
971        let next = match &mut self.state {
972            State::Closure(state) => state
973                .next(&self.config, &self.cancellation, self.phases.recorder())
974                .map(|result| result.map(|model| (model, None))),
975            State::Formula(state) => state.next(
976                &self.config,
977                observations,
978                &self.cancellation,
979                self.phases.recorder(),
980            ),
981            State::Hybrid(state) => state.next(
982                &self.config,
983                observations,
984                &self.cancellation,
985                self.phases.recorder(),
986            ),
987            State::TerminalDefinitions(state) => state.next(
988                &self.config,
989                observations,
990                &self.cancellation,
991                self.phases.recorder(),
992            ),
993            State::Stopped(_) => None,
994        };
995        drop(solving);
996        next.map(|result| match result {
997            Ok((interpretation, score)) => Ok(AnswerSet {
998                subject: self.subject.clone(),
999                interpretation,
1000                score,
1001            }),
1002            Err(error) => {
1003                let mut failure = SolveFailure::from(error);
1004                failure.semantic = Some(Box::new(self.progress()));
1005                failure.phase_timings = self.phases.snapshot().map(Box::new);
1006                Err(failure)
1007            }
1008        })
1009    }
