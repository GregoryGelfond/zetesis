7const DEFAULT_MAX_SUBSTITUTIONS: usize = 10_000_000;

9/// Policy for one semantic session. Budgets retain their existing ownership:
10/// Formula search/objective work is cumulative. Independent closure work is per
11/// candidate; shared CPU rounds separate collective source and per-world work.
12/// No field selects an output format, file path, source loader or writer.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub struct SolveConfig {
15    /// Execution backend: the CPU (the default) runs source joins or static
16    /// closure scans on an owned Rayon pool; a GPU backend requires a physical
17    /// device of its API and runs exact integer batches there, with general
18    /// formulas propagated on the GPU and their residual search exact on the CPU.
19    pub backend: Backend,
20    /// Materialization policy within the admitted input's supported profiles.
21    pub grounder: Grounder,
22    /// Relational source traversal; the sharing strategies require CPU execution.
23    pub source_batching: SourceBatching,
24    /// Exact membership policy; prepared formula inputs retain their original theory.
25    pub oracle: Oracle,
26    /// How the formula route proposes candidates and queries the reduct.
27    pub search: SearchMethod,
28    /// Enable optional host timing; semantic/resource counters remain independent.
29    pub stats: bool,
30    /// Cumulative streamed-constraint limits for a hybrid formula session.
31    /// Independent of core candidate/reduct work; eager sessions do not use them.
32    /// The default substitution ceiling matches the ordinary session allowance.
33    pub constraints: zetesis_themelios::ConstraintCheckLimits,
34    /// Maximum yielded models or retained optimum ties; zero requests all.
35    pub models: usize,
36    /// Cumulative candidate restriction or formula encoding and search work.
37    pub max_search_work: u64,
38    /// Cumulative formula branch decisions.
39    pub max_search_decisions: u64,
40    /// Cumulative prepared-order and selected-model construction work per formula
41    /// session, including failed attempts. Independent of membership and scoring.
42    pub max_model_work: u64,
43    /// Peak live model-construction metadata: the prepared order and private
44    /// selection/publication buffers, including actual growth overlap. Excludes
45    /// immutable catalog payload, previously published models and allocator/Arc
46    /// bookkeeping. This is a named capacity allowance, not process RSS.
47    pub max_model_bytes: usize,
48    /// Distinct candidate projections retained to exclude previously visited keys.
49    pub max_projection_entries: u64,
50    /// Logical nodes in the retained candidate-projection trie.
51    pub max_projection_nodes: usize,
52    /// Named projection-history capacity and conservative growth overlap.
53    /// This excludes allocator overhead and the separate authored CNF encoding.
54    pub max_projection_bytes: usize,
55    /// Cumulative objective preparation and evaluation work, including refusals.
56    pub max_objective_work: u64,
57    /// Cumulative optional bound-generation work, excluding shared preparation;
58    /// zero disables pruning while prepared scoring remains available.
59    pub max_objective_bound_work: u64,
60    /// Complete objective bindings per verified model.
61    pub max_objective_bindings: u64,
62    /// Distinct objective keys per verified model.
63    pub max_objective_keys: usize,
64    /// Objective key payload bytes per verified model.
65    pub max_objective_key_bytes: usize,
66    /// Maximum retained incumbent models.
67    pub max_optimal_models: usize,
68    /// Maximum atoms across retained incumbents.
69    pub max_optimal_atoms: usize,
70    /// Maximum portable incumbent payload bytes: each distinct occurrence
71    /// catalog's encoding once, selected positions per retained model, and one
72    /// shared best-score record. Equal-content separate catalogs count separately.
73    /// Excludes canonical identities outside occurrence maps, spare vector/hash
74    /// capacity, owner-index entries, allocator/Arc overhead, subjects, execution
75    /// state and transient old/new replacement overlap; this is not RSS.
76    pub max_optimal_bytes: usize,
77    /// Maximum candidates per owned batch.
78    pub batch_size: NonZeroUsize,
79    /// Worker count: the closure route's pool, and the walkers of the
80    /// region tree under the regions method, one being the scalar walk.
81    /// CPU closure setup conservatively reserves `max_closure_bytes` per worker;
82    /// [`Self::for_allowance`] keeps that product within the collective ceiling.
83    /// The library default is four; the command uses the host's available
84    /// parallelism, or one when availability is unknown.
85    pub workers: NonZeroUsize,
86    /// Worker count for unresolved formula queries: under the clauses search,
87    /// and under regions with one CPU walker or general device propagation.
88    /// Complete device certificates use scalar validation with no residual
89    /// pool. Multiple CPU region workers decide their own leaves in `workers`.
90    pub completion_workers: NonZeroUsize,
91    /// Named cold reduct preparation and each query's retained capacity.
92    /// The immutable reduct is prepared once per original theory. Parallel
93    /// completion also admits its shared owner against the collective ceiling.
94    /// Shared theory payload and allocator metadata are excluded.
95    pub max_reduct_bytes: u64,
96    /// Shared prepared reduct, worker query capacities and transient/result bytes, excluding
97    /// thread stacks, allocator/table overhead, scalar cursor and GPU storage.
98    /// Optional class preparation/checking uses this ceiling independently;
99    /// it is not a combined cap on class storage plus completion storage.
100    pub max_completion_scratch_bytes: u64,
101    /// Maximum candidate seeds or formula candidates.
102    pub max_candidates: u64,
103    /// Maximum incrementally retained gate atoms.
104    pub max_carrier_atoms: usize,
105    /// Logical candidate restriction payload during source preparation.
106    /// This excludes allocator slack and source relation/tree metadata.
107    pub max_candidate_bytes: usize,
108    /// Maximum charged CPU oracle work per candidate, or shared host source
109    /// work per lazy GPU batch. Shared CPU rounds count record visits plus
110    /// antecedent tests per world; independent joins and device units differ.
111    /// Formula execution also applies this ceiling per independent original or
112    /// frozen-reduct verification call, and per certified candidate check. It
113    /// does not replace cumulative `max_search_work` or device propagation limits.
114    pub max_work: u64,
115    /// Named storage for one independent lazy CPU closure construction.
116    /// Input seeds and completed model retention belong to separate owners.
117    /// CPU closure setup conservatively admits every assigned worker at this
118    /// allowance, including eager and shared execution, so
119    /// `workers * max_closure_bytes` must not exceed `max_closure_batch_bytes`;
120    /// that route refuses an excessive or overflowing product before allocating
121    /// its pool, compiling static rules or initializing candidates. Formula and
122    /// device execution do not use this reservation.
123    pub max_closure_bytes: usize,
124    /// Collective independent CPU preparation, cached workspace and active
125    /// closure allowance. Also bounds immutable query preparation bytes.
126    /// Shared-round and GPU batch storage remain separate.
127    pub max_closure_batch_bytes: usize,
128    /// Maximum device work per formula candidate, independent of CPU work.
129    /// Propagation charges setup and sweeps; tight support charges its complete
130    /// scan. Insufficient mandatory work is refused before submission.
131    pub gpu_formula_work: u32,
132    /// Maximum device propagation sweeps per formula candidate. Zero performs
133    /// original-truth setup and leaves undecided candidates to exact CPU search.
134    /// This ceiling does not apply to the tight-support scan.
135    pub gpu_formula_rounds: u32,
136    /// Collective source work per shared CPU batch, or immutable query
137    /// preparation work for independent CPU closure. Candidate work is separate.
138    pub max_source_work: u64,
139    /// Maximum derived independent CPU atoms, demanded shared CPU/GPU batch
140    /// catalog atoms, eager atoms, or copied candidate restriction atom occurrences.
141    /// Shared catalogs and candidate preparation are collective, not per world.
142    pub max_atoms: usize,
143    /// Maximum substitutions in explicit static lowering.
144    pub max_substitutions: usize,
145    /// Maximum rules in explicit static lowering.
146    pub max_ground_rules: usize,
147    /// Maximum accounted pending/GPU batch payload bytes. Lazy GPU splits this
148    /// allowance equally between source coordination and transient transport.
149    /// Shared CPU rounds use the full allowance for source and world state;
150    /// allocator, tree and driver overhead are outside the payload bound.
151    pub max_batch_bytes: u64,
152}

212impl SolveConfig {
213    /// The workers that walk the region tree together and decide its
214    /// leaves themselves: several, under the regions method on a CPU
215    /// backend. A device backend uses separate bounded candidate producers;
216    /// those workers do not decide membership. One worker or clause search
217    /// uses the scalar proposer, so all these alternatives return `None`.
218    #[must_use]
219    pub fn region_workers(&self) -> Option<NonZeroUsize> {
220        (self.search == SearchMethod::Regions
221            && self.workers.get() > 1
222            && self.backend == Backend::Cpu)
223            .then_some(self.workers)
224    }
225}

233impl SolveConfig {
234    /// Two gibibytes: the memory allowance the byte ceilings of
235    /// [`Self::DEFAULT`] are the shares of.
236    pub const REFERENCE_MEMORY: u64 = 2 * 1024 * 1024 * 1024;
237
238    /// The defaults for a session allowed `memory` bytes over `workers`:
239    /// [`Self::DEFAULT`] with the ten byte ceilings that bound retained
240    /// storage scaled by `memory` over [`Self::REFERENCE_MEMORY`], each
241    /// saturating at its type's maximum, and the per-closure allowance each
242    /// worker's share of the scaled collective closure ceiling, so that the
243    /// product checked by CPU closure setup holds. The projection, objective
244    /// key, incumbent, model construction, reduct, completion scratch, candidate, collective
245    /// closure and batch ceilings scale; work, count and structural
246    /// ceilings, and the ceilings of source admission, do not. Constant
247    /// time.
248    #[must_use]
249    pub fn for_allowance(memory: u64, workers: NonZeroUsize) -> Self {
250        let scale_u64 = |default: u64| {
251            let scaled =
252                u128::from(default) * u128::from(memory) / u128::from(Self::REFERENCE_MEMORY);
253            u64::try_from(scaled).unwrap_or(u64::MAX)
254        };
255        let scale_usize = |default: usize| {
256            usize::try_from(scale_u64(u64::try_from(default).unwrap_or(u64::MAX)))
257                .unwrap_or(usize::MAX)
258        };
259        let max_closure_batch_bytes = scale_usize(Self::DEFAULT.max_closure_batch_bytes);
260        Self {
261            workers,
262            max_projection_bytes: scale_usize(Self::DEFAULT.max_projection_bytes),
263            max_objective_key_bytes: scale_usize(Self::DEFAULT.max_objective_key_bytes),
264            max_optimal_bytes: scale_usize(Self::DEFAULT.max_optimal_bytes),
265            max_model_bytes: scale_usize(Self::DEFAULT.max_model_bytes),
266            max_reduct_bytes: scale_u64(Self::DEFAULT.max_reduct_bytes),
267            max_completion_scratch_bytes: scale_u64(Self::DEFAULT.max_completion_scratch_bytes),
268            max_candidate_bytes: scale_usize(Self::DEFAULT.max_candidate_bytes),
269            max_closure_bytes: max_closure_batch_bytes / workers.get(),
270            max_closure_batch_bytes,
271            max_batch_bytes: scale_u64(Self::DEFAULT.max_batch_bytes),
272            ..Self::DEFAULT
273        }
274    }
275}

277impl SolveConfig {
278    /// Validate execution-policy combinations before admitting a source.
279    ///
280    /// This performs no admission, execution, allocation or device discovery.
281    /// A later prepared input can impose additional representation constraints.
282    /// Execution-specific resource checks happen only when their route is set
283    /// up. CPU closure setup checks the worker reservation before allocating or
284    /// initializing execution state. After policy validation, a pre-cancelled
285    /// session stops before those resource checks and allocations.
286    ///
287    /// # Errors
288    /// Refuses incompatible oracle, grounding and shared-source policies.
289    pub fn validate(&self) -> Result<(), crate::SolveError> {
290        crate::engine::validate_combination(self)
291    }
292
293    /// Validate policies for finite formula admission before constructing it.
294    ///
295    /// This does not establish that any source is admissible or a device exists.
296    ///
297    /// # Errors
298    /// Refuses lazy grounding and relational shared-source policies.
299    pub fn validate_formula(&self) -> Result<(), crate::SolveError> {
300        if self.grounder == Grounder::Lazy {
301            return Err(crate::SolveError::UnsupportedOracle {
302                backend: self.backend,
303                grounder: self.grounder,
304            });
305        }
306        crate::engine::validate_countermodel(self)
307    }
308
309    /// Validate the execution policy for a producer core with streamed source
310    /// constraints. This does not admit source or construct an executor.
311    ///
312    /// # Errors
313    /// Refuses eager materialization, closure membership, device execution and
314    /// shared relational source rounds for this initial CPU formula profile.
315    pub fn validate_hybrid(&self) -> Result<(), crate::SolveError> {
316        if self.source_batching != SourceBatching::Independent {
317            return Err(crate::SolveError::UnsupportedSourceBatching);
318        }
319        if self.backend.is_gpu() {
320            return Err(crate::SolveError::HybridBackend {
321                backend: self.backend,
322            });
323        }
324        if self.oracle == Oracle::Closure || self.grounder == Grounder::Eager {
325            return Err(crate::SolveError::PreparedInput {
326                profile: crate::PreparedProfile::Hybrid,
327                oracle: self.oracle,
328                grounder: self.grounder,
329            });
330        }
331        Ok(())
332    }
