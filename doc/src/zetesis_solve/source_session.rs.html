1//! Exact source evaluation under a complete dependency-order certificate.
2//!
3//! The evaluator establishes membership before ordinary bounded model
4//! construction. No candidate theory is substituted for the original source.

19pub(crate) struct SourceSession<'a> {
20    owner: &'a AcyclicSource,
21    finished: bool,
22    checked: u64,
23    verified: u64,
24    state: Option<SearchState>,
25    construction: Option<Account>,
26}
27
28impl<'a> SourceSession<'a> {
29    pub(crate) fn new(
30        owner: &'a AcyclicSource,
31        observations: &mut impl ExecutionSink,
32    ) -> Result<Self, SolveError> {
33        observations.record(crate::ExecutionObservation::SourceEvaluation)?;
34        Ok(Self {
35            owner,
36            finished: false,
37            checked: 0,
38            verified: 0,
39            state: None,
40            construction: None,
41        })
42    }
43
44    pub(crate) fn next(
45        &mut self,
46        config: &SolveConfig,
47        cancellation: &Cancellation,
48        phases: &Recorder,
49    ) -> Option<Result<(Model, Option<Score>), SolveError>> {
50        if self.finished {
51            return None;
52        }
53        // Source preparation already established the exact interpretation.
54        // This session consumes it once, under its own publication allowances.
55        self.finished = true;
56        if let Err(stop) = cancellation.poll() {
57            self.state = Some(SearchState::Interrupted(Interruption::Preparation(stop)));
58            return None;
59        }
60        self.checked = 1;
61        let Some(positions) = self.owner.selected_positions() else {
62            self.state = Some(SearchState::Exhausted);
63            return None;
64        };
65        self.verified = 1;
66        let construction = self.construction.get_or_insert_with(Account::default);
67        let model = phases.measure(SolvePhase::ModelConstruction, || {
68            let order = construction.prepare(self.owner.atom_catalog(), config, cancellation)?;
69            construction.select(&order, positions.iter().copied(), config, cancellation)
70        });
71        match model {
72            Ok(model) => {
73                // The class certificate establishes uniqueness. A requested
74                // count of one does not erase already established exhaustion.
75                self.state = Some(SearchState::Exhausted);
76                Some(Ok((model, None)))
77            }
78            Err(Failure::Interrupted(stop)) => {
79                self.state = Some(SearchState::Interrupted(Interruption::ModelConstruction(
80                    stop,
81                )));
82                None
83            }
84            Err(Failure::Run(error)) => Some(Err(error)),
85        }
86    }
87
88    pub(crate) const fn finished(&self) -> bool {
89        self.finished
90    }
91
92    pub(crate) const fn stop(&mut self) {
93        self.finished = true;
94    }
95
96    pub(crate) fn outcome(&self) -> SemanticOutcome {
97        SemanticOutcome {
98            projection: None,
99            subject: Some(crate::Subject::Source(self.owner.clone())),
100            selection: Some(AnswerSelection::All),
101            verified: self.verified,
102            scored: 0,
103            retained: 0,
104            search_state: self.state,
105            optimization: None,
106            objective_work: 0,
107            checked: self.checked,
108            gate_atoms: 0,
109            candidate_statistics: None,
110            countermodel_statistics: None,
111            formula_execution: None,
112            lazy_execution: None,
113            shared_execution: None,
114            closure_execution: None,
115            query_execution: None,
116            hybrid_execution: None,
117            terminal_execution: None,
118            source_evaluation: Some(self.owner.statistics()),
119            model_construction: self.construction.as_ref().map(Account::statistics),
120        }
121    }
