14/// Cumulative search ceilings. Zero permits no operation of that kind.
15#[derive(Clone, Copy, Debug)]
16pub struct SearchLimits {
17    /// Charged operations, cumulatively: region narrowing and traversal, the
18    /// original index's construction, proper-subset queries, candidate
19    /// restrictions and bounds. Stable-model enumeration also charges optional
20    /// certificate work here.
21    pub max_work: u64,
22    /// Maximum region splits, in the candidate tree and in every proper-subset
23    /// query's tree together.
24    pub max_decisions: u64,
25}
26impl Default for SearchLimits {
27    fn default() -> Self {
28        Self {
29            max_work: 100_000_000,
30            max_decisions: 1_000_000,
31        }
32    }
33}

35/// Accounting for charged search and certificate operations, including failures.
36/// A panicking parallel worker conservatively consumes a reservation whose
37/// actual-work receipt could not be returned, and reports `WorkerPanicked`.
38#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
39pub struct SearchStatistics {
40    /// Charged primitive operations, including reservations lost to worker panic.
41    pub work: u64,
42    /// Region splits charged against [`SearchLimits::max_decisions`].
43    pub decisions: u64,
44}

46pub(crate) struct Budget<'a, Q = LocalQuota> {
47    pub(crate) quota: Q,
48    pub(crate) limits: SearchLimits,
49    pub(crate) cancellation: &'a Cancellation,
50    pub(crate) statistics: SearchStatistics,
51}
52impl<Q: Quota> Budget<'_, Q> {
53    pub(crate) fn tick(&mut self) -> Result<(), Incomplete> {
54        self.cancellation.poll()?;
55        self.quota
56            .work(self.statistics.work, self.limits.max_work)?;
57        self.statistics.work += 1;
58        Ok(())
59    }
60    pub(crate) fn decide(&mut self) -> Result<(), Incomplete> {
61        self.tick()?;
62        self.quota
63            .decision(self.statistics.decisions, self.limits.max_decisions)?;
64        self.statistics.decisions += 1;
65        Ok(())
66    }
67    /// Grant between one and `wanted` permits to an operation that spends
68    /// them one per read: control is polled once for the batch, and the
69    /// granted permits count as spent until [`Self::refund`] returns the rest.
70    pub(crate) fn reserve_up_to(&mut self, wanted: u64) -> Result<u64, Incomplete> {
71        zetesis_algebra::Quota::reserve(&mut self.incremental(), wanted)
72    }
73    /// Run a finite accounted kernel once under its full reservation, rather
74    /// than treating one partial lease grant as the kernel's complete limit.
75    /// Actual returned work is committed on success and refusal alike.
76    pub(crate) fn with_reservation<T>(
77        &mut self,
78        wanted: u64,
79        run: impl FnOnce(u64) -> (T, u64),
80    ) -> Result<T, Incomplete> {
81        self.cancellation.poll()?;
82        let reservation =
83            self.quota
84                .reserve_kernel(self.statistics.work, self.limits.max_work, wanted)?;
85        if reservation.allowance() == 0 {
86            return Err(Incomplete::WorkLimit);
87        }
88        let (result, used) =
89            Meter::under_reservation(reservation, |granted| Ok::<_, Incomplete>(run(granted)))
90                .map_err(reservation_failure)?;
91        self.statistics.work = self
92            .statistics
93            .work
94            .checked_add(used)
95            .ok_or(Incomplete::CounterOverflow)?;
96        Ok(result)
97    }
98
99    /// Return permits granted by [`Self::reserve_up_to`] and not spent.
100    pub(crate) fn refund(&mut self, unspent: u64) {
101        zetesis_algebra::Quota::refund(&mut self.incremental(), unspent);
102    }
103    /// Supply an atomic minimum through the quota's ownership-aware boundary.
104    pub(crate) fn replenish(
105        &mut self,
106        unused: &mut u64,
107        minimum: u64,
108        capacity: u64,
109    ) -> Result<(), Failure<Incomplete>> {
110        Q::replenish(self.incremental(), unused, minimum, capacity)
111    }
112
113    fn incremental(&mut self) -> quota::Incremental<'_, Q> {
114        quota::Incremental {
115            quota: &self.quota,
116            work: &mut self.statistics.work,
117            ceiling: self.limits.max_work,
118            cancellation: self.cancellation,
119        }
120    }
121
122    /// The work left before the ceiling, for an operation that charges its
123    /// own work and reports it afterwards through [`Self::charge`].
124    pub(crate) fn remaining_work(&self) -> u64 {
125        self.limits.max_work.saturating_sub(self.statistics.work)
126    }
127    /// Charge an operation's work, one poll for the lot, reserved through the
128    /// quota as ticks would be. The work may be already performed, or known
129    /// in advance and charged before it is performed, as the original index
130    /// is; a refused charge adds nothing.
131    pub(crate) fn charge(&mut self, work: u64) -> Result<(), Incomplete> {
132        self.cancellation.poll()?;
133        self.quota
134            .charge(self.statistics.work, self.limits.max_work, work)?;
135        Meter::charge_receipt(&mut self.statistics.work, u128::from(work), u64::MAX)
136            .map_err(|_| Incomplete::CounterOverflow)
137    }
138}

140/// Preserve callback errors and the existing invalid-receipt classification
141/// at the whole-kernel boundary, without another control observation.
142pub(crate) fn reservation_failure(error: Failure<Incomplete>) -> Incomplete {
143    match error {
144        Failure::Refused(error) => error,
145        Failure::Stopped(Stop::InvalidProgram) => Incomplete::InvalidWitness,
146        Failure::Stopped(other) => Incomplete::Verification(other),
147    }
148}
149
150pub(crate) fn increment(counter: &mut u64) -> Result<(), Incomplete> {
151    *counter = CheckedSum
152        .combine(Some(*counter), Some(1))
153        .ok_or(Incomplete::CounterOverflow)?;
154    Ok(())
155}

157pub(crate) fn storage<T>(count: usize) -> Result<Vec<T>, Incomplete> {
158    let mut result = Vec::new();
159    result
160        .try_reserve_exact(count)
161        .map_err(|_| Incomplete::Allocation)?;
162    Ok(result)
163}
