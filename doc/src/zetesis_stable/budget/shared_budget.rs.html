1//! Scoped work permits for joined queries and parallel region walks.
2//!
3//! Committed work + available permits + outstanding grants equals the initial
4//! allowance. A lease consumes its grant locally, then commits used permits and
5//! returns unused ones when it settles: before its holder waits, delivers or
6//! reserves, and on every exit. A settled lease refills on its next use.
7//! Outstanding grants are not reported as spent work. Decision reservations
8//! remain individually shared and nonblocking.
9
10use std::sync::atomic::{AtomicU64, Ordering};
11use std::sync::{Condvar, Mutex, MutexGuard};
12use std::time::Duration;
13
14use crate::{Cancellation, Incomplete, SearchLimits, SearchStatistics};
15use zetesis_algebra::{Failure, Meter, POLL_INTERVAL, PermitLease};

17/// A scheduling policy, not a measured crossover: an ordinary incremental
18/// grant prefers this many permits. A larger atomic minimum can require a
19/// larger window; no window changes the performed-work charge. A region walker
20/// keeps its lease across its
21/// consecutive regions, which commonly use a few hundred permits each, so one
22/// grant serves many regions and the shared ledger lock is taken about twice
23/// per grant rather than twice per region. Accounted kernels reserve their
24/// complete bounded allowance separately. No grant changes the charged
25/// operation sequence, so the work limit still bounds the identical total; a
26/// larger grant only coarsens how the shared allowance is parcelled out among
27/// workers and how often a lease rechecks cooperative control.
28const WORK_QUANTUM: u64 = 16384;
29/// A waiting query rechecks cooperative control at least once per timed wait.
30/// Lock acquisition and OS scheduling do not provide a hard response deadline.
31const CONTROL_WAIT: Duration = Duration::from_millis(1);
32
33pub(crate) struct SharedBudget {
34    limits: SearchLimits,
35    work: Mutex<Work>,
36    returned: Condvar,
37    decisions: AtomicU64,
38}
39
40struct Work {
41    spent: u64,
42    available: u64,
43    outstanding: u64,
44}
45
46impl Work {
47    /// None means another lease can still return permits, not exhaustion.
48    fn grant(&mut self) -> Result<Option<u64>, Incomplete> {
49        self.grant_at_least(1)
50    }
51
52    /// A buffered atomic operation needs its minimum before it can run. Peers
53    /// can satisfy a shortfall only while they retain outstanding grants. Keep
54    /// the usual preferred window, enlarging it only for a larger atomic bound.
55    fn grant_at_least(&mut self, minimum: u64) -> Result<Option<u64>, Incomplete> {
56        if self.available < minimum {
57            return if self.outstanding == 0 {
58                Err(Incomplete::WorkLimit)
59            } else {
60                Ok(None)
61            };
62        }
63        let grant = self.available.min(WORK_QUANTUM.max(minimum));
64        self.available -= grant;
65        self.outstanding += grant;
66        Ok(Some(grant))
67    }
68
69    /// An accounted kernel needs its allowance before entering. Wait for
70    /// outstanding grants when they can still cover a shortfall; otherwise
71    /// the remaining allowance bounds a possibly incomplete kernel attempt.
72    fn reserve(&mut self, wanted: u64) -> Option<u64> {
73        if self.available < wanted && self.outstanding != 0 {
74            return None;
75        }
76        let granted = wanted.min(self.available);
77        self.available -= granted;
78        self.outstanding += granted;
79        Some(granted)
80    }
81}

83impl SharedBudget {
84    pub(crate) fn new(limits: SearchLimits, spent: SearchStatistics) -> Self {
85        Self {
86            limits,
87            work: Mutex::new(Work {
88                spent: spent.work,
89                available: limits.max_work.saturating_sub(spent.work),
90                outstanding: 0,
91            }),
92            returned: Condvar::new(),
93            decisions: AtomicU64::new(spent.decisions),
94        }
95    }
96
97    pub(crate) fn lease<'a>(&'a self, cancellation: &'a Cancellation) -> WorkLease<'a> {
98        WorkLease {
99            shared: self,
100            cancellation,
101            permits: PermitLease::new(),
102        }
103    }
104
105    fn lock(&self) -> MutexGuard<'_, Work> {
106        // Only infallible conservation updates run while held: no query work,
107        // allocation, user callback or control poll can unwind through this lock.
108        self.work
109            .lock()
110            .unwrap_or_else(std::sync::PoisonError::into_inner)
111    }
112
113    fn decide(&self) -> Result<(), Incomplete> {
114        self.decisions
115            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |spent| {
116                Meter::check_wide_count(u128::from(spent), 1, u128::from(self.limits.max_decisions))
117                    .ok()
118                    .map(|_| spent + 1)
119            })
120            .map(|_| ())
121            .map_err(|_| Incomplete::DecisionLimit)
122    }
123
124    /// Commit `work` permits at once, for work done outside any lease. The
125    /// permits are taken from those available; when they do not cover it
126    /// the charge is refused as exhaustion without waiting for outstanding
127    /// grants or kernel reservations to return, so held allowances can cause
128    /// an early refusal.
129    pub(crate) fn charge(&self, work: u64) -> Result<(), Incomplete> {
130        let mut permits = self.lock();
131        // Outstanding grants are unavailable to this direct charge. The
132        // conservation invariant keeps the currently spendable ceiling in u64.
133        let ceiling = permits.spent + permits.available;
134        let spent = Meter::check_receipt(permits.spent, u128::from(work), ceiling)
135            .map_err(|_| Incomplete::WorkLimit)?;
136        permits.available -= work;
137        permits.spent = spent;
138        Ok(())
139    }
140
141    /// The work committed so far, while leases may still hold permits: a
142    /// lower bound on the work spent, exact once every lease has dropped.
143    pub(crate) fn snapshot(&self, statistics: &mut SearchStatistics) {
144        let work = self.lock();
145        statistics.work = work.spent;
146        statistics.decisions = self.decisions.load(Ordering::Relaxed);
147    }
148
149    /// Every lease has dropped and all workers have joined before this read,
150    /// so the snapshot is exact.
151    pub(crate) fn record(&self, statistics: &mut SearchStatistics) {
152        debug_assert_eq!(self.lock().outstanding, 0);
153        self.snapshot(statistics);
154    }
155}

157/// A query owns at most one grant and never waits while retaining a permit.
158/// The Cells are local to that query; the lease is deliberately not Sync.
159pub(crate) struct WorkLease<'a> {
160    shared: &'a SharedBudget,
161    cancellation: &'a Cancellation,
162    permits: PermitLease,
163}
164
165impl WorkLease<'_> {
166    /// Reserve an accounted kernel's finite work bound before it runs.
167    /// Settle this lease first, so waiting cannot retain the permits needed
168    /// by another reservation. Returned attempts settle their reported work;
169    /// an unwinding attempt conservatively consumes its whole reservation.
170    pub(crate) fn reserve(&mut self, wanted: u64) -> Result<WorkReservation<'_>, Incomplete> {
171        self.settle();
172        loop {
173            self.cancellation.poll()?;
174            let mut work = self.shared.lock();
175            if let Some(granted) = work.reserve(wanted) {
176                return Ok(WorkReservation {
177                    shared: self.shared,
178                    granted,
179                    used: granted,
180                });
181            }
182            drop(
183                self.shared
184                    .returned
185                    .wait_timeout(work, CONTROL_WAIT)
186                    .unwrap_or_else(std::sync::PoisonError::into_inner),
187            );
188        }
189    }
190
191    pub(crate) fn tick(&self) -> Result<(), Incomplete> {
192        self.permits.tick(|| self.refill())
193    }
194
195    pub(crate) fn decide(&self) -> Result<(), Incomplete> {
196        self.shared.decide()
197    }
198
199    /// Consume `amount` permits, refilling grant by grant; a refill that
200    /// finds the allowance exhausted fails after the permits already taken.
201    pub(crate) fn take(&self, amount: u64) -> Result<(), Incomplete> {
202        self.permits.take(amount, || self.refill())
203    }
204
205    /// Take between one and `wanted` permits from the current grant,
206    /// refilling it first when it is spent.
207    pub(crate) fn take_up_to(&self, wanted: u64) -> Result<u64, Incomplete> {
208        self.permits.take_up_to(wanted, || self.refill())
209    }
210
211    /// Return permits taken by [`Self::take_up_to`] and not spent to the
212    /// current grant, from which they came.
213    pub(crate) fn give_back(&self, unspent: u64) {
214        self.permits.give_back(unspent);
215    }
216
217    /// Replenish a common meter's unused borrow. Transfers within the current
218    /// grant keep the existing poll cadence and avoid the shared lock. Before
219    /// renewal, every unused borrowed permit returns to its original grant.
220    pub(crate) fn replenish(
221        &self,
222        work: &mut u64,
223        cancellation: &Cancellation,
224        unused: &mut u64,
225        minimum: u64,
226        capacity: u64,
227    ) -> Result<(), Failure<Incomplete>> {
228        while *unused < minimum {
229            cancellation
230                .poll()
231                .map_err(Incomplete::from)
232                .map_err(Failure::Refused)?;
233            let wanted = POLL_INTERVAL.min(capacity - *unused);
234            let granted = if let Ok(granted) = self.permits.take_up_to(wanted, || Err(())) {
235                granted
236            } else {
237                self.give_back(*unused);
238                *work -= *unused;
239                *unused = 0;
240                self.settle();
241                self.refill_with(|ledger| ledger.grant_at_least(minimum))
242                    .map_err(Failure::Refused)?;
243                // Renewal installed a positive grant. Take its first
244                // prefix now, without another control observation.
245                self.permits
246                    .take_up_to(POLL_INTERVAL.min(capacity), || {
247                        Err(Incomplete::InvalidWitness)
248                    })
249                    .map_err(Failure::Refused)?
250            };
251            *work += granted;
252            *unused += granted;
253        }
254        Ok(())
255    }
256
257    fn refill(&self) -> Result<(), Incomplete> {
258        self.refill_with(Work::grant)
259    }
260
261    fn refill_with(
262        &self,
263        grant: impl Fn(&mut Work) -> Result<Option<u64>, Incomplete>,
264    ) -> Result<(), Incomplete> {
265        loop {
266            if let Err(error) = self.cancellation.poll() {
267                // A refused renewal still commits the consumed grant before
268                // returning, without acquiring another allowance.
269                self.settle();
270                return Err(error.into());
271            }
272            let mut work = self.shared.lock();
273            // The old grant is wholly consumed. Settle it and acquire its
274            // replacement under one lock; no unused permits become available.
275            let renewed = self.permits.renew(|prior| {
276                work.outstanding -= prior.granted();
277                work.spent += prior.used();
278                grant(&mut work)
279            });
280            match renewed {
281                Ok(true) => return Ok(()),
282                Err(error) => {
283                    // Settling the last outstanding grant establishes true
284                    // exhaustion. Peers must wake even without a permit return.
285                    drop(work);
286                    self.shared.returned.notify_all();
287                    return Err(error);
288                }
289                Ok(false) => {}
290            }
291            // Another query can still return unused permits. The timed wait
292            // releases this sole shared lock, and the next iteration polls.
293            // This lease retains no grant and has enabled no waiting peer.
294            drop(
295                self.shared
296                    .returned
297                    .wait_timeout(work, CONTROL_WAIT)
298                    .unwrap_or_else(std::sync::PoisonError::into_inner),
299            );
300        }
301    }
302
303    /// Commit the consumed part of the current grant and return its unused
304    /// part; the lease stays usable and refills on its next use. A holder calls
305    /// this before any wait, delivery or reservation, so a peer never waits for
306    /// permits that this lease is not using.
307    pub(crate) fn settle(&self) {
308        self.permits.settle(|prior| {
309            {
310                let mut work = self.shared.lock();
311                work.outstanding -= prior.granted();
312                work.spent += prior.used();
313                work.available += prior.unused();
314            }
315            self.shared.returned.notify_all();
316        });
317    }
318}
319
320impl Drop for WorkLease<'_> {
321    fn drop(&mut self) {
322        self.settle();
323    }
324}

326/// An accounted operation's pre-reserved allowance. A returned attempt records
327/// its actual work and refunds the unused part. Without a returned receipt,
328/// including unwinding, the whole grant is charged; the caller must report
329/// that failure without establishing enumeration coverage.
330pub(crate) struct WorkReservation<'a> {
331    shared: &'a SharedBudget,
332    granted: u64,
333    used: u64,
334}
335
336impl zetesis_algebra::Reservation for WorkReservation<'_> {
337    fn allowance(&self) -> u64 {
338        self.granted
339    }
340
341    fn commit(mut self, used: u64) {
342        self.used = used;
343    }
344}
345
346impl Drop for WorkReservation<'_> {
347    fn drop(&mut self) {
348        {
349            let mut work = self.shared.lock();
350            work.outstanding -= self.granted;
351            work.spent += self.used;
352            work.available += self.granted - self.used;
353        }
354        self.shared.returned.notify_all();
355    }
356}
