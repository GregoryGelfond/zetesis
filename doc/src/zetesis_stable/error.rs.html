5/// Why no logical decision is available. Every variant leaves coverage open.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum Incomplete {
8    /// Shared cancellation was observed.
9    Cancelled,
10    /// The absolute deadline expired.
11    Deadline,
12    /// Charged SAT work reached its inclusive ceiling.
13    WorkLimit,
14    /// Starting another decision would exceed the ceiling.
15    DecisionLimit,
16    /// Checking another classical candidate would exceed the ceiling.
17    CandidateLimit,
18    /// Pending candidate storage would exceed the caller's byte ceiling.
19    PendingBytes,
20    /// Concurrent completion workspaces and ordered results exceed logical scratch.
21    CompletionScratch,
22    /// A proper-subset query's named workspace storage exceeds its explicit
23    /// byte ceiling.
24    ReductStorage {
25        /// Required named vector/map capacity, excluding shared source payload.
26        required: u128,
27        /// Inclusive ceiling for this query workspace.
28        limit: u128,
29    },
30    /// Additional native-extrema narrowing index capacity exceeds its owner allowance.
31    /// Legacy Boolean index capacity and transient construction peak are excluded.
32    NativeNarrowingStorage {
33        /// Actual retained capacity, or a guaranteed lower bound before construction.
34        required: u128,
35        /// Inclusive allowance for this prepared index owner.
36        limit: u128,
37    },
38    /// A retained batch exceeds the current call's candidate ceiling.
39    BatchCandidateLimit,
40    /// Scalar iteration cannot discard an outstanding candidate batch.
41    PendingBatch,
42    /// Fallible storage reservation failed.
43    Allocation,
44    /// A candidate belongs to a different admitted theory instance.
45    WrongTheory,
46    /// A candidate-only restriction declares a different semantic atom count.
47    RestrictionUniverse {
48        /// Original semantic atom count.
49        expected: usize,
50        /// Restriction's declared atom count.
51        actual: usize,
52    },
53    /// A fused iterator cannot restart after exhaustion or terminal failure.
54    ClosedEnumerator,
55    /// A certificate cannot be configured after candidate generation has begun.
56    LateCertificate,
57    /// The streamed constraints' region check stopped. Its typed receipt is
58    /// retained by [`crate::StreamedConstraints::take_failure`]; this never
59    /// means exhaustion.
60    RegionFilter,
61    /// A region check returned a decision for an atom that is not open.
62    InvalidRegionConsequence,
63    /// Streamed constraints cannot be installed after candidate traversal began.
64    LateRegionFilter,
65    /// Candidate bounds cannot be declared after candidate traversal began:
66    /// the producers' frontier order is fixed for the whole stream.
67    LateCandidateBounds,
68    /// A candidate bound came from another objective plan than the active
69    /// bound's; scores of different plans are not comparable.
70    CandidateBoundPlan,
71    /// A candidate bound's score does not strictly improve on the active
72    /// bound's, so it could readmit candidates that bound pruned.
73    CandidateBoundNotImproving,
74    /// Installed streamed constraints cannot be replaced.
75    RegionFilterAlreadySet,
76    /// A learned-row store cannot be installed after candidate traversal
77    /// began.
78    LateLearnedStore,
79    /// An installed learned-row store cannot be replaced.
80    LearnedStoreAlreadySet,
81    /// Optional candidate certification stopped without establishing membership.
82    Certificate(crate::CertificateError),
83    /// A mandatory arithmetic source read failed; the exact source fault is retained.
84    ArithmeticSource(zetesis_ferraris::AdmissionError),
85    /// Independent formula evaluation stopped without validating a witness.
86    Verification(Stop),
87    /// An internal SAT witness failed independent semantic validation.
88    InvalidWitness,
89    /// A worker unwound before it could account for its remaining search.
90    WorkerPanicked,
91    /// An accounting counter cannot be incremented without overflow.
92    CounterOverflow,
93}
94impl Refusal for Incomplete {
95    fn is_resource_refusal(&self) -> bool {
96        match self {
97            Self::WorkLimit
98            | Self::DecisionLimit
99            | Self::CandidateLimit
100            | Self::PendingBytes
101            | Self::CompletionScratch
102            | Self::ReductStorage { .. }
103            | Self::NativeNarrowingStorage { .. }
104            | Self::BatchCandidateLimit
105            | Self::Allocation => true,
106            Self::Verification(stop) => stop.is_resource_refusal(),
107            Self::ArithmeticSource(error) => error.is_resource_refusal(),
108            Self::Certificate(error) => match error {
109                crate::CertificateError::Tight(error) => error.is_resource_refusal(),
110                crate::CertificateError::Positive(error) => error.is_resource_refusal(),
111                crate::CertificateError::Stratified(error) => error.is_resource_refusal(),
112                crate::CertificateError::Evaluation(error) => error.is_resource_refusal(),
113            },
114            Self::Cancelled
115            | Self::Deadline
116            | Self::PendingBatch
117            | Self::WrongTheory
118            | Self::RestrictionUniverse { .. }
119            | Self::ClosedEnumerator
120            | Self::LateCertificate
121            | Self::RegionFilter
122            | Self::InvalidRegionConsequence
123            | Self::LateRegionFilter
124            | Self::LateCandidateBounds
125            | Self::CandidateBoundPlan
126            | Self::CandidateBoundNotImproving
127            | Self::RegionFilterAlreadySet
128            | Self::LateLearnedStore
129            | Self::LearnedStoreAlreadySet
130            | Self::InvalidWitness
131            | Self::WorkerPanicked
132            | Self::CounterOverflow => false,
133        }
134    }
135
136    fn from_stop(stop: Stop) -> Self {
137        Self::from(stop)
138    }
139}

141impl From<Stop> for Incomplete {
142    fn from(error: Stop) -> Self {
143        match error {
144            Stop::Cancelled => Self::Cancelled,
145            Stop::Deadline => Self::Deadline,
146            Stop::Allocation => Self::Allocation,
147            Stop::WrongProgram => Self::WrongTheory,
148            _ => Self::Verification(error),
149        }
150    }
151}
152impl fmt::Display for Incomplete {
153    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
154        match self {
155            Self::Cancelled => f.write_str("SAT operation cancelled"),
156            Self::Deadline => f.write_str("SAT operation deadline expired"),
157            Self::WorkLimit => f.write_str("SAT work limit reached"),
158            Self::DecisionLimit => f.write_str("SAT decision limit reached"),
159            Self::CandidateLimit => f.write_str("stable candidate limit reached"),
160            Self::CompletionScratch => f.write_str("completion logical scratch byte limit reached"),
161            Self::ReductStorage { required, limit } => write!(
162                f,
163                "reduct query storage requires {required} bytes; limit is {limit}"
164            ),
165            Self::NativeNarrowingStorage { required, limit } => write!(
166                f,
167                "native aggregate search index needs {required} bytes; limit is {limit}"
168            ),
169            Self::PendingBytes => f.write_str("pending candidate byte limit reached"),
170            Self::BatchCandidateLimit => f.write_str("pending batch candidate limit reached"),
171            Self::PendingBatch => f.write_str("an unresolved candidate batch remains"),
172            Self::Allocation => f.write_str("SAT storage reservation failed"),
173            Self::WrongTheory => f.write_str("candidate belongs to a different theory"),
174            Self::RestrictionUniverse { expected, actual } => write!(
175                f,
176                "candidate restriction declares {actual} atoms; expected {expected}"
177            ),
178            Self::ClosedEnumerator => f.write_str("candidate enumeration is already closed"),
179            Self::LateCertificate => {
180                f.write_str("certificate configuration requires an unstarted candidate stream")
181            }
182            Self::RegionFilter => {
183                f.write_str("candidate-region filter failed; inspect its receipt")
184            }
185            Self::InvalidRegionConsequence => {
186                f.write_str("candidate-region consequence does not decide an open atom")
187            }
188            Self::LateRegionFilter => {
189                f.write_str("candidate-region filtering requires an unstarted stream")
190            }
191            Self::LateCandidateBounds => {
192                f.write_str("declaring candidate bounds requires an unstarted stream")
193            }
194            Self::CandidateBoundPlan => {
195                f.write_str("a candidate bound must come from the active bound's objective plan")
196            }
197            Self::CandidateBoundNotImproving => {
198                f.write_str("a candidate bound's score must strictly improve on the active bound's")
199            }
200            Self::RegionFilterAlreadySet => {
201                f.write_str("candidate-region filter is already installed")
202            }
203            Self::LateLearnedStore => {
204                f.write_str("sharing learned rows requires an unstarted stream")
205            }
206            Self::LearnedStoreAlreadySet => f.write_str("a learned-row store is already installed"),
207            Self::Certificate(error) => write!(f, "certified membership: {error}"),
208            Self::ArithmeticSource(error) => write!(f, "arithmetic source: {error}"),
209            Self::Verification(error) => write!(f, "independent verification: {error}"),
210            Self::InvalidWitness => f.write_str("SAT witness failed independent verification"),
211            Self::WorkerPanicked => f.write_str("SAT worker panicked before completing its search"),
212            Self::CounterOverflow => f.write_str("SAT accounting counter overflow"),
213        }
214    }
215}
216impl std::error::Error for Incomplete {
217    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
218        match self {
219            Self::ArithmeticSource(error) => Some(error),
220            Self::Verification(error) => Some(error),
221            Self::Certificate(error) => Some(error),
222            _ => None,
223        }
224    }
