51/// Whole-operation ceilings for a membership check or stable-model enumeration.
52#[derive(Clone, Copy, Debug)]
53pub struct Limits {
54    /// Cumulative region search, certificate and query work and decisions
55    /// across a run.
56    pub search: SearchLimits,
57    /// Maximum classical candidates checked during enumeration; not model count.
58    pub max_candidates: u64,
59    /// Per-call work ceiling for independent original/reduct formula evaluation.
60    pub max_verification_work: u64,
61    /// Retained capacity of each proper-subset query's workspace.
62    /// Optional inner arithmetic preparation applies the same ceiling to
63    /// its own peak of simultaneously live storage. If preparation exceeds
64    /// it, the query continues with ordinary reduct search.
65    /// Original candidate generator ordering uses the same ceiling independently
66    /// for its transient graph and retained ranks; a named byte refusal declines
67    /// that optional preference, with its work prefix retained in search work.
68    pub max_reduct_bytes: u64,
69    /// Additional retained native-extrema narrowing index bytes per owner.
70    /// This excludes legacy Boolean indexes and transient construction peak.
71    /// Successful preparation admits actual capacity before publishing an index.
72    pub max_native_narrowing_bytes: u64,
73    /// Allowances of the rows a walk learns: each tier's rows and capture,
74    /// the shared store's rows, and what and when a tier shares. Exhausting
75    /// one disables that learning and never ends the enumeration.
76    pub learning: zetesis_ferraris::LearningLimits,
77}

84            max_reduct_bytes: crate::reduct::DEFAULT_BYTES,

91impl Limits {
92    /// What these limits leave to a later stream of the same run once an
93    /// earlier stream has spent `spent`: the cumulative work, decision and
94    /// candidate ceilings less what it charged, at least zero. Per-call and
95    /// capacity ceilings are each stream's own and stay as they are. A run
96    /// that searches in passes constructs each later pass's stream with this,
97    /// so its passes together stay within the run's limits.
98    #[must_use]
99    pub const fn remaining_after(self, spent: &Statistics) -> Self {
100        Self {
101            search: SearchLimits {
102                max_work: self.search.max_work.saturating_sub(spent.search.work),
103                max_decisions: self
104                    .search
105                    .max_decisions
106                    .saturating_sub(spent.search.decisions),
107            },
108            max_candidates: self.max_candidates.saturating_sub(spent.candidates),
109            ..self
110        }
111    }

114/// Low-level membership verdict data. Native [`check`] returns a verdict for its
115/// supplied subject; the enum itself carries no binding to that subject or its
116/// theory and can also be constructed by callers. For a record that retains the
117/// subject of an actual native check, use [`crate::check_interpretation`].
118#[derive(Clone, Debug)]
119pub enum Check {
120    /// Original model whose frozen reduct has no proper-subset model.
121    Stable,
122    /// The candidate does not satisfy the original theory.
123    NotModel,
124    /// Independently validated proper-subset model of the frozen reduct.
125    NonMinimal(Interpretation),
126    /// No semantic membership decision can be made within available resources.
127    Inconclusive(Incomplete),
128}
129impl Check {
130    /// Whether this verdict is the `Stable` variant. This inspects data only;
131    /// it does not run a check or authenticate a caller-constructed verdict.
132    #[must_use]
133    pub fn accepted(&self) -> bool {
134        matches!(self, Self::Stable)
135    }
136}

189/// Check stability through the proper-subset query of the frozen reduct, a
190/// region tree narrowed by the reduct's readings. A countermodel is validated
191/// independently before it refutes the candidate. No enumeration of all
192/// subsets or invocation of an external solver occurs.
193/// `max_candidates` applies only to [`StableModels`].
194#[must_use]
195pub fn check(
196    theory: &Theory,
197    candidate: &Interpretation,
198    limits: Limits,
199    cancellation: &Cancellation,
200) -> Check {
201    let mut budget = Budget {
202        quota: crate::budget::LocalQuota,
203        limits: limits.search,
204        cancellation,
205        statistics: SearchStatistics::default(),
206    };
207    let mut statistics = Statistics::default();
208    match fresh_membership(theory, candidate, limits, &mut budget, &mut statistics) {
209        Ok(result) => result,
210        Err(error) => Check::Inconclusive(error),
211    }
212}
213
214fn fresh_membership(
215    theory: &Theory,
216    candidate: &Interpretation,
217    limits: Limits,
218    budget: &mut Budget<'_, impl Quota>,
219    statistics: &mut Statistics,
220) -> Result<Check, Incomplete> {
221    let original = original_model(theory, candidate, limits, budget, statistics);
222    if !original? {
223        return Ok(Check::NotModel);
224    }
225    let started = timing::start(statistics.phase_timings.as_ref());
226    // The one standalone holder: its index is built here, inside the reduct
227    // phase, and charged to the reduct's region counts.
228    let mut original = OriginalIndex::new(theory, limits.max_native_narrowing_bytes);
229    let result = original
230        .ensure(budget.cancellation, |work| {
231            budget.charge(work)?;
232            statistics.reduct.regions.work = statistics
233                .reduct
234                .regions
235                .work
236                .checked_add(work)
237                .ok_or(Incomplete::CounterOverflow)?;
238            Ok(())
239        })
240        .and_then(|index| {
241            crate::reduct::State::default().check(
242                theory,
243                Some(index),
244                candidate,
245                limits,
246                budget,
247                statistics,
248            )
249        });
250    timing::finish(&mut statistics.phase_timings, Phase::Reduct, started);
251    result
252}

254pub(crate) fn original_model(
255    theory: &Theory,
256    candidate: &Interpretation,
257    limits: Limits,
258    budget: &Budget<'_, impl Quota>,
259    statistics: &mut Statistics,
260) -> Result<bool, Incomplete> {
261    let started = timing::start(statistics.phase_timings.as_ref());
262    let original = (|| {
263        budget.cancellation.poll()?;
264        if !theory.same_instance(candidate.theory()) {
265            return Err(Incomplete::WrongTheory);
266        }
267        models(theory, candidate, verification(limits), budget.cancellation)
268            .map_err(Incomplete::from)
269    })();
270    timing::finish(
271        &mut statistics.phase_timings,
272        Phase::OriginalValidation,
273        started,
274    );
275    original
276}

278/// Validate a proposed countermodel independently: it must be a proper
279/// subset of the candidate and model the candidate's frozen reduct. Every
280/// proposer, the frozen-positive closure and the region query alike, is held
281/// to this.
282pub(crate) fn checked_countermodel(
283    theory: &Theory,
284    candidate: &Interpretation,
285    subset: Interpretation,
286    limits: Limits,
287    budget: &mut Budget<'_, impl Quota>,
288    statistics: &mut Statistics,
289) -> Result<Check, Incomplete> {
290    validated_countermodel(theory, candidate, subset, limits, budget, statistics)
291        .map(Check::NonMinimal)
292}
293
294/// [`checked_countermodel`]'s validation, returning the countermodel it
295/// accepts: a served region of a published query records it as its query's
296/// witness rather than as a verdict.
297pub(crate) fn validated_countermodel(
298    theory: &Theory,
299    candidate: &Interpretation,
300    subset: Interpretation,
301    limits: Limits,
302    budget: &mut Budget<'_, impl Quota>,
303    statistics: &mut Statistics,
304) -> Result<Interpretation, Incomplete> {
305    let mut proper = false;
306    for atom in 0..theory.atom_count() {
307        budget.tick()?;
308        if subset.contains(atom) && !candidate.contains(atom) {
309            return Err(Incomplete::InvalidWitness);
310        }
311        proper |= candidate.contains(atom) && !subset.contains(atom);
312    }
313    if !proper
314        || !models_reduct(
315            theory,
316            candidate,
317            &subset,
318            verification(limits),
319            budget.cancellation,
320        )?
321    {
322        return Err(Incomplete::InvalidWitness);
323    }
324    increment(&mut statistics.countermodels)?;
325    Ok(subset)
326}

406    /// Enumerate by regions with up to `workers` walking the tree at once:
407    /// the walk begins on the calling thread and starts the workers once it
408    /// has run, in wall time since it began, for longer than starting them
409    /// costs (see [`Self::with_region_workers_engaging`]). Each worker has its
410    /// own knowledge, budget lease and reduct query state, sharing the
411    /// immutable original index and stealing from per-worker deques of
412    /// regions still to visit. A candidate's proper-subset query that runs
413    /// long while a worker is idle is divided among the workers
414    /// ([`Self::set_query_division`]). Deque growth can return an allocation
415    /// refusal.
416    /// The models arrive in the scalar walk's order until the workers start,
417    /// and in the schedule's order after, which is not a property of the
418    /// result and differs between runs; the family is exact. One worker is
419    /// the scalar regions method.
420    ///
421    /// Use [`Self::prepare_with_region_workers`] to retain the construction receipt on failure.
422    ///
423    /// # Errors

434    /// Enumerate by regions with up to `workers` walking the tree at once,
435    /// starting them as `engagement` says.
436    ///
437    /// The walk begins at the first pull on the calling thread, and returns the
438    /// models it verifies there in the scalar walk's order. Once its engagement
439    /// delay has passed, it starts `workers` threads, which continue from its
440    /// remaining frontier. The delay is wall time since the walk began, the
441    /// caller's time between pulls included, checked between the walk's steps.
442    /// Models then arrive in the schedule's order, which is not a property of the
443    /// result; the family is exact under every engagement.
444    ///
445    /// The enumeration owns the threads and joins them when it is exhausted,
446    /// stopped or dropped. Until it engages it owns one worker's workspace instead,
447    /// released when the walk ends, engages, stops or is dropped. One worker is the
448    /// scalar regions method, and `engagement` is then ignored.
449    ///
450    /// # Errors
451    /// Refuses admission, work limits, cancellation or allocation, as
452    /// [`Self::with_region_workers`] does. Threads are created at engagement, not
453    /// here. A thread creation the host refuses then stops the enumeration with
454    /// `Incomplete::Allocation`, after the models already returned: the threads
455    /// already started are joined and the models they queued are discarded. A
456    /// refused thread creation never establishes exhaustion.
457    ///
458    /// # Costs
459    /// A walk that ends within its delay starts no thread. Against
460    /// `Engagement::Immediate`, a walk that engages finishes at most about its
461    /// delay plus one step later. Until engagement the walk advances only inside
462    /// pulls, so the caller's per-model work does not overlap it.
463    pub fn with_region_workers_engaging(
464        theory: &Theory,
465        workers: std::num::NonZeroUsize,
466        engagement: Engagement,
467        limits: Limits,
468        cancellation: Cancellation,
469    ) -> Result<Self, Incomplete> {
470        Self::prepare_route(
471            theory,
472            limits,
473            cancellation,
474            Route::Workers {
475                workers,
476                engagement,
477            },
478        )
479        .result
480    }

547    fn prepare_route(
548        theory: &Theory,
549        limits: Limits,
550        cancellation: Cancellation,
551        route: Route,
552    ) -> ConstructionAttempt {
553        let route = match route {
554            Route::Workers { workers, .. } | Route::Producers(workers) if workers.get() == 1 => {
555                Route::Scalar
556            }
557            route => route,
558        };
559        let mut budget = Budget {
560            quota: crate::budget::LocalQuota,
561            limits: limits.search,
562            cancellation: &cancellation,
563            statistics: SearchStatistics::default(),
564        };
565        // Allocate the terminal holder before a conclusive preparation. The
566        // completed proof must not be followed by a scheduler allocation.
567        let mut refuted = Box::new(RefutedRoot {
568            parts: None,
569            statistics: RegionSearchStatistics::default(),
570            filter: None,
571        });
572        let opening = regions::open_attempt(
573            theory,
574            limits.max_reduct_bytes,
575            limits.learning,
576            &mut budget,
577        );
578        let mut regions = opening.statistics;
579        // The opening's producers seed the theory's index, so a candidate's
580        // inner preparation reads them without a second extraction.
581        let proposer = opening.result.and_then(|opened| {
582            let producers = opened.parts.producers().cloned();
583            if opened.root.is_none() {
584                refuted.parts = Some(opened.parts);
585                refuted.statistics = opened.statistics;
586                return Ok((Proposer::Refuted(refuted), producers));
587            }
588            let proposer = match route {
589                Route::Scalar => RegionSearch::new(opened, limits.max_native_narrowing_bytes)
590                    .map(|search| Proposer::Regions(Box::new(search))),
591                Route::Workers {
592                    workers,
593                    engagement,
594                } => ParallelRegions::new(
595                    opened,
596                    workers,
597                    engagement,
598                    limits,
599                    cancellation.clone(),
600                    &mut budget,
601                )
602                .map(|search| Proposer::Parallel(Box::new(search))),
603                Route::Producers(workers) => {
604                    RegionProposals::new(opened, workers, limits.max_native_narrowing_bytes)
605                        .map(|search| Proposer::Proposals(Box::new(search)))
606                }
607            }?;
608            Ok((proposer, producers))
609        });
610        if proposer.is_err() {
611            regions.generator_order.retained_bytes = 0;
612            regions.parity.retained_bytes = 0;
613            regions.cardinality.retained_bytes = 0;
614        }
615        let statistics = Statistics {
616            search: budget.statistics,
617            regions: Some(regions),
618            ..Default::default()
619        };
620        let result = proposer.map(|(proposer, producers)| Self {
621            theory: theory.clone(),
622            proposer,
623            limits,
624            cancellation,
625            statistics,
626            terminal: false,
627            exhausted: false,
628            pending_error: None,
629            batch: batch::State::default(),
630            certificate: None,
631            certificate_workspace: certified::Workspace::default(),
632            determined_candidates: None,
633            bound_generation: 0,
634            active_bound: None,
635            earlier: None,
636            reduct: crate::reduct::State::default(),
637            index: OriginalIndex::seeded(theory, limits.max_native_narrowing_bytes, producers),
638        });
639        let statistics = result.as_ref().map_or(statistics, Self::statistics);
640        ConstructionAttempt { result, statistics }
641    }

656    /// Enable coarse host timing from this point onward. Repeated calls retain
657    /// existing measurements. No timing clock is read while timing is
658    /// disabled; a parallel region walk reads the clock between its
659    /// coordinator's steps until its workers start, to keep its engagement
660    /// deadline. A walk's timing is fixed when the walk begins.

671    /// Append a classical candidate-only constraint. The region proposer
672    /// narrows the regions still to visit by it and continues; a positive
673    /// cursor tests the least interpretation against these same restrictions.
674    /// Original theory and reduct acceptance stay unchanged.
675    ///
676    /// The restriction must use the original semantic atom count and index
677    /// meanings. Its separate immutable instance is expected. Restrictions
678    /// accumulate by conjunction; this method cannot widen a candidate region.
679    /// Exhaustion afterwards proves coverage only of that constrained region.
680    /// An optimizer must separately justify that omitted candidates are dominated.
681    ///
682    /// # Errors
683    /// Refuses a different atom count, a closed iterator, a pending failure,
684    /// capacity, work, cancellation or allocation; charged work remains
685    /// cumulative.
686    pub fn restrict_candidates(&mut self, restriction: &Theory) -> Result<(), Incomplete> {
687        if self.terminal {
688            return Err(Incomplete::ClosedEnumerator);
689        }
690        if let Some(error) = self.pending_error {
691            return Err(error);
692        }
693        if restriction.atom_count() != self.theory.atom_count() {
694            return Err(Incomplete::RestrictionUniverse {
695                expected: self.theory.atom_count(),
696                actual: restriction.atom_count(),
697            });
698        }
699        let count = self
700            .statistics
701            .candidate_restrictions
702            .checked_add(1)
703            .ok_or(Incomplete::CounterOverflow)?;
704        let mut budget = Budget {
705            quota: crate::budget::LocalQuota,
706            limits: self.limits.search,
707            cancellation: &self.cancellation,
708            statistics: self.statistics.search,
709        };
710        let result = self.proposer.restrict(restriction, &mut budget);
711        self.statistics.search = budget.statistics;
712        if result.is_ok() {
713            self.statistics.candidate_restrictions = count;
714        }
715        result
716    }

718    /// Install the bound an objective plan compiled at an incumbent's `score`:
719    /// a candidate-only restriction to the interpretations costing less than
720    /// the incumbent, or at most as much, under that `plan`.
721    ///
722    /// # Checked
723    /// The first bound is accepted as given. Every later bound must come from
724    /// the same `plan`, and its `score` must be strictly better than the active
725    /// bound's (`Score::compare_costs` is `Less`); otherwise it is refused with
726    /// [`Incomplete::CandidateBoundPlan`] or
727    /// [`Incomplete::CandidateBoundNotImproving`], and the active bound stays.
728    /// A plan's bound at a strictly better score keeps only interpretations its
729    /// bound at the worse score keeps too, strict or not, so regions pruned
730    /// under an earlier bound never hold a candidate a later bound admits.
731    /// `restriction` must be the bound `plan` compiled at `score`; this
732    /// operation checks the plan and the score, not the restriction's formula.
733    /// It must use the original atom count and semantic index meanings. Bounds
734    /// never support original atoms and never change the original theory or
735    /// its reduct.
736    ///
737    /// Region search retains only the latest bound separately from permanent
738    /// [`Self::restrict_candidates`] constraints. Active worker snapshots may
739    /// finish under an older bound, and already pending candidates are retained;
740    /// their original membership checks still apply. An optimizer must compare
741    /// every returned model with its current incumbent. Exhaustion covers the
742    /// remaining constrained family, not the original unrestricted world view.
743    ///
744    /// # Errors
745    /// Refuses a closed iterator, a pending error, a different atom count,
746    /// another plan or a score that does not improve, generation overflow,
747    /// resource exhaustion or cancellation. Failed setup leaves the active
748    /// bound and generation unchanged; admitted work stays charged. Region
749    /// knowledge retains its existing infallible allocation contract; this
750    /// operation does not make all search allocation fallible.
751    pub fn improve_candidate_bound(
752        &mut self,
753        plan: PlanIdentity,
754        score: &Score,
755        restriction: &Theory,
756    ) -> Result<(), Incomplete> {
757        if self.terminal {
758            return Err(Incomplete::ClosedEnumerator);
759        }
760        if let Some(error) = self.pending_error {
761            return Err(error);
762        }
763        if restriction.atom_count() != self.theory.atom_count() {
764            return Err(Incomplete::RestrictionUniverse {
765                expected: self.theory.atom_count(),
766                actual: restriction.atom_count(),
767            });
768        }
769        if let Some((active, previous)) = &self.active_bound {
770            if *active != plan {
771                return Err(Incomplete::CandidateBoundPlan);
772            }
773            if score.compare_costs(previous) != std::cmp::Ordering::Less {
774                return Err(Incomplete::CandidateBoundNotImproving);
775            }
776        }
777        let generation = self
778            .bound_generation
779            .checked_add(1)
780            .ok_or(Incomplete::CounterOverflow)?;
781        let count = self
782            .statistics
783            .candidate_restrictions
784            .checked_add(1)
785            .ok_or(Incomplete::CounterOverflow)?;
786        let mut budget = Budget {
787            quota: crate::budget::LocalQuota,
788            limits: self.limits.search,
789            cancellation: &self.cancellation,
790            statistics: self.statistics.search,
791        };
792        let result = self.proposer.tighten(restriction, generation, &mut budget);
793        self.statistics.search = budget.statistics;
794        if result.is_ok() {
795            self.bound_generation = generation;
796            self.active_bound = Some((plan, score.clone()));
797            self.statistics.candidate_restrictions = count;
798        }
799        result
800    }

802    /// Declare, before the stream's first proposal, that the caller will
803    /// improve candidate bounds ([`Self::improve_candidate_bound`]) as it
804    /// commits answers, as an optimizing search does. Region producers
805    /// ([`Self::with_region_producers`]) then share one frontier
806    /// ([`FrontierOrder::Shared`]) for the whole stream: every producer takes
807    /// the region any producer published most recently, so a slate fills from
808    /// the area the latest splits opened, and older, shallower regions wait for
809    /// the tighter bounds that refute them near the root. Without the
810    /// declaration each producer keeps its own deque, the order an enumeration
811    /// prefers. Other routes are unaffected, and answer sets are unchanged
812    /// either way.
813    ///
814    /// # Errors
815    /// Refuses a closed or started stream, so one order governs every slate
816    /// and the region statistics' `frontier_order` names the order the
817    /// stream's producers take regions in.
818    pub fn expect_candidate_bounds(&mut self) -> Result<(), Incomplete> {
819        if self.terminal {
820            return Err(Incomplete::ClosedEnumerator);
821        }
822        if self.started() {
823            return Err(Incomplete::LateCandidateBounds);
824        }
825        if let Proposer::Proposals(proposals) = &mut self.proposer {
826            proposals.share_frontier();
827        }
828        // A slate produced ahead would be produced under a bound its
829        // predecessor's answers may tighten, so such a stream never proposes ahead.
830        self.batch.bounds_expected = true;
831        Ok(())
832    }

927    /// Fuse future pulls and join candidate workers without cancelling the
928    /// caller's shared token. Statistics retain every joined worker's prefix;
929    /// unresolved batch entries remain visible in their pending receipt.
930    /// Queued worker models are discarded, not counted as delivered answers.
931    /// An early stop cannot establish exhaustion; prior exhaustion is retained.
932    ///
933    /// # Errors
934    /// Reports the first worker stop or a failure to merge joined receipts.
935    /// All workers are joined even on failure; the stream remains fused.
936    pub fn stop(&mut self) -> Result<(), Incomplete> {
937        self.terminal = true;
938        let production = self.proposer.finish_production();
939        if let Proposer::Parallel(parallel) = &mut self.proposer {
940            let result = parallel.stop();
941            let joined = parallel.search_statistics();
942            // Before the first pull, certificate preparation can have charged
943            // the coordinator while the workers still hold construction totals.
944            self.statistics.search.work = self.statistics.search.work.max(joined.work);
945            self.statistics.search.decisions =
946                self.statistics.search.decisions.max(joined.decisions);
947            return result;
948        }
949        production
950    }

991    /// This stream's own receipts, which decide whether it has started.
992    fn own_statistics(&self) -> Statistics {
993        let mut statistics = match &self.proposer {
994            Proposer::Refuted(root) => Statistics {
995                regions: Some(root.statistics),
996                ..self.statistics
997            },
998            Proposer::Regions(regions) => Statistics {
999                regions: Some(regions.statistics()),
1000                ..self.statistics
1001            },
1002            Proposer::Proposals(proposals) => Statistics {
1003                regions: Some(proposals.statistics()),
1004                ..self.statistics
1005            },
1006            Proposer::Parallel(parallel) if self.determined_candidates.is_none() => {
1007                let merged = parallel.merged();
1008                let merged = &merged;
1009                let mut certified = self.statistics.certified;
1010                if let (Some(into), Some(from)) = (certified.as_mut(), merged.certified.as_ref()) {
1011                    into.checks = from.checks;
1012                    into.stable = from.stable;
1013                    into.refuted = from.refuted;
1014                    into.failed = from.failed;
1015                    into.checking_work = from.checking_work;
1016                    into.tight_check_peak_bytes = from.tight_check_peak_bytes;
1017                    into.positive_check_peak_bytes = from.positive_check_peak_bytes;
1018                    into.stratified_check_peak_bytes = from.stratified_check_peak_bytes;
1019                }
1020                Statistics {
1021                    regions: Some(parallel.statistics()),
1022                    candidates: merged.candidates,
1023                    countermodel_queries: merged.countermodel_queries,
1024                    countermodels: merged.countermodels,
1025                    certified,
1026                    // The walk's coordinator and worker phases, plus the
1027                    // enumeration's index build before the walk began.
1028                    phase_timings: crate::timing::combined(
1029                        merged.phase_timings,
1030                        self.statistics.phase_timings,
1031                    ),
1032                    reduct: crate::ReductStatistics {
1033                        original_work: merged.reduct.original_work,
1034                        frozen_positive_attempts: merged.reduct.frozen_positive_attempts,
1035                        frozen_positive_completed: merged.reduct.frozen_positive_completed,
1036                        frozen_positive_preparation_work: merged
1037                            .reduct
1038                            .frozen_positive_preparation_work,
1039                        frozen_positive_work: merged.reduct.frozen_positive_work,
1040                        peak_workspace_bytes: merged.reduct.peak_workspace_bytes,
1041                        regions: merged.reduct.regions,
1042                        inner_preparation: merged.reduct.inner_preparation,
1043                        ..self.statistics.reduct
1044                    },
1045                    ..self.statistics
1046                }
1047            }
1048            Proposer::Parallel(parallel) => Statistics {
1049                regions: Some(parallel.statistics()),
1050                ..self.statistics
1051            },
1052        };

1054            .proposer

1056            .map(crate::region_filter::Filter::statistics);

1058    }

1060    /// Report `earlier`'s receipts with this stream's, as one search over the
1061    /// same theory that an optimizer runs in passes, such as a tie pass after
1062    /// the pass that proved its optimum: from now on [`Self::statistics`] and
1063    /// [`Self::batch_statistics`] report `earlier`'s final receipts, its own
1064    /// earlier passes included, followed by this stream's. Work and counts
1065    /// add, peaks take the larger, and what describes the route as it stands
1066    /// is this stream's. Membership, exhaustion and every limit stay this
1067    /// stream's own; only the receipts combine, so a run that must stay within
1068    /// its limits across passes constructs this stream with
1069    /// [`Limits::remaining_after`] `earlier`'s statistics. Calling it again
1070    /// replaces the earlier receipts.
1071    ///
1072    /// # Errors

1075        if !earlier.theory.same_instance(&self.theory) {
1076            return Err(Incomplete::WrongTheory);
1077        }
1078        self.earlier = Some(Box::new(passes::Earlier {
1079            statistics: earlier.statistics(),
1080            batch: earlier.batch_statistics(),
1081        }));
1082        Ok(())
1083    }

1114impl Iterator for StableModels {
1115    type Item = Result<Interpretation, Incomplete>;
1116    fn next(&mut self) -> Option<Self::Item> {
1117        if self.terminal {
1118            return None;
1119        }
1120        if !self.batch.pending.is_empty() {
1121            self.terminal = true;
1122            let _ = self.proposer.finish_production();
1123            return Some(Err(Incomplete::PendingBatch));
1124        }
1125        if let Some(error) = self.pending_error.take() {
1126            self.terminal = true;
1127            let _ = self.proposer.finish_production();
1128            return Some(Err(error));
1129        }
1130        match self.advance() {
1131            Ok(Some(model)) => Some(Ok(model)),
1132            Ok(None) => {
1133                self.terminal = true;
1134                if let Err(error) = self.proposer.finish_production() {
1135                    return Some(Err(error));
1136                }
1137                self.exhausted = true;
1138                None
1139            }
1140            Err(error) => {
1141                // Terminal evidence includes every worker's settled prefix.
1142                // Cleanup cannot replace the interruption that stopped this pull.
1143                let _ = self.stop();
1144                Some(Err(error))
1145            }
1146        }
1147    }
1148}

1272    /// Record the original index's work, charged when a region walk first
1273    /// needed it, once in this proposer's region counts.
1274    fn record_index_work(&mut self, work: u64) -> Result<(), Incomplete> {
1275        let counts = match self {
1276            Self::Refuted(root) => &mut root.statistics.counts,
1277            Self::Regions(regions) => regions.counts_mut(),
1278            Self::Parallel(parallel) => parallel.counts_mut(),
1279            Self::Proposals(proposals) => proposals.counts_mut(),
1280        };
1281        counts.work = counts
1282            .work
1283            .checked_add(work)
1284            .ok_or(Incomplete::CounterOverflow)?;
1285        Ok(())
1286    }
1287
1288    /// The next classical candidate, from the positive cursor when one is
1289    /// active and else from this proposer's own walk, or `None` when the
1290    /// candidate space is covered. A proposal is refused, not returned, once
1291    /// the candidate ceiling is reached; the caller admits it. A region walk
1292    /// reads the index `walk` carries, which [`walk_index`] built.
1293    fn propose(
1294        &mut self,
1295        theory: &Theory,
1296        walk: Walk<'_>,
1297        limits: Limits,
1298        certificate: Option<&std::sync::Arc<certified::Certification>>,
1299        budget: &mut Budget<'_>,
1300        statistics: &mut Statistics,
1301    ) -> Result<Option<Proposal>, Incomplete> {
1302        let index = match walk {
1303            Walk::Determined(candidates) => {
1304                return candidates
1305                    .propose(self, theory, limits, budget, statistics)
1306                    .map(|candidate| candidate.map(Proposal::Candidate));
1307            }
1308            Walk::Index(index) => index,
1309        };
1310        let proposal = match self {
1311            Self::Refuted(_) => return Ok(None),
1312            Self::Regions(regions) => {
1313                let proposal =
1314                    regions.propose(theory, index, budget, &mut statistics.phase_timings)?;
1315                if proposal.is_some() && statistics.candidates >= limits.max_candidates {
1316                    return Err(Incomplete::CandidateLimit);
1317                }
1318                proposal
1319            }
1320            Self::Parallel(parallel) => {
1321                let timed = statistics.phase_timings.is_some();
1322                return Ok(parallel
1323                    .propose(index, certificate, timed, budget)?
1324                    .map(Proposal::Stable));
1325            }
1326            Self::Proposals(proposals) => {
1327                let subject = index.subject(theory)?;
1328                let mut output = crate::budget::storage(1)?;
1329                let produced = proposals.fill(
1330                    subject,
1331                    1,
1332                    limits.max_candidates.saturating_sub(statistics.candidates),
1333                    budget,
1334                    &mut output,
1335                    statistics.phase_timings.is_some(),
1336                );
1337                if let Some(timings) = statistics.phase_timings.as_mut() {
1338                    timings
1339                        .original_validation
1340                        .merge(produced.original_validation);
1341                }
1342                // A slate of one candidate has one hand, which a peer reserves
1343                // only after the first producer emits, and then finds the
1344                // output spoken for, so no second producer can stop after the
1345                // first emits. Batched calls retain that richer outcome.
1346                if let Some(error) = produced.stopped {
1347                    return Err(error);
1348                }
1349                output.pop()
1350            }
1351        };
1352        Ok(proposal.map(Proposal::Candidate))
1353    }

1355    fn tighten(
1356        &mut self,
1357        restriction: &Theory,
1358        generation: u64,
1359        budget: &mut Budget<'_>,
1360    ) -> Result<(), Incomplete> {
1361        match self {
1362            Self::Refuted(_) => Ok(()),
1363            Self::Regions(regions) => regions.tighten(restriction, generation, budget),
1364            Self::Parallel(parallel) => parallel.tighten(restriction, generation, budget),
1365            Self::Proposals(proposals) => proposals.tighten(restriction, generation, budget),
1366        }
1367    }

1369    /// Restrict every later proposal to the classical models of `restriction`.
1370    fn restrict(
1371        &mut self,
1372        restriction: &Theory,
1373        budget: &mut Budget<'_>,
1374    ) -> Result<(), Incomplete> {
1375        match self {
1376            Self::Refuted(_) => Ok(()),
1377            Self::Regions(regions) => regions.restrict(restriction, budget),
1378            Self::Parallel(parallel) => parallel.restrict(restriction, budget),
1379            Self::Proposals(proposals) => proposals.restrict(restriction, budget),
1380        }
1381    }

1384fn advance(
1385    membership_input: Membership<'_>,
1386    proposer: &mut Proposer,
1387    budget: &mut Budget<'_>,
1388    statistics: &mut Statistics,
1389) -> Result<Option<Interpretation>, Incomplete> {
1390    // No index, certificate workspace or candidate storage is needed after an
1391    // original-answer preparation has already established the empty family.
1392    if proposer.root_refuted() {
1393        return Ok(None);
1394    }
1395    let Membership {
1396        theory,
1397        limits,
1398        certificate,
1399        certificate_workspace,
1400        reduct,
1401        index,
1402        mut determined_candidates,
1403    } = membership_input;
1404    let index = if determined_candidates.is_none() {
1405        Some(walk_index(
1406            index,
1407            proposer,
1408            budget,
1409            &mut statistics.phase_timings,
1410        )?)
1411    } else {
1412        None
1413    };
1414    loop {
1415        // The parallel walk's workers time their own phases; the wait for
1416        // their models is not a phase.
1417        let started = match proposer {
1418            Proposer::Parallel(_) if determined_candidates.is_none() => None,
1419            _ => timing::start(statistics.phase_timings.as_ref()),
1420        };
1421        let walk = match (determined_candidates.as_deref_mut(), index) {
1422            (Some(candidates), _) => Walk::Determined(candidates),
1423            (None, Some(index)) => Walk::Index(index),
1424            (None, None) => return Err(Incomplete::InvalidWitness),
1425        };
1426        let proposal = proposer.propose(theory, walk, limits, certificate, budget, statistics);
1427        timing::finish(&mut statistics.phase_timings, Phase::Candidates, started);
1428        let candidate = match proposal? {
1429            None => return Ok(None),
1430            Some(Proposal::Stable(model)) => {
1431                // Decided by a worker; the coordinator only counts it.
1432                increment(&mut statistics.stable_models)?;
1433                return Ok(Some(model));
1434            }
1435            Some(Proposal::Candidate(candidate)) => candidate,
1436        };
1437        increment(&mut statistics.candidates)?;
1438        let decision: Decision = if let Some(certificate) = certificate {
1439            certified::classify(
1440                certificate,
1441                certificate_workspace,
1442                &candidate,
1443                limits,
1444                budget.cancellation,
1445                statistics,
1446                &mut budget.statistics,
1447            )?
1448            .into()
1449        } else {
1450            reduct
1451                .check(
1452                    theory,
1453                    index.map(std::sync::Arc::as_ref),
1454                    &candidate,
1455                    limits,
1456                    budget,
1457                    statistics,
1458                )?
1459                .into()
1460        };
1461        if decision == Decision::Invalid {
1462            return Err(Incomplete::InvalidWitness);
1463        }
1464        if decision == Decision::Stable {
1465            increment(&mut statistics.stable_models)?;
1466            return Ok(Some(candidate));
1467        }
1468    }
1469}

1471/// The original index a region walk reads, built and charged before the
1472/// walk's first step: this is the one build site of every enumeration route,
1473/// scalar, parallel and batched, and it runs on the coordinator before any
1474/// worker starts.
1475///
1476/// The build is charged to `budget` first, one unit per node and operand
1477/// occurrence, and recorded once in the proposer's region counts in the same
1478/// step, so search work and
1479/// `regions.work` agree on every exit, a build that fails after its charge
1480/// was admitted included. Its time, when timed, is one call of the candidate
1481/// phase; a built index costs neither work nor a timed call.
1482fn walk_index<'i>(
1483    original: &'i mut OriginalIndex,
1484    proposer: &mut Proposer,
1485    budget: &mut Budget<'_>,
1486    timings: &mut Option<crate::SearchPhaseTimings>,
1487) -> Result<&'i std::sync::Arc<IndexedTheory>, Incomplete> {
1488    if original.get().is_some() {
1489        return original.get().ok_or(Incomplete::InvalidWitness);
1490    }
1491    let started = timing::start(timings.as_ref());
1492    let built = original.ensure(budget.cancellation, |work| {
1493        budget.charge(work)?;
1494        proposer.record_index_work(work)
1495    });
1496    timing::finish(timings, Phase::Candidates, started);
1497    built
1498}
