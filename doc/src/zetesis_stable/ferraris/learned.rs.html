1//! The learned rows one solve's candidate streams share.
2//!
3//! A [`LearnedStore`] holds, by consequence class, the rows its streams' tiers
4//! delivered. Its public surface is creating an empty store and handing it to
5//! streams ([`crate::StableModels::set_learned_store`]): no public operation
6//! adds, seeds, edits or reads a row, so a caller chooses which streams share
7//! a store, never what is learned. A walk seeds a tier from the store before
8//! the tier's first narrowing and delivers it at the walk's boundaries: when
9//! the walk ends, when a parallel walk's coordinator hands over to its
10//! workers, at the GPU route's slate joins, and each time the walk has visited
11//! the owner's `publish_regions` regions since the tier last delivered. None
12//! of these is inside a narrowing. One lock serializes the store; the tier
13//! pays for every read.
14
15use std::sync::{Mutex, MutexGuard, PoisonError};
16
17use zetesis_ferraris::LearningStore;
18
19/// The learned rows the candidate streams of one solve share, by consequence
20/// class. Create an empty store and hand it to every stream that should
21/// share; rows enter it only from those streams' own narrowing.
22#[derive(Debug, Default)]
23pub struct LearnedStore {
24    rows: Mutex<LearningStore>,
25}
26
27impl LearnedStore {
28    /// An empty store.
29    #[must_use]
30    pub fn new() -> Self {
31        Self::default()
32    }
33
34    /// Exclusive access for one seeding or delivery. Retention publishes a
35    /// row only after every fallible step succeeds, so a panic while the lock
36    /// was held leaves only complete rows.
37    pub(crate) fn lock(&self) -> MutexGuard<'_, LearningStore> {
38        self.rows.lock().unwrap_or_else(PoisonError::into_inner)
39    }
40}
41
42/// A walk's sharing state beside its scratch's tier: whether the tier has
43/// drawn the store's rows, and the regions the walk visited since the tier
44/// last delivered.
45#[derive(Debug, Default)]
46pub(crate) struct Tier {
47    pub(super) seeded: bool,
48    pub(super) visited: u64,
49}
50
51impl Tier {
52    /// Draw the store's rows again before the tier's next narrowing, as a
53    /// producer does after each slate's join.
54    pub(crate) fn reseed(&mut self) {
55        self.seeded = false;
56    }
57}
