1//! The original theory's narrowing index: one owner, built when a walk first
2//! needs it.
3//!
4//! An enumeration or a standalone membership check owns one
5//! [`OriginalIndex`]. Its candidate walk and every proper-subset query of its
6//! reduct read the same immutable [`IndexedTheory`], so the index is built at
7//! most once per owner, and only by [`OriginalIndex::ensure`]: no other code
8//! can construct an `IndexedTheory`. A run decided without a region walk, by a
9//! positive certificate, builds none.
10
11use std::sync::{Arc, Mutex, OnceLock, PoisonError};
12
13use zetesis_algebra::Cancellation;
14use zetesis_ferraris::{Narrower, Producers, RegionLimits, Theory};
15
16use super::regions::stopped;
17use crate::Incomplete;

19/// One immutable original-theory index. Construction binds the index to the
20/// exact admitted instance; equal independently admitted DAGs are not its
21/// subject. Candidate and reduct traversals share this owner, never their
22/// mutable knowledge.
23#[derive(Debug)]
24pub(crate) struct IndexedTheory {
25    theory: Theory,
26    narrower: Narrower,
27    /// The theory's producers when every root is a rule, taken from the
28    /// opened root when the owner was seeded with them, else extracted once by
29    /// the first inner preparation that needs them and charged to it; what
30    /// every candidate's inner preparation reads.
31    producers: OnceLock<Option<Arc<Producers>>>,
32    /// Serializes the one extraction, so that concurrent preparations
33    /// neither extract nor charge it twice.
34    extraction: Mutex<()>,
35}
36
37impl IndexedTheory {
38    pub(crate) fn theory(&self) -> &Theory {
39        &self.theory
40    }
41
42    pub(crate) fn narrower(&self) -> &Narrower {
43        &self.narrower
44    }
45
46    /// The theory's producers, extracted on the first call under `charge`
47    /// (the extraction's visits) and kept for every later one, which charges
48    /// nothing; one extraction at a time, so the charge is paid once however
49    /// many preparations ask together.
50    ///
51    /// # Errors
52    /// Cancellation, the extraction's stop or the charge's refusal, after
53    /// which the next call extracts again.
54    pub(crate) fn producers(
55        &self,
56        cancellation: &Cancellation,
57        charge: impl FnOnce(u64) -> Result<(), Incomplete>,
58    ) -> Result<Option<&Producers>, Incomplete> {
59        if let Some(producers) = self.producers.get() {
60            return Ok(producers.as_deref());
61        }
62        let _one_at_a_time = self
63            .extraction
64            .lock()
65            .unwrap_or_else(PoisonError::into_inner);
66        if let Some(producers) = self.producers.get() {
67            return Ok(producers.as_deref());
68        }
69        let extraction =
70            zetesis_ferraris::producers(&self.theory, RegionLimits::default(), cancellation)
71                .map_err(stopped)?;
72        charge(extraction.work)?;
73        Ok(self
74            .producers
75            .get_or_init(|| extraction.producers.map(Arc::new))
76            .as_deref())
77    }
78
79    /// Authenticate before borrowing the indexed subject for any traversal.
80    pub(crate) fn subject(&self, theory: &Theory) -> Result<(&Theory, &Narrower), Incomplete> {
81        if self.theory.same_instance(theory) {
82            Ok((&self.theory, &self.narrower))
83        } else {
84            Err(Incomplete::WrongTheory)
85        }
86    }
87}

89/// The admitted theory and, once a walk has needed it, its index.
90#[derive(Debug)]
91pub(crate) struct OriginalIndex {
92    theory: Theory,
93    index: Option<Arc<IndexedTheory>>,
94    native_byte_limit: u64,
95    /// The opened root's extraction, when the owner was seeded with it: the
96    /// index hands these producers to its first preparation instead of
97    /// extracting them again.
98    producers: Seed,
99}
100
101/// Whether an index owner carries the opened root's extraction.
102#[derive(Debug)]
103enum Seed {
104    Unseeded,
105    /// Extracted at the opening: the producers, or none when the theory is
106    /// outside the support fragment.
107    Extracted(Option<Arc<Producers>>),
108}
109
110impl OriginalIndex {
111    /// An owner of the theory's index, not yet built.
112    pub(crate) fn new(theory: &Theory, native_byte_limit: u64) -> Self {
113        Self {
114            theory: theory.clone(),
115            index: None,
116            native_byte_limit,
117            producers: Seed::Unseeded,
118        }
119    }
120
121    /// An owner seeded with the producers the opened root extracted, `None`
122    /// when the extraction found the theory outside the support fragment.
123    pub(crate) fn seeded(
124        theory: &Theory,
125        native_byte_limit: u64,
126        producers: Option<Arc<Producers>>,
127    ) -> Self {
128        Self {
129            producers: Seed::Extracted(producers),
130            ..Self::new(theory, native_byte_limit)
131        }
132    }
133
134    /// The index, if a walk has built it; never builds.
135    pub(crate) fn get(&self) -> Option<&Arc<IndexedTheory>> {
136        self.index.as_ref()
137    }
138
139    /// The index, built on the first call that finds none.
140    ///
141    /// Before building, `charge` receives the index's documented cost: one
142    /// work unit per node and operand occurrence of the theory, which equals
143    /// the built narrower's [`Narrower::work`], known before the narrower is built. A refused charge
144    /// builds nothing and leaves the owner unbuilt. An admitted charge stays
145    /// admitted even if the build then fails, as a failed narrowing keeps
146    /// its admitted prefix: `charge` is where the caller records it, in the
147    /// budget and in the receipt that counts index work, so the two agree on
148    /// every exit. A built index is returned without calling `charge` again.
149    ///
150    /// # Errors
151    /// Cancellation, the charge or native-storage refusal, or allocation failure.
152    pub(crate) fn ensure(
153        &mut self,
154        cancellation: &Cancellation,
155        charge: impl FnOnce(u64) -> Result<(), Incomplete>,
156    ) -> Result<&Arc<IndexedTheory>, Incomplete> {
157        cancellation.poll()?;
158        if let Some(index) = &self.index {
159            admit(&index.narrower, self.native_byte_limit)?;
160        }
161        let index = if let Some(index) = self.index.take() {
162            index
163        } else {
164            let work = u64::try_from(
165                self.theory.nodes().len() as u128 + self.theory.parts().occurrences() as u128,
166            )
167            .map_err(|_| Incomplete::CounterOverflow)?;
168            charge(work)?;
169            let narrower = prepare(&self.theory, self.native_byte_limit, cancellation)?;
170            debug_assert_eq!(narrower.work(), work, "the charge is the index's work");
171            Arc::new(IndexedTheory {
172                theory: self.theory.clone(),
173                narrower,
174                producers: match std::mem::replace(&mut self.producers, Seed::Unseeded) {
175                    Seed::Unseeded => OnceLock::new(),
176                    Seed::Extracted(producers) => OnceLock::from(producers),
177                },
178                extraction: Mutex::new(()),
179            })
180        };
181        Ok(self.index.insert(index))
182    }
183}

185/// Construct an original, restriction or objective-bound index under the same
186/// additional native-storage policy. The original owner retains its own logical
187/// work receipt; restrictions and bounds preserve their existing work contract.
188///
189/// The check bounds retained publication, not peak construction memory. Legacy
190/// Boolean index storage and constructor-internal cancellation are unchanged.
191pub(super) fn prepare(
192    theory: &Theory,
193    native_byte_limit: u64,
194    cancellation: &Cancellation,
195) -> Result<Narrower, Incomplete> {
196    cancellation.poll()?;
197    // Every live native cut retains at least its original node identity, even
198    // when it becomes constant. Unused numeric descriptors imply no index.
199    let mut minimum = 0u128;
200    if !theory.parts().extremum_cuts().is_empty() {
201        for &node in theory.nodes() {
202            cancellation.poll()?;
203            if node.extremum_cut_descriptor().is_some() {
204                minimum += size_of::<usize>() as u128;
205            }
206        }
207    }
208    check_bytes(minimum, native_byte_limit)?;
209    #[cfg(test)]
210    hooks::observe(theory)?;
211    // Allocation failure remains Allocation, not a fabricated storage receipt.
212    // The limit bounds retained publication; temporary construction allocations
213    // and allocator excess can precede the final actual-capacity check.
214    let narrower = Narrower::try_new(theory).map_err(stopped)?;
215    cancellation.poll()?;
216    admit(&narrower, native_byte_limit)?;
217    Ok(narrower)
218}
219
220pub(super) fn admit(narrower: &Narrower, native_byte_limit: u64) -> Result<(), Incomplete> {
221    check_bytes(
222        narrower.extremum_index_bytes() + narrower.capacity_index_bytes(),
223        native_byte_limit,
224    )
225}
226
227fn check_bytes(required: u128, limit: u64) -> Result<(), Incomplete> {
228    if required > u128::from(limit) {
229        Err(Incomplete::NativeNarrowingStorage {
230            required,
231            limit: u128::from(limit),
232        })
233    } else {
234        Ok(())
235    }
236}

238/// Test-only observation of the construction site: a count of builds per
239/// admitted instance, and an injected build failure. Neither exists in a
240/// non-test build.
241#[cfg(test)]
242pub(crate) mod hooks {
243    use std::cell::Cell;
244    use std::sync::{Mutex, PoisonError};
245
246    use zetesis_ferraris::Theory;
247
248    use crate::Incomplete;
249
250    /// Every instance whose index construction was entered, once per entry.
251    /// Instances are compared by identity, so concurrently running tests,
252    /// each with its own admitted theory, never count each other's builds.
253    static BUILDS: Mutex<Vec<Theory>> = Mutex::new(Vec::new());
254
255    thread_local! {
256        /// Fail the next construction entered on this thread.
257        static FAIL: Cell<bool> = const { Cell::new(false) };
258    }
259
260    pub(super) fn observe(theory: &Theory) -> Result<(), Incomplete> {
261        BUILDS
262            .lock()
263            .unwrap_or_else(PoisonError::into_inner)
264            .push(theory.clone());
265        if FAIL.with(Cell::take) {
266            Err(Incomplete::Allocation)
267        } else {
268            Ok(())
269        }
270    }
271
272    /// Index constructions entered for this exact instance, on any thread.
273    pub(crate) fn builds(theory: &Theory) -> usize {
274        BUILDS
275            .lock()
276            .unwrap_or_else(PoisonError::into_inner)
277            .iter()
278            .filter(|built| built.same_instance(theory))
279            .count()
280    }
281
282    /// Make the next construction entered on this thread refuse its
283    /// allocation, as `Narrower::try_new` refuses one.
284    pub(crate) fn fail_next_build() {
285        FAIL.with(|fail| fail.set(true));
286    }
287}
