1//! The receipts of a search an optimizer runs in passes over one theory: a
2//! later pass's stream reports its own work after the earlier passes'.

9/// The final receipts of the passes before this stream's.
10#[derive(Clone, Copy, Debug)]
11pub(super) struct Earlier {
12    pub(super) statistics: Statistics,
13    pub(super) batch: BatchStatistics,
14}

16/// `earlier` followed by `later`: work, counts and installations summed, peaks
17/// the larger, and what describes the route as it stands (the frontier's
18/// order and current size, its producers and workers, the certificate plan,
19/// the generator order's status and retained bytes) the later pass's. Sums
20/// saturate at the counters' maximum, which no search reaches.
21pub(super) fn statistics(earlier: &Statistics, later: &Statistics) -> Statistics {
22    Statistics {
23        search: crate::SearchStatistics {
24            work: earlier.search.work.saturating_add(later.search.work),
25            decisions: earlier
26                .search
27                .decisions
28                .saturating_add(later.search.decisions),
29        },
30        candidate_restrictions: earlier
31            .candidate_restrictions
32            .saturating_add(later.candidate_restrictions),
33        candidates: earlier.candidates.saturating_add(later.candidates),
34        countermodel_queries: earlier
35            .countermodel_queries
36            .saturating_add(later.countermodel_queries),
37        countermodels: earlier.countermodels.saturating_add(later.countermodels),
38        stable_models: earlier.stable_models.saturating_add(later.stable_models),
39        phase_timings: crate::timing::combined(earlier.phase_timings, later.phase_timings),
40        certified: match (&earlier.certified, &later.certified) {
41            (Some(earlier), Some(later)) => Some(certified(earlier, later)),
42            (earlier, later) => later.or(*earlier),
43        },
44        regions: match (earlier.regions, later.regions) {
45            (Some(earlier), Some(later)) => Some(regions(&earlier, later)),
46            (earlier, later) => later.or(earlier),
47        },
48        region_filter: match (earlier.region_filter, later.region_filter) {
49            (Some(earlier), Some(later)) => Some(crate::RegionFilterStatistics {
50                preparations: earlier.preparations.saturating_add(later.preparations),
51                checks: earlier.checks.saturating_add(later.checks),
52                refuted: earlier.refuted.saturating_add(later.refuted),
53                failed: earlier.failed.saturating_add(later.failed),
54                overflowed: earlier.overflowed || later.overflowed,
55            }),
56            (earlier, later) => later.or(earlier),
57        },
58        reduct: crate::ReductStatistics {
59            original_work: earlier
60                .reduct
61                .original_work
62                .saturating_add(later.reduct.original_work),
63            frozen_positive_attempts: earlier
64                .reduct
65                .frozen_positive_attempts
66                .saturating_add(later.reduct.frozen_positive_attempts),
67            frozen_positive_completed: earlier
68                .reduct
69                .frozen_positive_completed
70                .saturating_add(later.reduct.frozen_positive_completed),
71            frozen_positive_preparation_work: earlier
72                .reduct
73                .frozen_positive_preparation_work
74                .saturating_add(later.reduct.frozen_positive_preparation_work),
75            frozen_positive_work: earlier
76                .reduct
77                .frozen_positive_work
78                .saturating_add(later.reduct.frozen_positive_work),
79            peak_workspace_bytes: earlier
80                .reduct
81                .peak_workspace_bytes
82                .max(later.reduct.peak_workspace_bytes),
83            regions: counts(&earlier.reduct.regions, later.reduct.regions),
84            inner_preparation: earlier
85                .reduct
86                .inner_preparation
87                .saturating_add(later.reduct.inner_preparation),
88        },
89    }
90}

92/// Batch receipts in the same sense: the pending candidates and the current
93/// validation workspace are the later pass's.
94pub(super) fn batch(earlier: &BatchStatistics, later: BatchStatistics) -> BatchStatistics {
95    BatchStatistics {
96        original_validation_work: earlier
97            .original_validation_work
98            .saturating_add(later.original_validation_work),
99        checker_calls: earlier.checker_calls.saturating_add(later.checker_calls),
100        completion_calls: earlier
101            .completion_calls
102            .saturating_add(later.completion_calls),
103        committed: earlier.committed.saturating_add(later.committed),
104        propagated: earlier.propagated.saturating_add(later.propagated),
105        residuals: earlier.residuals.saturating_add(later.residuals),
106        lookahead_slates: earlier
107            .lookahead_slates
108            .saturating_add(later.lookahead_slates),
109        ..later
110    }
111}

136fn regions(
137    earlier: &RegionSearchStatistics,
138    mut later: RegionSearchStatistics,
139) -> RegionSearchStatistics {
140    later.counts = counts(&earlier.counts, later.counts);
141    later.generator_order.work = earlier
142        .generator_order
143        .work
144        .saturating_add(later.generator_order.work);
145    later.generator_order.peak_bytes = earlier
146        .generator_order
147        .peak_bytes
148        .max(later.generator_order.peak_bytes);
149    later.parity.preparation_work = earlier
150        .parity
151        .preparation_work
152        .saturating_add(later.parity.preparation_work);
153    later.parity.preparation_bytes = earlier
154        .parity
155        .preparation_bytes
156        .max(later.parity.preparation_bytes);
157    later.parity.phases = preparation_work(earlier.parity.phases, later.parity.phases);
158    later.parity.region_passes = earlier
159        .parity
160        .region_passes
161        .saturating_add(later.parity.region_passes);
162    later.parity.forced = earlier.parity.forced.saturating_add(later.parity.forced);
163    later.parity.refuted = earlier.parity.refuted.saturating_add(later.parity.refuted);
164    later.cardinality.preparation_work = earlier
165        .cardinality
166        .preparation_work
167        .saturating_add(later.cardinality.preparation_work);
168    later.cardinality.preparation_bytes = earlier
169        .cardinality
170        .preparation_bytes
171        .max(later.cardinality.preparation_bytes);
172    later.cardinality.phases =
173        preparation_work(earlier.cardinality.phases, later.cardinality.phases);
174    later.cardinality.region_passes = earlier
175        .cardinality
176        .region_passes
177        .saturating_add(later.cardinality.region_passes);
178    later.cardinality.refuted = earlier
179        .cardinality
180        .refuted
181        .saturating_add(later.cardinality.refuted);
182    if let (Some(earlier), Some(later)) = (earlier.frontier, later.frontier.as_mut()) {
183        *later = RegionFrontierStatistics {
184            peak_regions: earlier.peak_regions.max(later.peak_regions),
185            peak_capacity: earlier.peak_capacity.max(later.peak_capacity),
186            peak_retained_bytes: earlier.peak_retained_bytes.max(later.peak_retained_bytes),
187            ..*later
188        };
189    }
190    later
191}

193fn preparation_work(
194    earlier: zetesis_ferraris::PreparationWork,

200        reduction: earlier.reduction.saturating_add(later.reduction),
201        discovery: earlier.discovery.saturating_add(later.discovery),
202        candidate_checks: earlier

204            .saturating_add(later.candidate_checks),

206        root_seed: earlier.root_seed.saturating_add(later.root_seed),

210fn counts(earlier: &RegionCounts, later: RegionCounts) -> RegionCounts {
211    let (e, l) = (&earlier.support_explanations, later.support_explanations);
212    RegionCounts {
213        regions: earlier.regions.saturating_add(later.regions),
214        refuted: earlier.refuted.saturating_add(later.refuted),
215        leaves: earlier.leaves.saturating_add(later.leaves),
216        propagations: earlier.propagations.saturating_add(later.propagations),
217        held: earlier.held.saturating_add(later.held),
218        cut: earlier.cut.saturating_add(later.cut),
219        work: earlier.work.saturating_add(later.work),
220        support_work: earlier.support_work.saturating_add(later.support_work),
221        founded_work: earlier.founded_work.saturating_add(later.founded_work),
222        arithmetic: earlier.arithmetic.saturating_add(later.arithmetic),
223        support_explanations: SupportExplanationStatistics {
224            attempts: e.attempts.saturating_add(l.attempts),
225            certified: e.certified.saturating_add(l.certified),
226            matched: e.matched.saturating_add(l.matched),
227            cut: e.cut.saturating_add(l.cut),
228            forced: e.forced.saturating_add(l.forced),
229            refuted: e.refuted.saturating_add(l.refuted),
230            declined: e.declined.saturating_add(l.declined),
231            work: e.work.saturating_add(l.work),
232            peak_bytes: e.peak_bytes.max(l.peak_bytes),
233        },
234    }
235}
