1//! Runtime attribution carried by the existing regional receipt.
2//!
3//! These are observations, not allowances. No membership subject produces
4//! them, and copying knowledge only contributes to the ordinary work field.
5
6use zetesis_algebra::region::SubjectStep;
7use zetesis_ferraris::ArithmeticAttempt;
8
9use crate::Incomplete;
10
11/// Private dimensions of `RegionCounts`, projected onto its public arithmetic
12/// summaries. Actual record headers include this payload exactly once.
13#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
14pub(crate) struct ArithmeticCounts {
15    pub(crate) parity_passes: u64,
16    pub(crate) parity_forced: u64,
17    pub(crate) parity_refuted: u64,
18    pub(crate) cardinality_passes: u64,
19    pub(crate) cardinality_refuted: u64,
20}
21
22/// Attribution is fixed by the two concrete candidate-only adapter arms.
23#[derive(Clone, Copy)]
24pub(super) enum ArithmeticKind {
25    Parity,
26    Cardinality,
27}
28
29impl ArithmeticKind {
30    /// Failed attempts have no completed batch, but retain real P1 decisions.
31    /// A refutation completes the batch at its absorbing result. Bookkeeping
32    /// neither polls nor debits the Meter after that result.
33    pub(super) fn counts<E>(self, attempt: &ArithmeticAttempt<E>) -> ArithmeticCounts {
34        let passes = u64::from(attempt.result.is_ok());
35        let refuted = u64::from(matches!(&attempt.result, Ok(SubjectStep::Refuted)));
36        match self {
37            Self::Parity => ArithmeticCounts {
38                parity_passes: passes,
39                // Fresh applications partition the original atom positions;
40                // their sum is bounded by this admitted attempt's word work.
41                parity_forced: attempt.statistics.held + attempt.statistics.cut,
42                parity_refuted: refuted,
43                ..ArithmeticCounts::default()
44            },
45            Self::Cardinality => ArithmeticCounts {
46                cardinality_passes: passes,
47                cardinality_refuted: refuted,
48                ..ArithmeticCounts::default()
49            },
50        }
51    }
52}
53
54impl ArithmeticCounts {
55    /// `RegionCounts` commits this together with all its other checked sums.
56    pub(crate) fn checked_add(self, other: Self) -> Result<Self, Incomplete> {
57        Ok(Self {
58            parity_passes: self
59                .parity_passes
60                .checked_add(other.parity_passes)
61                .ok_or(Incomplete::CounterOverflow)?,
62            parity_forced: self
63                .parity_forced
64                .checked_add(other.parity_forced)
65                .ok_or(Incomplete::CounterOverflow)?,
66            parity_refuted: self
67                .parity_refuted
68                .checked_add(other.parity_refuted)
69                .ok_or(Incomplete::CounterOverflow)?,
70            cardinality_passes: self
71                .cardinality_passes
72                .checked_add(other.cardinality_passes)
73                .ok_or(Incomplete::CounterOverflow)?,
74            cardinality_refuted: self
75                .cardinality_refuted
76                .checked_add(other.cardinality_refuted)
77                .ok_or(Incomplete::CounterOverflow)?,
78        })
79    }
80
81    /// Existing returned multipass/live snapshots saturate their observations;
82    /// they are never reused as a live search allowance or debited again.
83    pub(crate) fn saturating_add(self, other: Self) -> Self {
84        Self {
85            parity_passes: self.parity_passes.saturating_add(other.parity_passes),
86            parity_forced: self.parity_forced.saturating_add(other.parity_forced),
87            parity_refuted: self.parity_refuted.saturating_add(other.parity_refuted),
88            cardinality_passes: self
89                .cardinality_passes
90                .saturating_add(other.cardinality_passes),
91            cardinality_refuted: self
92                .cardinality_refuted
93                .saturating_add(other.cardinality_refuted),
94        }
95    }
96}
