14/// The membership machinery of one enumeration or worker. The proper-subset
15/// query reads the original theory's index, which this state never holds or
16/// builds: its owner, the enumeration's or the standalone check's
17/// `OriginalIndex`, lends it to each check. Mutable evaluation and query
18/// knowledge remain private to each check or worker.
19#[derive(Debug, Default)]
20pub(crate) struct State {
21    pub(crate) workspace: ReductWorkspace,
22}
23
24impl State {
25    /// Decide a classical candidate's membership. The proper-subset query
26    /// reads `index`, lent by its owner; a check lent none is refused as an
27    /// invalid witness, since every candidate comes from a walk that built
28    /// the index.
29    pub(crate) fn check(
30        &mut self,
31        theory: &Theory,
32        index: Option<&IndexedTheory>,
33        candidate: &Interpretation,
34        limits: Limits,
35        budget: &mut Budget<'_, impl Quota>,
36        statistics: &mut Statistics,
37    ) -> Result<Check, Incomplete> {
38        match self.check_publishing(
39            (theory, index),
40            candidate,
41            limits,
42            budget,
43            statistics,
44            &mut Inline,
45        )? {
46            Queried::Decided(check) => Ok(check),
47            Queried::Published(never) => match never {},
48        }
49    }
50
51    /// [`Self::check`], whose proper-subset query may publish its pending
52    /// regions through `publication` and end here undecided. A published
53    /// query's retained capacity is recorded and admitted as a decided one's.
54    pub(crate) fn check_publishing<P: Publication>(
55        &mut self,
56        (theory, index): (&Theory, Option<&IndexedTheory>),
57        candidate: &Interpretation,
58        limits: Limits,
59        budget: &mut Budget<'_, impl Quota>,
60        statistics: &mut Statistics,
61        publication: &mut P,
62    ) -> Result<Queried<P::Receipt>, Incomplete> {
63        budget.cancellation.poll()?;
64        if !theory.same_instance(candidate.theory()) {
65            return Err(Incomplete::WrongTheory);
66        }
67        let index = index.ok_or(Incomplete::InvalidWitness)?;
68        // Authenticate the lent index before any evaluation work.
69        index.subject(theory)?;
70        let query = ReductQuery::new(index);
71        let result = (|| {
72            let mut evaluated =
73                self.workspace
74                    .evaluate(candidate, limits, budget.cancellation, statistics)?;
75            if !evaluated.truth.is_model() {
76                return Ok(Queried::Decided(Check::NotModel));
77            }
78            let started = timing::start(statistics.phase_timings.as_ref());
79            increment(&mut statistics.countermodel_queries)?;
80            let result =
81                match positive_closure(&mut evaluated, candidate, limits, budget, statistics) {
82                    Ok(Some(verdict)) => Ok(Queried::Decided(verdict)),
83                    Ok(None) => query.check_publishing(
84                        &evaluated.truth,
85                        limits,
86                        budget,
87                        (statistics, evaluated.narrowing),
88                        publication,
89                    ),
90                    Err(error) => Err(error),
91                };
92            timing::finish(&mut statistics.phase_timings, Phase::Reduct, started);
93            result
94        })();
95        self.workspace
96            .finish_check(result, limits.max_reduct_bytes, statistics)
97    }
98}

100/// Try exact least closure before asking the general proper-subset query.
101/// Its accounted prefix is retained on every result; declined grammar alone
102/// falls through. A strict-subset result uses the ordinary witness validator.
103fn positive_closure(
104    evaluated: &mut super::RegionEvaluation<'_>,
105    candidate: &Interpretation,
106    limits: Limits,
107    budget: &mut Budget<'_, impl Quota>,
108    statistics: &mut Statistics,
109) -> Result<Option<Check>, Incomplete> {
110    use zetesis_ferraris::{
111        EvaluationError, EvaluationLimits, FrozenPositiveOutcome, FrozenPositiveWorkspace,
112    };
113
114    increment(&mut statistics.reduct.frozen_positive_attempts)?;
115    let wanted = u64::try_from(FrozenPositiveWorkspace::work_bound(candidate.theory()))
116        .map_err(|_| Incomplete::CounterOverflow)?;
117    let available = u128::from(limits.max_reduct_bytes)
118        .checked_sub(evaluated.positive_other_bytes)
119        .ok_or(Incomplete::ReductStorage {
120            required: evaluated.positive_other_bytes,
121            limit: u128::from(limits.max_reduct_bytes),
122        })?;
123    let cancellation = budget.cancellation;
124    let attempt = budget.with_reservation(wanted, |granted| {
125        let attempt = evaluated.positive.check(
126            &evaluated.truth,
127            EvaluationLimits {
128                max_work: granted,
129                max_bytes: usize::try_from(available).unwrap_or(usize::MAX),
130            },
131            cancellation,
132        );
133        let used = attempt.work;
134        (attempt, used)
135    })?;
136    statistics.reduct.frozen_positive_work = statistics
137        .reduct
138        .frozen_positive_work
139        .checked_add(attempt.work)
140        .ok_or(Incomplete::CounterOverflow)?;
141    statistics.reduct.frozen_positive_preparation_work = statistics
142        .reduct
143        .frozen_positive_preparation_work
144        .checked_add(attempt.preparation_work)
145        .ok_or(Incomplete::CounterOverflow)?;
146    statistics.reduct.peak_workspace_bytes = statistics
147        .reduct
148        .peak_workspace_bytes
149        .max(evaluated.positive_other_bytes + attempt.retained_bytes);
150    let outcome = attempt.result.map_err(|error| match error {
151        EvaluationError::Stopped(zetesis_algebra::Stop::WorkLimit) => Incomplete::WorkLimit,
152        EvaluationError::Stopped(stop) => Incomplete::from(stop),
153        EvaluationError::Storage { required, .. } => Incomplete::ReductStorage {
154            required: evaluated.positive_other_bytes + required,
155            limit: u128::from(limits.max_reduct_bytes),
156        },
157    })?;
158    let verdict = match outcome {
159        FrozenPositiveOutcome::Declined => return Ok(None),
160        FrozenPositiveOutcome::NotModel => Check::NotModel,
161        FrozenPositiveOutcome::Stable => Check::Stable,
162        FrozenPositiveOutcome::NonMinimal(subset) => crate::ferraris::checked_countermodel(
163            candidate.theory(),
164            candidate,
165            subset,
166            limits,
167            budget,
168            statistics,
169        )?,
170    };
171    increment(&mut statistics.reduct.frozen_positive_completed)?;
172    Ok(Some(verdict))
173}
