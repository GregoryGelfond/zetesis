1//! A JSON document's answers, read back as spelled atoms.
2//!
3//! The document's atom table is every record's spelled atoms in document
4//! order, and a record refers to its atoms by index into it, so the table
5//! read up to the last record resolves any record.
6
7use serde_json::Value;
8
9/// The record's full model as spelled atoms, in the record's order.
10///
11/// # Panics
12/// Panics if `document` or `record` does not have the JSON document's shape.
13#[must_use]
14pub fn spelled(document: &Value, record: &Value) -> Vec<Value> {
15    let table = table(document);
16    record["model"]["full_model"]
17        .as_array()
18        .unwrap()
19        .iter()
20        .map(|index| table[usize::try_from(index.as_u64().unwrap()).unwrap()].clone())
21        .collect()
22}
23
24/// The document's atom table.
25fn table(document: &Value) -> Vec<Value> {
26    document["models"]
27        .as_array()
28        .unwrap()
29        .iter()
30        .flat_map(|record| record["model"]["atoms"].as_array().unwrap().iter().cloned())
31        .collect()
32}

34#[cfg(test)]
35mod tests {
36    use super::*;
37    use serde_json::json;
38
39    #[test]
40    fn a_record_resolves_atoms_an_earlier_record_spelled() {
41        let document = json!({"models": [
42            {"model": {"atoms": ["a", "b"], "full_model": [0, 1]}},
43            {"model": {"atoms": ["c"], "full_model": [2, 0]}},
44        ]});
45        let second = &document["models"][1];
46        assert_eq!(spelled(&document, second), [json!("c"), json!("a")]);
47    }
48}
