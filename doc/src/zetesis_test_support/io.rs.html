1//! Sinks that fail on purpose, for tests of output failure paths.
2//!
3//! Each models one failure a destination can meet: a bounded destination that
4//! fills ([`BoundedWriter`], and [`BoundedText`] for text), a closed stream
5//! ([`Closed`]), and a failure at a chosen record ([`FailAt`]). A failing byte
6//! sink returns a broken pipe whose message a test can compare.
7
8use std::fmt;
9use std::io::{self, Write};
10
11/// The message of the error a full [`BoundedWriter`] returns.
12pub const FULL: &str = "bounded sink full";
13
14/// The message of the error a [`Closed`] sink returns.
15pub const CLOSED: &str = "sink closed";
16
17/// The message of the error a [`FailAt`] sink returns from its marker on.
18pub const FAILED: &str = "sink failed at its marker";
19
20/// A byte sink that accepts at most a fixed number of bytes and keeps the
21/// prefix it accepted.
22///
23/// A write that would exceed the capacity is accepted in part. Once the sink
24/// is full, every non-empty write fails with a broken pipe ([`FULL`]); an
25/// empty write succeeds.
26#[derive(Debug)]
27pub struct BoundedWriter {
28    remaining: usize,
29    bytes: Vec<u8>,
30}
31
32impl BoundedWriter {
33    /// A sink accepting at most `capacity` bytes.
34    #[must_use]
35    pub fn new(capacity: usize) -> Self {
36        Self {
37            remaining: capacity,
38            bytes: Vec::new(),
39        }
40    }
41
42    /// The bytes accepted so far.
43    #[must_use]
44    pub fn bytes(&self) -> &[u8] {
45        &self.bytes
46    }
47}
48
49impl Write for BoundedWriter {
50    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
51        if bytes.is_empty() {
52            return Ok(0);
53        }
54        if self.remaining == 0 {
55            return Err(io::Error::new(io::ErrorKind::BrokenPipe, FULL));
56        }
57        let count = bytes.len().min(self.remaining);
58        self.bytes.extend_from_slice(&bytes[..count]);
59        self.remaining -= count;
60        Ok(count)
61    }
62
63    fn flush(&mut self) -> io::Result<()> {
64        Ok(())
65    }
66}
67
68/// A byte sink whose every write fails with a broken pipe ([`CLOSED`]), as a
69/// closed stream does. Flushing succeeds.
70#[derive(Debug)]
71pub struct Closed;
72
73impl Write for Closed {
74    fn write(&mut self, _: &[u8]) -> io::Result<usize> {
75        Err(io::Error::new(io::ErrorKind::BrokenPipe, CLOSED))
76    }
77
78    fn flush(&mut self) -> io::Result<()> {
79        Ok(())
80    }
81}

83/// A byte sink that accepts every write until one contains its marker, then
84/// fails that write and every later one with a broken pipe ([`FAILED`]).
85///
86/// A test places the failure at a chosen record, such as the first line of a
87/// statistics block, without affecting the records before it.
88#[derive(Debug)]
89pub struct FailAt {
90    marker: &'static [u8],
91    failed: bool,
92}
93
94impl FailAt {
95    /// A sink failing from the first write containing `marker`.
96    #[must_use]
97    pub fn new(marker: &'static [u8]) -> Self {
98        Self {
99            marker,
100            failed: false,
101        }
102    }
103}
104
105impl Write for FailAt {
106    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
107        self.failed |= bytes
108            .windows(self.marker.len())
109            .any(|window| window == self.marker);
110        if self.failed {
111            return Err(io::Error::new(io::ErrorKind::BrokenPipe, FAILED));
112        }
113        Ok(bytes.len())
114    }
115
116    fn flush(&mut self) -> io::Result<()> {
117        Ok(())
118    }
119}

121/// A text sink that accepts at most a fixed number of bytes and keeps the
122/// text it accepted.
123///
124/// [`fmt::Write`] has no partial writes, so a write that would exceed the
125/// capacity is refused whole.
126#[derive(Debug)]
127pub struct BoundedText {
128    capacity: usize,
129    text: String,
130}
131
132impl BoundedText {
133    /// A sink accepting at most `capacity` bytes of text.
134    #[must_use]
135    pub fn new(capacity: usize) -> Self {
136        Self {
137            capacity,
138            text: String::new(),
139        }
140    }
141
142    /// The text accepted so far.
143    #[must_use]
144    pub fn text(&self) -> &str {
145        &self.text
146    }
147}
148
149impl fmt::Write for BoundedText {
150    fn write_str(&mut self, text: &str) -> fmt::Result {
151        if text.len() > self.capacity - self.text.len() {
152            return Err(fmt::Error);
153        }
154        self.text.push_str(text);
155        Ok(())
156    }
157}

159#[cfg(test)]
160mod tests {
161    use super::*;
162    use std::fmt::Write as _;
163
164    #[test]
165    fn a_bounded_writer_keeps_the_accepted_prefix_and_then_refuses() {
166        let mut sink = BoundedWriter::new(3);
167        assert_eq!(sink.write(b"ab").unwrap(), 2);
168        assert_eq!(sink.write(b"cde").unwrap(), 1);
169        let error = sink.write(b"f").unwrap_err();
170        assert_eq!(error.kind(), io::ErrorKind::BrokenPipe);
171        assert_eq!(error.to_string(), FULL);
172        assert_eq!(sink.bytes(), b"abc");
173    }
174
175    #[test]
176    fn a_full_bounded_writer_accepts_an_empty_write() {
177        let mut sink = BoundedWriter::new(0);
178        assert_eq!(sink.write(b"").unwrap(), 0);
179    }
180
181    #[test]
182    fn a_closed_sink_refuses_every_write_with_a_broken_pipe() {
183        let error = Closed.write(b"a").unwrap_err();
184        assert_eq!(error.kind(), io::ErrorKind::BrokenPipe);
185        assert_eq!(error.to_string(), CLOSED);
186        assert!(Closed.flush().is_ok());
187    }
188
189    #[test]
190    fn a_fail_at_sink_fails_from_the_write_containing_its_marker() {
191        let mut sink = FailAt::new(b"Statistics:");
192        assert_eq!(sink.write(b"Answer: 1\n").unwrap(), 10);
193        let error = sink.write(b"Statistics: 1\n").unwrap_err();
194        assert_eq!(error.kind(), io::ErrorKind::BrokenPipe);
195        assert_eq!(error.to_string(), FAILED);
196        assert!(sink.write(b"later\n").is_err());
197    }
198
199    #[test]
200    fn bounded_text_refuses_a_write_that_would_exceed_it_whole() {
201        let mut sink = BoundedText::new(4);
202        sink.write_str("abc").unwrap();
203        assert!(sink.write_str("de").is_err());
204        assert_eq!(sink.text(), "abc");
205        sink.write_str("d").unwrap();
206        assert_eq!(sink.text(), "abcd");
207    }
208}
