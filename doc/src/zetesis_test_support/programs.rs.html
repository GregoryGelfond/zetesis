10/// Every admitted rule in order, including closed facts, as borrowed templates.
11/// Tests comparing admission routes use the complete logical sequence even when
12/// one route stores facts separately from its explicit templates.
13#[must_use]
14pub fn rules(program: &Program) -> impl ExactSizeIterator<Item = TemplateRef<'_>> {
15    program.rules().map(|rule| match rule {
16        RuleRef::Template { template, .. } => template,
17        RuleRef::Fact(fact) => fact.as_template(),
18    })
19}

21/// The atom `name(values…)`, whose arity is the number of values.
22///
23/// # Panics
24/// Panics if `name` is empty.
25#[must_use]
26pub fn atom(name: &str, values: Vec<Value>) -> Atom {
27    let predicate = Predicate::new(name, values.len()).expect("nonempty test name");
28    Atom::new(predicate, values).expect("matching arity")
29}
30
31/// The atom pattern `name(terms…)`, whose arity is the number of terms.
32///
33/// # Panics
34/// Panics if `name` is empty.
35#[must_use]
36pub fn pattern(name: &str, terms: Vec<Term>) -> AtomPattern {
37    let predicate = Predicate::new(name, terms.len()).expect("nonempty test name");
38    AtomPattern::new(predicate, terms).expect("matching arity")
39}

41/// The atom `name(values…)` over numbers, whose arity is the number of values.
42///
43/// # Panics
44/// Panics if `name` is empty.
45#[must_use]
46pub fn numbered(name: &str, values: &[i32]) -> Atom {
47    atom(name, values.iter().copied().map(Value::Number).collect())
48}

50/// The atom `name(value)` over one number.
51///
52/// # Panics
53/// Panics if `name` is empty.
54#[must_use]
55pub fn unary(name: &str, value: i32) -> Atom {
56    numbered(name, &[value])
57}

59/// The atom `name(values…)` whose predicate carries `sign`.
60///
61/// # Panics
62/// Panics if `name` is empty.
63#[must_use]
64pub fn signed(name: &str, sign: Sign, values: Vec<Value>) -> Atom {
65    let predicate = Predicate::with_sign(name, values.len(), sign).expect("nonempty test name");
66    Atom::new(predicate, values).expect("matching arity")
67}

69/// The atom `name(values…)` over numbers whose predicate carries `sign`.
70///
71/// # Panics
72/// Panics if `name` is empty.
73#[must_use]
74pub fn signed_numbered(name: &str, sign: Sign, values: &[i32]) -> Atom {
75    signed(
76        name,
77        sign,
78        values.iter().copied().map(Value::Number).collect(),

100/// The model holding the nullary atoms `names`.
101///
102/// # Panics
103/// Panics if a name is empty.
104#[must_use]
105pub fn model(names: &[&str]) -> Model {
106    Model::new(names.iter().map(|name| nullary(name))).expect("distinct test atoms")
107}

109/// The variable numbered `index` as a term.
110#[must_use]
111pub fn variable(index: usize) -> Term {

115/// The value of `nodes` under the default value limits.
116///
117/// # Panics
118/// Panics if `nodes` do not form one value within those limits.
119#[must_use]
120pub fn value(nodes: Vec<ValueNode>) -> Value {
121    Value::from_nodes(nodes, ValueLimits::default()).expect("a valid test value")
122}

124/// The positive function node `name` of `arity` arguments.
125#[must_use]
126pub fn function(name: &str, arity: usize) -> ValueNode {
127    ValueNode::Function {
128        name: name.into(),
129        sign: Sign::Positive,
130        arity,
131    }

134/// The number `value` as a constant term.
135#[must_use]
136pub fn number(value: i32) -> Term {

140/// The fact `name(values…).`
141///
142/// # Panics
143/// Panics if `name` is empty.
144#[must_use]
145pub fn fact(name: &str, values: Vec<Term>) -> Template {
146    Template::new(Some(pattern(name, values)), vec![], vec![], vec![], vec![])
147}

149/// The program of `templates` under the default admission limits.
150///
151/// # Panics
152/// Panics if the templates are not a safe, finite program.
153#[must_use]
154pub fn program(templates: Vec<Template>) -> Program {
155    Program::new(
156        templates,
157        zetesis_core::FactTables::default(),
158        AdmissionLimits::default(),
159    )
160    .expect("safe finite test program")
161}

163#[cfg(test)]
164mod tests {
165    use super::*;
166
167    #[test]
168    fn an_atom_takes_its_arity_from_its_values() {
169        let built = atom("p", vec![Value::Number(1), Value::Symbol("a".into())]);
170        assert_eq!(built.predicate().arity(), 2);
171        assert_eq!(
172            built.values(),
173            [Value::Number(1), Value::Symbol("a".into())]
174        );
175    }
176
177    #[test]
178    fn a_nullary_atom_and_pattern_have_no_arguments() {
179        assert_eq!(nullary("a").predicate().arity(), 0);
180        assert!(nullary("a").values().is_empty());
181        assert_eq!(nullary_pattern("a").predicate().arity(), 0);
182    }
183
184    #[test]
185    fn numbered_atoms_carry_their_numbers() {
186        let built = numbered("p", &[1, 2]);
187        assert_eq!(built.predicate().arity(), 2);
188        assert_eq!(built.values(), [Value::Number(1), Value::Number(2)]);
189        assert_eq!(unary("p", 3), numbered("p", &[3]));
190    }
191
192    #[test]
193    fn signed_atoms_carry_their_sign() {
194        let negative = signed("q", Sign::Negative, vec![Value::Number(1)]);
195        assert_eq!(negative.predicate().sign(), Sign::Negative);
196        assert_eq!(negative.values(), [Value::Number(1)]);
197        assert_eq!(signed_numbered("q", Sign::Negative, &[1]), negative);
198        assert_eq!(unary("q", 1).predicate().sign(), Sign::Positive);
199    }
200
201    #[test]
202    fn a_model_holds_its_nullary_atoms() {
203        let built = model(&["a", "b"]);
204        assert_eq!(built.atoms().len(), 2);
205        assert!(built.contains(&nullary("a")));
206        assert!(built.contains(&nullary("b")));
207    }
208
209    #[test]
210    fn a_variable_term_carries_its_index() {
211        assert_eq!(variable(2), Term::Variable(2));
212    }
213
214    #[test]
215    fn a_function_node_is_positive_with_its_arity() {
216        assert_eq!(
217            function("f", 2),
218            ValueNode::Function {
219                name: "f".into(),
220                sign: Sign::Positive,
221                arity: 2,
222            }
223        );
224    }
225
226    #[test]
227    fn a_value_is_built_from_its_nodes() {
228        assert_eq!(value(vec![ValueNode::Number(3)]), Value::Number(3));
229    }
230
231    #[test]
232    fn a_program_holds_the_facts_it_is_built_from() {
233        let built = program(vec![
234            fact("p", vec![number(1)]),
235            fact("q", vec![number(2), number(3)]),
236        ]);
237        assert_eq!(built.templates().len(), 2);
238    }
