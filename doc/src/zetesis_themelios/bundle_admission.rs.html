18/// Admission budgets across an already bounded, loaded source graph. File,
19/// byte, and include-depth limits belong to [`crate::BundleLimits`] at loading.
20#[derive(Clone, Copy, Debug)]
21pub struct BundleAdmissionOptions {
22    /// Total syntax nodes visited across all unique original files.
23    pub max_syntax_nodes: usize,
24    /// Maximum syntax depth within any original file.
25    pub max_syntax_depth: usize,
26    /// Maximum original body elements in each rule, before deduplication.
27    pub max_body_elements: usize,
28    /// Independent limits on the compiled relational program.
29    pub core_limits: AdmissionLimits,
30}
31
32impl Default for BundleAdmissionOptions {
33    fn default() -> Self {
34        let single = AdmissionOptions::default();
35        Self {
36            max_syntax_nodes: single.max_syntax_nodes,
37            max_syntax_depth: single.max_syntax_depth,
38            max_body_elements: single.max_body_elements,
39            core_limits: single.core_limits,
40        }
41    }
42}

44/// An admitted relational program and its complete original source catalog.
45/// Every template origin resolves through [`Self::bundle`].
46#[derive(Debug)]
47pub struct AdmittedBundle {
48    program: Program,
49    bundle: SourceBundle,
50    template_origins: Vec<Vec<Location>>,
51    metadata: SourceMetadata,
52    expansion: ExpansionUsage,
53}
54
55impl AdmittedBundle {
56    /// The independently checked core program.
57    #[must_use]
58    pub fn program(&self) -> &Program {
59        &self.program
60    }
61    /// Original source bytes, identities, paths, and include occurrences.
62    #[must_use]
63    pub fn bundle(&self) -> &SourceBundle {
64        &self.bundle
65    }
66    /// Original parsed rule locations in compiled template order. Definitions
67    /// remain in the original source catalog; they are not claimed as rule spans.
68    #[must_use]
69    pub fn template_origins(&self) -> &[Vec<Location>] {
70        &self.template_origins
71    }
72    /// Global source declarations and display policy, retaining every original
73    /// metadata occurrence. The underlying program and model identity are full.
74    #[must_use]
75    pub fn metadata(&self) -> &SourceMetadata {
76        &self.metadata
77    }
78    /// Expansion charges this admission accepted, each under its ceiling.
79    #[must_use]
80    pub fn expansion_usage(&self) -> &ExpansionUsage {
81        &self.expansion
82    }
83    /// Consume the result when original source evidence is no longer required.
84    #[must_use]
85    pub fn into_program(self) -> Program {
86        self.program
87    }
88}

90/// A typed semantic bundle refusal, separate from source-graph loading errors.
91#[derive(Debug)]
92pub enum BundleAdmissionError {
93    /// The extended source or independent core admission door refused its stage.
94    Expansion(ExpansionFailure),
95    /// A later explicit root has a different lexical identity from the same
96    /// source's first root or include occurrence.
97    RootAlias {
98        /// First selected lexical filename.
99        first: PathBuf,
100        /// Later original root spelling.
101        repeated: PathBuf,
102        /// Original source span of the repeated root.
103        location: Location,
104    },
105    /// Different lexical include paths refer to the same canonical source.
106    /// clingo can parse these separately, which the deduplicated graph does not
107    /// represent; admission refuses instead of dropping repeated definitions.
108    IncludeAlias {
109        /// First resolved lexical path in include traversal order.
110        first: PathBuf,
111        /// Later resolved lexical path to the same canonical source.
112        repeated: PathBuf,
113        /// Original include occurrence that revisits the source.
114        location: Location,
115    },
116    /// A lexical path's dot-normalized identity differs from the canonical
117    /// source, for example through a symlink. Relative include semantics under
118    /// that spelling are not represented by the canonical graph.
119    IncludeRedirection {
120        /// Original include path resolved from its including lexical path.
121        requested: PathBuf,
122        /// Canonical source identity retained by the loader.
123        canonical: PathBuf,
124        /// Original include occurrence.
125        location: Location,
126    },
127}
128
129impl From<ExpansionFailure> for BundleAdmissionError {
130    fn from(error: ExpansionFailure) -> Self {
131        Self::Expansion(error)
132    }
133}
134impl From<AdmissionFailure> for BundleAdmissionError {
135    fn from(error: AdmissionFailure) -> Self {
136        Self::Expansion(error.into())
137    }
138}
139
140impl BundleAdmissionError {
141    /// Located diagnostics whose identities resolve in the retained bundle.
142    #[must_use]
143    pub fn diagnostics(&self) -> Vec<Diagnostic> {
144        match self {
145            Self::Expansion(error) => error.diagnostics(),
146            Self::RootAlias { location, .. }
147            | Self::IncludeAlias { location, .. }
148            | Self::IncludeRedirection { location, .. } => {
149                vec![crate::diagnostic::diagnostic(
150                    "include-identity",
151                    self.to_string(),
152                    *location,
153                )]
154            }
155        }
156    }
157}

159impl fmt::Display for BundleAdmissionError {
160    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
161        match self {
162            Self::Expansion(error) => error.fmt(f),
163            Self::RootAlias {
164                first, repeated, ..
165            } => write!(
166                f,
167                "bundle admission cannot merge lexical root aliases {} and {}",
168                first.display(),
169                repeated.display()
170            ),
171            Self::IncludeAlias {
172                first, repeated, ..
173            } => write!(
174                f,
175                "bundle admission cannot merge lexical include aliases {} and {}",
176                first.display(),
177                repeated.display()
178            ),
179            Self::IncludeRedirection {
180                requested,
181                canonical,
182                ..
183            } => write!(
184                f,
185                "bundle admission cannot redirect lexical include {} to {}",
186                requested.display(),
187                canonical.display()
188            ),
189        }
190    }
191}
192impl std::error::Error for BundleAdmissionError {
193    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
194        match self {
195            Self::Expansion(error) => Some(error),
196            _ => None,
197        }
198    }
199}

201/// A failure retains the full loaded source catalog, so diagnostics never lose
202/// the original bytes behind their source identities. No partial core program
203/// is returned.
204#[derive(Debug)]
205pub struct BundleAdmissionFailure {
206    bundle: SourceBundle,
207    error: Box<BundleAdmissionError>,
208}
209impl BundleAdmissionFailure {
210    /// Original files needed to render every diagnostic.
211    #[must_use]
212    pub fn bundle(&self) -> &SourceBundle {
213        &self.bundle
214    }
215    /// The typed semantic refusal.
216    #[must_use]
217    pub fn error(&self) -> &BundleAdmissionError {
218        &self.error
219    }
220    /// Source-qualified diagnostic view.
221    #[must_use]
222    pub fn diagnostics(&self) -> Vec<Diagnostic> {
223        self.error.diagnostics()
224    }
225    /// Recover the loaded sources, for example to retry with different budgets.
226    #[must_use]
227    pub fn into_bundle(self) -> SourceBundle {
228        self.bundle
229    }
230}
231impl fmt::Display for BundleAdmissionFailure {
232    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
233        self.error.fmt(f)
234    }
235}
236impl std::error::Error for BundleAdmissionFailure {
237    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
238        Some(self.error.as_ref())
239    }
240}

242/// Admit the extended profile across an original include graph, interpreting
243/// its ordered roots and captured include resolutions. Rules remain in the
244/// parameter-free base part, whether implicit or explicitly declared. Every file
245/// is checked before raising; owned statements are combined
246/// through themelios's provenance-merging constructor without concatenating or
247/// regenerating text. Scalar constants are global, unambiguous, and acyclic.
248///
249/// Identical resolved lexical include paths are included once, as clingo does.
250/// Alias paths and symlink redirections are conservatively refused because the
251/// canonical loader cannot recover their separate parse occurrences. Ordinary
252/// `..` traversal to a unique source is supported. Positive `#defined` and
253/// signature/empty `#show` are global source/display metadata. Named or
254/// parameterized `#program` parts and other unsupported constructs are refused.
255/// No external engine is invoked.
256///
257/// # Errors
258/// Returns a typed refusal together with the complete original source bundle.
259/// Syntax/expansion budgets count globally; no successful partial program is
260/// exposed after a refusal.
261pub fn admit_bundle_extended(
262    bundle: SourceBundle,
263    options: BundleAdmissionOptions,
264    limits: ExpansionLimits,
265) -> Result<AdmittedBundle, BundleAdmissionFailure> {
266    admit_bundle_with_control(bundle, options, limits, None)
267}

269/// Admit the original bundle with shared cancellation and deadline control.
270/// Each bounded upstream source inspection is bracketed by polls; expansion
271/// uses the same token throughout its existing checked operations.
272///
273/// # Errors
274/// Retains the original bundle with ordinary admission errors or interruption.
275pub fn admit_bundle_extended_with_cancellation(
276    bundle: SourceBundle,
277    options: BundleAdmissionOptions,
278    limits: ExpansionLimits,
279    cancellation: &zetesis_cpu::Cancellation,
280) -> Result<AdmittedBundle, BundleAdmissionFailure> {
281    admit_bundle_with_control(bundle, options, limits, Some(cancellation.clone()))
282}
283
284fn admit_bundle_with_control(
285    bundle: SourceBundle,
286    options: BundleAdmissionOptions,
287    limits: ExpansionLimits,
288    cancellation: Option<zetesis_cpu::Cancellation>,
289) -> Result<AdmittedBundle, BundleAdmissionFailure> {
290    match compile_bundle(&bundle, options, limits, cancellation) {
291        Ok(compiled) => Ok(AdmittedBundle {
292            program: compiled.program,
293            bundle,
294            template_origins: compiled.template_origins,
295            metadata: compiled.metadata,
296            expansion: compiled.expansion,
297        }),
298        Err(error) => Err(BundleAdmissionFailure {
299            bundle,
300            error: Box::new(error),
301        }),
302    }
303}
304
305fn compile_bundle(
306    bundle: &SourceBundle,
307    options: BundleAdmissionOptions,
308    limits: ExpansionLimits,
309    cancellation: Option<zetesis_cpu::Cancellation>,
310) -> Result<extended::Compilation, BundleAdmissionError> {
311    let budget = crate::expansion::Budget::new(limits, options.core_limits.max_templates)
312        .with_cancellation(cancellation);
313    budget.poll(crate::ProgramSite::program())?;
314    check_include_identity(bundle)?;
315    let mut definitions = BTreeMap::new();
316    let mut statements = Vec::new();
317    let mut visited_nodes = 0;
318    let mut metadata_count = 0;
319    let mut source_metadata = metadata::Builder::default();
320    for source in bundle.sources() {
321        budget.poll(crate::ProgramSite::source(Location {
322            source: source.id(),
323            span: source.source().span(),
324        }))?;
325        let local = AdmissionOptions {
326            source_id: source.id(),
327            max_source_bytes: 0,
328            max_syntax_nodes: options.max_syntax_nodes - visited_nodes,
329            max_syntax_depth: options.max_syntax_depth,
330            max_body_elements: options.max_body_elements,
331            core_limits: options.core_limits,
332        };
333        let nodes = profile::check_bundle(source.parsed(), local).map_err(|error| match error {
334            AdmissionFailure::Limit {
335                resource: InputLimit::SyntaxNodes,
336                observed,
337                location,
338                ..
339            } => AdmissionFailure::Limit {
340                resource: InputLimit::SyntaxNodes,
341                limit: options.max_syntax_nodes,
342                observed: visited_nodes.saturating_add(observed),
343                location,
344            },
345            other => other,
346        })?;
347        visited_nodes += nodes;
348        extended::check_definitions_in(source.parsed(), limits, &mut definitions)?;
349        metadata::check_count(source.parsed(), limits, &mut metadata_count)?;
350        let raised = raise(source.parsed());
351        if !raised.diagnostics().is_empty() {
352            return Err(AdmissionFailure::Raise(raised.diagnostics().to_vec()).into());
353        }
354        budget.poll(crate::ProgramSite::source(Location {
355            source: source.id(),
356            span: source.source().span(),
357        }))?;
358        metadata::collect(raised.program(), &mut source_metadata)?;
359        statements.extend(
360            raised
361                .program()
362                .statements()
363                .filter(|carrier| !matches!(carrier.get(), Statement::Include(_)))
364                .cloned(),
365        );
366    }
367    let entry = bundle
368        .get(bundle.entry())
369        .expect("a loaded bundle has its entry");
370    let location = Location {
371        source: entry.id(),
372        span: entry.source().span(),
373    };
374    let source = SourceProgram::of_nodes(statements);
375    Ok(extended::compile_owned(
376        &source,
377        options.core_limits,
378        budget,
379        location.into(),
380        source_metadata.finish(location.into())?,
381    )?)
382}

384pub(crate) fn check_include_identity(bundle: &SourceBundle) -> Result<(), BundleAdmissionError> {
385    let mut paths = BTreeMap::new();
386    for root in bundle.roots() {
387        if let Some(first) = paths.get(&root.id()) {
388            let first: &PathBuf = first;
389            if first.as_os_str() != root.requested_path().as_os_str() {
390                let source = bundle.get(root.id()).expect("a loaded root has a source");
391                return Err(BundleAdmissionError::RootAlias {
392                    first: first.clone(),
393                    repeated: root.requested_path().to_path_buf(),
394                    location: Location {
395                        source: source.id(),
396                        span: source.source().span(),
397                    },
398                });
399            }
400            continue;
401        }
402        paths.insert(root.id(), root.requested_path().to_path_buf());
403        let mut pending = vec![(root.id(), 0)];
404        while let Some((id, next)) = pending.last_mut() {
405            let source = bundle.get(*id).expect("a loaded graph has every source");
406            let Some(edge) = source.includes().get(*next) else {
407                pending.pop();
408                continue;
409            };
410            *next += 1;
411            let requested = edge.resolved_path();
412            if let Some(first) = paths.get(&edge.target()) {
413                if first.as_os_str() != requested.as_os_str() {
414                    return Err(BundleAdmissionError::IncludeAlias {
415                        first: first.clone(),
416                        repeated: requested.to_path_buf(),
417                        location: edge.location(),
418                    });
419                }
420                continue;
421            }
422            let target = bundle
423                .get(edge.target())
424                .expect("a loaded include has its target");
425            let checked = match edge.resolution() {
426                crate::IncludeResolution::IncludingDirectory => source
427                    .path()
428                    .parent()
429                    .expect("a canonical source has a parent")
430                    .join(edge.requested_path()),
431                crate::IncludeResolution::Absolute | crate::IncludeResolution::WorkingDirectory => {
432                    bundle.working_directory().join(edge.requested_path())
433                }
434            };
435            if dot_normalized(&checked) != target.path() {
436                return Err(BundleAdmissionError::IncludeRedirection {
437                    requested: requested.to_path_buf(),
438                    canonical: target.path().to_path_buf(),
439                    location: edge.location(),
440                });
441            }
442            paths.insert(edge.target(), requested.to_path_buf());
443            pending.push((edge.target(), 0));
444        }
445    }
446    Ok(())
447}

449fn dot_normalized(path: &Path) -> PathBuf {
450    let mut result = PathBuf::new();
451    for component in path.components() {
452        match component {
453            Component::CurDir => {}
454            Component::ParentDir => {
455                result.pop();
456            }
457            other => result.push(other.as_os_str()),
458        }
459    }
460    result
