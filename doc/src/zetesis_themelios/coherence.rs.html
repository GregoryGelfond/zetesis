1//! Signed identity and coherence at the complete source-admission boundary.
2//!
3//! Constraints join equal tuples of opposite signs. They never supply support,
4//! enumerate a domain product, or invent a predicate absent from the source.
5

16fn patterns(template: &Template) -> impl Iterator<Item = &AtomPattern> {
17    template
18        .head()
19        .into_iter()
20        .chain(template.positive())
21        .chain(template.gate_true())
22        .chain(template.gate_false())
23}

25pub(crate) fn core_sign(sign: themelios_program::symbol::Sign) -> Sign {
26    match sign {
27        themelios_program::symbol::Sign::Positive => Sign::Positive,
28        themelios_program::symbol::Sign::Negative => Sign::Negative,
29    }
30}

32pub(crate) fn source_sign(sign: Sign) -> themelios_program::symbol::Sign {
33    match sign {
34        Sign::Positive => themelios_program::symbol::Sign::Positive,
35        Sign::Negative => themelios_program::symbol::Sign::Negative,
36    }
37}

39pub(crate) fn append<E: From<AdmissionFailure>>(
40    templates: &mut Vec<Template>,
41    origins: &mut Vec<Vec<Location>>,
42    limits: AdmissionLimits,
43    fallback: Location,
44    mut charge: impl FnMut(ExpansionResource, u128, Location) -> Result<(), E>,
45) -> Result<(), E> {
46    // Preserve existing unsigned budgets and avoid a registry allocation on
47    // the common unsigned route. Traversal is bounded by the source templates.
48    if !templates
49        .iter()
50        .flat_map(patterns)
51        .any(|pattern| pattern.predicate().sign() == Sign::Negative)
52    {
53        return Ok(());
54    }
55    let mut signatures = BTreeMap::new();
56    for (index, template) in templates.iter().enumerate() {
57        let location = origins[index].first().copied().unwrap_or(fallback);
58        for pattern in patterns(template) {
59            charge(ExpansionResource::TermWork, 1, location)?;
60            signatures.entry(pattern.predicate()).or_insert(index);
61        }
62    }
63    let mut generated = Vec::new();
64    let mut evidence = Vec::new();
65    for (&negative, &index) in &signatures {
66        if negative.sign() != Sign::Negative {
67            continue;
68        }
69        let location = origins[index].first().copied().unwrap_or(fallback);
70        charge(
71            ExpansionResource::ScalarBytes,
72            negative.name().len() as u128,
73            location,
74        )?;
75        let positive = Predicate::new(negative.name(), negative.arity())
76            .expect("the original predicate has a nonempty name");
77        let Some(&opposite) = signatures.get(&positive) else {
78            continue;
79        };
80        check_core_limits(
81            index,
82            negative.arity(),
83            templates.len() + generated.len() + 1,
84            limits,
85            location,
86        )?;
87        charge(ExpansionResource::Templates, 1, location)?;
88        charge(
89            ExpansionResource::TermWork,
90            2 * negative.arity() as u128 + 2,
91            location,
92        )?;
93        charge(
94            ExpansionResource::ScalarBytes,
95            negative.name().len() as u128,
96            location,
97        )?;
98        charge(
99            ExpansionResource::Origins,
100            origins[index].len() as u128 + origins[opposite].len() as u128,
101            location,
102        )?;
103        let pattern = |predicate: Predicate| {
104            AtomPattern::new(
105                predicate,
106                (0..negative.arity()).map(Term::Variable).collect(),
107            )
108            .expect("one shared dense slot per argument")
109        };
110        generated.push(Template::new(
111            None,
112            vec![pattern(positive), pattern(negative.clone())],
113            Vec::new(),
114            Vec::new(),
115            Vec::new(),
116        ));
117        let mut locations = origins[index].clone();
118        locations.extend_from_slice(&origins[opposite]);
119        locations.sort_unstable();
120        locations.dedup();
121        evidence.push(locations);
122    }
123    templates.extend(generated);
124    origins.extend(evidence);
125    Ok(())
126}

128fn check_core_limits(
129    index: usize,
130    arity: usize,
131    count: usize,
132    limits: AdmissionLimits,
133    location: Location,
134) -> Result<(), AdmissionFailure> {
135    for (resource, actual, limit) in [
136        (AdmissionResource::Templates, count, limits.max_templates),
137        (
138            AdmissionResource::PredicateArity,
139            arity,
140            limits.max_predicate_arity,
141        ),
142        (
143            AdmissionResource::Variables,
144            arity,
145            limits.max_variables_per_template,
146        ),
147        (AdmissionResource::PositiveBody, 2, limits.max_positive_body),
148    ] {
149        if actual > limit {
150            return Err(AdmissionFailure::Core {
151                error: AdmissionError::LimitExceeded {
152                    resource,
153                    limit,
154                    actual,
155                    template: Some(index),
156                },
157                location,
158            });
159        }
160    }
161    Ok(())
162}
