1//! One classified source sequence with evidence parallel to its two streams.
2
3use zetesis_core::{
4    AdmissionError, AdmissionLimits, AdmissionOrigin, FactSegment, FactTables, Predicate, Program,
5    Template, Value,
6};
7
8/// A source rule is classified once, before native admission.
9pub(crate) enum Rule {
10    Template(Template),
11    Fact {
12        predicate: Predicate,
13        values: Vec<Value>,
14    },
15}
16
17/// Construction-time order, compact across adjacent fact families. Native fact
18/// anchors count templates; evidence continues to count original fact rows.
19pub(crate) enum Item {
20    Template(usize),
21    Facts { end: usize },
22}
23
24#[derive(Clone, Copy)]
25pub(crate) enum Occurrence {
26    Template(usize),
27    Fact(usize),
28}
29
30pub(crate) struct Compiled<P> {
31    pub templates: Vec<Template>,
32    pub facts: FactTables,
33    pub template_origins: Vec<Vec<P>>,
34    pub fact_origins: Vec<Vec<P>>,
35    pub order: Vec<Item>,
36}
37
38impl<P> Default for Compiled<P> {
39    fn default() -> Self {
40        Self {
41            templates: Vec::new(),
42            facts: FactTables::default(),
43            template_origins: Vec::new(),
44            fact_origins: Vec::new(),
45            order: Vec::new(),
46        }
47    }
48}
49
50#[derive(Debug)]
51pub(crate) struct Admitted<P> {
52    pub program: Program,
53    pub template_origins: Vec<Vec<P>>,
54    pub fact_origins: Vec<Vec<P>>,
55}
56
57impl<P: Clone> Compiled<P> {
58    pub(crate) fn len(&self) -> usize {
59        self.templates.len() + self.facts.len()
60    }
61
62    pub(crate) fn origins(&self, occurrence: Occurrence) -> &[P] {
63        match occurrence {
64            Occurrence::Template(index) => &self.template_origins[index],
65            Occurrence::Fact(index) => &self.fact_origins[index],
66        }
67    }
68
69    pub(crate) fn push(&mut self, rule: Rule, origins: Vec<P>) -> Result<(), AdmissionError> {
70        match rule {
71            Rule::Template(template) => {
72                let index = self.templates.len();
73                let failure = || AdmissionError::Canonical {
74                    error: zetesis_core::catalog::Error::Allocation,
75                    origin: AdmissionOrigin::Template(index),
76                };
77                self.templates.try_reserve(1).map_err(|_| failure())?;
78                self.template_origins
79                    .try_reserve(1)
80                    .map_err(|_| failure())?;
81                self.order.try_reserve(1).map_err(|_| failure())?;
82                self.templates.push(template);
83                self.template_origins.push(origins);
84                self.order.push(Item::Template(index));
85                Ok(())
86            }
87            Rule::Fact { predicate, values } => {
88                let first = self.facts.len();
89                self.facts
90                    .push_row(self.templates.len(), predicate, values)
91                    .map_err(|error| AdmissionError::Canonical {
92                        error,
93                        origin: AdmissionOrigin::Fact(first),
94                    })?;
95                self.record_facts(first, &origins, || Ok(()), |error| error)
96            }
97        }
98    }
99
100    pub(crate) fn product<E>(
101        &mut self,
102        predicate: Predicate,
103        columns: Vec<Vec<FactSegment>>,
104        origins: &[P],
105        before: impl FnMut() -> Result<(), E>,
106        failure: impl Fn(AdmissionError) -> E,
107    ) -> Result<(), E> {
108        let first = self.facts.len();
109        self.facts
110            .push_product(self.templates.len(), predicate, columns)
111            .map_err(|error| {
112                failure(AdmissionError::Canonical {
113                    error,
114                    origin: AdmissionOrigin::Fact(first),
115                })
116            })?;
117        self.record_facts(first, origins, before, failure)
118    }
119
120    fn record_facts<E>(
121        &mut self,
122        first: usize,
123        origins: &[P],
124        mut before: impl FnMut() -> Result<(), E>,
125        failure: impl Fn(AdmissionError) -> E,
126    ) -> Result<(), E> {
127        let end = self.facts.len();
128        if first == end {
129            return Ok(());
130        }
131        let allocation = || {
132            failure(AdmissionError::Canonical {
133                error: zetesis_core::catalog::Error::Allocation,
134                origin: AdmissionOrigin::Fact(first),
135            })
136        };
137        self.fact_origins
138            .try_reserve(end - first)
139            .map_err(|_| allocation())?;
140        self.order.try_reserve(1).map_err(|_| allocation())?;
141        for _ in first..end {
142            before()?;
143            self.fact_origins.push(origins.to_vec());
144        }
145        if let Some(Item::Facts { end: previous }) = self.order.last_mut() {
146            *previous = end;
147        } else {
148            self.order.push(Item::Facts { end });
149        }
150        Ok(())
151    }
152
153    pub(crate) fn admit<E>(
154        self,
155        limits: AdmissionLimits,
156        failure: impl FnOnce(AdmissionError, &[P]) -> E,
157    ) -> Result<Admitted<P>, E> {
158        let Self {
159            templates,
160            facts,
161            template_origins,
162            fact_origins,
163            order: _,
164        } = self;
165        let program = Program::new(templates, facts, limits).map_err(|error| {
166            let origins = error
167                .template_index()
168                .and_then(|index| template_origins.get(index))
169                .or_else(|| error.fact_index().and_then(|index| fact_origins.get(index)))
170                .map_or(&[][..], Vec::as_slice);
171            failure(error, origins)
172        })?;
173        Ok(Admitted {
174            program,
175            template_origins,
176            fact_origins,
177        })
178    }
179}
