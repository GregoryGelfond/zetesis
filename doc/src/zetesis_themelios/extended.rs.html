27/// Admit a bounded extension of S0: unannotated acyclic scalar `#const`
28/// definitions; checked ground scalar arithmetic in ordinary S0 terms; and
29/// finite pools/intervals in ordinary fact heads. A bare undefined constant name
30/// remains a symbolic value. Definition names do not rename predicates.
31///
32/// Intervals have scalar integer endpoints; arithmetic over a set, variable
33/// arithmetic, and pools/intervals in non-fact rules remain unsupported. A
34/// descending interval denotes no facts. Fact products are bounded before
35/// allocation. The resulting core program uses the existing S0 semantics.
36/// Original source bytes and parsed rule locations are preserved; no source text
37/// is regenerated. Signed `#defined` signatures and signature/empty `#show`
38/// are retained as metadata; display selection never changes model identity.
39/// Includes require bundle admission and other directives remain unsupported.
40///
41/// # Errors
42/// Refuses every parser/raiser diagnostic, unsupported construct, ambiguous or
43/// cyclic definition, undefined/overflowing scalar operation, expansion limit,
44/// or independent core admission error. No successful partial program is
45/// returned. This partial profile refuses undefined arithmetic rather than
46/// emulating clingo's warning-and-drop behavior.
47pub fn admit_extended(
48    text: String,
49    options: AdmissionOptions,
50    limits: ExpansionLimits,
51) -> Result<Admitted, ExpansionFailure> {
52    ParsedSource::new(text, options)?
53        .admit_extended(limits)
54        .map_err(SourceFailure::into_error)
55}

65pub(crate) fn admit_parsed(
66    source: ParsedSource,
67    limits: ExpansionLimits,
68) -> Result<Admitted, SourceFailure<ExpansionFailure>> {
69    match compile_parsed(&source, limits) {
70        Ok(compiled) => Ok(Admitted {
71            program: compiled.program,
72            source: source.into_source(),
73            template_origins: compiled.template_origins,
74            metadata: compiled.metadata,
75            expansion: compiled.expansion,
76        }),
77        Err(error) => Err(SourceFailure::new(source, error)),
78    }
79}

81fn compile_parsed(
82    source: &ParsedSource,
83    limits: ExpansionLimits,
84) -> Result<Compilation, ExpansionFailure> {
85    let parsed = source.parsed();
86    let options = source.options();
87    profile::check_extended(parsed, options)?;
88    check_definitions(parsed, limits)?;
89    metadata::check_count(parsed, limits, &mut 0)?;
90    let raised = raise(parsed);
91    if !raised.diagnostics().is_empty() {
92        return Err(AdmissionFailure::Raise(raised.diagnostics().to_vec()).into());
93    }
94    let mut source_metadata = metadata::Builder::default();
95    metadata::collect(raised.program(), &mut source_metadata)?;
96    let location = Location {
97        source: source.source().id(),
98        span: source.source().span(),
99    };
100    compile_owned(
101        raised.program(),
102        options.core_limits,
103        limits,
104        location.into(),
105        source_metadata.finish(location.into())?,
106    )
107}

109/// Compile the source into the program with its template origins and the
110/// expansion charges it accepted, as one compilation with `metadata`.
111pub(crate) fn compile_owned(
112    source: &SourceProgram,
113    core_limits: AdmissionLimits,
114    limits: ExpansionLimits,
115    location: ProgramSite,
116    metadata: SourceMetadata,
117) -> Result<Compilation, ExpansionFailure> {
118    let mut budget = Budget::new(limits, core_limits.max_templates);
119    let (program, template_origins) = compile_relational(
120        source,
121        core_limits,
122        &mut budget,
123        location,
124        FormulaPurpose::Ordinary,
125        |carrier, _| parsed_origins(carrier),
126        |location| ProgramSite::from(*location),
127    )?;
128    Ok(Compilation {
129        program,
130        template_origins,
131        metadata,
132        expansion: budget.usage(),
133    })
134}

136/// Share normalization, fact expansion and signed coherence while retaining the
137/// caller's evidence representation and one cumulative preparation budget.
138pub(crate) fn compile_relational<P: Clone + Ord>(
139    source: &SourceProgram,
140    core_limits: AdmissionLimits,
141    budget: &mut Budget,
142    fallback: ProgramSite,
143    purpose: FormulaPurpose,
144    evidence: impl Fn(&WithProvenance<Statement>, ProgramSite) -> Vec<P>,
145    site: impl Fn(&P) -> ProgramSite,
146) -> Result<(Program, Vec<Vec<P>>), ExpansionFailure> {
147    budget.poll(fallback)?;
148    let constants = resolve(source, budget, fallback)?;
149    let (mut templates, mut origins) =
150        compile_extended(source, &constants, budget, fallback, purpose, evidence)?;
151    budget.poll(fallback)?;
152    crate::coherence::append_with(
153        &mut templates,
154        &mut origins,
155        core_limits,
156        |resource, count, evidence| {
157            budget.charge(resource, count, evidence.first().map_or(fallback, &site))
158        },
159        |error, evidence| {
160            ExpansionFailure::from(AdmissionFailure::Core {
161                error,
162                location: evidence.first().map_or(fallback, &site),
163            })
164        },
165    )?;
166    budget.poll(fallback)?;
167    let program = Program::new(templates, core_limits).map_err(|error| {
168        let location = error
169            .template_index()
170            .and_then(|index| origins.get(index))
171            .and_then(|evidence| evidence.first())
172            .map_or(fallback, site);
173        AdmissionFailure::Core { error, location }
174    })?;
175    budget.poll(fallback)?;
176    Ok((program, origins))
177}

187pub(crate) fn check_definitions_in(
188    parsed: &Parse<ast::Program>,
189    limits: ExpansionLimits,
190    names: &mut BTreeMap<String, Location>,
191) -> Result<(), ExpansionFailure> {
192    for statement in parsed.tree().statements() {
193        let ast::Statement::Const(constant) = statement else {
194            continue;
195        };
196        let location = parsed.location(constant.syntax().text_range());
197        if constant.annotation().is_some() {
198            return Err(ExpansionFailure::ConstantPolicy {
199                location: location.into(),
200            });
201        }
202        let Some(name) = constant.name() else {
203            return Err(unsupported(ProfileFeature::Statement, location).into());
204        };
205        if let Some(first) = names.insert(name.text().to_owned(), location) {
206            return Err(ExpansionFailure::DuplicateConstant {
207                name: name.text().to_owned(),
208                first: first.into(),
209                duplicate: location.into(),
210            });
211        }
212        check(
213            ExpansionResource::Constants,
214            names.len() as u128,
215            limits.max_constants,
216            location.into(),
217        )?;
218    }
219    Ok(())
220}

228pub(crate) fn resolve(
229    source: &SourceProgram,
230    budget: &mut Budget,
231    fallback: ProgramSite,
232) -> Result<BTreeMap<String, Symbol>, ExpansionFailure> {
233    let mut definitions: BTreeMap<String, Definition<'_>> = BTreeMap::new();
234    for (index, carrier) in source.statements().enumerate() {
235        let location = origin(carrier, fallback.with_statement(StatementId::new(index)));
236        budget.poll(location)?;
237        let Statement::Const(constant) = carrier.get() else {
238            continue;
239        };
240        if constant.policy.is_some() {
241            return Err(ExpansionFailure::ConstantPolicy { location });
242        }
243        // Canonical declarations are identified by their content. Additional
244        // provenance records evidence, not another definition of the name.
245        if let Some(first) = definitions
246            .get(constant.name.as_str())
247            .map(|entry| entry.location)
248        {
249            return Err(ExpansionFailure::DuplicateConstant {
250                name: constant.name.as_str().to_owned(),
251                first,
252                duplicate: location,
253            });
254        }
255        budget.check_constants(definitions.len().saturating_add(1), location)?;
256        definitions.insert(
257            constant.name.as_str().to_owned(),
258            Definition {
259                constant,
260                location,
261                dependencies: Vec::new(),
262            },
263        );
264    }
265    let names: BTreeSet<_> = definitions.keys().cloned().collect();
266    for definition in definitions.values_mut() {
267        let mut dependencies = BTreeSet::new();
268        for term in definition.constant.value.subterms() {
269            budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
270            if let Term::Symbolic(symbol) = term {
271                for nested in symbol.subsymbols() {
272                    if !symbol.arguments().is_empty() {
273                        budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
274                    }
275                    if let Some(name) = constant_name(nested).filter(|name| names.contains(*name)) {
276                        dependencies.insert(name.to_owned());
277                    }
278                }
279            }
280        }
281        definition.dependencies = dependencies.into_iter().collect();
282    }
283    let mut values = BTreeMap::new();
284    let mut active = BTreeSet::new();
285    for name in definitions.keys() {
286        if values.contains_key(name) {
287            continue;
288        }
289        active.insert(name.clone());
290        let mut stack = vec![(name.clone(), 0)];
291        while let Some((name, next)) = stack.last_mut() {
292            let definition = &definitions[name];
293            budget.poll(definition.location)?;
294            if let Some(dependency) = definition.dependencies.get(*next) {
295                *next += 1;
296                if values.contains_key(dependency) {
297                    continue;
298                }
299                if active.contains(dependency) {
300                    let mut names: Vec<_> = stack.iter().map(|(name, _)| name.clone()).collect();
301                    names.push(dependency.clone());
302                    return Err(ExpansionFailure::ConstantCycle {
303                        names,
304                        location: definition.location,
305                    });
306                }
307                active.insert(dependency.clone());
308                stack.push((dependency.clone(), 0));
309                continue;
310            }
311            let value = definition.constant.value.clone().try_fold(|parts| {
312                normalize_node(Term::from(parts), &values, budget, definition.location)
313            })?;
314            let TermParts::Symbolic(value) = value.into_parts() else {
315                return Err(ExpansionFailure::Evaluation {
316                    error: EvalError::Undefined,
317                    location: definition.location,
318                });
319            };
320            values.insert(name.clone(), value);
321            active.remove(name);
322            stack.pop();
323        }
324    }
325    Ok(values)
326}

330/// Normalize each statement and compile it. A statement whose every term
331/// normalization would return as it is, charged and validated by
332/// [`kept_as_is`], is compiled from the original carrier: the rewrite would
333/// hand back the same statement, stamped with a transformation tag that
334/// nothing after this point reads, at the cost of copying and rebuilding it.
335/// Every other statement is rebuilt by the rewrite, alone in a program of
336/// its own so that no two statements merge.
337fn compile_extended<P: Clone>(
338    source: &SourceProgram,
339    constants: &BTreeMap<String, Symbol>,
340    budget: &mut Budget,
341    fallback: ProgramSite,
342    purpose: FormulaPurpose,
343    evidence: impl Fn(&WithProvenance<Statement>, ProgramSite) -> Vec<P>,
344) -> Result<Compiled<P>, ExpansionFailure> {
345    let mut compiled = (Vec::new(), Vec::new());
346    for (index, carrier) in source.statements().enumerate() {
347        let location = origin(carrier, fallback.with_statement(StatementId::new(index)));
348        budget.poll(location)?;
349        if !purpose.includes(carrier.get())
350            || matches!(
351                carrier.get(),
352                Statement::Const(_) | Statement::Defined(_) | Statement::Show(_)
353            )
354        {
355            continue;
356        }
357        let locations = evidence(carrier, location);
358        if kept_as_is(carrier.get(), constants, budget, location)? {
359            emit(carrier, locations, budget, location, &mut compiled)?;
360            continue;
361        }
362        let mut normalizer = Normalizer {
363            constants,
364            budget,
365            location,
366            failure: None,
367        };
368        let rewritten = rewrite(SourceProgram::of_nodes([carrier.clone()]), &mut normalizer);
369        if let Some(failure) = normalizer.failure {
370            return Err(failure);
371        }
372        budget.poll(location)?;
373        let normalized_carrier = rewritten
374            .statements()
375            .next()
376            .expect("a rule rewrite keeps its statement");
377        emit(
378            normalized_carrier,
379            locations,
380            budget,
381            location,
382            &mut compiled,
383        )?;
384    }
385    Ok(compiled)
386}

388/// Emit one normalized statement's templates, each with the statement's
389/// original evidence: its facts, expanded, or the one template of its rule.
390fn emit<P: Clone>(
391    statement: &WithProvenance<Statement>,
392    locations: Vec<P>,
393    budget: &mut Budget,
394    location: ProgramSite,
395    (templates, origins): &mut Compiled<P>,
396) -> Result<(), ExpansionFailure> {
397    if let Some(facts) = fact_expansion::facts(statement, budget, location)? {
398        budget.charge(
399            ExpansionResource::Origins,
400            (facts.len() as u128).saturating_mul(locations.len() as u128),
401            location,
402        )?;
403        for fact in facts {
404            templates.push(fact);
405            origins.push(locations.clone());
406        }
407    } else {
408        budget.charge(ExpansionResource::Templates, 1, location)?;
409        budget.charge(
410            ExpansionResource::Origins,
411            locations.len() as u128,
412            location,
413        )?;
414        templates.push(compile::statement(statement, location)?);
415        origins.push(locations);
416    }
417    Ok(())
418}

420/// Whether normalization would return `statement` exactly as it is. Its
421/// terms are charged and validated as the rewrite would charge and validate
422/// them, in the rewrite's order, on a tentative budget that becomes the
423/// budget only when every term is a [`Leaf`]; otherwise the budget is left
424/// as it was, and the rewrite charges the statement from the start. A
425/// failure is the one the rewrite would raise first, since it arises before
426/// any term the rewrite would rebuild. Nothing is cloned but the budget.
427fn kept_as_is(
428    statement: &Statement,
429    constants: &BTreeMap<String, Symbol>,
430    budget: &mut Budget,
431    location: ProgramSite,
432) -> Result<bool, ExpansionFailure> {
433    let Statement::Rule(rule) = statement else {
434        return Ok(false);
435    };
436    let mut tentative = budget.clone();
437    let kept = every_term(rule, &mut |term| match Leaf::of(term, constants) {
438        Some(leaf) => leaf.account(&mut tentative, location).map(|()| true),
439        None => Ok(false),
440    })?;
441    if kept {
442        *budget = tentative;
443    }
444    Ok(kept)
445}

447/// Apply `keep` to the terms of a rule of the shape this profile compiles,
448/// in the order the rewrite rebuilds them: the head, then each body element;
449/// an atom's arguments in order, a comparison's first term and then its
450/// steps. `Ok(false)` at the first term `keep` refuses, or at once for a
451/// head or body element of another shape, which the rewrite descends in an
452/// order this walk does not reproduce.
453fn every_term(
454    rule: &Rule,
455    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
456) -> Result<bool, ExpansionFailure> {
457    // A head or body element of any other shape, including one the pinned
458    // themelios tier may add, is left to the rewrite.
459    let head = match rule.head().get() {
460        Head::Literal(literal) => literal_terms(literal, keep)?,
461        Head::Falsum | Head::Verum => true,
462        _ => false,
463    };
464    if !head {
465        return Ok(false);
466    }
467    for element in rule.body().get().elements() {
468        let kept = match element.get() {
469            BodyElement::Literal(literal) => literal_terms(literal, keep)?,
470            _ => false,
471        };
472        if !kept {
473            return Ok(false);
474        }
475    }
476    Ok(true)
477}
478
479fn literal_terms(
480    literal: &Literal,
481    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
482) -> Result<bool, ExpansionFailure> {
483    match &literal.inner {
484        LiteralInner::Atom(atom) => {
485            for term in atom.get().argument_terms() {
486                if !keep(term)? {
487                    return Ok(false);
488                }
489            }
490            Ok(true)
491        }
492        LiteralInner::Comparison(comparison) => {
493            let comparison = comparison.get();
494            if !keep(comparison.first())? {
495                return Ok(false);
496            }
497            for (_, term) in comparison.steps() {
498                if !keep(term)? {
499                    return Ok(false);
500                }
501            }
502            Ok(true)
503        }
504        LiteralInner::True | LiteralInner::False => Ok(true),
505    }
506}

508/// A term [`normalize_node`] returns exactly as it is, charged and validated
509/// by [`Leaf::account`]: a variable, or an argument-free scalar naming no
510/// definition. Every other term is rebuilt, or, for a pool or interval
511/// former, kept around terms that may not be.
512#[derive(Clone, Copy)]
513enum Leaf<'a> {
514    Variable,
515    Scalar(&'a Symbol),
516}
517
518impl<'a> Leaf<'a> {
519    fn of(term: &'a Term, constants: &BTreeMap<String, Symbol>) -> Option<Self> {
520        match term {
521            Term::Variable(_) => Some(Self::Variable),
522            Term::Symbolic(symbol)
523                if symbol.arguments().is_empty() && defined(symbol, constants).is_none() =>
524            {
525                Some(Self::Scalar(symbol))
526            }
527            _ => None,
528        }
529    }
530
531    /// Charge and validate as [`normalize_node`] does for this term: its unit
532    /// of term work first, then a scalar's validation and payload.
533    fn account(self, budget: &mut Budget, location: ProgramSite) -> Result<(), ExpansionFailure> {
534        budget.charge(ExpansionResource::TermWork, 1, location)?;
535        match self {
536            Self::Variable => Ok(()),
537            Self::Scalar(symbol) => scalar_leaf(symbol, budget, location),
538        }
539    }
540}

573/// Statement evidence retains logical identity independently of parsed spans.
574/// Constructed input contributes its caller-assigned site rather than a fabricated location.
575pub(crate) fn program_sites(
576    carrier: &WithProvenance<Statement>,
577    fallback: ProgramSite,
578) -> Vec<ProgramSite> {
579    let mut sites: Vec<_> = carrier
580        .provenance()
581        .origins()
582        .filter_map(|origin| match origin {
583            Origin::Parsed(location) => Some(fallback.with_location(*location)),
584            Origin::Constructed | Origin::Transformed(_) => None,
585        })
586        .collect();
587    if sites.is_empty() {
588        sites.push(fallback);
589    }
590    sites

593/// Attach actual parsed evidence without changing a caller-assigned statement ID.
594pub(crate) fn origin(
595    carrier: &WithProvenance<Statement>,
596    fallback: impl Into<ProgramSite>,
597) -> ProgramSite {
598    let fallback = fallback.into();
599    carrier
600        .provenance()
601        .origins()
602        .find_map(|origin| match origin {
603            Origin::Parsed(location) => Some(fallback.with_location(*location)),
604            Origin::Constructed | Origin::Transformed(_) => None,
605        })
606        .unwrap_or(fallback)
607}

609struct Normalizer<'a> {
610    constants: &'a BTreeMap<String, Symbol>,
611    budget: &'a mut Budget,
612    location: ProgramSite,
613    failure: Option<ExpansionFailure>,
614}
615
616impl Rewrite for Normalizer<'_> {
617    fn tag(&self) -> TransformTag {
618        TransformTag::new("zetesis-scalar-expansion")
619    }
620    fn rewrite_term(&mut self, term: Term) -> Term {
621        if self.failure.is_some() {
622            return term;
623        }
624        match normalize_node(term, self.constants, self.budget, self.location) {
625            Ok(term) => term,
626            Err(error) => {
627                self.failure = Some(error);
628                // Only a failed private intermediate uses this placeholder;
629                // the entire rewrite is refused before compilation.
630                Term::Symbolic(Symbol::Number(0))
631            }
632        }
633    }
634}

636pub(crate) fn normalize_node(
637    term: Term,
638    constants: &BTreeMap<String, Symbol>,
639    budget: &mut Budget,
640    location: ProgramSite,
641) -> Result<Term, ExpansionFailure> {
642    budget.charge(ExpansionResource::TermWork, 1, location)?;
643    match &term {
644        Term::Variable(_) | Term::Pool(_) | Term::Interval { .. } => Ok(term),
645        Term::Symbolic(symbol) if !symbol.arguments().is_empty() => {
646            budget.charge(
647                ExpansionResource::ScalarBytes,
648                symbol_bytes(symbol),
649                location,
650            )?;
651            let resolved =
652                symbol
653                    .clone()
654                    .try_fold(|parts| -> Result<Symbol, ExpansionFailure> {
655                        budget.charge(ExpansionResource::TermWork, 1, location)?;
656                        let value = Symbol::from(parts);
657                        if let Some(replacement) = defined(&value, constants) {
658                            budget.charge(
659                                ExpansionResource::ScalarBytes,
660                                symbol_bytes(replacement),
661                                location,
662                            )?;
663                            Ok(replacement.clone())
664                        } else {
665                            Ok(value)
666                        }
667                    })?;
668            compile::validate_scalar(&resolved, location)?;
669            Ok(Term::Symbolic(resolved))
670        }
671        Term::Symbolic(symbol) => {
672            if let Some(value) = defined(symbol, constants) {
673                budget.charge(
674                    ExpansionResource::ScalarBytes,
675                    symbol_bytes(value),
676                    location,
677                )?;
678                Ok(Term::Symbolic(value.clone()))
679            } else {
680                scalar_leaf(symbol, budget, location)?;
681                Ok(term)
682            }
683        }
684        Term::UnaryOperation {
685            operator: themelios_program::term::UnaryOp::Negate,
686            argument,
687        } if matches!(argument.as_ref(), Term::Symbolic(Symbol::Function { .. })) => {
688            let Term::Symbolic(Symbol::Function {
689                name,
690                arguments,
691                sign,
692            }) = argument.as_ref()
693            else {
694                unreachable!("checked function operand")
695            };
696            budget.charge(
697                ExpansionResource::ScalarBytes,
698                symbol_bytes(match argument.as_ref() {
699                    Term::Symbolic(symbol) => symbol,
700                    _ => unreachable!("checked symbol"),
701                }),
702                location,
703            )?;
704            Ok(Term::Symbolic(Symbol::Function {
705                name: name.clone(),
706                arguments: arguments.clone(),
707                sign: match sign {
708                    Sign::Positive => Sign::Negative,
709                    Sign::Negative => Sign::Positive,
710                },
711            }))
712        }
713        Term::Function { .. } | Term::Tuple(_) if term.is_ground() => {
714            for child in term.subterms() {
715                budget.charge(ExpansionResource::TermWork, 1, location)?;
716                if matches!(child, Term::Pool(_) | Term::Interval { .. }) {
717                    return Err(unsupported(ProfileFeature::Term, location).into());
718                }
719            }
720            let symbol = term
721                .evaluate()
722                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
723            compile::validate_scalar(&symbol, location)?;
724            Ok(Term::Symbolic(symbol))
725        }
726        Term::UnaryOperation { .. } | Term::BinaryOperation { .. } | Term::Absolute(_) => {
727            let symbol = term
728                .evaluate()
729                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
730            compile::validate_scalar(&symbol, location)?;
731            Ok(Term::Symbolic(symbol))
732        }
733        _ => Err(unsupported(ProfileFeature::Term, location).into()),
734    }
735}
736
737fn constant_name(symbol: &Symbol) -> Option<&str> {
738    match symbol {
739        Symbol::Function {
740            name,
741            arguments,
742            sign: Sign::Positive,
743        } if arguments.is_empty() => Some(name.as_str()),
744        _ => None,
745    }
746}

748fn symbol_bytes(symbol: &Symbol) -> u128 {
749    crate::structural_value::symbol_bytes(symbol)
