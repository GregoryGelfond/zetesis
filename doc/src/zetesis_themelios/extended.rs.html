23pub(crate) fn check_definitions_in(
24    parsed: &Parse<ast::Program>,
25    limits: ExpansionLimits,
26    names: &mut BTreeMap<String, Location>,
27) -> Result<(), ExpansionFailure> {
28    for statement in parsed.tree().statements() {
29        let ast::Statement::Const(constant) = statement else {
30            continue;
31        };
32        let location = parsed.location(constant.syntax().text_range());
33        if constant.annotation().is_some() {
34            return Err(ExpansionFailure::ConstantPolicy {
35                location: location.into(),
36            });
37        }
38        let Some(name) = constant.name() else {
39            return Err(unsupported(ProfileFeature::Statement, location).into());
40        };
41        if let Some(first) = names.insert(name.text().to_owned(), location) {
42            return Err(ExpansionFailure::DuplicateConstant {
43                name: name.text().to_owned(),
44                first: first.into(),
45                duplicate: location.into(),
46            });
47        }
48        check(
49            ExpansionResource::Constants,
50            names.len() as u128,
51            limits.max_constants,
52            location.into(),
53        )?;
54    }
55    Ok(())
56}

64pub(crate) fn resolve(
65    source: &SourceProgram,
66    budget: &mut Budget,
67    fallback: ProgramSite,
68) -> Result<BTreeMap<String, Symbol>, ExpansionFailure> {
69    let mut definitions: BTreeMap<String, Definition<'_>> = BTreeMap::new();
70    for (index, carrier) in source.statements().enumerate() {
71        let location = origin(carrier, fallback.with_statement(StatementId::new(index)));
72        budget.poll(location)?;
73        let Statement::Const(constant) = carrier.get() else {
74            continue;
75        };
76        if constant.policy.is_some() {
77            return Err(ExpansionFailure::ConstantPolicy { location });
78        }
79        // Canonical declarations are identified by their content. Additional
80        // provenance records evidence, not another definition of the name.
81        if let Some(first) = definitions
82            .get(constant.name.as_str())
83            .map(|entry| entry.location)
84        {
85            return Err(ExpansionFailure::DuplicateConstant {
86                name: constant.name.as_str().to_owned(),
87                first,
88                duplicate: location,
89            });
90        }
91        budget.check_constants(definitions.len().saturating_add(1), location)?;
92        definitions.insert(
93            constant.name.as_str().to_owned(),
94            Definition {
95                constant,
96                location,
97                dependencies: Vec::new(),
98            },
99        );
100    }
101    let names: BTreeSet<_> = definitions.keys().cloned().collect();
102    for definition in definitions.values_mut() {
103        let mut dependencies = BTreeSet::new();
104        for term in definition.constant.value.subterms() {
105            budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
106            if let Term::Symbolic(symbol) = term {
107                for nested in symbol.subsymbols() {
108                    if !symbol.arguments().is_empty() {
109                        budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
110                    }
111                    if let Some(name) = constant_name(nested).filter(|name| names.contains(*name)) {
112                        dependencies.insert(name.to_owned());
113                    }
114                }
115            }
116        }
117        definition.dependencies = dependencies.into_iter().collect();
118    }
119    let mut values = BTreeMap::new();
120    let mut active = BTreeSet::new();
121    for name in definitions.keys() {
122        if values.contains_key(name) {
123            continue;
124        }
125        active.insert(name.clone());
126        let mut stack = vec![(name.clone(), 0)];
127        while let Some((name, next)) = stack.last_mut() {
128            let definition = &definitions[name];
129            budget.poll(definition.location)?;
130            if let Some(dependency) = definition.dependencies.get(*next) {
131                *next += 1;
132                if values.contains_key(dependency) {
133                    continue;
134                }
135                if active.contains(dependency) {
136                    let mut names: Vec<_> = stack.iter().map(|(name, _)| name.clone()).collect();
137                    names.push(dependency.clone());
138                    return Err(ExpansionFailure::ConstantCycle {
139                        names,
140                        location: definition.location,
141                    });
142                }
143                active.insert(dependency.clone());
144                stack.push((dependency.clone(), 0));
145                continue;
146            }
147            let value = definition.constant.value.clone().try_fold(|parts| {
148                normalize_node(Term::from(parts), &values, budget, definition.location)
149            })?;
150            let TermParts::Symbolic(value) = value.into_parts() else {
151                return Err(ExpansionFailure::Evaluation {
152                    error: EvalError::Undefined,
153                    location: definition.location,
154                });
155            };
156            values.insert(name.clone(), value);
157            active.remove(name);
158            stack.pop();
159        }
160    }
161    Ok(values)
162}

164/// The normalization contract a kept formula statement must match. Sentinels
165/// `#inf` and `#sup` consume one unit of term work without scalar payload work.
166#[derive(Clone, Copy, PartialEq, Eq)]
167pub(crate) enum Normalization {
168    Formula,
169}

171/// Whether `normalization` would return `statement` exactly as it is. Its
172/// terms are charged and validated as the rewrite would charge and validate
173/// them, in the rewrite's order, on a tentative budget that becomes the
174/// budget only when every term is a [`Leaf`]; otherwise the budget is left
175/// as it was, and the rewrite charges the statement from the start. A
176/// failure is the one the rewrite would raise first, since it arises before
177/// any term the rewrite would rebuild. Nothing is cloned but the budget.
178pub(crate) fn kept_as_is(
179    statement: &Statement,
180    constants: &BTreeMap<String, Symbol>,
181    normalization: Normalization,
182    budget: &mut Budget,
183    location: ProgramSite,
184) -> Result<bool, ExpansionFailure> {
185    let Statement::Rule(rule) = statement else {
186        return Ok(false);
187    };
188    #[cfg(test)]
189    if testing::rewrites_every_statement() {
190        return Ok(false);
191    }
192    let mut tentative = budget.clone();
193    let kept = every_term(rule, &mut |term| match Leaf::of(term, constants) {
194        Some(leaf) => leaf
195            .account(normalization, &mut tentative, location)
196            .map(|()| true),
197        None => Ok(false),
198    })?;
199    if kept {
200        *budget = tentative;
201        #[cfg(test)]
202        testing::note_kept();
203    }
204    Ok(kept)
205}

207/// Apply `keep` to the terms of a rule of the shape this profile compiles,
208/// in the order the rewrite rebuilds them: the head, then each body element;
209/// an atom's arguments in order, a comparison's first term and then its
210/// steps. `Ok(false)` at the first term `keep` refuses, or at once for a
211/// head or body element of another shape, which the rewrite descends in an
212/// order this walk does not reproduce.
213fn every_term(
214    rule: &Rule,
215    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
216) -> Result<bool, ExpansionFailure> {
217    // A head or body element of any other shape, including one the pinned
218    // themelios tier may add, is left to the rewrite.
219    let head = match rule.head().get() {
220        Head::Literal(literal) => literal_terms(literal, keep)?,
221        Head::Falsum | Head::Verum => true,
222        _ => false,
223    };
224    if !head {
225        return Ok(false);
226    }
227    for element in rule.body().get().elements() {
228        let kept = match element.get() {
229            BodyElement::Literal(literal) => literal_terms(literal, keep)?,
230            _ => false,
231        };
232        if !kept {
233            return Ok(false);
234        }
235    }
236    Ok(true)
237}
238
239fn literal_terms(
240    literal: &Literal,
241    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
242) -> Result<bool, ExpansionFailure> {
243    match &literal.inner {
244        LiteralInner::Atom(atom) => {
245            for term in atom.get().argument_terms() {
246                if !keep(term)? {
247                    return Ok(false);
248                }
249            }
250            Ok(true)
251        }
252        LiteralInner::Comparison(comparison) => {
253            let comparison = comparison.get();
254            if !keep(comparison.first())? {
255                return Ok(false);
256            }
257            for (_, term) in comparison.steps() {
258                if !keep(term)? {
259                    return Ok(false);
260                }
261            }
262            Ok(true)
263        }
264        LiteralInner::True | LiteralInner::False => Ok(true),
265    }
266}

268/// A term [`normalize_node`] returns exactly as it is, charged and validated
269/// by [`Leaf::account`]: a variable, an argument-free scalar naming no
270/// definition, or a ground symbol with arguments none of whose parts names a
271/// definition. Raise yields such a symbol in its canonical form, and the
272/// normalizer's symbolic branch folds it back to itself. Every other term is
273/// rebuilt, or, for a pool or interval former, kept around terms that may
274/// not be.
275#[derive(Clone, Copy)]
276enum Leaf<'a> {
277    Variable,
278    Scalar(&'a Symbol),
279    Compound(&'a Symbol),
280}
281
282impl<'a> Leaf<'a> {
283    fn of(term: &'a Term, constants: &BTreeMap<String, Symbol>) -> Option<Self> {
284        match term {
285            Term::Variable(_) => Some(Self::Variable),
286            Term::Symbolic(symbol)
287                if symbol.arguments().is_empty() && defined(symbol, constants).is_none() =>
288            {
289                Some(Self::Scalar(symbol))
290            }
291            Term::Symbolic(symbol)
292                if !symbol.arguments().is_empty()
293                    && symbol
294                        .subsymbols()
295                        .all(|part| defined(part, constants).is_none()) =>
296            {
297                Some(Self::Compound(symbol))
298            }
299            _ => None,
300        }
301    }
302
303    /// Charge and validate as `normalization` does for this term: its unit of
304    /// term work first; then a scalar's validation and payload, except a
305    /// sentinel the formula route keeps; or a compound symbol's payload, one
306    /// unit for each of its parts as the fold visits them, and its validation.
307    fn account(
308        self,
309        normalization: Normalization,
310        budget: &mut Budget,
311        location: ProgramSite,
312    ) -> Result<(), ExpansionFailure> {
313        budget.charge(ExpansionResource::TermWork, 1, location)?;
314        match self {
315            Self::Variable => Ok(()),
316            Self::Scalar(Symbol::Infimum | Symbol::Supremum)
317                if normalization == Normalization::Formula =>
318            {
319                Ok(())
320            }
321            Self::Scalar(symbol) => scalar_leaf(symbol, budget, location),
322            Self::Compound(symbol) => {
323                budget.charge(
324                    ExpansionResource::ScalarBytes,
325                    symbol_bytes(symbol),
326                    location,
327                )?;
328                for _ in symbol.subsymbols() {
329                    budget.charge(ExpansionResource::TermWork, 1, location)?;
330                }
331                compile::validate_scalar(symbol, location)?;
332                Ok(())
333            }
334        }
335    }
336}

338/// A switch for tests that compare a kept statement with its rewrite: while it
339/// is set on a thread, [`kept_as_is`] keeps nothing there. The statements kept
340/// on a thread are counted, so a test can assert which path it took.
341#[cfg(test)]
342pub(crate) mod testing {
343    use std::cell::Cell;
344
345    thread_local! {
346        static REWRITE_EVERY_STATEMENT: Cell<bool> = const { Cell::new(false) };
347        static KEPT: Cell<usize> = const { Cell::new(0) };
348    }
349
350    pub(crate) fn rewrites_every_statement() -> bool {
351        REWRITE_EVERY_STATEMENT.with(Cell::get)
352    }
353
354    pub(super) fn note_kept() {
355        KEPT.with(|kept| kept.set(kept.get() + 1));
356    }
357
358    /// The statements kept on this thread so far.
359    pub(crate) fn kept_statements() -> usize {
360        KEPT.with(Cell::get)
361    }
362
363    /// Run `body` with every statement rewritten on this thread.
364    pub(crate) fn with_every_statement_rewritten<T>(body: impl FnOnce() -> T) -> T {
365        let previous = REWRITE_EVERY_STATEMENT.with(|rewrite| rewrite.replace(true));
366        let result = body();
367        REWRITE_EVERY_STATEMENT.with(|rewrite| rewrite.set(previous));
368        result
369    }
370}

406/// Statement evidence retains logical identity independently of parsed spans.
407/// Constructed input contributes its caller-assigned site rather than a fabricated location.
408pub(crate) fn program_sites(
409    carrier: &WithProvenance<Statement>,
410    fallback: ProgramSite,
411) -> Vec<ProgramSite> {
412    let mut sites: Vec<_> = carrier
413        .provenance()
414        .origins()
415        .filter_map(|origin| match origin {
416            Origin::Parsed(location) => Some(fallback.with_location(*location)),
417            Origin::Constructed | Origin::Transformed(_) => None,
418        })
419        .collect();
420    if sites.is_empty() {
421        sites.push(fallback);
422    }
423    sites
424}

426/// Attach actual parsed evidence without changing a caller-assigned statement ID.
427pub(crate) fn origin(
428    carrier: &WithProvenance<Statement>,
429    fallback: impl Into<ProgramSite>,
430) -> ProgramSite {
431    let fallback = fallback.into();
432    carrier
433        .provenance()
434        .origins()
435        .find_map(|origin| match origin {
436            Origin::Parsed(location) => Some(fallback.with_location(*location)),
437            Origin::Constructed | Origin::Transformed(_) => None,
438        })
439        .unwrap_or(fallback)
440}

442pub(crate) fn normalize_node(
443    term: Term,
444    constants: &BTreeMap<String, Symbol>,
445    budget: &mut Budget,
446    location: ProgramSite,
447) -> Result<Term, ExpansionFailure> {
448    budget.charge(ExpansionResource::TermWork, 1, location)?;
449    match &term {
450        Term::Variable(_) | Term::Pool(_) | Term::Interval { .. } => Ok(term),
451        Term::Symbolic(symbol) if !symbol.arguments().is_empty() => {
452            budget.charge(
453                ExpansionResource::ScalarBytes,
454                symbol_bytes(symbol),
455                location,
456            )?;
457            let resolved =
458                symbol
459                    .clone()
460                    .try_fold(|parts| -> Result<Symbol, ExpansionFailure> {
461                        budget.charge(ExpansionResource::TermWork, 1, location)?;
462                        let value = Symbol::from(parts);
463                        if let Some(replacement) = defined(&value, constants) {
464                            budget.charge(
465                                ExpansionResource::ScalarBytes,
466                                symbol_bytes(replacement),
467                                location,
468                            )?;
469                            Ok(replacement.clone())
470                        } else {
471                            Ok(value)
472                        }
473                    })?;
474            compile::validate_scalar(&resolved, location)?;
475            Ok(Term::Symbolic(resolved))
476        }
477        Term::Symbolic(symbol) => {
478            if let Some(value) = defined(symbol, constants) {
479                budget.charge(
480                    ExpansionResource::ScalarBytes,
481                    symbol_bytes(value),
482                    location,
483                )?;
484                Ok(Term::Symbolic(value.clone()))
485            } else {
486                scalar_leaf(symbol, budget, location)?;
487                Ok(term)
488            }
489        }
490        Term::UnaryOperation {
491            operator: themelios_program::term::UnaryOp::Negate,
492            argument,
493        } if matches!(argument.as_ref(), Term::Symbolic(Symbol::Function { .. })) => {
494            let Term::Symbolic(Symbol::Function {
495                name,
496                arguments,
497                sign,
498            }) = argument.as_ref()
499            else {
500                unreachable!("checked function operand")
501            };
502            budget.charge(
503                ExpansionResource::ScalarBytes,
504                symbol_bytes(match argument.as_ref() {
505                    Term::Symbolic(symbol) => symbol,
506                    _ => unreachable!("checked symbol"),
507                }),
508                location,
509            )?;
510            Ok(Term::Symbolic(Symbol::Function {
511                name: name.clone(),
512                arguments: arguments.clone(),
513                sign: match sign {
514                    Sign::Positive => Sign::Negative,
515                    Sign::Negative => Sign::Positive,
516                },
517            }))
518        }
519        Term::Function { .. } | Term::Tuple(_) if term.is_ground() => {
520            for child in term.subterms() {
521                budget.charge(ExpansionResource::TermWork, 1, location)?;
522                if matches!(child, Term::Pool(_) | Term::Interval { .. }) {
523                    return Err(unsupported(ProfileFeature::Term, location).into());
524                }
525            }
526            let symbol = term
527                .evaluate()
528                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
529            compile::validate_scalar(&symbol, location)?;
530            Ok(Term::Symbolic(symbol))
531        }
532        Term::UnaryOperation { .. } | Term::BinaryOperation { .. } | Term::Absolute(_) => {
533            let symbol = term
534                .evaluate()
535                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
536            compile::validate_scalar(&symbol, location)?;
537            Ok(Term::Symbolic(symbol))
538        }
539        _ => Err(unsupported(ProfileFeature::Term, location).into()),
540    }
541}

543fn constant_name(symbol: &Symbol) -> Option<&str> {
544    match symbol {
545        Symbol::Function {
546            name,
547            arguments,
548            sign: Sign::Positive,
549        } if arguments.is_empty() => Some(name.as_str()),
550        _ => None,
551    }
