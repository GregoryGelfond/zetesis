26/// Admit a bounded extension of S0: unannotated acyclic scalar `#const`
27/// definitions; checked ground scalar arithmetic in ordinary S0 terms; and
28/// finite pools/intervals in ordinary fact heads. A bare undefined constant name
29/// remains a symbolic value. Definition names do not rename predicates.
30///
31/// Intervals have scalar integer endpoints; arithmetic over a set, variable
32/// arithmetic, and pools/intervals in non-fact rules remain unsupported. A
33/// descending interval denotes no facts. Fact products are bounded before
34/// allocation. The resulting core program uses the existing S0 semantics.
35/// Original source bytes and parsed rule locations are preserved; no source text
36/// is regenerated. Signed `#defined` signatures and signature/empty `#show`
37/// are retained as metadata; display selection never changes model identity.
38/// Includes require bundle admission and other directives remain unsupported.
39///
40/// # Errors
41/// Refuses every parser/raiser diagnostic, unsupported construct, ambiguous or
42/// cyclic definition, undefined/overflowing scalar operation, expansion limit,
43/// or independent core admission error. No successful partial program is
44/// returned. This partial profile refuses undefined arithmetic rather than
45/// emulating clingo's warning-and-drop behavior.
46pub fn admit_extended(
47    text: String,
48    options: AdmissionOptions,
49    limits: ExpansionLimits,
50) -> Result<Admitted, ExpansionFailure> {
51    ParsedSource::new(text, options)?
52        .admit_extended(limits)
53        .map_err(SourceFailure::into_error)
54}

73pub(crate) fn admit_parsed_recording(
74    source: ParsedSource,
75    limits: ExpansionLimits,
76    cancellation: Option<zetesis_cpu::Cancellation>,
77    recorder: Option<&mut crate::automatic::handoff::Recorder>,
78) -> Result<Admitted, SourceFailure<ExpansionFailure>> {
79    match compile_parsed(&source, limits, cancellation, recorder) {
80        Ok(compiled) => Ok(Admitted {
81            program: compiled.program,
82            source: source.into_source(),
83            template_origins: compiled.template_origins,
84            fact_origins: compiled.fact_origins,
85            metadata: compiled.metadata,
86            expansion: compiled.expansion,
87        }),
88        Err(error) => Err(SourceFailure::new(source, error)),
89    }

92fn compile_parsed(
93    source: &ParsedSource,
94    limits: ExpansionLimits,
95    cancellation: Option<zetesis_cpu::Cancellation>,
96    mut recorder: Option<&mut crate::automatic::handoff::Recorder>,
97) -> Result<Compilation, ExpansionFailure> {
98    let parsed = source.parsed();
99    let options = source.options();
100    let location = Location {
101        source: source.source().id(),
102        span: source.source().span(),
103    };
104    let budget =
105        Budget::new(limits, options.core_limits.max_templates).with_cancellation(cancellation);
106    budget.poll(location.into())?;
107    if let Some(recorder) = recorder.as_deref_mut() {
108        recorder.check_profile(parsed, options, false)?;
109    } else {
110        profile::check_extended(parsed, options)?;
111    }
112    check_definitions(parsed, limits)?;
113    metadata::check_count(parsed, limits, &mut 0)?;
114    let raised = crate::automatic::handoff::raise(parsed, recorder.as_deref_mut())?;
115    budget.poll(location.into())?;
116    let mut source_metadata = metadata::Builder::default();
117    metadata::collect(&raised, &mut source_metadata)?;
118    let result = compile_owned(
119        &raised,
120        options.core_limits,
121        budget,
122        location.into(),
123        source_metadata.finish(location.into())?,
124    );
125    if result.is_err()
126        && let Some(recorder) = recorder
127    {
128        recorder.retain_program(raised);
129    }
130    result
131}

133/// Compile the source into the program with its template and fact origins and the
134/// expansion charges it accepted, as one compilation with `metadata`.
135pub(crate) fn compile_owned(
136    source: &SourceProgram,
137    core_limits: AdmissionLimits,
138    mut budget: Budget,
139    location: ProgramSite,
140    metadata: SourceMetadata,
141) -> Result<Compilation, ExpansionFailure> {
142    let compiled = compile_relational(
143        source,
144        core_limits,
145        &mut budget,
146        location,
147        FormulaPurpose::Ordinary,
148        |carrier, _| parsed_origins(carrier),
149        |location| ProgramSite::from(*location),
150    )?;
151    budget.poll(location)?;
152    Ok(Compilation {
153        program: compiled.program,
154        template_origins: compiled.template_origins,
155        fact_origins: compiled.fact_origins,
156        metadata,
157        expansion: budget.usage(),
158    })
159}

161/// Share normalization, fact expansion and signed coherence while retaining the
162/// caller's evidence representation and one cumulative preparation budget.
163pub(crate) fn compile_relational<P: Clone + Ord>(
164    source: &SourceProgram,
165    core_limits: AdmissionLimits,
166    budget: &mut Budget,
167    fallback: ProgramSite,
168    purpose: FormulaPurpose,
169    evidence: impl Fn(&WithProvenance<Statement>, ProgramSite) -> Vec<P>,
170    site: impl Fn(&P) -> ProgramSite,
171) -> Result<compile::CompiledProgram<P>, ExpansionFailure> {
172    budget.poll(fallback)?;
173    let constants = resolve(source, budget, fallback)?;
174    let mut compiled = compile_extended(source, &constants, budget, fallback, purpose, evidence)?;
175    budget.poll(fallback)?;
176    crate::coherence::append_with(
177        &mut compiled,
178        core_limits,
179        |resource, count, evidence| {
180            budget.charge(resource, count, evidence.first().map_or(fallback, &site))
181        },
182        |error, evidence| {
183            ExpansionFailure::from(AdmissionFailure::Core {
184                error,
185                location: evidence.first().map_or(fallback, &site),
186            })
187        },
188    )?;
189    budget.poll(fallback)?;
190    let compiled = compiled.admit(core_limits, |error, origins| {
191        let location = origins.first().map_or(fallback, &site);
192        AdmissionFailure::Core { error, location }
193    })?;
194    budget.poll(fallback)?;
195    Ok(compiled)
196}

206pub(crate) fn check_definitions_in(
207    parsed: &Parse<ast::Program>,
208    limits: ExpansionLimits,
209    names: &mut BTreeMap<String, Location>,
210) -> Result<(), ExpansionFailure> {
211    for statement in parsed.tree().statements() {
212        let ast::Statement::Const(constant) = statement else {
213            continue;
214        };
215        let location = parsed.location(constant.syntax().text_range());
216        if constant.annotation().is_some() {
217            return Err(ExpansionFailure::ConstantPolicy {
218                location: location.into(),
219            });
220        }
221        let Some(name) = constant.name() else {
222            return Err(unsupported(ProfileFeature::Statement, location).into());
223        };
224        if let Some(first) = names.insert(name.text().to_owned(), location) {
225            return Err(ExpansionFailure::DuplicateConstant {
226                name: name.text().to_owned(),
227                first: first.into(),
228                duplicate: location.into(),
229            });
230        }
231        check(
232            ExpansionResource::Constants,
233            names.len() as u128,
234            limits.max_constants,
235            location.into(),
236        )?;
237    }
238    Ok(())
239}

247pub(crate) fn resolve(
248    source: &SourceProgram,
249    budget: &mut Budget,
250    fallback: ProgramSite,
251) -> Result<BTreeMap<String, Symbol>, ExpansionFailure> {
252    let mut definitions: BTreeMap<String, Definition<'_>> = BTreeMap::new();
253    for (index, carrier) in source.statements().enumerate() {
254        let location = origin(carrier, fallback.with_statement(StatementId::new(index)));
255        budget.poll(location)?;
256        let Statement::Const(constant) = carrier.get() else {
257            continue;
258        };
259        if constant.policy.is_some() {
260            return Err(ExpansionFailure::ConstantPolicy { location });
261        }
262        // Canonical declarations are identified by their content. Additional
263        // provenance records evidence, not another definition of the name.
264        if let Some(first) = definitions
265            .get(constant.name.as_str())
266            .map(|entry| entry.location)
267        {
268            return Err(ExpansionFailure::DuplicateConstant {
269                name: constant.name.as_str().to_owned(),
270                first,
271                duplicate: location,
272            });
273        }
274        budget.check_constants(definitions.len().saturating_add(1), location)?;
275        definitions.insert(
276            constant.name.as_str().to_owned(),
277            Definition {
278                constant,
279                location,
280                dependencies: Vec::new(),
281            },
282        );
283    }
284    let names: BTreeSet<_> = definitions.keys().cloned().collect();
285    for definition in definitions.values_mut() {
286        let mut dependencies = BTreeSet::new();
287        for term in definition.constant.value.subterms() {
288            budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
289            if let Term::Symbolic(symbol) = term {
290                for nested in symbol.subsymbols() {
291                    if !symbol.arguments().is_empty() {
292                        budget.charge(ExpansionResource::TermWork, 1, definition.location)?;
293                    }
294                    if let Some(name) = constant_name(nested).filter(|name| names.contains(*name)) {
295                        dependencies.insert(name.to_owned());
296                    }
297                }
298            }
299        }
300        definition.dependencies = dependencies.into_iter().collect();
301    }
302    let mut values = BTreeMap::new();
303    let mut active = BTreeSet::new();
304    for name in definitions.keys() {
305        if values.contains_key(name) {
306            continue;
307        }
308        active.insert(name.clone());
309        let mut stack = vec![(name.clone(), 0)];
310        while let Some((name, next)) = stack.last_mut() {
311            let definition = &definitions[name];
312            budget.poll(definition.location)?;
313            if let Some(dependency) = definition.dependencies.get(*next) {
314                *next += 1;
315                if values.contains_key(dependency) {
316                    continue;
317                }
318                if active.contains(dependency) {
319                    let mut names: Vec<_> = stack.iter().map(|(name, _)| name.clone()).collect();
320                    names.push(dependency.clone());
321                    return Err(ExpansionFailure::ConstantCycle {
322                        names,
323                        location: definition.location,
324                    });
325                }
326                active.insert(dependency.clone());
327                stack.push((dependency.clone(), 0));
328                continue;
329            }
330            let value = definition.constant.value.clone().try_fold(|parts| {
331                normalize_node(Term::from(parts), &values, budget, definition.location)
332            })?;
333            let TermParts::Symbolic(value) = value.into_parts() else {
334                return Err(ExpansionFailure::Evaluation {
335                    error: EvalError::Undefined,
336                    location: definition.location,
337                });
338            };
339            values.insert(name.clone(), value);
340            active.remove(name);
341            stack.pop();
342        }
343    }
344    Ok(values)
345}

347/// Normalize each statement and compile it. A statement whose every term
348/// normalization would return as it is, charged and validated by
349/// [`kept_as_is`], is compiled from the original carrier: the rewrite would
350/// hand back the same statement, stamped with a transformation tag that
351/// nothing after this point reads, at the cost of copying and rebuilding it.
352/// Every other statement is rebuilt by the rewrite, alone in a program of
353/// its own so that no two statements merge.
354fn compile_extended<P: Clone>(
355    source: &SourceProgram,
356    constants: &BTreeMap<String, Symbol>,
357    budget: &mut Budget,
358    fallback: ProgramSite,
359    purpose: FormulaPurpose,
360    evidence: impl Fn(&WithProvenance<Statement>, ProgramSite) -> Vec<P>,
361) -> Result<compile::Compiled<P>, ExpansionFailure> {
362    let mut compiled = compile::Compiled::default();
363    for (index, carrier) in source.statements().enumerate() {
364        let location = origin(carrier, fallback.with_statement(StatementId::new(index)));
365        budget.poll(location)?;
366        if !purpose.includes(carrier.get())
367            || matches!(
368                carrier.get(),
369                Statement::Const(_) | Statement::Defined(_) | Statement::Show(_)
370            )
371        {
372            continue;
373        }
374        let locations = evidence(carrier, location);
375        if kept_as_is(
376            carrier.get(),
377            constants,
378            Normalization::Extended,
379            budget,
380            location,
381        )? {
382            emit(carrier, locations, budget, location, &mut compiled)?;
383            continue;
384        }
385        let mut normalizer = Normalizer {
386            constants,
387            budget,
388            location,
389            failure: None,
390        };
391        let rewritten = rewrite(SourceProgram::of_nodes([carrier.clone()]), &mut normalizer);
392        if let Some(failure) = normalizer.failure {
393            return Err(failure);
394        }
395        budget.poll(location)?;
396        let normalized_carrier = rewritten
397            .statements()
398            .next()
399            .expect("a rule rewrite keeps its statement");
400        emit(
401            normalized_carrier,
402            locations,
403            budget,
404            location,
405            &mut compiled,
406        )?;
407    }
408    Ok(compiled)
409}

411/// Emit one normalized statement into the shared classified source owner.
412/// Checked products retain compact columns, with evidence for every occurrence.
413fn emit<P: Clone>(
414    statement: &WithProvenance<Statement>,
415    locations: Vec<P>,
416    budget: &mut Budget,
417    location: ProgramSite,
418    compiled: &mut compile::Compiled<P>,
419) -> Result<(), ExpansionFailure> {
420    if let Some(facts) = fact_expansion::facts(statement, budget, location)? {
421        budget.charge(
422            ExpansionResource::Origins,
423            (facts.len() as u128).saturating_mul(locations.len() as u128),
424            location,
425        )?;
426        for product in facts.products() {
427            let mut columns = Vec::new();
428            columns
429                .try_reserve_exact(product.predicate().arity())
430                .map_err(|error| ExpansionFailure::Allocation { error, location })?;
431            for source in product.columns() {
432                let mut column = Vec::new();
433                for cell in source {
434                    budget.poll(location)?;
435                    column
436                        .try_reserve(1)
437                        .map_err(|error| ExpansionFailure::Allocation { error, location })?;
438                    column.push(match cell {
439                        fact_expansion::Cell::Scalar(value) => {
440                            zetesis_core::FactSegment::Value(value.clone())
441                        }
442                        fact_expansion::Cell::Interval(range) => {
443                            zetesis_core::FactSegment::Interval {
444                                first: *range.start(),
445                                last: *range.end(),
446                            }
447                        }
448                    });
449                }
450                columns.push(column);
451            }
452            compiled.product(
453                product.predicate().clone(),
454                columns,
455                &locations,
456                || budget.poll(location),
457                |error| ExpansionFailure::from(AdmissionFailure::Core { error, location }),
458            )?;
459        }
460    } else {
461        budget.charge(ExpansionResource::Templates, 1, location)?;
462        budget.charge(
463            ExpansionResource::Origins,
464            locations.len() as u128,
465            location,
466        )?;
467        compiled
468            .push(compile::statement(statement, location)?, locations)
469            .map_err(|error| AdmissionFailure::Core { error, location })?;
470    }
471    Ok(())
472}

483/// Whether `normalization` would return `statement` exactly as it is. Its
484/// terms are charged and validated as the rewrite would charge and validate
485/// them, in the rewrite's order, on a tentative budget that becomes the
486/// budget only when every term is a [`Leaf`]; otherwise the budget is left
487/// as it was, and the rewrite charges the statement from the start. A
488/// failure is the one the rewrite would raise first, since it arises before
489/// any term the rewrite would rebuild. Nothing is cloned but the budget.
490pub(crate) fn kept_as_is(
491    statement: &Statement,
492    constants: &BTreeMap<String, Symbol>,
493    normalization: Normalization,
494    budget: &mut Budget,
495    location: ProgramSite,
496) -> Result<bool, ExpansionFailure> {
497    let Statement::Rule(rule) = statement else {
498        return Ok(false);
499    };
500    #[cfg(test)]
501    if testing::rewrites_every_statement() {
502        return Ok(false);
503    }
504    let mut tentative = budget.clone();
505    let kept = every_term(rule, &mut |term| match Leaf::of(term, constants) {
506        Some(leaf) => leaf
507            .account(normalization, &mut tentative, location)
508            .map(|()| true),
509        None => Ok(false),
510    })?;
511    if kept {
512        *budget = tentative;
513        #[cfg(test)]
514        testing::note_kept();
515    }
516    Ok(kept)
517}

519/// Apply `keep` to the terms of a rule of the shape this profile compiles,
520/// in the order the rewrite rebuilds them: the head, then each body element;
521/// an atom's arguments in order, a comparison's first term and then its
522/// steps. `Ok(false)` at the first term `keep` refuses, or at once for a
523/// head or body element of another shape, which the rewrite descends in an
524/// order this walk does not reproduce.
525fn every_term(
526    rule: &Rule,
527    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
528) -> Result<bool, ExpansionFailure> {
529    // A head or body element of any other shape, including one the pinned
530    // themelios tier may add, is left to the rewrite.
531    let head = match rule.head().get() {
532        Head::Literal(literal) => literal_terms(literal, keep)?,
533        Head::Falsum | Head::Verum => true,
534        _ => false,
535    };
536    if !head {
537        return Ok(false);
538    }
539    for element in rule.body().get().elements() {
540        let kept = match element.get() {
541            BodyElement::Literal(literal) => literal_terms(literal, keep)?,
542            _ => false,
543        };
544        if !kept {
545            return Ok(false);
546        }
547    }
548    Ok(true)
549}
550
551fn literal_terms(
552    literal: &Literal,
553    keep: &mut impl FnMut(&Term) -> Result<bool, ExpansionFailure>,
554) -> Result<bool, ExpansionFailure> {
555    match &literal.inner {
556        LiteralInner::Atom(atom) => {
557            for term in atom.get().argument_terms() {
558                if !keep(term)? {
559                    return Ok(false);
560                }
561            }
562            Ok(true)
563        }
564        LiteralInner::Comparison(comparison) => {
565            let comparison = comparison.get();
566            if !keep(comparison.first())? {
567                return Ok(false);
568            }
569            for (_, term) in comparison.steps() {
570                if !keep(term)? {
571                    return Ok(false);
572                }
573            }
574            Ok(true)
575        }
576        LiteralInner::True | LiteralInner::False => Ok(true),
577    }
578}

580/// A term [`normalize_node`] returns exactly as it is, charged and validated
581/// by [`Leaf::account`]: a variable, an argument-free scalar naming no
582/// definition, or a ground symbol with arguments none of whose parts names a
583/// definition. Raise yields such a symbol in its canonical form, and the
584/// normalizer's symbolic branch folds it back to itself. Every other term is
585/// rebuilt, or, for a pool or interval former, kept around terms that may
586/// not be.
587#[derive(Clone, Copy)]
588enum Leaf<'a> {
589    Variable,
590    Scalar(&'a Symbol),
591    Compound(&'a Symbol),
592}
593
594impl<'a> Leaf<'a> {
595    fn of(term: &'a Term, constants: &BTreeMap<String, Symbol>) -> Option<Self> {
596        match term {
597            Term::Variable(_) => Some(Self::Variable),
598            Term::Symbolic(symbol)
599                if symbol.arguments().is_empty() && defined(symbol, constants).is_none() =>
600            {
601                Some(Self::Scalar(symbol))
602            }
603            Term::Symbolic(symbol)
604                if !symbol.arguments().is_empty()
605                    && symbol
606                        .subsymbols()
607                        .all(|part| defined(part, constants).is_none()) =>
608            {
609                Some(Self::Compound(symbol))
610            }
611            _ => None,
612        }
613    }
614
615    /// Charge and validate as `normalization` does for this term: its unit of
616    /// term work first; then a scalar's validation and payload, except a
617    /// sentinel the formula route keeps; or a compound symbol's payload, one
618    /// unit for each of its parts as the fold visits them, and its validation.
619    fn account(
620        self,
621        normalization: Normalization,
622        budget: &mut Budget,
623        location: ProgramSite,
624    ) -> Result<(), ExpansionFailure> {
625        budget.charge(ExpansionResource::TermWork, 1, location)?;
626        match self {
627            Self::Variable => Ok(()),
628            Self::Scalar(Symbol::Infimum | Symbol::Supremum)
629                if normalization == Normalization::Formula =>
630            {
631                Ok(())
632            }
633            Self::Scalar(symbol) => scalar_leaf(symbol, budget, location),
634            Self::Compound(symbol) => {
635                budget.charge(
636                    ExpansionResource::ScalarBytes,
637                    symbol_bytes(symbol),
638                    location,
639                )?;
640                for _ in symbol.subsymbols() {
641                    budget.charge(ExpansionResource::TermWork, 1, location)?;
642                }
643                compile::validate_scalar(symbol, location)?;
644                Ok(())
645            }
646        }
647    }
648}

650/// A switch for tests that compare a kept statement with its rewrite: while it
651/// is set on a thread, [`kept_as_is`] keeps nothing there. The statements kept
652/// on a thread are counted, so a test can assert which path it took.
653#[cfg(test)]
654pub(crate) mod testing {
655    use std::cell::Cell;
656
657    thread_local! {
658        static REWRITE_EVERY_STATEMENT: Cell<bool> = const { Cell::new(false) };
659        static KEPT: Cell<usize> = const { Cell::new(0) };
660    }
661
662    pub(crate) fn rewrites_every_statement() -> bool {
663        REWRITE_EVERY_STATEMENT.with(Cell::get)
664    }
665
666    pub(super) fn note_kept() {
667        KEPT.with(|kept| kept.set(kept.get() + 1));
668    }
669
670    /// The statements kept on this thread so far.
671    pub(crate) fn kept_statements() -> usize {
672        KEPT.with(Cell::get)
673    }
674
675    /// Run `body` with every statement rewritten on this thread.
676    pub(crate) fn with_every_statement_rewritten<T>(body: impl FnOnce() -> T) -> T {
677        let previous = REWRITE_EVERY_STATEMENT.with(|rewrite| rewrite.replace(true));
678        let result = body();
679        REWRITE_EVERY_STATEMENT.with(|rewrite| rewrite.set(previous));
680        result
681    }
682}

687/// The value an argument-free scalar names, when it names a definition.
688fn defined<'a>(symbol: &Symbol, constants: &'a BTreeMap<String, Symbol>) -> Option<&'a Symbol> {
689    constant_name(symbol).and_then(|name| constants.get(name))
690}

692/// Validate an argument-free scalar normalization keeps and charge its payload.
693fn scalar_leaf(
694    symbol: &Symbol,
695    budget: &mut Budget,
696    location: ProgramSite,
697) -> Result<(), ExpansionFailure> {
698    compile::validate_scalar(symbol, location)?;
699    budget.charge(
700        ExpansionResource::ScalarBytes,
701        symbol_bytes(symbol),
702        location,
703    )?;
704    Ok(())
705}

707pub(crate) fn parsed_origins(carrier: &WithProvenance<Statement>) -> Vec<Location> {
708    carrier
709        .provenance()
710        .origins()
711        .filter_map(|origin| match origin {
712            Origin::Parsed(location) => Some(*location),
713            Origin::Constructed | Origin::Transformed(_) => None,
714        })
715        .collect()
716}

718/// Statement evidence retains logical identity independently of parsed spans.
719/// Constructed input contributes its caller-assigned site rather than a fabricated location.
720pub(crate) fn program_sites(
721    carrier: &WithProvenance<Statement>,
722    fallback: ProgramSite,
723) -> Vec<ProgramSite> {
724    let mut sites: Vec<_> = carrier
725        .provenance()
726        .origins()
727        .filter_map(|origin| match origin {
728            Origin::Parsed(location) => Some(fallback.with_location(*location)),
729            Origin::Constructed | Origin::Transformed(_) => None,
730        })
731        .collect();
732    if sites.is_empty() {
733        sites.push(fallback);
734    }
735    sites
736}
737
738/// Attach actual parsed evidence without changing a caller-assigned statement ID.
739pub(crate) fn origin(
740    carrier: &WithProvenance<Statement>,
741    fallback: impl Into<ProgramSite>,
742) -> ProgramSite {
743    let fallback = fallback.into();
744    carrier
745        .provenance()
746        .origins()
747        .find_map(|origin| match origin {
748            Origin::Parsed(location) => Some(fallback.with_location(*location)),
749            Origin::Constructed | Origin::Transformed(_) => None,
750        })
751        .unwrap_or(fallback)
752}
753
754struct Normalizer<'a> {
755    constants: &'a BTreeMap<String, Symbol>,
756    budget: &'a mut Budget,
757    location: ProgramSite,
758    failure: Option<ExpansionFailure>,
759}
760
761impl Rewrite for Normalizer<'_> {
762    fn tag(&self) -> TransformTag {
763        TransformTag::new("zetesis-scalar-expansion")
764    }
765    fn rewrite_term(&mut self, term: Term) -> Term {
766        if self.failure.is_some() {
767            return term;
768        }
769        match normalize_node(term, self.constants, self.budget, self.location) {
770            Ok(term) => term,
771            Err(error) => {
772                self.failure = Some(error);
773                // Only a failed private intermediate uses this placeholder;
774                // the entire rewrite is refused before compilation.
775                Term::Symbolic(Symbol::Number(0))
776            }
777        }
778    }
779}
780
781pub(crate) fn normalize_node(
782    term: Term,
783    constants: &BTreeMap<String, Symbol>,
784    budget: &mut Budget,
785    location: ProgramSite,
786) -> Result<Term, ExpansionFailure> {
787    budget.charge(ExpansionResource::TermWork, 1, location)?;
788    match &term {
789        Term::Variable(_) | Term::Pool(_) | Term::Interval { .. } => Ok(term),
790        Term::Symbolic(symbol) if !symbol.arguments().is_empty() => {
791            budget.charge(
792                ExpansionResource::ScalarBytes,
793                symbol_bytes(symbol),
794                location,
795            )?;
796            let resolved =
797                symbol
798                    .clone()
799                    .try_fold(|parts| -> Result<Symbol, ExpansionFailure> {
800                        budget.charge(ExpansionResource::TermWork, 1, location)?;
801                        let value = Symbol::from(parts);
802                        if let Some(replacement) = defined(&value, constants) {
803                            budget.charge(
804                                ExpansionResource::ScalarBytes,
805                                symbol_bytes(replacement),
806                                location,
807                            )?;
808                            Ok(replacement.clone())
809                        } else {
810                            Ok(value)
811                        }
812                    })?;
813            compile::validate_scalar(&resolved, location)?;
814            Ok(Term::Symbolic(resolved))
815        }
816        Term::Symbolic(symbol) => {
817            if let Some(value) = defined(symbol, constants) {
818                budget.charge(
819                    ExpansionResource::ScalarBytes,
820                    symbol_bytes(value),
821                    location,
822                )?;
823                Ok(Term::Symbolic(value.clone()))
824            } else {
825                scalar_leaf(symbol, budget, location)?;
826                Ok(term)
827            }
828        }
829        Term::UnaryOperation {
830            operator: themelios_program::term::UnaryOp::Negate,
831            argument,
832        } if matches!(argument.as_ref(), Term::Symbolic(Symbol::Function { .. })) => {
833            let Term::Symbolic(Symbol::Function {
834                name,
835                arguments,
836                sign,
837            }) = argument.as_ref()
838            else {
839                unreachable!("checked function operand")
840            };
841            budget.charge(
842                ExpansionResource::ScalarBytes,
843                symbol_bytes(match argument.as_ref() {
844                    Term::Symbolic(symbol) => symbol,
845                    _ => unreachable!("checked symbol"),
846                }),
847                location,
848            )?;
849            Ok(Term::Symbolic(Symbol::Function {
850                name: name.clone(),
851                arguments: arguments.clone(),
852                sign: match sign {
853                    Sign::Positive => Sign::Negative,
854                    Sign::Negative => Sign::Positive,
855                },
856            }))
857        }
858        Term::Function { .. } | Term::Tuple(_) if term.is_ground() => {
859            for child in term.subterms() {
860                budget.charge(ExpansionResource::TermWork, 1, location)?;
861                if matches!(child, Term::Pool(_) | Term::Interval { .. }) {
862                    return Err(unsupported(ProfileFeature::Term, location).into());
863                }
864            }
865            let symbol = term
866                .evaluate()
867                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
868            compile::validate_scalar(&symbol, location)?;
869            Ok(Term::Symbolic(symbol))
870        }
871        Term::UnaryOperation { .. } | Term::BinaryOperation { .. } | Term::Absolute(_) => {
872            let symbol = term
873                .evaluate()
874                .map_err(|error| ExpansionFailure::Evaluation { error, location })?;
875            compile::validate_scalar(&symbol, location)?;
876            Ok(Term::Symbolic(symbol))
877        }
878        _ => Err(unsupported(ProfileFeature::Term, location).into()),
879    }
880}

882fn constant_name(symbol: &Symbol) -> Option<&str> {
883    match symbol {
884        Symbol::Function {
885            name,
886            arguments,
887            sign: Sign::Positive,
888        } if arguments.is_empty() => Some(name.as_str()),
889        _ => None,
890    }
891}
