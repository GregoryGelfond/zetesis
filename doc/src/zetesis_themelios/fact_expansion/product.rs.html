15pub(super) struct Product {

21impl Product {

23    pub(super) fn retained_bytes(&self) -> usize {
24        self.predicate.name().len()
25            + self.columns.capacity() * size_of::<Column>()
26            + self
27                .columns
28                .iter()
29                .map(Column::retained_bytes)
30                .sum::<usize>()
31    }

33    pub(super) fn prepare(
34        atom: &Atom,
35        arguments: &[SourceTerm],
36        logical_bytes: &mut u128,
37        budget: &mut Budget,
38        location: ProgramSite,
39    ) -> Result<Option<Self>, ExpansionFailure> {
40        let mut sizes = reserved(arguments.len(), budget, location)?;
41        for term in arguments {
42            sizes.push(columns::size(term, budget, location)?);
43        }
44        // Check every argument, including siblings of an empty interval.
45        let count = sizes
46            .iter()
47            .fold(1_u128, |count, (size, _)| count.saturating_mul(*size));
48        budget.charge(ExpansionResource::Templates, count, location)?;
49        let intermediate = sizes
50            .iter()
51            .fold(0_u128, |sum, (size, _)| sum.saturating_add(*size));
52        budget.charge(
53            ExpansionResource::Values,
54            intermediate.saturating_add(count.saturating_mul(arguments.len() as u128)),
55            location,
56        )?;
57        if count == 0 {
58            return Ok(None);
59        }
60        budget.charge(
61            ExpansionResource::ScalarBytes,
62            (atom.name.as_str().len() as u128).saturating_mul(count),
63            location,
64        )?;
65        logical_family(
66            atom,
67            &sizes,
68            count,
69            intermediate,
70            logical_bytes,
71            budget,
72            location,
73        )?;
74        let predicate = Predicate::with_sign(
75            atom.name.as_str(),
76            arguments.len(),
77            crate::coherence::core_sign(atom.sign),
78        )
79        .map_err(|error| AdmissionFailure::Construction { error, location })?;
80        let mut columns = reserved(arguments.len(), budget, location)?;
81        let mut copied_bytes = 0_u128;
82        for (term, (width, _)) in arguments.iter().zip(sizes) {
83            let (column, payload) = Column::prepare(term, budget, location)?;
84            copied_bytes = copied_bytes.saturating_add((count / width).saturating_mul(payload));
85            columns.push(column);
86        }
87        let mut product = Self {
88            predicate,
89            columns,
90            rows: usize::try_from(count).expect("Templates charge has a usize ceiling"),
91        };
92        product.charge_copies(copied_bytes, budget, location)?;
93        Ok(Some(product))
94    }
95

108    /// Every scalar occurrence is selected count / column.width times. When
109    /// that complete traffic fits, no product traversal is needed to admit it.
110    /// The refusal path uses the same positions as emission to retain the old
111    /// accepted prefix and first offending per-argument charge.
112    fn charge_copies(
113        &mut self,
114        bytes: u128,
115        budget: &mut Budget,
116        location: ProgramSite,
117    ) -> Result<(), ExpansionFailure> {
118        if bytes <= budget.remaining_scalar_bytes() {
119            return budget.charge(ExpansionResource::ScalarBytes, bytes, location);
120        }
121        for _ in 0..self.rows {
122            budget.poll(location)?;
123            for column in &self.columns {
124                budget.charge(
125                    ExpansionResource::ScalarBytes,
126                    column.value_bytes(),
127                    location,
128                )?;
129            }
130            self.advance(budget, location)?;
131        }
132        unreachable!("complete copy traffic exceeded the remaining allowance")
133    }
134
135    fn advance(&mut self, budget: &Budget, location: ProgramSite) -> Result<(), ExpansionFailure> {
136        for column in self.columns.iter_mut().rev() {
137            budget.poll(location)?;
138            if column.advance() {
139                break;
140            }
141        }
142        Ok(())
143    }
144}

146/// Retain the existing logical family envelope even though it is no longer
147/// materialized. It is independent of actual Vec capacity and process RSS.
148fn logical_family(
149    atom: &Atom,
150    sizes: &[(u128, u128)],
151    count: u128,
152    intermediate: u128,
153    retained: &mut u128,
154    budget: &Budget,
155    location: ProgramSite,
156) -> Result<(), ExpansionFailure> {
157    let payload = sizes
158        .iter()
159        .fold(0_u128, |sum, (_, bytes)| sum.saturating_add(*bytes));
160    let output = count.saturating_mul(
161        size_of::<Template>() as u128
162            + sizes.len() as u128 * size_of::<Term>() as u128
163            + atom.name.as_str().len() as u128
164            + payload,
165    );
166    *retained = retained.saturating_add(output);
167    let columns = intermediate
168        .saturating_mul(size_of::<Value>() as u128)
169        .saturating_add(payload)
170        .saturating_add(
171            sizes.len() as u128 * (size_of::<Vec<Value>>() + size_of::<usize>()) as u128,
172        );
173    budget.check_family(retained.saturating_add(columns), location)
174}
