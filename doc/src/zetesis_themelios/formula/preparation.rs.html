14pub(crate) struct Preparation {
15    pub(crate) catalog: crate::formula_support::SupportCatalog,
16    pub(crate) accounting: crate::formula_support::Accounting,
17    pub(crate) program: formula_ir::Prepared,
18    pub(crate) budget: Budget,
19    pub(crate) limits: FormulaLimits,
20    pub(crate) options: crate::grounding_options::Execution,
21    pub(crate) location: ProgramSite,
22}

24impl Preparation {
25    pub(super) fn new(
26        program: formula_ir::Prepared,
27        catalog: crate::formula_support::SupportCatalog,
28        accounting: crate::formula_support::Accounting,
29        budget: Budget,
30        limits: &FormulaLimits,
31        location: ProgramSite,
32    ) -> Self {
33        Self {
34            program,
35            catalog,
36            accounting,
37            budget,
38            // The deferred receipt owns one immutable snapshot; its later
39            // compilation and grounding helpers borrow this configuration.
40            limits: *limits,
41            options: crate::grounding_options::Execution::default(),
42            location,
43        }
44    }
45
46    fn ground_hybrid(
47        self,
48        observer: Option<&dyn GroundingObserver>,
49    ) -> Result<(Compiled, Option<crate::formula_hybrid::Constraints>), FormulaFailure> {
50        grounding_observer::observe(observer, || formula_ground::ground_hybrid(self, observer))
51    }
52
53    fn ground(
54        self,
55        observer: Option<&dyn GroundingObserver>,
56        count_plan: Option<crate::formula_count_plan::Request<'_>>,
57    ) -> Result<Compiled, FormulaFailure> {
58        grounding_observer::observe(observer, || {
59            formula_ground::ground(self, observer, count_plan)
60        })
61    }
62}

64/// A checked canonical input that has not completed possible support or
65/// materialized a formula theory. Consuming it resumes the original budgets;
66/// callers cannot replace those budgets at the materialization boundary.
67pub struct PreparedFormula {
68    preparation: Preparation,
69    source: Owner,
70    metadata: SourceMetadata,
71}

142    /// Attempt finite-domain guards during final instantiation of an exact
143    /// normalized positive flat program. Disabled by default. Inapplicable,
144    /// Unknown or Stopped analysis leaves the complete join unchanged.
145    ///
146    /// The analysis borrows this exact normalized source and keeps its logical
147    /// population limits separate from support/guard capacity. Its work and all
148    /// bridge/guard work consume the original cumulative formula budget.
149    /// Analysis uses bounded infallible collections and has no internal control
150    /// hook. A configured preparation token is polled around that bounded call;
151    /// it does not provide a hard real-time deadline or allocator/RSS bound.
152    /// Observers receive the typed analysis outcome.
153    #[must_use]
154    pub const fn with_domain_analysis(mut self, limits: Option<crate::DomainLimits>) -> Self {
155        self.preparation.options.domains = limits;
156        self

159    /// Materialize producers and ineligible constraints while retaining ordinary
160    /// atom/scalar integrity constraints for repeated bounded satisfaction checks.
161    /// Complete possible support and original arithmetic admission still run.
162    /// This explicit schedule requires indexed joins. Objectives are retained;
163    /// solving must check the original constraints before scoring a core answer.
164    /// No solver runs; core answer sets still require the retained constraints.
165    ///
166    /// # Errors
167    /// Returns a typed capability, arithmetic, allocation or resource refusal,
168    /// identifying its original statement when applicable.
169    pub fn ground_hybrid(self) -> Result<crate::HybridFormula, FormulaFailure> {
170        self.ground_hybrid_with_observer(None)
171    }

173    /// Hybrid materialization with the same source phase observer as eager
174    /// grounding. Counts describe actual retained core nodes and admission work.
175    ///
176    /// # Errors
177    /// Returns the same failures as [`Self::ground_hybrid`].
178    pub fn ground_hybrid_with_observer(
179        self,
180        observer: Option<&dyn GroundingObserver>,
181    ) -> Result<crate::HybridFormula, FormulaFailure> {
182        let (compiled, constraints) = self
183            .preparation
184            .ground_hybrid(observer)
185            .map_err(|error| self.source.retain_failure(error))?;
186        Ok(crate::HybridFormula::new(
187            compiled,
188            constraints,
189            self.source,
190            self.metadata,
191        ))
192    }

194    /// Materialize the complete formula theory using the retained preparation.
195    /// This computes no answer sets and invokes no solver.
196    ///
197    /// # Errors
198    /// Returns a typed grounding, arithmetic or resource refusal. A successful
199    /// preparation does not guarantee successful materialization.
200    pub fn ground(self) -> Result<AdmittedFormula, FormulaFailure> {
201        self.ground_with_observer(None)
202    }

204    /// Materialize a certified base and defer complete terminal positive
205    /// definition groups when their source/IR correspondence is established and
206    /// at least one producer has a variable. A group shares a predicate name,
207    /// arity and sign; its ground producers defer with its variable producers.
208    /// Ground-only groups remain in the base, including rules with nonempty
209    /// bodies. If no group is selected, materialize the original theory. This
210    /// physical policy does not narrow the mathematical terminal classification.
211    /// Both routes retain preparation charges and compute no answer sets.
212    ///
213    /// # Errors
214    /// Returns source, allocation or resource refusals; an applicable terminal
215    /// plan is never silently replaced after its materialization has failed.
216    pub fn ground_adaptive(
217        self,
218    ) -> Result<crate::FormulaMaterialization<AdmittedFormula>, FormulaFailure> {
219        self.ground_adaptive_with_observer(None)
220    }

222    /// Adaptive materialization with caller-owned grounding observations.
223    ///
224    /// # Errors
225    /// Returns the same failures as [`Self::ground_adaptive`].
226    pub fn ground_adaptive_with_observer(
227        self,
228        observer: Option<&dyn GroundingObserver>,
229    ) -> Result<crate::FormulaMaterialization<AdmittedFormula>, FormulaFailure> {
230        use crate::formula_terminal::Materialized;
231        Ok(
232            match crate::formula_terminal::materialize(self.preparation, observer)
233                .map_err(|error| self.source.retain_failure(error))?
234            {
235                Materialized {
236                    compiled,
237                    core: _,
238                    terminal: None,
239                } => crate::FormulaMaterialization::Complete(AdmittedFormula {
240                    compiled,
241                    source: self.source,
242                    metadata: self.metadata,
243                }),
244                Materialized {
245                    compiled,
246                    core,
247                    terminal: Some(extension),
248                } => crate::FormulaMaterialization::Terminal(crate::TerminalFormula::new(
249                    compiled,
250                    core,
251                    extension,
252                    self.source,
253                    self.metadata,
254                )),
255            },
256        )
257    }

259    /// Lazy materialization: the terminal partition defers certified terminal
260    /// definitions, as [`Self::ground_adaptive`] does, but the base (or, with
261    /// nothing deferred, the program) is grounded under the hybrid schedule:
262    /// its producer core is instantiated and its eligible integrity constraints
263    /// are streamed. A terminal owner's base is then [`crate::BaseKind::Hybrid`],
264    /// even with no eligible constraint. Objective declarations disable terminal
265    /// deferral and produce [`crate::FormulaMaterialization::Complete`] with a
266    /// [`crate::HybridFormula`], retaining every producer. Score its answers only
267    /// after complete streamed constraint acceptance.
268    ///
269    /// # Errors
270    /// Returns the failures of [`Self::ground_adaptive`] and
271    /// [`Self::ground_hybrid`].
272    pub fn ground_lazy(
273        self,
274    ) -> Result<crate::FormulaMaterialization<crate::HybridFormula>, FormulaFailure> {
275        self.ground_lazy_with_observer(None)
276    }

278    /// Lazy materialization with caller-owned grounding observations.

280    /// # Errors

282    pub fn ground_lazy_with_observer(
283        self,
284        observer: Option<&dyn GroundingObserver>,
285    ) -> Result<crate::FormulaMaterialization<crate::HybridFormula>, FormulaFailure> {
286        let materialized = crate::formula_terminal::materialize_lazy(self.preparation, observer)
287            .map_err(|error| self.source.retain_failure(error))?;

291    /// Materialize the same original theory while independently attempting
292    /// bounded source-count partition consequences. Planning is opt-in; its
293    /// status is retained by [`AdmittedFormula::count_plan`]. Its control applies
294    /// only to planning, not to existing source materialization. No solver runs.
295    /// Capture reuses complete source joins and canonical guard evaluations;
296    /// [`crate::CountPlan`] describes the bounded applicability and cost model.
297    /// The outer observer boundary includes optional planning; source phase work
298    /// counters remain separate from [`crate::CountPlanStatistics`].
299    ///
300    /// # Errors
301    /// Returns existing source-grounding failures. Optional planning failures
302    /// retain the successfully admitted original and an incomplete plan status.
303    pub fn ground_with_count_plan(
304        self,
305        limits: crate::CountPlanLimits,
306        cancellation: &zetesis_cpu::Cancellation,
307        observer: Option<&dyn GroundingObserver>,
308    ) -> Result<AdmittedFormula, FormulaFailure> {
309        let request = crate::formula_count_plan::Request {
310            limits,
311            cancellation,
312        };
313        let compiled = self
314            .preparation
315            .ground(observer, Some(request))
316            .map_err(|error| self.source.retain_failure(error))?;
317        Ok(AdmittedFormula {
318            compiled,
319            source: self.source,
320            metadata: self.metadata,
321        })
322    }
323
324    /// Like [`Self::ground`], observing only the actual eager-grounding interval.

326    /// The caller retains its observer on failure.
327    ///
328    /// # Errors
329    /// Returns the same failures as [`Self::ground`].
330    pub fn ground_with_observer(
331        self,
332        observer: Option<&dyn GroundingObserver>,
333    ) -> Result<AdmittedFormula, FormulaFailure> {
334        let compiled = self
335            .preparation
336            .ground(observer, None)
337            .map_err(|error| self.source.retain_failure(error))?;
338        Ok(AdmittedFormula {
339            compiled,
340            source: self.source,
341            metadata: self.metadata,
342        })
343    }

598/// The owner a lazy materialization admits: a hybrid owner when nothing is
599/// deferred, else a terminal owner with a hybrid base.
600fn lazy_outcome(
601    materialized: crate::formula_terminal::Materialized,
602    source: crate::formula_owner::Owner,
603    metadata: crate::SourceMetadata,
604) -> crate::FormulaMaterialization<crate::HybridFormula> {
605    use crate::formula_terminal::{Core, Materialized};
606    match materialized {
607        Materialized {
608            compiled,
609            core,
610            terminal: Some(extension),
611        } => crate::FormulaMaterialization::Terminal(crate::TerminalFormula::new(
612            compiled, core, extension, source, metadata,
613        )),
614        Materialized {
615            compiled,
616            core: Core::Hybrid(constraints),
617            terminal: None,
618        } => crate::FormulaMaterialization::Complete(crate::HybridFormula::new(
619            compiled,
620            constraints.map(|constraints| *constraints),
621            source,
622            metadata,
623        )),
624        Materialized {
625            core: Core::Eager,
626            terminal: None,
627            ..
628        } => unreachable!("lazy materialization grounds an undeferred program as hybrid"),
629    }
