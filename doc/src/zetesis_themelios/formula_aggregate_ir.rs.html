20impl Compiler<'_> {
21    pub(super) fn condition(
22        &mut self,
23        condition: &Condition,
24        variables: &mut Variables,
25    ) -> Result<Vec<LiteralIr>, FormulaFailure> {
26        let mut result = Vec::new();
27        for (index, literal) in condition.literals().enumerate() {
28            if index >= self.options.max_body_elements {
29                return Err(AdmissionFailure::Limit {
30                    resource: InputLimit::BodyElements,
31                    limit: self.options.max_body_elements,
32                    observed: index + 1,
33                    location: self.location,
34                }
35                .into());
36            }
37            self.literal_into(literal.get(), variables, &mut result)?;
38        }
39        Ok(result)
40    }
41    pub(super) fn comparison(
42        &mut self,
43        left: &Term,
44        relation: Relation,
45        right: &Term,
46        variables: &mut Variables,
47    ) -> Result<LiteralIr, FormulaFailure> {
48        // Closed tuples are ordinary complete values. Tuple destructuring is
49        // retained only when a syntactic tuple still contains local variables.
50        if matches!(left, Term::Tuple(_)) || matches!(right, Term::Tuple(_)) {
51            if !matches!(relation, Relation::Eq | Relation::Neq) {
52                return Err(unsupported(ProfileFeature::Term, self.location).into());
53            }
54            Ok(LiteralIr::TupleCompare(
55                self.tuple_expressions(left, variables)?,
56                relation,
57                self.tuple_expressions(right, variables)?,
58            ))
59        } else {
60            Ok(LiteralIr::Compare(
61                self.expression(left, variables)?,
62                relation,
63                self.expression(right, variables)?,
64            ))
65        }
66    }
67    fn tuple_expressions(
68        &mut self,
69        term: &Term,
70        variables: &mut Variables,
71    ) -> Result<Vec<Expression>, FormulaFailure> {
72        match term {
73            Term::Tuple(terms) => terms
74                .iter()
75                .map(|term| self.expression(term, variables))
76                .collect(),
77            Term::Symbolic(Symbol::Tuple(values)) => values
78                .iter()
79                .map(|value| self.expression(&Term::Symbolic(value.clone()), variables))
80                .collect(),
81            _ => Err(unsupported(ProfileFeature::Term, self.location).into()),
82        }
83    }
84    pub(super) fn choice_guards(
85        &mut self,
86        choice: &Choice,
87        variables: &mut Variables,
88        values: &mut Vec<LiteralIr>,
89    ) -> Result<Vec<AggregateGuard>, FormulaFailure> {
90        self.guards(
91            choice.left_guard().map(WithProvenance::get),
92            choice.right_guard().map(WithProvenance::get),
93            variables,
94            values,
95        )
96    }
97    pub(super) fn guards(
98        &mut self,
99        left: Option<&Guard>,
100        right: Option<&Guard>,
101        variables: &mut Variables,
102        values: &mut Vec<LiteralIr>,
103    ) -> Result<Vec<AggregateGuard>, FormulaFailure> {
104        let mut guards = Vec::new();
105        if let Some(left) = left {
106            guards.push(AggregateGuard {
107                relation: reverse(left.relation.unwrap_or(Relation::Le)),
108                bound: self.aggregate_expression(&left.term, variables, values)?,
109            });
110        }
111        if let Some(right) = right {
112            guards.push(AggregateGuard {
113                relation: right.relation.unwrap_or(Relation::Le),
114                bound: self.aggregate_expression(&right.term, variables, values)?,
115            });
116        }
117        Ok(guards)
118    }
119    fn aggregate_expression(
120        &mut self,
121        term: &Term,
122        variables: &mut Variables,
123        values: &mut Vec<LiteralIr>,
124    ) -> Result<Expression, FormulaFailure> {
125        if let Some(value) = sentinel(term) {
126            let key = self.value(&value)?;
127            return Ok(Expression {
128                nodes: vec![Operation::Constant(self.source.scalar_key(
129                    &key,
130                    self.limits,
131                    self.counters,
132                    self.location,
133                )?)],
134            });
135        }
136        self.ranged_expression(term, variables, values)
137    }
138    pub(super) fn body_guards(
139        &mut self,
140        source: &Body,
141        variables: &mut Variables,
142        values: &mut Vec<LiteralIr>,
143    ) -> Result<Vec<Vec<AggregateGuard>>, FormulaFailure> {
144        let mut result = Vec::new();
145        for element in source.elements() {
146            if let BodyElement::Aggregate { aggregate, .. } = element.get() {
147                let guarded: &dyn HasGuards = match aggregate {
148                    Aggregate::Function(value) => value,
149                    Aggregate::Set(value) => value,
150                };
151                result.push(self.guards(
152                    guarded.left_guard().map(WithProvenance::get),
153                    guarded.right_guard().map(WithProvenance::get),
154                    variables,
155                    values,
156                )?);
157            }
158        }
159        Ok(result)
160    }
161    pub(super) fn body_aggregates(
162        &mut self,
163        source: &Body,
164        guards: Vec<Vec<AggregateGuard>>,
165        assignments: Vec<Option<usize>>,
166        variables: &Variables,
167        body: &mut Vec<LiteralIr>,
168    ) -> Result<(), FormulaFailure> {
169        let mut guards = guards.into_iter();
170        let mut assignments = assignments.into_iter();
171        for element in source.elements() {
172            if let BodyElement::Aggregate {
173                negation,
174                aggregate,
175            } = element.get()
176            {
177                let binding = assignments
178                    .next()
179                    .expect("one binding marker per aggregate");
180                let mut aggregate = self.aggregate(
181                    *negation,
182                    aggregate,
183                    guards.next().expect("one guard list per aggregate"),
184                    variables,
185                    binding,
186                )?;
187                self.assignment_scope(&aggregate)?;
188                aggregate.family_inputs =
189                    self.assignment_family_inputs(&aggregate, variables.count)?;
190                body.push(LiteralIr::Aggregate(aggregate));
191            }
192        }
193        Ok(())
194    }
195    pub(super) fn aggregate_term(
196        &mut self,
197        term: &Term,
198        variables: &mut Variables,
199    ) -> Result<CoreTerm, FormulaFailure> {
200        if !matches!(term, Term::Variable(_) | Term::Symbolic(_)) {
201            return Err(unsupported(ProfileFeature::Aggregate, self.location).into());
202        }
203        self.domain_term(term, variables)
204    }

206    /// Explicit scalar roots in rule, aggregate and witness contexts. Objective
207    /// fields use `objective_term` instead and do not enlarge this root selection.
208    pub(super) fn domain_term(
209        &mut self,
210        term: &Term,
211        variables: &mut Variables,
212    ) -> Result<CoreTerm, FormulaFailure> {
213        match term {
214            Term::Symbolic(symbol) => {
215                self.budget
216                    .charge(crate::ExpansionResource::TermWork, 1, self.location)?;
217                let value = crate::compile::scalar(symbol, self.location)?;
218                Ok(CoreTerm::Constant(self.root_scalar(&value)?))
219            }
220            _ => self.objective_term(term, variables),
221        }
222    }
223    fn aggregate(
224        &mut self,
225        negation: DefaultNegation,
226        aggregate: &Aggregate,
227        guards: Vec<AggregateGuard>,
228        variables: &Variables,
229        binding: Option<usize>,
230    ) -> Result<AggregateIr, FormulaFailure> {
231        let mut elements = Vec::new();
232        let function = match aggregate {
233            Aggregate::Function(aggregate) => {
234                if !matches!(
235                    aggregate.function(),
236                    AggregateFunction::Count
237                        | AggregateFunction::Sum
238                        | AggregateFunction::SumPlus
239                        | AggregateFunction::Min
240                        | AggregateFunction::Max
241                ) {
242                    return Err(unsupported(ProfileFeature::Aggregate, self.location).into());
243                }
244                for (index, element) in aggregate.elements().enumerate() {
245                    let family = LocalFamily(index);
246                    let fields: Vec<_> = element.get().terms().collect();
247                    for (terms, source_condition) in
248                        self.local_alternatives(&fields, element.get().condition())?
249                    {
250                        elements.push(self.aggregate_tuple_element(
251                            &terms,
252                            &source_condition,
253                            aggregate.function(),
254                            variables,
255                            family,
256                        )?);
257                    }
258                }
259                aggregate.function()
260            }
261            Aggregate::Set(aggregate) => {
262                for (index, element) in aggregate.elements().enumerate() {
263                    let family = LocalFamily(index);
264                    let (literal, source_condition) = match element.get() {
265                        SetElement::Literal(literal) => (literal, Condition::new([])),
266                        SetElement::ConditionalLiteral(value) => {
267                            (&value.literal, value.condition.clone())
268                        }
269                    };
270                    for alternative in self.condition_alternatives(&source_condition)? {
271                        for selected in self.literal_alternatives(literal)? {
272                            let mut local = variables.clone();
273                            let mut condition = self.condition(&alternative, &mut local)?;
274                            let themelios_program::program::LiteralInner::Atom(atom) =
275                                &selected.inner
276                            else {
277                                return Err(
278                                    unsupported(ProfileFeature::Aggregate, self.location).into()
279                                );
280                            };
281                            if selected.negation != DefaultNegation::None {
282                                return Err(
283                                    unsupported(ProfileFeature::Aggregate, self.location).into()
284                                );
285                            }
286                            let atom =
287                                self.positive_atom_key(atom.get(), &mut local, &mut condition)?;
288                            self.bindings(&mut condition, &mut local)?;
289                            local.safety(self.location)?;
290                            elements.push(AggregateElementIr {
291                                family,
292                                key: AggregateKey::Atom(atom),
293                                condition,
294                                variables: local.count,
295                            });
296                        }
297                    }
298                }
299                AggregateFunction::Count
300            }
301        };
302        let id = self.next_aggregate;
303        self.next_aggregate += 1;
304        Ok(AggregateIr {
305            id,
306            binding,
307            family_inputs: Vec::new(),
308            negation,
309            function,
310            guards,
311            elements,
312        })
313    }

315    fn aggregate_tuple_element(
316        &mut self,
317        terms: &[Term],
318        source_condition: &Condition,
319        function: AggregateFunction,
320        variables: &Variables,
321        family: LocalFamily,
322    ) -> Result<AggregateElementIr, FormulaFailure> {
323        let mut local = variables.clone();
324        let mut condition = self.condition(source_condition, &mut local)?;
325        let tuple = terms
326            .iter()
327            .map(|term| {
328                if matches!(term, Term::Variable(_) | Term::Symbolic(_)) {
329                    self.aggregate_term(term, &mut local)
330                } else {
331                    self.generated_term(term, &mut local, &mut condition)
332                }
333            })
334            .collect::<Result<Vec<_>, _>>()?;
335        if matches!(function, AggregateFunction::Min | AggregateFunction::Max) && tuple.is_empty() {
336            return Err(unsupported(ProfileFeature::Aggregate, self.location).into());
337        }
338        self.bindings(&mut condition, &mut local)?;
339        local.safety(self.location)?;
340        Ok(AggregateElementIr {
341            family,
342            key: AggregateKey::Tuple(tuple),
343            condition,
344            variables: local.count,
345        })
346    }
347}
348fn sentinel(term: &Term) -> Option<Value> {
349    match term {
350        Term::Symbolic(Symbol::Infimum) => Some(Value::Infimum),
351        Term::Symbolic(Symbol::Supremum) => Some(Value::Supremum),
352        _ => None,
353    }
354}
355fn reverse(relation: Relation) -> Relation {
356    match relation {
357        Relation::Eq => Relation::Eq,
358        Relation::Neq => Relation::Neq,
359        Relation::Lt => Relation::Gt,
360        Relation::Le => Relation::Ge,
361        Relation::Gt => Relation::Lt,
362        Relation::Ge => Relation::Le,
363    }
