1//! Exact analysis facts over prevalidated source leaves, not exported core values.
2//!
3//! A recipe is admitted only after `fact_expansion` checked the complete family.
4//! Its integer segments and leaf positions describe that immutable normalized
5//! carrier. Neither preparation readers nor later public inspection replay
6//! normalization, arithmetic, admission or the structured-value bridge.

84/// Preparation reserves with typed limits; already-admitted public inspection
85/// uses upstream's ordinary infallible allocation boundary. The same traversal
86/// and private locator invariant serve both readers.
87pub(super) trait Allocation {
88    type Failure;
89    fn poll(&mut self) -> Result<(), Self::Failure>;
90    fn reserve_prepaid<T>(
91        &mut self,
92        values: &mut Vec<T>,
93        count: usize,
94    ) -> Result<(), Self::Failure>;
95    fn reserve<T>(&mut self, values: &mut Vec<T>, count: usize) -> Result<(), Self::Failure>;
96}

98pub(crate) struct Checked<'a> {
99    pub(crate) budget: &'a mut Budget,
100    pub(crate) site: ProgramSite,
101}

103impl Allocation for Checked<'_> {
104    type Failure = FormulaFailure;
105    fn reserve_prepaid<T>(
106        &mut self,
107        values: &mut Vec<T>,
108        count: usize,
109    ) -> Result<(), Self::Failure> {
110        self.budget.poll(self.site)?;
111        values
112            .try_reserve_exact(count)
113            .map_err(|error| crate::ExpansionFailure::Allocation {
114                error,
115                location: self.site,
116            })?;
117        Ok(())
118    }
119    fn poll(&mut self) -> Result<(), Self::Failure> {
120        Ok(self.budget.poll(self.site)?)
121    }
122    fn reserve<T>(&mut self, values: &mut Vec<T>, count: usize) -> Result<(), Self::Failure> {
123        self.budget.poll(self.site)?;
124        crate::formula_pattern::reserve_amortized(values, count, self.budget, self.site)
125    }
126}

148impl Families {
149    /// Record a family whose whole shape and expansion have already passed
150    /// `fact_expansion`. Empty products are omitted, but every scalar leaf keeps
151    /// its source ordinal so later reads do not depend on omitted products.
152    pub(crate) fn record(
153        &mut self,
154        atom: &Atom,
155        provenance: &Provenance,
156        owner: StatementId,
157        expected: usize,
158        allocation: &mut Checked<'_>,
159    ) -> Result<usize, FormulaFailure> {
160        let family = self.sources.len();
161        let mut scalar = 0;
162        let mut rows = 0;
163        for arguments in atom.alternatives() {
164            let first_column = self.columns.len();
165            let first_segment = self.segments.len();
166            for term in arguments {
167                self.record_column(term, &mut scalar, allocation)?;
168            }
169            if self.columns[first_column..]
170                .iter()
171                .any(|column| column.width == 0)
172            {
173                self.columns.truncate(first_column);
174                self.segments.truncate(first_segment);
175                continue;
176            }
177            let mut count = 1_usize;
178            for column in self.columns[first_column..].iter_mut().rev() {
179                column.stride = count;
180                count = count
181                    .checked_mul(column.width)
182                    .expect("the admitted product fits usize");
183            }
184            if count == 0 {
185                self.columns.truncate(first_column);
186                self.segments.truncate(first_segment);
187                continue;
188            }
189            rows += count;
190            allocation.reserve(&mut self.products, 1)?;
191            self.products.push(Product {
192                family,
193                columns: first_column..self.columns.len(),
194                rows: count,
195            });
196        }
197        assert_eq!(
198            rows, expected,
199            "source recipe and accepted expansion have the same rows"
200        );
201        allocation.reserve(&mut self.sources, 1)?;
202        self.sources.push(Family {
203            owner,
204            normalized: None,
205            provenance: provenance.clone(),
206        });
207        Ok(family)
208    }
209
210    /// One column preserves source leaf order; empty intervals add no segment.
211    fn record_column(
212        &mut self,
213        term: &Term,
214        scalar: &mut usize,
215        allocation: &mut Checked<'_>,
216    ) -> Result<(), FormulaFailure> {
217        let start = self.segments.len();
218        let mut width = 0;
219        let mut pending = Vec::new();
220        allocation.reserve(&mut pending, 1)?;
221        pending.push(term);
222        while let Some(term) = pending.pop() {
223            allocation.budget.poll(allocation.site)?;
224            let (value, count) = match term {
225                Term::Pool(items) => {
226                    allocation.reserve(&mut pending, items.len())?;
227                    pending.extend(items.iter().rev());
228                    continue;
229                }
230                Term::Symbolic(_) => {
231                    let index = *scalar;
232                    *scalar += 1;
233                    (SegmentValue::Symbol(index), 1)
234                }
235                Term::Interval { lower, upper } => {
236                    let (Term::Symbolic(lower), Term::Symbolic(upper)) = (&**lower, &**upper)
237                    else {
238                        unreachable!("fact expansion accepted symbolic interval endpoints")
239                    };
240                    let (Symbol::Number(lower), Symbol::Number(upper)) = (lower, upper) else {
241                        continue;
242                    };
243                    if lower > upper {
244                        continue;
245                    }
246                    let count = usize::try_from(i64::from(*upper) - i64::from(*lower) + 1)
247                        .expect("the admitted family fits usize");
248                    (SegmentValue::Integers(*lower), count)
249                }
250                _ => unreachable!("fact expansion accepted only symbolic leaves and intervals"),
251            };
252            width += count;
253            allocation.reserve(&mut self.segments, 1)?;
254            self.segments.push(Segment { end: width, value });
255        }
256        allocation.reserve(&mut self.columns, 1)?;
257        self.columns.push(Column {
258            segments: start..self.segments.len(),
259            width,
260            stride: 1,
261        });
262        Ok(())
263    }

281    pub(super) fn read<'a, A: Allocation>(
282        &'a self,
283        source: &'a Program,
284        allocation: &mut A,
285    ) -> Result<Reader<'a>, A::Failure> {
286        allocation.poll()?;
287        if self.sources.is_empty() {
288            return Ok(Reader {
289                families: self,
290                inputs: Vec::new(),
291            });
292        }
293        let mut originals = Vec::new();
294        for statement in source.statements() {
295            allocation.reserve(&mut originals, 1)?;
296            originals.push(statement);
297        }
298        let mut inputs = Vec::new();
299        allocation.reserve(&mut inputs, self.sources.len())?;
300        for family in &self.sources {
301            allocation.poll()?;
302            let statement = family.normalized.as_ref().map_or_else(
303                || originals[family.owner.index()],
304                |program| {
305                    program
306                        .statements()
307                        .next()
308                        .expect("one normalized family carrier")
309                },
310            );
311            let atom = crate::fact_expansion::fact_atom(statement.get())
312                .expect("validated fact recipe carrier");
313            let mut symbols = Vec::new();
314            let mut pending = Vec::new();
315            for arguments in atom.alternatives() {
316                for term in arguments {
317                    allocation.reserve(&mut pending, 1)?;
318                    pending.push(term);
319                    while let Some(term) = pending.pop() {
320                        match term {
321                            Term::Pool(items) => {
322                                allocation.reserve(&mut pending, items.len())?;
323                                pending.extend(items.iter().rev());
324                            }
325                            Term::Symbolic(symbol) => {
326                                allocation.reserve(&mut symbols, 1)?;
327                                symbols.push(symbol);
328                            }
329                            Term::Interval { .. } => {}
330                            _ => unreachable!("validated fact source recipe"),
331                        }
332                    }
333                }
334            }
335            inputs.push(Input { atom, symbols });
336        }
337        Ok(Reader {
338            families: self,
339            inputs,
340        })
341    }
342}

344impl Reader<'_> {
345    pub(super) fn canonical(
346        &self,
347        allocation: &mut Checked<'_>,
348    ) -> Result<Vec<Fact>, FormulaFailure> {
349        let mut map = BTreeMap::<Key<'_, '_>, Fact>::new();
350        for (product, plan) in self.families.products.iter().enumerate() {
351            for row in 0..plan.rows {
352                allocation
353                    .budget
354                    .charge(ExpansionResource::TermWork, 1, allocation.site)?;
355                // As for Owners, this names BTreeMap payload, not allocator bookkeeping.
356                allocation.budget.charge(
357                    ExpansionResource::ScalarBytes,
358                    size_of::<(Key<'_, '_>, Fact)>() as u128,
359                    allocation.site,
360                )?;
361                let position = Position { product, row };
362                match map.entry(Key {
363                    reader: self,
364                    position,
365                }) {
366                    std::collections::btree_map::Entry::Vacant(entry) => {
367                        entry.insert(Fact {
368                            position,
369                            others: Vec::new(),
370                        });
371                    }
372                    std::collections::btree_map::Entry::Occupied(mut entry) => {
373                        let fact = entry.get_mut();
374                        let first = self.families.products[fact.position.product].family;
375                        if plan.family != first && fact.others.last() != Some(&plan.family) {
376                            allocation.reserve(&mut fact.others, 1)?;
377                            fact.others.push(plan.family);
378                        }
379                    }
380                }
381            }
382        }
383        let mut facts = Vec::new();
384        allocation.reserve(&mut facts, map.len())?;
385        for fact in map.into_values() {
386            allocation.poll()?;
387            facts.push(fact);
388        }
389        Ok(facts)
390    }

400    pub(super) fn signature(&self, position: Position) -> (Sign, &Name, usize) {
401        let atom = self.input(position).atom;
402        (atom.sign, &atom.name, self.product(position).columns.len())
403    }
404
405    fn symbols(&self, position: Position) -> impl Iterator<Item = Scalar<'_>> {
406        let product = self.product(position);
407        self.families.columns[product.columns.clone()]
408            .iter()
409            .map(move |column| {
410                let offset = (position.row / column.stride) % column.width;
411                let segments = &self.families.segments[column.segments.clone()];
412                let index = segments.partition_point(|segment| segment.end <= offset);
413                let segment = &segments[index];
414                match segment.value {
415                    SegmentValue::Symbol(index) => {
416                        Scalar::Symbol(self.input(position).symbols[index])
417                    }
418                    SegmentValue::Integers(first) => {
419                        let start = if index == 0 {
420                            0
421                        } else {
422                            segments[index - 1].end
423                        };
424                        let number = i64::from(first)
425                            + i64::try_from(offset - start).expect("admitted i32 interval width");
426                        Scalar::Number(
427                            i32::try_from(number).expect("position in admitted i32 interval"),
428                        )
429                    }
430                }
431            })
432    }

434    fn compare(&self, left: Position, right: Position) -> Ordering {
435        let left_atom = self.input(left).atom;
436        let right_atom = self.input(right).atom;
437        left_atom
438            .sign
439            .cmp(&right_atom.sign)
440            .then_with(|| left_atom.name.cmp(&right_atom.name))
441            .then_with(|| self.symbols(left).cmp(self.symbols(right)))
442    }

444    pub(super) fn atom(&self, fact: &Fact) -> Atom {
445        self.atom_at(fact.position)

448    pub(super) fn atom_at(&self, position: Position) -> Atom {
449        let source = self.input(position).atom;
450        Atom {
451            sign: source.sign,
452            name: source.name.clone(),
453            arguments: Arguments::Single(
454                self.symbols(position)

461    pub(super) fn representative(&self, position: Position) -> WithProvenance<Statement> {
462        let provenance = &self.families.sources[self.product(position).family].provenance;
463        statement(self.atom_at(position), provenance.clone())

466    pub(super) fn statement(&self, fact: &Fact) -> WithProvenance<Statement> {
467        statement(self.atom(fact), self.provenance(fact).into_owned())
468    }
469
470    pub(super) fn provenance(&self, fact: &Fact) -> Cow<'_, Provenance> {
471        let first = &self.families.sources[self.product(fact.position).family].provenance;
472        if fact.others.is_empty() {
473            Cow::Borrowed(first)
474        } else {
475            Cow::Owned(
476                fact.others
477                    .iter()
478                    .fold(first.clone(), |provenance, &family| {
479                        provenance.merge(self.families.sources[family].provenance.clone())
480                    }),
481            )
482        }
483    }

485    /// The last source family determines the fact carrier's content on ingest;
486    /// all fact children are constructed, while root evidence is unioned.
487    pub(super) fn last_owner(&self, fact: &Fact) -> StatementId {
488        let family = fact
489            .others
490            .last()
491            .copied()
492            .unwrap_or(self.product(fact.position).family);
493        self.families.sources[family].owner
494    }

504    /// A fact is `Statement::Rule` with a positive atom literal and empty body.
505    /// These comparisons pin the reviewed themelios derived field/variant order;
506    /// no source value is copied merely to compare a virtual fact with a rule.
507    pub(super) fn compare_statement(&self, fact: &Fact, statement: &Statement) -> Ordering {
508        let Statement::Rule(rule) = statement else {
509            return Ordering::Less;
510        };
511        let Head::Literal(literal) = rule.head().get() else {
512            return Ordering::Less;
513        };
514        let negation = DefaultNegation::None.cmp(&literal.negation);
515        if negation != Ordering::Equal {
516            return negation;
517        }
518        let LiteralInner::Atom(atom) = &literal.inner else {
519            return Ordering::Less;
520        };
521        let source = self.input(fact.position).atom;
522        let order = source
523            .sign
524            .cmp(&atom.get().sign)
525            .then_with(|| source.name.cmp(&atom.get().name));
526        if order != Ordering::Equal {
527            return order;
528        }
529        let Arguments::Single(terms) = &atom.get().arguments else {
530            return Ordering::Less;
531        };
532        let mut left = self.symbols(fact.position);
533        let mut right = terms.iter();
534        loop {
535            let order = match (left.next(), right.next()) {
536                (Some(left), Some(right)) => match right {
537                    Term::Symbolic(right) => left.cmp(&Scalar::Symbol(right)),
538                    Term::Variable(_) => Ordering::Greater,
539                    _ => Ordering::Less,
540                },
541                (None, None) => break,
542                (None, Some(_)) => Ordering::Less,
543                (Some(_), None) => Ordering::Greater,
544            };
545            if order != Ordering::Equal {
546                return order;
547            }
548        }
549        if rule.body().get().elements().next().is_none() {
550            Ordering::Equal
551        } else {
552            Ordering::Less
553        }
554    }
555}

561struct Key<'r, 's> {
562    reader: &'r Reader<'s>,
563    position: Position,
564}
565impl PartialEq for Key<'_, '_> {
566    fn eq(&self, other: &Self) -> bool {
567        self.cmp(other) == Ordering::Equal
568    }
569}
570impl Eq for Key<'_, '_> {}
571impl PartialOrd for Key<'_, '_> {
572    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
573        Some(self.cmp(other))
574    }
575}
576impl Ord for Key<'_, '_> {
577    fn cmp(&self, other: &Self) -> Ordering {
578        self.reader.compare(self.position, other.position)
579    }

582enum Scalar<'a> {
583    Symbol(&'a Symbol),
584    Number(i32),
585}
586impl Scalar<'_> {
587    fn owned(self) -> Symbol {
588        match self {
589            Self::Symbol(symbol) => symbol.clone(),
590            Self::Number(number) => Symbol::Number(number),
591        }
592    }
593}
594impl PartialEq for Scalar<'_> {
595    fn eq(&self, other: &Self) -> bool {
596        self.cmp(other) == Ordering::Equal
597    }
598}
599impl Eq for Scalar<'_> {}
600impl PartialOrd for Scalar<'_> {
601    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
602        Some(self.cmp(other))
603    }
604}
605impl Ord for Scalar<'_> {
606    fn cmp(&self, other: &Self) -> Ordering {
607        match (self, other) {
608            (Self::Symbol(left), Self::Symbol(right)) => left.cmp(right),
609            (Self::Symbol(left), Self::Number(right)) => (*left).cmp(&Symbol::Number(*right)),
610            (Self::Number(left), Self::Symbol(right)) => Symbol::Number(*left).cmp(right),
611            (Self::Number(left), Self::Number(right)) => left.cmp(right),
612        }
613    }
