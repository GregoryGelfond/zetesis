1//! One nonnegative packed interval inside the finite assignment-sum operation.
2//!
3//! A bit at s means that some optional subset has value s above the seed, so
4//! the sum seed + s is reachable, with existence (not the number of witnesses)
5//! retained. Descending shifts read the previous occurrence state, so repeated
6//! weights still make separate contributions. The caller can enter once and
7//! leave once; no input list or cache is retained.
8

15const WORD_BITS: usize = u64::BITS as usize;
16
17#[derive(Clone, Copy)]
18pub(super) struct Plan {
19    /// The largest reachable sum above the seed.
20    maximum: usize,
21    words: usize,
22}
23
24impl Plan {
25    /// A packed route for a run that still starts at its `seed`, so every
26    /// value lies at or above it.
27    pub(super) fn for_runs(
28        values: &Numbers,
29        seed: i32,
30        weight: i32,
31        context: &mut Context<'_, &Computation<'_, '_>>,
32    ) -> Result<Option<Self>, FormulaFailure> {
33        if values.slice().first() != Some(&seed) {
34            return Ok(None);
35        }
36        let last = *values.slice().last().expect("the seed is retained");
37        let Some(plan) = Self::select(
38            values.len(),
39            i64::from(last) - i64::from(seed),
40            seed,
41            weight,
42            context.work.location,
43        )?
44        else {
45            return Ok(None);
46        };
47        // Entry temporarily retains the canonical run and the packed owner.
48        // Conservatively keep both existing run leases in this proposal; the
49        // unused spare is released before the selected allocation is attempted.
50        let requested = size_of::<Buffer<u64>>() as u128 + 8 * plan.words as u128;
51        Ok(fits_extra_bytes(requested, context)?.then_some(plan))
52    }
53
54    /// `maximum` is the largest sum above `seed` before adding `weight`. The
55    /// largest resulting sum, `seed` plus the new maximum, must fit `i32`.
56    fn select(
57        count: usize,
58        maximum: i64,
59        seed: i32,
60        weight: i32,
61        location: crate::ProgramSite,
62    ) -> Result<Option<Self>, FormulaFailure> {
63        if weight < 0 {
64            return Ok(None);
65        }
66        let maximum = maximum + i64::from(weight);
67        if i32::try_from(i64::from(seed) + maximum).is_err() {
68            return Err(overflow(location));
69        }
70        let maximum = usize::try_from(maximum).expect("nonnegative maximum");
71        let words = (maximum + 1).div_ceil(WORD_BITS);
72        let cells = count as u128;
73        let word_count = words as u128;
74        // Sorted merging visits M old values and K union values, then pushes K
75        // outputs: at least 3M units since K >= M. Entry plus one update and
76        // eventual enumeration takes at most M + 4W + K + 4 logical units,
77        // excluding shared allocation-copy receipts. 4W+4 < M pays even for
78        // the conversion when this is the final contribution. No lookahead.
79        // Geometric word capacity is bounded by 2W requested cells; requiring
80        // 16W <= 4M makes it no larger than the next merge run's minimum payload.
81        if 4 * word_count + 4 >= cells || 16 * word_count > 4 * cells {
82            return Ok(None);
83        }
84        // This is a work comparison, not an advance debit or a reason to
85        // choose a more expensive fallback near the work ceiling. The selected
86        // kernel charges its actual visits through the unchanged shared account.
87        Ok(Some(Self { maximum, words }))
88    }
89}
90
91// Read the remaining allowance only when a new allocation is proposed. Buffer
92// still admits actual capacity and replacement overlap; a selected refusal is
93// an error and never retries the more expensive representation.
94fn fits_extra_bytes(
95    requested: u128,
96    context: &mut Context<'_, &Computation<'_, '_>>,
97) -> Result<bool, FormulaFailure> {
98    let lease = context.computation.lease();
99    let available =
100        context
101            .computation
102            .allowance(&lease, context.work.limits, context.work.location)?;
103    Ok(requested <= available as u128)
104}

106struct Packed {
107    words: Buffer<u64>,
108    /// The sum every bit offset is taken from.
109    seed: i32,
110    maximum: usize,
111    count: usize,
112}
113
114pub(super) fn finish(
115    values: Numbers,
116    seed: i32,
117    first: i32,
118    weights: impl IntoIterator<Item = i32>,
119    plan: Plan,
120    context: &mut Context<'_, &Computation<'_, '_>>,
121) -> Result<Numbers, FormulaFailure> {
122    let mut packed = Packed::from_values(values, seed, plan, context)?;
123    packed.add(first, plan, context)?;
124    let mut weights = weights.into_iter();
125    for weight in weights.by_ref() {
126        tick(context)?;
127        let Some(plan) = packed.next_plan(weight, context)? else {
128            let values = packed.into_numbers(context)?;
129            let mut runs = Runs::from_values(values, context)?;
130            // This occurrence already paid its dispatch/cancellation charge.
131            runs.add(weight, context)?;
132            return runs.finish(weights, context);
133        };
134        packed.add(weight, plan, context)?;
135    }
136    packed.into_numbers(context)
137}

139impl Packed {
140    fn next_plan(
141        &self,
142        weight: i32,
143        context: &mut Context<'_, &Computation<'_, '_>>,
144    ) -> Result<Option<Plan>, FormulaFailure> {
145        let maximum = i64::try_from(self.maximum).expect("admitted maximum");
146        let Some(plan) = Plan::select(
147            self.count,
148            maximum,
149            self.seed,
150            weight,
151            context.work.location,
152        )?
153        else {
154            return Ok(None);
155        };
156        // Retained updates need no new lease or header. A replacement keeps
157        // the already-accounted old words live, so only the requested new
158        // payload is additional here. Growth within capacity needs no probe.
159        if plan.words > self.words.capacity() && !fits_extra_bytes(8 * plan.words as u128, context)?
160        {
161            return Ok(None);
162        }
163        Ok(Some(plan))
164    }
165
166    fn from_values(
167        values: Numbers,
168        seed: i32,
169        plan: Plan,
170        context: &mut Context<'_, &Computation<'_, '_>>,
171    ) -> Result<Self, FormulaFailure> {
172        let above = |sum: i32| {
173            usize::try_from(i64::from(sum) - i64::from(seed)).expect("sums at or above the seed")
174        };
175        let maximum = above(*values.slice().last().expect("the seed is retained"));
176        let count = values.len();
177        let mut words = Buffer::new(
178            context.computation,
179            context.work.limits,
180            context.work.counters,
181            context.work.location,
182        )?;
183        words.resize(
184            plan.words,
185            0,
186            context.computation,
187            context.work.limits,
188            context.work.counters,
189            context.work.location,
190        )?;
191        for &sum in values.iter() {
192            tick(context)?;
193            let sum = above(sum);
194            words.slice_mut()[sum / WORD_BITS] |= 1 << (sum % WORD_BITS);
195        }
196        // The old canonical run and the new words overlap during conversion;
197        // both leases remain live until this point, then the run is released.
198        drop(values);
199        Ok(Self {
200            words,
201            seed,
202            maximum,
203            count,
204        })
205    }
206
207    fn add(
208        &mut self,
209        weight: i32,
210        plan: Plan,
211        context: &mut Context<'_, &Computation<'_, '_>>,
212    ) -> Result<(), FormulaFailure> {
213        let weight = usize::try_from(weight).expect("nonnegative packed contribution");
214        if weight == 0 {
215            return Ok(());
216        }
217        if plan.words > self.words.len() {
218            self.words.resize(
219                plan.words,
220                0,
221                context.computation,
222                context.work.limits,
223                context.work.counters,
224                context.work.location,
225            )?;
226        }
227        let old_words = (self.maximum + 1).div_ceil(WORD_BITS);
228        let shift = weight % WORD_BITS;
229        for source in (0..old_words).rev() {
230            tick(context)?;
231            let bits = self.words.slice()[source];
232            let target = source + weight / WORD_BITS;
233            self.words.slice_mut()[target] |= bits << shift;
234            if shift != 0 && target + 1 < self.words.len() {
235                self.words.slice_mut()[target + 1] |= bits >> (WORD_BITS - shift);
236            }
237        }
238        let mut count = 0;
239        for &word in self.words.iter() {
240            tick(context)?;
241            count += word.count_ones() as usize;
242            let limit = context.work.limits.max_assignment_values as u128;
243            // Preserve the first excess distinct-value count used by merging,
244            // even when one packed word crosses several values at once.
245            ceiling(
246                FormulaResource::AssignmentValues,
247                (count as u128).min(limit + 1),
248                limit,
249                context.work.location,
250            )?;
251        }
252        self.maximum = plan.maximum;
253        self.count = count;
254        Ok(())
255    }
256
257    fn into_numbers(
258        self,
259        context: &mut Context<'_, &Computation<'_, '_>>,
260    ) -> Result<Numbers, FormulaFailure> {
261        let mut values = Buffer::new(
262            context.computation,
263            context.work.limits,
264            context.work.counters,
265            context.work.location,
266        )?;
267        values.reserve(
268            self.count,
269            context.computation,
270            context.work.limits,
271            context.work.counters,
272            context.work.location,
273        )?;
274        for (index, &word) in self.words.iter().enumerate() {
275            tick(context)?;
276            let mut remaining = word;
277            while remaining != 0 {
278                let bit = remaining.trailing_zeros() as usize;
279                let above = i64::try_from(index * WORD_BITS + bit).expect("an admitted offset");
280                let sum = i32::try_from(i64::from(self.seed) + above)
281                    .expect("every set bit is an admitted i32 sum");
282                values.push(
283                    sum,
284                    context.computation,
285                    context.work.limits,
286                    context.work.counters,
287                    context.work.location,
288                )?;
289                remaining &= remaining - 1;
290            }
291        }
292        Ok(values)
293    }
