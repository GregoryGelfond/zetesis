18    pub(super) fn assignment_targets(
19        &self,
20        source: &Body,
21        guards: &[Vec<AggregateGuard>],
22        variables: &mut Variables,
23    ) -> Result<Vec<Option<usize>>, FormulaFailure> {
24        let mut result = Vec::new();
25        for (element, guards) in source
26            .elements()
27            .filter(|element| matches!(element.get(), BodyElement::Aggregate { .. }))
28            .zip(guards)
29        {
30            let [guard] = guards.as_slice() else {
31                result.push(None);
32                continue;
33            };
34            let [Operation::Variable(target)] = guard.bound.nodes.as_slice() else {
35                result.push(None);
36                continue;
37            };
38            if guard.relation != Relation::Eq || variables.safe.contains(target) {
39                result.push(None);
40                continue;
41            }
42            let BodyElement::Aggregate {
43                negation: DefaultNegation::None,
44                aggregate: Aggregate::Function(aggregate),
45            } = element.get()
46            else {
47                result.push(None);
48                continue;
49            };
50            if !matches!(
51                aggregate.function(),
52                AggregateFunction::Count
53                    | AggregateFunction::Sum
54                    | AggregateFunction::SumPlus
55                    | AggregateFunction::Min
56                    | AggregateFunction::Max
57            ) || !variables.named.values().any(|variable| variable == target)
58            {
59                return Err(unsupported(ProfileFeature::AggregateAssignment, self.location).into());
60            }
61            variables.safe.insert(*target);
62            result.push(Some(*target));
63        }
64        Ok(result)
65    }
66    pub(super) fn assignment_scope(
67        &mut self,
68        aggregate: &AggregateIr,
69    ) -> Result<(), FormulaFailure> {
70        let Some(target) = aggregate.binding else {
71            return Ok(());
72        };
73        for element in &aggregate.elements {
74            if self.element_uses(element, target)? {
75                return Err(FormulaFailure::UnsafeVariable {
76                    variable: target,
77                    location: self.location,
78                });
79            }
80        }
81        Ok(())
82    }
83    pub(super) fn assignment_context(
84        &mut self,
85        body: &[LiteralIr],
86        choice_guards: &[AggregateGuard],
87        aggregate_values: &[bool],
88    ) -> Result<bool, FormulaFailure> {
89        let mut consumed = false;
90        for (target, dependent) in aggregate_values.iter().enumerate() {
91            self.scope_work(1)?;
92            if !dependent {
93                continue;
94            }
95            for guard in choice_guards {
96                // A head bound consumes the completed proposal row. The
97                // original aggregate equality still guards the entire choice;
98                // neither a proposal nor its cardinality establishes support.
99                consumed |= self.expression_uses(&guard.bound, target)?;
100            }
101            for literal in body {
102                self.scope_work(1)?;
103                // Outer body consumers share one contract across all heads:
104                // consume completed proposal values while retaining the
105                // original equality and body activation. Local element scopes
106                // are compiled separately from this completed outer frame.
107                if matches!(
108                    literal,
109                    LiteralIr::Atom(DefaultNegation::Not | DefaultNegation::NotNot, _)
110                        | LiteralIr::ProjectedAtom(
111                            DefaultNegation::Not | DefaultNegation::NotNot,
112                            _,
113                        )
114                        | LiteralIr::Compare(..)
115                        | LiteralIr::ArgumentCheck { .. }
116                        | LiteralIr::TupleCompare(..)
117                        | LiteralIr::Guard(_)
118                        | LiteralIr::HeadGuard(_)
119                        | LiteralIr::Bind { .. }
120                        | LiteralIr::Range { .. }
121                        | LiteralIr::Conditional(_)
122                ) {
123                    // Negative gates read completed values without binding or
124                    // pruning possible rows. Grounding retains their original
125                    // polarity, including the existential projection formula.
126                    // Range generators follow the checked plan; a previously
127                    // bound range target remains a completed-row membership test.
128                    // Universal conditions seed their local joins with this
129                    // complete outer frame. Their implications and consequent
130                    // alternatives remain formulas, never support-row filters.
131                    consumed |= self.literal_uses(literal, target)?;
132                    continue;
133                }
134                if let LiteralIr::Aggregate(other) = literal {
135                    // Its own equality guard supplies the target; scope checks
136                    // forbid reading that target inside its own elements.
137                    // Another producer may read it only after the checked plan
138                    // completes its inputs. Such a dependency is still a value
139                    // consumer for the objective-observer admission contract.
140                    // A nonbinding aggregate reads guards, element keys and
141                    // element conditions from the complete proposal row, with
142                    // independently scoped local witnesses. Its comparison
143                    // and default negation remain original formula operations;
144                    // the guard neither generates values nor filters support.
145                    if other.binding != Some(target) {
146                        consumed |= self.literal_uses(literal, target)?;
147                    }
148                    continue;
149                }
150                // Other consumers keep their established restrictions.
151                if self.literal_uses(literal, target)? {
152                    return Err(
153                        unsupported(ProfileFeature::AggregateAssignment, self.location).into(),
154                    );
155                }
156            }
157        }
158        Ok(consumed)
159    }

161    /// Summarize the normal rule after its complete body and head plan.
162    /// Every non-generated outer read outside the positive witness patterns
163    /// participates. This includes all generator expressions, so transitive
164    /// scalar/aggregate inputs are covered without a second dependency graph.
165    /// Positive matching and prefix errors still run for every base row.
166    /// Rich heads and unplanned scopes retain their existing traversal.
167    pub(super) fn continuation_inputs(
168        &mut self,
169        body: &[LiteralIr],
170        head: &crate::formula_ir::HeadIr,
171        variables: usize,
172        plan: &crate::formula_assignment_plan::Plan,
173    ) -> Result<Option<Vec<usize>>, FormulaFailure> {
174        let crate::formula_ir::HeadIr::Normal(Some(head)) = head else {
175            return Ok(None);
176        };
177        let mut count = 0;
178        for variable in 0..variables {
179            count += usize::from(self.continuation_uses(body, *head, plan, variable)?);
180        }
181        self.budget.charge(
182            ExpansionResource::ScalarBytes,
183            count as u128 * std::mem::size_of::<usize>() as u128,
184            self.location,
185        )?;
186        let mut inputs = Vec::with_capacity(count);
187        for variable in 0..variables {
188            if self.continuation_uses(body, *head, plan, variable)? {
189                inputs.push(variable);
190            }
191        }
192        Ok(Some(inputs))
193    }

195    fn continuation_uses(
196        &mut self,
197        body: &[LiteralIr],
198        head: AtomPattern,
199        plan: &crate::formula_assignment_plan::Plan,
200        variable: usize,
201    ) -> Result<bool, FormulaFailure> {
202        self.scope_work(plan.steps.len())?;
203        if plan.steps.iter().any(|step| step.produced == variable) {
204            return Ok(false);
205        }
206        if self.pattern_uses(head, variable)? {
207            return Ok(true);
208        }
209        for literal in body {
210            self.scope_work(1)?;
211            if matches!(
212                literal,
213                LiteralIr::Atom(DefaultNegation::None, _) | LiteralIr::PatternAtom(_)
214            ) {
215                continue;
216            }
217            if self.literal_uses(literal, variable)? {
218                return Ok(true);
219            }
220        }
221        Ok(false)
222    }

224    /// Retain only inherited reads that determine an assignment's complete
225    /// tuple and eligibility family. This uses the same scope analysis as the
226    /// proposal scheduler, before synthetic head slots exist. Two charged
227    /// scans reserve the exact summary size, once per source aggregate.
228    pub(super) fn assignment_family_inputs(
229        &mut self,
230        aggregate: &AggregateIr,
231        variables: usize,
232    ) -> Result<Vec<usize>, FormulaFailure> {
233        if aggregate.binding.is_none() {
234            return Ok(Vec::new());
235        }
236        let mut count = 0;
237        for variable in 0..variables {
238            count += usize::from(self.family_uses(aggregate, variable)?);
239        }
240        self.budget.charge(
241            ExpansionResource::ScalarBytes,
242            count as u128 * std::mem::size_of::<usize>() as u128,
243            self.location,
244        )?;
245        let mut inputs = Vec::with_capacity(count);
246        for variable in 0..variables {
247            if self.family_uses(aggregate, variable)? {
248                inputs.push(variable);
249            }
250        }
251        Ok(inputs)
252    }

254    fn family_uses(
255        &mut self,
256        aggregate: &AggregateIr,
257        variable: usize,
258    ) -> Result<bool, FormulaFailure> {
259        self.scope_work(1)?;
260        for element in &aggregate.elements {
261            if self.element_uses(element, variable)? {
262                return Ok(true);
263            }
264        }
265        Ok(false)
266    }

268    pub(super) fn scope_work(&mut self, count: usize) -> Result<(), FormulaFailure> {
269        self.budget
270            .charge(ExpansionResource::TermWork, count as u128, self.location)?;
271        Ok(())
272    }

274    pub(super) fn expression_uses(
275        &mut self,
276        expression: &Expression,
277        variable: usize,
278    ) -> Result<bool, FormulaFailure> {
279        self.scope_work(expression.nodes.len())?;
280        Ok(expression.inputs().any(|input| input == variable))

283    fn pattern_uses(&mut self, atom: AtomPattern, variable: usize) -> Result<bool, FormulaFailure> {
284        let atom = self
285            .source
286            .pattern_ref(atom, self.limits, self.counters, self.location)?;
287        self.budget.charge(
288            ExpansionResource::TermWork,
289            atom.terms().len() as u128,
290            self.location,
291        )?;
292        Ok(atom
293            .terms()
294            .iter()
295            .any(|term| matches!(term, TemplateTerm::Variable(slot) if slot == variable)))
296    }

298    fn projection_uses(
299        &mut self,
300        projection: &Projection,
301        variable: usize,
302    ) -> Result<bool, FormulaFailure> {
303        match projection {
304            Projection::Arguments { terms, .. } => {
305                self.scope_work(terms.len())?;
306                Ok(terms.contains(&Some(Term::Variable(variable))))
307            }
308            Projection::Witnesses {
309                bindings, inputs, ..
310            } => {
311                if variable >= *inputs {
312                    return Ok(false);
313                }
314                for literal in bindings {
315                    if self.literal_uses(literal, variable)? {
316                        return Ok(true);
317                    }
318                }
319                Ok(false)
320            }
321        }
322    }

324    pub(super) fn element_uses(
325        &mut self,
326        element: &AggregateElementIr,
327        variable: usize,
328    ) -> Result<bool, FormulaFailure> {
329        // Callers supply an outer-frame slot. Element-local slots extend that
330        // frame, so matching an inherited index retains correlation without
331        // treating local witnesses or later synthetic head slots as inputs.
332        self.scope_work(1)?;
333        let key_uses = match &element.key {
334            AggregateKey::Tuple(terms) => {
335                self.scope_work(terms.len())?;
336                terms.contains(&Term::Variable(variable))
337            }
338            AggregateKey::Atom(atom) => self.pattern_uses(*atom, variable)?,
339        };
340        if key_uses {
341            return Ok(true);
342        }
343        for literal in &element.condition {
344            if self.literal_uses(literal, variable)? {
345                return Ok(true);
346            }
347        }
348        Ok(false)
349    }

351    fn literal_uses(
352        &mut self,
353        literal: &LiteralIr,
354        variable: usize,
355    ) -> Result<bool, FormulaFailure> {
356        self.scope_work(1)?;
357        Ok(match literal {
358            LiteralIr::Atom(_, atom) => self.pattern_uses(*atom, variable)?,
359            LiteralIr::PatternAtom(pattern) => {
360                let flat = self.source.pattern_ref(
361                    pattern.atom,
362                    self.limits,
363                    self.counters,
364                    self.location,
365                )?;
366                let pattern = pattern.bind(flat);
367                self.budget.charge(
368                    ExpansionResource::TermWork,
369                    pattern.node_count() as u128,
370                    self.location,
371                )?;
372                pattern.slots().any(|slot| slot == variable)
373            }
374            LiteralIr::ProjectedAtom(_, projection) => {
375                self.projection_uses(projection, variable)?
376            }
377            LiteralIr::Compare(left, _, right)
378            | LiteralIr::ArgumentCheck {
379                captured: left,
380                value: right,
381            } => self.expression_uses(left, variable)? || self.expression_uses(right, variable)?,
382            LiteralIr::TupleCompare(left, _, right) => {
383                for term in left.iter().chain(right) {
384                    if self.expression_uses(term, variable)? {
385                        return Ok(true);
386                    }
387                }
388                false
389            }
390            LiteralIr::Guard(guard) | LiteralIr::HeadGuard(guard) => {
391                for expression in guard.expressions() {
392                    if self.expression_uses(expression, variable)? {
393                        return Ok(true);
394                    }
395                }
396                false
397            }
398            LiteralIr::Conditional(conditional) => {
399                if self.consequent_uses(&conditional.consequent, variable)? {
400                    return Ok(true);
401                }
402                for literal in &conditional.condition {
403                    if self.literal_uses(literal, variable)? {
404                        return Ok(true);
405                    }
406                }
407                false
408            }
409            LiteralIr::Aggregate(aggregate) => {
410                for guard in &aggregate.guards {
411                    if self.expression_uses(&guard.bound, variable)? {
412                        return Ok(true);
413                    }
414                }
415                // Whether the aggregate supplies an output does not change
416                // its element reads. In particular, a comparison aggregate
417                // can consume another aggregate's completed proposal.
418                for element in &aggregate.elements {
419                    if self.element_uses(element, variable)? {
420                        return Ok(true);
421                    }
422                }
423                false
424            }
425            LiteralIr::Bind { target, value } => {
426                *target == variable || self.expression_uses(value, variable)?
427            }
428            LiteralIr::Range {
429                target,
430                lower,
431                upper,
432                ..
433            } => {
434                *target == variable
435                    || self.expression_uses(lower, variable)?
436                    || self.expression_uses(upper, variable)?
437            }
438        })
439    }

441    /// Visit every local alternative before the enclosing condition scope is read.
442    fn consequent_uses(
443        &mut self,
444        consequent: &Consequent,
445        variable: usize,
446    ) -> Result<bool, FormulaFailure> {
447        let mut uses = false;
448        match consequent {
449            Consequent::Atoms(_, alternatives) => {
450                for alternative in alternatives {
451                    uses |= match &alternative.operand {
452                        ConsequentOperand::Atom(atom) => self.pattern_uses(*atom, variable)?,
453                        ConsequentOperand::Projection(projection) => {
454                            self.projection_uses(projection, variable)?
455                        }
456                    };
457                    for literal in &alternative.bindings {
458                        uses |= self.literal_uses(literal, variable)?;
459                    }
460                }
461            }
462            Consequent::Guards(alternatives) => {
463                for alternative in alternatives {
464                    for expression in alternative.guard.expressions() {
465                        uses |= self.expression_uses(expression, variable)?;
466                    }
467                    for literal in &alternative.bindings {
468                        uses |= self.literal_uses(literal, variable)?;
469                    }
470                }
471            }
472            Consequent::Guard(guard) => {
473                for expression in guard.expressions() {
474                    uses |= self.expression_uses(expression, variable)?;
475                }
476            }
477        }
478        Ok(uses)
479    }
