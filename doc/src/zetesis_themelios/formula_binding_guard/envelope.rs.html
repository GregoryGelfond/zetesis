1//! Directed finite integer envelopes for an existing binding scope.
2//!
3//! Each inequality has the mathematical form `sum(a_i * X_i) <= bound`.
4//! An endpoint is proposed only from the required endpoints of every other
5//! nonzero term. Each endpoint is installed once; at most twice the number of
6//! variables can be installed. This is a finite coverage analysis, not integer
7//! constraint solving. The original whole guard filters the proposed rows.
8

40pub(super) fn binding<'a>(
41    literals: impl Iterator<Item = &'a LiteralIr>,
42    variables: usize,
43    safe: &BTreeSet<usize>,
44    source: &crate::formula_support::components::Admission<'_>,
45    budget: &mut Budget,
46    work: GroundingWork<'_>,
47) -> Result<Option<(usize, i32, i32)>, FormulaFailure> {
48    let GroundingWork {
49        limits,
50        counters,
51        location,
52    } = work;
53    let mut capture = Capture {
54        reader: Reader {
55            source,
56            limits,
57            counters,
58            variables,
59            budget,
60            location,
61        },
62        inequalities: Vec::new(),
63        capacity: None,
64    };
65    for literal in literals {
66        capture.literal(literal)?;
67    }
68    capture.reader.budget.charge(
69        ExpansionResource::ScalarBytes,
70        2 * variables as u128 * std::mem::size_of::<Interval>() as u128,
71        location,
72    )?;
73    let mut intervals = vec![Interval::default(); variables];
74    let mut proposals = vec![Interval::default(); variables];
75    // Each successful round installs an absent endpoint. Numeric tightening
76    // never controls termination, even on contradictory cyclic comparisons.
77    for _ in 0..variables.saturating_mul(2) {
78        capture.reader.work(variables as u128)?;
79        proposals.fill(Interval::default());
80        for inequality in &capture.inequalities {
81            for (target, proposal) in proposals.iter_mut().enumerate() {
82                capture.reader.work(1)?;
83                match inequality.derive(target, &intervals, &mut capture.reader) {
84                    Ok(Some(endpoint)) => {
85                        // All proposals in a round read the same earlier
86                        // endpoints. Select the strongest simultaneous bound,
87                        // independently of source comparison order.
88                        proposal.include(endpoint);
89                    }
90                    Ok(None) => {}
91                    Err(error) if capacity_failure(&error) => capture.capacity = Some(error),
92                    Err(error) => return Err(error),
93                }
94            }
95        }
96        let mut progress = false;
97        for (interval, proposal) in intervals.iter_mut().zip(&proposals) {
98            capture.reader.work(1)?;
99            if let Some(lower) = proposal.lower {
100                interval.lower = Some(lower);
101                progress = true;
102            }
103            if let Some(upper) = proposal.upper {
104                interval.upper = Some(upper);
105                progress = true;
106            }
107        }
108        if !progress {
109            break;
110        }
111    }
112    for (target, interval) in intervals.iter().enumerate() {
113        capture.reader.work(1)?;
114        if !safe.contains(&target)
115            && let Some((lower, upper)) = interval.range()
116        {
117            return Ok(Some((target, lower, upper)));
118        }
119    }
120    if let Some(error) = capture.capacity {
121        return Err(error);
122    }
123    Ok(None)
124}

126struct Capture<'a, 'source> {
127    reader: Reader<'a, 'source>,
128    inequalities: Vec<Inequality>,
129    capacity: Option<FormulaFailure>,
130}
131
132impl Capture<'_, '_> {
133    fn literal(&mut self, literal: &LiteralIr) -> Result<(), FormulaFailure> {
134        self.reader.work(1)?;
135        match literal {
136            LiteralIr::Compare(left, relation, right) => self.comparison(left, *relation, right)?,
137            LiteralIr::Guard(Guard::Comparisons {
138                negation: DefaultNegation::None | DefaultNegation::NotNot,
139                comparisons,
140            }) => {
141                for comparison in comparisons {
142                    if let GuardComparison::Scalar(left, relation, right) = comparison {
143                        self.comparison(left, *relation, right)?;
144                    }
145                }
146            }
147            LiteralIr::Bind { target, value } => self.generated(*target, value, value)?,
148            LiteralIr::Range {
149                target,
150                lower,
151                upper,
152                binder: true,
153            } => self.generated(*target, lower, upper)?,
154            _ => {}
155        }
156        Ok(())
157    }
158
159    fn comparison(
160        &mut self,
161        left: &Expression,
162        relation: Relation,
163        right: &Expression,
164    ) -> Result<(), FormulaFailure> {
165        if relation == Relation::Neq {
166            return Ok(());
167        }
168        let result = self.normalized(left, relation, right);
169        match result {
170            Err(error) if capacity_failure(&error) => self.capacity = Some(error),
171            other => other?,
172        }
173        Ok(())
174    }
175
176    fn normalized(
177        &mut self,
178        left: &Expression,
179        relation: Relation,
180        right: &Expression,
181    ) -> Result<(), FormulaFailure> {
182        let left = self.reader.expression(left)?;
183        let right = self.reader.expression(right)?;
184        let (Some(left), Some(right)) = (left, right) else {
185            return Ok(());
186        };
187        match relation {
188            Relation::Le | Relation::Lt => {
189                self.inequality(&left, &right, relation == Relation::Lt)?;
190            }
191            Relation::Ge | Relation::Gt => {
192                self.inequality(&right, &left, relation == Relation::Gt)?;
193            }
194            Relation::Eq => {
195                self.inequality(&left, &right, false)?;
196                self.inequality(&right, &left, false)?;
197            }
198            Relation::Neq => {}
199        }
200        Ok(())
201    }
202
203    fn inequality(
204        &mut self,
205        left: &Affine,
206        right: &Affine,
207        strict: bool,
208    ) -> Result<(), FormulaFailure> {
209        self.reader.work(1 + self.reader.variables as u128)?;
210        self.reader.budget.charge(
211            ExpansionResource::ScalarBytes,
212            std::mem::size_of::<Inequality>() as u128
213                + self.reader.variables as u128 * std::mem::size_of::<i64>() as u128,
214            self.reader.location,
215        )?;
216        let difference = self.reader.combine(left, right, -1)?;
217        let bound = coefficient(
218            -i128::from(difference.constant) - i128::from(strict),
219            self.reader.location,
220        )?;
221        self.inequalities.push(Inequality {
222            coefficients: difference.coefficients,
223            bound,
224        });
225        Ok(())
226    }
227
228    fn generated(
229        &mut self,
230        target: usize,
231        lower: &Expression,
232        upper: &Expression,
233    ) -> Result<(), FormulaFailure> {
234        // Existing generators retain their own dependency contract. Only closed
235        // numeric results are reused here as static endpoint evidence; a bound
236        // relational variable is not a closed source endpoint.
237        if lower.inputs().next().is_some() || upper.inputs().next().is_some() {
238            return Ok(());
239        }
240        let lower = self.reader.expression(lower)?;
241        let upper = self.reader.expression(upper)?;
242        let (Some(lower), Some(upper)) = (lower, upper) else {
243            return Ok(());
244        };
245        let variable = self.reader.variable(target)?;
246        self.inequality(&lower, &variable, false)?;
247        self.inequality(&variable, &upper, false)
248    }
249}

251impl Inequality {
252    fn derive(
253        &self,
254        target: usize,
255        intervals: &[Interval],
256        reader: &mut Reader<'_, '_>,
257    ) -> Result<Option<Endpoint>, FormulaFailure> {
258        let coefficient = self.coefficients[target];
259        if coefficient == 0 {
260            return Ok(None);
261        }
262        let lower = coefficient < 0;
263        if (if lower {
264            intervals[target].lower
265        } else {
266            intervals[target].upper
267        })
268        .is_some()
269        {
270            return Ok(None);
271        }
272        let mut remainder = i128::from(self.bound);
273        for (index, (&weight, interval)) in self.coefficients.iter().zip(intervals).enumerate() {
274            reader.work(1)?;
275            if index == target || weight == 0 {
276                continue;
277            }
278            let Some(endpoint) = (if weight > 0 {
279                interval.lower
280            } else {
281                interval.upper
282            }) else {
283                return Ok(None);
284            };
285            remainder = remainder
286                .checked_sub(i128::from(weight) * i128::from(endpoint))
287                .ok_or_else(|| bound_capacity(reader.location))?;
288        }
289        let quotient = inverse_bound(remainder, coefficient, reader.location)?;
290        let endpoint = if lower {
291            quotient.clamp(i128::from(i32::MIN), i128::from(i32::MAX) + 1)
292        } else {
293            quotient.clamp(i128::from(i32::MIN) - 1, i128::from(i32::MAX))
294        };
295        let endpoint =
296            i64::try_from(endpoint).expect("clamped i32 endpoint or adjacent empty boundary");
297        Ok(Some(if lower {
298            Endpoint::Lower(endpoint)
299        } else {
300            Endpoint::Upper(endpoint)
301        }))
302    }
303}

305/// Solve `coefficient * x <= remainder` over mathematical integers. The caller
306/// supplies a nonzero coefficient and chooses the indicated lower/upper side.
307fn inverse_bound(
308    remainder: i128,
309    coefficient: i64,
310    location: ProgramSite,
311) -> Result<i128, FormulaFailure> {
312    let divisor = i128::from(coefficient);
313    if coefficient < 0 {
314        // ceil(r / a), a < 0, equals -floor(r / -a).
315        remainder
316            .div_euclid(-divisor)
317            .checked_neg()
318            .ok_or_else(|| bound_capacity(location))
319    } else {
320        Ok(remainder.div_euclid(divisor))
321    }
322}

324impl Interval {
325    fn include(&mut self, endpoint: Endpoint) {
326        match endpoint {
327            Endpoint::Lower(value) => {
328                self.lower = Some(self.lower.map_or(value, |previous| previous.max(value)));
329            }
330            Endpoint::Upper(value) => {
331                self.upper = Some(self.upper.map_or(value, |previous| previous.min(value)));
332            }
333        }
334    }
335
336    fn range(self) -> Option<(i32, i32)> {
337        let (lower, upper) = (self.lower?, self.upper?);
338        Some(if lower > upper {
339            (1, 0)
340        } else {
341            (
342                i32::try_from(lower).expect("nonempty lower endpoint is i32"),
343                i32::try_from(upper).expect("nonempty upper endpoint is i32"),
344            )
345        })
346    }
347}
348
349fn capacity_failure(error: &FormulaFailure) -> bool {
350    matches!(
351        error,
352        FormulaFailure::Limit {
353            resource: FormulaResource::BindingCoefficientBits | FormulaResource::BindingBoundBits,
354            ..
355        }
356    )
357}
358
359fn bound_capacity(location: ProgramSite) -> FormulaFailure {
360    FormulaFailure::Limit {
361        resource: FormulaResource::BindingBoundBits,
362        limit: u128::from(i128::BITS),
363        observed: u128::from(i128::BITS) + 1,
364        location,
365    }
366}

368#[cfg(test)]
369mod tests {
370    use super::inverse_bound;
371    use crate::ProgramSite;
372    use crate::{FormulaFailure, FormulaResource};
373    use themelios_base::source::SourceId;
374    use themelios_base::span::{ByteOffset, Span};
375
376    fn location() -> ProgramSite {
377        ProgramSite::source(themelios_base::span::Location {
378            source: SourceId::new(1),
379            span: Span::empty(ByteOffset::new(0)),
380        })
381    }
382
383    #[test]
384    fn inverse_bounds_match_integer_inequalities() {
385        for coefficient in [-7_i64, -3, -1, 1, 3, 7] {
386            for remainder in -20_i128..=20 {
387                let bound = inverse_bound(remainder, coefficient, location()).unwrap();
388                for value in -24_i128..=24 {
389                    assert_eq!(
390                        i128::from(coefficient) * value <= remainder,
391                        if coefficient < 0 {
392                            value >= bound
393                        } else {
394                            value <= bound
395                        }
396                    );
397                }
398            }
399        }
400    }
401
402    #[test]
403    fn unrepresentable_inverse_bound_reports_capacity() {
404        assert!(matches!(
405            inverse_bound(i128::MIN, -1, location()),
406            Err(FormulaFailure::Limit {
407                resource: FormulaResource::BindingBoundBits,
408                limit: 128,
409                observed: 129,
410                ..
411            })
412        ));
413    }
