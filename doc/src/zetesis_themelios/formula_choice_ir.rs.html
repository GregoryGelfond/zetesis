1//! Choice keys distinguish counted entries after local pool expansion.
2
3use themelios_program::program::{Choice, Identity};
4
5use crate::FormulaFailure;
6use crate::formula_ir::{Compiler, Element, HeadElementKey, LocalFamily, Variables};
7
8impl Compiler<'_> {
9    pub(super) fn choice_elements(
10        &mut self,
11        choice: &Choice,
12        source: Option<&Choice>,
13        variables: &Variables,
14    ) -> Result<Vec<Element>, FormulaFailure> {
15        let mut elements = Vec::new();
16        // The whole-rule pool rewrite reconstructs local head alternatives.
17        // Compile each original counted entry's local product independently.
18        // Every product alternative receives its occurrence key before local
19        // variable substitutions; equal alternatives remain distinct entries.
20        for (index, element) in source.unwrap_or(choice).elements().enumerate() {
21            let family = LocalFamily(index);
22            let identity = element.get().identity();
23            if identity == Identity::ByOccurrence {
24                self.dependency_projection = true;
25            }
26            for literal in self.literal_alternatives(element.get().literal())? {
27                for alternative in self.condition_alternatives(element.get().condition())? {
28                    let mut local = variables.clone();
29                    self.head_global_literal(&literal, &mut local)?;
30                    let mut condition = self.condition(&alternative, &mut local)?;
31                    let (head, body_variables) =
32                        self.element_head(&literal, &mut local, &mut condition)?;
33                    let key = match identity {
34                        Identity::ByContent => HeadElementKey::Atom,
35                        Identity::ByOccurrence => HeadElementKey::Occurrence(elements.len()),
36                    };
37                    self.variable_limit(&local)?;
38                    local.safety(self.location)?;
39                    elements.push(Element {
40                        family,
41                        key,
42                        head,
43                        condition,
44                        body_variables,
45                        variables: local.count,
46                    });
47                }
48            }
49        }
50        Ok(elements)
51    }
52}
