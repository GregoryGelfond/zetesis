1//! Bounded alternative syntax for compilation and dependency-only analysis.
2//!
3//! A conjunction of projected conditional literals is deliberately not asserted
4//! equivalent to the runtime inner disjunction. Its purpose is to retain every
5//! signature, polarity and parsed atom origin without passing pools upstream.
6

18impl Compiler<'_> {
19    pub(super) fn conditional_atoms(&mut self, atom: &Atom) -> Result<Vec<Atom>, FormulaFailure> {
20        let (count, admitted) = atom_width(atom);
21        let mut scan = Payload::default();
22        scan.visit_atom(atom);
23        self.budget
24            .charge(ExpansionResource::TermWork, scan.nodes, self.location)?;
25        if admitted != scan.pools {
26            return Err(unsupported(ProfileFeature::Term, self.location).into());
27        }
28        // Values limits bound source alternatives, including duplicate values.
29        // Byte accounting reserves four term/text copy equivalents and atom
30        // carriers for source selection; the independent analysis ceiling bounds nodes.
31        self.budget
32            .charge(ExpansionResource::Values, count, self.location)?;
33        self.budget.charge(
34            ExpansionResource::ScalarBytes,
35            count
36                .saturating_mul(scan.bytes.saturating_mul(4) + std::mem::size_of::<Atom>() as u128),
37            self.location,
38        )?;
39        self.budget.charge(
40            ExpansionResource::TermWork,
41            count.saturating_mul(scan.nodes),
42            self.location,
43        )?;
44        ceiling(
45            FormulaResource::AnalysisNodes,
46            count.saturating_mul(scan.nodes),
47            self.limits.max_analysis_nodes as u128,
48            self.location,
49        )?;
50        let count = usize::try_from(count).expect("finite value ceiling proves usize");
51        Ok((0..count)
52            .map(|position| select_atom(atom.clone(), position))
53            .collect())
54    }

56    pub(super) fn conditional_projection(
57        &mut self,
58        statement: &WithProvenance<Statement>,
59        projection_nodes: &mut u128,
60    ) -> Result<Vec<WithProvenance<Statement>>, FormulaFailure> {
61        let mut payload = Payload::default();
62        payload.visit_statement(statement.get());
63        if payload.pools == 0 {
64            return Ok(vec![statement.clone()]);
65        }
66        let mut width = Width(1);
67        width.visit_statement(statement.get());
68        let count = width.0;
69        self.budget
70            .charge(ExpansionResource::TermWork, payload.nodes, self.location)?;
71        self.budget
72            .charge(ExpansionResource::Values, count, self.location)?;
73        *projection_nodes =
74            projection_nodes.saturating_add(count.saturating_mul(payload.nodes + 1));
75        ceiling(
76            FormulaResource::AnalysisNodes,
77            *projection_nodes,
78            self.limits.max_analysis_nodes as u128,
79            self.location,
80        )?;
81        self.budget.charge(
82            ExpansionResource::ScalarBytes,
83            count.saturating_mul(payload.bytes.saturating_mul(4)),
84            self.location,
85        )?;
86        self.budget.charge(
87            ExpansionResource::TermWork,
88            count.saturating_mul(payload.nodes.saturating_mul(4)),
89            self.location,
90        )?;
91        self.dependency_projection = true;
92        let count = usize::try_from(count).expect("bounded analysis alternatives");
93        let mut result = Vec::with_capacity(count);
94        for position in 0..count {
95            let mut projection = Projection { position };
96            let rewritten = rewrite(Program::of_nodes([statement.clone()]), &mut projection);
97            debug_assert_eq!(projection.position, 0);
98            result.push(
99                rewritten
100                    .statements()
101                    .next()
102                    .expect("projection retains statement")
103                    .clone(),
104            );
105        }
106        Ok(result)
107    }
108}

110/// Cartesian syntax choices preserve every signed signature and argument arity.
111/// Local quantifiers are deliberately not interpreted by this analysis view.
112struct Width(u128);
113impl Visit for Width {
114    fn visit_atom(&mut self, atom: &Atom) {
115        if let Arguments::Pooled(alternatives) = &atom.arguments {
116            self.0 = self.0.saturating_mul(alternatives.len() as u128);
117        }
118        for term in atom.argument_terms().flat_map(Term::subterms) {
119            self.visit_term(term);
120        }
121    }
122    fn visit_term(&mut self, term: &Term) {
123        if let Term::Pool(items) = term {
124            self.0 = self.0.saturating_mul(items.len() as u128);
125        }
126    }
127}
128struct Projection {
129    position: usize,
130}
131impl Projection {
132    fn take(&mut self, width: usize) -> usize {
133        let selected = self.position % width;
134        self.position /= width;
135        selected
136    }
137}
138impl Rewrite for Projection {
139    fn tag(&self) -> TransformTag {
140        TransformTag::new("zetesis-pool-dependencies")
141    }
142    fn rewrite_term(&mut self, term: Term) -> Term {
143        if let Term::Pool(items) = &term {
144            items[self.take(items.len())].clone()
145        } else {
146            term
147        }
148    }
149    fn rewrite_atom(&mut self, mut atom: Atom) -> Atom {
150        // Rewrite's atom hook owns term traversal, as in the upstream default.
151        atom.arguments = match atom.arguments {
152            Arguments::Single(terms) => Arguments::Single(
153                terms
154                    .into_iter()
155                    .map(|term| term.fold(|parts| self.rewrite_term(Term::from(parts))))
156                    .collect(),
157            ),
158            Arguments::Pooled(alternatives) => {
159                let alternatives: Vec<Vec<Term>> = alternatives
160                    .into_iter()
161                    .map(|terms| {
162                        terms
163                            .into_iter()
164                            .map(|term| term.fold(|parts| self.rewrite_term(Term::from(parts))))
165                            .collect()
166                    })
167                    .collect();
168                Arguments::Single(alternatives[self.take(alternatives.len())].clone())
169            }
170        };
171        atom
172    }
173}

175#[derive(Default)]
176struct Payload {
177    nodes: u128,
178    bytes: u128,
179    pools: usize,
180}
181impl Visit for Payload {
182    fn visit_atom(&mut self, atom: &Atom) {
183        self.nodes += 1;
184        self.bytes += atom.name.as_str().len() as u128;
185        self.pools += usize::from(matches!(atom.arguments, Arguments::Pooled(_)));
186        for term in atom.argument_terms().flat_map(Term::subterms) {
187            self.visit_term(term);
188        }
189    }
190    fn visit_term(&mut self, term: &Term) {
191        self.nodes += 1;
192        self.bytes += std::mem::size_of::<Term>() as u128;
193        self.pools += usize::from(matches!(term, Term::Pool(_)));
194        self.bytes += match term {
195            Term::Function { name, .. } => name.as_str().len() as u128,
196            Term::Symbolic(symbol) => crate::structural_value::symbol_bytes(symbol),
197            Term::Variable(themelios_program::term::Variable::Named(name)) => {
198                name.as_str().len() as u128
199            }
200            _ => 0,
201        };
202    }
