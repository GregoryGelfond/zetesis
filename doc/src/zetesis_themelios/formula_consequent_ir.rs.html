1//! Finite consequent alternatives share value evaluation and structural matching.
2//!
3//! Relational witnesses retain their complete captured atom. Generated values
4//! use the shared interval fold; each alternative keeps its own local scope.
5
6use crate::formula_support::components::{Pattern, Term as CoreTerm};
7use themelios_program::program::{Arguments, Atom, DefaultNegation, Relation};
8use themelios_program::term::{Term, Variable};
9
10use crate::formula::ceiling;
11use crate::formula_conditional_ir::{Alternative, ConsequentOperand};
12use crate::formula_ir::{Compiler, Expression, LiteralIr, Operation, Variables};
13use crate::{FormulaFailure, FormulaResource};
14
15impl Compiler<'_> {
16    pub(super) fn consequent_alternative(
17        &mut self,
18        atom: &Atom,
19        negation: DefaultNegation,
20        condition: &Variables,
21    ) -> Result<Alternative, FormulaFailure> {
22        let Arguments::Single(arguments) = &atom.arguments else {
23            unreachable!("conditional pools were selected")
24        };
25        let work = arguments.iter().flat_map(Term::subterms).count() as u128;
26        let mut local =
27            self.alternative_scope(work, std::mem::size_of::<Alternative>(), condition)?;
28        let mut bindings = Vec::new();
29        let missing = arguments
30            .iter()
31            .flat_map(Term::subterms)
32            .any(|term| match term {
33                Term::Variable(Variable::Anonymous) => true,
34                Term::Variable(Variable::Named(name)) => !local.named.contains_key(name.as_str()),
35                _ => false,
36            });
37        let operand = if let Some(LiteralIr::ProjectedAtom(_, projection)) =
38            self.projected_atom(atom, negation, &mut local, &mut bindings)?
39        {
40            ConsequentOperand::Projection(projection)
41        } else if missing && negation == DefaultNegation::None {
42            ConsequentOperand::Atom(self.positive_atom_key(atom, &mut local, &mut bindings)?)
43        } else {
44            ConsequentOperand::Atom(self.consequent_atom(
45                atom,
46                arguments,
47                &mut local,
48                &mut bindings,
49            )?)
50        };
51        self.bindings(&mut bindings, &mut local)?;
52        self.variable_limit(&local)?;
53        local.safety(self.location)?;
54        Ok(Alternative {
55            operand,
56            bindings,
57            variables: local.count,
58        })
59    }
60
61    fn consequent_atom(
62        &mut self,
63        atom: &Atom,
64        arguments: &[Term],
65        local: &mut Variables,
66        bindings: &mut Vec<LiteralIr>,
67    ) -> Result<Pattern, FormulaFailure> {
68        ceiling(
69            FormulaResource::Arity,
70            arguments.len() as u128,
71            self.options.core_limits.max_predicate_arity as u128,
72            self.location,
73        )?;
74        let mut terms = Vec::new();
75        for term in arguments {
76            if matches!(term, Term::Symbolic(_) | Term::Variable(_)) {
77                terms.push(self.domain_term(term, local)?);
78            } else {
79                let value = self.ranged_expression(term, local, bindings)?;
80                let target = self.consequent_slot(local)?;
81                bindings.push(LiteralIr::Compare(
82                    Expression {
83                        nodes: vec![Operation::Variable(target)],
84                    },
85                    Relation::Eq,
86                    value,
87                ));
88                terms.push(CoreTerm::Variable(target));
89            }
90        }
91        let predicate = self.predicate(
92            atom.name.as_str(),
93            arguments.len(),
94            crate::coherence::core_sign(atom.sign),
95        )?;
96        self.pattern_from_parts(predicate, &terms)
97    }
98
99    pub(super) fn consequent_slot(
100        &self,
101        variables: &mut Variables,
102    ) -> Result<usize, FormulaFailure> {
103        ceiling(
104            FormulaResource::Variables,
105            variables.count as u128 + 1,
106            self.options.core_limits.max_variables_per_template as u128,
107            self.location,
108        )?;
109        Ok(variables.slot(&Variable::Anonymous))
110    }
