18/// Independent limits for optional capture, partition selection and emission.
19/// Zero never means unlimited; ordinary source grounding has separate budgets.
20#[derive(Clone, Copy, Debug)]
21pub struct CountPlanLimits {
22    /// Retained applicable grounded count groups.
23    pub max_groups: usize,
24    /// Total retained semantic head occurrences across groups.
25    pub max_members: usize,
26    /// Copied source origins across captured groups and emitted consequences.
27    pub max_origins: usize,
28    /// Attempted global-group partitions, including unsuccessful covers.
29    pub max_attempts: usize,
30    /// Conservative cumulative logical allocation payload across optional work.
31    /// Existing source/theory/atom storage, allocator metadata and excess
32    /// allocator capacity are excluded. Source validation transfers a stack-only
33    /// certificate; optional capture retains only members and origins.
34    pub max_bytes: u64,
35    /// Cumulative capture, matching, partition and emission operations.
36    pub max_work: u64,
37    /// Per-attempt complete partition validation, also capped by remaining work.
38    pub partition: partition::Limits,
39    /// Final candidate restriction storage, independent of the original theory.
40    pub theory: zetesis_ferraris::AdmissionLimits,
41    /// Per-consequence exact count lowering, capped by remaining work/nodes.
42    pub aggregate: AggregateLimits,
43}
44impl Default for CountPlanLimits {
45    fn default() -> Self {
46        Self {
47            max_groups: 1_024,
48            max_members: 65_536,
49            max_origins: 16_384,
50            max_attempts: 1_024,
51            max_bytes: 64 * 1024 * 1024,
52            max_work: 10_000_000,
53            partition: partition::Limits::default(),
54            theory: zetesis_ferraris::AdmissionLimits::default(),
55            aggregate: AggregateLimits::default(),
56        }
57    }
58}

62pub enum CountPlanResource {

66    Members,

77/// Incomplete optional work; no variant denotes source refusal or logical UNSAT.
78#[derive(Clone, Copy, Debug, PartialEq, Eq)]
79pub enum CountPlanFailureKind {
80    /// An inclusive optional resource ceiling was exceeded.
81    Limit(CountPlanResource),
82    /// Checked dimensions or accounting overflowed.
83    Overflow,
84    /// Source-produced node/atom evidence failed its internal correspondence.
85    SourceMapping,
86    /// Independent partition validation stopped.
87    Partition(partition::Error),
88    /// Existing exact count translation stopped.
89    Aggregate(zetesis_ferraris::AggregateError),
90    /// Independent consequence theory admission stopped.
91    Theory(zetesis_ferraris::AdmissionError),
92    /// Optional control or fallible allocation stopped planning.
93    Stopped(Stop),
94}

96/// Completed or attempted optional work, independent of source-grounding work.
97#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
98pub struct CountPlanStatistics {
99    /// Charged operations, including an incomplete prefix.
100    pub work: u64,
101    /// Captured applicable groups.
102    pub groups: usize,
103    /// Captured head occurrences.
104    pub members: usize,
105    /// Copied source locations.
106    pub origins: usize,
107    /// Attempted complete covers.
108    pub attempts: usize,
109    /// Local lower bounds after complete consequence emission.
110    pub consequences: usize,
111    /// Cumulative conservative payload admitted against the optional allowance,

118/// Located optional failure with its accounted work prefix.

129        self.kind
130    }

132    #[must_use]
133    pub const fn location(&self) -> Location {
134        self.location
135    }
136    /// Optional work attempted before the failure.
137    #[must_use]

140    }

144        write!(f, "optional count planning stopped: {:?}", self.kind)
145    }

149/// A source-derived candidate restriction with a closed construction boundary.
150///
151/// The initial profile uses complete ordinary-choice and bijective count-head
152/// groups whose coalesced eligibility is canonical truth. An integer global
153/// lower bound and a disjoint, complete partition of upper-bounded groups imply
154/// stronger local lower bounds. Capacity groups must be unconditional or share
155/// exactly the global activation formula; every consequence keeps that formula.
156/// Disequality supplies no interval premise. A group with any nonnumeric logical
157/// bound supplies no numeric certificate; independent numeric groups remain
158/// eligible. Original source guard evaluation and numeric limits remain unchanged.
159///
160/// Discovery greedily visits retained grounded groups without backtracking.
161/// It can miss a valid alternative cover. With G groups, M retained member
162/// occurrences, R original roots and A original atoms, discovery charges at most
163/// O(G M² + G² R + G A) work: complete membership probes and root evidence scans
164/// are charged conservatively. Emission uses the bounded threshold compiler.
165/// Temporary storage includes retained groups and coverage maps, a copied
166/// activation DAG prefix, origins and the independently owned restriction.
167/// Ordinary grounding retains none of these optional descriptors. Logical work
168/// and conservative cumulative payload have separate inclusive ceilings.
169///
170/// The original theory entails this restriction under the retained source
171/// premises. This implementation obligation is tested, not a Rust refinement
172/// theorem. The plan proves neither stable membership nor exhaustive search.
173#[derive(Debug)]
174pub struct CountPlan {
175    original: Theory,
176    restriction: Theory,
177    origins: Vec<Location>,
178    statistics: CountPlanStatistics,
179}
180impl CountPlan {
181    /// Exact original reduct subject. Its immutable storage is shared.
182    #[must_use]
183    pub fn original_theory(&self) -> &Theory {
184        &self.original
185    }
186    /// Independent original-candidate consequence theory, using the original
187    /// semantic atom indices. Apply before proposal, never as a reduct subject.
188    #[must_use]
189    pub fn restriction(&self) -> &Theory {
190        &self.restriction
191    }
192    /// Source count premises supporting the emitted consequences.
193    #[must_use]
194    pub fn origins(&self) -> &[Location] {
195        &self.origins
196    }
197    /// Completed optional planning accounting.
198    #[must_use]
199    pub const fn statistics(&self) -> CountPlanStatistics {
200        self.statistics
201    }
202    /// Number of strictly stronger guarded local lower bounds.
203    #[must_use]
204    pub const fn consequence_count(&self) -> usize {
205        self.statistics.consequences
206    }
207}

209/// Borrowed outcome of independently requested source-count planning.
210#[derive(Clone, Copy, Debug)]
211pub enum CountPlanStatus<'a> {
212    /// Ordinary grounding retained no optional descriptors or planning budget.
213    NotRequested,
214    /// The selected bounded planning policy emitted no stronger restriction.
215    /// This does not establish that no other cover or consequence exists.
216    NoPlan(CountPlanStatistics),
217    /// Complete source-derived candidate restriction.
218    Ready(&'a CountPlan),
219    /// Optional work stopped; the successfully admitted original remains usable.
220    Incomplete(&'a CountPlanFailure),
221}
222
223#[derive(Debug)]
224pub(crate) enum Outcome {
225    NotRequested,
226    NoPlan(CountPlanStatistics),
227    Ready(CountPlan),
228    Incomplete(CountPlanFailure),
229}
230impl Outcome {
231    pub(crate) fn view(&self) -> CountPlanStatus<'_> {
232        match self {
233            Self::NotRequested => CountPlanStatus::NotRequested,
234            Self::NoPlan(stats) => CountPlanStatus::NoPlan(*stats),
235            Self::Ready(plan) => CountPlanStatus::Ready(plan),
236            Self::Incomplete(error) => CountPlanStatus::Incomplete(error),
237        }
238    }
239}

247struct Work {
248    limits: CountPlanLimits,
249    cancellation: Cancellation,
250    statistics: CountPlanStatistics,
251    location: Location,
252}
253impl Work {
254    fn poll(&self) -> Result<(), CountPlanFailureKind> {
255        self.cancellation
256            .poll()
257            .map_err(CountPlanFailureKind::Stopped)
258    }
259    fn charge(&mut self, count: u64) -> Result<(), CountPlanFailureKind> {
260        self.poll()?;
261        self.record(count)
262    }
263    fn record(&mut self, count: u64) -> Result<(), CountPlanFailureKind> {
264        if count > self.limits.max_work - self.statistics.work {
265            return Err(CountPlanFailureKind::Limit(CountPlanResource::Work));
266        }
267        self.statistics.work += count;
268        Ok(())
269    }
270    fn payload(&mut self, bytes: u64) -> Result<(), CountPlanFailureKind> {
271        let total = self
272            .statistics
273            .storage_bytes
274            .checked_add(bytes)
275            .ok_or(CountPlanFailureKind::Overflow)?;
276        if total > self.limits.max_bytes {
277            return Err(CountPlanFailureKind::Limit(CountPlanResource::Bytes));
278        }
279        self.statistics.storage_bytes = total;
280        Ok(())
281    }
282    fn vector<T>(&mut self, count: usize) -> Result<Vec<T>, CountPlanFailureKind> {
283        self.poll()?;
284        self.payload(bytes::<T>(count)?)?;
285        let mut output = Vec::new();
286        output
287            .try_reserve_exact(count)
288            .map_err(|_| CountPlanFailureKind::Stopped(Stop::Allocation))?;
289        Ok(output)
290    }
291    fn failure(&self, kind: CountPlanFailureKind) -> CountPlanFailure {
292        CountPlanFailure {
293            kind,
294            location: self.location,
295            statistics: self.statistics,
296        }
297    }
298}
299fn bytes<T>(count: usize) -> Result<u64, CountPlanFailureKind> {
300    count
301        .checked_mul(size_of::<T>())
302        .and_then(|n| u64::try_from(n).ok())
303        .ok_or(CountPlanFailureKind::Overflow)
304}
305fn ceiling(
306    actual: usize,
307    maximum: usize,
308    resource: CountPlanResource,
309) -> Result<(), CountPlanFailureKind> {
310    if actual > maximum {
311        Err(CountPlanFailureKind::Limit(resource))
312    } else {
313        Ok(())
314    }
