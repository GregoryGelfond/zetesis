1//! Greedy source-group covers and strictly stronger conditional consequences.
2//!
3//! Each global group considers capacity groups in their retained grounding order,
4//! choosing a group only if disjoint from the current cover. There is no
5//! backtracking: an early large group can hide a better later cover. This finite
6//! discovery policy is incomplete; every emitted consequence remains sound.
7
8use crate::ProgramSite;
9use zetesis_ferraris::{Theory, partition};

11use super::capture::Group;
12use super::{
13    CountPlan, CountPlanFailureKind as Fault, CountPlanResource as Resource, Work, ceiling,
14};

16pub(super) struct Consequence {
17    pub location: ProgramSite,
18    pub body: usize,
19    pub members: Vec<usize>,
20    pub lower: usize,
21    pub origins: Vec<ProgramSite>,
22}

24pub(super) fn plan(
25    groups: &[Group],
26    theory: &Theory,
27    work: &mut Work,
28) -> Result<Option<CountPlan>, Fault> {
29    work.poll()?;
30    let mut consequences = Vec::new();
31    for (global_index, global) in groups.iter().enumerate() {
32        work.location = global.location;
33        work.charge(1)?;
34        if !global.unconditional_members || global.lower == 0 || global.members.len() < 2 {
35            continue;
36        }
37        validate_source(global, theory, work)?;
38        let next = work
39            .statistics
40            .attempts
41            .checked_add(1)
42            .ok_or(Fault::Overflow)?;
43        ceiling(next, work.limits.max_attempts, Resource::Attempts)?;
44        work.statistics.attempts = next;
45        let Some(indices) = cover(global_index, groups, theory, work)? else {
46            continue;
47        };
48        let plan = partition(global, &indices, groups, theory, work)?;
49        // The first profile emits only improved local bounds. An inconsistent
50        // cover is left to the original theory, rather than asserting falsum.
51        if plan.inconsistent() {
52            continue;
53        }
54        append_consequences(global, &indices, groups, &plan, &mut consequences, work)?;
55    }
56    if consequences.is_empty() {
57        work.poll()?;
58        return Ok(None);
59    }
60    let (restriction, origins) = super::emit::restriction(theory, &consequences, work)?;
61    work.statistics.consequences = consequences.len();
62    work.poll()?;
63    Ok(Some(CountPlan {
64        original: theory.clone(),
65        restriction,
66        origins,
67        statistics: work.statistics,
68    }))
69}

71fn validate_source(group: &Group, theory: &Theory, work: &mut Work) -> Result<(), Fault> {
72    work.charge(u64::try_from(theory.roots().len()).map_err(|_| Fault::Overflow)? + 1)?;
73    if group.body >= theory.nodes().len()
74        || group.bound_root >= theory.nodes().len()
75        || !theory.roots().contains(&group.asserted_root)
76    {
77        return Err(Fault::SourceMapping);
78    }
79    Ok(())
80}

82fn cover(
83    global_index: usize,
84    groups: &[Group],
85    theory: &Theory,
86    work: &mut Work,
87) -> Result<Option<Vec<usize>>, Fault> {
88    // Capture accepts members only through checked Node::Atom lookups in the
89    // grounder's atom catalog. That catalog only appends and transfers its
90    // insertion order unchanged into this final theory, so every member index
91    // is below theory.atom_count(), even if later grounding added more atoms.
92    // Source-formula correspondence still needs validate_source below.
93    let global = &groups[global_index];
94    let mut covered = work.vector::<bool>(theory.atom_count())?;
95    for _ in 0..theory.atom_count() {
96        work.charge(1)?;
97        covered.push(false);
98    }
99    let mut indices = work.vector(groups.len())?;
100    let mut count = 0_usize;
101    for (index, group) in groups.iter().enumerate() {
102        work.charge(1)?;
103        if !group.unconditional_members
104            || index == global_index
105            || group.upper == group.members.len()
106            || (group.body != 1 && group.body != global.body)
107        {
108            continue;
109        }
110        let mut fits = true;
111        for &atom in &group.members {
112            work.charge(u64::try_from(global.members.len()).map_err(|_| Fault::Overflow)? + 1)?;
113            if covered[atom] || global.members.binary_search(&atom).is_err() {
114                fits = false;
115                break;
116            }
117        }
118        if !fits {
119            continue;
120        }
121        validate_source(group, theory, work)?;
122        for &atom in &group.members {
123            work.charge(1)?;
124            covered[atom] = true;
125        }
126        count = count
127            .checked_add(group.members.len())
128            .ok_or(Fault::Overflow)?;
129        indices.push(index);
130    }
131    Ok((count == global.members.len() && indices.len() >= 2).then_some(indices))
132}
133fn partition(
134    global: &Group,
135    indices: &[usize],
136    groups: &[Group],
137    theory: &Theory,
138    work: &mut Work,
139) -> Result<partition::Plan, Fault> {
140    let mut caps = work.vector(indices.len())?;
141    for &index in indices {
142        work.charge(1)?;
143        caps.push(partition::Group {
144            members: &groups[index].members,
145            upper: groups[index].upper,
146        });
147    }
148    let mut limits = work.limits.partition;
149    limits.max_work = limits
150        .max_work
151        .min(work.limits.max_work - work.statistics.work);
152    limits.max_bytes = limits
153        .max_bytes
154        .min(work.limits.max_bytes - work.statistics.storage_bytes);
155    let plan = partition::Plan::new(
156        partition::Premises {
157            atom_count: theory.atom_count(),
158            members: &global.members,
159            lower: global.lower,
160            groups: &caps,
161        },
162        limits,
163        &work.cancellation,
164    );
165    let statistics = match &plan {
166        Ok(plan) => plan.statistics(),
167        Err(error) => error.statistics(),
168    };
169    work.record(statistics.work)?;
170    if statistics.construction_bytes <= work.limits.max_bytes - work.statistics.storage_bytes {
171        work.payload(statistics.construction_bytes)?;
172    }
173    let plan = plan.map_err(Fault::Partition)?;
174    Ok(plan)
175}
176fn append_consequences(
177    global: &Group,
178    indices: &[usize],
179    groups: &[Group],
180    plan: &partition::Plan,
181    consequences: &mut Vec<Consequence>,
182    work: &mut Work,
183) -> Result<(), Fault> {
184    for (consequence, &index) in plan.consequences().zip(indices) {
185        work.charge(1)?;
186        if consequence.lower <= groups[index].lower {
187            continue;
188        }
189        let mut members = work.vector(consequence.members.len())?;
190        for &atom in consequence.members {
191            work.charge(1)?;
192            members.push(atom);
193        }
194        let origin_count = global
195            .origins
196            .len()
197            .checked_add(
198                indices
199                    .iter()
200                    .try_fold(0_usize, |n, &i| n.checked_add(groups[i].origins.len()))
201                    .ok_or(Fault::Overflow)?,
202            )
203            .ok_or(Fault::Overflow)?;
204        let total = work
205            .statistics
206            .origins
207            .checked_add(origin_count)
208            .ok_or(Fault::Overflow)?;
209        ceiling(total, work.limits.max_origins, Resource::Origins)?;
210        let mut origins = work.vector(origin_count)?;
211        for &origin in global
212            .origins
213            .iter()
214            .chain(indices.iter().flat_map(|&i| &groups[i].origins))
215        {
216            work.charge(1)?;
217            origins.push(origin);
218        }
219        work.statistics.origins = total;
220        work.payload(super::bytes::<Consequence>(1)?)?;
221        consequences
222            .try_reserve_exact(1)
223            .map_err(|_| Fault::Stopped(zetesis_cpu::Stop::Allocation))?;
224        consequences.push(Consequence {
225            location: global.location,
226            body: global.body,
227            members,
228            lower: consequence.lower,
229            origins,
230        });
231    }
232    Ok(())
