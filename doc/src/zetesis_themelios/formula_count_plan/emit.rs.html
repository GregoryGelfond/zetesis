11pub(super) fn restriction(
12    original: &Theory,
13    consequences: &[Consequence],
14    work: &mut Work,
15) -> Result<(Theory, Vec<ProgramSite>), Fault> {
16    let limit = work.limits.theory;
17    if original.atom_count() > limit.max_atoms {
18        return Err(Fault::Theory(zetesis_ferraris::AdmissionError::Limit));
19    }
20    // A bounded prefix copy preserves every original activation ID exactly.
21    // Its nodes are not asserted: only the new consequence roots are roots of
22    // this separate candidate theory. No old root ID is replaced or aliased.
23    let prefix = consequences
24        .iter()
25        .map(|c| c.body)
26        .max()
27        .unwrap_or(0)
28        .checked_add(1)
29        .ok_or(Fault::Overflow)?;
30    if prefix > limit.max_nodes {
31        return Err(Fault::Theory(zetesis_ferraris::AdmissionError::Limit));
32    }
33    let mut nodes = FormulaNodes::default();
34    for index in 0..prefix {
35        let node = original.view().node(index).map_err(Fault::Theory)?;
36        let copied = push(&mut nodes, node, work)?;
37        debug_assert_eq!(copied, index, "admitted rows keep their original IDs");
38    }
39    let mut roots = work.vector(consequences.len())?;
40    if consequences.len() > limit.max_roots {
41        return Err(Fault::Theory(zetesis_ferraris::AdmissionError::Limit));
42    }
43    let origin_count = consequences
44        .iter()
45        .try_fold(0_usize, |n, c| n.checked_add(c.origins.len()))
46        .ok_or(Fault::Overflow)?;
47    let total_origins = work
48        .statistics
49        .origins
50        .checked_add(origin_count)
51        .ok_or(Fault::Overflow)?;
52    super::ceiling(
53        total_origins,
54        work.limits.max_origins,
55        super::CountPlanResource::Origins,
56    )?;
57    let mut origins = work.vector(origin_count)?;
58    work.statistics.origins = total_origins;
59    for consequence in consequences {
60        work.location = consequence.location;
61        work.charge(1)?;
62        let lower = lower(&mut nodes, consequence, work)?;
63        let root = if consequence.body == 1 {
64            lower
65        } else {
66            push(&mut nodes, NodeView::Implies(consequence.body, lower), work)?
67        };
68        roots.push(root);
69        for &origin in &consequence.origins {
70            work.charge(1)?;
71            origins.push(origin);
72        }
73    }
74    let admission =
75        2 * nodes.view().len() as u128 + nodes.parts().occurrences() as u128 + roots.len() as u128;
76    work.charge(u64::try_from(admission).map_err(|_| Fault::Overflow)?)?;
77    let result = Theory::new(original.atom_count(), nodes.into_parts(), roots, limit)
78        .map_err(Fault::Theory)?;
79    work.poll()?;
80    Ok((result, origins))
81}

83fn lower(
84    nodes: &mut FormulaNodes,
85    consequence: &Consequence,
86    work: &mut Work,
87) -> Result<usize, Fault> {
88    let limit = work.limits.theory;
89    let mut elements = work.vector(consequence.members.len())?;
90    for &atom in &consequence.members {
91        let node = push(nodes, NodeView::Atom(atom), work)?;
92        elements.push(AggregateElement {
93            weight: 1,
94            condition: node,
95        });
96    }
97    let mut limits = work.limits.aggregate;
98    limits.max_nodes = limits.max_nodes.min(limit.max_nodes);
99    limits.max_operands = limits.max_operands.min(limit.max_operands);
100    limits.max_work = limits
101        .max_work
102        .min(work.limits.max_work - work.statistics.work);
103    // Positive unit-weight Ge emits two constants and at most two nodes
104    // per member/threshold cell. Its two usize rows have width lower+1.
105    // Preflight that exact algorithm's worst-case logical payload before
106    // calling the separately limited, transactional aggregate constructor.
107    let cells = consequence
108        .lower
109        .checked_add(1)
110        .and_then(|n| n.checked_mul(2))
111        .ok_or(Fault::Overflow)?;
112    let appended = consequence
113        .members
114        .len()
115        .checked_mul(consequence.lower)
116        .and_then(|n| n.checked_mul(2))
117        .and_then(|n| n.checked_add(2))
118        .ok_or(Fault::Overflow)?;
119    work.payload(super::bytes::<usize>(cells)?)?;
120    work.payload(super::bytes::<Node>(appended)?)?;
121    let build = append_aggregate(
122        nodes,
123        &elements,
124        AggregateComparison::Ge,
125        i64::try_from(consequence.lower).map_err(|_| Fault::Overflow)?,
126        limits,
127        &work.cancellation,
128    );
129    let statistics = match &build {
130        Ok(build) => build.statistics(),
131        Err(error) => error.statistics(),
132    };
133    work.record(statistics.work)?;
134    let build = build.map_err(Fault::Aggregate)?;
135    Ok(build.root())
136}

138fn push(nodes: &mut FormulaNodes, node: NodeView<'_>, work: &mut Work) -> Result<usize, Fault> {
139    let edges = match node {
140        NodeView::And(row) | NodeView::Or(row) => row.len(),
141        NodeView::Implies(_, _) => 2,
142        NodeView::Atom(_) | NodeView::False => 0,
143    };
144    let wide = matches!(node, NodeView::And(row) | NodeView::Or(row) if row.len() >= 3);
145    let copies = if wide { edges } else { 0 };
146    work.charge(
147        u64::try_from(1_usize.saturating_add(edges).saturating_add(copies))
148            .map_err(|_| Fault::Overflow)?,
149    )?;
150    if nodes.view().len() >= work.limits.theory.max_nodes
151        || nodes.parts().occurrences() as u128 + edges as u128
152            > work.limits.theory.max_operands as u128
153    {
154        return Err(Fault::Theory(zetesis_ferraris::AdmissionError::Limit));
155    }
156    work.payload(super::bytes::<Node>(1)?)?;
157    work.payload(super::bytes::<usize>(copies)?)?;
158    let mut transaction = nodes.transaction();
159    let index = transaction
160        .push(
161            node,
162            work.limits.theory.max_nodes,
163            work.limits.theory.max_operands,
164        )
165        .map_err(Fault::Theory)?;
166    transaction.commit();
167    Ok(index)
168}
