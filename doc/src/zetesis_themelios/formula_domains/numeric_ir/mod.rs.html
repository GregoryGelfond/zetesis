30/// A borrowed numeric-domain report over the prepared program's canonical terms.
31/// The reader authenticates every retained key; argument bounds are conservative
32/// source bounds and establish neither truth nor answer-set membership.
33#[derive(Clone, Copy)]
34pub struct NumericAnalysis<'a, 'source> {
35    analysis: &'a Analysis<'a>,
36    read: CatalogRead<'source>,
37}
38impl std::fmt::Debug for NumericAnalysis<'_, '_> {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        f.debug_struct("NumericAnalysis")
41            .field("status", &self.analysis.values.status())
42            .field("statistics", &self.analysis.values.statistics())
43            .finish_non_exhaustive()
44    }
45}
46impl<'a, 'source> NumericAnalysis<'a, 'source> {
47    /// Bounds, finite values, producer sites and logical completion evidence.
48    #[must_use]
49    pub fn analysis(self) -> &'a numeric::Analysis<TermKey, ProgramSite> {
50        &self.analysis.values
51    }
52    /// Resolve a retained term without copying its logical value.
53    #[must_use]
54    pub fn catalog(self) -> CatalogRead<'source> {
55        self.read
56    }
57    /// Completed diagnostic populations, or None after a logical analysis stop.
58    /// Zero counts in a completed census remain distinct from unavailable results.
59    #[must_use]
60    pub fn census(self) -> Option<NumericCensus> {
61        self.analysis.census
62    }
63
64    /// Reports in non-fact compiled occurrence order. No reports are published
65    /// after an analysis stop. Equal source sites do not merge occurrences.
66    #[must_use]
67    pub fn rules(self) -> impl ExactSizeIterator<Item = NumericRuleAnalysis<'a>> + 'a {
68        self.analysis
69            .occurrences
70            .iter()
71            .enumerate()
72            .map(move |(occurrence, rule)| NumericRuleAnalysis {
73                rule,
74                values: &self.analysis.values,
75                occurrence,
76            })
77    }
78
79    /// Signed predicates and their argument coordinates in this analysis.
80    ///
81    /// Each entry resolves against the same canonical prefix as the terms.
82    pub fn arguments(
83        self,
84    ) -> impl Iterator<Item = Result<(PredicateRef<'source>, usize, numeric::ArgumentId), ReadError>> + 'a
85    where
86        'source: 'a,
87    {
88        self.analysis.predicates.iter().flat_map(move |predicate| {
89            predicate
90                .arguments
91                .iter()
92                .enumerate()
93                .map(move |(column, &argument)| {
94                    self.read
95                        .predicate(&predicate.key)
96                        .map(|name| (name, column, argument))
97                })
98        })
99    }
100}

102impl Analysis<'_> {
103    pub(crate) fn view<'a, 'source>(
104        &'a self,
105        read: CatalogRead<'source>,
106    ) -> NumericAnalysis<'a, 'source> {
107        NumericAnalysis {
108            analysis: self,
109            read,
110        }
111    }
112
113    pub(crate) fn argument(
114        &self,
115        predicate: PredicateRef<'_>,
116        column: usize,
117        read: CatalogRead<'_>,
118        limits: &FormulaLimits,
119        counters: &mut Counters,
120        location: ProgramSite,
121    ) -> Result<Option<numeric::ArgumentId>, FormulaFailure> {
122        let mut low = 0;
123        let mut high = self.predicates.len();
124        while low < high {
125            counters.work(limits, location)?;
126            let middle = low + (high - low) / 2;
127            let retained = read
128                .predicate(&self.predicates[middle].key)
129                .map_err(|error| read_failure(error, location))?;
130            match predicate.compare_ref_with(retained, || counters.work(limits, location))? {
131                std::cmp::Ordering::Less => high = middle,
132                std::cmp::Ordering::Greater => low = middle + 1,
133                std::cmp::Ordering::Equal => {
134                    return Ok(self.predicates[middle].arguments.get(column).copied());
135                }
136            }
137        }
138        Ok(None)
139    }
140}

142/// The caller accounts work and preserves the original fault and location.
143/// Other analysis statistics are meaningful only for a published result.
144pub(crate) struct Failure {
145    pub(crate) error: numeric::Error<ReadError>,
146    pub(crate) work: u64,
147    pub(crate) context: Option<ProgramSite>,
148}
149
150pub(crate) fn analyze<'prepared>(
151    prepared: &'prepared Prepared,
152    components: TemplateComponentsRef<'_>,
153    read: CatalogRead<'_>,
154    limits: Limits,
155    diagnostics: bool,
156) -> Result<Analysis<'prepared>, Failure> {
157    let reader = Reader(read);
158    let mut lower = Lowering {
159        reader: Reader(read),
160        components,
161        predicates: Vec::new(),
162        finite_width: limits.max_values_per_argument,
163        diagnostics,
164        occurrences: Vec::new(),
165        totality: Vec::new(),
166    };
167    let values = numeric::analyze(&reader, limits, |input| lower.program(prepared, input))
168        .map_err(|failure| Failure {
169            error: failure.error,
170            work: failure.statistics.work,
171            context: failure.context,
172        })?;
173    // A stopped analysis publishes no usable argument coordinates.
174    let (predicates, occurrences, totality) = if values.status() == numeric::Status::FixedPoint {
175        (lower.predicates, lower.occurrences, lower.totality)
176    } else {
177        (Vec::new(), Vec::new(), Vec::new())
178    };
179    Ok(Analysis {
180        values,
181        predicates,
182        occurrences,
183        census: None,
184        totality,
185    })
186}

188pub(crate) fn failure(error: numeric::Error<ReadError>, location: ProgramSite) -> FormulaFailure {
189    match error {
190        numeric::Error::Allocation(error) => FormulaFailure::AtomAllocation { error, location },
191        numeric::Error::Read(error) => read_failure(error, location),
192        // Logical limits are settled as stopped analysis inside the engine.
193        // An escaped stop or malformed private IR is not an empty domain.
194        numeric::Error::Limit(_) | numeric::Error::InvalidInput => components::missing(location),
195    }
196}
197
198fn read_failure(error: ReadError, location: ProgramSite) -> FormulaFailure {
199    FormulaFailure::TemplateCatalog {
200        error: zetesis_core::TemplateCatalogFailure::Read(error),
201        location,
202    }
