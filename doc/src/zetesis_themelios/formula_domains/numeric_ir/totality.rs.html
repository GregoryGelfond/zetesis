1//! Static scalar totality for possible-head production.
2//!
3//! The lowering requests every scalar operand of an eligible rule, in literal
4//! order, against that exact neutral rule. Its complete-binding domains include
5//! every compiled producer, rather than one intermediate support snapshot.
6//! Successful requests retain only the original rule borrow and occurrence.
7//! They grant prefix selection, not a computed-key map, source-family evidence
8//! or a witness that a complete binding exists.
9
10use zetesis_core::TemplateComponentsRef;
11use zetesis_domain::numeric;
12
13use super::{Analysis, census};
14use crate::formula_ir::{HeadIr, LiteralIr, RuleIr};
15use crate::formula_support::Counters;
16use crate::{FormulaFailure, FormulaLimits};

18pub(super) struct Request<'source> {
19    pub(super) index: usize,
20    pub(super) source: &'source RuleIr,
21    pub(super) rule: numeric::RuleId,
22}
23
24/// Every complete binding of this rule has defined scalar body expressions.
25/// Private construction requires the finished all-producer analysis. A prefix
26/// can still contain an out-of-domain value; its evaluator stays checked.
27pub(crate) struct TotalBody<'source> {
28    source: &'source RuleIr,
29}
30
31impl TotalBody<'_> {
32    pub(crate) fn belongs_to(&self, source: &RuleIr) -> bool {
33        std::ptr::eq(self.source, source)
34    }
35}
36
37/// Request execution evidence only where Complete coverage otherwise prevents
38/// scalar prefix selection. The shared readiness check establishes positive
39/// slot coverage after the analysis; syntax alone never issues a certificate.
40pub(super) fn requested(source: &RuleIr) -> bool {
41    matches!(source.head, HeadIr::Normal(Some(_)))
42        && source.variables == source.body_variables
43        && source.bindings.is_none()
44        && crate::formula_support::family::partial(&source.body)
45        && source.body.iter().all(|literal| {
46            matches!(
47                literal,
48                LiteralIr::Atom(..) | LiteralIr::Compare(_) | LiteralIr::ArgumentCheck(_)
49            )
50        })
51}

53impl<'source> Analysis<'source> {
54    /// Retain sparse certificates in original occurrence order. The temporary
55    /// query graph and numeric result are released with this analysis. Metadata
56    /// is bounded by admitted rules and each attempted publication is charged.
57    pub(crate) fn total_bodies(
58        &self,
59        components: TemplateComponentsRef<'_>,
60        limits: &FormulaLimits,
61        counters: &mut Counters,
62    ) -> Result<Vec<(usize, TotalBody<'source>)>, FormulaFailure> {
63        let mut result = Vec::new();
64        if self.values.status() != numeric::Status::FixedPoint {
65            return Ok(result);
66        }
67        for request in &self.totality {
68            counters.work(limits, request.source.location)?;
69            if !self.total(request, components, limits, counters)? {
70                continue;
71            }
72            counters.work(limits, request.source.location)?;
73            result
74                .try_reserve(1)
75                .map_err(|error| FormulaFailure::AtomAllocation {
76                    error,
77                    location: request.source.location,
78                })?;
79            result.push((
80                request.index,
81                TotalBody {
82                    source: request.source,
83                },
84            ));
85        }
86        Ok(result)
87    }
88
89    fn total(
90        &self,
91        request: &Request<'_>,
92        components: TemplateComponentsRef<'_>,
93        limits: &FormulaLimits,
94        counters: &mut Counters,
95    ) -> Result<bool, FormulaFailure> {
96        let source = request.source;
97        let missing = || crate::formula_support::components::missing(source.location);
98        let bounds = self.values.rule(request.rule).ok_or_else(missing)?;
99        let Some(binding) = census::readiness(source, components, limits, counters)? else {
100            return Ok(false);
101        };
102        if bounds.domains().len() != binding.len() {
103            return Err(missing());
104        }
105        for (depth, domain) in binding.iter().zip(bounds.domains()) {
106            counters.work(limits, source.location)?;
107            if depth.is_none()
108                || (domain.integrality == numeric::Integrality::Integer
109                    && domain.interval == numeric::Interval::Bottom)
110            {
111                return Ok(false);
112            }
113        }
114        let mut query = 0;
115        let mut total = true;
116        for literal in &source.body {
117            counters.work(limits, source.location)?;
118            let comparison = match literal {
119                LiteralIr::Compare(comparison) | LiteralIr::ArgumentCheck(comparison) => comparison,
120                LiteralIr::Atom(..) => continue,
121                _ => return Err(missing()),
122            };
123            for expression in [comparison.left(), comparison.right()] {
124                counters.work(limits, source.location)?;
125                total &=
126                    bounds.integer(query).is_some() || census::leaf(expression, Some(&binding));
127                query += 1;
128            }
129        }
130        if query != bounds.integer_query_count() {
131            return Err(missing());
132        }
133        Ok(total)
134    }
135}
