1//! The prepared program as items: rules, and facts held in fact blocks.
2//!
3//! A fact statement `p(t₁, …, tₙ).` expands to one product per argument
4//! alternative (`fact_expansion`). Compilation stores the facts in the fact
5//! arena instead of building a rule with an empty body for each of its rows:
6//! a block's predicate is admitted once, its cells select admitted occurrences,
7//! and the statement's evidence is stored once. A fact block is one item in one
8//! of two layouts. A statement with one argument alternative of one row supplies
9//! a run row; consecutive such statements of one predicate share a run, stored
10//! by column with one evidence range per statement. Every other argument
11//! alternative becomes a product, the Cartesian product of its columns.
12//!
13//! Readers traverse the prepared program as [`Item`]s: a rule, or a fact block
14//! standing exactly where its fact rules stood, its rows in expansion order
15//! (alternatives in order, last argument fastest). Every consumer that visits
16//! the rules in order visits items in the same order, so wherever the rule form
17//! discovered, staged, interned or emitted a fact rule's head, the block's row
18//! is discovered, staged, interned or emitted at the same point. A run is one
19//! item: it keeps its statements' rows in order, and a reader that counts items
20//! counts it once.
21//!
22//! Product columns retain scalars and integer segments; an integer segment
23//! records its numeric start and a certified consecutive span of scalar
24//! occurrence coordinates. It compresses this arena, not the source vocabulary
25//! or its scalar occurrences. Runs retain one scalar buffer per column.
26//! Storage is O(segments + run cells + evidence + blocks + columns). Every vector of an arena
27//! grows only through a caller's [`LeasedReserve`], which admits the growth
28//! before it is allocated; filling reserved room allocates nothing. Reading a
29//! row allocates nothing: a product's cell is selected from its column by the
30//! row's mixed-radix position and a binary search of its column's segments.
31

48/// Reserves room in a vector before anything is written to it, admitting the
49/// allocation under `lease` first. The lease holds every buffer its caller
50/// grows through it, so its receipt less `values`'s own capacity is the bytes
51/// it holds besides `values`.
52pub(crate) trait LeasedReserve {
53    /// Reserve room for `additional` more entries in `values`.
54    ///
55    /// # Errors
56    /// A refused admission, or an allocation failure, before `values` changes.
57    fn reserve_leased<T>(
58        &mut self,
59        values: &mut Vec<T>,
60        additional: usize,
61        lease: &mut StorageLease,
62    ) -> Result<(), FormulaFailure>;
63}

65/// The prepared program in item order.
66///
67/// Every rule has exactly one rule slot, and the `k`-th rule slot in item
68/// order names rule `k`. The rules therefore keep their relative order, a
69/// rule's index among the rules is stable for readers that never see facts,
70/// and a reader that pairs the rule slots with the rules in their stored order
71/// pairs each slot with its own rule. Each operation that adds, removes or
72/// moves rules keeps this: a rule is only ever appended through
73/// [`Self::push_rule`], which records its slot.
74#[derive(Default)]
75pub(crate) struct Items {
76    rules: Vec<RuleIr>,
77    facts: FactArena,
78    order: Vec<Slot>,
79}

186    /// Keep the rules `keep` accepts and every fact block, in their order.
187    /// `keep` decides each rule once, in rule order. Allocates nothing: the
188    /// kept rules are compacted in place.
189    pub(crate) fn retain_rules(&mut self, mut keep: impl FnMut(&RuleIr) -> bool) {
190        let rules = &mut self.rules;
191        // Rule slots name the rules in their stored order (the type's
192        // invariant), so the slots visit the rules in order: below `kept` lie
193        // the rules kept so far, from `kept` to the rule visited the rules
194        // dropped so far, and above it the rules not yet visited.
195        let mut next = 0;
196        let mut kept = 0;
197        self.order.retain_mut(|slot| match slot {
198            Slot::Rule(index) => {
199                debug_assert_eq!(*index, next, "the k-th rule slot names rule k");
200                next += 1;
201                if keep(&rules[*index]) {
202                    rules.swap(kept, *index);
203                    *index = kept;
204                    kept += 1;
205                    true
206                } else {
207                    false
208                }
209            }
210            Slot::Facts(_) => true,
211        });
212        rules.truncate(kept);
213    }

215    /// Reserve, before anything moves, the room moving the items `selected`
216    /// marks (one flag per item, nonzero to move) needs:
217    /// - under `moved_lease`, the moved sequence's rules and items, and its
218    ///   arena's vectors, each for the summed requirement of everything that
219    ///   moves into it;
220    /// - under `kept_lease`, the replacement vectors of this sequence's kept
221    ///   rules and items.
222    ///
223    /// The moved arena holds no lease of its own: its bytes are
224    /// `moved_lease`'s.
225    ///
226    /// # Errors
227    /// A refused admission or allocation, before anything moves.
228    ///
229    /// # Panics
230    /// Panics if `selected` does not hold one flag per item.
231    pub(crate) fn reserve_split(
232        &self,
233        selected: &[u8],
234        room: &mut impl LeasedReserve,
235        moved_lease: &mut StorageLease,
236        kept_lease: &mut StorageLease,
237    ) -> Result<SplitRoom, FormulaFailure> {
238        assert_eq!(selected.len(), self.order.len(), "one flag per item");
239        let mut moving = 0;
240        let mut moving_rules = 0;
241        let mut growth = Growth::default();
242        for (slot, flag) in self.order.iter().zip(selected) {
243            if *flag == 0 {
244                continue;
245            }
246            moving += 1;
247            match *slot {
248                Slot::Rule(_) => moving_rules += 1,
249                Slot::Facts(block) => growth.add(self.facts.footprint(block)),
250            }
251        }
252        let mut split = SplitRoom {
253            moved: Self::default(),
254            kept_rules: Vec::new(),
255            kept_order: Vec::new(),
256        };
257        room.reserve_leased(&mut split.moved.rules, moving_rules, moved_lease)?;
258        room.reserve_leased(&mut split.moved.order, moving, moved_lease)?;
259        split.moved.facts.reserve_in(growth, room, moved_lease)?;
260        for (slot, flag) in self.order.iter().zip(selected) {
261            if let (Slot::Facts(block), true) = (*slot, *flag != 0)
262                && let Layout::Run { columns, .. } = &self.facts.blocks[block].layout
263            {
264                split.moved.facts.reserve_new_run(
265                    columns.len(),
266                    self.facts.blocks[block].rows,
267                    room,
268                    moved_lease,
269                )?;
270            }
271        }
272        room.reserve_leased(
273            &mut split.kept_rules,
274            self.rules.len() - moving_rules,
275            kept_lease,
276        )?;
277        room.reserve_leased(&mut split.kept_order, self.order.len() - moving, kept_lease)?;
278        Ok(split)
279    }

281    /// Move the items `selected` marks into the `room` that
282    /// [`Self::reserve_split`] reserved for them, keeping both sequences'
283    /// order, and return the moved sequence. Allocates nothing.
284    ///
285    /// The moved fact blocks are copied. Their storage stays in this arena,
286    /// which keeps its capacity and lease, and they are no longer its items:
287    /// until this arena is dropped, a moved block's cells and evidence are
288    /// held twice, here and in the moved sequence, each under its own lease.
289    ///
290    /// # Panics
291    /// Panics if `selected` does not hold one flag per item.
292    pub(crate) fn split_off(&mut self, selected: &[u8], room: SplitRoom) -> Self {
293        assert_eq!(selected.len(), self.order.len(), "one flag per item");
294        let SplitRoom {
295            mut moved,
296            mut kept_rules,
297            mut kept_order,
298        } = room;
299        // Rule slots name the rules in their stored order (the type's
300        // invariant), so the k-th rule slot takes the k-th rule.
301        let mut rules = std::mem::take(&mut self.rules).into_iter();
302        let mut run_column = 0;
303        for (&slot, flag) in self.order.iter().zip(selected) {
304            match (slot, *flag != 0) {
305                (Slot::Rule(_), true) => {
306                    push_reserved(&mut moved.order, Slot::Rule(moved.rules.len()));
307                    let rule = rules.next().expect("one rule per rule slot");
308                    push_reserved(&mut moved.rules, rule);
309                }
310                (Slot::Rule(_), false) => {
311                    push_reserved(&mut kept_order, Slot::Rule(kept_rules.len()));
312                    let rule = rules.next().expect("one rule per rule slot");
313                    push_reserved(&mut kept_rules, rule);
314                }
315                (Slot::Facts(block), true) => {
316                    push_reserved(&mut moved.order, Slot::Facts(moved.facts.blocks.len()));
317                    self.facts
318                        .copy_block(block, &mut moved.facts, &mut run_column);
319                }
320                (Slot::Facts(block), false) => push_reserved(&mut kept_order, Slot::Facts(block)),
321            }
322        }
323        self.rules = kept_rules;
324        self.order = kept_order;
325        moved
326    }

328    /// The retained capacities of the rules' and arena's vectors, and of each
329    /// rule's body and origins.
330    pub(crate) fn retained_bytes(&self) -> u128 {
331        self.rules.capacity() as u128 * size_of::<RuleIr>() as u128
332            + self
333                .rules
334                .iter()
335                .map(|rule| {
336                    rule.body.capacity() as u128 * size_of::<crate::formula_ir::LiteralIr>() as u128
337                        + rule.origins.capacity() as u128 * size_of::<ProgramSite>() as u128
338                })
339                .sum::<u128>()
340            + self.order.capacity() as u128 * size_of::<Slot>() as u128
341            + self.facts.retained_bytes() as u128
342    }

448/// Facts stored by block, with each statement's evidence once.
449#[derive(Default)]
450pub(crate) struct FactArena {
451    blocks: Vec<Block>,
452    /// Product column segments, with admitted scalar occurrence coordinates.
453    segments: Vec<Segment>,
454    /// Product columns, in argument order within each block.
455    columns: Vec<ProductColumn>,
456    /// Each run's argument columns; one admitted occurrence per row per column.
457    run_columns: Vec<Vec<Scalar>>,
458    /// Statement evidence: once per statement of a product, once per run row.
459    evidence: Vec<ProgramSite>,
460    /// Each run row's evidence end, an offset into `evidence`.
461    ends: Vec<usize>,
462    /// All vector headers and buffers above, including nested run-column buffers.
463    lease: Option<StorageLease>,

469struct Block {

471    arity: usize,

476enum Layout {
477    /// The Cartesian product of `columns`, last column fastest; every row
478    /// shares the statement's `evidence`.
479    Product {
480        columns: Range<usize>,
481        evidence: Range<usize>,
482    },
483    /// One buffer per argument. Row `i`'s evidence ends at `ends[ends + i]`
484    /// and begins where the previous row's ended, or at `evidence` initially.
485    Run {
486        columns: Range<usize>,
487        ends: usize,
488        evidence: usize,
489    },

492/// One product column: its segment range, expanded width, and the number of
493/// rows that pass before it advances. The segments need not lie in argument
494/// order in their buffer: compilation admits later columns first.
495#[derive(Clone, Debug)]
496pub(crate) struct ProductColumn {
497    segments: Range<usize>,
498    width: usize,
499    stride: usize,
500}
501
502impl ProductColumn {
503    /// A column over complete, consecutive segments in its scratch buffer.
504    pub(crate) fn new(segments: Range<usize>, width: usize, stride: usize) -> Self {
505        Self {
506            segments,
507            width,
508            stride,
509        }
510    }
511
512    fn cell(&self, segments: &[Segment], offset: usize) -> Scalar {
513        let segments = &segments[self.segments.clone()];
514        let segment = &segments[segments.partition_point(|segment| segment.end <= offset)];
515        segment.get(offset - (segment.end - segment.len()))
516    }
517}

519/// One source scalar, or consecutive integers whose actual admitted occurrence
520/// coordinates were also consecutive. `end` is its exclusive expanded position
521/// within its column, so rows locate segments without expanding them.
522#[derive(Clone, Copy, Debug)]
523pub(crate) struct Segment {
524    end: usize,
525    values: ColumnSegment,
526}
527
528/// One compact column segment. Scalars and spans are occurrence coordinates in
529/// this preparation's source components, never canonical value identifiers.
530/// An integer segment certifies both its consecutive numbers and its admitted
531/// consecutive coordinates; readers resolve only the coordinates they need.
532#[derive(Clone, Copy, Debug)]
533pub(crate) enum ColumnSegment {
534    Scalar(Scalar),
535    Integers { first: i32, scalars: ScalarSpan },
536}
537
538impl Segment {
539    /// One scalar at its expanded column position.
540    pub(crate) fn scalar(position: usize, scalar: Scalar) -> Self {
541        Self {
542            end: position + 1,
543            values: ColumnSegment::Scalar(scalar),
544        }
545    }
546
547    /// The first admitted number of an integer segment.
548    pub(crate) fn integer(position: usize, number: i32, scalar: Scalar) -> Self {
549        Self {
550            end: position + 1,
551            values: ColumnSegment::Integers {
552                first: number,
553                scalars: ScalarSpan::new(scalar),
554            },
555        }
556    }
557
558    /// Coalesce only consecutive numbers and consecutive observed coordinates.
559    /// A repeated value or an occurrence-coordinate gap starts a new segment.
560    pub(crate) fn append(&mut self, number: i32, scalar: Scalar) -> bool {
561        let ColumnSegment::Integers { first, scalars } = &mut self.values else {
562            return false;
563        };
564        let next = i64::try_from(scalars.len())
565            .ok()
566            .and_then(|width| i64::from(*first).checked_add(width));
567        if next != Some(i64::from(number)) || !scalars.append(scalar) {
568            return false;
569        }
570        self.end += 1;
571        true
572    }
573
574    fn len(self) -> usize {
575        match self.values {
576            ColumnSegment::Scalar(_) => 1,
577            ColumnSegment::Integers { scalars, .. } => scalars.len(),
578        }
579    }
580
581    fn get(self, offset: usize) -> Scalar {
582        match self.values {
583            ColumnSegment::Scalar(scalar) => {
584                debug_assert_eq!(offset, 0);
585                scalar
586            }
587            ColumnSegment::Integers { scalars, .. } => scalars.get(offset),
588        }
589    }
590}

592/// A statement's evidence in the arena, shared by the blocks it expands to.
593#[derive(Clone, Debug)]
594pub(crate) struct Evidence(Range<usize>);
595
596impl FactArena {
597    /// What one compiled argument alternative adds to the arena, in constant
598    /// time from its layout scratch and any statement evidence not yet stored.
599    /// A run row needs an end; a product needs columns. Continuing the last
600    /// run needs no new block. The compiler decides which layout it appends;
601    /// only the arena translates that decision into vector growth.
602    pub(crate) fn growth(
603        columns: &[ProductColumn],
604        segments: &[Segment],
605        new_evidence: Option<&[ProgramSite]>,
606        run: bool,
607        continuing: bool,
608    ) -> Growth {
609        debug_assert!(!continuing || run, "only a run can be continued");
610        Growth {
611            blocks: usize::from(!continuing),
612            segments: if run { 0 } else { segments.len() },
613            columns: if run { 0 } else { columns.len() },
614            run_columns: if run && !continuing { columns.len() } else { 0 },
615            evidence: new_evidence.map_or(0, <[ProgramSite]>::len),
616            ends: usize::from(run),
617        }
618    }
619
620    /// Reserve `growth` under the arena's lease, installed by `install` on
621    /// first use, admitting each vector's growth before it is allocated.
622    ///
623    /// # Errors
624    /// A refused admission or allocation, before the arena changes.
625    pub(crate) fn reserve(
626        &mut self,
627        growth: Growth,
628        install: impl FnOnce() -> StorageLease,
629        room: &mut impl LeasedReserve,
630    ) -> Result<(), FormulaFailure> {
631        let mut lease = self.lease.take().unwrap_or_else(install);
632        let reserved = self.reserve_in(growth, room, &mut lease);
633        self.lease = Some(lease);
634        reserved
635    }
636
637    /// Reserve `growth` in each vector under `lease`.
638    fn reserve_in(
639        &mut self,
640        growth: Growth,
641        room: &mut impl LeasedReserve,
642        lease: &mut StorageLease,
643    ) -> Result<(), FormulaFailure> {
644        room.reserve_leased(&mut self.blocks, growth.blocks, lease)?;
645        room.reserve_leased(&mut self.segments, growth.segments, lease)?;
646        room.reserve_leased(&mut self.columns, growth.columns, lease)?;
647        room.reserve_leased(&mut self.run_columns, growth.run_columns, lease)?;
648        room.reserve_leased(&mut self.evidence, growth.evidence, lease)?;
649        room.reserve_leased(&mut self.ends, growth.ends, lease)
650    }

652    /// Reserve every argument buffer before publishing a run row. On failure
653    /// no row is appended; any successfully reserved buffers stay in the arena
654    /// under its restored lease, including new empty column headers.
655    pub(crate) fn reserve_row(
656        &mut self,
657        existing: Option<usize>,
658        arity: usize,
659        room: &mut impl LeasedReserve,
660    ) -> Result<Range<usize>, FormulaFailure> {
661        let mut lease = self.lease.take().expect("arena growth installed its lease");
662        let result = match existing {
663            Some(block) => {
664                let Layout::Run { columns, .. } = &self.blocks[block].layout else {
665                    unreachable!("only an open run can be continued")
666                };
667                let columns = columns.clone();
668                self.reserve_run_cells(columns.clone(), 1, room, &mut lease)
669                    .map(|()| columns)
670            }
671            None => self.reserve_new_run(arity, 1, room, &mut lease),
672        };
673        self.lease = Some(lease);
674        result
675    }
676
677    /// Append empty headers in already reserved room, then reserve all their
678    /// cells. Keeping the headers on failure keeps each buffer owned and leased.
679    fn reserve_new_run(
680        &mut self,
681        arity: usize,
682        rows: usize,
683        room: &mut impl LeasedReserve,
684        lease: &mut StorageLease,
685    ) -> Result<Range<usize>, FormulaFailure> {
686        let first = self.run_columns.len();
687        for _ in 0..arity {
688            push_reserved(&mut self.run_columns, Vec::new());
689        }
690        let columns = first..self.run_columns.len();
691        self.reserve_run_cells(columns.clone(), rows, room, lease)?;
692        Ok(columns)
693    }
694
695    fn reserve_run_cells(
696        &mut self,
697        columns: Range<usize>,
698        rows: usize,
699        room: &mut impl LeasedReserve,
700        lease: &mut StorageLease,
701    ) -> Result<(), FormulaFailure> {
702        for column in &mut self.run_columns[columns] {
703            room.reserve_leased(column, rows, lease)?;
704        }
705        Ok(())
706    }

724    /// Store a product's columns and segments in reserved room. A column's
725    /// segment coordinates are translated; scalar coordinates still refer to
726    /// the same preparation's source components.
727    pub(crate) fn push_product(
728        &mut self,
729        predicate: Predicate,
730        columns: &[ProductColumn],
731        segments: &[Segment],
732        evidence: &Evidence,
733    ) -> usize {
734        let segment_start = self.segments.len();
735        extend_reserved(&mut self.segments, segments);
736        let first = self.columns.len();
737        for column in columns {
738            push_reserved(
739                &mut self.columns,
740                ProductColumn {
741                    segments: segment_start + column.segments.start
742                        ..segment_start + column.segments.end,
743                    width: column.width,
744                    stride: column.stride,
745                },
746            );
747        }
748        let block = self.blocks.len();
749        push_reserved(
750            &mut self.blocks,
751            Block {
752                predicate,
753                arity: columns.len(),
754                rows: columns
755                    .first()
756                    .map_or(1, |column| column.width.saturating_mul(column.stride)),
757                layout: Layout::Product {
758                    columns: first..self.columns.len(),
759                    evidence: evidence.0.clone(),
760                },
761            },
762        );
763        block
764    }

766    /// Append one row to the already reserved argument buffers.
767    pub(crate) fn push_row(&mut self, columns: Range<usize>, cells: &[Option<Scalar>]) {
768        debug_assert_eq!(columns.len(), cells.len());
769        for (column, cell) in self.run_columns[columns].iter_mut().zip(cells) {
770            push_reserved(column, cell.expect("every first-row value is admitted"));
771        }
772    }

774    /// Start a run over the columns whose first row was just appended.
775    pub(crate) fn push_run(
776        &mut self,
777        predicate: Predicate,
778        columns: Range<usize>,
779        evidence: &Evidence,
780    ) -> usize {
781        let block = self.blocks.len();
782        push_reserved(
783            &mut self.blocks,
784            Block {
785                predicate,
786                arity: columns.len(),
787                rows: 1,
788                layout: Layout::Run {
789                    columns,
790                    ends: self.ends.len(),
791                    evidence: evidence.0.start,
792                },
793            },
794        );
795        push_reserved(&mut self.ends, evidence.0.end);
796        block
797    }

799    /// Continue the run `block`, in reserved room, with the row of cells
800    /// pushed last, whose evidence was stored last.
801    pub(crate) fn extend_run(&mut self, block: usize, evidence: &Evidence) {
802        debug_assert_eq!(
803            self.ends.last().copied(),
804            Some(evidence.0.start),
805            "a run's rows store their evidence contiguously"
806        );
807        push_reserved(&mut self.ends, evidence.0.end);
808        self.blocks[block].rows += 1;
809    }

826    /// The retained capacities of the arena's vectors.
827    pub(crate) fn retained_bytes(&self) -> usize {
828        self.blocks.capacity() * size_of::<Block>()
829            + self.segments.capacity() * size_of::<Segment>()
830            + self.columns.capacity() * size_of::<ProductColumn>()
831            + self.run_columns.capacity() * size_of::<Vec<Scalar>>()
832            + self
833                .run_columns
834                .iter()
835                .map(|column| column.capacity() * size_of::<Scalar>())
836                .sum::<usize>()
837            + self.evidence.capacity() * size_of::<ProgramSite>()
838            + self.ends.capacity() * size_of::<usize>()
839    }

847    /// What [`Self::copy_block`] appends for `block`, per vector.
848    fn footprint(&self, block: usize) -> Growth {
849        let source = FactBlock { arena: self, block };
850        let mut growth = Growth {
851            blocks: 1,
852            ..Growth::default()
853        };
854        match &self.blocks[block].layout {
855            Layout::Product { columns, evidence } => {
856                growth.columns = columns.len();
857                growth.segments = self.columns[columns.clone()]
858                    .iter()
859                    .map(|column| column.segments.len())
860                    .sum();
861                growth.evidence = evidence.len();
862            }
863            Layout::Run { columns, .. } => {
864                growth.run_columns = columns.len();
865                growth.ends = self.blocks[block].rows;
866                growth.evidence = source.rows().map(|row| row.evidence().len()).sum();
867            }
868        }
869        growth
870    }

872    /// Append a copy of `block` to `target`, in the room its
873    /// [`Self::footprint`] reserved there.
874    fn copy_block(&self, block: usize, target: &mut Self, run_column: &mut usize) {
875        let source = &self.blocks[block];
876        let facts = FactBlock { arena: self, block };
877        let layout = match &source.layout {
878            Layout::Product { columns, evidence } => {
879                let first = target.columns.len();
880                for column in &self.columns[columns.clone()] {
881                    let start = target.segments.len();
882                    extend_reserved(
883                        &mut target.segments,
884                        &self.segments[column.segments.clone()],
885                    );
886                    push_reserved(
887                        &mut target.columns,
888                        ProductColumn {
889                            segments: start..target.segments.len(),
890                            width: column.width,
891                            stride: column.stride,
892                        },
893                    );
894                }
895                let start = target.evidence.len();
896                extend_reserved(&mut target.evidence, &self.evidence[evidence.clone()]);
897                Layout::Product {
898                    columns: first..target.columns.len(),
899                    evidence: start..target.evidence.len(),
900                }
901            }
902            Layout::Run { columns, .. } => {
903                let first = *run_column;
904                for column in &self.run_columns[columns.clone()] {
905                    extend_reserved(&mut target.run_columns[*run_column], column);
906                    *run_column += 1;
907                }
908                let ends = target.ends.len();
909                let evidence = target.evidence.len();
910                for row in facts.rows() {
911                    extend_reserved(&mut target.evidence, row.evidence());
912                    push_reserved(&mut target.ends, target.evidence.len());
913                }
914                Layout::Run {
915                    columns: first..*run_column,
916                    ends,
917                    evidence,
918                }
919            }
920        };
921        push_reserved(
922            &mut target.blocks,
923            Block {
924                predicate: source.predicate,
925                arity: source.arity,
926                rows: source.rows,
927                layout,
928            },
929        );
930    }

933/// A fact block: one item holding rows of one predicate, in one of two
934/// layouts. A product holds one statement alternative's rows as the Cartesian
935/// product of its columns; a run holds consecutive one-row statements, one row
936/// each.
937#[derive(Clone, Copy)]
938pub(crate) struct FactBlock<'a> {
939    arena: &'a FactArena,
940    block: usize,
941}
942
943impl<'a> FactBlock<'a> {
944    fn block(self) -> &'a Block {
945        &self.arena.blocks[self.block]
946    }
947
948    /// The predicate, admitted once for the block.
949    pub(crate) fn predicate(self) -> Predicate {
950        self.block().predicate
951    }
952
953    /// The block's arity.
954    pub(crate) fn arity(self) -> usize {
955        self.block().arity
956    }
957
958    /// The number of rows, at least one.
959    pub(crate) fn len(self) -> usize {
960        self.block().rows
961    }
962
963    /// The site of the block's first statement.
964    pub(crate) fn location(self) -> ProgramSite {
965        self.rows()
966            .next()
967            .expect("a stored block has a row")
968            .location()
969    }
970
971    /// Every admitted cell in the previous logical traversal order: columns
972    /// in argument order for a product, rows in emission order for a run.
973    /// Expanding a certified span produces only occurrence coordinates, with
974    /// no allocation or owned value reconstruction.
975    pub(crate) fn cells(self) -> impl Iterator<Item = Scalar> + use<'a> {
976        let product = matches!(self.block().layout, Layout::Product { .. });
977        let products = self
978            .columns()
979            .take(if product { self.arity() } else { 0 })
980            .flat_map(FactColumn::segments)
981            .flat_map(|segment| {
982                let (scalar, span) = match segment {
983                    ColumnSegment::Scalar(scalar) => (Some(scalar), None),
984                    ColumnSegment::Integers { scalars, .. } => (None, Some(scalars)),
985                };
986                scalar.into_iter().chain(
987                    span.into_iter()
988                        .flat_map(|span| (0..span.len()).map(move |offset| span.get(offset))),
989                )
990            });
991        let runs = (0..if product { 0 } else { self.len() })
992            .flat_map(move |index| Row { facts: self, index }.cells());
993        products.chain(runs)
994    }
995
996    /// Argument columns in argument order, without Cartesian row expansion.
997    /// Product columns contain compact integer spans and scalar cells; run
998    /// columns contain their existing cells in row order. Reading allocates
999    /// nothing. The caller owns cancellation and work for the segments read.
1000    pub(crate) fn columns(self) -> impl ExactSizeIterator<Item = FactColumn<'a>> + use<'a> {
1001        (0..self.arity()).map(move |argument| FactColumn {
1002            facts: self,
1003            argument,
1004        })
1005    }
1006
1007    /// The rows in expansion order.
1008    pub(crate) fn rows(self) -> impl Iterator<Item = Row<'a>> + use<'a> {
1009        (0..self.len()).map(move |index| Row { facts: self, index })
1010    }
1011}

1013/// One borrowed argument column, over its arena's original admitted cells.
1014#[derive(Clone, Copy)]
1015pub(crate) struct FactColumn<'a> {
1016    facts: FactBlock<'a>,
1017    argument: usize,
1018}
1019
1020impl<'a> FactColumn<'a> {
1021    /// Compact product segments, or the run's scalar occurrences, in column
1022    /// order. An interval's last number is `first + scalars.len() - 1`, within
1023    /// i32 by construction. Its coordinates resolve only against the owning
1024    /// preparation's `TemplateComponentsRef`; no canonical-ID adjacency is assumed.
1025    pub(crate) fn segments(self) -> impl Iterator<Item = ColumnSegment> + use<'a> {
1026        let arena = self.facts.arena;
1027        let (segments, scalars) = match &self.facts.block().layout {
1028            Layout::Product { columns, .. } => {
1029                let column = &arena.columns[columns.start + self.argument];
1030                (&arena.segments[column.segments.clone()], &[][..])
1031            }
1032            Layout::Run { columns, .. } => (
1033                &[][..],
1034                arena.run_columns[columns.start + self.argument].as_slice(),
1035            ),
1036        };
1037        segments
1038            .iter()
1039            .map(|segment| segment.values)
1040            .chain(scalars.iter().copied().map(ColumnSegment::Scalar))
1041    }
1042}

1044/// One row of a fact block: one fact.
1045#[derive(Clone, Copy)]
1046pub(crate) struct Row<'a> {
1047    facts: FactBlock<'a>,
1048    index: usize,
1049}
1050
1051impl<'a> Row<'a> {
1052    /// The row's admitted cells, in argument order.
1053    pub(crate) fn cells(self) -> impl Iterator<Item = Scalar> + use<'a> {
1054        let arena = self.facts.arena;
1055        let block = self.facts.block();
1056        let index = self.index;
1057        (0..block.arity).map(move |column| match &block.layout {
1058            Layout::Product { columns, .. } => {
1059                let column = &arena.columns[columns.start + column];
1060                column.cell(&arena.segments, (index / column.stride) % column.width)
1061            }
1062            Layout::Run { columns, .. } => arena.run_columns[columns.start + column][index],
1063        })
1064    }
1065
1066    /// The evidence of the row's statement.
1067    pub(crate) fn evidence(self) -> &'a [ProgramSite] {
1068        let arena = self.facts.arena;
1069        match &self.facts.block().layout {
1070            Layout::Product { evidence, .. } => &arena.evidence[evidence.clone()],
1071            Layout::Run { ends, evidence, .. } => {
1072                let start = if self.index == 0 {
1073                    *evidence
1074                } else {
1075                    arena.ends[ends + self.index - 1]
1076                };
1077                &arena.evidence[start..arena.ends[ends + self.index]]
1078            }
1079        }
1080    }
1081
1082    /// The site of the row's statement: its evidence's first entry, as for the
1083    /// fact rule it replaces.
1084    pub(crate) fn location(self) -> ProgramSite {
1085        self.evidence()[0]
1086    }
1087}
