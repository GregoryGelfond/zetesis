1//! Theory and condition coordinates select the shared source authority.
2//!
3//! First emission fixes a local dense coordinate. Source identity discovery and
4//! support membership are separate; only sparse integer maps live in this owner.
5
6use crate::ProgramSite;
7use zetesis_core::Sign;
8use zetesis_core::catalog::AtomRef;
9
10use crate::formula_support::{Computation, Counters, SourceAtom, SourceSelection};
11use crate::{FormulaFailure, FormulaLimits, FormulaResource};
12
13pub(super) struct Catalog(SourceSelection);
14
15impl Catalog {
16    pub(super) fn new(
17        computation: &Computation<'_, '_>,
18        counters: &Counters,
19        limits: &FormulaLimits,
20        location: ProgramSite,
21    ) -> Result<Self, FormulaFailure> {
22        SourceSelection::new(computation, limits, counters, location).map(Self)
23    }
24
25    pub(super) fn len(&self) -> usize {
26        self.0.len()
27    }
28
29    pub(super) fn source(
30        &self,
31        local: usize,
32        counters: &mut Counters,
33        limits: &FormulaLimits,
34        location: ProgramSite,
35    ) -> Result<SourceAtom, FormulaFailure> {
36        self.0.source(local, limits, counters, location)
37    }
38
39    pub(super) fn get<'read>(
40        &self,
41        id: usize,
42        computation: &'read Computation<'_, '_>,
43        counters: &mut Counters,
44        limits: &FormulaLimits,
45        location: ProgramSite,
46    ) -> Result<AtomRef<'read>, FormulaFailure> {
47        self.0.atom(id, computation, limits, counters, location)
48    }
49
50    pub(super) fn position(
51        &self,
52        atom: &SourceAtom,
53        counters: &mut Counters,
54        limits: &FormulaLimits,
55        location: ProgramSite,
56    ) -> Result<Option<usize>, FormulaFailure> {
57        self.0.position(atom, limits, counters, location)
58    }
59
60    pub(super) fn insert(
61        &mut self,
62        atom: &SourceAtom,
63        bound: (FormulaResource, usize),
64        computation: &Computation<'_, '_>,
65        counters: &mut Counters,
66        limits: &FormulaLimits,
67        location: ProgramSite,
68    ) -> Result<(usize, bool), FormulaFailure> {
69        self.0
70            .insert(atom, bound, computation, limits, counters, location)
71    }
72
73    pub(super) fn find(
74        &self,
75        local: usize,
76        sign: Sign,
77        computation: &Computation<'_, '_>,
78        counters: &mut Counters,
79        limits: &FormulaLimits,
80        location: ProgramSite,
81    ) -> Result<Option<usize>, FormulaFailure> {
82        let source = self.0.source(local, limits, counters, location)?;
83        let Some(other) = computation.signed(&source, sign, limits, counters, location)? else {
84            return Ok(None);
85        };
86        self.0.position(&other, limits, counters, location)
87    }
88
89    pub(super) fn into_selection(self) -> SourceSelection {
90        self.0
91    }
92}
