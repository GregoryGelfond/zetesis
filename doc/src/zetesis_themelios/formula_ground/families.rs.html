41/// One completed carrier. The Binding retains its canonical vocabulary and its
42/// existing lease; the separate lease counts only this shared owner's remainder.
43pub(crate) struct Produced<'source> {
44    values: Binding<'static>,
45    pub(super) replay: Option<tuples::Replay<'source>>,
46    lease: StorageLease,
47}

49impl<'source> Produced<'source> {
50    /// Retain completed values in one accounted shared owner. Refusal drops the
51    /// unpublished values; the registry receives no partially built carrier.
52    fn new(
53        values: Binding<'static>,
54        replay: Option<tuples::Replay<'source>>,
55        context: Context<'_, &Computation<'_, '_>>,
56    ) -> Result<Rc<Self>, FormulaFailure> {
57        let Context { computation, work } = context;
58        let GroundingWork {
59            limits,
60            counters,
61            location,
62        } = work;
63        let lease = computation.lease();
64        counters.work(limits, location)?;
65        let replay_header = replay
66            .as_ref()
67            .map_or(0, |_| size_of::<tuples::Replay<'source>>());
68        let bytes =
69            size_of::<Self>() - size_of::<Binding>() - replay_header + 2 * size_of::<usize>();
70        admit_shared_owner(&lease, bytes, computation, limits, location)?;
71        counters.work(limits, location)?;
72        let mut produced = Rc::new(Produced {
73            values,
74            replay,
75            lease,
76        });
77        let allocated = Rc::get_mut(&mut produced).expect("new unpublished carrier owner");
78        allocated.lease.observe(bytes, location)?;
79        computation.storage_observed(&allocated.lease, 0, 0, limits, counters, location)?;
80        Ok(produced)
81    }
82}

84/// Growing queries retain the established local carrier; one final Builder can
85/// instead share its already completed stage. Both expose the same value frame.
86pub(crate) enum Carrier<'source> {
87    Owned(Binding<'static>),
88    Shared(Rc<Produced<'source>>),
89}
90impl Deref for Carrier<'_> {
91    type Target = Binding<'static>;
92    fn deref(&self) -> &Self::Target {
93        match self {
94            Self::Owned(values) => values,
95            Self::Shared(produced) => &produced.values,
96        }
97    }
98}
99impl Carrier<'_> {
100    pub(crate) fn owns_binding_header(&self) -> bool {
101        matches!(self, Self::Owned(_))
102    }
103}

105impl<'query, 'source> Workspace<'query, 'source> {
106    pub(super) fn new(
107        context: Context<'_, &Computation<'query, 'source>>,
108    ) -> Result<Rc<Self>, FormulaFailure> {
109        let Context { computation, work } = context;
110        let GroundingWork {
111            limits,
112            counters,
113            location,
114        } = work;
115        let terms = TermTable::new(computation, limits, counters, location)?;
116        let rows = Contexts::new(computation, limits, counters, location)?;
117        // TermTable and Contexts already account their own complete headers.
118        // Rc's two control words and all remaining inline fields are new owners.
119        let bytes =
120            size_of::<Self>() - size_of::<TermTable>() - size_of::<Contexts<Family<'source>>>()
121                + 2 * size_of::<usize>();
122        let lease = computation.lease();
123        admit_shared_owner(&lease, bytes, computation, limits, location)?;
124        counters.work(limits, location)?;
125        let mut owner = Rc::new(Self {
126            support: computation.query_support(),
127            enabled: Cell::new(false),
128            terms: RefCell::new(terms),
129            rows: RefCell::new(rows),
130            cached_elements: Cell::new(0),
131            cached_roots: Cell::new(0),
132            lease,
133        });
134        let allocated = Rc::get_mut(&mut owner).expect("new unpublished shared owner");
135        allocated.lease.observe(bytes, location)?;
136        computation.storage_observed(&allocated.lease, 0, 0, limits, counters, location)?;
137        Ok(owner)
138    }

167    /// Only the completed-support capability can enable producer retention.
168    /// Other builders still use this same owner's ordinary native cache.

170        &self,
171        queries: &CompletedQueries<'source>,

174        self.authenticate(queries.support(), location)?;
175        self.enabled.set(true);

179    pub(crate) fn authenticate(
180        &self,
181        support: &Support<'_>,
182        location: ProgramSite,
183    ) -> Result<(), FormulaFailure> {
184        if std::ptr::eq(
185            std::ptr::from_ref(self.support).cast::<()>(),
186            std::ptr::from_ref(support).cast::<()>(),
187        ) {
188            Ok(())
189        } else {
190            Err(FormulaFailure::SupportRelation {
191                error: zetesis_core::relation::Failure::Owner,
192                location,
193            })
194        }
195    }

197    pub(super) fn key(
198        &self,
199        inputs: &[usize],
200        assignment: &Binding,
201        context: Context<'_, &Computation<'_, '_>>,
202    ) -> Result<Buffer<Option<usize>>, FormulaFailure> {
203        let Context { computation, work } = context;
204        let GroundingWork {
205            limits,
206            counters,
207            location,
208        } = work;
209        self.authenticate(computation.query_support(), location)?;
210        self.terms(GroundingWork::new(limits, counters, location))?
211            .authenticate_owner(computation, limits, counters, location)?;
212        // Empty projections still authenticate the live vocabulary witness.
213        assignment
214            .slots()
215            .prefix(0)
216            .expect("the empty prefix exists")
217            .bind_with(computation.read(), || counters.work(limits, location))
218            .map_err(|error| crate::formula_binding::failure(error, location))?;
219        let mut key = Buffer::new(computation, limits, counters, location)?;
220        key.resize(inputs.len(), None, computation, limits, counters, location)?;
221        for (at, &slot) in inputs.iter().enumerate() {
222            counters.work(limits, location)?;
223            let value = assignment
224                .slots()
225                .key(slot)
226                .map_err(|error| crate::formula_binding::assignment(error, location))?;
227            key.slice_mut()[at] = value
228                .as_ref()
229                .map(|value| {
230                    self.terms_mut(GroundingWork::new(limits, counters, location))?
231                        .insert(value, None, computation, limits, counters, location)
232                })
233                .transpose()?;
234        }
235        Ok(key)
236    }

238    pub(crate) fn values(
239        self: &Rc<Self>,
240        aggregate: Ir<'_, 'source, AggregateIr>,
241        assignment: &Binding,
242        support: &Support<'source>,
243        budget: &mut Budget,
244        context: Context<'_, &mut Computation<'_, '_>>,
245    ) -> Result<Carrier<'source>, FormulaFailure> {
246        let Context { computation, work } = context;
247        let GroundingWork {
248            limits,
249            counters,
250            location,
251        } = work;
252        self.authenticate(support, location)?;
253        self.authenticate(computation.query_support(), location)?;
254        counters.work(limits, location)?;
255        let Ir::Source(source) = aggregate else {
256            return crate::formula_assignment::values(
257                aggregate,
258                assignment,
259                support,
260                budget,
261                Context::new(computation, limits, counters, location),
262            )
263            .map(Carrier::Owned);
264        };
265        let enabled = self.enabled.get();
266        #[cfg(test)]
267        let enabled = enabled && testing::enabled();
268        if !enabled {
269            return crate::formula_assignment::values(
270                aggregate,
271                assignment,
272                support,
273                budget,
274                Context::new(computation, limits, counters, location),
275            )
276            .map(Carrier::Owned);
277        }
278        let key = self.key(
279            &source.family_inputs,
280            assignment,
281            Context::new(&*computation, limits, counters, location),
282        )?;
283        {
284            counters.work(limits, location)?;
285            let rows = self.rows.borrow();
286            let slot = rows.find(source.id, &key, computation, limits, counters, location)?;
287            if let Some(slot) = slot {
288                let family = rows.get(slot).expect("published family coordinate");
289                if let Some(original) = family.source
290                    && !std::ptr::eq(original, source)
291                {
292                    return Err(FormulaFailure::SupportRelation {
293                        error: zetesis_core::relation::Failure::Owner,
294                        location,
295                    });
296                }
297                if let Some(produced) = &family.produced {
298                    counters.work(limits, location)?;
299                    #[cfg(test)]
300                    testing::record(testing::Event::Reused);
301                    return Ok(Carrier::Shared(Rc::clone(produced)));
302                }
303            }
304        }
305        // Registry and term borrows ended before complete source construction.
306        let produced = self.produce(
307            aggregate,
308            assignment,
309            support,
310            budget,
311            Context::new(&mut *computation, limits, counters, location),
312        )?;
313        counters.work(limits, location)?;
314        let mut rows = self.rows.borrow_mut();
315        // Nested construction may append rows; integer coordinates remain stable.
316        // A completed nested stage is never overwritten by this draft.
317        let current = rows.find(source.id, &key, computation, limits, counters, location)?;
318        if let Some(slot) = current {
319            let family = rows.get_mut(slot).expect("existing native family");
320            if let Some(existing) = &family.produced {
321                counters.work(limits, location)?;
322                #[cfg(test)]
323                testing::record(testing::Event::Reused);
324                return Ok(Carrier::Shared(Rc::clone(existing)));
325            }
326            counters.work(limits, location)?;
327            family.source = Some(source);
328            family.produced = Some(Rc::clone(&produced));
329        } else {
330            rows.insert(
331                source.id,
332                &key,
333                Family {
334                    source: Some(source),
335                    produced: Some(Rc::clone(&produced)),
336                    native: None,
337                },
338                Context::new(&*computation, limits, counters, location),
339            )?;
340        }
341        #[cfg(test)]
342        testing::record(testing::Event::Produced);
343        Ok(Carrier::Shared(produced))
344    }

346    /// Construct every producer-owned component before exposing a registry row.
347    /// Refusal may retain interned vocabulary coordinates, never a partial stage.
348    fn produce(
349        &self,
350        aggregate: Ir<'_, 'source, AggregateIr>,
351        assignment: &Binding,
352        support: &Support<'source>,
353        budget: &mut Budget,
354        context: Context<'_, &mut Computation<'_, '_>>,
355    ) -> Result<Rc<Produced<'source>>, FormulaFailure> {
356        let Context { computation, work } = context;
357        let GroundingWork {
358            limits,
359            counters,
360            location,
361        } = work;
362        let mut trace = tuples::Trace::new(
363            self,
364            Context::new(&*computation, limits, counters, location),
365        )?;
366        let collected = crate::formula_assignment::collect(
367            aggregate,
368            assignment,
369            support,
370            budget,
371            &mut trace,
372            Context::new(&mut *computation, limits, counters, location),
373        )?;
374        let values = collected.carrier(
375            crate::formula_assignment::Kind::of(aggregate.function),
376            Context::new(&mut *computation, limits, counters, location),
377        )?;
378        let replay = trace.finish(
379            &collected,
380            Context::new(&*computation, limits, counters, location),
381        )?;
382        drop(collected);
383        Produced::new(
384            values,
385            replay,
386            Context::new(&*computation, limits, counters, location),
387        )
388    }

391/// Admit a future owner without reporting its unallocated capacity as observed.
392/// The caller publishes the actual lease immediately after Rc allocation.
393fn admit_shared_owner(
394    lease: &StorageLease,
395    bytes: usize,
396    computation: &Computation<'_, '_>,
397    limits: &crate::FormulaLimits,
398    location: ProgramSite,
399) -> Result<(), FormulaFailure> {
400    let available = computation.allowance(lease, limits, location)?;
401    ceiling(
402        FormulaResource::SupportBytes,
403        (limits.max_support_bytes - available) as u128 + bytes as u128,
404        limits.max_support_bytes as u128,
405        location,
406    )
407}

413    #[derive(Clone, Copy, Debug, Default)]
414    pub(crate) struct Counts {
415        pub(crate) produced: usize,
416        pub(crate) reused: usize,
417        pub(crate) carrier_borrowed: usize,
418        pub(crate) consumed: usize,
419    }
420    #[derive(Clone, Copy)]
421    pub(crate) enum Event {
422        Produced,
423        Reused,
424        CarrierBorrowed,
425        Consumed,
426    }
427    thread_local! {
428        static ENABLED: Cell<bool> = const { Cell::new(true) };
429        static COUNTS: Cell<Counts> = const { Cell::new(Counts {
430            produced: 0, reused: 0, carrier_borrowed: 0, consumed: 0,
431        }) };
432    }
433    pub(super) fn enabled() -> bool {
434        ENABLED.get()
435    }
436    pub(crate) fn counts() -> Counts {
437        COUNTS.get()
438    }
439    pub(crate) fn record(event: Event) {
440        COUNTS.with(|cell| {
441            let mut counts = cell.get();
442            match event {
443                Event::Produced => counts.produced += 1,
444                Event::Reused => counts.reused += 1,
445                Event::CarrierBorrowed => counts.carrier_borrowed += 1,
446                Event::Consumed => counts.consumed += 1,
447            }
448            cell.set(counts);
449        });
450    }
451    pub(crate) fn scoped<T>(enabled: bool, action: impl FnOnce() -> T) -> T {
452        struct Restore(bool, Counts);
453        impl Drop for Restore {
454            fn drop(&mut self) {
455                ENABLED.set(self.0);
456                COUNTS.set(self.1);
457            }
458        }
459        let _restore = Restore(ENABLED.replace(enabled), COUNTS.replace(Counts::default()));
460        action()
461    }
462}
