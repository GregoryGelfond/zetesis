1//! Shared construction metadata, distinct from final root-owned evidence.
2//!
3//! Each atom owns two chain headers and a reference to its first source entry. Entries live
4//! in shared append arenas: producers retain insertion order and origins retain
5//! full `Location` order without duplicates. Links never cross arena owners.
6//! This removes per-atom staging allocations, at the cost of one link per entry.
7//! The arena remains live while the public root-owned origin vectors are built.
8

49impl Metadata {
50    /// The caller has admitted this atom and its first source occurrence.
51    /// Existing atom/producer work ticks cover fixed publication; shared arena
52    /// relocations, origin searches/inserts and final evidence copies are extra.
53    pub(super) fn atom(
54        &mut self,
55        location: Location,
56        counters: &mut Counters,
57        limits: &FormulaLimits,
58    ) -> Result<(), FormulaFailure> {
59        counters.charge_work(
60            growth(&self.atoms) + growth(&self.origins),
61            limits,
62            location,
63        )?;
64        reserve(&mut self.atoms, 1, location)?;
65        reserve(&mut self.origins, 1, location)?;
66        let first = append(&mut self.origins, location, None);
67        self.atoms.push(Atom {
68            first_origin: first,
69            producers: Chain::default(),
70            origins: Chain {
71                first: Some(first),
72                last: Some(first),
73                len: 1,
74            },
75        });
76        Ok(())
77    }
78
79    pub(super) fn location(&self, atom: usize) -> Location {
80        self.origins[self.atoms[atom].first_origin.get() - 1].value
81    }
82
83    pub(super) fn producer(
84        &mut self,
85        atom: usize,
86        antecedent: usize,
87        counters: &mut Counters,
88        limits: &FormulaLimits,
89        location: Location,
90    ) -> Result<(), FormulaFailure> {
91        counters.charge_work(growth(&self.producers), limits, location)?;
92        reserve(&mut self.producers, 1, location)?;
93        let entry = append(&mut self.producers, antecedent, None);
94        let chain = &mut self.atoms[atom].producers;
95        if let Some(last) = chain.last {
96            self.producers[last.get() - 1].next = Some(entry);
97        } else {
98            chain.first = Some(entry);
99        }
100        chain.last = Some(entry);
101        chain.len += 1;
102        Ok(())
103    }
104
105    pub(super) fn origin(
106        &mut self,
107        atom: usize,
108        location: Location,
109        budget: &mut Budget,
110        counters: &mut Counters,
111        limits: &FormulaLimits,
112    ) -> Result<(), FormulaFailure> {
113        let position = self.find_origin(atom, location);
114        let (previous, next, work) = match position {
115            Position::Present { work } => {
116                return counters.charge_work(work, limits, location);
117            }
118            Position::Vacant {
119                previous,
120                next,
121                work,
122            } => (previous, next, work),
123        };
124        // Keep first-occurrence admission ahead of the work refusal. Search is
125        // bounded by the already admitted origin chain; refusal publishes no link.
126        budget.charge(ExpansionResource::Origins, 1, location)?;
127        counters.charge_work(work + 1 + growth(&self.origins), limits, location)?;
128        reserve(&mut self.origins, 1, location)?;
129        let entry = append(&mut self.origins, location, next);
130        let chain = &mut self.atoms[atom].origins;
131        if let Some(previous) = previous {
132            self.origins[previous.get() - 1].next = Some(entry);
133        } else {
134            chain.first = Some(entry);
135        }
136        if next.is_none() {
137            chain.last = Some(entry);
138        }
139        chain.len += 1;
140        Ok(())
141    }
142
143    fn find_origin(&self, atom: usize, location: Location) -> Position {
144        let chain = self.atoms[atom].origins;
145        let last = chain.last.expect("every atom retains its first origin");
146        match location.cmp(&self.origins[last.get() - 1].value) {
147            Ordering::Equal => return Position::Present { work: 1 },
148            Ordering::Greater => {
149                return Position::Vacant {
150                    previous: Some(last),
151                    next: None,
152                    work: 1,
153                };
154            }
155            Ordering::Less => {}
156        }
157        let mut previous = None;
158        let mut next = chain.first;
159        let mut work = 1;
160        while let Some(index) = next {
161            let entry = &self.origins[index.get() - 1];
162            work += 1;
163            match location.cmp(&entry.value) {
164                Ordering::Equal => return Position::Present { work },
165                Ordering::Less => {
166                    return Position::Vacant {
167                        previous,
168                        next,
169                        work,
170                    };
171                }
172                Ordering::Greater => {
173                    previous = next;
174                    next = entry.next;
175                }
176            }
177        }
178        unreachable!("the larger tail bounds this sorted insertion search")
179    }

181    pub(super) fn producers(
182        &self,
183        atom: usize,
184    ) -> impl ExactSizeIterator<Item = usize> + FusedIterator + '_ {
185        Values::new(&self.producers, self.atoms[atom].producers)
186    }
187
188    pub(super) fn origins(
189        &self,
190        atom: usize,
191    ) -> impl ExactSizeIterator<Item = Location> + FusedIterator + '_ {
192        Values::new(&self.origins, self.atoms[atom].origins)
193    }
194
195    /// Called only after root-count and emitted-origin admission. Reserve the
196    /// required final view once; charge both link traversal and location copying.
197    pub(super) fn copy_origins(
198        &self,
199        atom: usize,
200        counters: &mut Counters,
201        limits: &FormulaLimits,
202    ) -> Result<Vec<Location>, FormulaFailure> {
203        let origins = self.origins(atom);
204        let location = self.location(atom);
205        counters.charge_work(origins.len() as u128 * 2, limits, location)?;
206        let mut result = Vec::new();
207        reserve(&mut result, origins.len(), location)?;
208        result.extend(origins);
209        Ok(result)
210    }
211}

213enum Position {
214    Present {
215        work: u128,
216    },
217    Vacant {
218        previous: Link,
219        next: Link,
220        work: u128,
221    },

224struct Values<'a, T> {
225    entries: &'a [Entry<T>],
226    next: Link,
227    remaining: usize,
228}
229
230impl<'a, T> Values<'a, T> {
231    fn new(entries: &'a [Entry<T>], chain: Chain) -> Self {
232        Self {
233            entries,
234            next: chain.first,
235            remaining: chain.len,
236        }
237    }
238}
239
240impl<T: Copy> Iterator for Values<'_, T> {
241    type Item = T;
242
243    fn next(&mut self) -> Option<T> {
244        let index = self.next?;
245        let entry = &self.entries[index.get() - 1];
246        self.next = entry.next;
247        self.remaining -= 1;
248        Some(entry.value)
249    }
250
251    fn size_hint(&self) -> (usize, Option<usize>) {
252        (self.remaining, Some(self.remaining))
253    }
254}
255
256impl<T: Copy> ExactSizeIterator for Values<'_, T> {}
257impl<T: Copy> FusedIterator for Values<'_, T> {}

259fn growth<T>(entries: &Vec<T>) -> u128 {
260    if entries.len() == entries.capacity() {
261        entries.len() as u128
262    } else {
263        0
264    }
265}
266
267fn reserve<T>(
268    entries: &mut Vec<T>,
269    additional: usize,
270    location: Location,
271) -> Result<(), FormulaFailure> {
272    entries
273        .try_reserve(additional)
274        .map_err(|error| FormulaFailure::MetadataAllocation { error, location })
275}
276
277/// The reservation already succeeded. Every entry is nonzero-sized, so Vec's
278/// isize-sized allocation bound makes len + 1 representable. Links encode
279/// absence explicitly; zero is never a valid arena position.
280fn append<T>(entries: &mut Vec<Entry<T>>, value: T, next: Link) -> NonZeroUsize {
281    let index = NonZeroUsize::new(entries.len() + 1).expect("one-based arena position");
282    entries.push(Entry { value, next });
283    index
