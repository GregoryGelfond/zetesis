1//! Complete positive witnesses share lazy formula construction with their join.
2//!
3//! Only a selected complete descendant may publish an atom or group. Later
4//! siblings reuse exactly the prefix left intact by join undo. Matched rows
5//! supply identities; support membership supplies no truth premise.
6//!
7//! Balanced groups have at most three children. Consecutive traversal columns
8//! split into two or three nearly equal ranges recursively, so native rows keep
9//! reusable subbodies rather than flattening every complete body. Metadata is
10//! O(body width), grouping depth logarithmic. A non-prefix group of two or
11//! three atomic witnesses can additionally retain reached row coordinates and
12//! the canonical node already compiled for them. Only successful complete
13//! descendants populate these maps; their separate storage and actual lookup,
14//! insertion and movement work belong to the existing source workspace.
15

30#[derive(Clone, Copy)]
31enum Input {
32    Atom(usize),
33    Group(usize),
34}
35
36#[derive(Clone, Copy)]
37enum Children {
38    Pair([Input; 2]),
39    Triple([Input; 3]),
40}
41impl Children {
42    fn slice(&self) -> &[Input] {
43        match self {
44            Self::Pair(inputs) => inputs,
45            Self::Triple(inputs) => inputs,
46        }
47    }
48}
49
50struct Group {
51    // Postorder ends are monotone; undo invalidates one suffix.
52    end: usize,
53    children: Children,
54    node: Option<usize>,
55    // A map is confined to this Body and therefore to one immutable ordered
56    // Join/support owner. Row positions from another occurrence/rule/snapshot
57    // can never query it. Its values borrow the Builder's append-only nodes;
58    // no graph rollback or owner replacement occurs during this Body's life.
59    compiled: Option<CoordinateMap<[usize; 3], usize>>,
60}
61
62impl Group {
63    fn reusable(&self) -> bool {
64        matches!(self.children.slice().first(), Some(Input::Atom(first)) if *first > 0)
65            && self
66                .children
67                .slice()
68                .iter()
69                .all(|input| matches!(input, Input::Atom(_)))
70    }
71}
72
73struct Body {
74    atoms: MatchedNodes,
75    groups: Buffer<Group>,
76}
77
78impl Body {
79    fn new(
80        width: usize,
81        builder: &mut Builder<'_, '_, '_>,
82        rule: &RuleIr,
83    ) -> Result<Self, FormulaFailure> {
84        let atoms = MatchedNodes::new(width, builder, rule.location)?;
85        let mut groups = Buffer::new(
86            builder.computation,
87            builder.limits,
88            builder.counters,
89            rule.location,
90        )?;
91        branch(0..width, &mut groups, builder, rule)?;
92        if groups.len() == 1 {
93            return Ok(Self { atoms, groups });
94        }
95        #[cfg(test)]
96        if !testing::reuse_enabled() {
97            return Ok(Self { atoms, groups });
98        }
99        for group in groups.slice_mut() {
100            builder.work(rule.location)?;
101            if !group.reusable() {
102                continue;
103            }
104            // The leased Group buffer already owns the full inline map header.
105            // No tuple or formula is precomputed from the possible populations.
106            group.compiled = Some(
107                CoordinateMap::new(
108                    builder.computation,
109                    builder.limits,
110                    builder.counters,
111                    rule.location,
112                )?
113                .into_embedded(rule.location)?,
114            );
115        }
116        Ok(Self { atoms, groups })
117    }
118
119    fn emit(
120        &mut self,
121        row: &witnesses::WitnessRow<'_, '_>,
122        builder: &mut Builder<'_, '_, '_>,
123        rule: &RuleIr,
124    ) -> Result<usize, FormulaFailure> {
125        #[cfg(test)]
126        if !testing::reuse_enabled() {
127            return self.prefix_emit(row, builder, rule);
128        }
129        if self.groups.len() == 1 {
130            return self.prefix_emit(row, builder, rule);
131        }
132        self.group(self.groups.len() - 1, row, builder, rule)
133    }
134
135    /// Recursion follows the existing balanced tree; its depth is logarithmic
136    /// in body width. A compiled leaf group owns its atom materialization as a
137    /// whole: hits never leave per-atom prefix state that a later miss could
138    /// mistakenly interpret as current.
139    fn group(
140        &mut self,
141        index: usize,
142        row: &witnesses::WitnessRow<'_, '_>,
143        builder: &mut Builder<'_, '_, '_>,
144        rule: &RuleIr,
145    ) -> Result<usize, FormulaFailure> {
146        builder.work(rule.location)?;
147        let group = &self.groups.slice()[index];
148        if group.end <= row.unchanged {
149            return Ok(group
150                .node
151                .expect("unchanged group has a selected predecessor"));
152        }
153        let children = group.children;
154        let mut key = [0; 3];
155        let compiled = group.compiled.is_some();
156        if let Some(map) = &group.compiled {
157            for (slot, &input) in children.slice().iter().enumerate() {
158                builder.work(rule.location)?;
159                let Input::Atom(at) = input else {
160                    unreachable!("only atomic leaf groups retain row coordinates")
161                };
162                key[slot] = row.rows[at].position();
163            }
164            #[cfg(test)]
165            testing::record(testing::Point::Lookup);
166            if let Some(node) = map.find(
167                &key,
168                builder.computation,
169                builder.limits,
170                builder.counters,
171                rule.location,
172            )? {
173                #[cfg(test)]
174                testing::record(testing::Point::Reuse);
175                // Observe cancellation at the reuse boundary before publication.
176                builder.work(rule.location)?;
177                self.groups.slice_mut()[index].node = Some(node);
178                return Ok(node);
179            }
180        }
181        let mut values = [0; 3];
182        for (slot, &input) in children.slice().iter().enumerate() {
183            builder.work(rule.location)?;
184            values[slot] = match input {
185                Input::Atom(at) => {
186                    #[cfg(test)]
187                    if compiled || at >= row.unchanged {
188                        testing::record(testing::Point::Atom);
189                    }
190                    if compiled {
191                        // A compiled group never seeds per-atom prefix state.
192                        builder.matched_atom(row.rows[at], rule.location)?
193                    } else {
194                        self.atoms.dispatched(at, row, builder, rule.location)?
195                    }
196                }
197                Input::Group(child) => self.group(child, row, builder, rule)?,
198            };
199        }
200        #[cfg(test)]
201        testing::record(testing::Point::Group);
202        // The original ordered positive operands, including duplicates, enter
203        // the same sole canonical node index. Support is never assumed true.
204        let node = builder.node(
205            NodeView::And(&values[..children.slice().len()]),
206            rule.location,
207        )?;
208        if let Some(map) = &mut self.groups.slice_mut()[index].compiled {
209            map.insert(
210                key,
211                node,
212                None,
213                Context::new(
214                    &*builder.computation,
215                    builder.limits,
216                    builder.counters,
217                    rule.location,
218                ),
219            )?;
220        }
221        self.groups.slice_mut()[index].node = Some(node);
222        Ok(node)
223    }
224
225    fn prefix_node(&self, input: Input) -> usize {
226        match input {
227            Input::Atom(index) => self.atoms.cached(index),
228            Input::Group(index) => self.groups.slice()[index]
229                .node
230                .expect("completed child group"),
231        }
232    }
233
234    fn prefix_emit(
235        &mut self,
236        row: &witnesses::WitnessRow<'_, '_>,
237        builder: &mut Builder<'_, '_, '_>,
238        rule: &RuleIr,
239    ) -> Result<usize, FormulaFailure> {
240        for index in row.unchanged..row.rows.len() {
241            #[cfg(test)]
242            testing::record(testing::Point::Atom);
243            self.atoms.refresh(index, row, builder, rule.location)?;
244        }
245        // Locate the changed suffix without rescanning every group. Every
246        // inspected boundary remains a charged operation.
247        let mut start = 0;
248        let mut end = self.groups.len();
249        while start < end {
250            builder.work(rule.location)?;
251            let middle = start + (end - start) / 2;
252            if self.groups.slice()[middle].end <= row.unchanged {
253                start = middle + 1;
254            } else {
255                end = middle;
256            }
257        }
258        for index in start..self.groups.len() {
259            let mut values = [0; 3];
260            let children = self.groups.slice()[index].children;
261            for (slot, &input) in children.slice().iter().enumerate() {
262                builder.work(rule.location)?;
263                values[slot] = self.prefix_node(input);
264            }
265            #[cfg(test)]
266            testing::record(testing::Point::Group);
267            // Children are atomic formulas or nonempty positive groups, never
268            // Boolean constants. Complete ordered duplicates remain operands.
269            let node = builder.node(
270                NodeView::And(&values[..children.slice().len()]),
271                rule.location,
272            )?;
273            self.groups.slice_mut()[index].node = Some(node);
274        }
275        Ok(self
276            .groups
277            .slice()
278            .last()
279            .expect("nonempty body groups")
280            .node
281            .expect("completed root group"))
282    }
283}

285/// At most three strictly smaller positive ranges per recursion give a
286/// logarithmic stack bound for every admitted source body width.
287fn branch(
288    range: Range<usize>,
289    groups: &mut Buffer<Group>,
290    builder: &mut Builder<'_, '_, '_>,
291    rule: &RuleIr,
292) -> Result<Input, FormulaFailure> {
293    builder.work(rule.location)?;
294    let width = range.len();
295    if width == 1 {
296        return Ok(Input::Atom(range.start));
297    }
298    let count = if width <= 3 {
299        width
300    } else {
301        width.div_ceil(3).clamp(2, 3)
302    };
303    let mut children = [Input::Atom(range.start); 3];
304    let mut start = range.start;
305    for (index, child) in children.iter_mut().enumerate().take(count) {
306        let end = start + width / count + usize::from(index < width % count);
307        *child = branch(start..end, groups, builder, rule)?;
308        start = end;
309    }
310    let children = if count == 2 {
311        Children::Pair([children[0], children[1]])
312    } else {
313        Children::Triple(children)
314    };
315    let index = groups.len();
316    groups.push(
317        Group {
318            end: range.end,
319            children,
320            node: None,
321            compiled: None,
322        },
323        builder.computation,
324        builder.limits,
325        builder.counters,
326        rule.location,
327    )?;
328    Ok(Input::Group(index))
329}

331/// Decline without advancing the supplied join when this class does not apply.
332pub(super) fn rule(
333    builder: &mut Builder<'_, '_, '_>,
334    rule: &RuleIr,
335    outer: &mut Join<'_, '_>,
336) -> Result<bool, FormulaFailure> {
337    #[cfg(test)]
338    if !testing::enabled() {
339        return Ok(false);
340    }
341    let Some(pattern) = witnesses::eligible(rule, builder.limits, builder.counters)? else {
342        return Ok(false);
343    };
344    debug_assert_eq!(rule.body_variables, rule.variables);
345    #[cfg(test)]
346    testing::record(testing::Point::Run);
347    outer.retain_witnesses(Context::new(
348        builder.computation,
349        builder.limits,
350        builder.counters,
351        rule.location,
352    ))?;
353    let mut prepared = None;
354    while let Some(row) = outer.next_witness_row(
355        None,
356        builder.budget,
357        Context::new(
358            builder.computation,
359            builder.limits,
360            builder.counters,
361            rule.location,
362        ),
363    )? {
364        debug_assert_eq!(row.rows.len(), rule.body.len());
365        #[cfg(test)]
366        testing::record(testing::Point::Witness);
367        if prepared.is_none() {
368            let head = RowHead::new(
369                pattern,
370                &row,
371                builder.computation,
372                builder.limits,
373                builder.counters,
374                rule.location,
375            )?;
376            prepared = Some((head, Body::new(row.rows.len(), builder, rule)?));
377        }
378        let (head, body) = prepared.as_mut().expect("selected producer preparation");
379        let body = body.emit(&row, builder, rule)?;
380        let atom = head.atom(
381            &row,
382            builder.computation,
383            builder.limits,
384            builder.counters,
385            rule.location,
386        )?;
387        let head = builder.atom_identity(&atom, rule.location)?;
388        let formula = builder.node(NodeView::Implies(body, head), rule.location)?;
389        builder.root(formula, rule)?;
390        builder.producer(head, body, rule)?;
391        #[cfg(test)]
392        testing::record(testing::Point::Publish);
393    }
394    Ok(true)
