1//! Scoped body validation and original-model objective truth share formula lowering.
2//!
3//! A transient builder borrows complete Support and selects the shared source
4//! authority while owning its node table and aggregate cache. Objective rows use their explicit scratch
5//! ceilings even when filters or numeric selection later discard them. Rejected
6//! rule rows with nested scopes or generators use the theory ceilings for
7//! their discarded validation scratch; flat rejected rows validate atom keys.
8//! Both share cumulative source work. Only final reachable objective Condition
9//! nodes consume the retained query ceiling. No roots, producers, coherence or
10//! support-guard completion from scratch reach the original program.

39/// Source activity uses the same complete scoped lowering as original-model
40/// queries, with theory scratch ceilings and no retained query publication.
41pub(crate) fn source_activity<'source>(
42    literals: &[LiteralIr],
43    binding: &Binding,
44    support: &Support<'source>,
45    eligibility: &SourceEligibility,
46    context: &mut Context<'_, '_, 'source>,
47) -> Result<Activity, FormulaFailure> {
48    validate_with_purpose(literals, binding, support, context, Purpose::Validation)?
49        .source_activity(eligibility, context)
50}

52pub(super) fn validate_with_purpose<'source>(
53    literals: &[LiteralIr],
54    binding: &Binding,
55    support: &Support<'source>,
56    context: &mut Context<'_, '_, 'source>,
57    purpose: Purpose,
58) -> Result<ValidatedBody, FormulaFailure> {
59    // The builder borrows the cumulative account through every outcome,
60    // including an unwind during scoped lowering.
61    let mut builder = Builder::empty(
62        context.computation,
63        context.limits,
64        context.budget,
65        context.counters,
66        purpose,
67        None,
68        context.location,
69    )?;
70    let result = (|| {
71        builder.initialize(context.location)?;
72        builder.body(literals, binding, context.location, support)
73    })();
74    let root = result?;
75    Ok(ValidatedBody {
76        atoms: builder.catalog,
77        nodes: builder.nodes.into_parts(),
78        root,
79    })
80}

83    /// Fold independent source-atom possibilities over the lowered DAG. An
84    /// optional atom and its negation remain optional even when correlated;
85    /// this is neither satisfiability nor a frozen-reduct evaluation.
86    pub(super) fn source_activity(
87        &self,
88        eligibility: &SourceEligibility,
89        context: &mut Context<'_, '_, '_>,
90    ) -> Result<Activity, FormulaFailure> {
91        let count = self.root + 1;
92        let mut values = Buffer::new(
93            context.computation,
94            context.limits,
95            context.counters,
96            context.location,
97        )?;
98        values.resize(
99            count,
100            Activity::Absent,
101            context.computation,
102            context.limits,
103            context.counters,
104            context.location,
105        )?;
106        for index in 0..count {
107            let node = self.nodes.view().node(index).expect("owned formula node");
108            context.counters.work(context.limits, context.location)?;
109            let activity = match node {
110                Node::False => Activity::Absent,
111                Node::Atom(atom) => {
112                    let source = self.atoms.source(
113                        atom,
114                        context.counters,
115                        context.limits,
116                        context.location,
117                    )?;
118                    eligibility.source_activity(&source, context)?
119                }
120                Node::And(row) | Node::Or(row) => {
121                    let conjunction = matches!(node, Node::And(_));
122                    let mut activity = if conjunction {
123                        Activity::Required
124                    } else {
125                        Activity::Absent
126                    };
127                    for &child in row {
128                        context.counters.work(context.limits, context.location)?;
129                        activity = if conjunction {
130                            activity.min(values.slice()[child])
131                        } else {
132                            activity.max(values.slice()[child])
133                        };
134                    }
135                    activity
136                }
137                Node::Threshold(row) => {
138                    let mut lower = 0_i128;
139                    let mut upper = 0_i128;
140                    for (&condition, &weight) in row.conditions.iter().zip(row.weights) {
141                        context.counters.work(context.limits, context.location)?;
142                        let activity = values.slice()[condition];
143                        if activity == Activity::Required {
144                            lower += i128::from(weight);
145                        }
146                        if activity != Activity::Absent {
147                            upper += i128::from(weight);
148                        }
149                    }
150                    if lower >= row.bound {
151                        Activity::Required
152                    } else if upper < row.bound {
153                        Activity::Absent
154                    } else {
155                        Activity::Optional
156                    }
157                }
158                Node::ExtremumCut(cut) => {
159                    let mut activity = match cut.endpoint_truth() {
160                        Some(true) => Activity::Required,
161                        Some(false) | None => Activity::Absent,
162                    };
163                    if cut.endpoint_truth().is_none() {
164                        for (&condition, &rank) in cut.row.conditions.iter().zip(cut.row.ranks) {
165                            context.counters.work(context.limits, context.location)?;
166                            if cut.qualifies(rank) {
167                                activity = activity.max(values.slice()[condition]);
168                            }
169                        }
170                    }
171                    activity
172                }
173                Node::Implies(left, right) => {
174                    context
175                        .counters
176                        .charge_work(2, context.limits, context.location)?;
177                    values.slice()[left].negate().max(values.slice()[right])
178                }
179            };
180            values.slice_mut()[index] = activity;
181        }
182        Ok(values.slice()[self.root])
183    }

185    /// A canonical false body cannot contribute a source projection instance.
186    pub(super) fn is_false(&self) -> bool {
187        self.root == super::FALSUM
188    }
189    /// Boolean implication is translated only for a complete original model.
190    /// The same rewrite is invalid for frozen Ferraris reducts.
191    /// Select the complete backward-closed formula slice before translating it.
192    /// Physical numeric children remain selected even when a guard is constant.
193    fn ancestor_mask(
194        &self,
195        context: &mut Context<'_, '_, '_>,
196    ) -> Result<Buffer<bool>, FormulaFailure> {
197        let mut needed = Buffer::new(
198            context.computation,
199            context.limits,
200            context.counters,
201            context.location,
202        )?;
203        needed.resize(
204            self.nodes.view().len(),
205            false,
206            context.computation,
207            context.limits,
208            context.counters,
209            context.location,
210        )?;
211        context.counters.work(context.limits, context.location)?;
212        needed.slice_mut()[self.root] = true;
213        // Backward edges make one descending pass sufficient to find ancestors.
214        for index in (0..=self.root).rev() {
215            context.counters.work(context.limits, context.location)?;
216            if needed.slice()[index] {
217                let pair;
218                let row = match self.nodes.view().node(index).expect("owned formula node") {
219                    Node::And(row) | Node::Or(row) => row,
220                    Node::Threshold(row) => row.conditions,
221                    Node::ExtremumCut(cut) => cut.row.conditions,
222                    Node::Implies(a, b) => {
223                        pair = [a, b];
224                        &pair
225                    }
226                    Node::Atom(_) | Node::False => &[],
227                };
228                for &child in row {
229                    context.counters.work(context.limits, context.location)?;
230                    needed.slice_mut()[child] = true;
231                }
232            }
233        }
234        Ok(needed)
235    }

237    pub(super) fn condition(
238        self,
239        context: &mut Context<'_, '_, '_>,
240    ) -> Result<PendingCondition, FormulaFailure> {
241        let needed = self.ancestor_mask(context)?;
242        let mut mapping = Buffer::new(
243            context.computation,
244            context.limits,
245            context.counters,
246            context.location,
247        )?;
248        mapping.resize(
249            self.root + 1,
250            0,
251            context.computation,
252            context.limits,
253            context.counters,
254            context.location,
255        )?;
256        let mut query = Query::new(context)?;
257        for index in 0..=self.root {
258            let node = self.nodes.view().node(index).expect("owned formula node");
259            context.counters.work(context.limits, context.location)?;
260            if !needed.slice()[index] {
261                continue;
262            }
263            let node = match node {
264                Node::False => ConditionNode::Boolean(false),
265                Node::Atom(atom) => ConditionNode::Atom(atom),
266                Node::And(row) | Node::Or(row) => {
267                    let conjunction = matches!(node, Node::And(_));
268                    let mut previous = None;
269                    for &child in row {
270                        context.counters.work(context.limits, context.location)?;
271                        let next = mapping.slice()[child];
272                        previous = Some(match previous {
273                            None => next,
274                            Some(previous) => query.node(
275                                if conjunction {
276                                    ConditionNode::And(previous, next)
277                                } else {
278                                    ConditionNode::Or(previous, next)
279                                },
280                                context,
281                            )?,
282                        });
283                    }
284                    mapping.slice_mut()[index] =
285                        previous.expect("stored groups have at least two children");
286                    continue;
287                }
288                Node::Threshold(row) => {
289                    mapping.slice_mut()[index] =
290                        threshold::lower(row, mapping.slice(), &mut query, context)?;
291                    continue;
292                }
293                Node::ExtremumCut(cut) => {
294                    mapping.slice_mut()[index] =
295                        threshold::lower_extremum(cut, mapping.slice(), &mut query, context)?;
296                    continue;
297                }
298                Node::Implies(left, right) => {
299                    let negated = query.node(ConditionNode::Not(mapping.slice()[left]), context)?;
300                    ConditionNode::Or(negated, mapping.slice()[right])
301                }
302            };
303            mapping.slice_mut()[index] = query.node(node, context)?;
304        }
305        // The root has the greatest reachable index and is appended last.
306        debug_assert_eq!(mapping.slice()[self.root] + 1, query.len());
307        Ok(PendingCondition::new(self.atoms.into_selection(), query))
308    }
