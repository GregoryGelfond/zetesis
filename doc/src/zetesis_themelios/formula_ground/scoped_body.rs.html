1//! Scoped body validation and original-model objective truth share formula lowering.
2//!
3//! Scoped builders borrow complete Support and select the shared source
4//! authority while owning their node tables and aggregate caches. One source
5//! producer pass retains its immutable lowering over completed Support. Staged
6//! activity shares completed readings until publication; direct exact publication
7//! requires fresh readings. Independent validation keeps transient builders.
8//! Objective rows use their explicit scratch ceilings even when filters or
9//! numeric selection later discard them. Rejected rule rows with nested scopes
10//! or generators use the theory ceilings for their discarded validation scratch;
11//! flat rejected rows validate atom keys.
12//! Both share cumulative source work. Only final reachable objective Condition
13//! nodes consume the retained query ceiling. No roots, producers, coherence or
14//! support-guard completion from scratch reach the original program.

38pub(super) fn validate<'source>(
39    literals: &[LiteralIr],
40    binding: &Binding,
41    support: &Support<'source>,
42    context: &mut Context<'_, '_, 'source>,
43) -> Result<ValidatedBody, FormulaFailure> {
44    validate_with_purpose(literals, binding, support, context, Purpose::Objective)
45}

47pub(super) fn validate_with_purpose<'source>(
48    literals: &[LiteralIr],
49    binding: &Binding,
50    support: &Support<'source>,
51    context: &mut Context<'_, '_, 'source>,
52    purpose: Purpose,
53) -> Result<ValidatedBody, FormulaFailure> {
54    // The builder borrows the cumulative account through every outcome,
55    // including an unwind during scoped lowering.
56    let mut builder = Builder::empty(
57        context.computation,
58        context.limits,
59        context.budget,
60        context.counters,
61        purpose,
62        None,
63        context.location,
64    )?;
65    let result = (|| {
66        builder.initialize(context.location)?;
67        builder.body(literals, binding, context.location, support)
68    })();
69    let root = result?;
70    Ok(ValidatedBody {
71        atoms: builder.catalog,
72        nodes: builder.nodes.into_parts(),
73        root,
74    })
75}

94    /// A canonical false body cannot contribute a source projection instance.
95    pub(super) fn is_false(&self) -> bool {
96        self.root == super::FALSUM
97    }
98    /// Boolean implication is translated only for a complete original model.
99    /// The same rewrite is invalid for frozen Ferraris reducts.
100    /// Select the complete backward-closed formula slice before translating it.
101    /// Physical numeric children remain selected even when a guard is constant.
102    fn ancestor_mask(
103        &self,
104        context: &mut Context<'_, '_, '_>,
105    ) -> Result<Buffer<bool>, FormulaFailure> {
106        let mut needed = Buffer::new(
107            context.computation,
108            context.limits,
109            context.counters,
110            context.location,
111        )?;
112        needed.resize(
113            self.nodes.view().len(),
114            false,
115            context.computation,
116            context.limits,
117            context.counters,
118            context.location,
119        )?;
120        context.counters.work(context.limits, context.location)?;
121        needed.slice_mut()[self.root] = true;
122        // Backward edges make one descending pass sufficient to find ancestors.
123        for index in (0..=self.root).rev() {
124            context.counters.work(context.limits, context.location)?;
125            if needed.slice()[index] {
126                let pair;
127                let row = match self.nodes.view().node(index).expect("owned formula node") {
128                    Node::And(row) | Node::Or(row) => row,
129                    Node::Threshold(row) => row.conditions,
130                    Node::ExtremumCut(cut) => cut.row.conditions,
131                    Node::Implies(a, b) => {
132                        pair = [a, b];
133                        &pair
134                    }
135                    Node::Atom(_) | Node::False => &[],
136                };
137                for &child in row {
138                    context.counters.work(context.limits, context.location)?;
139                    needed.slice_mut()[child] = true;
140                }
141            }
142        }
143        Ok(needed)
144    }
145
146    pub(super) fn condition(
147        self,
148        context: &mut Context<'_, '_, '_>,
149    ) -> Result<PendingCondition, FormulaFailure> {
150        let needed = self.ancestor_mask(context)?;
151        let mut mapping = Buffer::new(
152            context.computation,
153            context.limits,
154            context.counters,
155            context.location,
156        )?;
157        mapping.resize(
158            self.root + 1,
159            0,
160            context.computation,
161            context.limits,
162            context.counters,
163            context.location,
164        )?;
165        let mut query = Query::new(context)?;
166        for index in 0..=self.root {
167            let node = self.nodes.view().node(index).expect("owned formula node");
168            context.counters.work(context.limits, context.location)?;
169            if !needed.slice()[index] {
170                continue;
171            }
172            let node = match node {
173                Node::False => ConditionNode::Boolean(false),
174                Node::Atom(atom) => ConditionNode::Atom(atom),
175                Node::And(row) | Node::Or(row) => {
176                    let conjunction = matches!(node, Node::And(_));
177                    let mut previous = None;
178                    for &child in row {
179                        context.counters.work(context.limits, context.location)?;
180                        let next = mapping.slice()[child];
181                        previous = Some(match previous {
182                            None => next,
183                            Some(previous) => query.node(
184                                if conjunction {
185                                    ConditionNode::And(previous, next)
186                                } else {
187                                    ConditionNode::Or(previous, next)
188                                },
189                                context,
190                            )?,
191                        });
192                    }
193                    mapping.slice_mut()[index] =
194                        previous.expect("stored groups have at least two children");
195                    continue;
196                }
197                Node::Threshold(row) => {
198                    mapping.slice_mut()[index] =
199                        threshold::lower(row, mapping.slice(), &mut query, context)?;
200                    continue;
201                }
202                Node::ExtremumCut(cut) => {
203                    mapping.slice_mut()[index] =
204                        threshold::lower_extremum(cut, mapping.slice(), &mut query, context)?;
205                    continue;
206                }
207                Node::Implies(left, right) => {
208                    let negated = query.node(ConditionNode::Not(mapping.slice()[left]), context)?;
209                    ConditionNode::Or(negated, mapping.slice()[right])
210                }
211            };
212            mapping.slice_mut()[index] = query.node(node, context)?;
213        }
214        // The root has the greatest reachable index and is appended last.
215        debug_assert_eq!(mapping.slice()[self.root] + 1, query.len());
216        Ok(PendingCondition::new(self.atoms.into_selection(), query))
217    }
218}
