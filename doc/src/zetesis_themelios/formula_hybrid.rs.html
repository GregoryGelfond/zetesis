74/// A materialized producer core with the integrity constraints streamed over it:

96/// One immutable original program: a materialized producer core and prepared
97/// integrity constraints over the same atom envelope. Admission closes the
98/// completed support, keeping the canonical base and only the relations the
99/// streamed constraints read; when no constraint is streamed, the support and
100/// plan storage are released instead.
101///
102/// Cloning shares all retained source, atoms and indexes. The core's answer sets
103/// are proposals; only those satisfying the streamed constraints are answer
104/// sets of this owner. Objectives are scored only after this complete check.
105#[derive(Clone)]
106pub struct HybridFormula(Arc<Admitted>);

112            .field("streamed_instances", &self.streamed_instances())
113            .finish_non_exhaustive()
114    }

278/// One check's join-planning and scalar budget; shared scalar receipts are
279/// reported to `allowance`. Planning uses the caller's work ceiling, not an
280/// unrelated source-admission default.
281fn check_budget(limits: ConstraintCheckLimits, allowance: Option<ConstraintAllowance>) -> Budget {
282    let budget = Budget::new(
283        ExpansionLimits {
284            max_term_work: usize::try_from(limits.max_work).unwrap_or(usize::MAX),
285            max_scalar_bytes: limits.max_scalar_bytes,
286            ..ExpansionLimits::default()
287        },
288        0,
289    );
290    match allowance {
291        Some(allowance) => budget.with_allowance(allowance),
292        None => budget,
293    }
294}

390    /// Prepare one independent cumulative checker over the completed support.

404    /// Prepare an independent checker whose charges share `allowance` with all
405    /// other attached checkers, including final full-model checking. Cancellation
406    /// is polled during charged setup work and before publishing the checker.
407    /// Accepted setup charges are published before return, including on failure.
408    /// Checks publish their charges when they return, fail or unwind; live
409    /// shared statistics can lag the operation's exact local receipt.
410    ///
411    /// # Errors
412    /// Returns a typed preparation/resource/control refusal with its local
413    /// accepted receipt. `allowance.statistics()` retains the combined receipt.
414    pub fn checker_with_allowance(
415        &self,
416        allowance: &ConstraintAllowance,
417        cancellation: &Cancellation,
418    ) -> Result<ConstraintChecker<'_>, ConstraintCheckFailure> {
419        cancellation.poll().map_err(|stop| ConstraintCheckFailure {
420            cause: ConstraintCheckCause::Stopped(stop),
421            statistics: ConstraintCheckStatistics::default(),
422        })?;
423        let limits = allowance.limits();
424        let checker = self.prepare_checker(
425            limits,
426            Counters::with_allowance(allowance.clone(), cancellation),
427            Some(allowance.clone()),
428        )?;
429        cancellation.poll().map_err(|stop| ConstraintCheckFailure {
430            cause: ConstraintCheckCause::Stopped(stop),
431            statistics: checker.statistics(),
432        })?;
433        Ok(checker)
434    }
435
436    fn prepare_checker(
437        &self,
438        limits: ConstraintCheckLimits,
439        mut counters: Counters,
440        allowance: Option<ConstraintAllowance>,
441    ) -> Result<ConstraintChecker<'_>, ConstraintCheckFailure> {
442        let prepared = if let Some(constraints) = &self.0.constraints {
443            let mut formula_limits = constraints.limits;
444            formula_limits.max_work = limits.max_work;
445            formula_limits.max_substitutions = limits.max_substitutions;
446            let completed = constraints
447                .support
448                .snapshot(&formula_limits, &mut counters, constraints.location)
449                .map_err(|error| ConstraintCheckFailure {
450                    cause: ConstraintCheckCause::Source(Box::new(
451                        self.0.source.retain_failure(error),
452                    )),
453                    statistics: ConstraintCheckStatistics {
454                        work: counters.accounting.work,
455                        substitutions: counters.accounting.substitutions,
456                        scalar_bytes: 0,
457                    },
458                })?;
459            Some(PreparedConstraints {
460                source: constraints,
461                completed,
462                limits: formula_limits,
463                index: None,
464                rows: None,
465                plans: Vec::new(),
466                predicates: Vec::new(),
467                shared_incremental: &self.0.incremental,
468                incremental_plan: None,
469                incremental: None,
470                incremental_eligible: None,
471            })
472        } else {
473            None
474        };
475        Ok(ConstraintChecker {
476            owner: self,
477            prepared,
478            budget: check_budget(limits, allowance.clone()),
479            limits,
480            allowance,
481            settled: (0, 0),
482            settled_scalar_bytes: 0,
483            consequence_active: false,
484            accounting: counters.into_accounting(),
485        })
486    }
487}

489/// Allowances for each check of a checker (one model, region-refutation scan,
490/// or complete candidate closure including all of its consequence passes),
491/// independent of source admission and reduct-oracle work. A checker's first
492/// check also covers its preparation; each later check is measured from the
493/// charges accepted when the previous one ended. They bound the work spent on
494/// any one candidate, never the number of candidates. Zero is a zero
495/// allowance, never unlimited. Per-operation binding/storage bounds remain
496/// those admitted with the source.
497#[derive(Clone, Copy, Debug, PartialEq, Eq)]
498pub struct ConstraintCheckLimits {
499    /// Charged join, scalar and typed atom-lookup work in one check.
500    /// The same allowance independently bounds source join-planning term work.
501    pub max_work: u64,
502    /// Complete substitutions visited in one check, including false filters.
503    pub max_substitutions: u64,
504    /// Bytes requested in one check for capture-delta cells during structural-pattern
505    /// matching, charged through source expansion's `ScalarBytes` resource.
506    /// The historical field name is retained; these cells borrow canonical
507    /// terms. ID-only binding copies and frozen constructor lookups add no
508    /// scalar-byte charge. Their storage uses the admitted support allowance.
509    pub max_scalar_bytes: usize,
510}
511impl ConstraintCheckLimits {
512    /// Finite defaults matching ordinary source work and expansion allowances.
513    pub const DEFAULT: Self = Self {
514        max_work: 10_000_000,
515        max_substitutions: 1_000_000,
516        max_scalar_bytes: 16_777_216,
517    };
518}
519impl Default for ConstraintCheckLimits {
520    fn default() -> Self {
521        Self::DEFAULT
522    }
523}

563/// A check stopped without establishing satisfaction or a violation.
564#[derive(Debug)]
565pub enum ConstraintCheckCause {
566    /// The model catalog or supplied region theory has a different owner.
567    WrongProgram,
568    /// A continuation was requested without an unfinished consequence closure.
569    NoActiveRegion,
570    /// Region coordinates do not span the authenticated original atom catalog.
571    WrongRegionSize {
572        /// Original catalog dimension.
573        expected: usize,
574        /// Supplied region dimension.
575        actual: usize,
576    },
577    /// Preparation of the bounded catalog lookup refused allocation or found
578    /// duplicate typed atoms. Source-work stops use `Source` instead.
579    Index(AtomIndexError<Box<FormulaFailure>>),
580    /// Caller cancellation or deadline, never semantic rejection.
581    Stopped(Stop),
582    /// Located join/evaluation/allocation/resource refusal.
583    Source(Box<FormulaFailure>),
584}
585impl ConstraintCheckCause {
586    /// Report a work or substitution refusal against the check's own
587    /// allowance: subtract the charges `start` accepted before the check.
588    fn relative_to(self, start: (u64, u64)) -> Self {
589        let relative = |error: Box<FormulaFailure>| match *error {
590            FormulaFailure::Limit {
591                resource:
592                    resource @ (crate::FormulaResource::Work | crate::FormulaResource::Substitutions),
593                limit,
594                observed,
595                location,
596            } => {
597                let base = u128::from(if resource == crate::FormulaResource::Work {
598                    start.0
599                } else {
600                    start.1
601                });
602                Box::new(FormulaFailure::Limit {
603                    resource,
604                    limit: limit.saturating_sub(base),
605                    observed: observed.saturating_sub(base),
606                    location,
607                })
608            }
609            other => Box::new(other),
610        };
611        match self {
612            Self::Source(error) => Self::Source(relative(error)),
613            Self::Index(AtomIndexError::Stopped(error)) => {
614                Self::Index(AtomIndexError::Stopped(relative(error)))
615            }
616            other => other,
617        }
618    }
619
620    fn retain_input(self, owner: &Owner) -> Self {
621        match self {
622            Self::Source(error) => Self::Source(Box::new(owner.retain_failure(*error))),
623            Self::Index(AtomIndexError::Stopped(error)) => Self::Index(AtomIndexError::Stopped(
624                Box::new(owner.retain_failure(*error)),
625            )),
626            other => other,
627        }
628    }
629}
630impl fmt::Display for ConstraintCheckCause {
631    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
632        match self {
633            Self::WrongProgram => f.write_str("constraint candidate belongs to another program"),
634            Self::NoActiveRegion => {
635                f.write_str("constraint consequence has no active region closure")
636            }
637            Self::WrongRegionSize { expected, actual } => {
638                write!(
639                    f,
640                    "constraint region has {actual} atoms; expected {expected}"
641                )
642            }
643            Self::Index(error) => error.fmt(f),
644            Self::Stopped(stop) => stop.fmt(f),
645            Self::Source(error) => error.fmt(f),
646        }
647    }
648}
649impl std::error::Error for ConstraintCheckCause {
650    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
651        match self {
652            Self::WrongProgram | Self::NoActiveRegion | Self::WrongRegionSize { .. } => None,
653            Self::Index(error) => Some(error),
654            Self::Stopped(error) => Some(error),
655            Self::Source(error) => Some(error.as_ref()),
656        }
657    }

660/// Typed refusal together with all charges accepted before it.
661#[derive(Debug)]
662pub struct ConstraintCheckFailure {
663    /// Exact cause; the original non-clone allocation error is preserved.
664    pub cause: ConstraintCheckCause,
665    /// Cumulative checker charges, including earlier complete checks.
666    pub statistics: ConstraintCheckStatistics,
667}
668impl ConstraintCheckFailure {
669    /// Cancellation/deadline classification without discarding the source error.
670    #[must_use]
671    pub fn stop(&self) -> Option<Stop> {
672        match &self.cause {
673            ConstraintCheckCause::Stopped(reason) => Some(*reason),
674            ConstraintCheckCause::Source(error)
675            | ConstraintCheckCause::Index(AtomIndexError::Stopped(error)) => error.interruption(),
676            ConstraintCheckCause::WrongProgram
677            | ConstraintCheckCause::NoActiveRegion
678            | ConstraintCheckCause::WrongRegionSize { .. }
679            | ConstraintCheckCause::Index(_) => None,
680        }
681    }
682}
683impl fmt::Display for ConstraintCheckFailure {
684    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
685        self.cause.fmt(f)
686    }
687}

689    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
690        Some(&self.cause)
691    }

694/// Mutable per-session work and scalar state over an immutable admitted owner.
695/// No grounded constraint instances are retained. Pending consequences belong
696/// to one authenticated closure. Successfully completed no-consequence scans
697/// can survive into another closure when both decision masks authenticate a
698/// narrowing and the affected reads permit reuse. Ordinary checks invalidate
699/// candidate evidence.
700/// The checker can move between threads. Grounding observation and each check's
701/// runtime control remain local to the synchronous operation that uses them.
702pub struct ConstraintChecker<'a> {
703    owner: &'a StreamedCore,
704    prepared: Option<PreparedConstraints<'a>>,
705    /// The current check's scalar budget, shared by every consequence pass.
706    budget: Budget,
707    /// The ceilings each check gets for itself.
708    limits: ConstraintCheckLimits,
709    /// The shared receipt this checker reports to, if any.
710    allowance: Option<ConstraintAllowance>,
711    /// Work and substitutions accepted when the previous check ended: the
712    /// next check's allowance is measured from here, so the first check's
713    /// includes this checker's preparation.
714    settled: (u64, u64),
715    /// Scalar bytes requested by every finished check's budget.
716    settled_scalar_bytes: usize,
717    /// An explicit First pass opened one candidate closure. Its counters and
718    /// scalar budget survive Hold/Cut results and unwind until that closure ends.
719    consequence_active: bool,
720    accounting: Accounting,
721}
722/// A nonempty source and its borrowed snapshot are present or absent together.
723struct PreparedConstraints<'a> {
724    source: &'a Constraints,
725    completed: CompletedSupport<'a>,
726    limits: FormulaLimits,
727    /// Original dense IDs, borrowed from the core's index on first candidate use
728    /// (and built there by the first checker that needs it). Final-model row
729    /// selection and complete bodies query the model's canonical true set.
730    index: Option<&'a CatalogIndex>,
731    /// Source occurrence IDs mapped once into the original dense catalog.
732    rows: Option<SourceRows<'a>>,
733    /// Lazily prepared after each rule's first successful predicate gate.
734    plans: Vec<Option<PreparedRule<'a>>>,
735    /// Occurrence-local predicate windows, borrowed from external source/index
736    /// owners and prepared before the rule's first region predicate gate.
737    predicates: Vec<Option<RulePredicates<'a>>>,
738    /// This exact core alone publishes the immutable mapping. Admitting a
739    /// borrow once also survives discarded mutable state after an unwind.
740    shared_incremental: &'a OnceLock<consequences::IncrementalPlan>,
741    incremental_plan: Option<&'a consequences::IncrementalPlan>,
742    incremental: Option<consequences::Incremental<'a>>,
743    incremental_eligible: Option<bool>,
744}
745
746impl<'a> PreparedConstraints<'a> {
747    fn selection<'check>(
748        &'check self,
749        rule_index: usize,
750        region: &'check Region,
751    ) -> Selection<'check, 'a> {
752        Selection {
753            rows: self.rows.as_ref().expect("prepared before region scan"),
754            index: self.index.expect("prepared before region scan").lookup(),
755            predicates: self.predicates.get(rule_index).and_then(Option::as_ref),
756            region,
757        }
758    }
759
760    /// Resolve every atom occurrence of this rule before its first region gate.
761    /// This performs no binding or scalar evaluation. It may charge/refuse
762    /// preparation for later occurrences before an earlier predicate would have
763    /// rejected the rule; successful views retain no candidate decisions.
764    fn prepare_predicates(
765        &mut self,
766        rule_index: usize,
767        counters: &mut Counters,
768    ) -> Result<(), FormulaFailure> {
769        if self.predicates.is_empty() {
770            self.predicates = RulePredicates::slots(
771                &self.source.rules,
772                &mut self.completed,
773                &self.limits,
774                counters,
775            )?;
776        }
777        if self.predicates[rule_index].is_none() {
778            self.predicates[rule_index] = Some(RulePredicates::new(
779                &self.source.rules[rule_index],
780                self.index.expect("prepared before region scan").lookup(),
781                &mut self.completed,
782                &self.limits,
783                counters,
784            )?);
785        }
786        Ok(())
787    }
788
789    fn prepare_rule(
790        &mut self,
791        rule_index: usize,
792        budget: &mut Budget,
793        counters: &mut Counters,
794    ) -> Result<(), FormulaFailure> {
795        if self.plans.is_empty() {
796            self.plans = PreparedRule::slots(
797                &self.source.rules,
798                &mut self.completed,
799                &self.limits,
800                counters,
801            )?;
802        }
803        if self.plans[rule_index].is_none() {
804            self.plans[rule_index] = Some(PreparedRule::new(
805                &self.source.rules[rule_index],
806                &mut self.completed,
807                &self.limits,
808                budget,
809                counters,
810            )?);
811        }
812        Ok(())
813    }
814
815    fn prepare_selection(
816        &mut self,
817        core: &'a StreamedCore,
818        counters: &mut Counters,
819    ) -> Result<(), ConstraintCheckCause> {
820        self.prepare_index(core, counters)?;
821        if self.rows.is_none() {
822            let positions = self
823                .prepare_positions(core, counters)
824                .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
825            let rows = SourceRows::attach(
826                &self.completed,
827                positions,
828                &self.limits,
829                counters,
830                self.source.location,
831            )
832            .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
833            // Every reservation checked its actual capacity before any publication.
834            self.completed.retain_workspace(
835                usize::try_from(rows.retained_bytes()).expect("admitted support bytes fit usize"),
836            );
837            self.rows = Some(rows);
838        }
839        Ok(())
840    }
841
842    /// Borrow the core's row positions, building them first when no checker
843    /// has: one charged index probe per kept support row, against this
844    /// checker's work and support ceilings. Races, refusals and the ledger
845    /// follow [`Self::prepare_index`].
846    fn prepare_positions(
847        &mut self,
848        core: &'a StreamedCore,
849        counters: &mut Counters,
850    ) -> Result<&'a selection::RowPositions, FormulaFailure> {
851        let location = self.source.location;
852        counters.work(&self.limits, location)?;
853        if core.0.rows.get().is_none() {
854            let built = selection::RowPositions::prepare(
855                &self.completed,
856                self.index.expect("prepared before its rows").lookup(),
857                &self.limits,
858                counters,
859                location,
860            )?;
861            // A racing checker may have published first: keep its positions.
862            let _ = core.0.rows.set(built);
863        }
864        let positions = core
865            .0
866            .rows
867            .get()
868            .expect("published above or by a racing checker");
869        let retained = positions.retained_bytes();
870        self.completed
871            .admit_workspace(retained, &self.limits, counters, location)?;
872        self.completed
873            .retain_workspace(usize::try_from(retained).expect("admitted support bytes fit usize"));
874        Ok(positions)
875    }
876
877    /// Borrow the core's index, building it first when no checker has. A
878    /// borrow costs one unit of work; a build costs O(n log n) comparisons
879    /// over the core's n atoms, charged to this checker, which also admits
880    /// its preparation peak. Checkers racing on the first use each build and
881    /// one publishes; none waits on another, so each observes its own
882    /// cancellation. A refused build publishes nothing. Either way the
883    /// checker's support ledger counts the index's retained bytes, as it
884    /// counts the shared support base it reads.
885    fn prepare_index(
886        &mut self,
887        core: &'a StreamedCore,
888        counters: &mut Counters,
889    ) -> Result<(), ConstraintCheckCause> {
890        if self.index.is_some() {
891            return Ok(());
892        }
893        let location = self.source.location;
894        counters
895            .work(&self.limits, location)
896            .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
897        if core.0.index.get().is_none() {
898            let built = self.build_index(&core.0.compiled.atoms, counters)?;
899            // A racing checker may have published first: keep its index.
900            let _ = core.0.index.set(built);
901        }
902        let index = core
903            .0
904            .index
905            .get()
906            .expect("published above or by a racing checker");
907        let retained = index.retained_bytes();
908        self.completed
909            .admit_workspace(retained, &self.limits, counters, location)
910            .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
911        self.completed
912            .retain_workspace(usize::try_from(retained).expect("admitted support bytes fit usize"));
913        self.index = Some(index);
914        Ok(())
915    }
916
917    fn build_index(
918        &mut self,
919        atoms: &AtomCatalog,
920        counters: &mut Counters,
921    ) -> Result<CatalogIndex, ConstraintCheckCause> {
922        let location = self.source.location;
923        // Two retained integer orders and one preparation scratch order, all
924        // bounded by the admitted atom count. The index reserves fallibly and
925        // charges each comparison/write; it never copies atom payloads.
926        let requested = size_of::<CatalogIndex>() as u128
927            + size_of::<Vec<usize>>() as u128
928            + 3 * atoms.atoms().len() as u128 * size_of::<usize>() as u128;
929        self.completed
930            .admit_workspace(requested, &self.limits, counters, location)
931            .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
932        let index = CatalogIndex::new_with(atoms, || {
933            counters.work(&self.limits, location).map_err(Box::new)
934        })
935        .map_err(|error| match error {
936            AtomIndexError::Stopped(error) => ConstraintCheckCause::Source(error),
937            other => ConstraintCheckCause::Index(other),
938        })?;
939        self.completed
940            .admit_workspace(
941                index.preparation_peak_bytes(),
942                &self.limits,
943                counters,
944                location,
945            )
946            .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))?;
947        Ok(index)
948    }
949}

951#[derive(Clone, Copy)]
952enum Candidate<'a> {
953    Model(&'a Model),
954    Region(&'a Theory, &'a Region),
955}

957impl ConstraintChecker<'_> {
958    /// Current cumulative resource receipt, including failed-check prefixes.
959    #[must_use]
960    pub fn statistics(&self) -> ConstraintCheckStatistics {
961        ConstraintCheckStatistics {
962            work: self.accounting.work,
963            substitutions: self.accounting.substitutions,
964            scalar_bytes: self
965                .settled_scalar_bytes
966                .saturating_add(self.budget.usage().scalar_bytes),
967        }
968    }
969
970    /// Check original constraint satisfaction against an exact-catalog model.
971    /// Joins borrow completed possible support. After exact source-row mapping,
972    /// positive rows absent from the supplied model are excluded before matching.
973    /// Unmapped rows retain the ordinary path. Every completed body separately
974    /// queries that same model; row selection lends no positive truth proof.
975    /// Shared support is never treated as candidate truth. A violation returns
976    /// early; successful satisfaction exhausts every necessary family.
977    /// Runtime control is polled at charged source work boundaries and before
978    /// publishing either verdict. Source arithmetic admission has already run.
979    ///
980    /// # Errors
981    /// Wrong catalog, cancellation/deadline or a checked resource/evaluation
982    /// refusal. A failed scan certifies neither satisfaction nor exhaustion.
983    pub fn check(
984        &mut self,
985        model: &Model,
986        cancellation: &Cancellation,
987    ) -> Result<ConstraintVerdict, ConstraintCheckFailure> {
988        self.examine(Candidate::Model(model), cancellation)
989            .map(|location| match location {
990                Some(site) => ConstraintVerdict::Violated { site },
991                None => ConstraintVerdict::Satisfied,
992            })
993    }
994
995    /// Refute an original candidate region only when one admitted constraint
996    /// body is true throughout it. Positive and double-negated atoms must be
997    /// held; default-negated atoms must be cut. Open atoms establish neither.
998    ///
999    /// The supplied theory must be this owner's exact core, and the region must
1000    /// span its dense catalog. These checks authenticate the coordinate convention,
1001    /// not the provenance of a raw `Region`. No region or candidate is mutated.
1002    /// Never use this operation to read a candidate's frozen reduct.
1003    ///
1004    /// First use borrows the core's typed catalog index and source-row
1005    /// positions (building each when no checker of this core has) and pairs
1006    /// the positions with this checker's occurrence maps; later checks reuse
1007    /// both. Necessary predicate and held-row selections
1008    /// precede binding/scalar evaluation, over the same completed support and
1009    /// shared evaluator as `check`. Missing row IDs remain eligible until the
1010    /// full body check. Source arithmetic admission has already completed.
1011    /// Failure preserves every accepted charge; `NotRefuted` does not prove
1012    /// satisfaction. This does not construct a filtered support certificate.
1013    ///
1014    /// # Errors
1015    /// Wrong theory/dimension, cancellation, or a checked index/source refusal.
1016    pub fn check_region(
1017        &mut self,
1018        theory: &Theory,
1019        region: &Region,
1020        cancellation: &Cancellation,
1021    ) -> Result<ConstraintRegionVerdict, ConstraintCheckFailure> {
1022        self.examine(Candidate::Region(theory, region), cancellation)
1023            .map(|location| match location {
1024                Some(site) => ConstraintRegionVerdict::Refuted { site },
1025                None => ConstraintRegionVerdict::NotRefuted,
1026            })
1027    }
1028
1029    fn authenticate(&self, candidate: Candidate<'_>) -> Result<(), ConstraintCheckCause> {
1030        match candidate {
1031            Candidate::Model(model) => {
1032                if !model.catalog().same_owner(self.owner.atom_catalog()) {
1033                    return Err(ConstraintCheckCause::WrongProgram);
1034                }
1035            }
1036            Candidate::Region(theory, region) => {
1037                if !theory.same_instance(self.owner.core_theory()) {
1038                    return Err(ConstraintCheckCause::WrongProgram);
1039                }
1040                let expected = self.owner.atom_catalog().atoms().len();
1041                if region.len() != expected {
1042                    return Err(ConstraintCheckCause::WrongRegionSize {
1043                        expected,
1044                        actual: region.len(),
1045                    });
1046                }
1047            }
1048        }
1049        Ok(())
1050    }
1051
1052    fn examine(
1053        &mut self,
1054        candidate: Candidate<'_>,
1055        cancellation: &Cancellation,
1056    ) -> Result<Option<ProgramSite>, ConstraintCheckFailure> {
1057        self.invalidate_consequences();
1058        if self.consequence_active {
1059            self.settle_check();
1060        }
1061        let start = self.prepare_pass(cancellation);
1062        let result = self.authenticate(candidate).and_then(|()| {
1063            cancellation.poll().map_err(ConstraintCheckCause::Stopped)?;
1064            let verdict = self
1065                .accounting
1066                .with_cancellation(cancellation, |counters| {
1067                    if let Some(prepared) = &mut self.prepared {
1068                        prepared.prepare_selection(self.owner, counters)?;
1069                    }
1070                    self.budget
1071                        .with_settled_receipt(|budget| {
1072                            Self::scan(self.prepared.as_mut(), budget, counters, candidate)
1073                        })
1074                        .map_err(|error| ConstraintCheckCause::Source(Box::new(error)))
1075                })?;
1076            cancellation.poll().map_err(ConstraintCheckCause::Stopped)?;
1077            Ok(verdict)
1078        });
1079        let statistics = self.statistics();
1080        self.finish_check();
1081        result.map_err(|cause| ConstraintCheckFailure {
1082            cause: cause.relative_to(start).retain_input(&self.owner.0.source),
1083            statistics,
1084        })
1085    }
1086
1087    /// Publication at each pass is separate from renewal at the closure boundary.
1088    fn prepare_pass(&mut self, cancellation: &Cancellation) -> (u64, u64) {
1089        let start = self.settled;
1090        self.budget.set_cancellation(Some(cancellation.clone()));
1091        if let Some(prepared) = &mut self.prepared {
1092            prepared.limits.max_work = start.0.saturating_add(self.limits.max_work);
1093            prepared.limits.max_substitutions =
1094                start.1.saturating_add(self.limits.max_substitutions);
1095        }
1096        start
1097    }
1098
1099    fn invalidate_consequences(&mut self) {
1100        if let Some(prepared) = &mut self.prepared
1101            && let Some(incremental) = &mut prepared.incremental
1102        {
1103            incremental.reset();
1104        }
1105    }
1106
1107    fn finish_check(&mut self) {
1108        self.invalidate_consequences();
1109        self.settle_check();
1110    }
1111
1112    /// Renew one closure's allowance independently of retained semantic evidence.
1113    fn settle_check(&mut self) {
1114        self.settled = (self.accounting.work, self.accounting.substitutions);
1115        self.settled_scalar_bytes = self.statistics().scalar_bytes;
1116        self.budget = check_budget(self.limits, self.allowance.clone());
1117        self.consequence_active = false;
1118    }
1119
1120    fn scan(
1121        prepared: Option<&mut PreparedConstraints<'_>>,
1122        budget: &mut Budget,
1123        counters: &mut Counters,
1124        candidate: Candidate<'_>,
1125    ) -> Result<Option<ProgramSite>, FormulaFailure> {
1126        let Some(prepared) = prepared else {
1127            return Ok(None);
1128        };
1129        let constraints = prepared.source;
1130        for (rule_index, rule) in constraints.rules.iter().enumerate() {
1131            if matches!(candidate, Candidate::Region(..)) {
1132                prepared.prepare_predicates(rule_index, counters)?;
1133            }
1134            if let Candidate::Region(_, region) = candidate
1135                && !prepared.selection(rule_index, region).possible(
1136                    rule,
1137                    &prepared.completed,
1138                    &prepared.limits,
1139                    counters,
1140                )?
1141            {
1142                continue;
1143            }
1144            prepared.prepare_rule(rule_index, budget, counters)?;
1145            let model_selection;
1146            let region_selection;
1147            let filter: &dyn RowFilter = match candidate {
1148                Candidate::Model(model) => {
1149                    model_selection = ModelSelection {
1150                        rows: prepared.rows.as_ref().expect("prepared before model scan"),
1151                        model,
1152                    };
1153                    &model_selection
1154                }
1155                Candidate::Region(_, region) => {
1156                    region_selection = prepared.selection(rule_index, region);
1157                    &region_selection
1158                }
1159            };
1160            let queries = prepared.completed.queries(
1161                crate::JoinStrategy::Indexed,
1162                &prepared.limits,
1163                counters,
1164                rule.location,
1165            )?;
1166            let mut computation = queries.computation(rule.location)?;
1167            let mut join = prepared.plans[rule_index]
1168                .as_ref()
1169                .expect("prepared above")
1170                .rows(
1171                    &queries,
1172                    Some(filter),
1173                    &computation,
1174                    &prepared.limits,
1175                    budget,
1176                    counters,
1177                )?;
1178            while let Some(row) = join.next_row(
1179                &mut computation,
1180                &prepared.limits,
1181                budget,
1182                counters,
1183                rule.location,
1184            )? {
1185                if row.passes
1186                    && body(
1187                        &rule.body,
1188                        &row.values,
1189                        row.positives.as_ref(),
1190                        candidate,
1191                        prepared.index.map(CatalogIndex::lookup),
1192                        prepared.predicates.get(rule_index).and_then(Option::as_ref),
1193                        Context::new(&computation, &prepared.limits, counters, rule.location),
1194                    )?
1195                {
1196                    return Ok(Some(rule.location));
1197                }
1198            }
1199        }
1200        Ok(None)
1201    }
1202}

1204pub(crate) fn eligible(rule: &RuleIr) -> bool {
1205    matches!(rule.head, HeadIr::Normal(None))
1206        && rule.body.iter().all(|literal| {
1207            !matches!(
1208                literal,
1209                LiteralIr::ProjectedAtom(..) | LiteralIr::Conditional(_) | LiteralIr::Aggregate(_)
1210            )
1211        })
1212}

1214/// `scan` pairs a row loan with the same immutable region borrowed by its
1215/// filter, and with that row's binding. Body identity authenticates the literal
1216/// occurrences; it does not establish region identity independently. Models
1217/// always query their own interpretation, even if a loan is supplied.
1218fn body(
1219    literals: &[LiteralIr],
1220    binding: &Binding<'_>,
1221    positives: Option<&PositiveRows<'_>>,
1222    candidate: Candidate<'_>,
1223    index: Option<AtomLookup<'_, '_>>,
1224    predicates: Option<&RulePredicates<'_>>,
1225    context: Context<'_, &crate::formula_support::Computation<'_, '_>>,
1226) -> Result<bool, FormulaFailure> {
1227    let Context {
1228        computation,
1229        work:
1230            GroundingWork {
1231                limits,
1232                counters,
1233                location,
1234            },
1235    } = context;
1236    // Positive source evidence is local to this exact body and immutable
1237    // region. A final model always authenticates its own selected atoms.
1238    let positives = matches!(candidate, Candidate::Region(..))
1239        && positives.is_some_and(|proof| proof.covers(literals));
1240    let mut view = None;
1241    for (occurrence, literal) in literals.iter().enumerate() {
1242        let Some((negation, pattern)) = literal_atom(literal) else {
1243            continue;
1244        };
1245        counters.work(limits, location)?;
1246        if positives && negation == DefaultNegation::None {
1247            continue;
1248        }
1249        let binding_view = if let Some(view) = view {
1250            view
1251        } else {
1252            let checked = binding.view(computation.read(), limits, counters, location)?;
1253            view = Some(checked);
1254            checked
1255        };
1256        let key = computation
1257            .static_pattern(*pattern, limits, counters, location)?
1258            .key(binding_view)
1259            .map_err(|error| FormulaFailure::UnsafeVariable {
1260                variable: error.variable,
1261                location,
1262            })?;
1263        let truth = match candidate {
1264            Candidate::Model(model) => {
1265                let present = model
1266                    .lookup()
1267                    .get_key_with(&key, || counters.work(limits, location))?
1268                    .is_some();
1269                match negation {
1270                    DefaultNegation::Not => !present,
1271                    DefaultNegation::None | DefaultNegation::NotNot => present,
1272                }
1273            }
1274            Candidate::Region(_, region) => {
1275                let row = if let Some(prepared) =
1276                    predicates.and_then(|prepared| prepared.at(literals, occurrence))
1277                {
1278                    prepared.get_key_with(&key, || counters.work(limits, location))?
1279                } else {
1280                    index
1281                        .expect("prepared before region scan")
1282                        .get_key_with(&key, || counters.work(limits, location))?
1283                };
1284                // Passing source rows contributed every occurrence to the
1285                // completed catalog at admission, including unsupported atoms.
1286                let position = row
1287                    .ok_or(FormulaFailure::SupportRelation {
1288                        error: zetesis_core::relation::Failure::Owner,
1289                        location,
1290                    })?
1291                    .position();
1292                match negation {
1293                    DefaultNegation::Not => region.is_cut(position),
1294                    DefaultNegation::None | DefaultNegation::NotNot => region.is_held(position),
1295                }
1296            }
1297        };
1298        if !truth {
1299            return Ok(false);
1300        }
1301    }
1302    Ok(true)
1303}
1304
1305fn literal_atom(
1306    literal: &LiteralIr,
1307) -> Option<(
1308    DefaultNegation,
1309    &crate::formula_support::components::Pattern,
1310)> {
1311    match literal {
1312        LiteralIr::Atom(negation, pattern) => Some((*negation, pattern)),
1313        LiteralIr::PatternAtom(pattern) => Some((DefaultNegation::None, &pattern.atom)),
1314        // The join owns scalar comparisons and generated bindings.
1315        LiteralIr::Compare(..)
1316        | LiteralIr::TupleCompare(..)
1317        | LiteralIr::ArgumentCheck { .. }
1318        | LiteralIr::Guard(_)
1319        | LiteralIr::HeadGuard(_)
1320        | LiteralIr::Bind { .. }
1321        | LiteralIr::Range { .. } => None,
1322        LiteralIr::ProjectedAtom(..) | LiteralIr::Conditional(_) | LiteralIr::Aggregate(_) => {
1323            unreachable!("only admitted ordinary constraints are streamed")
1324        }
1325    }
