1//! Necessary occurrence reads refine a bounded set of changed atoms.
2//!
3//! This replaces the existing per-predicate flag, not the dependency owner.
4//! Overflow keeps ordinary invalidation. A change that makes an occurrence
5//! false only disables body instances, regardless of its default negation.
6//! Enabling positive occurrences may anchor a completed negative scan; enabling
7//! default-negated occurrences still require a full scan. Generated rules
8//! decline relevant changes. Slots, including repeated
9//! names and lowered constructor temporaries, are independent wildcards here:
10//! failing a constant or constructor check excludes a read; passing proves none.
11
12use super::{ChangedAtoms, PreparedConstraints, literal_atom};
13use crate::FormulaFailure;
14use crate::formula_ir::{LiteralIr, RuleIr};
15use crate::formula_pattern::{ArgumentPattern, PatternNode};
16use crate::formula_support::{Counters, GroundingWork};
17use themelios_program::program::DefaultNegation;
18use zetesis_core::catalog::{AtomRef, TermRef};
19use zetesis_core::{TemplateComponentsRef, TemplateTerm};

21#[derive(Clone, Copy)]
22pub(super) enum Change {
23    None,
24    Atoms(ChangedAtoms),
25    /// Changes exceeded the bounded set; no complete atom list is retained.
26    Untracked,
27}
28
29impl Change {
30    pub(super) fn include(&mut self, atom: usize) {
31        *self = match *self {
32            Self::None => Self::Atoms(ChangedAtoms::one(atom)),
33            Self::Atoms(mut atoms) => {
34                if atoms.include(atom) {
35                    Self::Atoms(atoms)
36                } else {
37                    Self::Untracked
38                }
39            }
40            Self::Untracked => Self::Untracked,
41        };
42    }
43}

45/// The ordered dependency row has one group per original atom occurrence.
46/// Both it and the canonical changed IDs belong to this exact prepared core.
47/// Clean means no possible read changed or every affected occurrence became
48/// false. `PositiveDelta` covers enabling positive occurrences with retained
49/// newly held atoms. Enabling default-negated occurrences require Full.
50/// Reuse requires monotone, consistent bounds and completed negative evidence;
51/// Clean also preserves a completed batch pending application of all its units.
52/// Full declines reuse. Relevant generated frames also decline; unchanged
53/// generated rules retain their prior clean evidence.
54pub(super) fn classify(
55    rule: &RuleIr,
56    dependencies: &[usize],
57    changes: &[Change],
58    prepared: &PreparedConstraints<'_>,
59    region: &zetesis_cpu::regions::Region,
60    counters: &mut Counters,
61) -> Result<super::Scan, FormulaFailure> {
62    let mut groups = dependencies.iter();
63    let mut required = super::Scan::Clean;
64    let mut generated = false;
65    let mut relevant = false;
66    for literal in &rule.body {
67        counters.work(&prepared.limits, rule.location)?;
68        generated |= crate::formula_binding_cursor::target(literal).is_some();
69        if literal_atom(literal).is_none() {
70            continue;
71        }
72        let &group = groups.next().expect("one group per original occurrence");
73        match changes[group] {
74            Change::None => {}
75            Change::Untracked => return Ok(super::Scan::Full),
76            Change::Atoms(atoms) => {
77                for &position in atoms.as_slice() {
78                    counters.work(&prepared.limits, rule.location)?;
79                    let atom = prepared
80                        .index
81                        .expect("prepared source owner")
82                        .catalog()
83                        .atoms()
84                        .at(position)
85                        .expect("authenticated changed atom");
86                    let components = prepared.completed.components().ok_or_else(|| {
87                        crate::formula_support::components::missing(rule.location)
88                    })?;
89                    if may_read(
90                        literal,
91                        atom,
92                        components,
93                        &mut GroundingWork::new(&prepared.limits, counters, rule.location),
94                    )? {
95                        relevant = true;
96                        let (negation, _) = literal_atom(literal).expect("an atom occurrence");
97                        match region.decision(position) {
98                            // Disabling one read cannot create a violation or
99                            // unit. Keep inspecting: an alias later in this
100                            // body can read the same atom with another polarity.
101                            Some(held) if held != (negation != DefaultNegation::Not) => {}
102                            Some(true) if negation == DefaultNegation::None => required
103                                .include(super::Scan::PositiveDelta(ChangedAtoms::one(position))),
104                            Some(_) | None => return Ok(super::Scan::Full),
105                        }
106                        if required == super::Scan::Full {
107                            return Ok(required);
108                        }
109                    }
110                }
111            }
112        }
113    }
114    debug_assert!(groups.next().is_none());
115    Ok(if generated && relevant {
116        super::Scan::Full
117    } else {
118        required
119    })
120}

122/// Only immutable constants and constructor descriptors constrain the read.
123/// Canonical term comparison/navigation supplies equality and charges. There is
124/// no binding, term construction, captured equality, allocation or truth read.
125fn may_read(
126    literal: &LiteralIr,
127    atom: AtomRef<'_>,
128    components: TemplateComponentsRef<'_>,
129    work: &mut GroundingWork<'_>,
130) -> Result<bool, FormulaFailure> {
131    let (_, pattern) = literal_atom(literal).expect("an original atom occurrence");
132    let pattern = pattern.get(components, work.limits, work.counters, work.location)?;
133    // The plan already establishes signed predicate equality for this group.
134    for (column, term) in pattern.terms().iter().enumerate() {
135        work.counters.work(work.limits, work.location)?;
136        if let TemplateTerm::Constant(expected) = term {
137            let actual = atom.values().get(column).expect("same predicate arity");
138            if !actual
139                .equals_ref_with(expected, || work.counters.work(work.limits, work.location))?
140            {
141                return Ok(false);
142            }
143        }
144    }
145    if let LiteralIr::PatternAtom(pattern) = literal {
146        for argument in &pattern.arguments {
147            work.counters.work(work.limits, work.location)?;
148            let value = atom
149                .values()
150                .get(argument.position)
151                .expect("admitted argument");
152            if !argument_matches(argument, value, components, work)? {
153                return Ok(false);
154            }
155        }
156    }
157    Ok(true)
158}

160fn argument_matches(
161    pattern: &ArgumentPattern,
162    value: TermRef<'_>,
163    components: TemplateComponentsRef<'_>,
164    work: &mut GroundingWork<'_>,
165) -> Result<bool, FormulaFailure> {
166    let mut offset = 0;
167    for node in &pattern.nodes {
168        let Some(actual) =
169            value.subterm_with(offset, || work.counters.work(work.limits, work.location))?
170        else {
171            return Ok(false);
172        };
173        work.counters.work(work.limits, work.location)?;
174        match node {
175            PatternNode::Constructor(constructor) => {
176                let descriptor = actual.descriptor();
177                work.counters.charge_work(
178                    descriptor.text_bytes() as u128,
179                    work.limits,
180                    work.location,
181                )?;
182                let expected =
183                    constructor.get(components, work.limits, work.counters, work.location)?;
184                if descriptor != expected {
185                    return Ok(false);
186                }
187                offset += 1;
188            }
189            PatternNode::Constant(constant) => {
190                let expected =
191                    constant.get(components, work.limits, work.counters, work.location)?;
192                if !actual
193                    .equals_ref_with(expected, || work.counters.work(work.limits, work.location))?
194                {
195                    return Ok(false);
196                }
197                offset += actual.expanded_nodes();
198            }
199            PatternNode::Slot(_) | PatternNode::Wildcard => offset += actual.expanded_nodes(),
200        }
201    }
202    Ok(offset == value.expanded_nodes())
203}
