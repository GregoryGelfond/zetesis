1//! Scoped optimization observations reuse rule bodies without semantic producers.
2//!
3//! Its compiler has a private empty value domain. Objective-only constants and
4//! aggregate tuples therefore never enlarge the original program's universe.

20impl Compiler<'_> {
21    pub(super) fn objective_statement(
22        &mut self,
23        statement: &WithProvenance<Statement>,
24        origins: &[Location],
25        objectives: &mut Vec<ObjectiveIr>,
26        declarations: &mut Vec<Location>,
27        projection_nodes: &mut u128,
28    ) -> Result<Option<Vec<WithProvenance<Statement>>>, FormulaFailure> {
29        if let Statement::WeakConstraint(weak) = statement.get()
30            && selection::scoped(weak, self.budget, self.location)?
31        {
32            let analyzed = self.conditional_projection(statement, projection_nodes)?;
33            self.weak_objective(weak, origins, objectives, declarations)?;
34            return Ok(Some(analyzed));
35        }
36        let normalized = crate::formula_weak::normalize(statement, self.budget, self.location)?;
37        let statement = normalized.as_ref().unwrap_or(statement);
38        if let Statement::Optimize(optimize) = statement.get() {
39            self.objectives(
40                optimize,
41                origins,
42                &crate::extended::parsed_origins(statement),
43                objectives,
44                declarations,
45            )?;
46            return Ok(Some(
47                self.conditional_projection(statement, projection_nodes)?,
48            ));
49        }
50        Ok(None)
51    }

53    pub(super) fn weak_objective(
54        &mut self,
55        weak: &WeakConstraint,
56        origins: &[Location],
57        objectives: &mut Vec<ObjectiveIr>,
58        declarations: &mut Vec<Location>,
59    ) -> Result<(), FormulaFailure> {
60        let body_origins = || {
61            weak.body().provenance().origins().chain(
62                weak.body()
63                    .get()
64                    .elements()
65                    .flat_map(|element| element.provenance().origins()),
66            )
67        };
68        self.budget.charge(
69            ExpansionResource::Origins,
70            (origins.len() as u128)
71                .saturating_mul(2)
72                .saturating_add(body_origins().count() as u128),
73            self.location,
74        )?;
75        let mut evidence = Vec::new();
76        evidence
77            .try_reserve_exact(origins.len().saturating_add(body_origins().count()))
78            .map_err(|_| FormulaFailure::Objective {
79                error: zetesis_objective::AdmissionError::Allocation,
80                location: self.location,
81            })?;
82        evidence.extend_from_slice(origins);
83        for origin in weak.body().provenance().origins().chain(
84            weak.body()
85                .get()
86                .elements()
87                .flat_map(|element| element.provenance().origins()),
88        ) {
89            if let Origin::Parsed(location) = origin {
90                evidence.push(*location);
91            }
92        }
93        evidence.sort_unstable();
94        evidence.dedup();
95        let fields: Vec<_> = std::iter::once(weak.weight().term())
96            .chain(weak.weight().priority())
97            .chain(weak.terms())
98            .collect();
99        declarations.extend_from_slice(origins);
100        let family = ObjectiveFamily(objectives.len());
101        for body in self.body_alternatives(weak.body().get())? {
102            let alternatives = self.local_alternatives(&fields, &Condition::new([]))?;
103            for (terms, _) in &alternatives {
104                let mut terms = terms.iter();
105                let mut weight =
106                    themelios_program::program::weight(terms.next().expect("weight field").clone());
107                if weak.weight().priority().is_some() {
108                    weight = weight.at_priority(terms.next().expect("priority field").clone());
109                }
110                self.budget.charge(
111                    ExpansionResource::Origins,
112                    evidence.len() as u128,
113                    self.location,
114                )?;
115                let objective = self.scoped_objective(
116                    &body,
117                    &weight,
118                    terms,
119                    evidence.clone(),
120                    WeightPolarity::AsWritten,
121                    family,
122                )?;
123                objectives.push(objective);
124            }
125        }
126        Ok(())
127    }

129    pub(super) fn scoped_element(
130        &mut self,
131        element: &OptimizeElement,
132        origins: Vec<Location>,
133        polarity: WeightPolarity,
134        family: ObjectiveFamily,
135    ) -> Result<ObjectiveIr, FormulaFailure> {
136        let body = crate::formula_weak::body(element.condition(), self.budget, self.location)?;
137        self.scoped_objective(
138            &body,
139            element.weight(),
140            element.terms(),
141            origins,
142            polarity,
143            family,
144        )
145    }

147    pub(super) fn element_needs_scope(
148        &mut self,
149        element: &OptimizeElement,
150    ) -> Result<bool, FormulaFailure> {
151        for term in std::iter::once(element.weight().term())
152            .chain(element.weight().priority())
153            .chain(element.terms())
154            .flat_map(Term::subterms)
155        {
156            self.budget
157                .charge(ExpansionResource::TermWork, 1, self.location)?;
158            if matches!(term, Term::Interval { .. }) {
159                return Ok(true);
160            }
161        }
162        selection::condition(element.condition(), self.budget, self.location)
163    }

165    fn scoped_objective<'source>(
166        &mut self,
167        source: &Body,
168        weight: &Weight,
169        terms: impl Iterator<Item = &'source Term>,
170        origins: Vec<Location>,
171        polarity: WeightPolarity,
172        family: ObjectiveFamily,
173    ) -> Result<ObjectiveIr, FormulaFailure> {
174        // Objective roots have their own selection over the same canonical
175        // authority. Preserve the enclosing domain even when this scope refuses.
176        let empty = self.empty_domain()?;
177        let domain = std::mem::replace(&mut self.domain, empty);
178        let result = self.scoped_body(source, weight, terms, origins, polarity, family);
179        self.domain = domain;
180        result
181    }

183    fn scoped_field(
184        &mut self,
185        term: &Term,
186        variables: &mut Variables,
187        body: &mut Vec<LiteralIr>,
188    ) -> Result<super::ObjectiveField, FormulaFailure> {
189        if term
190            .subterms()
191            .any(|term| matches!(term, Term::Interval { .. }))
192        {
193            return self
194                .ranged_expression(term, variables, body)
195                .map(super::ObjectiveField::Expression);
196        }
197        self.objective_field(term, variables)
198    }

200    fn scoped_body<'source>(
201        &mut self,
202        source: &Body,
203        source_weight: &Weight,
204        terms: impl Iterator<Item = &'source Term>,
205        origins: Vec<Location>,
206        polarity: WeightPolarity,
207        family: ObjectiveFamily,
208    ) -> Result<ObjectiveIr, FormulaFailure> {
209        let mut variables = Variables::default();
210        // Fields belong to the outer scope, even when only a local aggregate
211        // mentions the same name. Local conditions cannot establish their safety.
212        let mut body = Vec::new();
213        let weight = self.scoped_field(source_weight.term(), &mut variables, &mut body)?;
214        let priority = match source_weight.priority() {
215            Some(term)
216                if term
217                    .subterms()
218                    .any(|term| matches!(term, Term::Interval { .. })) =>
219            {
220                self.ranged_expression(term, &mut variables, &mut body)?
221            }
222            other => self.objective_priority(other, &mut variables)?,
223        };
224        let tuple = terms
225            .map(|term| self.scoped_field(term, &mut variables, &mut body))
226            .collect::<Result<Vec<_>, _>>()?;
227        self.body_literals(source, &mut variables, &mut body)?;
228        let guards = self.body_guards(source, &mut variables, &mut body)?;
229        let assignments = self.assignment_targets(source, &guards, &mut variables)?;
230        self.bindings(&mut body, &mut variables)?;
231        variables.safety(self.location)?;
232        self.body_aggregates(source, guards, assignments, &variables, &mut body)?;
233        self.body_conditionals(source, &variables, &mut body)?;
234        let bindings = self.assignment_plan(&body, variables.count, variables.count, &[])?;
235        self.variable_limit(&variables)?;
236        let positive = body
237            .iter()
238            .filter_map(|literal| match literal {
239                LiteralIr::Atom(DefaultNegation::None, atom) => Some(*atom),
240                LiteralIr::PatternAtom(pattern) => Some(pattern.atom),
241                _ => None,
242            })
243            .collect();
244        Ok(ObjectiveIr {
245            family,
246            weight,
247            priority,
248            tuple,
249            positive,
250            filters: Vec::new(),
251            polarity,
252            priority_sources: BTreeSet::new(),
253            needs_eligibility_query: true,
254            condition: ObjectiveCondition::Body {
255                literals: body,
256                bindings,
257                filters: source.elements().filter(|element| matches!(element.get(), BodyElement::Literal(literal) if matches!(literal.inner, LiteralInner::Comparison(_)))).count(),
258            },
259            variables: variables.count,
260            origins,
261            location: self.location,
262        })
263    }
