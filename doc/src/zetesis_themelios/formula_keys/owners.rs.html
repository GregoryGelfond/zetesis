1//! Original ownership through canonical collection of analyzed statements.
2//!
3//! Compilation appends an already-canonical family for each original statement.
4//! A family of length other than one is ineligible for an in-place keyed rewrite.
5//! Canonical collection can also merge outputs from different owners, or repeated
6//! outputs from one owner: every content collision is likewise ineligible.
7//!
8//! The sidecar retains only statement IDs. Its temporary map borrows content;
9//! neither statement payloads nor provenance are copied. Map iteration gives the
10//! canonical statement order consumed by `Program::of_nodes`. This relies on the
11//! private producer invariant: normalization/projection outputs have passed the
12//! upstream Program door, and expanded facts contain only already-canonical
13//! symbolic arguments; normalized weak objectives rewrap canonical fields.
14//! It is not a constructor over arbitrary raw statements.
15//!
16//! For m emitted carriers this uses O(m log m) content comparisons and O(m)
17//! identity/map scratch. Each comparison reads bounded statement structure.
18//! The term-work unit charged here is one carrier lookup, not each node visited
19//! by upstream Ord; byte charges cover named sidecar/map payload, not allocator
20//! bookkeeping. Canonical Program construction retains its upstream cost.
21

30/// One entry per canonical analyzed carrier, in its content order. An absent
31/// owner means expansion or content merging prevents an in-place rewrite.
32pub(crate) struct Owners {
33    entries: Vec<Option<StatementId>>,
34}
35
36/// Record the analyzed family just emitted by one original statement. Every
37/// original is recorded once; a repeated count of one must not be used to hide
38/// multiple emissions from the same source compilation. The owners grow
39/// geometrically, so recording a program's statements one by one reallocates
40/// them a logarithmic number of times.
41pub(crate) fn record(
42    owners: &mut Vec<Option<StatementId>>,
43    owner: StatementId,
44    count: usize,
45    budget: &mut Budget,
46    site: ProgramSite,
47) -> Result<(), FormulaFailure> {
48    crate::formula_pattern::reserve_amortized(owners, count, budget, site)?;
49    let identity = (count == 1).then_some(owner);
50    owners.extend(std::iter::repeat_n(identity, count));
51    Ok(())
52}

54impl Owners {
55    /// Collect these already-canonical carriers and their parallel ownership
56    /// vector together, so the sidecar cannot be paired with another collection.
57    /// Two equal carriers clear ownership even if their original ID agrees.
58    pub(crate) fn collect_summary(
59        statements: Vec<WithProvenance<Statement>>,
60        owners: Vec<Option<StatementId>>,
61        representatives: Vec<WithProvenance<Statement>>,
62        sources: Vec<StatementId>,
63        budget: &mut Budget,
64        site: ProgramSite,
65    ) -> Result<(Program, Self, Vec<Option<StatementId>>), FormulaFailure> {
66        assert_eq!(
67            statements.len(),
68            owners.len(),
69            "every analyzed carrier has an owner slot"
70        );
71        assert_eq!(
72            statements.len(),
73            sources.len(),
74            "source identity for every non-fact emission"
75        );
76        let mut content = BTreeMap::new();
77        for ((statement, owner), source) in statements.iter().zip(&owners).zip(sources) {
78            budget.charge(ExpansionResource::TermWork, 1, site)?;
79            // Named key/value payload; BTreeMap allocator overhead is outside
80            // the expansion allowance, as for the neighboring compiler maps.
81            budget.charge(
82                ExpansionResource::ScalarBytes,
83                size_of::<(&Statement, (Option<StatementId>, Option<StatementId>))>() as u128,
84                site,
85            )?;
86            content
87                .entry(statement.get())
88                .and_modify(|entry: &mut (Option<StatementId>, Option<StatementId>)| {
89                    entry.0 = None;
90                    entry.1 = Some(source);
91                })
92                .or_insert((*owner, Some(source)));
93        }
94        for statement in &representatives {
95            budget.charge(ExpansionResource::TermWork, 1, site)?;
96            budget.charge(
97                ExpansionResource::ScalarBytes,
98                size_of::<(&Statement, (Option<StatementId>, Option<StatementId>))>() as u128,
99                site,
100            )?;
101            content
102                .entry(statement.get())
103                .and_modify(|entry| entry.0 = None)
104                .or_insert((None, None));
105        }
106        let mut entries = Vec::new();
107        crate::formula_pattern::reserve(&mut entries, content.len(), budget, site)?;
108        let mut nonfacts = Vec::new();
109        crate::formula_pattern::reserve(&mut nonfacts, content.len(), budget, site)?;
110        for (owner, nonfact) in content.into_values() {
111            entries.push(owner);
112            nonfacts.push(nonfact);
113        }
114        drop(owners);
115        // A representative must never replace a real carrier's nested evidence.
116        let program = Program::of_nodes(representatives.into_iter().chain(statements));
117        assert_eq!(
118            program.statements().count(),
119            entries.len(),
120            "canonical analyzed collection preserves sidecar positions",
121        );
122        Ok((program, Self { entries }, nonfacts))
123    }

125    #[cfg(test)]
126    pub(crate) fn collect(
127        statements: Vec<WithProvenance<Statement>>,
128        owners: Vec<Option<StatementId>>,
129        budget: &mut Budget,
130        site: ProgramSite,
131    ) -> Result<(Program, Self), FormulaFailure> {
132        let sources = (0..statements.len()).map(StatementId::new).collect();
133        Self::collect_summary(statements, owners, Vec::new(), sources, budget, site)
134            .map(|(program, owners, _)| (program, owners))
135    }
136
137    /// An owner may be asked only for this canonical analyzed position.
138    pub(crate) fn at(&self, index: usize) -> Option<StatementId> {
139        self.entries
140            .get(index)
141            .copied()
142            .expect("the owner sidecar covers every canonical analyzed position")
143    }
144}

155    fn fact(name: &str) -> WithProvenance<Statement> {
156        Program::of([Rule::fact(Atom::constant(Name::new(name).unwrap()))])
157            .statements()
158            .next()
159            .unwrap()
160            .clone()
161    }

163    fn collect(families: Vec<(usize, Vec<WithProvenance<Statement>>)>) -> (Program, Owners) {
164        let site = ProgramSite::program();
165        let mut budget = Budget::new(ExpansionLimits::default(), 100);
166        let mut statements = Vec::new();
167        let mut owners = Vec::new();
168        for (owner, family) in families {
169            record(
170                &mut owners,
171                StatementId::new(owner),
172                family.len(),
173                &mut budget,
174                site,
175            )
176            .unwrap();
177            statements.extend(family);
178        }
179        Owners::collect(statements, owners, &mut budget, site).unwrap()
180    }

207    /// Record `statements` one-statement families, returning the owners, the
208    /// number of reallocations and the budget charged.
209    fn record_one_by_one(statements: usize) -> (Vec<Option<StatementId>>, usize, Budget) {
210        let site = ProgramSite::program();
211        let mut budget = Budget::new(ExpansionLimits::default(), 100);
212        let mut owners = Vec::new();
213        let mut reallocations = 0;
214        for index in 1..=statements {
215            let capacity = owners.capacity();
216            record(&mut owners, StatementId::new(index), 1, &mut budget, site).unwrap();
217            reallocations += usize::from(owners.capacity() != capacity);
218        }
219        (owners, reallocations, budget)
220    }

222    #[test]
223    fn recording_reallocates_owners_logarithmically() {
224        // One exact reservation per statement would reallocate, and copy, the
225        // owners once for every statement recorded.
226        let statements = 1024_usize;
227        let (_, reallocations, _) = record_one_by_one(statements);
228        assert!(
229            reallocations <= 1 + statements.ilog2() as usize,
230            "{reallocations} reallocations for {statements} statements",
231        );
232    }

234    #[test]
235    fn every_reserved_owner_slot_is_charged() {
236        let (owners, _, budget) = record_one_by_one(100);
237        assert!(owners.capacity() > owners.len());
238        assert_eq!(
239            budget.usage().scalar_bytes,
240            owners.capacity() * size_of::<Option<StatementId>>(),

244    fn compiled(text: &str, max_key_work: u64) -> crate::AdmittedFormula {
245        let parsed =
246            crate::ParsedSource::new(text.into(), crate::AdmissionOptions::default()).unwrap();
247        let raised = themelios_program::raise::raise(parsed.parsed());
248        assert!(raised.diagnostics().is_empty());
249        // The outer statements are constructed, so no parsed coordinate can
250        // supply the keyed-rewrite identity exercised here.
251        let program = Program::of(
252            raised
253                .into_program()
254                .statements()
255                .map(|carrier| carrier.get().clone()),
256        );
257        crate::prepare_program_formula(
258            std::sync::Arc::new(program),
259            crate::ProgramAdmissionOptions::default(),
260            ExpansionLimits::default(),
261            crate::FormulaLimits {
262                max_key_work,
263                ..crate::FormulaLimits::default()
264            },
265        )
266        .unwrap()
267        .ground()
268        .unwrap()
269    }

271    fn family(
272        input: &crate::AdmittedFormula,
273    ) -> std::collections::BTreeSet<Vec<themelios_program::Symbol>> {
274        let mut search = zetesis_sat::StableModels::new(
275            input.theory(),
276            zetesis_sat::Limits::default(),
277            zetesis_cpu::Cancellation::default(),
278        )
279        .unwrap();
280        let result = search
281            .by_ref()
282            .map(|model| {
283                let mut atoms: Vec<_> = model
284                    .unwrap()
285                    .atoms()
286                    .map(|index| {
287                        crate::symbols::atom_with(input.atoms().at(index).unwrap(), 4096, || {
288                            Ok::<_, std::convert::Infallible>(())
289                        })
290                        .unwrap()
291                    })
292                    .collect();
293                atoms.sort_unstable();
294                atoms
295            })
296            .collect();
297        assert!(search.exhausted());
298        result
299    }

301    #[test]
302    fn constant_normalization_keeps_a_unique_compiler_owner() {
303        let source = "#const target=a. key(a). value(0;1). \
304            1{p(K,V):value(V)}1 :- key(K). :- p(target,V), V != 1.";
305        let rewritten = compiled(source, crate::FormulaLimits::default().max_key_work);
306        let original = compiled(source, 0);
307        assert_eq!(rewritten.keyed_constraints(), 1);
308        assert_eq!(original.keyed_constraints(), 0);
309        let actual = family(&rewritten);
310        assert_eq!(actual.len(), 1);
311        assert_eq!(actual, family(&original));
312    }

314    #[test]
315    fn expanded_compiler_collisions_remain_original() {
316        let source = "#const selected=1. key(a;b). value(0;1). \
317            1{p(K,V):value(V)}1 :- key(K). \
318            :- p((a;b),V), V != selected. :- p(a,V), V != 1.";
319        let inspected = compiled(source, crate::FormulaLimits::default().max_key_work);
320        let original = compiled(source, 0);
321        assert_eq!(inspected.keyed_constraints(), 0);
322        let actual = family(&inspected);
323        assert_eq!(actual.len(), 1);
324        assert_eq!(actual, family(&original));
