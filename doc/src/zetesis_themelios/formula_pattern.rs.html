1//! Positive structural patterns match support rows transactionally.
2//!
3//! Whole argument captures retain the original atom in the emitted formula.
4//! Nested variables expose subvalues; anonymous nodes neither equate occurrences
5//! nor introduce named slots. Plans and matching traverse flat preorder storage.

17pub(crate) struct PatternAtom {
18    pub atom: components::Pattern,
19    pub arguments: Vec<ArgumentPattern>,

21pub(crate) struct ArgumentPattern {
22    pub position: usize,
23    pub nodes: Vec<PatternNode>,
24}
25pub(crate) enum PatternNode {
26    Constructor(components::Constructor),
27    Constant(components::Scalar),
28    Slot(usize),
29    Wildcard,
30}

32/// A retained plan paired with its source's resolved flat atom.
33#[derive(Clone, Copy)]
34pub(crate) struct Pattern<'a> {
35    source: &'a PatternAtom,
36    atom: PatternRef<'a>,
37}
38impl PatternAtom {
39    pub(crate) fn bind<'a>(&'a self, atom: PatternRef<'a>) -> Pattern<'a> {
40        Pattern { source: self, atom }
41    }
42}
43impl<'a> Pattern<'a> {
44    pub(crate) fn atom(self) -> PatternRef<'a> {
45        self.atom
46    }
47    pub(crate) fn node_count(self) -> usize {
48        self.atom.terms().len()
49            + self
50                .source
51                .arguments
52                .iter()
53                .map(|argument| argument.nodes.len())
54                .sum::<usize>()
55    }
56    pub(crate) fn slots(self) -> impl Iterator<Item = usize> + 'a {
57        self.atom
58            .terms()
59            .iter()
60            .filter_map(|term| match term {
61                TemplateTerm::Variable(slot) => Some(slot),
62                TemplateTerm::Constant(_) => None,
63            })
64            .chain(self.source.arguments.iter().flat_map(|argument| {
65                argument.nodes.iter().filter_map(|node| {
66                    if let PatternNode::Slot(slot) = node {
67                        Some(*slot)
68                    } else {
69                        None
70                    }
71                })
72            }))
73    }
74
75    /// No incoming slot changes unless the caller commits the complete delta.
76    /// Captures borrow the support prefix; no whole value or subtree is copied.
77    pub(crate) fn matches<'source>(
78        self,
79        atom: zetesis_core::relation::Row<'_, 'source>,
80        values: BindingView<'_>,
81        computation: &Computation<'_, '_>,
82        context: &mut MatchContext<'_>,
83    ) -> Result<Option<Vec<(usize, TermRef<'source>)>>, FormulaFailure> {
84        context.work()?;
85        if !atom
86            .predicate()
87            .equals_ref_with(self.atom.predicate(), || context.work())?
88        {
89            return Ok(None);
90        }
91        context.charge(self.node_count() as u128)?;
92        let slots = self.slots().count();
93        let mut delta = Vec::new();
94        reserve(&mut delta, slots, context.budget, context.work.location)?;
95        for (column, term) in self.atom.terms().iter().enumerate() {
96            let value = atom.value(column).expect("checked pattern arity");
97            let agrees = match term {
98                TemplateTerm::Constant(expected) => context.equal(value, expected)?,
99                TemplateTerm::Variable(slot) => bind(slot, value, values, &mut delta, context)?,
100            };
101            if !agrees {
102                return Ok(None);
103            }
104        }
105        for argument in &self.source.arguments {
106            let value = atom
107                .value(argument.position)
108                .expect("checked argument position");
109            let mut offset = 0;
110            for node in &argument.nodes {
111                let Some(actual) = value.subterm_with(offset, || context.work())? else {
112                    return Ok(None);
113                };
114                // Equal constructor arities preserve preorder alignment. A
115                // capture consumes its complete subtree, retaining its identity.
116                context.work()?;
117                let descriptor = actual.descriptor();
118                context.charge(descriptor_bytes(descriptor))?;
119                if let PatternNode::Constructor(constructor) = node {
120                    let expected = computation.static_constructor(
121                        *constructor,
122                        context.work.limits,
123                        context.work.counters,
124                        context.work.location,
125                    )?;
126                    if expected != descriptor {
127                        return Ok(None);
128                    }
129                    offset += 1;
130                    continue;
131                }
132                let agrees = match node {
133                    PatternNode::Slot(slot) => bind(*slot, actual, values, &mut delta, context)?,
134                    PatternNode::Constant(expected) => {
135                        let expected = computation.static_scalar(
136                            *expected,
137                            context.work.limits,
138                            context.work.counters,
139                            context.work.location,
140                        )?;
141                        context.equal(actual, expected)?
142                    }
143                    PatternNode::Wildcard => true,
144                    PatternNode::Constructor(_) => unreachable!("shape consumed above"),
145                };
146                if !agrees {
147                    return Ok(None);
148                }
149                context.work()?;
150                offset += actual.expanded_nodes();
151            }
152            debug_assert_eq!(offset, value.expanded_nodes());
153        }
154        Ok(Some(delta))
155    }
156}

158pub(super) struct MatchContext<'a> {
159    pub work: GroundingWork<'a>,
160    pub budget: &'a mut Budget,
161}
162impl MatchContext<'_> {
163    fn work(&mut self) -> Result<(), FormulaFailure> {
164        self.work
165            .counters
166            .work(self.work.limits, self.work.location)
167    }
168    /// Charge `amount` units at once, so a refusal states that requirement.
169    fn charge(&mut self, amount: u128) -> Result<(), FormulaFailure> {
170        self.work
171            .counters
172            .charge_work(amount, self.work.limits, self.work.location)
173    }
174    fn equal(&mut self, left: TermRef<'_>, right: TermRef<'_>) -> Result<bool, FormulaFailure> {
175        left.equals_ref_with(right, || self.work())
176    }
177}
178
179fn bind<'source>(
180    slot: usize,
181    value: TermRef<'source>,
182    values: BindingView<'_>,
183    delta: &mut Vec<(usize, TermRef<'source>)>,
184    context: &mut MatchContext<'_>,
185) -> Result<bool, FormulaFailure> {
186    context.work()?;
187    if let Some(bound) = values.get(slot) {
188        return context.equal(value, bound);
189    }
190    // Repeated names see staged bindings without changing the incoming frame.
191    for (target, bound) in delta.iter() {
192        context.work()?;
193        if *target == slot {
194            return context.equal(value, *bound);
195        }
196    }
197    delta.push((slot, value));
198    Ok(true)
199}

201fn descriptor_bytes(node: ValueNodeRef<'_>) -> u128 {
202    match node {
203        ValueNodeRef::String(text)
204        | ValueNodeRef::Symbol(text)
205        | ValueNodeRef::Function { name: text, .. } => text.len() as u128,
206        _ => 0,
207    }
208}

210/// Reserve selected owned cells before growing a buffer; allocator overhead is excluded.
211pub(super) fn reserve<T>(
212    buffer: &mut Vec<T>,
213    additional: usize,
214    budget: &mut Budget,
215    location: ProgramSite,
216) -> Result<(), FormulaFailure> {
217    let required = buffer
218        .len()
219        .checked_add(additional)
220        .ok_or_else(|| allocation(location))?;
221    if required > buffer.capacity() {
222        budget.charge(
223            ExpansionResource::ScalarBytes,
224            (required - buffer.capacity()) as u128 * std::mem::size_of::<T>() as u128,
225            location,
226        )?;
227        buffer
228            .try_reserve_exact(additional)
229            .map_err(|_| allocation(location))?;
230    }
231    Ok(())
232}
233pub(super) fn push<T>(
234    buffer: &mut Vec<T>,
235    value: T,
236    budget: &mut Budget,
237    location: ProgramSite,
238) -> Result<(), FormulaFailure> {
239    if buffer.len() == buffer.capacity() {
240        let additional = buffer.len().max(1);
241        reserve(buffer, additional, budget, location)?;
242    }
243    buffer.push(value);
244    Ok(())
245}
246fn allocation(location: ProgramSite) -> FormulaFailure {
247    AdmissionFailure::Construction {
248        error: ConstructionError::Value(ValueError::Allocation),
249        location,
250    }
251    .into()
252}
