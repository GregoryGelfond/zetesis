1//! Positive structural patterns match support rows transactionally.
2//!
3//! Whole argument captures retain the original atom in the emitted formula.
4//! Nested variables expose subvalues; anonymous nodes neither equate occurrences
5//! nor introduce named slots. Plans and matching traverse flat preorder storage.

27    Constant(components::Scalar),

32/// A retained plan paired with its source's resolved flat atom.
33#[derive(Clone, Copy)]
34pub(crate) struct Pattern<'a> {
35    source: &'a PatternAtom,
36    atom: PatternRef<'a>,
37}
38impl PatternAtom {
39    pub(crate) fn bind<'a>(&'a self, atom: PatternRef<'a>) -> Pattern<'a> {
40        Pattern { source: self, atom }
41    }
42}
43impl<'a> Pattern<'a> {
44    pub(crate) fn atom(self) -> PatternRef<'a> {
45        self.atom
46    }
47    pub(crate) fn arguments(self) -> &'a [ArgumentPattern] {
48        &self.source.arguments
49    }
50
51    pub(crate) fn node_count(self) -> usize {
52        self.atom.terms().len()
53            + self
54                .source
55                .arguments
56                .iter()
57                .map(|argument| argument.nodes.len())
58                .sum::<usize>()
59    }
60    pub(crate) fn slots(self) -> impl Iterator<Item = usize> + 'a {
61        self.atom
62            .terms()
63            .iter()
64            .filter_map(|term| match term {
65                TemplateTerm::Variable(slot) => Some(slot),
66                TemplateTerm::Constant(_) => None,
67            })
68            .chain(self.source.arguments.iter().flat_map(|argument| {
69                argument.nodes.iter().filter_map(|node| {
70                    if let PatternNode::Slot(slot) = node {
71                        Some(*slot)
72                    } else {
73                        None
74                    }
75                })
76            }))
77    }
78
79    /// No incoming slot changes unless the caller commits the complete delta.
80    /// Captures borrow the support prefix; no whole value or subtree is copied.
81    /// The caller owns and leases reusable capture capacity. Only a successful
82    /// match leaves staged bindings; every mismatch or refusal clears them.
83    pub(crate) fn matches<'source>(
84        self,
85        atom: zetesis_core::relation::Row<'_, 'source>,
86        values: BindingView<'_>,
87        computation: &Computation<'_, '_>,
88        delta: &mut Vec<(usize, TermRef<'source>)>,
89        context: &mut MatchContext<'_>,
90    ) -> Result<bool, FormulaFailure> {
91        delta.clear();
92        let result = self.match_into(atom, values, computation, delta, context);
93        if !matches!(result, Ok(true)) {
94            delta.clear();
95        }
96        result
97    }
98
99    fn match_into<'source>(
100        self,
101        atom: zetesis_core::relation::Row<'_, 'source>,
102        values: BindingView<'_>,
103        computation: &Computation<'_, '_>,
104        delta: &mut Vec<(usize, TermRef<'source>)>,
105        context: &mut MatchContext<'_>,
106    ) -> Result<bool, FormulaFailure> {
107        context.work()?;
108        if !atom
109            .predicate()
110            .equals_ref_with(self.atom.predicate(), || context.work())?
111        {
112            return Ok(false);
113        }
114        context.charge(self.node_count() as u128)?;
115        let slots = self.slots().count();
116        context.reserve(delta, slots, computation)?;
117        for (column, term) in self.atom.terms().iter().enumerate() {
118            let value = atom.value(column).expect("checked pattern arity");
119            let agrees = match term {
120                TemplateTerm::Constant(expected) => context.equal(value, expected)?,
121                TemplateTerm::Variable(slot) => bind(slot, value, values, delta, context)?,
122            };
123            if !agrees {
124                return Ok(false);
125            }
126        }
127        for argument in &self.source.arguments {
128            let value = atom
129                .value(argument.position)
130                .expect("checked argument position");
131            let mut offset = 0;
132            for node in &argument.nodes {
133                let Some(actual) = value.subterm_with(offset, || context.work())? else {
134                    return Ok(false);
135                };
136                // Equal constructor arities preserve preorder alignment. A
137                // capture consumes its complete subtree, retaining its identity.
138                context.work()?;
139                let descriptor = actual.descriptor();
140                context.charge(descriptor_bytes(descriptor))?;
141                if let PatternNode::Constructor(constructor) = node {
142                    let expected = computation.static_constructor(
143                        *constructor,
144                        context.work.limits,
145                        context.work.counters,
146                        context.work.location,
147                    )?;
148                    if expected != descriptor {
149                        return Ok(false);
150                    }
151                    offset += 1;
152                    continue;
153                }
154                let agrees = match node {
155                    PatternNode::Slot(slot) => bind(*slot, actual, values, delta, context)?,
156                    PatternNode::Constant(expected) => {
157                        let expected = computation.static_scalar(
158                            *expected,
159                            context.work.limits,
160                            context.work.counters,
161                            context.work.location,
162                        )?;
163                        context.equal(actual, expected)?
164                    }
165                    PatternNode::Wildcard => true,
166                    PatternNode::Constructor(_) => unreachable!("shape consumed above"),
167                };
168                if !agrees {
169                    return Ok(false);
170                }
171                context.work()?;
172                offset += actual.expanded_nodes();
173            }
174            debug_assert_eq!(offset, value.expanded_nodes());
175        }
176        Ok(true)
177    }
178}

180pub(super) struct MatchContext<'a> {
181    pub work: GroundingWork<'a>,
182    pub budget: &'a mut Budget,
183    /// The caller's existing owner, including every other retained buffer.
184    pub lease: &'a mut StorageLease,
185    pub other_bytes: usize,
186}
187impl MatchContext<'_> {
188    fn reserve<T>(
189        &mut self,
190        delta: &mut Vec<T>,
191        slots: usize,
192        computation: &Computation<'_, '_>,
193    ) -> Result<(), FormulaFailure> {
194        let previous = delta.capacity();
195        let requested = slots.max(previous);
196        if requested > previous {
197            self.budget.charge(
198                ExpansionResource::ScalarBytes,
199                (requested - previous) as u128 * size_of::<T>() as u128,
200                self.work.location,
201            )?;
202        }
203        let result = crate::formula_support::reserve_exact(
204            delta,
205            slots,
206            self.lease,
207            self.other_bytes,
208            Context::new(
209                computation,
210                self.work.limits,
211                self.work.counters,
212                self.work.location,
213            ),
214        )
215        .map_err(|error| match error {
216            FormulaFailure::TermAssignment {
217                error:
218                    zetesis_core::catalog::AssignmentError::Storage(
219                        zetesis_core::catalog::Error::Allocation,
220                    ),
221                ..
222            } => allocation(self.work.location),
223            error => error,
224        });
225        // The live lease already records any capacity retained on refusal.
226        // Account allocator-provided excess without charging reused cells.
227        let excess = if delta.capacity() > requested {
228            self.budget
229                .charge(
230                    ExpansionResource::ScalarBytes,
231                    (delta.capacity() - requested) as u128 * size_of::<T>() as u128,
232                    self.work.location,
233                )
234                .map_err(FormulaFailure::from)
235        } else {
236            Ok(())
237        };
238        result.and(excess)
239    }
240
241    fn work(&mut self) -> Result<(), FormulaFailure> {
242        self.work
243            .counters
244            .work(self.work.limits, self.work.location)
245    }
246    /// Charge `amount` units at once, so a refusal states that requirement.
247    fn charge(&mut self, amount: u128) -> Result<(), FormulaFailure> {
248        self.work
249            .counters
250            .charge_work(amount, self.work.limits, self.work.location)
251    }
252    fn equal(&mut self, left: TermRef<'_>, right: TermRef<'_>) -> Result<bool, FormulaFailure> {
253        left.equals_ref_with(right, || self.work())
254    }
255}
256
257fn bind<'source>(
258    slot: usize,
259    value: TermRef<'source>,
260    values: BindingView<'_>,
261    delta: &mut Vec<(usize, TermRef<'source>)>,
262    context: &mut MatchContext<'_>,
263) -> Result<bool, FormulaFailure> {
264    context.work()?;
265    if let Some(bound) = values.get(slot) {
266        return context.equal(value, bound);
267    }
268    // Repeated names see staged bindings without changing the incoming frame.
269    for (target, bound) in delta.iter() {
270        context.work()?;
271        if *target == slot {
272            return context.equal(value, *bound);
273        }
274    }
275    delta.push((slot, value));
276    Ok(true)
277}

279fn descriptor_bytes(node: ValueNodeRef<'_>) -> u128 {
280    match node {
281        ValueNodeRef::String(text)
282        | ValueNodeRef::Symbol(text)
283        | ValueNodeRef::Function { name: text, .. } => text.len() as u128,
284        _ => 0,
285    }
286}

288/// Reserve selected owned cells before growing a buffer; allocator overhead is excluded.
289pub(super) fn reserve<T>(
290    buffer: &mut Vec<T>,
291    additional: usize,
292    budget: &mut Budget,
293    location: ProgramSite,
294) -> Result<(), FormulaFailure> {
295    let required = buffer
296        .len()
297        .checked_add(additional)
298        .ok_or_else(|| allocation(location))?;
299    if required > buffer.capacity() {
300        budget.charge(
301            ExpansionResource::ScalarBytes,
302            (required - buffer.capacity()) as u128 * std::mem::size_of::<T>() as u128,
303            location,
304        )?;
305        buffer
306            .try_reserve_exact(additional)
307            .map_err(|_| allocation(location))?;
308    }
309    Ok(())
310}
311pub(super) fn push<T>(
312    buffer: &mut Vec<T>,
313    value: T,
314    budget: &mut Budget,
315    location: ProgramSite,
316) -> Result<(), FormulaFailure> {
317    if buffer.len() == buffer.capacity() {
318        let additional = buffer.len().max(1);
319        reserve(buffer, additional, budget, location)?;
320    }
321    buffer.push(value);
322    Ok(())
323}
324fn allocation(location: ProgramSite) -> FormulaFailure {
325    AdmissionFailure::Construction {
326        error: ConstructionError::Value(ValueError::Allocation),
327        location,
328    }
329    .into()
330}

396    /// A standalone matcher caller owns the same Vec header/capacity receipt
397    /// that production includes inside the Join lease.
398    struct Scratch<'source> {
399        delta: Vec<(usize, TermRef<'source>)>,
400        lease: StorageLease,
401    }
402    impl<'source> Scratch<'source> {
403        fn new(computation: &Computation<'_, '_>) -> Self {
404            let mut lease = computation.lease();
405            lease
406                .observe(size_of::<Vec<(usize, TermRef<'_>)>>(), location())
407                .unwrap();
408            Self {
409                delta: Vec::new(),
410                lease,
411            }
412        }
413        fn matches(
414            &mut self,
415            pattern: Pattern<'_>,
416            row: zetesis_core::relation::Row<'_, 'source>,
417            incoming: BindingView<'_>,
418            context: Context<'_, &Computation<'_, '_>>,
419            budget: &mut Budget,
420        ) -> Result<bool, FormulaFailure> {
421            pattern.matches(
422                row,
423                incoming,
424                context.computation,
425                &mut self.delta,
426                &mut MatchContext {
427                    work: context.work,
428                    budget,
429                    lease: &mut self.lease,
430                    other_bytes: size_of::<Vec<(usize, TermRef<'_>)>>(),
431                },
432            )
433        }
434    }
