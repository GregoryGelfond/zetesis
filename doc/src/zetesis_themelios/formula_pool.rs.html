1//! Bounded source-occurrence pools before ordinary formula compilation.
2//!
3//! An outer cursor selects each pooled literal independently, emitting complete
4//! rules. Within a choice, selections instead add elements to the same group.
5//! No constructor closure, textual substitution or upstream unpooling is used.
6//! Pool-free alternatives retain the existing scalar/range admission contract.

28impl Compiler<'_> {
29    /// Weak constraints have the same outer body-product law as rules. Local
30    /// aggregate and conditional scopes remain untouched for their consumers.
31    pub(super) fn body_alternatives(&mut self, body: &Body) -> Result<Vec<Body>, FormulaFailure> {
32        let mut payload = Footprint::default();
33        payload.visit_body(body);
34        self.budget.charge(
35            ExpansionResource::TermWork,
36            payload.nodes.saturating_mul(2),
37            self.location,
38        )?;
39        self.budget.charge(
40            ExpansionResource::ScalarBytes,
41            payload.bytes.saturating_mul(2),
42            self.location,
43        )?;
44        let source = WithProvenance::constructed(Statement::Rule(
45            themelios_program::program::Rule::new(Head::Falsum, body.clone()),
46        ));
47        let mut nodes = 0;
48        let Some(mut cursor) = Cursor::new(
49            &source,
50            0,
51            self.limits,
52            &mut nodes,
53            self.budget,
54            self.location,
55        )?
56        else {
57            return Ok(vec![body.clone()]);
58        };
59        self.budget.check_family(
60            (cursor.remaining as u128).saturating_mul(
61                cursor
62                    .bytes
63                    .saturating_add(std::mem::size_of::<Body>() as u128),
64            ),
65            self.location,
66        )?;
67        let mut result = Vec::new();
68        while let Some(statement) = cursor.next(self.budget, self.location)? {
69            let Statement::Rule(rule) = statement.get() else {
70                unreachable!("body cursor retains rule")
71            };
72            result.push(rule.body().get().clone());
73        }
74        Ok(result)
75    }

77    /// Append a bounded family of whole rules, retaining every source occurrence
78    /// even when the separate analyzed program later deduplicates equal rules.
79    pub(super) fn source_rules(
80        &mut self,
81        statement: &WithProvenance<Statement>,
82        origins: &[ProgramSite],
83        projection_nodes: &mut u128,
84        rules: &mut Vec<RuleIr>,
85        analyzed: &mut Vec<WithProvenance<Statement>>,
86    ) -> Result<(), FormulaFailure> {
87        // Local pooling reconstructs choice element carriers. Retain the
88        // original counted entries so choice compilation expands each local
89        // product once, before assigning occurrence keys.
90        let choice_source = match statement.get() {
91            Statement::Rule(rule) => match rule.head().get() {
92                Head::Choice(choice) => Some(choice),
93                _ => None,
94            },
95            _ => None,
96        };
97        if let Some(mut cursor) = Cursor::new(
98            statement,
99            origins.len(),
100            self.limits,
101            projection_nodes,
102            self.budget,
103            self.location,
104        )? {
105            while let Some(statement) = cursor.next(self.budget, self.location)? {
106                let Statement::Rule(rule) = statement.get() else {
107                    unreachable!("pool expansion retains a rule")
108                };
109                rules.push(self.rule(rule, origins.to_vec(), choice_source)?);
110                analyzed.extend(self.conditional_projection(&statement, projection_nodes)?);
111            }
112        } else {
113            self.budget
114                .charge(ExpansionResource::Templates, 1, self.location)?;
115            self.budget.charge(
116                ExpansionResource::Origins,
117                origins.len() as u128,
118                self.location,
119            )?;
120            let Statement::Rule(rule) = statement.get() else {
121                return Err(unsupported(ProfileFeature::Statement, self.location).into());
122            };
123            rules.push(self.rule(rule, origins.to_vec(), choice_source)?);
124            analyzed.extend(self.conditional_projection(statement, projection_nodes)?);
125        }
126        Ok(())
127    }
128}

130/// Owns only occurrence positions, never the Cartesian output. Equal values do
131/// not merge these positions. Each returned statement is compiled separately.
132struct Cursor<'a> {
133    source: &'a WithProvenance<Statement>,
134    widths: Vec<usize>,
135    positions: Vec<usize>,
136    remaining: usize,
137    nodes: u128,
138    bytes: u128,
139}
140
141impl<'a> Cursor<'a> {
142    fn new(
143        source: &'a WithProvenance<Statement>,
144        origins: usize,
145        limits: &FormulaLimits,
146        projection_nodes: &mut u128,
147        budget: &mut Budget,
148        location: ProgramSite,
149    ) -> Result<Option<Self>, FormulaFailure> {
150        let mut scan = Footprint {
151            nodes: 1,
152            ..Footprint::default()
153        };
154        scan.visit_statement(source.get());
155        budget.charge(ExpansionResource::TermWork, scan.nodes, location)?;
156        if scan.pools == 0 {
157            return Ok(None);
158        }
159        let Statement::Rule(rule) = source.get() else {
160            return Err(unsupported(ProfileFeature::Term, location).into());
161        };
162        budget.charge(
163            ExpansionResource::ScalarBytes,
164            (scan.pools as u128).saturating_mul(
165                (std::mem::size_of::<u128>() + 2 * std::mem::size_of::<usize>()) as u128,
166            ),
167            location,
168        )?;
169        let Shape {
170            widths,
171            local_copies,
172        } = shape(rule, scan.pools);
173        // Local scopes validate and expand their own alternatives during IR
174        // compilation. Only ordinary head/body occurrences drive this cursor.
175        let count = widths
176            .iter()
177            .fold(1_u128, |count, width| count.saturating_mul(*width));
178        budget.charge(ExpansionResource::Templates, count, location)?;
179        budget.charge(
180            ExpansionResource::Origins,
181            count.saturating_mul(origins as u128),
182            location,
183        )?;
184        let copies = local_copies.saturating_add(1);
185        let nodes = scan.nodes.saturating_mul(copies);
186        // A conservative cumulative projection bound is checked before cloning;
187        // final analysis independently counts the exact, deduplicated program.
188        *projection_nodes = projection_nodes.saturating_add(nodes.saturating_mul(count));
189        crate::formula::ceiling(
190            FormulaResource::AnalysisNodes,
191            *projection_nodes,
192            limits.max_analysis_nodes as u128,
193            location,
194        )?;
195        budget.charge(
196            ExpansionResource::Values,
197            scan.nodes.saturating_mul(copies).saturating_mul(count),
198            location,
199        )?;
200        budget.check_family(
201            scan.bytes
202                .saturating_mul(copies)
203                .saturating_mul(4)
204                .saturating_add(
205                    scan.nodes
206                        .saturating_mul(copies)
207                        .saturating_mul(std::mem::size_of::<BodyElement>() as u128),
208                )
209                .saturating_add((widths.len() as u128).saturating_mul(
210                    (std::mem::size_of::<u128>() + 2 * std::mem::size_of::<usize>()) as u128,
211                )),
212            location,
213        )?;
214        let positions = vec![0; widths.len()];
215        let widths = widths
216            .into_iter()
217            .map(|width| usize::try_from(width).expect("bounded pool width"))
218            .collect();
219        Ok(Some(Self {
220            source,
221            widths,
222            positions,
223            remaining: usize::try_from(count).expect("bounded by usize template limit"),
224            nodes: nodes.saturating_mul(4),
225            bytes: scan.bytes.saturating_mul(copies).saturating_mul(4),
226        }))
227    }
228
229    fn next(
230        &mut self,
231        budget: &mut Budget,
232        location: ProgramSite,
233    ) -> Result<Option<WithProvenance<Statement>>, FormulaFailure> {
234        if self.remaining == 0 {
235            return Ok(None);
236        }
237        // Reserve four equivalents of counted term cells and text payload for
238        // the owned rewrite, selected children and retained analysis statement.
239        // Other AST carriers, provenance and allocator overhead are excluded;
240        // source/syntax and projection-node limits bound structural expansion.
241        budget.charge(ExpansionResource::TermWork, self.nodes, location)?;
242        budget.charge(ExpansionResource::ScalarBytes, self.bytes, location)?;
243        let mut selector = Select {
244            positions: self.positions.iter(),
245        };
246        let expanded = rewrite(
247            Program::of_nodes([self.source.clone()]),
248            &mut Expand {
249                selector: &mut selector,
250            },
251        );
252        assert!(
253            selector.positions.next().is_none(),
254            "all pool occurrences selected"
255        );
256        let statement = expanded
257            .statements()
258            .next()
259            .expect("one rule remains one rule")
260            .clone();
261        self.remaining -= 1;
262        for (position, width) in self.positions.iter_mut().zip(&self.widths).rev() {
263            *position += 1;
264            if *position < *width {
265                break;
266            }
267            *position = 0;
268        }
269        Ok(Some(statement))
270    }
271}

273struct Shape {
274    widths: Vec<u128>,
275    local_copies: u128,
276}
277
278fn shape(rule: &themelios_program::program::Rule, capacity: usize) -> Shape {
279    let mut widths = Vec::with_capacity(capacity);
280    let mut local_copies = 0_u128;
281    match rule.head().get() {
282        Head::Literal(literal) => {
283            outer(literal, &mut widths);
284        }
285        Head::Disjunction(head) => {
286            for element in head.elements() {
287                if unconditional(element.get().condition()) {
288                    outer(element.get().literal(), &mut widths);
289                }
290            }
291        }
292        Head::Choice(choice) => {
293            guard_width(choice.left_guard(), &mut widths);
294            guard_width(choice.right_guard(), &mut widths);
295            for element in choice.elements() {
296                if let LiteralInner::Atom(atom) = &element.get().literal().inner {
297                    let (count, _) = atom_width(atom.get());
298                    local_copies = local_copies.saturating_add(count.saturating_sub(1));
299                }
300            }
301        }
302        Head::Aggregate(aggregate) => {
303            guard_width(aggregate.left_guard(), &mut widths);
304            guard_width(aggregate.right_guard(), &mut widths);
305        }
306        _ => {}
307    }
308    for element in rule.body().get().elements() {
309        if let BodyElement::Literal(literal) = element.get() {
310            outer(literal, &mut widths);
311        } else if let BodyElement::Aggregate { aggregate, .. } = element.get() {
312            let guarded: &dyn HasGuards = match aggregate {
313                Aggregate::Function(value) => value,
314                Aggregate::Set(value) => value,
315            };
316            guard_width(guarded.left_guard(), &mut widths);
317            guard_width(guarded.right_guard(), &mut widths);
318        }
319    }
320    Shape {
321        widths,
322        local_copies,
323    }
324}

326fn outer(literal: &Literal, widths: &mut Vec<u128>) {
327    let (width, pools) = literal_width(literal);
328    if pools != 0 {
329        widths.push(width);
330    }
331}
332
333fn unconditional(condition: &Condition) -> bool {
334    condition.literals().all(|literal| {
335        matches!(
336            (&literal.get().inner, literal.get().negation),
337            (
338                LiteralInner::True,
339                DefaultNegation::None | DefaultNegation::NotNot
340            ) | (LiteralInner::False, DefaultNegation::Not)
341        )
342    })
343}

345fn guard_width(guard: Option<&WithProvenance<Guard>>, widths: &mut Vec<u128>) {
346    if let Some(guard) = guard
347        && matches!(guard.get().term, Term::Pool(_))
348    {
349        widths.push(term_width(&guard.get().term) as u128);
350    }
351}
352
353fn term_width(term: &Term) -> usize {
354    if let Term::Pool(items) = term {
355        items.len()
356    } else {
357        1
358    }
359}

361pub(super) fn atom_width(atom: &Atom) -> (u128, usize) {
362    let mut pools = usize::from(matches!(atom.arguments, Arguments::Pooled(_)));
363    let count = atom.alternatives().fold(0_u128, |sum, arguments| {
364        let product = arguments.iter().fold(1_u128, |product, term| {
365            pools += usize::from(matches!(term, Term::Pool(_)));
366            product.saturating_mul(term_width(term) as u128)
367        });
368        sum.saturating_add(product)
369    });
370    (count, pools)
371}

373pub(super) fn select_atom(mut atom: Atom, mut position: usize) -> Atom {
374    let arguments = atom
375        .alternatives()
376        .find_map(|arguments| {
377            let width = arguments
378                .iter()
379                .fold(1_usize, |width, term| width * term_width(term));
380            if position >= width {
381                position -= width;
382                None
383            } else {
384                let mut selected = arguments.to_vec();
385                for term in selected.iter_mut().rev() {
386                    if let Term::Pool(items) = term {
387                        let index = position % items.len();
388                        position /= items.len();
389                        *term = items[index].clone();
390                    }
391                }
392                Some(selected)
393            }
394        })
395        .expect("bounded occurrence position");
396    atom.arguments = Arguments::Single(arguments);
397    atom
398}

400struct Select<'a> {
401    positions: std::slice::Iter<'a, usize>,
402}
403impl Select<'_> {
404    fn guard(&mut self, guard: Option<&WithProvenance<Guard>>) -> Option<Guard> {
405        guard.map(|guard| {
406            let mut guard = guard.get().clone();
407            if let Term::Pool(items) = &guard.term {
408                guard.term =
409                    items[*self.positions.next().expect("guard occurrence position")].clone();
410            }
411            guard
412        })
413    }
414}
415impl Rewrite for Select<'_> {
416    fn tag(&self) -> TransformTag {
417        TransformTag::new("zetesis-finite-pools")
418    }
419    fn rewrite_atom(&mut self, atom: Atom) -> Atom {
420        if atom_width(&atom).1 == 0 {
421            atom
422        } else {
423            select_atom(
424                atom,
425                *self.positions.next().expect("atom occurrence position"),
426            )
427        }
428    }
429    fn rewrite_comparison(&mut self, comparison: Comparison) -> Comparison {
430        let pooled = std::iter::once(comparison.first())
431            .chain(comparison.steps().map(|(_, term)| term))
432            .any(|term| matches!(term, Term::Pool(_)));
433        if pooled {
434            select_comparison(
435                &comparison,
436                *self
437                    .positions
438                    .next()
439                    .expect("comparison occurrence position"),
440            )
441        } else {
442            comparison
443        }
444    }
445}

447struct Expand<'a, 'b> {
448    selector: &'a mut Select<'b>,
449}
450impl Rewrite for Expand<'_, '_> {
451    fn tag(&self) -> TransformTag {
452        self.selector.tag()
453    }
454    fn rewrite_body(&mut self, body: Body) -> Body {
455        // The conditional compiler owns inner-OR alternatives. Outer pool
456        // positions never select or duplicate a universal consequent.
457        Body::new(body.elements().map(|element| match element.get() {
458            BodyElement::Literal(literal) => {
459                BodyElement::Literal(self.selector.rewrite_literal(literal.clone()))
460            }
461            BodyElement::Aggregate {
462                negation,
463                aggregate,
464            } => {
465                let aggregate = match aggregate {
466                    Aggregate::Function(value) => Aggregate::Function(FunctionAggregate::new(
467                        self.selector.guard(value.left_guard()),
468                        value.function(),
469                        value.elements().map(|element| element.get().clone()),
470                        self.selector.guard(value.right_guard()),
471                    )),
472                    Aggregate::Set(value) => Aggregate::Set(SetAggregate::new(
473                        self.selector.guard(value.left_guard()),
474                        value.elements().map(|element| element.get().clone()),
475                        self.selector.guard(value.right_guard()),
476                    )),
477                };
478                BodyElement::Aggregate {
479                    negation: *negation,
480                    aggregate,
481                }
482            }
483            other => other.clone(),
484        }))
485    }
486    fn rewrite_head(&mut self, head: Head) -> Head {
487        if let Head::Disjunction(disjunction) = head {
488            return Head::Disjunction(Disjunction::new(disjunction.elements().map(|element| {
489                let literal = element.get().literal();
490                DisjunctionElement::new(
491                    if unconditional(element.get().condition()) {
492                        self.selector.rewrite_literal(literal.clone())
493                    } else {
494                        literal.clone()
495                    },
496                    element.get().condition().clone(),
497                )
498            })));
499        }
500        if let Head::Aggregate(value) = head {
501            return Head::Aggregate(HeadAggregate::new(
502                self.selector.guard(value.left_guard()),
503                value.function(),
504                value.elements().map(|element| element.get().clone()),
505                self.selector.guard(value.right_guard()),
506            ));
507        }
508        let Head::Choice(choice) = head else {
509            return self.selector.rewrite_head(head);
510        };
511        let mut elements = Vec::new();
512        for element in choice.elements() {
513            let literal = element.get().literal();
514            if let LiteralInner::Atom(atom) = &literal.inner {
515                let (count, _) = atom_width(atom.get());
516                for index in 0..usize::try_from(count).expect("bounded local copies") {
517                    elements.push(ChoiceElement::new(
518                        Literal {
519                            negation: literal.negation,
520                            inner: LiteralInner::Atom(
521                                atom.clone().map(|atom| select_atom(atom, index)),
522                            ),
523                        },
524                        element.get().condition().clone(),
525                    ));
526                }
527            } else {
528                elements.push(element.get().clone());
529            }
530        }
531        // The enclosing rule/head retains parsed origins. Synthesized element
532        // nodes truthfully carry constructed provenance, like expanded facts.
533        Head::Choice(Choice::new(
534            self.selector.guard(choice.left_guard()),
535            elements,
536            self.selector.guard(choice.right_guard()),
537        ))
538    }
539}

541/// Logical syntax-copy units shared by source normalization and terminal
542/// partitioning. These exclude other AST carriers, provenance and allocator
543/// overhead; they are not retained-capacity or whole-heap measurements.
544pub(crate) fn source_copy_cost(statement: &Statement) -> (u128, u128) {
545    let mut footprint = Footprint {
546        nodes: 1,
547        ..Footprint::default()
548    };
549    footprint.visit_statement(statement);
550    (footprint.nodes, footprint.bytes)
551}
552
553#[derive(Default)]
554struct Footprint {
555    nodes: u128,
556    bytes: u128,
557    pools: usize,
558}
559impl Visit for Footprint {
560    fn visit_atom(&mut self, atom: &Atom) {
561        self.nodes += 1;
562        self.bytes += atom.name.as_str().len() as u128;
563        self.pools += usize::from(matches!(atom.arguments, Arguments::Pooled(_)));
564        for term in atom.argument_terms() {
565            for node in term.subterms() {
566                self.visit_term(node);
567            }
568        }
569    }
570    fn visit_term(&mut self, term: &Term) {
571        self.nodes += 1;
572        self.bytes += std::mem::size_of::<Term>() as u128;
573        self.pools += usize::from(matches!(term, Term::Pool(_)));
574        if let Term::Function { name, .. } = term {
575            self.bytes += name.as_str().len() as u128;
576        }
577        if let Term::Symbolic(symbol) = term {
578            self.bytes += crate::structural_value::symbol_bytes(symbol);
579        }
580        if let Term::Variable(themelios_program::term::Variable::Named(name)) = term {
581            self.bytes += name.as_str().len() as u128;
582        }
583    }
