1//! Bounded source-occurrence pools before ordinary formula compilation.
2//!
3//! An outer cursor selects each pooled literal independently, emitting complete
4//! rules. Within a choice, selections instead add elements to the same group.
5//! No constructor closure, textual substitution or upstream unpooling is used.
6//! Pool-free alternatives retain the existing scalar/range admission contract.

28impl Compiler<'_> {
29    /// Weak constraints have the same outer body-product law as rules. Local
30    /// aggregate and conditional scopes remain untouched for their consumers.
31    pub(super) fn body_alternatives(&mut self, body: &Body) -> Result<Vec<Body>, FormulaFailure> {
32        let mut payload = Footprint::default();
33        payload.visit_body(body);
34        self.budget.charge(
35            ExpansionResource::TermWork,
36            payload.nodes.saturating_mul(2),
37            self.location,
38        )?;
39        self.budget.charge(
40            ExpansionResource::ScalarBytes,
41            payload.bytes.saturating_mul(2),
42            self.location,
43        )?;
44        let source = WithProvenance::constructed(Statement::Rule(
45            themelios_program::program::Rule::new(Head::Falsum, body.clone()),
46        ));
47        let mut nodes = 0;
48        let Some(mut cursor) = Cursor::new(
49            &source,
50            0,
51            self.limits,
52            &mut nodes,
53            self.budget,
54            self.location,
55        )?
56        else {
57            return Ok(vec![body.clone()]);
58        };
59        let mut result = Vec::new();
60        while let Some(statement) = cursor.next(self.budget, self.location)? {
61            let Statement::Rule(rule) = statement.get() else {
62                unreachable!("body cursor retains rule")
63            };
64            result.push(rule.body().get().clone());
65        }
66        Ok(result)
67    }

69    /// Append a bounded family of whole rules, retaining every source occurrence
70    /// even when the separate analyzed program later deduplicates equal rules.
71    pub(super) fn source_rules(
72        &mut self,
73        statement: &WithProvenance<Statement>,
74        origins: &[Location],
75        projection_nodes: &mut u128,
76        rules: &mut Vec<RuleIr>,
77        analyzed: &mut Vec<WithProvenance<Statement>>,
78    ) -> Result<(), FormulaFailure> {
79        // Local atom pooling reconstructs choice element carriers. Boolean
80        // literals and their conditions remain equal, so their exact source
81        // keys are recovered from this same rule before that reconstruction.
82        let choice_source = match statement.get() {
83            Statement::Rule(rule) => match rule.head().get() {
84                Head::Choice(choice) => Some(choice),
85                _ => None,
86            },
87            _ => None,
88        };
89        if let Some(mut cursor) = Cursor::new(
90            statement,
91            origins.len(),
92            self.limits,
93            projection_nodes,
94            self.budget,
95            self.location,
96        )? {
97            while let Some(statement) = cursor.next(self.budget, self.location)? {
98                let Statement::Rule(rule) = statement.get() else {
99                    unreachable!("pool expansion retains a rule")
100                };
101                rules.push(self.rule(rule, origins.to_vec(), choice_source)?);
102                analyzed.extend(self.conditional_projection(&statement, projection_nodes)?);
103            }
104        } else {
105            self.budget
106                .charge(ExpansionResource::Templates, 1, self.location)?;
107            self.budget.charge(
108                ExpansionResource::Origins,
109                origins.len() as u128,
110                self.location,
111            )?;
112            let Statement::Rule(rule) = statement.get() else {
113                return Err(unsupported(ProfileFeature::Statement, self.location).into());
114            };
115            rules.push(self.rule(rule, origins.to_vec(), choice_source)?);
116            analyzed.extend(self.conditional_projection(statement, projection_nodes)?);
117        }
118        Ok(())
119    }
120}

122/// Owns only occurrence positions, never the Cartesian output. Equal values do
123/// not merge these positions. Each returned statement is compiled separately.
124struct Cursor<'a> {
125    source: &'a WithProvenance<Statement>,
126    widths: Vec<usize>,
127    positions: Vec<usize>,
128    remaining: usize,
129    nodes: u128,
130    bytes: u128,
131}
132
133impl<'a> Cursor<'a> {
134    fn new(
135        source: &'a WithProvenance<Statement>,
136        origins: usize,
137        limits: &FormulaLimits,
138        projection_nodes: &mut u128,
139        budget: &mut Budget,
140        location: Location,
141    ) -> Result<Option<Self>, FormulaFailure> {
142        let mut scan = Footprint {
143            nodes: 1,
144            ..Footprint::default()
145        };
146        scan.visit_statement(source.get());
147        budget.charge(ExpansionResource::TermWork, scan.nodes, location)?;
148        if scan.pools == 0 {
149            return Ok(None);
150        }
151        let Statement::Rule(rule) = source.get() else {
152            return Err(unsupported(ProfileFeature::Term, location).into());
153        };
154        budget.charge(
155            ExpansionResource::ScalarBytes,
156            (scan.pools as u128).saturating_mul(
157                (std::mem::size_of::<u128>() + 2 * std::mem::size_of::<usize>()) as u128,
158            ),
159            location,
160        )?;
161        let Shape {
162            widths,
163            local_copies,
164        } = shape(rule, scan.pools);
165        // Local scopes validate and expand their own alternatives during IR
166        // compilation. Only ordinary head/body occurrences drive this cursor.
167        let count = widths
168            .iter()
169            .fold(1_u128, |count, width| count.saturating_mul(*width));
170        budget.charge(ExpansionResource::Templates, count, location)?;
171        budget.charge(
172            ExpansionResource::Origins,
173            count.saturating_mul(origins as u128),
174            location,
175        )?;
176        let copies = local_copies.saturating_add(1);
177        let nodes = scan.nodes.saturating_mul(copies);
178        // A conservative cumulative projection bound is checked before cloning;
179        // final analysis independently counts the exact, deduplicated program.
180        *projection_nodes = projection_nodes.saturating_add(nodes.saturating_mul(count));
181        crate::formula::ceiling(
182            FormulaResource::AnalysisNodes,
183            *projection_nodes,
184            limits.max_analysis_nodes as u128,
185            location,
186        )?;
187        budget.charge(
188            ExpansionResource::Values,
189            scan.nodes.saturating_mul(copies).saturating_mul(count),
190            location,
191        )?;
192        let positions = vec![0; widths.len()];
193        let widths = widths
194            .into_iter()
195            .map(|width| usize::try_from(width).expect("bounded pool width"))
196            .collect();
197        Ok(Some(Self {
198            source,
199            widths,
200            positions,
201            remaining: usize::try_from(count).expect("bounded by usize template limit"),
202            nodes: nodes.saturating_mul(4),
203            bytes: scan.bytes.saturating_mul(copies).saturating_mul(4),
204        }))
205    }
206
207    fn next(
208        &mut self,
209        budget: &mut Budget,
210        location: Location,
211    ) -> Result<Option<WithProvenance<Statement>>, FormulaFailure> {
212        if self.remaining == 0 {
213            return Ok(None);
214        }
215        // Reserve four equivalents of counted term cells and text payload for
216        // the owned rewrite, selected children and retained analysis statement.
217        // Other AST carriers, provenance and allocator overhead are excluded;
218        // source/syntax and projection-node limits bound structural expansion.
219        budget.charge(ExpansionResource::TermWork, self.nodes, location)?;
220        budget.charge(ExpansionResource::ScalarBytes, self.bytes, location)?;
221        let mut selector = Select {
222            positions: self.positions.iter(),
223        };
224        let expanded = rewrite(
225            Program::of_nodes([self.source.clone()]),
226            &mut Expand {
227                selector: &mut selector,
228            },
229        );
230        assert!(
231            selector.positions.next().is_none(),
232            "all pool occurrences selected"
233        );
234        let statement = expanded
235            .statements()
236            .next()
237            .expect("one rule remains one rule")
238            .clone();
239        self.remaining -= 1;
240        for (position, width) in self.positions.iter_mut().zip(&self.widths).rev() {
241            *position += 1;
242            if *position < *width {
243                break;
244            }
245            *position = 0;
246        }
247        Ok(Some(statement))
248    }
249}

251struct Shape {
252    widths: Vec<u128>,
253    local_copies: u128,
254}
255
256fn shape(rule: &themelios_program::program::Rule, capacity: usize) -> Shape {
257    let mut widths = Vec::with_capacity(capacity);
258    let mut local_copies = 0_u128;
259    match rule.head().get() {
260        Head::Literal(literal) => {
261            outer(literal, &mut widths);
262        }
263        Head::Disjunction(head) => {
264            for element in head.elements() {
265                if unconditional(element.get().condition()) {
266                    outer(element.get().literal(), &mut widths);
267                }
268            }
269        }
270        Head::Choice(choice) => {
271            guard_width(choice.left_guard(), &mut widths);
272            guard_width(choice.right_guard(), &mut widths);
273            for element in choice.elements() {
274                if let LiteralInner::Atom(atom) = &element.get().literal().inner {
275                    let (count, _) = atom_width(atom.get());
276                    local_copies = local_copies.saturating_add(count.saturating_sub(1));
277                }
278            }
279        }
280        Head::Aggregate(aggregate) => {
281            guard_width(aggregate.left_guard(), &mut widths);
282            guard_width(aggregate.right_guard(), &mut widths);
283        }
284        _ => {}
285    }
286    for element in rule.body().get().elements() {
287        if let BodyElement::Literal(literal) = element.get() {
288            outer(literal, &mut widths);
289        } else if let BodyElement::Aggregate { aggregate, .. } = element.get() {
290            let guarded: &dyn HasGuards = match aggregate {
291                Aggregate::Function(value) => value,
292                Aggregate::Set(value) => value,
293            };
294            guard_width(guarded.left_guard(), &mut widths);
295            guard_width(guarded.right_guard(), &mut widths);
296        }
297    }
298    Shape {
299        widths,
300        local_copies,
301    }
302}
303
304fn outer(literal: &Literal, widths: &mut Vec<u128>) {
305    let (width, pools) = literal_width(literal);
306    if pools != 0 {
307        widths.push(width);
308    }
309}

311fn unconditional(condition: &Condition) -> bool {
312    condition.literals().all(|literal| {
313        matches!(
314            (&literal.get().inner, literal.get().negation),
315            (
316                LiteralInner::True,
317                DefaultNegation::None | DefaultNegation::NotNot
318            ) | (LiteralInner::False, DefaultNegation::Not)
319        )
320    })
321}

323fn guard_width(guard: Option<&WithProvenance<Guard>>, widths: &mut Vec<u128>) {
324    if let Some(guard) = guard
325        && matches!(guard.get().term, Term::Pool(_))
326    {
327        widths.push(term_width(&guard.get().term) as u128);
328    }
329}
330
331fn term_width(term: &Term) -> usize {
332    if let Term::Pool(items) = term {
333        items.len()
334    } else {
335        1
336    }
337}

339pub(super) fn atom_width(atom: &Atom) -> (u128, usize) {
340    let mut pools = usize::from(matches!(atom.arguments, Arguments::Pooled(_)));
341    let count = atom.alternatives().fold(0_u128, |sum, arguments| {
342        let product = arguments.iter().fold(1_u128, |product, term| {
343            pools += usize::from(matches!(term, Term::Pool(_)));
344            product.saturating_mul(term_width(term) as u128)
345        });
346        sum.saturating_add(product)
347    });
348    (count, pools)
349}

351pub(super) fn select_atom(mut atom: Atom, mut position: usize) -> Atom {
352    let arguments = atom
353        .alternatives()
354        .find_map(|arguments| {
355            let width = arguments
356                .iter()
357                .fold(1_usize, |width, term| width * term_width(term));
358            if position >= width {
359                position -= width;
360                None
361            } else {
362                let mut selected = arguments.to_vec();
363                for term in selected.iter_mut().rev() {
364                    if let Term::Pool(items) = term {
365                        let index = position % items.len();
366                        position /= items.len();
367                        *term = items[index].clone();
368                    }
369                }
370                Some(selected)
371            }
372        })
373        .expect("bounded occurrence position");
374    atom.arguments = Arguments::Single(arguments);
375    atom
376}

378struct Select<'a> {
379    positions: std::slice::Iter<'a, usize>,
380}
381impl Select<'_> {
382    fn guard(&mut self, guard: Option<&WithProvenance<Guard>>) -> Option<Guard> {
383        guard.map(|guard| {
384            let mut guard = guard.get().clone();
385            if let Term::Pool(items) = &guard.term {
386                guard.term =
387                    items[*self.positions.next().expect("guard occurrence position")].clone();
388            }
389            guard
390        })
391    }
392}
393impl Rewrite for Select<'_> {
394    fn tag(&self) -> TransformTag {
395        TransformTag::new("zetesis-finite-pools")
396    }
397    fn rewrite_atom(&mut self, atom: Atom) -> Atom {
398        if atom_width(&atom).1 == 0 {
399            atom
400        } else {
401            select_atom(
402                atom,
403                *self.positions.next().expect("atom occurrence position"),
404            )
405        }
406    }
407    fn rewrite_comparison(&mut self, comparison: Comparison) -> Comparison {
408        let pooled = std::iter::once(comparison.first())
409            .chain(comparison.steps().map(|(_, term)| term))
410            .any(|term| matches!(term, Term::Pool(_)));
411        if pooled {
412            select_comparison(
413                &comparison,
414                *self
415                    .positions
416                    .next()
417                    .expect("comparison occurrence position"),
418            )
419        } else {
420            comparison
421        }
422    }
423}

425struct Expand<'a, 'b> {
426    selector: &'a mut Select<'b>,
427}
428impl Rewrite for Expand<'_, '_> {
429    fn tag(&self) -> TransformTag {
430        self.selector.tag()
431    }
432    fn rewrite_body(&mut self, body: Body) -> Body {
433        // The conditional compiler owns inner-OR alternatives. Outer pool
434        // positions never select or duplicate a universal consequent.
435        Body::new(body.elements().map(|element| match element.get() {
436            BodyElement::Literal(literal) => {
437                BodyElement::Literal(self.selector.rewrite_literal(literal.clone()))
438            }
439            BodyElement::Aggregate {
440                negation,
441                aggregate,
442            } => {
443                let aggregate = match aggregate {
444                    Aggregate::Function(value) => Aggregate::Function(FunctionAggregate::new(
445                        self.selector.guard(value.left_guard()),
446                        value.function(),
447                        value.elements().map(|element| element.get().clone()),
448                        self.selector.guard(value.right_guard()),
449                    )),
450                    Aggregate::Set(value) => Aggregate::Set(SetAggregate::new(
451                        self.selector.guard(value.left_guard()),
452                        value.elements().map(|element| element.get().clone()),
453                        self.selector.guard(value.right_guard()),
454                    )),
455                };
456                BodyElement::Aggregate {
457                    negation: *negation,
458                    aggregate,
459                }
460            }
461            other => other.clone(),
462        }))
463    }
464    fn rewrite_head(&mut self, head: Head) -> Head {
465        if let Head::Disjunction(disjunction) = head {
466            return Head::Disjunction(Disjunction::new(disjunction.elements().map(|element| {
467                let literal = element.get().literal();
468                DisjunctionElement::new(
469                    if unconditional(element.get().condition()) {
470                        self.selector.rewrite_literal(literal.clone())
471                    } else {
472                        literal.clone()
473                    },
474                    element.get().condition().clone(),
475                )
476            })));
477        }
478        if let Head::Aggregate(value) = head {
479            return Head::Aggregate(HeadAggregate::new(
480                self.selector.guard(value.left_guard()),
481                value.function(),
482                value.elements().map(|element| element.get().clone()),
483                self.selector.guard(value.right_guard()),
484            ));
485        }
486        let Head::Choice(choice) = head else {
487            return self.selector.rewrite_head(head);
488        };
489        let mut elements = Vec::new();
490        for element in choice.elements() {
491            let literal = element.get().literal();
492            if let LiteralInner::Atom(atom) = &literal.inner {
493                let (count, _) = atom_width(atom.get());
494                for index in 0..usize::try_from(count).expect("bounded local copies") {
495                    elements.push(ChoiceElement::new(
496                        Literal {
497                            negation: literal.negation,
498                            inner: LiteralInner::Atom(
499                                atom.clone().map(|atom| select_atom(atom, index)),
500                            ),
501                        },
502                        element.get().condition().clone(),
503                    ));
504                }
505            } else {
506                elements.push(element.get().clone());
507            }
508        }
509        // The enclosing rule/head retains parsed origins. Synthesized element
510        // nodes truthfully carry constructed provenance, like expanded facts.
511        Head::Choice(Choice::new(
512            self.selector.guard(choice.left_guard()),
513            elements,
514            self.selector.guard(choice.right_guard()),
515        ))
516    }
517}

519/// Logical syntax-copy units shared by source normalization and terminal
520/// partitioning. These exclude other AST carriers, provenance and allocator
521/// overhead; they are not retained-capacity or whole-heap measurements.
522pub(crate) fn source_copy_cost(statement: &Statement) -> (u128, u128) {
523    let mut footprint = Footprint {
524        nodes: 1,
525        ..Footprint::default()
526    };
527    footprint.visit_statement(statement);
528    (footprint.nodes, footprint.bytes)
529}
530
531#[derive(Default)]
532struct Footprint {
533    nodes: u128,
534    bytes: u128,
535    pools: usize,
536}
537impl Visit for Footprint {
538    fn visit_atom(&mut self, atom: &Atom) {
539        self.nodes += 1;
540        self.bytes += atom.name.as_str().len() as u128;
541        self.pools += usize::from(matches!(atom.arguments, Arguments::Pooled(_)));
542        for term in atom.argument_terms() {
543            for node in term.subterms() {
544                self.visit_term(node);
545            }
546        }
547    }
548    fn visit_term(&mut self, term: &Term) {
549        self.nodes += 1;
550        self.bytes += std::mem::size_of::<Term>() as u128;
551        self.pools += usize::from(matches!(term, Term::Pool(_)));
552        if let Term::Function { name, .. } = term {
553            self.bytes += name.as_str().len() as u128;
554        }
555        if let Term::Symbolic(symbol) = term {
556            self.bytes += crate::structural_value::symbol_bytes(symbol);
557        }
558        if let Term::Variable(themelios_program::term::Variable::Named(name)) = term {
559            self.bytes += name.as_str().len() as u128;
560        }
561    }
