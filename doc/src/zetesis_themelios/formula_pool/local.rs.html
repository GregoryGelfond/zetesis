11impl Compiler<'_> {
12    pub(crate) fn literal_alternatives(
13        &mut self,
14        literal: &Literal,
15    ) -> Result<Vec<Literal>, FormulaFailure> {
16        Ok(self
17            .condition_alternatives(&Condition::new([literal.clone()]))?
18            .into_iter()
19            .map(|condition| {
20                condition
21                    .literals()
22                    .next()
23                    .expect("one selected literal")
24                    .get()
25                    .clone()
26            })
27            .collect())
28    }
29
30    /// Local condition alternatives retain independent variable environments.
31    pub(crate) fn condition_alternatives(
32        &mut self,
33        condition: &Condition,
34    ) -> Result<Vec<Condition>, FormulaFailure> {
35        Ok(self
36            .local_alternatives(&[], condition)?
37            .into_iter()
38            .map(|(_, condition)| condition)
39            .collect())
40    }

42    /// Fields and conditions form a product within one original element. The
43    /// caller decides whether those occurrences contribute a union, conjunction
44    /// or separate rules; this cursor never chooses that semantic composition.
45    pub(crate) fn local_alternatives(
46        &mut self,
47        fields: &[&Term],
48        condition: &Condition,
49    ) -> Result<Vec<(Vec<Term>, Condition)>, FormulaFailure> {
50        let mut scan = Footprint::default();
51        for term in fields.iter().flat_map(|term| term.subterms()) {
52            scan.visit_term(term);
53        }
54        scan.visit_condition(condition);
55        let count = fields
56            .iter()
57            .fold(1_u128, |n, term| n.saturating_mul(term_width(term) as u128));
58        let count = condition.literals().fold(count, |n, literal| {
59            n.saturating_mul(literal_width(literal.get()).0)
60        });
61        self.budget
62            .charge(ExpansionResource::TermWork, scan.nodes, self.location)?;
63        self.budget
64            .charge(ExpansionResource::Values, count, self.location)?;
65        self.budget.charge(
66            ExpansionResource::TermWork,
67            count.saturating_mul(scan.nodes + 1),
68            self.location,
69        )?;
70        let bytes = count.saturating_mul(
71            scan.bytes.saturating_mul(4)
72                + std::mem::size_of::<Condition>() as u128
73                + fields.len() as u128 * std::mem::size_of::<Term>() as u128,
74        );
75        self.budget
76            .charge(ExpansionResource::ScalarBytes, bytes, self.location)?;
77        self.budget.check_family(
78            bytes.saturating_add(count.saturating_mul(std::mem::size_of::<Vec<Term>>() as u128)),
79            self.location,
80        )?;
81        let count = usize::try_from(count).expect("finite value allowance bounds occurrences");
82        let mut result = Vec::with_capacity(count);
83        for mut position in 0..count {
84            self.budget.poll(self.location)?;
85            let mut stride = count;
86            let mut terms = Vec::with_capacity(fields.len());
87            for term in fields {
88                let width = term_width(term);
89                stride /= width;
90                terms.push(select_term(term, position / stride));
91                position %= stride;
92            }
93            let condition = Condition::new(condition.literals().map(|literal| {
94                let width =
95                    usize::try_from(literal_width(literal.get()).0).expect("bounded literal width");
96                stride /= width;
97                let selected = select_literal(literal.get(), position / stride);
98                position %= stride;
99                selected
100            }));
101            result.push((terms, condition));
102        }
103        Ok(result)
104    }
105}

107pub(super) fn literal_width(literal: &Literal) -> (u128, usize) {
108    match &literal.inner {
109        LiteralInner::Atom(atom) => atom_width(atom.get()),
110        LiteralInner::Comparison(comparison) => std::iter::once(comparison.get().first())
111            .chain(comparison.get().steps().map(|(_, term)| term))
112            .fold((1_u128, 0), |(width, pools), term| {
113                (
114                    width.saturating_mul(term_width(term) as u128),
115                    pools + usize::from(matches!(term, Term::Pool(_))),
116                )
117            }),
118        LiteralInner::True | LiteralInner::False => (1, 0),
119    }
120}

122pub(super) fn select_literal(literal: &Literal, position: usize) -> Literal {
123    Literal {
124        negation: literal.negation,
125        inner: match &literal.inner {
126            LiteralInner::Atom(atom) => {
127                LiteralInner::Atom(atom.clone().map(|atom| select_atom(atom, position)))
128            }
129            LiteralInner::Comparison(comparison) => LiteralInner::Comparison(
130                comparison
131                    .clone()
132                    .map(|comparison| select_comparison(&comparison, position)),
133            ),
134            other => other.clone(),
135        },
136    }
137}

139pub(super) fn select_comparison(comparison: &Comparison, mut position: usize) -> Comparison {
140    let mut select = |term: &Term| {
141        let width = term_width(term);
142        let term = select_term(term, position % width);
143        position /= width;
144        term
145    };
146    let first = select(comparison.first());
147    let mut steps = comparison.steps();
148    let (relation, term) = steps.next().expect("comparison has a step");
149    let mut result = Comparison::new(first, relation, select(term));
150    for (relation, term) in steps {
151        result = result.chain(relation, select(term));
152    }
153    result
154}
155
156fn select_term(term: &Term, position: usize) -> Term {
157    if let Term::Pool(items) = term {
158        items[position].clone()
159    } else {
160        term.clone()
161    }
