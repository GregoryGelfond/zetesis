1//! Finite term alternatives composed before their source scope is expanded.
2//!
3//! The upstream rewrite visits children first. Each child therefore has one
4//! outer pool at most; this pass distributes those alternatives through one
5//! constructor or operator. It never evaluates intervals or erases empty ranges:
6//! their endpoint checks and variable safety belong to the binding consumer.
7
8use themelios_base::span::Location;
9use themelios_program::term::{Term, TermParts};
10use themelios_program::transform::Visit;
11
12use super::Footprint;
13use crate::expansion::Budget;
14use crate::{ExpansionFailure, ExpansionResource};
15
16pub(crate) fn distribute(
17    term: Term,
18    budget: &mut Budget,
19    location: Location,
20) -> Result<Term, ExpansionFailure> {
21    // Pool-free nodes retain the original normalizer and its charging order.
22    let pooled = match &term {
23        Term::Function { arguments, .. }
24        | Term::Tuple(arguments)
25        | Term::External { arguments, .. } => arguments.iter().any(is_pool),
26        Term::UnaryOperation { argument, .. } | Term::Absolute(argument) => is_pool(argument),
27        Term::BinaryOperation { left, right, .. }
28        | Term::Interval {
29            lower: left,
30            upper: right,
31        } => is_pool(left) || is_pool(right),
32        _ => false,
33    };
34    if !pooled {
35        return Ok(term);
36    }
37    match term.into_parts() {
38        TermParts::Function { name, arguments } => product(
39            &arguments,
40            name.as_str().len(),
41            budget,
42            location,
43            |arguments| Term::Function {
44                name: name.clone(),
45                arguments,
46            },
47        ),
48        TermParts::Tuple(arguments) => product(&arguments, 0, budget, location, Term::Tuple),
49        TermParts::External { name, arguments } => product(
50            &arguments,
51            name.as_str().len(),
52            budget,
53            location,
54            |arguments| Term::External {
55                name: name.clone(),
56                arguments,
57            },
58        ),
59        TermParts::UnaryOperation { operator, argument } => {
60            product(&[argument], 0, budget, location, |mut arguments| {
61                Term::UnaryOperation {
62                    operator,
63                    argument: Box::new(arguments.pop().expect("one unary argument")),
64                }
65            })
66        }
67        TermParts::Absolute(argument) => {
68            product(&[argument], 0, budget, location, |mut arguments| {
69                Term::Absolute(Box::new(arguments.pop().expect("one absolute argument")))
70            })
71        }
72        TermParts::BinaryOperation {
73            operator,
74            left,
75            right,
76        } => product(&[left, right], 0, budget, location, |mut arguments| {
77            let right = arguments.pop().expect("right binary argument");
78            let left = arguments.pop().expect("left binary argument");
79            Term::BinaryOperation {
80                operator,
81                left: Box::new(left),
82                right: Box::new(right),
83            }
84        }),
85        TermParts::Interval { lower, upper } => {
86            product(&[lower, upper], 0, budget, location, |mut arguments| {
87                let upper = arguments.pop().expect("upper interval endpoint");
88                let lower = arguments.pop().expect("lower interval endpoint");
89                Term::Interval {
90                    lower: Box::new(lower),
91                    upper: Box::new(upper),
92                }
93            })
94        }
95        _ => unreachable!("only compound nodes with pooled children are distributed"),
96    }
97}

99fn is_pool(term: &Term) -> bool {
100    matches!(term, Term::Pool(_))
101}
102
103fn alternatives(term: &Term) -> &[Term] {
104    if let Term::Pool(items) = term {
105        items
106    } else {
107        std::slice::from_ref(term)
108    }
109}

111fn product(
112    arguments: &[Term],
113    name_bytes: usize,
114    budget: &mut Budget,
115    location: Location,
116    build: impl Fn(Vec<Term>) -> Term,
117) -> Result<Term, ExpansionFailure> {
118    let count = arguments.iter().fold(1_u128, |count, argument| {
119        count.saturating_mul(alternatives(argument).len() as u128)
120    });
121    let mut footprint = Footprint::default();
122    for term in arguments.iter().flat_map(Term::subterms) {
123        footprint.visit_term(term);
124    }
125    budget.charge(ExpansionResource::TermWork, footprint.nodes, location)?;
126    budget.charge(ExpansionResource::Values, count, location)?;
127    budget.charge(
128        ExpansionResource::TermWork,
129        count.saturating_mul(footprint.nodes.saturating_add(1)),
130        location,
131    )?;
132    // Reserve selected term/text copies, the output root and cursor cells before
133    // retaining the family. This is cumulative logical payload, not allocator RSS.
134    budget.charge(
135        ExpansionResource::ScalarBytes,
136        count
137            .saturating_mul(
138                footprint
139                    .bytes
140                    .saturating_mul(2)
141                    .saturating_add((std::mem::size_of::<Term>() + name_bytes) as u128),
142            )
143            .saturating_add(arguments.len() as u128 * std::mem::size_of::<usize>() as u128),
144        location,
145    )?;
146    let count = usize::try_from(count).expect("finite value allowance fits usize");
147    let mut output = Vec::with_capacity(count);
148    let mut positions = vec![0; arguments.len()];
149    for _ in 0..count {
150        let selected = arguments
151            .iter()
152            .zip(&positions)
153            .map(|(argument, &position)| alternatives(argument)[position].clone())
154            .collect();
155        output.push(build(selected));
156        for (position, argument) in positions.iter_mut().zip(arguments).rev() {
157            *position += 1;
158            if *position < alternatives(argument).len() {
159                break;
160            }
161            *position = 0;
162        }
163    }
164    Ok(Term::pool(output).expect("source pool products are nonempty"))
165}
