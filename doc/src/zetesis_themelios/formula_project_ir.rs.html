1//! Projection declarations reuse bounded source-rule compilation without becoming
2//! logical producers. Their temporary dependency projections are discarded;
3//! declarations contribute no statements to the retained source analysis.
4
5use crate::ProgramSite;
6use themelios_program::program::{
7    DefaultNegation, Literal, LiteralInner, Project, Rule, Statement,
8};
9use themelios_program::provenance::WithProvenance;
10
11use crate::FormulaFailure;
12use crate::formula_ir::{Compiler, RuleIr};
13
14impl Compiler<'_> {
15    pub(super) fn project_statement(
16        &mut self,
17        statement: &WithProvenance<Statement>,
18        origins: &[ProgramSite],
19        projection_nodes: &mut u128,
20        declarations: &mut Vec<RuleIr>,
21    ) -> Result<bool, FormulaFailure> {
22        let Statement::Project(project) = statement.get() else {
23            return Ok(false);
24        };
25        match project {
26            Project::Signature(_) => {}
27            Project::Atom { atom, body } => {
28                let literal = Literal {
29                    negation: DefaultNegation::None,
30                    inner: LiteralInner::Atom(atom.clone()),
31                };
32                let rule = WithProvenance::new(
33                    Statement::Rule(Rule::new(literal, body.get().clone())),
34                    statement.provenance().clone(),
35                );
36                let mut discarded = Vec::new();
37                let dependency_projection = self.dependency_projection;
38                // Declaration constants belong to this scope, not the logical
39                // rule domain. Aggregate IDs remain unique across both owners.
40                let empty = self.empty_domain()?;
41                let domain = std::mem::replace(&mut self.domain, empty);
42                let result = self.source_rules(
43                    &rule,
44                    origins,
45                    projection_nodes,
46                    declarations,
47                    &mut discarded,
48                );
49                self.domain = domain;
50                self.dependency_projection = dependency_projection;
51                result?;
52            }
53        }
54        Ok(true)
55    }
56}
