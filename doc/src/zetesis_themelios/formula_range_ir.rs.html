1//! Shared finite interval lowering for scoped structural value consumers.
2//!
3//! The iterative fold moves child plans in postorder and charges every moved
4//! operation. Each interval owns a private range slot; consumers retain their
5//! outer-rule, local-element, or conditional-alternative composition boundary.
6
7use crate::diagnostic::unsupported;
8use crate::formula_ir::{Compiler, Expression, LiteralIr, Operation, Variables};
9use crate::{ExpansionResource, FormulaFailure, ProfileFeature};
10use themelios_program::term::{Term, TermParts, UnaryOp, Variable};
11
12impl Compiler<'_> {
13    pub(super) fn ranged_expression(
14        &mut self,
15        term: &Term,
16        variables: &mut Variables,
17        bindings: &mut Vec<LiteralIr>,
18    ) -> Result<Expression, FormulaFailure> {
19        if !term
20            .subterms()
21            .any(|node| matches!(node, Term::Interval { .. }))
22        {
23            return self.expression(term, variables);
24        }
25        self.ranged_plan_preflight(term)?;
26        term.clone().try_fold(|parts| {
27            self.budget
28                .charge(ExpansionResource::TermWork, 1, self.location)?;
29            let (children, operation) = match parts {
30                TermParts::Variable(variable) => {
31                    return self.expression(&Term::Variable(variable), variables);
32                }
33                TermParts::Symbolic(symbol) => {
34                    return self.expression(&Term::Symbolic(symbol), variables);
35                }
36                TermParts::Interval { lower, upper } => {
37                    let target = self.consequent_slot(variables)?;
38                    self.budget.charge(
39                        ExpansionResource::ScalarBytes,
40                        (3 * std::mem::size_of::<LiteralIr>() + std::mem::size_of::<Operation>())
41                            as u128,
42                        self.location,
43                    )?;
44                    bindings.push(LiteralIr::Range {
45                        target,
46                        lower,
47                        upper,
48                        binder: false,
49                    });
50                    return Ok(Expression {
51                        nodes: vec![Operation::Variable(target)],
52                    });
53                }
54                TermParts::UnaryOperation {
55                    operator,
56                    mut argument,
57                } => {
58                    if operator == UnaryOp::Negate
59                        && let Some(Operation::Constructor(constructor)) = argument.nodes.last_mut()
60                        && let Some(shape) = self.source.negate_constructor(
61                            constructor.shape,
62                            self.limits,
63                            self.counters,
64                            self.location,
65                        )?
66                    {
67                        constructor.shape = shape;
68                        return Ok(argument);
69                    }
70                    (vec![argument], Pending::Unary(operator))
71                }
72                TermParts::BinaryOperation {
73                    operator,
74                    left,
75                    right,
76                } => (vec![left, right], Pending::Binary(operator)),
77                TermParts::Absolute(argument) => (vec![argument], Pending::Absolute),
78                TermParts::Function { name, arguments } => {
79                    let shape = self.source.constructor(
80                        zetesis_core::ValueNodeRef::Function {
81                            name: name.as_str(),
82                            sign: zetesis_core::Sign::Positive,
83                            arity: arguments.len(),
84                        },
85                        self.limits,
86                        self.counters,
87                        self.location,
88                    )?;
89                    (arguments, Pending::Constructor(shape))
90                }
91                TermParts::Tuple(arguments) => {
92                    let shape = self.source.constructor(
93                        zetesis_core::ValueNodeRef::Tuple {
94                            arity: arguments.len(),
95                        },
96                        self.limits,
97                        self.counters,
98                        self.location,
99                    )?;
100                    (arguments, Pending::Constructor(shape))
101                }
102                _ => return Err(unsupported(ProfileFeature::Term, self.location).into()),
103            };
104            self.join_value_plans(children, operation)
105        })
106    }
107
108    /// Admit the source fold and leaf-plan storage before cloning the tree.
109    fn ranged_plan_preflight(&mut self, term: &Term) -> Result<(), FormulaFailure> {
110        self.value_plan_preflight(term)?;
111        // Reserve the source fold and its leaf plans before cloning. Subsequent
112        // concatenations reserve their own operation storage before allocation.
113        let count = term.subterms().count() as u128;
114        self.budget
115            .charge(ExpansionResource::TermWork, count * 3, self.location)?;
116        let text = term
117            .subterms()
118            .map(|node| match node {
119                Term::Symbolic(symbol) => crate::structural_value::symbol_bytes(symbol),
120                Term::Function { name, .. } => name.as_str().len() as u128,
121                Term::Variable(Variable::Named(name)) => name.as_str().len() as u128,
122                _ => 0,
123            })
124            .sum::<u128>();
125        self.budget.charge(
126            ExpansionResource::ScalarBytes,
127            text + count * (std::mem::size_of::<Term>() + std::mem::size_of::<Operation>()) as u128,
128            self.location,
129        )?;
130        Ok(())
131    }
132
133    fn join_value_plans(
134        &mut self,
135        children: Vec<Expression>,
136        operation: Pending,
137    ) -> Result<Expression, FormulaFailure> {
138        let count = children
139            .iter()
140            .map(|child| child.nodes.len() as u128)
141            .sum::<u128>()
142            + 1;
143        self.budget
144            .charge(ExpansionResource::TermWork, count, self.location)?;
145        self.budget.charge(
146            ExpansionResource::ScalarBytes,
147            count * std::mem::size_of::<Operation>() as u128
148                + children.len() as u128 * std::mem::size_of::<usize>() as u128,
149            self.location,
150        )?;
151        let mut nodes = Vec::new();
152        let mut roots = Vec::new();
153        for child in children {
154            let offset = nodes.len();
155            for mut node in child.nodes {
156                match &mut node {
157                    Operation::Unary(_, input) | Operation::Absolute(input) => *input += offset,
158                    Operation::Binary(_, left, right) => {
159                        *left += offset;
160                        *right += offset;
161                    }
162                    Operation::Constructor(constructor) => {
163                        for input in &mut constructor.arguments {
164                            *input += offset;
165                        }
166                    }
167                    Operation::Variable(_) | Operation::Constant(_) => {}
168                }
169                nodes.push(node);
170            }
171            roots.push(nodes.len() - 1);
172        }
173        nodes.push(match operation {
174            Pending::Unary(operator) => Operation::Unary(operator, roots[0]),
175            Pending::Binary(operator) => Operation::Binary(operator, roots[0], roots[1]),
176            Pending::Absolute => Operation::Absolute(roots[0]),
177            Pending::Constructor(shape) => {
178                Operation::Constructor(Box::new(crate::formula_value::Constructor {
179                    shape,
180                    arguments: roots,
181                }))
182            }
183        });
184        Ok(Expression { nodes })
185    }
186}
187
188#[derive(Clone, Copy)]
189enum Pending {
190    Unary(UnaryOp),
191    Binary(themelios_program::term::BinaryOp),
192    Absolute,
193    Constructor(crate::formula_support::components::Constructor),
194}
