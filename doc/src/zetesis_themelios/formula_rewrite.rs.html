1//! Shared original-owner replacement and lazy fixed-fact evidence.

3use std::collections::BTreeMap;
4
5#[cfg(test)]
6use themelios_program::program::Program;
7use themelios_program::program::Rule;
8use themelios_program::provenance::Provenance;
9use zetesis_domain::{FactIndex, KeyWork, Stop};
10
11use crate::StatementId;
12
13pub(crate) struct Replacement {
14    pub(crate) provenance: Provenance,
15    pub(crate) rules: Vec<Rule>,
16    pub(crate) tag: &'static str,
17}
18
19pub(crate) type Replacements = BTreeMap<StatementId, Replacement>;
20
21/// Optional key work stops remain distinct from a refusal to construct the
22/// complete fact input under the mandatory preparation expansion budget.
23#[derive(Debug)]
24pub(crate) enum FactFailure {
25    Stopped(Stop),
26    Source(crate::FormulaFailure),
27}
28
29enum Input<'p> {
30    Summary(&'p crate::formula_analysis::summary::PreparationFacts<'p>),
31    #[cfg(test)]
32    Program(&'p Program),
33}
34
35/// The index borrows atoms owned by the outer preparation reader. It never owns
36/// the storage it indexes, and a stopped reading publishes no partial index.
37pub(crate) struct FixedFacts<'p> {
38    input: Input<'p>,
39    index: Option<FactIndex<'p>>,
40}
41
42impl<'p> FixedFacts<'p> {
43    pub(crate) const fn summary(
44        reader: &'p crate::formula_analysis::summary::PreparationFacts<'p>,
45    ) -> Self {
46        Self {
47            input: Input::Summary(reader),
48            index: None,
49        }
50    }
51
52    #[cfg(test)]
53    pub(crate) const fn new(program: &'p Program) -> Self {
54        Self {
55            input: Input::Program(program),
56            index: None,
57        }
58    }
59
60    pub(crate) fn read(
61        &mut self,
62        work: &mut KeyWork,
63        budget: &mut crate::expansion::Budget,
64        site: crate::ProgramSite,
65    ) -> Result<&FactIndex<'p>, FactFailure> {
66        if self.index.is_none() {
67            let index = match self.input {
68                Input::Summary(reader) => {
69                    let atoms = reader.atoms(budget, site).map_err(FactFailure::Source)?;
70                    FactIndex::read_parts(reader.summary.producers(), atoms, work)
71                }
72                #[cfg(test)]
73                Input::Program(program) => FactIndex::read(program, work),
74            }
75            .map_err(FactFailure::Stopped)?;
76            self.index = Some(index);
77        }
78        Ok(self.index.as_ref().expect("complete fixed-fact reading"))
79    }
80}
