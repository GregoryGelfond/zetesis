1//! Exact finite source evaluation for a complete signed acyclic normal program.
2//!
3//! The result is a source subject, not a replacement formula theory. Complete
4//! support and original-family arithmetic admission precede the one dependency
5//! ordered activity scan. The ordinary ground door remains the authority for
6//! arbitrary original and reduct queries.

19/// The complete source class either admits one exact subject or returns the
20/// original preparation, including all accepted classification charges.
21#[derive(Debug)]
22pub enum SourceAdmission<P> {
23    /// A complete signed acyclic normal source, evaluated exactly once.
24    Admitted(AcyclicSource),
25    /// The original preparation still owns its ordinary grounding/query door.
26    NotApplicable(P),
27}
28
29/// All accepted source preparation and evaluation charges. These are admission
30/// totals, not work attributed to later answer publication or a search worker.
31#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
32pub struct SourceEvaluationStatistics {
33    /// Cumulative formula work, including support, arithmetic and exact activity.
34    pub work: u64,
35    /// Cumulative complete substitutions accepted by the original formula budget.
36    pub substitutions: u64,
37}

39struct Admitted {
40    source: Owner,
41    metadata: SourceMetadata,
42    evaluation: Evaluated,
43}

45/// Immutable exact interpretation of one admitted acyclic source program.
46/// Clones share its original identity and canonical typed atom payload. No
47/// Boolean theory is synthesized and no formula-query contract is changed.
48#[derive(Clone)]
49pub struct AcyclicSource(Arc<Admitted>);

51impl fmt::Debug for AcyclicSource {
52    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
53        f.debug_struct("AcyclicSource")
54            .field("source", &self.0.source)
55            .field("atoms", &self.atom_catalog().atoms().len())
56            .field("satisfiable", &self.selected_positions().is_some())
57            .finish_non_exhaustive()
58    }
59}

200/// Classification keeps every accepted charge in the original preparation.
201/// An owned order retains the existing canonical source token; it cannot be
202/// applied to another independently prepared owner after either value moves.
203pub(crate) fn classify(
204    preparation: &mut Preparation,
205) -> Result<Option<AcyclicPlan>, FormulaFailure> {
206    preparation.accounting.controlled::<_, FormulaFailure>(
207        preparation.budget.cancellation(),
208        crate::grounding_observer::Work::default(),
209        preparation.location,
210        |counters| {
211            AcyclicPlan::prepare(
212                &preparation.program,
213                &preparation.catalog,
214                &preparation.limits,
215                counters,
216                preparation.location,
217            )
218        },
219    )
220}

222pub(crate) fn evaluate(
223    preparation: Preparation,
224    plan: AcyclicPlan,
225    observer: Option<&dyn GroundingObserver>,
226) -> Result<Evaluated, FormulaFailure> {
227    let Preparation {
228        program,
229        catalog,
230        mut accounting,
231        mut budget,
232        limits,
233        options,
234        location,
235    } = preparation;
236    let profile = Profile::new(observer);
237    let cancellation = budget.cancellation().cloned();
238    accounting.controlled::<_, FormulaFailure>(
239        cancellation.as_ref(),
240        profile.work(),
241        location,
242        |counters| {
243            counters.work(&limits, location)?;
244            if !plan.authenticates(&catalog) {
245                return Err(FormulaFailure::SupportRelation {
246                    error: zetesis_core::relation::Failure::Owner,
247                    location,
248                });
249            }
250            let (mut catalog, domains) = crate::formula_ground::complete_support(
251                &program,
252                catalog,
253                options.domains,
254                &mut budget,
255                &profile,
256                GroundingWork::new(&limits, counters, location),
257            )?;
258            // Complete-domain analysis may assist support; exact activity always reads
259            // the complete original rows and does not depend on an optional guard.
260            drop(domains);
261            let (exact, warnings) = {
262                let (completed, mut append) = catalog.split(&limits, counters, location)?;
263                let queries = completed.queries(options.joins, &limits, counters, location)?;
264                let mut computation = Computation::new(&mut append, queries.support());
265                let warnings =
266                    profile.phase(crate::GroundingPhase::SupportCompletion, None, || {
267                        crate::formula_ground::arithmetic::prepare(
268                            &program,
269                            queries.support(),
270                            &mut computation,
271                            &limits,
272                            &mut budget,
273                            counters,
274                        )
275                    })?;
276                let exact = formula_source_activity::evaluate_acyclic(
277                    &plan,
278                    &program,
279                    queries.support(),
280                    &profile,
281                    &mut formula_source_activity::Context {
282                        computation: &mut computation,
283                        limits: &limits,
284                        budget: &mut budget,
285                        counters,
286                        location,
287                    },
288                )?;
289                (exact, warnings)
290            };
291            let Exact {
292                atoms,
293                selected,
294                consistent,
295                lease,
296            } = exact;
297            let mut publication = Publication::new(&mut catalog, counters, location)?;
298            publication.observe_peak(size_of::<Admitted>() as u128, &limits, counters, location)?;
299            let atoms = publication.atoms(atoms, &limits, counters, location)?;
300            publication.retain(lease, &limits, counters, location)?;
301            counters.work(&limits, location)?;
302            let statistics = SourceEvaluationStatistics {
303                work: counters.accounting.work,
304                substitutions: counters.accounting.substitutions,
305            };
306            // All surviving maps have been admitted while their source and scratch
307            // coexist. The immutable output keeps the shared canonical payload only.
308            drop(publication);
309            drop(plan);
310            Ok(Evaluated {
311                atoms,
312                selected: consistent.then_some(selected),
313                warnings: warnings.into_values(),
314                statistics,
315                expansion: budget.usage(),
316                analysis: program.analysis,
317                analyzed: program.analyzed,
318                keyed_constraints: program.keyed_constraints,
319                key_analysis: program.key_analysis,
320            })
321        },
322    )
323}
