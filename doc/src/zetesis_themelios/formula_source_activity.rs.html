1//! Shared finite source activity for declarations and certified evaluation.
2//!
3//! Required/possible activity is a source grounding abstraction, not answer-set
4//! realization. Optional means retained as a grounding possibility; it does not
5//! assert that several optional literals can hold together. For example,
6//! `a,not a` can retain a priority while its original-model query is always false.
7//! Choices stay optional even when constraints force an answer.
8//! The original theory is never read or changed by this certificate.
9//! Completed support and whole-producer refinement establish source coverage;
10//! the separate `model_query` module constructs original-model objective queries
11//! only after row eligibility. Its retained-node limit does not bound activity.
12//! A private strict dependency-order certificate permits an exact producer scan
13//! through the same activity operations. Only that complete class, followed by
14//! constraint and coherence checks, can establish an original answer.
15

42fn ordinary(literals: &[LiteralIr]) -> bool {
43    literals.iter().all(|literal| {
44        matches!(
45            literal,
46            LiteralIr::Atom(..)
47                | LiteralIr::Compare(..)
48                | LiteralIr::Guard(_)
49                | LiteralIr::HeadGuard(_)
50                | LiteralIr::Bind { .. }
51                | LiteralIr::Range { .. }
52        )
53    })
54}

61/// A completed source atom can be absent, possible or required. This order
62/// makes finite conjunction the minimum and alternative producers the maximum.
63#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
64pub(crate) enum Activity {
65    Absent,
66    Optional,
67    Required,
68}

70impl Activity {
71    pub(crate) fn negate(self) -> Self {
72        match self {
73            Self::Absent => Self::Required,
74            Self::Optional => Self::Optional,
75            Self::Required => Self::Absent,
76        }
77    }

100impl Context<'_, '_, '_> {
101    fn work(&mut self) -> Result<(), FormulaFailure> {
102        self.counters.work(self.limits, self.location)
103    }

105    fn atom(
106        &mut self,
107        pattern: AtomPattern,
108        binding: &Binding<'_>,
109    ) -> Result<SourceAtom, FormulaFailure> {
110        let pattern =
111            self.computation
112                .static_pattern(pattern, self.limits, self.counters, self.location)?;
113        self.computation
114            .atom(pattern, binding, self.limits, self.counters, self.location)
115    }
116
117    fn pattern_matches(
118        &mut self,
119        pattern: AtomPattern,
120        expected: &Signature,
121    ) -> Result<bool, FormulaFailure> {
122        let pattern =
123            self.computation
124                .static_pattern(pattern, self.limits, self.counters, self.location)?;
125        self.matches_signature(pattern.predicate(), expected)
126    }
127
128    /// Compare the canonical predicate to an upstream analysis key without
129    /// copying its spelling into a second execution predicate.
130    fn matches_signature(
131        &mut self,
132        predicate: PredicateRef<'_>,
133        expected: &Signature,
134    ) -> Result<bool, FormulaFailure> {
135        self.work()?;
136        if crate::coherence::source_sign(predicate.sign()) != expected.sign
137            || predicate.arity() != expected.arity as usize
138        {
139            return Ok(false);
140        }
141        let actual = predicate.name();
142        self.counters
143            .charge_work(actual.len() as u128, self.limits, self.location)?;
144        Ok(actual == expected.name.as_str())
145    }
146
147    /// Whether a fact block's predicate is `expected`, read once for its rows.
148    fn produces_facts(
149        &mut self,
150        block: FactBlock<'_>,
151        expected: &Signature,
152    ) -> Result<bool, FormulaFailure> {
153        let predicate = self.computation.static_predicate(
154            block.predicate(),
155            self.limits,
156            self.counters,
157            self.location,
158        )?;
159        self.matches_signature(predicate, expected)
160    }
161
162    fn produces(&mut self, rule: &RuleIr, expected: &Signature) -> Result<bool, FormulaFailure> {
163        match &rule.head {
164            HeadIr::Normal(Some(head)) => self.pattern_matches(*head, expected),
165            HeadIr::Choice(group) => {
166                for element in &group.elements {
167                    self.work()?;
168                    if let Some(head) = element.head.positive_atom()
169                        && self.pattern_matches(*head, expected)?
170                    {
171                        return Ok(true);
172                    }
173                }
174                Ok(false)
175            }
176            HeadIr::Disjunction(_) | HeadIr::ConditionalDisjunction { .. } => {
177                for head in rule
178                    .head
179                    .disjuncts()
180                    .filter_map(|head| head.positive_atom())
181                {
182                    self.work()?;
183                    if self.pattern_matches(*head, expected)? {
184                        return Ok(true);
185                    }
186                }
187                Ok(false)
188            }
189            HeadIr::Normal(None) => Ok(false),
190        }
191    }
192
193    fn allocation(&self) -> FormulaFailure {
194        FormulaFailure::Objective {
195            error: zetesis_objective::AdmissionError::Allocation,
196            location: self.location,
197        }
198    }

200    fn scalar(&mut self, literal: &LiteralIr, binding: &Binding) -> Result<bool, FormulaFailure> {
201        match literal {
202            LiteralIr::Compare(comparison) => {
203                let ([left, right], relation) = comparison.operands();
204                let left = formula_support::expression(
205                    left,
206                    binding,
207                    self.computation,
208                    self.limits,
209                    self.counters,
210                    self.location,
211                )?;
212                let right = formula_support::expression(
213                    right,
214                    binding,
215                    self.computation,
216                    self.limits,
217                    self.counters,
218                    self.location,
219                )?;
220                let read = self.computation.read();
221                self.counters.work(self.limits, self.location)?;
222                let left = read.term(&left).map_err(|error| {
223                    crate::formula_binding::assignment(
224                        zetesis_core::catalog::AssignmentError::Read(error),
225                        self.location,
226                    )
227                })?;
228                self.counters.work(self.limits, self.location)?;
229                let right = read.term(&right).map_err(|error| {
230                    crate::formula_binding::assignment(
231                        zetesis_core::catalog::AssignmentError::Read(error),
232                        self.location,
233                    )
234                })?;
235                formula_support::compare(
236                    left,
237                    relation,
238                    right,
239                    self.limits,
240                    self.counters,
241                    self.location,
242                )
243            }
244            LiteralIr::Guard(guard) | LiteralIr::HeadGuard(guard) => guard.evaluate(
245                binding,
246                self.computation,
247                self.limits,
248                self.counters,
249                self.location,
250            ),
251            LiteralIr::Bind { .. } | LiteralIr::Range { .. } => Ok(true),
252            _ => Err(refusal(self.location)),
253        }
254    }

256    /// Reserve the coexisting closure, pending traversal and remaining set
257    /// before inserting any new predicate. Legacy certificate storage remains
258    /// included throughout this independent certificate's construction.
259    fn closure<'a>(
260        &mut self,
261        graph: &themelios_analysis::depend::DependencyGraph,
262        retained: usize,
263        conditions: impl IntoIterator<Item = &'a [LiteralIr]>,
264    ) -> Result<BTreeSet<Signature>, FormulaFailure> {
265        let mut relevant = BTreeSet::new();
266        let mut pending = Vec::new();
267        for condition in conditions {
268            self.source_roots(condition, retained, &mut relevant, &mut pending)?;
269        }
270        while let Some(predicate) = pending.pop() {
271            self.work()?;
272            for (_, dependency) in graph.edges_from(&predicate) {
273                self.work()?;
274                if !relevant.contains(dependency) {
275                    self.discover(dependency.clone(), retained, &mut relevant, &mut pending)?;
276                }
277            }
278        }
279        Ok(relevant)
280    }
281
282    fn discover(
283        &mut self,
284        predicate: Signature,
285        retained: usize,
286        relevant: &mut BTreeSet<Signature>,
287        pending: &mut Vec<Signature>,
288    ) -> Result<(), FormulaFailure> {
289        if relevant.contains(&predicate) {
290            return Ok(());
291        }
292        self.entries(retained.saturating_add(relevant.len().saturating_add(1).saturating_mul(3)))?;
293        pending.try_reserve(1).map_err(|_| self.allocation())?;
294        relevant.insert(predicate.clone());
295        pending.push(predicate);
296        Ok(())
297    }
298
299    fn entries(&self, count: usize) -> Result<(), FormulaFailure> {
300        ceiling(
301            FormulaResource::ObjectivePresenceEntries,
302            count as u128,
303            self.limits.max_objective_presence_entries as u128,
304            self.location,
305        )
306    }
307}

309impl SourceEligibility {
310    /// Discover all scoped atom dependencies and complete their shared activity.
311    /// This never adds the observing conditions to logical producers.
312    pub(crate) fn build_from_conditions<'a, 'source>(
313        prepared: &'source Prepared,
314        completed_support: &CompletedQueries<'source>,
315        retained: usize,
316        conditions: impl IntoIterator<Item = &'a [LiteralIr]>,
317        context: &mut Context<'_, '_, 'source>,
318    ) -> Result<Self, FormulaFailure> {
319        let support = completed_support.support();
320        let graph = prepared.analysis.dependencies();
321        let relevant = context.closure(graph, retained, conditions)?;
322        // The dependency sets coexist with the completed atom table. These
323        // logical planning slots are independent of allocator representation.
324        let temporary = retained.saturating_add(relevant.len().saturating_mul(3));
325        context.entries(temporary)?;
326        let mut remaining = relevant.clone();
327        let mut completed = BTreeSet::<Signature>::new();
328        let mut result = Self::new(context)?;
329        while !remaining.is_empty() {
330            let mut ready = None;
331            for predicate in &remaining {
332                context.work()?;
333                let mut available = true;
334                for (_, dependency) in graph.edges_from(predicate) {
335                    context.work()?;
336                    available &= !relevant.contains(dependency) || completed.contains(dependency);
337                }
338                if available {
339                    ready = Some(predicate.clone());
340                    break;
341                }
342            }
343            let Some(predicate) = ready else {
344                result.cyclic(prepared, support, &remaining, temporary, context)?;
345                break;
346            };
347            result.predicate(prepared, &predicate, support, temporary, context)?;
348            remaining.remove(&predicate);
349            completed.insert(predicate);
350        }
351        Ok(result)
352    }

354    fn predicate<'source>(
355        &mut self,
356        prepared: &'source Prepared,
357        predicate: &Signature,
358        support: &Support<'source>,
359        temporary: usize,
360        context: &mut Context<'_, '_, 'source>,
361    ) -> Result<(), FormulaFailure> {
362        self.round = Some(authority::Round::new(context));
363        let result = self
364            .derive_predicate(
365                prepared,
366                predicate,
367                support,
368                temporary.saturating_add(self.activity.len()),
369                context,
370            )
371            .and_then(|()| self.publish_predicate(context));
372        self.round = None;
373        result
374    }

376    fn derive_predicate<'source>(
377        &mut self,
378        prepared: &'source Prepared,
379        predicate: &Signature,
380        support: &Support<'source>,
381        temporary: usize,
382        context: &mut Context<'_, '_, 'source>,
383    ) -> Result<(), FormulaFailure> {
384        // A fact block is a precise producer: a normal head over an empty body.
385        let mut precise = true;
386        for rule in prepared.rules_without_facts() {
387            context.work()?;
388            if !context.produces(rule, predicate)? {
389                continue;
390            }
391            context.location = rule.location;
392            precise &= match &rule.head {
393                HeadIr::Choice(_) | HeadIr::Normal(_) => true,
394                HeadIr::Disjunction(heads) => {
395                    heads.iter().all(|head| head.positive_atom().is_some())
396                }
397                HeadIr::ConditionalDisjunction { .. } => false,
398            };
399        }
400        if precise {
401            for item in prepared.items().iter() {
402                context.work()?;
403                match item {
404                    Item::Rule(_, rule) => {
405                        if !context.produces(rule, predicate)? {
406                            continue;
407                        }
408                        self.rule(rule, predicate, support, temporary, context)?;
409                    }
410                    Item::Facts(block) => {
411                        context.location = block.location();
412                        if !context.produces_facts(block, predicate)? {
413                            continue;
414                        }
415                        self.facts(block, temporary, context)?;
416                    }
417                }
418            }
419        } else {
420            // Unsupported head classification retains completed Support as an
421            // upper carrier. Rich bodies of classified producers instead share
422            // the original scoped lowering and independent activity fold.
423            for candidate in support.predicates() {
424                context.work()?;
425                if context.matches_signature(candidate, predicate)? {
426                    self.possible_into_round(candidate, support, temporary, context)?;
427                    break;
428                }
429            }
430        }
431        Ok(())
432    }

434    fn rule<'source>(
435        &mut self,
436        rule: &'source RuleIr,
437        predicate: &Signature,
438        support: &Support<'source>,
439        temporary: usize,
440        context: &mut Context<'_, '_, 'source>,
441    ) -> Result<(), FormulaFailure> {
442        context.location = rule.location;
443        let mut outer = Join::rule(
444            Ir::Source(rule),
445            support,
446            context.computation,
447            context.limits,
448            context.budget,
449            context.counters,
450        )?;
451        while let Some(binding) = outer.next(
452            context.computation,
453            context.limits,
454            context.budget,
455            context.counters,
456            rule.location,
457        )? {
458            let body =
459                self.producer_activity(&rule.body, &rule.body_binding(&binding), support, context)?;
460            match &rule.head {
461                HeadIr::Normal(Some(head)) => {
462                    self.retain(*head, &binding, body, temporary, context)?;
463                }
464                HeadIr::Disjunction(heads) => {
465                    if heads.iter().any(|head| head.positive_atom().is_none()) {
466                        return Err(refusal(rule.location));
467                    }
468                    for head in heads.iter().filter_map(|head| head.positive_atom()) {
469                        if context.pattern_matches(*head, predicate)? {
470                            self.retain(
471                                *head,
472                                &binding,
473                                body.min(Activity::Optional),
474                                temporary,
475                                context,
476                            )?;
477                        }
478                    }
479                }
480                HeadIr::ConditionalDisjunction { .. } => {
481                    // Rich local eligibility uses completed support above;
482                    // model truth remains with the exact original query.
483                    return Err(refusal(rule.location));
484                }
485                HeadIr::Choice(group) => {
486                    for element in &group.elements {
487                        let Some(head) = element.head.positive_atom() else {
488                            continue;
489                        };
490                        if !context.pattern_matches(*head, predicate)? {
491                            continue;
492                        }
493                        let mut local = Join::element(
494                            Ir::Source(element),
495                            &binding,
496                            support,
497                            context.budget,
498                            crate::formula_support::Context::new(
499                                &*context.computation,
500                                context.limits,
501                                context.counters,
502                                rule.location,
503                            ),
504                        )?;
505                        while let Some(row) = local.next(
506                            context.computation,
507                            context.limits,
508                            context.budget,
509                            context.counters,
510                            rule.location,
511                        )? {
512                            let eligible = self.producer_activity(
513                                &element.condition,
514                                &element.body_binding(&row),
515                                support,
516                                context,
517                            )?;
518                            self.retain(
519                                *head,
520                                &row,
521                                body.min(eligible).min(Activity::Optional),
522                                temporary,
523                                context,
524                            )?;
525                        }
526                    }
527                }
528                HeadIr::Normal(None) => unreachable!("only producer heads selected"),
529            }
530        }
531        Ok(())
532    }

534    /// Retain each fact row with activity `Required`, which is what [`Self::rule`]
535    /// retains for a fact rule's one empty binding: the producer activity of an
536    /// empty body.
537    fn facts(
538        &mut self,
539        block: FactBlock<'_>,
540        temporary: usize,
541        context: &mut Context<'_, '_, '_>,
542    ) -> Result<(), FormulaFailure> {
543        let mut head = crate::formula_support::facts::FactHead::new(
544            block,
545            context.computation,
546            context.limits,
547            context.counters,
548            context.location,
549        )?;
550        for row in block.rows() {
551            context.location = row.location();
552            let atom = head.atom(row, context.computation, context.limits, context.counters)?;
553            self.stage(&atom, Activity::Required, temporary, context)?;
554        }
555        Ok(())
556    }

558    fn retain(
559        &mut self,
560        pattern: AtomPattern,
561        binding: &Binding,
562        activity: Activity,
563        temporary: usize,
564        context: &mut Context<'_, '_, '_>,
565    ) -> Result<(), FormulaFailure> {
566        // Validate the complete assigned atom even when its producer is absent.
567        // Identity admission changes neither activity nor support membership.
568        let atom = context.atom(pattern, binding)?;
569        if activity == Activity::Absent && !self.exact {
570            return Ok(());
571        }
572        self.stage(&atom, activity, temporary, context)
573    }

575    fn producer_activity<'source>(
576        &self,
577        literals: &[LiteralIr],
578        binding: &Binding,
579        support: &Support<'source>,
580        context: &mut Context<'_, '_, 'source>,
581    ) -> Result<Activity, FormulaFailure> {
582        if ordinary(literals) {
583            self.activity(literals, binding, context)
584        } else {
585            crate::formula_ground::source_activity(literals, binding, support, self, context)
586        }

589    /// Source truth coverage is evaluated without constructing or charging a
590    /// retained model query. Every literal is visited even after known absence.
591    pub(crate) fn activity(
592        &self,
593        literals: &[LiteralIr],
594        binding: &Binding,
595        context: &mut Context<'_, '_, '_>,
596    ) -> Result<Activity, FormulaFailure> {
597        self.fold_activity(literals, binding, false, context)
598    }
599
600    fn fold_activity(
601        &self,
602        literals: &[LiteralIr],
603        binding: &Binding,
604        selected: bool,
605        context: &mut Context<'_, '_, '_>,
606    ) -> Result<Activity, FormulaFailure> {
607        let mut result = Activity::Required;
608        for literal in literals {
609            context.work()?;
610            let activity = match literal {
611                LiteralIr::Atom(negation, pattern) => {
612                    let activity = self.pattern_activity(*pattern, binding, context)?;
613                    if *negation == DefaultNegation::Not {
614                        activity.negate()
615                    } else {
616                        activity
617                    }
618                }
619                LiteralIr::Aggregate(_)
620                | LiteralIr::Conditional(_)
621                | LiteralIr::ProjectedAtom(..) => Activity::Optional,
622                LiteralIr::PatternAtom(pattern) => {
623                    self.pattern_activity(pattern.atom, binding, context)?
624                }
625                LiteralIr::Compare(..) if selected => Activity::Required,
626                // The complete Join already checked these ordinary data filters.
627                LiteralIr::ArgumentCheck { .. } | LiteralIr::TupleCompare(..) => Activity::Required,
628                _ => {
629                    if context.scalar(literal, binding)? {
630                        Activity::Required
631                    } else {
632                        Activity::Absent
633                    }
634                }
635            };
636            result = result.min(activity);
637        }
638        Ok(result)
639    }
