1//! Exact consequence scan under a complete signed acyclic source certificate.
2//! The activity cells and canonical source identities are the same ones used
3//! by source eligibility; only this stronger certificate permits exact output.
4
5use super::{Activity, Context, SourceEligibility};
6use crate::formula_facts::{FactBlock, Item};

9use crate::grounding_observer::Profile;

12pub(crate) struct Exact {
13    pub(crate) atoms: SourceSelection,
14    pub(crate) selected: Vec<usize>,
15    pub(crate) consistent: bool,
16    pub(crate) lease: StorageLease,
17}

19impl SourceEligibility {
20    // The private AcyclicPlan is the sole caller which enables this mode.
21    // Direct publication is safe because a producer reads only strictly earlier
22    // predicates. Alternative producers still combine through the same max.
23    pub(super) fn stage_exact(
24        &mut self,
25        atom: &SourceAtom,
26        activity: Activity,
27        context: &mut Context<'_, '_, '_>,
28    ) -> Result<(), FormulaFailure> {
29        let (id, _) = self.atoms.insert(
30            atom,
31            (FormulaResource::Atoms, context.limits.theory.max_atoms),
32            context.computation,
33            context.limits,
34            context.counters,
35            context.location,
36        )?;
37        let additional = self.atoms.len().saturating_sub(self.activity.len());
38        crate::formula_support::reserve(
39            &mut self.activity,
40            additional,
41            &mut self.lease,
42            Self::header_bytes(),
43            crate::formula_support::Context::new(
44                &*context.computation,
45                context.limits,
46                context.counters,
47                context.location,
48            ),
49        )?;
50        context
51            .counters
52            .charge_work(additional as u128 + 1, context.limits, context.location)?;
53        self.activity.resize(self.atoms.len(), Activity::Absent);
54        self.activity[id] = self.activity[id].max(activity);
55        Ok(())
56    }
57}

59pub(crate) fn evaluate_acyclic(
60    plan: &AcyclicPlan,
61    source: &crate::formula_ir::Prepared,
62    support: &crate::formula_support::Support<'_>,
63    profile: &Profile<'_>,
64    context: &mut Context<'_, '_, '_>,
65) -> Result<Exact, FormulaFailure> {
66    let mut result = SourceEligibility::new(context)?;
67    result.exact = true;
68    let mut consistent = true;
69    for &(_, position) in plan.items() {
70        match source.items().get(position) {
71            Item::Rule(_, rule) => {
72                context.location = rule.location;
73                consistent &= profile.phase(
74                    GroundingPhase::RuleInstantiation,
75                    Some(rule.location),
76                    || result.rule_exact(rule, support, context),
77                )?;
78            }
79            Item::Facts(block) => {
80                context.location = block.location();
81                profile.phase(
82                    GroundingPhase::RuleInstantiation,
83                    Some(block.location()),
84                    || result.facts_exact(block, context),
85                )?;
86            }
87        }
88    }
89    consistent &= profile.phase(GroundingPhase::Coherence, None, || result.coherent(context))?;
90    result.into_exact(consistent, context)
91}

93impl SourceEligibility {
94    /// Stage each fact row `Required`: a body-free rule's one passing binding
95    /// has that activity, since the fold over no literal starts from it.
96    fn facts_exact(
97        &mut self,
98        block: FactBlock<'_>,
99        context: &mut Context<'_, '_, '_>,
100    ) -> Result<(), FormulaFailure> {
101        let mut head = crate::formula_support::facts::FactHead::new(
102            block,
103            context.computation,
104            context.limits,
105            context.counters,
106            context.location,
107        )?;
108        for row in block.rows() {
109            context.location = row.location();
110            let atom = head.atom(row, context.computation, context.limits, context.counters)?;
111            self.stage_exact(&atom, Activity::Required, context)?;
112        }
113        Ok(())
114    }
115
116    fn rule_exact(
117        &mut self,
118        rule: &crate::formula_ir::RuleIr,
119        support: &crate::formula_support::Support<'_>,
120        context: &mut Context<'_, '_, '_>,
121    ) -> Result<bool, FormulaFailure> {
122        let mut consistent = true;
123        let mut join = Join::rule(
124            rule,
125            support,
126            context.computation,
127            context.limits,
128            context.budget,
129            context.counters,
130        )?;
131        // This is the maintained complete-column selection, after independent
132        // original-family admission. Rejected rows still validate their keys.
133        join.select_total_body(rule, context.computation, context.limits, context.counters)?;
134        while let Some(row) = join.next_row(
135            context.computation,
136            context.limits,
137            context.budget,
138            context.counters,
139            rule.location,
140        )? {
141            if !row.passes {
142                crate::formula_ground::validate_atom_keys(
143                    &rule.body,
144                    &row.values,
145                    context.computation,
146                    context.limits,
147                    context.counters,
148                    rule.location,
149                    |variable, location| FormulaFailure::UnsafeVariable { variable, location },
150                )?;
151                continue;
152            }
153            let body = self.fold_activity(&rule.body, &row.values, true, context)?;
154            match &rule.head {
155                HeadIr::Normal(Some(head)) => {
156                    self.retain(*head, &row.values, body, 0, context)?;
157                }
158                HeadIr::Normal(None) => consistent &= body != Activity::Required,
159                _ => unreachable!("complete normal source certificate"),
160            }
161        }
162        Ok(consistent)
163    }
164
165    fn coherent(&self, context: &mut Context<'_, '_, '_>) -> Result<bool, FormulaFailure> {
166        let mut consistent = true;
167        for (id, activity) in self.activity.iter().enumerate() {
168            context.work()?;
169            if *activity != Activity::Required {
170                continue;
171            }
172            let atom = self
173                .atoms
174                .source(id, context.limits, context.counters, context.location)?;
175            let value = context.computation.source_atom(
176                &atom,
177                context.limits,
178                context.counters,
179                context.location,
180            )?;
181            let opposite = match value.predicate().sign() {
182                zetesis_core::Sign::Positive => zetesis_core::Sign::Negative,
183                zetesis_core::Sign::Negative => zetesis_core::Sign::Positive,
184            };
185            if let Some(other) = context.computation.signed(
186                &atom,
187                opposite,
188                context.limits,
189                context.counters,
190                context.location,
191            )? {
192                consistent &= self.lookup(&other, context)? != Activity::Required;
193            }
194        }
195        Ok(consistent)
196    }
197
198    fn into_exact(
199        self,
200        consistent: bool,
201        context: &mut Context<'_, '_, '_>,
202    ) -> Result<Exact, FormulaFailure> {
203        let mut selected = Vec::new();
204        let mut lease = context.computation.lease();
205        let header = size_of::<Exact>() - size_of::<SourceSelection>();
206        lease.observe(header, context.location)?;
207        for (id, activity) in self.activity.iter().enumerate() {
208            context.work()?;
209            if *activity != Activity::Required {
210                continue;
211            }
212            crate::formula_support::reserve(
213                &mut selected,
214                1,
215                &mut lease,
216                header,
217                crate::formula_support::Context::new(
218                    &*context.computation,
219                    context.limits,
220                    context.counters,
221                    context.location,
222                ),
223            )?;
224            selected.push(id);
225        }
226        Ok(Exact {
227            atoms: self.atoms,
228            selected,
229            consistent,
230            lease,
231        })
232    }
