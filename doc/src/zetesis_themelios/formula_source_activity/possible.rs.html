1//! Completed possible relation rows supply identities, without copying payload.
2//! Membership covers possible truth and never establishes realizability.
3
4use super::{Activity, Context, SourceEligibility, authority::Round};
5use crate::FormulaFailure;
6use crate::formula_support::Support;
7
8impl SourceEligibility {
9    pub(super) fn possible(
10        &mut self,
11        predicate: zetesis_core::catalog::PredicateRef<'_>,
12        support: &Support<'_>,
13        temporary: usize,
14        context: &mut Context<'_, '_, '_>,
15    ) -> Result<(), FormulaFailure> {
16        self.round = Some(Round::new(context));
17        let result = self
18            .possible_into_round(
19                predicate,
20                support,
21                temporary.saturating_add(self.activity.len()),
22                context,
23            )
24            .and_then(|()| self.publish_predicate(context));
25        self.round = None;
26        result
27    }
28
29    pub(super) fn possible_into_round(
30        &mut self,
31        predicate: zetesis_core::catalog::PredicateRef<'_>,
32        support: &Support<'_>,
33        temporary: usize,
34        context: &mut Context<'_, '_, '_>,
35    ) -> Result<(), FormulaFailure> {
36        context.work()?;
37        for row in support.rows(predicate) {
38            context.work()?;
39            self.retain_atom(row.atom(), Activity::Optional, temporary, context)?;
40        }
41        Ok(())
42    }
43}
