163/// descendant serves only vocabulary-scoped term lookups.

166    /// a hybrid base's.

173impl CompletedCatalog {
174    /// Close this support, keeping only the relations of the predicates the
175    /// atoms of `rules` name. The close is charged and its peak admitted as any
176    /// close is.
177    pub(crate) fn into_streamed(
178        self,
179        rules: &[crate::formula_ir::RuleIr],
180        work: GroundingWork<'_>,
181    ) -> Result<StreamedSupport, FormulaFailure> {
182        let GroundingWork {
183            limits,
184            counters,
185            location,
186        } = work;
187        let keep = match self.catalog.component_view(limits, counters, location)? {
188            Some(view) => demand::read_predicates(rules, view, limits, counters, location)?,
189            None => None,
190        };
191        let max_bytes = limits.max_atom_storage_bytes;
192        let (closed, rows) = self
193            .catalog
194            .into_streamed(
195                keep.as_ref(),
196                GroundingWork::new(limits, counters, location),
197            )
198            .map_err(|failure| failure.into_parts().0)?;
199        let lookup = zetesis_core::atom_interner::AtomInterner::for_closed_catalog(
200            &closed.storage,
201            max_bytes,
202        )
203        .map_err(|error| FormulaFailure::AtomCatalog { error, location })?;
204        Ok(StreamedSupport {
205            closed: std::sync::Arc::new(closed),
206            rows,
207            lookup,
208            completion: self.completion,
209        })
210    }
211}

498/// Admit `amount` units before their effect: observe at a charged-range boundary
499/// or before a work refusal, check the ceiling, record the receipt, then count.
500/// Zero admits no units and never observes; use `Counters::poll` explicitly. Every charged unit runs
501/// on this path, so it inlines into its callers; the site is read only when a
502/// failure is built, on the cold path.
503#[inline]
504fn charge_work(
505    work: &mut u64,
506    control: Option<&accounting::AccountControl>,
507    allowance: Option<&mut crate::constraint_allowance::Pending>,
508    amount: u128,
509    limits: &FormulaLimits,
510    location: ProgramSite,
511) -> Result<(), FormulaFailure> {
512    check_work(*work, control, amount, limits, location)?;
513    count_work(
514        work,
515        allowance,
516        u64::try_from(amount).expect("charged work fits its u64 ceiling"),
517    );
518    Ok(())
519}

521/// Check the charged range and ceiling. A refused charge observes first; an
522/// admitted range observes only when it contains a cadence boundary.
523#[inline]
524fn check_work(
525    work: u64,
526    control: Option<&accounting::AccountControl>,
527    amount: u128,
528    limits: &FormulaLimits,
529    location: ProgramSite,
530) -> Result<(), FormulaFailure> {
531    let admission = zetesis_algebra::Meter::check_receipt(work, amount, limits.max_work);
532    if let Some(control) = control {
533        let observation = if admission.is_err() {
534            control.observe()
535        } else {
536            control.observe_charge(work, u64::try_from(amount).expect("admitted work fits u64"))
537        };
538        observation.map_err(|reason| interrupted(reason, location))?;
539    }
540    admission
541        .map(|_| ())
542        .map_err(|error| work_failure(error, limits.max_work, location))
543}

545/// Preserve the source account's saturated attempted-demand diagnostic.
546fn work_failure(
547    error: zetesis_algebra::MeterWorkError,
548    limit: u64,
549    location: ProgramSite,
550) -> FormulaFailure {
551    let (observed, limit) = match error {
552        zetesis_algebra::MeterWorkError::Limit { observed, limit } => (observed, limit),
553        zetesis_algebra::MeterWorkError::Overflow => (u128::MAX, u128::from(limit)),
554    };
555    FormulaFailure::Limit {
556        resource: FormulaResource::Work,
557        observed,
558        limit,
559        location,
560    }
561}

595        complete(

606/// Every round joins each selected rule under its domain guards, when the
607/// analysis prepared candidates: a row a guard rejects has no complete
608/// continuation the exclusion rule keeps, in this round as in the final one.
609fn complete(
610    mut catalog: SupportCatalog,
611    prepared: &Prepared,
612    mut producers: producers::Producers<'_>,
613    domains: Option<&crate::formula_domains::Domains<'_>>,
614    budget: &mut Budget,
615    work: GroundingWork<'_>,
616) -> Result<CompletedCatalog, FormulaFailure> {
617    let GroundingWork {
618        limits,
619        counters,
620        location: fallback,
621    } = work;
622    catalog.prepared_bytes(producers.bytes());
623    // Postings are kept only for columns a join can bind, decided before the
624    // first relation is created, since a column cannot gain postings later.
625    if let Some(view) = catalog.component_view(limits, counters, fallback)? {
626        let demand = demand::Demand::of(prepared, view, limits, counters, fallback)?;
627        catalog.install_demand(demand);
628    }
629    catalog.publish(limits, counters, fallback)?;
630    #[cfg(test)]
631    postings::begin_support();
632    let mut rounds = 0_u64;
633    // Rows are only appended, so each round extends the kept certificates.
634    let mut retained = queries::Retained::default();
635    loop {
636        ceiling(
637            FormulaResource::SupportRounds,
638            u128::from(rounds) + 1,
639            u128::from(limits.max_support_rounds),
640            fallback,
641        )?;
642        rounds += 1;
643        counters.record(Event::SupportRound);
644        let changed = {
645            // Rules alone are scheduled: a fact block has no producer slot,
646            // and enters support in the first round at its item position.
647            let mut schedule = producers.schedule();
648            let mut facts = (rounds == 1 && prepared.items().blocks().next().is_some())
649                .then(|| facts::FirstRound::new(prepared.items()));
650            let mut selected = schedule.next(limits, counters, fallback)?;
651            if selected.is_none() && producers.is_planned() && facts.is_none() {
652                // Every zero-input producer ran at bootstrap. Every subsequent
653                // new positive binding needs a changed predicate, whose complete
654                // posting would have selected its original producer. Thus this
655                // admitted round is unchanged without a new relation snapshot.
656                return finish(
657                    (catalog, producers, retained),
658                    GroundingWork::new(limits, counters, fallback),
659                );
660            }
661            counters.work(limits, fallback)?;
662            counters.record(Event::SupportSnapshotPreparation);
663            let (relations, mut delta) = catalog.split(limits, counters, fallback)?;
664            let kept = std::mem::take(&mut retained);
665            let support = Support::indexed_retaining(&relations, kept, limits, counters, fallback)?;
666            counters.observe_work(Event::SupportProductionWork, |counters| {
667                while let Some((index, deferred)) = next_producer(
668                    selected,
669                    &mut facts,
670                    (&producers, &mut delta, &support),
671                    (limits, counters, fallback),
672                )? {
673                    if let Some(facts) = &mut facts {
674                        facts.before_rule(index, &mut delta, &support, limits, counters)?;
675                    }
676                    let rule = &prepared.rules_without_facts()[index];
677                    counters.work(limits, rule.location)?;
678                    counters.record(Event::SupportProducerVisit);
679                    let mut variants = if deferred {
680                        counters.record(Event::SupportRichTraversal);
681                        delta::traversal(rule, &support, true)
682                    } else {
683                        producers.variants(index, rule, &support, rounds == 1, limits, counters)?
684                    };
685                    let round = (&mut delta, &support, domains);
686                    produce(
687                        (index, rule),
688                        &mut variants,
689                        round,
690                        budget,
691                        (limits, counters),
692                    )?;
693                    if !deferred {
694                        selected = schedule.next(limits, counters, fallback)?;
695                    }
696                }
697                Ok::<_, FormulaFailure>(())
698            })?;
699            // No join remains live. Release its query descriptor before the
700            // pending selection borrows the catalog's semantic ordering index.
701            retained = support.into_retained(fallback)?;
702            advance_round(
703                &mut delta,
704                producers.plan_mut(),
705                GroundingWork::new(limits, counters, fallback),
706            )?
707        };
708        if !changed {
709            // An unchanged round decides every component no deferred producer
710            // still feeds, so the earliest waiting producer was traversed.
711            debug_assert!(
712                !producers.pending(),
713                "a deferred producer was not traversed"
714            );
715            return finish(
716                (catalog, producers, retained),
717                GroundingWork::new(limits, counters, fallback),
718            );
719        }
720        catalog.publish(limits, counters, fallback)?;
721    }

724/// The round's next producer: the schedule's, then, once it is exhausted and
725/// the facts are in, each deferred rich producer whose read components reached
726/// their fixed point, flagged so its caller traverses it in full.
727fn next_producer<'source>(
728    selected: Option<usize>,
729    facts: &mut Option<facts::FirstRound<'_>>,
730    (producers, delta, support): (
731        &producers::Producers<'_>,
732        &mut SupportAppend<'source>,
733        &Support<'source>,
734    ),
735    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
736) -> Result<Option<(usize, bool)>, FormulaFailure> {
737    if let Some(index) = selected {
738        return Ok(Some((index, false)));
739    }
740    if let Some(facts) = facts.take() {
741        facts.finish(delta, support, limits, counters)?;
742    }
743    let ready = match producers.take_ready() {
744        Some(index) => Some(index),
745        None if producers.mark_ready(delta, limits, counters, location)? => producers.take_ready(),
746        None => None,
747    };
748    Ok(ready.map(|index| (index, true)))
749}

751/// Derive one producer's variants into this round's pending atoms, under the
752/// rule's domain guards and static totality when the analysis supplies them.
753fn produce<'source>(
754    (index, rule): (usize, &'source crate::formula_ir::RuleIr),
755    variants: &mut delta::Variants<'_, '_>,
756    (delta, support, domains): (
757        &mut SupportAppend<'source>,
758        &Support<'source>,
759        Option<&crate::formula_domains::Domains<'_>>,
760    ),
761    budget: &mut Budget,
762    (limits, counters): (&FormulaLimits, &mut Counters),
763) -> Result<(), FormulaFailure> {
764    let mut computation = Computation::new(delta, support);
765    let candidates = domains
766        .map(|domains| domains.for_rule(index, rule))
767        .transpose()?
768        .flatten();
769    let totality = domains
770        .map(|domains| domains.total_body(index, rule))
771        .transpose()?
772        .flatten();
773    let guards = match candidates {
774        Some(candidates) => {
775            support.domain_guards(rule, candidates, &computation, limits, counters)?
776        }
777        None => None,
778    };
779    derive_variants(
780        rule,
781        variants,
782        (guards.as_ref(), totality),
783        support,
784        budget,
785        Context::new(&mut computation, limits, counters, rule.location),
786    )
787}

789/// Prepare the next wake set only after the complete pending population has its
790/// publication order. Refusal leaves private round metadata, never completion.
791fn advance_round(
792    delta: &mut SupportAppend<'_>,
793    plan: Option<&mut producers::ProducerPlan<'_>>,
794    work: GroundingWork<'_>,
795) -> Result<bool, FormulaFailure> {
796    let GroundingWork {
797        limits,
798        counters,
799        location,
800    } = work;
801    if delta.is_empty() {
802        return Ok(false);
803    }
804    counters.observe_work(Event::SupportOrderWork, |counters| {
805        delta.order(limits, counters, location)
806    })?;
807    #[cfg(test)]
808    facts::rounds::record(delta.atoms());
809    if let Some(plan) = plan {
810        counters.observe_work(Event::SupportWakeWork, |counters| {
811            plan.advance(delta.atoms(), limits, counters, location)
812        })?;
813    }
814    Ok(true)
815}

842/// Visit one rule's ordered delta variants under the same admitted guards.
843fn derive_variants<'source>(
844    rule: &'source crate::formula_ir::RuleIr,
845    variants: &mut delta::Variants<'_, '_>,
846    (guards, totality): (
847        Option<&queries::Guards<'_, 'source>>,
848        Option<&crate::formula_domains::numeric_ir::TotalBody<'_>>,
849    ),
850    support: &Support<'source>,
851    budget: &mut Budget,
852    context: Context<'_, &mut Computation<'_, '_>>,
853) -> Result<(), FormulaFailure> {
854    let Context {
855        computation,
856        work:
857            GroundingWork {
858                limits,
859                counters,
860                location: _,
861            },
862    } = context;
863    // A successful map belongs to this exact rule and immutable round. The
864    // first cursor retains its ordinary mutable preparation; later variants
865    // borrow it only after that cursor has finished and released its state.
866    let mut preparation = None;
867    let mut next = variants.next(limits, counters)?;
868    while let Some(variant) = next {
869        if let Some(preparation) = &preparation {
870            let mut outer = multiples::RoundPreparation::rows(
871                preparation,
872                variant,
873                guards,
874                budget,
875                Context::new(computation, limits, counters, rule.location),
876            )?;
877            outer.select_round_multiples(rule, computation, limits, counters)?;
878            if let Some(totality) = totality {
879                outer.select_static_total_body(rule, totality, limits, counters)?;
880            }
881            derive_rule(
882                rule,
883                &mut outer,
884                support,
885                computation,
886                limits,
887                budget,
888                counters,
889            )?;
890            drop(outer);
891            next = variants.next(limits, counters)?;
892            continue;
893        }
894        let mut outer = Join::variant_rule(
895            Ir::Source(rule),
896            variant,
897            support,
898            guards,
899            budget,
900            Context::new(computation, limits, counters, rule.location),
901        )?;
902        let total = outer.select_round_multiples(rule, computation, limits, counters)?;
903        if let Some(totality) = totality {
904            outer.select_static_total_body(rule, totality, limits, counters)?;
905        }
906        derive_rule(
907            rule,
908            &mut outer,
909            support,
910            computation,
911            limits,
912            budget,
913            counters,
914        )?;
915        // Preserve the next-variant lookup before admitting any new owner. A
916        // final or only successful cursor has no reuse consumer.
917        next = variants.next(limits, counters)?;
918        if total && next.is_some() {
919            preparation = Some(multiples::RoundPreparation::retain(
920                Ir::Source(rule),
921                outer,
922                Context::new(computation, limits, counters, rule.location),
923            )?);
924        }
925    }
926    Ok(())
927}

929fn derive_rule<'source>(
930    rule: &'source crate::formula_ir::RuleIr,
931    outer: &mut Join<'_, 'source>,
932    support: &Support<'source>,
933    computation: &mut Computation<'_, '_>,
934    limits: &FormulaLimits,
935    budget: &mut Budget,
936    counters: &mut Counters,
937) -> Result<(), FormulaFailure> {
938    // Every cursor belongs to this immutable round, including each first-new
939    // variant. Family-evidence and external-check cursors never enter here.
940    outer.select_scalar_body(rule, computation, limits, counters)?;
941    outer.select_numeric_probes(rule, limits, counters)?;
942    if witnesses::derive(rule, outer, computation, limits, budget, counters)? {
943        return Ok(());
944    }
945    let mut derivation = Derivation {
946        normal_head: None,
947        computation,
948        limits,
949        budget,
950        counters,
951    };
952    let projected = match &rule.head {
953        HeadIr::Normal(head) => *head,
954        _ => None,
955    };
956    while let Some(binding) = derivation.next(outer, projected, rule.location)? {
957        match &rule.head {
958            HeadIr::Normal(Some(head)) => {
959                derivation.normal_head(*head, &binding, rule.location)?;
960            }
961            HeadIr::Disjunction(heads) => {
962                // Only positive occurrences can produce possible atoms.
963                // Neither default-negation mode supplies support.
964                for head in heads.iter().filter_map(|head| head.positive_atom()) {
965                    derivation.head(*head, &binding, rule.location)?;
966                }
967            }
968            HeadIr::ConditionalDisjunction { ordinary, elements } => {
969                for head in ordinary.iter().filter_map(|head| head.positive_atom()) {
970                    derivation.head(*head, &binding, rule.location)?;
971                }
972                for element in elements {
973                    let mut local = Join::local_head(
974                        Ir::Source(element.condition.as_slice()),
975                        &binding.prefix(element.outer_variables),
976                        element.body_variables..element.variables,
977                        support,
978                        derivation.budget,
979                        Context::new(
980                            derivation.computation,
981                            limits,
982                            derivation.counters,
983                            rule.location,
984                        ),
985                    )?;
986                    while let Some(binding) = derivation.next(&mut local, None, rule.location)? {
987                        if let Some(head) = element.head.positive_atom() {
988                            derivation.head(*head, &binding, rule.location)?;
989                        }
990                    }
991                }
992            }
993            HeadIr::Choice(group) => {
994                derivation.choice(group, &binding, support, rule.location)?;
995            }
996            HeadIr::Normal(None) => unreachable!("constraints never produce support"),
997        }
998    }
999    Ok(())
1000}

1002/// A support round publishes newly derived heads after complete checked binding.
1003/// Existing support and the current delta share one borrowed atom identity.
1004struct Derivation<'a, 'source, 'round> {
1005    normal_head: Option<heads::Head<'round>>,
1006    computation: &'a mut Computation<'source, 'round>,
1007    limits: &'a FormulaLimits,
1008    budget: &'a mut Budget,
1009    counters: &'a mut Counters,
1010}
1011
1012impl Derivation<'_, '_, '_> {
1013    /// The one normal head is prepared only after a complete selected binding.
1014    /// It borrows this round's immutable template, never a generated row.
1015    fn normal_head(
1016        &mut self,
1017        pattern: AtomPattern,
1018        binding: &Binding<'_>,
1019        location: ProgramSite,
1020    ) -> Result<(), FormulaFailure> {
1021        self.counters
1022            .observe_work(Event::SupportHeadWork, |counters| {
1023                if self.normal_head.is_none() {
1024                    self.normal_head = Some(heads::Head::new(
1025                        pattern,
1026                        self.computation,
1027                        self.limits,
1028                        counters,
1029                        location,
1030                    )?);
1031                }
1032                self.normal_head
1033                    .as_ref()
1034                    .expect("prepared normal head")
1035                    .derive(binding, self.computation, self.limits, counters, location)
1036            })
1037    }
1038
1039    /// Select a complete support binding before the consumer borrows its frame.
1040    /// Attribution observes admitted operations; it adds no work charges.
1041    fn next<'join>(
1042        &mut self,
1043        join: &'join mut Join<'_, '_>,
1044        projected: Option<AtomPattern>,
1045        location: ProgramSite,
1046    ) -> Result<Option<Binding<'join>>, FormulaFailure> {
1047        self.counters
1048            .observe_work(Event::SupportJoinWork, |counters| {
1049                join.next_support(
1050                    projected,
1051                    self.computation,
1052                    self.limits,
1053                    self.budget,
1054                    counters,
1055                    location,
1056                )
1057            })
1058    }
1059
1060    fn choice<'source>(
1061        &mut self,
1062        group: &'source crate::formula_ir::ChoiceIr,
1063        binding: &Binding,
1064        support: &Support<'source>,
1065        location: ProgramSite,
1066    ) -> Result<(), FormulaFailure> {
1067        crate::formula_head_aggregate::validate_group(
1068            Ir::Source(group),
1069            binding,
1070            support,
1071            self.budget,
1072            Context::new(self.computation, self.limits, self.counters, location),
1073        )?;
1074        for element in &group.elements {
1075            let mut local = Join::element(
1076                Ir::Source(element),
1077                binding,
1078                support,
1079                self.budget,
1080                Context::new(self.computation, self.limits, self.counters, location),
1081            )?;
1082            while let Some(binding) = self.next(&mut local, None, location)? {
1083                if let Some(head) = element.head.positive_atom() {
1084                    self.head(*head, &binding, location)?;
1085                }
1086            }
1087        }
1088        Ok(())
1089    }
1090
1091    fn head(
1092        &mut self,
1093        pattern: AtomPattern,
1094        binding: &Binding,
1095        location: ProgramSite,
1096    ) -> Result<(), FormulaFailure> {
1097        self.counters
1098            .observe_work(Event::SupportHeadWork, |counters| {
1099                let pattern =
1100                    self.computation
1101                        .static_pattern(pattern, self.limits, counters, location)?;
1102                let atom =
1103                    self.computation
1104                        .atom(pattern, binding, self.limits, counters, location)?;
1105                self.computation
1106                    .support(&atom, self.limits, counters, location)
1107            })
1108    }
1109}

1303        &self,
1304        position: &mut usize,
1305        limits: &FormulaLimits,
1306        counters: &mut Counters,
1307        location: ProgramSite,
1308    ) -> Result<Option<usize>, FormulaFailure> {
1309        let row = match self {
1310            Self::Indexed(rows) => {
1311                let row = rows.get(*position);
1312                if row.is_some() {
1313                    *position += 1;
1314                }
1315                row
1316            }
1317            Self::Owned(rows) => {
1318                let row = rows.slice().get(*position).copied();
1319                if row.is_some() {
1320                    *position += 1;
1321                }
1322                row
1323            }
1324            Self::Range(rows) => {
1325                counters.work(limits, location)?;
1326                let row = rows.slice().get(*position).copied();
1327                if row.is_some() {
1328                    *position += 1;
1329                    counters.record(Event::RangeProbeRow);
1330                }
1331                row
1332            }
1333            Self::Expression(rows) => {
1334                counters.work(limits, location)?;
1335                let row = rows.get(*position);
1336                if row.is_some() {
1337                    *position += 1;
1338                    counters.record(Event::ExpressionKeyRow);
1339                }
1340                row
1341            }
1342            Self::Table(rows) => {
1343                let row = rows.next(*position, limits, counters, location)?;
1344                if let Some(row) = row {
1345                    *position = row + 1;
1346                    counters.record(Event::TableRow);
1347                }
1348                row
1349            }
1350        };
1351        Ok(row)
1352    }

1902    fn fix_component(
1903        &mut self,
1904        fixed: &Binding<'_>,
1905        used: &[usize],
1906        expected: Option<&[bool]>,
1907        context: Context<'_, &Computation<'_, '_>>,
1908    ) -> Result<(), FormulaFailure> {
1909        let Context {
1910            computation: _,
1911            work,
1912        } = context;
1913        let GroundingWork {
1914            limits,
1915            counters,
1916            location,
1917        } = work;
1918        for slot in 0..fixed.len().min(self.values.len()) {
1919            counters.work(limits, location)?;
1920            let value = fixed
1921                .slots()
1922                .key(slot)
1923                .map_err(|error| crate::formula_binding::assignment(error, location))?;
1924            if expected.is_some_and(|expected| expected[slot] != value.is_some()) {
1925                return Err(FormulaFailure::SupportRelation {
1926                    error: zetesis_core::relation::Failure::Owner,
1927                    location,
1928                });
1929            }
1930            if let Some(value) = value {
1931                self.values.set(slot, &value, limits, counters, location)?;
1932            }
1933        }
1934        let mut used_cursor = 0;
1935        for (variable, slot) in self.slots.iter_mut().enumerate() {
1936            counters.work(limits, location)?;
1937            while used.get(used_cursor).is_some_and(|&used| used < variable) {
1938                counters.work(limits, location)?;
1939                used_cursor += 1;
1940            }
1941            if used.get(used_cursor) != Some(&variable) {
1942                *slot = Slot::Excluded;
1943            }
1944        }
1945        Ok(())
1946    }

2829    fn next_base(
2830        &mut self,
2831        ownership: Ownership,
2832        projected: Option<AtomPattern>,
2833        budget: &mut Budget,
2834        context: Context<'_, &mut Computation<'_, '_>>,
2835    ) -> Result<Option<Frame>, FormulaFailure> {
2836        let Context {
2837            computation,
2838            work:
2839                GroundingWork {
2840                    limits,
2841                    counters,
2842                    location,
2843                },
2844        } = context;
2845        self.prepare_scalar_comparisons(computation, limits, budget, counters, location)?;
2846        self.resume(limits, counters, location)?;
2847        self.comparisons = Comparisons::Deferred;
2848        if self.traversal == Traversal::Finished {
2849            return Ok(None);
2850        }
2851        loop {
2852            counters.work(limits, location)?;
2853            if self.skip_derived(projected, computation, limits, counters, location)? {
2854                if self.traversal == Traversal::Finished {
2855                    return Ok(None);
2856                }
2857                continue;
2858            }
2859            if self.plan.patterns.is_empty() {
2860                if self.traversal == Traversal::EmptyVisited {
2861                    self.traversal = Traversal::Finished;
2862                    return Ok(None);
2863                }
2864                self.traversal = Traversal::EmptyVisited;
2865                if !self.filter_prefix(computation, limits, counters, location)? {
2866                    continue;
2867                }
2868                self.comparisons = self.certificate();
2869                return self
2870                    .complete(ownership, computation, limits, counters, location)
2871                    .map(Some);
2872            }
2873            if self.depth == self.plan.patterns.len() {
2874                self.comparisons = self.certificate();
2875                return self
2876                    .complete(ownership, computation, limits, counters, location)
2877                    .map(Some);
2878            }
2879            let pattern = self.plan.patterns[self.depth];
2880            let row = self.advance_probe(
2881                pattern,
2882                Context::new(computation, limits, counters, location),
2883            )?;
2884            let atom = row.and_then(|row| self.resolutions[self.depth].rows()?.row(row));
2885            let Some(atom) = atom else {
2886                self.positions[self.depth] = 0;
2887                self.probes[self.depth] = None;
2888                if self.depth == 0 {
2889                    self.traversal = Traversal::Finished;
2890                    return Ok(None);
2891                }
2892                self.undo_at(self.depth - 1, limits, counters, location)?;
2893                self.depth -= 1;
2894                continue;
2895            };

2897            let selected = self.permits_row(pattern, atom, limits, counters, location)?;
2898            let permitted = selected.permits()
2899                && self
2900                    .positive_prefix
2901                    .as_ref()
2902                    .is_none_or(|prefix| prefix.permits(selected));
2903            if let Some(selection) = self.probe_selections.get_mut(self.depth) {
2904                selection.record(self.positions[self.depth] - 1, permitted);
2905            }
2906            if !permitted {
2907                continue;
2908            }
2909            let matches = self.match_row(
2910                pattern.pattern,
2911                atom,
2912                budget,
2913                Context::new(computation, limits, counters, location),
2914            )?;
2915            if matches && self.filter_prefix(computation, limits, counters, location)? {
2916                if let Some(witnesses) = &mut self.witnesses {
2917                    counters.work(limits, location)?;
2918                    witnesses.push(self.depth, atom);
2919                }
2920                // Pattern matching and prefix scalars must succeed before a
2921                // mapped row's identity can describe this completed binding.
2922                if let Some(prefix) = &mut self.positive_prefix {
2923                    prefix.advance(self.depth, pattern.source, selected);
2924                }
2925                self.depth += 1;
2926            } else {
2927                self.undo_at(self.depth, limits, counters, location)?;
2928            }
2929        }
2930    }
2931    /// Authenticate this depth's relation and filter as one immutable pair.
2932    /// A refused authentication publishes neither half; backtracking reuses a
2933    /// successfully resolved pair independently of the chosen probe strategy.
2934    fn resolve_probe_source(
2935        &mut self,
2936        pattern: PatternOccurrence<'a>,
2937        work: GroundingWork<'_>,
2938    ) -> Result<(), FormulaFailure> {
2939        let GroundingWork {
2940            limits,
2941            counters,
2942            location,
2943        } = work;
2944        // Relation ownership is independent of how the row source was prepared.
2945        // A preselected probe still needs this snapshot's owner before row access.
2946        if matches!(self.resolutions[self.depth], Resolution::Unresolved) {
2947            let rows = self
2948                .support
2949                .resolve(pattern.atom(), limits, counters, location)?;
2950            let filter_source = match (self.row_filter, rows) {
2951                (Some(filter), Some(source)) if source.row_count() != 0 => {
2952                    Some(filter.resolve(source.atoms, limits, counters, location)?)
2953                }
2954                _ => None,
2955            };

2957            // refused authentication. Backtracking keeps this immutable pair.
2958            self.resolutions[self.depth] = Resolution::Resolved {
2959                rows,
2960                filter_source,
2961            };
2962        }
2963        Ok(())

3074    fn indexed_probe(
3075        &self,
3076        posting: Option<&'source [usize]>,
3077        source: Option<&'source relations::RelationRows<'source>>,
3078        occurrence: usize,
3079        limits: &FormulaLimits,
3080        counters: &mut Counters,
3081        location: ProgramSite,
3082    ) -> Result<Probe<'a, 'source>, FormulaFailure> {
3083        let selected = self
3084            .pattern_rows
3085            .map(|pattern_rows| {
3086                pattern_rows.posting(self.depth, source, limits, counters, location)
3087            })
3088            .transpose()?
3089            .flatten();
3090        // Both postings are necessary conditions over the same ordered source.
3091        // Keep the smaller one; the ordinary matcher checks their conjunction.
3092        let posting = match (posting, selected) {
3093            (Some(indexed), Some(selected)) if selected.len() < indexed.len() => Some(selected),
3094            (None, Some(selected)) => Some(selected),
3095            (indexed, _) => indexed,
3096        };
3097        let total = source.map_or(0, relations::RelationRows::row_count);
3098        Ok(Probe::Indexed(if self.delta.is_some() {
3099            let old = source.map_or(0, relations::RelationRows::old_rows);
3100            let range = delta::interval(self.delta, occurrence, old, total);
3101            delta::Rows::within(posting, range, limits, counters, location)?
3102        } else {
3103            delta::Rows::all(posting, total)
3104        }))

3245    /// Evaluate the comparisons the current depth decides, once, and fold
3246    /// them into the prefix's verdict. A comparison that is defined and false
3247    /// excludes every substitution of the prefix, so the prefix is pruned and
3248    /// nothing beneath it is reached; an evaluation failure is retained, and
3249    /// becomes a refusal only if no comparison of the complete substitution
3250    /// excludes it.
3251    fn filter_prefix(
3252        &mut self,
3253        computation: &mut Computation<'_, '_>,
3254        limits: &FormulaLimits,
3255        counters: &mut Counters,
3256        location: ProgramSite,
3257    ) -> Result<bool, FormulaFailure> {
3258        if self.coverage == Coverage::Complete {
3259            return Ok(true);
3260        }
3261        if !self.scalar_selection.permits(
3262            self.literals.get(),
3263            &self.values,
3264            Context::new(&*computation, limits, counters, location),
3265        )? {
3266            self.verdicts[self.depth] = false;
3267            return Ok(false);
3268        }
3269        let depth = self.depth;
3270        let mut passes = depth == 0 || self.verdicts[depth - 1];
3271        for index in self.plan.decisions.decided_at(depth) {
3272            let compared = if let Some(operands) = self
3273                .scalar_selection
3274                .comparisons()
3275                .and_then(|view| view.operands(index))
3276            {
3277                // The selected operands have their own DAG. Original unary
3278                // projection descriptors name generated slots and cannot serve it.
3279                self.evaluation.source_comparison(
3280                    operands,
3281                    &self.values,
3282                    Context::new(computation, limits, counters, location),
3283                )
3284            } else {
3285                let comparison =
3286                    comparison(&self.literals[index]).expect("a decided literal is a comparison");
3287                let decided = integer_comparisons::decide(
3288                    comparison,
3289                    &self.values,
3290                    Context::new(&*computation, limits, counters, location),
3291                )?;
3292                let ([left, right], relation) = comparison.operands();
3293                if let Some(holds) = decided {
3294                    Ok(holds)
3295                } else if self.certified_total {
3296                    self.projections.compare(
3297                        index,
3298                        ([left, right], relation),
3299                        &self.values,
3300                        &mut self.evaluation,
3301                        Context::new(computation, limits, counters, location),
3302                    )
3303                } else {
3304                    self.evaluation.source_comparison(
3305                        ([left, right], relation),
3306                        &self.values,
3307                        Context::new(computation, limits, counters, location),
3308                    )
3309                }
3310            };
3311            match compared {
3312                Ok(false) if self.certified_total => {
3313                    // Complete source domains establish that later arithmetic
3314                    // cannot fail. False now excludes every extension without
3315                    // concealing a source diagnostic.
3316                    self.verdicts[depth] = false;
3317                    return Ok(false);
3318                }
3319                Ok(value) => passes &= value,
3320                Err(FormulaFailure::Expansion(ExpansionFailure::Evaluation { error, .. })) => {
3321                    // Retained, not raised: a comparison decided here or
3322                    // deeper may still exclude the substitution.
3323                    if self.failure.is_none() {
3324                        self.failure = Some((depth, error));
3325                    }
3326                }
3327                Err(error) => return Err(error),
3328            }
3329        }
3330        self.verdicts[depth] = passes;
3331        Ok(passes)
3332    }

3402        let Some(head) = projected else {

3414    /// Materialization, when requested, remains before complete-row filters.
3415    /// Its failure leaves the same unfinished depth and counters as before.
3416    fn complete(
3417        &mut self,
3418        ownership: Ownership,
3419        computation: &mut Computation<'_, '_>,
3420        limits: &FormulaLimits,
3421        counters: &mut Counters,
3422        location: ProgramSite,
3423    ) -> Result<Frame, FormulaFailure> {
3424        counters.substitution(limits, location)?;
3425        for (variable, slot) in self.slots.iter().enumerate() {
3426            if !self.values.is_bound(variable, location)? && matches!(slot, Slot::Relational) {
3427                return Err(FormulaFailure::UnsafeVariable { variable, location });
3428            }
3429        }
3430        let frame = match ownership {
3431            Ownership::Own => {
3432                let values = self
3433                    .values
3434                    .copied(computation, limits, counters, location)?;
3435                counters.record(Event::BindingSnapshot);
3436                Frame::Owned(values)
3437            }
3438            Ownership::Lend => {
3439                // The same admitted slot span is inspected for a lent frame.
3440                // No scalar payload is copied, and no snapshot is recorded.
3441                for _ in 0..self.values.len() {
3442                    counters.work(limits, location)?;
3443                }
3444                Frame::Current
3445            }
3446        };
3447        if !self.plan.patterns.is_empty() {
3448            self.traversal = Traversal::PendingUndo;
3449        }
3450        if matches!(ownership, Ownership::Own) {
3451            self.resume(limits, counters, location)?;
3452        }
3453        Ok(frame)
3454    }

3456    /// Exactly one undo follows a successfully completed nonempty join. Owned
3457    /// snapshots resume immediately; a lent row resumes on its next advance.
3458    fn resume(
3459        &mut self,
3460        limits: &FormulaLimits,
3461        counters: &mut Counters,
3462        location: ProgramSite,
3463    ) -> Result<(), FormulaFailure> {
3464        if self.traversal == Traversal::PendingUndo {
3465            self.undo_at(self.depth - 1, limits, counters, location)?;
3466            self.depth -= 1;
3467            self.traversal = Traversal::Searching;
3468        }
3469        Ok(())
3470    }
