10impl Admission<'_> {
11    pub(crate) fn predicate(
12        &mut self,
13        predicate: PredicateRef<'_>,
14        limits: &FormulaLimits,
15        counters: &mut Counters,
16        location: Location,
17    ) -> Result<Predicate, FormulaFailure> {
18        let outer = self.total(location)? as u128 - self.catalog.owner.storage_bytes();
19        let checked = owner_limits(limits, outer, location)?;
20        self.catalog.owner.restart_storage_peak();
21        let result = self
22            .catalog
23            .owner
24            .declare_predicate_with(predicate, checked, || counters.work(limits, location));
25        let observed = self.owner_observed(outer, limits, counters, location);
26        let declaration = result.map_err(|error| atom_failure(error, limits, outer, location))?;
27        observed?;
28        self.append(
29            limits,
30            counters,
31            location,
32            |components, read, allowance, counters| {
33                components
34                    .append_predicate_with(read, &declaration, allowance, || {
35                        counters.work(limits, location)
36                    })
37                    .map(Predicate)
38            },
39        )
40    }

42    pub(crate) fn pattern(
43        &mut self,
44        predicate: Predicate,
45        arguments: &[Term],
46        limits: &FormulaLimits,
47        counters: &mut Counters,
48        location: Location,
49    ) -> Result<Pattern, FormulaFailure> {
50        let mut terms = Vec::new();
51        let mut lease = self.lease();
52        let header = size_of::<Vec<TemplateTerm<'_>>>() + size_of::<StorageLease>();
53        lease.observe(header, location)?;
54        self.storage_observed(&lease, 0, header, limits, counters, location)?;
55        let required =
56            header as u128 + arguments.len() as u128 * size_of::<TemplateTerm<'_>>() as u128;
57        let outer_scratch = (self.total(location)? - lease.bytes()) as u128;
58        ceiling(
59            FormulaResource::SupportBytes,
60            outer_scratch + required,
61            limits.max_support_bytes as u128,
62            location,
63        )?;
64        counters.work(limits, location)?;
65        let reserved = terms.try_reserve_exact(arguments.len());
66        lease.observe(
67            header + terms.capacity() * size_of::<TemplateTerm<'_>>(),
68            location,
69        )?;
70        let observed = self.storage_observed(&lease, header, header, limits, counters, location);
71        reserved.map_err(|error| FormulaFailure::AtomAllocation { error, location })?;
72        observed?;
73
74        let components = self
75            .catalog
76            .components
77            .as_ref()
78            .ok_or_else(|| missing(location))?;
79        let outer = self.total(location)? as u128 - components.storage_bytes();
80        let allowance = remaining_u128(outer, limits, location)?;
81        let read = self.catalog.owner.read();
82        let bound = components
83            .bind_with(read, || counters.work(limits, location))
84            .map_err(|error| failure(error, limits, outer, location))?;
85        let predicate = predicate.get(bound, limits, counters, location)?;
86        counters.work(limits, location)?;
87        let declaration = read
88            .declare_existing(predicate)
89            .map_err(|error| read_failure(error, location))?;
90        counters.work(limits, location)?;
91        let predicate = read
92            .predicate(&declaration)
93            .map_err(|error| read_failure(error, location))?;
94        for argument in arguments {
95            counters.work(limits, location)?;
96            let value = detached(*argument, bound, read, limits, counters, location)?;
97            counters.work(limits, location)?;
98            terms.push(value);
99        }
100        let pattern = PatternRef::from_parts(predicate, &terms)
101            .map_err(|error| crate::AdmissionFailure::Construction { error, location })?;
102        let components = self
103            .catalog
104            .components
105            .as_mut()
106            .ok_or_else(|| missing(location))?;
107        components.restart_storage_peak();
108        let result = components
109            .append_pattern_with(read, pattern, allowance, || counters.work(limits, location));
110        let peak = outer + components.storage_peak_bytes();
111        counters.record(Event::SupportPeakBytes(peak));
112        let observed = ceiling(
113            FormulaResource::SupportBytes,
114            peak,
115            limits.max_support_bytes as u128,
116            location,
117        );
118        let position = result.map_err(|error| failure(error, limits, outer, location))?;
119        observed?;
120        Ok(Pattern(position))
121    }

123    pub(crate) fn filter(
124        &mut self,
125        equal: bool,
126        left: Term,
127        right: Term,
128        limits: &FormulaLimits,
129        counters: &mut Counters,
130        location: Location,
131    ) -> Result<Filter, FormulaFailure> {
132        self.append(
133            limits,
134            counters,
135            location,
136            |components, read, allowance, counters| {
137                let bound = components.bind_with(read, || counters.work(limits, location))?;
138                let left = detached(left, bound, read, limits, counters, location)
139                    .map_err(TemplateCatalogFailure::Stopped)?;
140                let right = detached(right, bound, read, limits, counters, location)
141                    .map_err(TemplateCatalogFailure::Stopped)?;
142                let filter = FilterRef::from_parts(equal, left, right);
143                components
144                    .append_filter_with(read, filter, allowance, || counters.work(limits, location))
145                    .map(Filter)
146            },
147        )
148    }

166// Detach the short component-metadata borrow without copying payload. Each
167// transient key is immediately resolved through the independently borrowed
168// canonical reader and dropped; argument scratch retains no per-cell witness.
169fn detached<'read>(
170    term: Term,
171    components: TemplateComponentsRef<'_>,
172    read: CatalogRead<'read>,
173    limits: &FormulaLimits,
174    counters: &mut Counters,
175    location: Location,
176) -> Result<TemplateTerm<'read>, FormulaFailure> {
177    match term.get(components, limits, counters, location)? {
178        TemplateTerm::Variable(variable) => Ok(TemplateTerm::Variable(variable)),
179        TemplateTerm::Constant(value) => {
180            counters.work(limits, location)?;
181            let key = read
182                .term_key(value)
183                .map_err(|error| read_failure(error, location))?;
184            counters.work(limits, location)?;
185            read.term(&key)
186                .map(TemplateTerm::Constant)
187                .map_err(|error| read_failure(error, location))
188        }
189    }
190}

192impl Term {
193    pub(crate) fn get<'a>(
194        self,
195        view: TemplateComponentsRef<'a>,
196        limits: &FormulaLimits,
197        counters: &mut Counters,
198        location: Location,
199    ) -> Result<TemplateTerm<'a>, FormulaFailure> {
200        match self {
201            Self::Variable(variable) => {
202                counters.work(limits, location)?;
203                Ok(TemplateTerm::Variable(variable))
204            }
205            Self::Constant(scalar) => scalar
206                .get(view, limits, counters, location)
207                .map(TemplateTerm::Constant),
208        }
209    }
210}
