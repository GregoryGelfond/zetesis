1//! The columns of each support predicate that a join can bind.
2//!
3//! Support postings serve two readers: a probe chooses the shortest posting
4//! among the columns its pattern binds, and totality reads a column's keys as
5//! its distinct values. Both read only columns a join can bind, so a column is
6//! demanded iff some join occurrence of its predicate holds there a constant,
7//! or a variable a join can have bound: one that occurs again in the same rule,
8//! in its body or its head, since factorization binds head variables before it
9//! probes the body. A predicate no join occurrence names demands nothing.
10//!
11//! Rules with an ordinary head and a flat body of atoms, comparisons and
12//! bindings are read exactly; there a negated atom demands nothing, since it
13//! is decided by exact lookup once its variables are bound. Choice and
14//! conditional heads, projections and every nested frame (aggregate elements,
15//! choice and conditional conditions, objectives, witnesses) demand every
16//! column holding a constant or a variable: a superset, since a column indexed
17//! without need only costs memory, while one left unindexed only costs the
18//! probe its posting.
19

32/// Demanded columns by signed predicate, computed once before support grows.
33#[derive(Debug, Default)]
34pub(crate) struct Demand {
35    columns: HashMap<Predicate, Vec<bool>>,
36}
37
38impl Demand {
39    /// Named capacity: one entry per predicate a join names, with its owned
40    /// signature and column flags. Hash-table control bytes are excluded.
41    pub(crate) fn retained_bytes(&self) -> usize {
42        self.columns.capacity() * size_of::<(Predicate, Vec<bool>)>()
43            + self
44                .columns
45                .iter()
46                .map(|(predicate, columns)| predicate.name().len() + columns.capacity())
47                .sum::<usize>()
48    }
49
50    /// The demanded columns of `predicate`; `None` when no join names it.
51    pub(crate) fn columns(
52        &self,
53        predicate: PredicateRef<'_>,
54        limits: &FormulaLimits,
55        counters: &mut Counters,
56        location: ProgramSite,
57    ) -> Result<Option<&[bool]>, FormulaFailure> {
58        counters.charge_work(predicate.name().len() as u128 + 1, limits, location)?;
59        Ok(self.columns.get(&owned(predicate)).map(Vec::as_slice))
60    }
61
62    /// Walk every rule, projection and objective of `prepared` once.
63    pub(crate) fn of(
64        prepared: &Prepared,
65        view: TemplateComponentsRef<'_>,
66        limits: &FormulaLimits,
67        counters: &mut Counters,
68        location: ProgramSite,
69    ) -> Result<Self, FormulaFailure> {
70        let mut walk = Walk {
71            view,
72            limits,
73            counters,
74            location,
75            found: HashMap::new(),
76        };
77        // Demand comes from body literals, and a fact has none.
78        for rule in prepared
79            .rules_without_facts()
80            .iter()
81            .chain(&prepared.projection)
82        {
83            walk.rule(rule)?;
84        }
85        for objective in &prepared.objectives {
86            for &atom in &objective.positive {
87                walk.every_column(atom)?;
88            }
89            occurrences::frame(objective.condition.literals(), &mut walk)?;
90        }
91        Ok(Self {
92            columns: walk.found,
93        })
94    }
95}

97struct Walk<'a, 'c> {
98    view: TemplateComponentsRef<'a>,
99    limits: &'c FormulaLimits,
100    counters: &'c mut Counters,
101    location: ProgramSite,
102    found: HashMap<Predicate, Vec<bool>>,
103}
104
105impl Walk<'_, '_> {
106    fn work(&mut self) -> Result<(), FormulaFailure> {
107        self.counters.work(self.limits, self.location)
108    }
109
110    fn rule(&mut self, rule: &RuleIr) -> Result<(), FormulaFailure> {
111        match &rule.head {
112            // The body and every element condition, as nested frames.
113            HeadIr::Choice(_) | HeadIr::ConditionalDisjunction { .. } => {
114                occurrences::rule(rule, self)?;
115            }
116            HeadIr::Normal(head) => {
117                let heads: Vec<_> = head.iter().copied().collect();
118                self.body(&rule.body, &heads)?;
119            }
120            HeadIr::Disjunction(heads) => {
121                let heads: Vec<_> = heads
122                    .iter()
123                    .filter_map(|head| head.atom().copied())
124                    .collect();
125                self.body(&rule.body, &heads)?;
126            }
127        }
128        Ok(())
129    }
130
131    /// A rule's body under an ordinary head: exact when flat, else as a
132    /// nested frame. Head occurrences count, since factorization binds them.
133    fn body(
134        &mut self,
135        literals: &[LiteralIr],
136        heads: &[super::components::Pattern],
137    ) -> Result<(), FormulaFailure> {
138        if !literals.iter().all(flat) {
139            return occurrences::frame(literals, self);
140        }
141        let mut occurrences: BTreeMap<usize, usize> = BTreeMap::new();
142        for &head in heads {
143            self.work()?;
144            for variable in self.pattern_variables(head)? {
145                *occurrences.entry(variable).or_default() += 1;
146            }
147        }
148        for literal in literals {
149            self.work()?;
150            for variable in self.variables(literal)? {
151                *occurrences.entry(variable).or_default() += 1;
152            }
153        }
154        // A negated atom is decided by exact lookup once safety has bound its
155        // variables, never by a posting probe: it demands nothing, though its
156        // variables count as occurrences above.
157        for literal in literals {
158            self.work()?;
159            if let LiteralIr::Atom(DefaultNegation::None, atom) = literal {
160                let pattern = atom.get(self.view, self.limits, self.counters, self.location)?;
161                let terms = pattern.terms();
162                let demanded = (0..terms.len())
163                    .map(|column| match terms.at(column) {
164                        Some(TemplateTerm::Constant(_)) => true,
165                        Some(TemplateTerm::Variable(variable)) => {
166                            occurrences.get(&variable).is_some_and(|&count| count > 1)
167                        }
168                        None => false,
169                    })
170                    .collect();
171                self.record(pattern.predicate(), demanded);
172            }
173        }
174        Ok(())
175    }
176
177    /// Demand every column of an occurrence that holds a constant or variable.
178    fn every_column(&mut self, atom: super::components::Pattern) -> Result<(), FormulaFailure> {
179        let pattern = atom.get(self.view, self.limits, self.counters, self.location)?;
180        let demanded = vec![true; pattern.terms().len()];
181        self.record(pattern.predicate(), demanded);
182        Ok(())
183    }
184
185    fn record(&mut self, predicate: PredicateRef<'_>, demanded: Vec<bool>) {
186        let known = self
187            .found
188            .entry(owned(predicate))
189            .or_insert_with(|| vec![false; demanded.len()]);
190        for (known, demanded) in known.iter_mut().zip(demanded) {
191            *known |= demanded;
192        }
193    }
194
195    fn pattern_variables(
196        &mut self,
197        atom: super::components::Pattern,
198    ) -> Result<Vec<usize>, FormulaFailure> {
199        let pattern = atom.get(self.view, self.limits, self.counters, self.location)?;
200        let terms = pattern.terms();
201        Ok((0..terms.len())
202            .filter_map(|column| match terms.at(column) {
203                Some(TemplateTerm::Variable(variable)) => Some(variable),
204                _ => None,
205            })
206            .collect())
207    }
208
209    /// The variables a flat literal reads or binds, once per occurrence.
210    fn variables(&mut self, literal: &LiteralIr) -> Result<Vec<usize>, FormulaFailure> {
211        Ok(match literal {
212            LiteralIr::Atom(_, atom) => self.pattern_variables(*atom)?,
213            LiteralIr::Compare(comparison) | LiteralIr::ArgumentCheck(comparison) => comparison
214                .left()
215                .inputs()
216                .chain(comparison.right().inputs())
217                .collect(),
218            LiteralIr::TupleCompare(left, _, right) => left
219                .iter()
220                .chain(right)
221                .flat_map(crate::formula_ir::Expression::inputs)
222                .collect(),
223            LiteralIr::Bind { target, value } => {
224                std::iter::once(*target).chain(value.inputs()).collect()
225            }
226            LiteralIr::Range {
227                target,
228                lower,
229                upper,
230                ..
231            } => std::iter::once(*target)
232                .chain(lower.inputs())
233                .chain(upper.inputs())
234                .collect(),
235            _ => unreachable!("only flat literals are counted"),
236        })
237    }
238}

240/// A nested frame is read conservatively: every column of an occurrence that
241/// holds a constant or a variable, and every column a projection names.
242impl occurrences::Visitor for Walk<'_, '_> {
243    fn work(&mut self) -> Result<(), FormulaFailure> {
244        Walk::work(self)
245    }
246
247    fn occurrence(&mut self, occurrence: Occurrence) -> Result<(), FormulaFailure> {
248        match occurrence {
249            Occurrence::Atom(atom) => self.every_column(atom),
250            Occurrence::Projected(predicate) => {
251                let predicate =
252                    predicate.get(self.view, self.limits, self.counters, self.location)?;
253                self.record(predicate, vec![true; predicate.arity()]);
254                Ok(())
255            }
256        }
257    }
258}

260/// A literal whose variables the exact reading counts.
261fn flat(literal: &LiteralIr) -> bool {
262    matches!(
263        literal,
264        LiteralIr::Atom(..)
265            | LiteralIr::Compare(..)
266            | LiteralIr::ArgumentCheck { .. }
267            | LiteralIr::TupleCompare(..)
268            | LiteralIr::Bind { .. }
269            | LiteralIr::Range { .. }
270    )
271}

273/// The predicates the atoms of `rules` name, any polarity; `None` when a rule
274/// holds a literal this walk does not read, so that every relation is kept.
275pub(crate) fn read_predicates<'r>(
276    rules: impl IntoIterator<Item = &'r RuleIr>,
277    view: TemplateComponentsRef<'_>,
278    limits: &FormulaLimits,
279    counters: &mut Counters,
280    location: ProgramSite,
281) -> Result<Option<std::collections::HashSet<Predicate>>, FormulaFailure> {
282    let mut found = std::collections::HashSet::new();
283    for rule in rules {
284        for literal in &rule.body {
285            counters.work(limits, location)?;
286            let atom = match literal {
287                LiteralIr::Atom(_, atom) => *atom,
288                LiteralIr::PatternAtom(atom) => atom.atom,
289                LiteralIr::Compare(..)
290                | LiteralIr::ArgumentCheck { .. }
291                | LiteralIr::TupleCompare(..)
292                | LiteralIr::Bind { .. }
293                | LiteralIr::Range { .. } => continue,
294                _ => return Ok(None),
295            };
296            let pattern = atom.get(view, limits, counters, location)?;
297            found.insert(owned(pattern.predicate()));
298        }
299    }
300    Ok(Some(found))
301}

303/// The owned key of a signed predicate.
304fn owned(predicate: PredicateRef<'_>) -> Predicate {
305    Predicate::with_sign(predicate.name(), predicate.arity(), predicate.sign())
306        .expect("an admitted predicate is a valid signature")
