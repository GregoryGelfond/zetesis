1//! Necessary integer equality/range postings for explicitly selected traversals.
2//!
3//! The query certificate covers every row of one source occurrence, not a
4//! meet of argument domains or a guessed integer type. The opposite operand
5//! must evaluate to an integer in the current prefix. A skipped row therefore
6//! either cannot match the occurrence or makes a plan-decided comparison defined
7//! and false. An ordering key is one column; equality may use a checked
8//! expression of columns. The ordinary residual comparison remains in the join.
9//!
10//! Flat normal source bodies opt in during eager emission or possible-head
11//! production. Explicit aggregate consumers additionally admit flat positive
12//! local conditions with covered bare-variable comparisons. Joins marked as
13//! original source evidence, hybrid capture/checks and direct-source exact
14//! activity never opt in. Scalar assignments may opt in after a whole-body
15//! totality certificate over that same immutable snapshot; their selected DAG
16//! operands retain original literal identities. Structural and nested scopes
17//! retain their old traversal unless the explicit local certificate applies.
18//! Every round query uses the occurrence's first-new
19//! interval against the one immutable snapshot index.
20

36impl<'a, 'source> Join<'a, 'source> {
37    /// Called only by eager instantiation or possible-head production.
38    /// No totality premise is borrowed from original-family admission: a
39    /// defined false plan-decided comparison excludes that row under the
40    /// ordinary filters contract, including independent deferred scalar faults.
41    /// Generated scalar bindings require an independent whole-body totality
42    /// certificate before their comparison view can select any source row.
43    /// Structural and nested scopes retain their existing traversal.
44    pub(crate) fn select_numeric_probes(
45        &mut self,
46        rule: &RuleIr,
47        limits: &crate::FormulaLimits,
48        counters: &mut super::Counters,
49    ) -> Result<(), FormulaFailure> {
50        counters.work(limits, rule.location)?;
51        if !std::ptr::eq(self.literals.get(), rule.body.as_slice())
52            || !matches!(rule.head, HeadIr::Normal(_))
53            || (rule.variables != rule.body_variables && !self.certified_total)
54        {
55            return Ok(());
56        }
57        let mut comparison_found = false;
58        for literal in &rule.body {
59            counters.work(limits, rule.location)?;
60            match literal {
61                LiteralIr::Compare(comparison) => {
62                    comparison_found |= selected_relation(comparison.relation());
63                }
64                LiteralIr::Atom(..) | LiteralIr::TupleCompare(..) => {}
65                LiteralIr::Bind { .. } if self.certified_total => {}
66                _ => return Ok(()),
67            }
68        }
69        self.expression_keys = comparison_found.then_some(Eligible);
70        Ok(())
71    }

73    /// Select numeric postings for one explicitly consumed positive local join.
74    /// Every scalar operand is a bare variable supplied by this prefix or a
75    /// positive flat occurrence. Thus a defined false selected comparison can
76    /// remove only rows that would fail the same residual comparison; there is
77    /// no skipped local arithmetic, generator or nested-family diagnostic.
78    /// The existing probe separately certifies its complete integer column and
79    /// current bound, retaining source order and all ordinary alias checks.
80    /// Call once before traversal. This changes neither the plan nor the
81    /// source-family/coverage authority.
82    pub(crate) fn select_numeric_local(
83        &mut self,
84        mut context: Context<'_, &Computation<'_, '_>>,
85    ) -> Result<(), FormulaFailure> {
86        if self.source_evidence {
87            return Ok(());
88        }
89        let mut comparison_found = false;
90        for literal in self.literals.get() {
91            tick(&mut context)?;
92            match literal {
93                LiteralIr::Atom(themelios_program::program::DefaultNegation::None, _) => {}
94                LiteralIr::Compare(comparison) => {
95                    let ([left, right], relation) = comparison.operands();
96                    let ([Operation::Variable(left)], [Operation::Variable(right)]) =
97                        (left.nodes.as_slice(), right.nodes.as_slice())
98                    else {
99                        return Ok(());
100                    };
101                    if !self.local_numeric_input(*left, &mut context)?
102                        || !self.local_numeric_input(*right, &mut context)?
103                    {
104                        return Ok(());
105                    }
106                    comparison_found |= selected_relation(relation);
107                }
108                _ => return Ok(()),
109            }
110        }
111        self.expression_keys = comparison_found.then_some(Eligible);
112        Ok(())
113    }

115    fn local_numeric_input(
116        &self,
117        slot: usize,
118        context: &mut Context<'_, &Computation<'_, '_>>,
119    ) -> Result<bool, FormulaFailure> {
120        tick(context)?;
121        if slot >= self.values.len() {
122            return Ok(false);
123        }
124        if self.values.is_bound(slot, context.work.location)? {
125            return Ok(true);
126        }
127        for occurrence in &self.plan.patterns {
128            tick(context)?;
129            let PositivePattern::Flat(pattern) = occurrence.pattern else {
130                return Ok(false);
131            };
132            if column(pattern, slot, context)?.is_some() {
133                return Ok(true);
134            }
135        }
136        Ok(false)
137    }

139    pub(super) fn expression_key_probe(
140        &self,
141        pattern: PatternOccurrence<'a>,
142        source: Option<&'source RelationRows<'source>>,
143        mut context: Context<'_, &Computation<'_, '_>>,
144    ) -> Result<Option<Probe<'a, 'source>>, FormulaFailure> {
145        if self.expression_keys.is_none() || self.source_evidence {
146            return Ok(None);
147        }
148        #[cfg(test)]
149        if tests::disabled() {
150            return Ok(None);
151        }
152        let (PositivePattern::Flat(flat), Some(source)) = (pattern.pattern, source) else {
153            return Ok(None);
154        };
155        if source.row_count() == 0 {
156            return Ok(None);
157        }
158        // Preserve expression-equality selection ahead of range selection:
159        // a diagonal equality must not be displaced by an earlier R<S filter.
160        for equality in [true, false] {
161            for index in self.plan.decisions.decided_at(self.depth) {
162                tick(&mut context)?;
163                let Some(([left, right], relation)) = self
164                    .scalar_selection
165                    .comparisons()
166                    .and_then(|view| view.operands(index))
167                    .or_else(|| {
168                        super::comparison(&self.literals[index])
169                            .map(crate::formula_ir::Comparison::operands)
170                    })
171                else {
172                    continue;
173                };
174                if !selected_relation(relation) || (relation == Relation::Eq) != equality {
175                    continue;
176                }
177                for (key, bound, relation) in
178                    [(left, right, relation), (right, left, reversed(relation))]
179                {
180                    if let Some(probe) = self.numeric_candidate(
181                        ([key, bound], relation),
182                        flat,
183                        source,
184                        pattern.source,
185                        &mut context,
186                    )? {
187                        return Ok(Some(probe));
188                    }
189                }
190            }
191        }
192        Ok(None)
193    }

195    fn numeric_candidate(
196        &self,
197        comparison: ([&Expression; 2], Relation),
198        pattern: zetesis_core::PatternRef<'_>,
199        source: &'source RelationRows<'source>,
200        occurrence: usize,
201        context: &mut Context<'_, &Computation<'_, '_>>,
202    ) -> Result<Option<Probe<'a, 'source>>, FormulaFailure> {
203        let ([key, bound], relation) = comparison;
204        // Stage 2a indexes one column, while 2b allows a checked expression.
205        if relation != Relation::Eq && !matches!(key.nodes.as_slice(), [Operation::Variable(_)]) {
206            return Ok(None);
207        }
208        if !self.key_available(key, bound, pattern, context)? {
209            return Ok(None);
210        }
211        let binding = self.values.view(
212            context.computation.read(),
213            context.work.limits,
214            context.work.counters,
215            context.work.location,
216        )?;
217        // The ordinary equality posting remains the first choice, including
218        // its existing delta clipping and residual alias/constant checks.
219        if let Some(posting) = self.support.probe_at(
220            Some(source),
221            pattern,
222            binding,
223            context.work.limits,
224            context.work.counters,
225            context.work.location,
226        )? {
227            return self
228                .indexed_probe(
229                    Some(posting),
230                    Some(source),
231                    occurrence,
232                    context.work.limits,
233                    context.work.counters,
234                    context.work.location,
235                )
236                .map(Some);
237        }
238        let mut scratch = buffer(context)?;
239        let wanted = evaluate(&bound.nodes, &mut scratch, context, |slot, context| {
240            tick(context)?;
241            let value =
242                self.values
243                    .read(slot, context.computation.read(), context.work.location)?;
244            Ok(number(value.descriptor()))
245        })?;
246        let Some(wanted) = wanted else {
247            return Ok(None);
248        };
249        if relation != Relation::Eq {
250            let [Operation::Variable(variable)] = key.nodes.as_slice() else {
251                unreachable!("range keys contain exactly one variable");
252            };
253            if let Some(probe) =
254                self.multiples_range_probe(source, *variable, (relation, wanted), context)?
255            {
256                return Ok(Some(probe));
257            }
258        }
259        let range = delta::interval(
260            self.delta,
261            occurrence,
262            source.old_rows(),
263            source.row_count(),
264        );
265        let query = Context::new(
266            context.computation,
267            context.work.limits,
268            context.work.counters,
269            context.work.location,
270        );
271        if relation == Relation::Eq {
272            self.support
273                .expression_key(source, key, pattern, wanted, range, query)
274                .map(|rows| rows.map(Probe::Expression))
275        } else {
276            self.support
277                .range_key(source, key, pattern, (relation, wanted), range, query)
278                .map(|rows| rows.map(Probe::Range))
279        }
280    }

282    fn key_available(
283        &self,
284        key: &Expression,
285        bound: &Expression,
286        pattern: zetesis_core::PatternRef<'_>,
287        context: &mut Context<'_, &Computation<'_, '_>>,
288    ) -> Result<bool, FormulaFailure> {
289        let mut unbound = false;
290        for operation in &key.nodes {
291            tick(context)?;
292            match operation {
293                Operation::Variable(slot) => {
294                    if column(pattern, *slot, context)?.is_none() {
295                        return Ok(false);
296                    }
297                    unbound |= !self.values.is_bound(*slot, context.work.location)?;
298                }
299                Operation::Constructor(_) => return Ok(false),
300                _ => {}
301            }
302        }
303        if !unbound {
304            return Ok(false);
305        }
306        for operation in &bound.nodes {
307            tick(context)?;
308            match operation {
309                Operation::Variable(slot)
310                    if !self.values.is_bound(*slot, context.work.location)? =>
311                {
312                    return Ok(false);
313                }
314                Operation::Constructor(_) => return Ok(false),
315                _ => {}
316            }
317        }
318        Ok(true)
319    }
320}

322fn selected_relation(relation: Relation) -> bool {
323    matches!(
324        relation,
325        Relation::Eq | Relation::Lt | Relation::Le | Relation::Gt | Relation::Ge
326    )
327}
328
329fn reversed(relation: Relation) -> Relation {
330    match relation {
331        Relation::Lt => Relation::Gt,
332        Relation::Le => Relation::Ge,
333        Relation::Gt => Relation::Lt,
334        Relation::Ge => Relation::Le,
335        relation => relation,
336    }
