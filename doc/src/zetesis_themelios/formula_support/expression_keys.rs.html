1//! Necessary integer equality/range postings for explicitly selected traversals.
2//!
3//! The query certificate covers every row of one source occurrence, not a
4//! meet of argument domains or a guessed integer type. The opposite operand
5//! must evaluate to an integer in the current prefix. A skipped row therefore
6//! either cannot match the occurrence or makes a plan-decided comparison defined
7//! and false. An ordering key is one column; equality may use a checked
8//! expression of columns. The ordinary residual comparison remains in the join.
9//!
10//! Flat normal source bodies opt in during eager emission or possible-head
11//! production. Family evidence, hybrid capture/checks and direct-source exact
12//! activity never opt in. Generated bindings/heads and structural or nested
13//! scopes retain their old traversal. Every round query uses the occurrence's
14//! first-new interval against the one immutable snapshot index.
15

31impl<'a, 'source> Join<'a, 'source> {
32    /// Called only by eager instantiation or possible-head production.
33    /// No totality premise is borrowed from original-family admission: a
34    /// defined false plan-decided comparison excludes that row under the
35    /// ordinary filters contract, including independent deferred scalar faults.
36    /// A source with generated bindings or structural/nested scopes retains
37    /// its current traversal until that broader failure inventory is reviewed.
38    pub(crate) fn select_numeric_probes(
39        &mut self,
40        rule: &RuleIr,
41        limits: &crate::FormulaLimits,
42        counters: &mut super::Counters,
43    ) -> Result<(), FormulaFailure> {
44        counters.work(limits, rule.location)?;
45        if !std::ptr::eq(self.literals.get(), rule.body.as_slice())
46            || !matches!(rule.head, HeadIr::Normal(_))
47            || rule.variables != rule.body_variables
48        {
49            return Ok(());
50        }
51        let mut comparison_found = false;
52        for literal in &rule.body {
53            counters.work(limits, rule.location)?;
54            match literal {
55                LiteralIr::Compare(comparison) => {
56                    comparison_found |= selected_relation(comparison.relation());
57                }
58                LiteralIr::Atom(..) | LiteralIr::TupleCompare(..) => {}
59                _ => return Ok(()),
60            }
61        }
62        self.expression_keys = comparison_found.then_some(Eligible);
63        Ok(())
64    }

66    pub(super) fn expression_key_probe(
67        &self,
68        pattern: PatternOccurrence<'a>,
69        source: Option<&'source RelationRows<'source>>,
70        mut context: Context<'_, &Computation<'_, '_>>,
71    ) -> Result<Option<Probe<'a, 'source>>, FormulaFailure> {
72        if self.expression_keys.is_none() || self.source_evidence {
73            return Ok(None);
74        }
75        #[cfg(test)]
76        if tests::disabled() {
77            return Ok(None);
78        }
79        let (PositivePattern::Flat(flat), Some(source)) = (pattern.pattern, source) else {
80            return Ok(None);
81        };
82        if source.row_count() == 0 {
83            return Ok(None);
84        }
85        // Preserve expression-equality selection ahead of range selection:
86        // a diagonal equality must not be displaced by an earlier R<S filter.
87        for equality in [true, false] {
88            for index in self.plan.decisions.decided_at(self.depth) {
89                tick(&mut context)?;
90                let Some(comparison) = super::comparison(&self.literals[index]) else {
91                    continue;
92                };
93                let ([left, right], relation) = comparison.operands();
94                if !selected_relation(relation) || (relation == Relation::Eq) != equality {
95                    continue;
96                }
97                for (key, bound, relation) in
98                    [(left, right, relation), (right, left, reversed(relation))]
99                {
100                    if let Some(probe) = self.numeric_candidate(
101                        ([key, bound], relation),
102                        flat,
103                        source,
104                        pattern.source,
105                        &mut context,
106                    )? {
107                        return Ok(Some(probe));
108                    }
109                }
110            }
111        }
112        Ok(None)
113    }

115    fn numeric_candidate(
116        &self,
117        comparison: ([&Expression; 2], Relation),
118        pattern: zetesis_core::PatternRef<'_>,
119        source: &'source RelationRows<'source>,
120        occurrence: usize,
121        context: &mut Context<'_, &Computation<'_, '_>>,
122    ) -> Result<Option<Probe<'a, 'source>>, FormulaFailure> {
123        let ([key, bound], relation) = comparison;
124        // Stage 2a indexes one column, while 2b allows a checked expression.
125        if relation != Relation::Eq && !matches!(key.nodes.as_slice(), [Operation::Variable(_)]) {
126            return Ok(None);
127        }
128        if !self.key_available(key, bound, pattern, context)? {
129            return Ok(None);
130        }
131        let binding = self.values.view(
132            context.computation.read(),
133            context.work.limits,
134            context.work.counters,
135            context.work.location,
136        )?;
137        // The ordinary equality posting remains the first choice, including
138        // its existing delta clipping and residual alias/constant checks.
139        if let Some(posting) = self.support.probe_at(
140            Some(source),
141            pattern,
142            binding,
143            context.work.limits,
144            context.work.counters,
145            context.work.location,
146        )? {
147            return self
148                .indexed_probe(
149                    Some(posting),
150                    Some(source),
151                    occurrence,
152                    context.work.limits,
153                    context.work.counters,
154                    context.work.location,
155                )
156                .map(Some);
157        }
158        let mut scratch = buffer(context)?;
159        let wanted = evaluate(&bound.nodes, &mut scratch, context, |slot, context| {
160            tick(context)?;
161            let value =
162                self.values
163                    .read(slot, context.computation.read(), context.work.location)?;
164            Ok(number(value.descriptor()))
165        })?;
166        let Some(wanted) = wanted else {
167            return Ok(None);
168        };
169        let range = delta::interval(
170            self.delta,
171            occurrence,
172            source.old_rows(),
173            source.row_count(),
174        );
175        let query = Context::new(
176            context.computation,
177            context.work.limits,
178            context.work.counters,
179            context.work.location,
180        );
181        if relation == Relation::Eq {
182            self.support
183                .expression_key(source, key, pattern, wanted, range, query)
184                .map(|rows| rows.map(Probe::Expression))
185        } else {
186            self.support
187                .range_key(source, key, pattern, (relation, wanted), range, query)
188                .map(|rows| rows.map(Probe::Range))
189        }
190    }
191
192    fn key_available(
193        &self,
194        key: &Expression,
195        bound: &Expression,
196        pattern: zetesis_core::PatternRef<'_>,
197        context: &mut Context<'_, &Computation<'_, '_>>,
198    ) -> Result<bool, FormulaFailure> {
199        let mut unbound = false;
200        for operation in &key.nodes {
201            tick(context)?;
202            match operation {
203                Operation::Variable(slot) => {
204                    if column(pattern, *slot, context)?.is_none() {
205                        return Ok(false);
206                    }
207                    unbound |= !self.values.is_bound(*slot, context.work.location)?;
208                }
209                Operation::Constructor(_) => return Ok(false),
210                _ => {}
211            }
212        }
213        if !unbound {
214            return Ok(false);
215        }
216        for operation in &bound.nodes {
217            tick(context)?;
218            match operation {
219                Operation::Variable(slot)
220                    if !self.values.is_bound(*slot, context.work.location)? =>
221                {
222                    return Ok(false);
223                }
224                Operation::Constructor(_) => return Ok(false),
225                _ => {}
226            }
227        }
228        Ok(true)
229    }
230}

232fn selected_relation(relation: Relation) -> bool {
233    matches!(
234        relation,
235        Relation::Eq | Relation::Lt | Relation::Le | Relation::Gt | Relation::Ge
236    )
237}
238
239fn reversed(relation: Relation) -> Relation {
240    match relation {
241        Relation::Lt => Relation::Gt,
242        Relation::Le => Relation::Ge,
243        Relation::Gt => Relation::Lt,
244        Relation::Ge => Relation::Le,
245        relation => relation,
246    }
