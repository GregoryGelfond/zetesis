1//! Successful scalar preparation for one rule in one immutable support round.
2//!
3//! The first eligible variant owns ordinary preparation and traversal. After it
4//! finishes, this owner retains that same successful expression map. Subsequent
5//! variants borrow the map but prepare their own join order and multiples target.
6//! The owner borrows both rule and support, so it cannot outlive publication.
7//! Its presence grants neither source completion nor generic table selection.
8
9use crate::FormulaFailure;
10use crate::expansion::Budget;
11use crate::formula_ir::RuleIr;
12use crate::formula_support::{
13    Computation, Context, Ir, Join, StorageLease, Support, delta, projections, queries,
14};
15
16pub(in crate::formula_support) struct RoundPreparation<'rule, 'source> {
17    rule: Ir<'rule, 'source, RuleIr>,
18    support: &'rule Support<'source>,
19    values: projections::ProjectionValues<'rule>,
20    lease: StorageLease,
21}
22
23impl<'rule, 'source> RoundPreparation<'rule, 'source> {
24    /// Move the first successful cursor's map without cloning its buffers. Its
25    /// lease continues to own the map header and payload; this owner charges
26    /// only its additional header, including the optional slot's padding.
27    /// Refusal publishes no owner and releases the consumed cursor normally.
28    pub(in crate::formula_support) fn retain(
29        rule: impl Into<Ir<'rule, 'source, RuleIr>>,
30        join: Join<'rule, 'source>,
31        context: Context<'_, &Computation<'_, '_>>,
32    ) -> Result<Self, FormulaFailure> {
33        let source_rule = rule.into();
34        let rule = source_rule.get();
35        let Context { computation, work } = context;
36        work.counters.work(work.limits, rule.location)?;
37        if !std::ptr::eq(join.literals.get(), rule.body.as_slice())
38            || join.delta.is_none()
39            || join.source_evidence
40            || join.certified_total
41            || join.multiples.is_none()
42        {
43            return Err(FormulaFailure::SupportRelation {
44                error: zetesis_core::relation::Failure::Owner,
45                location: rule.location,
46            });
47        }
48        let projections::Projections::Local(values) = join.projections else {
49            return Err(FormulaFailure::SupportRelation {
50                error: zetesis_core::relation::Failure::Owner,
51                location: rule.location,
52            });
53        };
54        let mut lease = computation.lease();
55        let header = size_of::<Option<Self>>() - size_of::<projections::ProjectionValues<'_>>();
56        lease.observe(header, rule.location)?;
57        computation.storage_observed(
58            &lease,
59            0,
60            header,
61            work.limits,
62            work.counters,
63            rule.location,
64        )?;
65        Ok(Self {
66            rule: source_rule,
67            support: join.support,
68            values,
69            lease,
70        })
71    }
72
73    /// Build a fresh variant with the same rule/source authority. The borrowed
74    /// map never grows. A prefix outside its covered domain still uses ordinary
75    /// checked evaluation. `select_round_multiples` separately checks whether
76    /// this variant's plan binds the divisor before its target.
77    pub(in crate::formula_support) fn rows<'a>(
78        &'a self,
79        variant: delta::Variant,
80        domains: Option<&'a queries::Guards<'a, 'source>>,
81        budget: &mut Budget,
82        context: Context<'_, &Computation<'_, '_>>,
83    ) -> Result<Join<'a, 'source>, FormulaFailure> {
84        context
85            .work
86            .counters
87            .work(context.work.limits, self.rule.location)?;
88        context
89            .computation
90            .allowance(&self.lease, context.work.limits, self.rule.location)?;
91        Join::variant_rule_with_total(
92            self.rule,
93            variant,
94            self.support,
95            domains,
96            Some(&self.values),
97            budget,
98            context,
99        )
100    }
101}
