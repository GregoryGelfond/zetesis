1//! The atom occurrences a frame holds, at every depth.
2//!
3//! One enumeration serves two readers: demand, which keeps postings for the
4//! columns a join can bind, and the producer registration, which records the
5//! predicates a rich producer reads. It matches every form of the IR with no
6//! wildcard arm, so a new form is a compile error here rather than a read
7//! silently missing from either reader.
8
9use super::components::{Pattern, Predicate};
10use crate::FormulaFailure;
11use crate::formula_conditional_ir::{Consequent, ConsequentOperand};
12use crate::formula_ir::{AggregateKey, HeadIr, LiteralIr, Projection, RuleIr};
13
14/// One atom occurrence, by its compiled coordinate.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub(super) enum Occurrence {
17    /// An atom pattern of either sign: a body atom, a structural occurrence,
18    /// an aggregate element's atom key, a conditional literal's atom, or a
19    /// projection's witness atom.
20    Atom(Pattern),
21    /// The predicate an argument projection reads.
22    Projected(Predicate),
23}
24
25/// A reader of occurrences. The walk charges one `work` unit per literal it
26/// visits, at every depth, before reporting that literal's occurrences.
27pub(super) trait Visitor {
28    fn work(&mut self) -> Result<(), FormulaFailure>;
29    fn occurrence(&mut self, occurrence: Occurrence) -> Result<(), FormulaFailure>;
30}
31
32/// The frames a traversal of `rule` reads: its body, then the condition of
33/// each head element. The atoms its head derives are not among them.
34pub(super) fn rule(rule: &RuleIr, visitor: &mut impl Visitor) -> Result<(), FormulaFailure> {
35    frame(&rule.body, visitor)?;
36    match &rule.head {
37        HeadIr::Normal(_) | HeadIr::Disjunction(_) => {}
38        HeadIr::Choice(group) => {
39            for element in &group.elements {
40                frame(&element.condition, visitor)?;
41            }
42        }
43        HeadIr::ConditionalDisjunction { elements, .. } => {
44            for element in elements {
45                frame(&element.condition, visitor)?;
46            }
47        }
48    }
49    Ok(())
50}

52/// Every occurrence of `literals` and of every frame nested in them: aggregate
53/// elements' conditions and atom keys, conditional literals' conditions,
54/// consequents and bindings, and projections' predicates and witnesses.
55pub(super) fn frame(
56    literals: &[LiteralIr],
57    visitor: &mut impl Visitor,
58) -> Result<(), FormulaFailure> {
59    let mut pending = vec![literals];
60    while let Some(literals) = pending.pop() {
61        for literal in literals {
62            visitor.work()?;
63            match literal {
64                LiteralIr::Atom(_, atom) => visitor.occurrence(Occurrence::Atom(*atom))?,
65                LiteralIr::PatternAtom(atom) => visitor.occurrence(Occurrence::Atom(atom.atom))?,
66                LiteralIr::ProjectedAtom(_, value) => projection(value, visitor, &mut pending)?,
67                LiteralIr::Conditional(value) => {
68                    pending.push(&value.condition);
69                    match &value.consequent {
70                        Consequent::Atoms(_, alternatives) => {
71                            for alternative in alternatives {
72                                pending.push(&alternative.bindings);
73                                match &alternative.operand {
74                                    ConsequentOperand::Atom(atom) => {
75                                        visitor.occurrence(Occurrence::Atom(*atom))?;
76                                    }
77                                    ConsequentOperand::Projection(value) => {
78                                        projection(value, visitor, &mut pending)?;
79                                    }
80                                }
81                            }
82                        }
83                        Consequent::Guards(alternatives) => {
84                            for alternative in alternatives {
85                                pending.push(&alternative.bindings);
86                            }
87                        }
88                        Consequent::Guard(_) => {}
89                    }
90                }
91                LiteralIr::Aggregate(value) => {
92                    for element in &value.elements {
93                        pending.push(&element.condition);
94                        match &element.key {
95                            AggregateKey::Atom(atom) => {
96                                visitor.occurrence(Occurrence::Atom(*atom))?;
97                            }
98                            AggregateKey::Tuple(_) => {}
99                        }
100                    }
101                }
102                LiteralIr::Compare(..)
103                | LiteralIr::ArgumentCheck { .. }
104                | LiteralIr::TupleCompare(..)
105                | LiteralIr::Guard(_)
106                | LiteralIr::HeadGuard(_)
107                | LiteralIr::Bind { .. }
108                | LiteralIr::Range { .. } => {}
109            }
110        }
111    }
112    Ok(())
113}

115fn projection<'ir>(
116    value: &'ir Projection,
117    visitor: &mut impl Visitor,
118    pending: &mut Vec<&'ir [LiteralIr]>,
119) -> Result<(), FormulaFailure> {
120    match value {
121        Projection::Arguments { predicate, .. } => {
122            visitor.occurrence(Occurrence::Projected(*predicate))?;
123        }
124        Projection::Witnesses { atom, bindings, .. } => {
125            visitor.occurrence(Occurrence::Atom(*atom))?;
126            pending.push(bindings);
127        }
128    }
129    Ok(())
130}
