1//! Completed necessary selection of one short, immutable indexed posting.
2//!
3//! The join owns one receipt per original positive occurrence. Its resolved
4//! relation and immutable row filter remain borrowed for that join's lifetime.
5//! Bits denote positions in the exact borrowed posting, never atom identities.
6//! Matching, scalar evaluation and positive-row evidence are not retained here.
7
8use super::{
9    Computation, Context, Counters, GroundingWork, Join, PatternOccurrence, Probe, RowFilter,
10    delta, reserve_exact,
11};
12use crate::{FormulaFailure, FormulaLimits, ProgramSite};
13
14#[derive(Default)]
15pub(super) struct Receipt<'a> {
16    posting: Option<&'a [usize]>,
17    selected: u64,
18    covered: usize,
19    permits_open: bool,
20    complete: bool,
21    replay: bool,
22}
23
24impl<'a> Receipt<'a> {
25    /// Only an exhausted traversal of this exact posting and eligibility mode
26    /// can be replayed. A different probe discards the previous evidence.
27    pub(super) fn start(&mut self, probe: &Probe<'a, '_>, permits_open: bool) {
28        let Probe::Indexed(delta::Rows::Posting(posting)) = probe else {
29            *self = Self::default();
30            return;
31        };
32        if !(2..=u64::BITS as usize).contains(&posting.len()) {
33            *self = Self::default();
34            return;
35        }
36        if self.complete
37            && self.permits_open == permits_open
38            && self
39                .posting
40                .is_some_and(|previous| std::ptr::eq(previous, *posting))
41        {
42            self.replay = true;
43        } else {
44            *self = Self {
45                posting: Some(posting),
46                permits_open,
47                ..Self::default()
48            };
49        }
50    }
51
52    pub(super) fn next(
53        &self,
54        probe: &Probe<'_, '_>,
55        position: &mut usize,
56        limits: &FormulaLimits,
57        counters: &mut Counters,
58        location: ProgramSite,
59    ) -> Result<Option<usize>, FormulaFailure> {
60        if !self.replay {
61            return probe.next(position, limits, counters, location);
62        }
63        counters.work(limits, location)?;
64        let posting = self.posting.expect("completed posting retained");
65        if *position >= posting.len() {
66            return Ok(None);
67        }
68        let remaining = self.selected & (u64::MAX << *position);
69        if remaining == 0 {
70            *position = posting.len();
71            return Ok(None);
72        }
73        let selected = remaining.trailing_zeros() as usize;
74        *position = selected + 1;
75        Ok(Some(posting[selected]))
76    }
77
78    /// Record only a successful necessary-selection decision, before matching.
79    /// A fault leaves a gap in coverage, so retrying cannot certify that probe.
80    pub(super) fn record(&mut self, position: usize, permitted: bool) {
81        if !self.replay
82            && self.posting.is_some_and(|posting| position < posting.len())
83            && position == self.covered
84        {
85            if permitted {
86                self.selected |= 1 << position;
87            }
88            self.covered += 1;
89        }
90    }
91
92    pub(super) fn finish(&mut self) {
93        if !self.replay {
94            self.complete = self
95                .posting
96                .is_some_and(|posting| self.covered == posting.len());
97        }
98    }
99}

101impl<'a> Join<'a, '_> {
102    /// Advance the current occurrence's authenticated source. Completed
103    /// necessary selection can omit rows; every offered row still belongs to
104    /// the live probe, and matching remains the enclosing join's responsibility.
105    pub(super) fn advance_probe(
106        &mut self,
107        pattern: PatternOccurrence<'a>,
108        context: Context<'_, &mut Computation<'_, '_>>,
109    ) -> Result<Option<usize>, FormulaFailure> {
110        let Context {
111            computation,
112            work:
113                GroundingWork {
114                    limits,
115                    counters,
116                    location,
117                },
118        } = context;
119        let preparing = self.probes[self.depth].is_none();
120        self.prepare_probe(
121            pattern,
122            Context::new(computation, limits, counters, location),
123        )?;
124        let probe = self.probes[self.depth].as_ref().expect("prepared probe");
125        let Some(selection) = self.probe_selections.get_mut(self.depth) else {
126            return probe.next(&mut self.positions[self.depth], limits, counters, location);
127        };
128        if preparing {
129            selection.start(
130                probe,
131                self.positive_prefix
132                    .as_ref()
133                    .is_none_or(super::rows::PositivePrefix::permits_open),
134            );
135        }
136        let row = selection.next(
137            probe,
138            &mut self.positions[self.depth],
139            limits,
140            counters,
141            location,
142        )?;
143        if row.is_none() {
144            selection.finish();
145        }
146        Ok(row)
147    }
148
149    /// The metadata is optional. Decline a capacity proposal that would displace
150    /// the existing join; once allocation starts, its actual capacity and any
151    /// failure retain the ordinary typed accounting boundary.
152    pub(super) fn prepare_selections(
153        &mut self,
154        context: Context<'_, &super::Computation<'_, '_>>,
155    ) -> Result<(), FormulaFailure> {
156        if self.plan.patterns.len() < 2
157            || !self.row_filter.is_some_and(RowFilter::immutable_selection)
158        {
159            return Ok(());
160        }
161        let Context {
162            computation,
163            work:
164                GroundingWork {
165                    limits,
166                    counters,
167                    location,
168                },
169        } = context;
170        let count = self.plan.patterns.len();
171        let other = self.storage_bytes();
172        let proposed = other as u128 + count as u128 * size_of::<Receipt<'_>>() as u128;
173        if proposed > computation.allowance(&self.lease, limits, location)? as u128 {
174            return Ok(());
175        }
176        reserve_exact(
177            &mut self.probe_selections,
178            count,
179            &mut self.lease,
180            other,
181            Context::new(computation, limits, counters, location),
182        )?;
183        for _ in 0..count {
184            counters.work(limits, location)?;
185            self.probe_selections.push(Receipt::default());
186        }
187        Ok(())
188    }
189}
