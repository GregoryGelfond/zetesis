1//! One authenticated dependency order for the complete signed normal source.
2//!
3//! Unlike positive support scheduling, every body predicate must have a
4//! strictly smaller component rank. No producer or negative edge is omitted.
5
6use super::{Memory, Node, find, graph_nodes, invalid};
7use crate::formula_facts::Item;
8use crate::formula_ir::{HeadIr, LiteralIr, Prepared};
9use crate::formula_support::{Counters, SourceScope, StorageLease, SupportCatalog};
10use crate::{AnalysisBasis, FormulaFailure, FormulaLimits, ProgramSite};
11use themelios_program::program::DefaultNegation;
12use zetesis_core::relation::Failure;
13
14pub(crate) struct AcyclicPlan {
15    scope: SourceScope,
16    // Component rank followed by item position: a rule's, or a fact block's
17    // where its fact rules stood. Constraints sort last.
18    items: Vec<(usize, usize)>,
19    _lease: StorageLease,
20}
21
22impl AcyclicPlan {
23    pub(crate) fn prepare(
24        source: &Prepared,
25        catalog: &SupportCatalog,
26        limits: &FormulaLimits,
27        counters: &mut Counters,
28        location: ProgramSite,
29    ) -> Result<Option<Self>, FormulaFailure> {
30        if !eligible(source, limits, counters, location)? {
31            return Ok(None);
32        }
33        let external = catalog.bytes(location)? as u128 + counters.workspace_bytes() as u128;
34        let mut memory = Memory::new(size_of::<Self>(), external, limits, counters, location);
35        memory.add(0)?;
36        let nodes = graph_nodes(source, &mut memory, limits, counters, location)?;
37        let components = catalog.component_view(limits, counters, location)?;
38        let mut items = Vec::new();
39        memory.reserve(&mut items, source.items().len())?;
40        for (position, item) in source.items().iter().enumerate() {
41            let rule = match item {
42                Item::Rule(_, rule) => rule,
43                Item::Facts(block) => {
44                    // A block's rows share its predicate, and so its rank;
45                    // a fact reads no predicate.
46                    let location = block.location();
47                    counters.work(limits, location)?;
48                    let view = components
49                        .ok_or_else(|| crate::formula_support::components::missing(location))?;
50                    let node = find(
51                        nodes.len(),
52                        |i| nodes[i].signature,
53                        block.predicate().get(view, limits, counters, location)?,
54                        limits,
55                        counters,
56                        location,
57                    )?;
58                    let rank = nodes[node]
59                        .component
60                        .ok_or_else(|| invalid(Failure::Owner, location))?;
61                    items.push((rank, position));
62                    continue;
63                }
64            };
65            counters.work(limits, rule.location)?;
66            let HeadIr::Normal(head) = &rule.head else {
67                unreachable!("class checked")
68            };
69            let rank = if let Some(head) = head {
70                let view = components
71                    .ok_or_else(|| crate::formula_support::components::missing(rule.location))?;
72                let node = find(
73                    nodes.len(),
74                    |i| nodes[i].signature,
75                    head.get(view, limits, counters, rule.location)?.predicate(),
76                    limits,
77                    counters,
78                    rule.location,
79                )?;
80                nodes[node]
81                    .component
82                    .ok_or_else(|| invalid(Failure::Owner, rule.location))?
83            } else {
84                usize::MAX
85            };
86            for literal in &rule.body {
87                counters.work(limits, rule.location)?;
88                let LiteralIr::Atom(_, atom) = literal else {
89                    continue;
90                };
91                let view = components
92                    .ok_or_else(|| crate::formula_support::components::missing(rule.location))?;
93                let node = find(
94                    nodes.len(),
95                    |i| nodes[i].signature,
96                    atom.get(view, limits, counters, rule.location)?.predicate(),
97                    limits,
98                    counters,
99                    rule.location,
100                )?;
101                let dependency = nodes[node]
102                    .component
103                    .ok_or_else(|| invalid(Failure::Owner, rule.location))?;
104                if dependency >= rank {
105                    return Err(invalid(Failure::Owner, rule.location));
106                }
107            }
108            items.push((rank, position));
109        }
110        // Sort physical source positions with the shared bounded operation;
111        // semantic signatures remain in the original analysis owner.
112        crate::formula_support::sort::by(
113            &mut items,
114            crate::formula_support::GroundingWork::new(limits, counters, location),
115            |left, right, _| Ok(left.cmp(right)),
116        )?;
117        memory.release(size_of::<Vec<Node<'_>>>() + nodes.capacity() * size_of::<Node<'_>>());
118        drop(nodes);
119        let mut lease = counters.lease();
120        lease.observe(memory.bytes, location)?;
121        Ok(Some(Self {
122            scope: catalog.source_scope().clone(),
123            items,
124            _lease: lease,
125        }))
126    }
127
128    pub(crate) fn authenticates(&self, catalog: &SupportCatalog) -> bool {
129        self.scope.same(catalog.source_scope())
130    }
131
132    /// The plan's (component rank, item position) entries, in scan order.
133    pub(crate) fn items(&self) -> &[(usize, usize)] {
134        &self.items
135    }
136}

138fn eligible(
139    source: &Prepared,
140    limits: &FormulaLimits,
141    counters: &mut Counters,
142    location: ProgramSite,
143) -> Result<bool, FormulaFailure> {
144    counters.work(limits, location)?;
145    if source.analysis_basis != AnalysisBasis::NormalizedProgram
146        || !source.objectives.is_empty()
147        || !source.objective_declarations.is_empty()
148        || source.project_selection.is_explicit()
149        || !source.projection.is_empty()
150    {
151        return Ok(false);
152    }
153    for component in source.analysis.dependencies().components() {
154        counters.work(limits, location)?;
155        if component.is_recursive() {
156            return Ok(false);
157        }
158    }
159    // A fact block is a normal producer with an empty body: always in class.
160    for rule in source.rules_without_facts() {
161        counters.work(limits, rule.location)?;
162        if !matches!(rule.head, HeadIr::Normal(_))
163            || rule.bindings.is_some()
164            || rule.body_variables != rule.variables
165        {
166            return Ok(false);
167        }
168        for literal in &rule.body {
169            counters.work(limits, rule.location)?;
170            if !matches!(
171                literal,
172                LiteralIr::Atom(DefaultNegation::None | DefaultNegation::Not, _)
173                    | LiteralIr::Compare(..)
174            ) {
175                return Ok(false);
176            }
177        }
178    }
179    Ok(true)
180}
