1//! What each producer reads, prepared once per completion on the full schedule.
2//!
3//! Without a producer plan every rule is registered, and its registration
4//! records once what a visit needs:
5//! - a certified producer's positive body occurrences, flat and structural,
6//!   whose delta variants it takes;
7//! - a rich producer's read set: the distinct predicates of every atom
8//!   occurrence its frames hold, at any depth, of either sign
9//!   ([`occurrences::rule`]).
10//!
11//! Every producer is visited in the first round. Afterwards a certified
12//! producer is visited only in a round after a predicate of one of its
13//! positive occurrences gained rows ([`Visits`]): its delta variants join
14//! rows of the last publication alone, so any other visit has no variant. A
15//! rich producer is visited in every round. A certified producer the
16//! component schedule holds back is visited like any other: its visit takes
17//! no variant and is counted as deferred, and the schedule traverses it
18//! outside the round's visits, whether or not it was visited.
19//!
20//! After the first round, a rich producer none of whose read predicates gained
21//! rows is not traversed. Every relation it reads is then the one its last
22//! traversal read, and support membership only grows, so another traversal
23//! would follow the same plan over the same rows and derive only heads already
24//! supported. Facts have no producer slot: a fact block enters support in the
25//! first round, and its rows wake a reading rich producer in the second.
26//!
27//! A rich producer whose reads all lie in dependency components before its
28//! heads' is instead traversed once, when those components reach their fixed
29//! point ([`components`]). Any other rich producer that proposes an
30//! assignment's values from elements over earlier decidable components is
31//! held back until those reach their fixed point, and traversed per change
32//! from then on. A producer of either kind is also held back for the atoms
33//! it negates in earlier decidable components, until those are complete.

64    reads: Vec<Occurrence>,

75/// The producers a round visits. After the first round a certified producer
76/// is visited only when a predicate of one of its positive occurrences
77/// gained rows at the last publication: its delta variants join new rows
78/// alone, so without one it has no variant and a visit derives nothing.
79struct Visits<'source> {
80    /// The certified producers' positive occurrences, by predicate.
81    wake: Wake<'source>,
82    /// Visited in every round: each rich producer, whose registration
83    /// decides at the visit whether it is traversed, and each certified
84    /// producer with an occurrence whose predicate the graph does not name.
85    always: Vec<usize>,
86}

88#[derive(Clone, Debug, PartialEq, Eq)]
89enum Producer {
90    /// A normal head over positive, negated and scalar body literals: its
91    /// positive occurrences in `inputs`.
92    Certified(Range<usize>),
93    /// Every other rule with a head: its read set in `reads`.
94    Rich(Range<usize>),
95    /// A constraint proposes no head and is never scheduled.
96    Constraint,
97}

99impl<'source> Registration<'source> {
100    /// Register every rule of `rules`, resolving read predicates through
101    /// `view` to deduplicate them. Charges one unit per rule and per body
102    /// literal classified; a rich rule's walk charges one unit per literal it
103    /// visits, at every depth, and deduplication one unit per occurrence
104    /// resolved and per comparison, with the compared names' lengths. The
105    /// retained vectors are admitted against the support byte ceiling beside
106    /// the `external` bytes other owners hold, and scratch is released before
107    /// returning.
108    pub(in crate::formula_support) fn prepare(
109        rules: &'source [RuleIr],
110        view: Option<TemplateComponentsRef<'_>>,
111        external: u128,
112        limits: &FormulaLimits,
113        counters: &mut Counters,
114        location: ProgramSite,
115    ) -> Result<Self, FormulaFailure> {
116        let mut memory = Memory::new(size_of::<Self>(), external, limits, counters, location);
117        memory.add(0)?;
118        let mut producers = Vec::new();
119        let mut inputs = Vec::new();
120        let mut reads = Vec::new();
121        memory.reserve(&mut producers, rules.len())?;
122        let mut found = Vec::new();
123        let mut distinct: Vec<(PredicateRef<'_>, Occurrence)> = Vec::new();
124        for rule in rules {
125            counters.work(limits, rule.location)?;
126            let producer =
127                match classify(rule, limits, counters)? {
128                    Kind::Constraint => Producer::Constraint,
129                    Kind::Certified(count) => {
130                        let start = inputs.len();
131                        memory.reserve(&mut inputs, count)?;
132                        inputs.extend(rule.body.iter().enumerate().filter_map(
133                            |(occurrence, literal)| positive(literal).map(|_| occurrence),
134                        ));
135                        Producer::Certified(start..inputs.len())
136                    }
137                    Kind::Rich => {
138                        found.clear();
139                        let mut walk = Walk {
140                            found: &mut found,
141                            memory: &mut memory,
142                            limits,
143                            counters,
144                            location: rule.location,
145                        };
146                        occurrences::rule(rule, &mut walk)?;
147                        let start = reads.len();
148                        if let Some(view) = view {
149                            distinct.clear();
150                            memory.reserve(&mut distinct, found.len())?;
151                            for &occurrence in &found {
152                                insert(
153                                    &mut distinct,
154                                    occurrence,
155                                    view,
156                                    limits,
157                                    counters,
158                                    rule.location,
159                                )?;
160                            }
161                            memory.reserve(&mut reads, distinct.len())?;
162                            reads.extend(distinct.iter().map(|&(_, occurrence)| occurrence));
163                        } else {
164                            // No component view: keep every coordinate, a superset.
165                            memory.reserve(&mut reads, found.len())?;
166                            reads.extend_from_slice(&found);
167                        }
168                        Producer::Rich(start..reads.len())
169                    }
170                };
171            producers.push(producer);
172        }
173        let scratch_bytes = found.capacity() * size_of::<Occurrence>()
174            + distinct.capacity() * size_of::<(PredicateRef<'_>, Occurrence)>();
175        drop(distinct);
176        drop(found);
177        memory.release(scratch_bytes);
178        Ok(Self {
179            rules,
180            producers,
181            inputs,
182            reads,
183            schedule: None,
184            wake: None,
185            bytes: memory.bytes,
186        })
187    }

189    /// Schedule by dependency component every producer that can wait for
190    /// components it reads. Its vectors are admitted beside the registration's
191    /// own bytes and `external`, and counted with them.
192    pub(in crate::formula_support) fn schedule(
193        &mut self,
194        prepared: &'source crate::formula_ir::Prepared,
195        view: TemplateComponentsRef<'_>,
196        external: u128,
197        (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
198    ) -> Result<(), FormulaFailure> {
199        let mut memory = Memory::new(0, external + self.bytes as u128, limits, counters, location);
200        memory.add(0)?;
201        let schedule = components::Schedule::prepare(
202            self,
203            prepared,
204            view,
205            &mut memory,
206            (limits, counters, location),
207        )?;
208        // Without a waiting producer every vector was scratch and is gone.
209        if let Some(schedule) = schedule {
210            memory.add(size_of::<components::Schedule<'_>>())?;
211            self.schedule = Some(Box::new(schedule));
212            self.bytes += memory.bytes;
213        }
214        Ok(())
215    }

217    /// Prepare the producers each round visits ([`Visits`]). Every producer
218    /// is visited in the first round, in rule order, so that a fact block
219    /// still enters support at its item position. The test-only reference
220    /// that traverses every rich producer in every round keeps visiting
221    /// every rule. The retained vectors are admitted beside the
222    /// registration's own bytes and `external`, and counted with them; the
223    /// graph's nodes are scratch. One unit per producer and per positive
224    /// occurrence, with each occurrence's resolution. This additional setup
225    /// work and storage can change tight-limit refusals; such a refusal
226    /// propagates; there is no fallback on refused wake-set preparation.
227    pub(in crate::formula_support) fn wake(
228        &mut self,
229        prepared: &'source crate::formula_ir::Prepared,
230        view: TemplateComponentsRef<'_>,
231        external: u128,
232        (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
233    ) -> Result<(), FormulaFailure> {
234        if every_round_traversed() {
235            return Ok(());
236        }
237        let mut memory = Memory::new(
238            size_of::<Visits<'_>>(),
239            external + self.bytes as u128,
240            limits,
241            counters,
242            location,
243        );
244        memory.add(0)?;
245        let nodes = super::graph_nodes(prepared, &mut memory, limits, counters, location)?;
246        let resolver = certain::Resolver {
247            nodes: &nodes,
248            view,
249            limits,
250        };
251        let mut rules = Vec::new();
252        let mut predicates = Vec::new();
253        let mut always = Vec::new();
254        // The two scratch vectors' headers; the retained ones are fields.
255        memory.add(size_of::<Vec<Option<Range<usize>>>>() + size_of::<Vec<usize>>())?;
256        memory.reserve(&mut rules, self.producers.len())?;
257        for (index, producer) in self.producers.iter().enumerate() {
258            counters.work(limits, location)?;
259            let rule = &self.rules[index];
260            let start = predicates.len();
261            match producer {
262                Producer::Constraint => {
263                    rules.push(None);
264                    continue;
265                }
266                Producer::Rich(_) => {
267                    memory.reserve(&mut always, 1)?;
268                    always.push(index);
269                }
270                Producer::Certified(inputs) => {
271                    let mut named = true;
272                    for &occurrence in &self.inputs[inputs.clone()] {
273                        counters.work(limits, rule.location)?;
274                        let pattern = positive(&rule.body[occurrence])
275                            .expect("a registered input is a positive occurrence");
276                        if let Some(node) = resolver.node(pattern, counters, rule.location)? {
277                            memory.reserve(&mut predicates, 1)?;
278                            predicates.push(node);
279                        } else {
280                            named = false;
281                        }
282                    }
283                    if !named {
284                        predicates.truncate(start);
285                        memory.reserve(&mut always, 1)?;
286                        always.push(index);
287                    }
288                }
289            }
290            rules.push(Some(start..predicates.len()));
291        }
292        let mut wake = Wake::prepare(
293            &nodes,
294            (&rules, &predicates),
295            false,
296            &mut memory,
297            limits,
298            counters,
299            location,
300        )?;
301        for (index, inputs) in rules.iter().enumerate() {
302            if inputs.is_some() {
303                wake.mark(index, limits, counters, location)?;
304            }
305        }
306        let scratch_bytes = size_of::<Vec<super::Node<'_>>>()
307            + nodes.capacity() * size_of::<super::Node<'_>>()
308            + size_of::<Vec<Option<Range<usize>>>>()
309            + rules.capacity() * size_of::<Option<Range<usize>>>()
310            + size_of::<Vec<usize>>()
311            + predicates.capacity() * size_of::<usize>();
312        drop(rules);
313        drop(predicates);
314        drop(nodes);
315        memory.release(scratch_bytes);
316        self.wake = Some(Visits { wake, always });
317        self.bytes += memory.bytes;
318        Ok(())
319    }

321    /// This round's producers, in rule order: the ones [`Self::advance`]
322    /// selected, or every rule without a wake set.
323    pub(in crate::formula_support) fn round(&self) -> super::Schedule<'_> {
324        match &self.wake {
325            Some(visits) => visits.wake.schedule(),
326            None => super::Schedule::all(self.rules),
327        }
328    }

330    /// Select the next round's producers from this complete round's
331    /// proposed new atoms: the certified producers one of whose positive
332    /// occurrences' predicates gained an atom, and every producer visited in
333    /// every round. One unit per producer selected, beside the wake set's.
334    pub(in crate::formula_support) fn advance<'atoms>(
335        &mut self,
336        delta: impl Iterator<Item = zetesis_core::catalog::AtomRef<'atoms>>,
337        limits: &FormulaLimits,
338        counters: &mut Counters,
339        location: ProgramSite,
340    ) -> Result<(), FormulaFailure> {
341        let Some(visits) = &mut self.wake else {
342            return Ok(());
343        };
344        visits.wake.advance(delta, limits, counters, location)?;
345        for &producer in &visits.always {
346            visits.wake.mark(producer, limits, counters, location)?;
347        }
348        Ok(())
349    }

381    /// The variants of the producer at `index`, which must be `rule`. A
382    /// certified producer takes its registered occurrences' delta variants. A
383    /// rich producer is traversed in the first round, and later exactly when
384    /// one of its read predicates gained rows in `support`; a visit that ends
385    /// at that test is recorded as unchanged. A deferred producer is never
386    /// traversed here, and a gated one only after the schedule traversed it,
387    /// and then by the same test. A held-back certified producer takes no
388    /// variant before the schedule traversed it and its delta variants
389    /// after: that traversal read every row of its round's snapshot, so the
390    /// rows published since are the new ones. The owner check is a pointer
391    /// comparison, covered by the visit's own charge.
392    pub(in crate::formula_support) fn variants<'a, 'rows>(
393        &'a self,
394        index: usize,
395        rule: &'a RuleIr,
396        support: &'a Support<'rows>,
397        first: bool,
398        limits: &FormulaLimits,
399        counters: &mut Counters,
400    ) -> Result<Variants<'a, 'rows>, FormulaFailure> {
401        let (Some(producer), Some(registered)) = (self.producers.get(index), self.rules.get(index))
402        else {
403            return Err(invalid(Failure::Owner, rule.location));
404        };
405        if !std::ptr::eq(rule, registered) {
406            return Err(invalid(Failure::Owner, rule.location));
407        }
408        match producer {
409            Producer::Certified(inputs) => {
410                let held = self
411                    .schedule
412                    .as_ref()
413                    .and_then(|schedule| schedule.state(index));
414                match held {
415                    Some(components::State::Waiting | components::State::Ready) => {
416                        counters.record(Event::SupportDeferredVisit);
417                        Ok(delta::traversal(rule, support, false))
418                    }
419                    // The schedule's traversal stood for the first round's.
420                    Some(components::State::Traversed) => Ok(delta::prepared_variants(
421                        rule,
422                        support,
423                        &self.inputs[inputs.clone()],
424                        false,
425                    )),
426                    // Never held. `Schedule::state` reports a released
427                    // producer as never held, so its state only completes
428                    // the match.
429                    None | Some(components::State::Released) => Ok(delta::prepared_variants(
430                        rule,
431                        support,
432                        &self.inputs[inputs.clone()],
433                        first,
434                    )),
435                }
436            }
437            Producer::Rich(_)
438                if let Some(state) = self.schedule.as_ref().and_then(|schedule| {
439                    schedule
440                        .state(index)
441                        .filter(|&state| !schedule.resumed(index, state))
442                }) =>
443            {
444                // A deferred producer is traversed only by the component
445                // schedule, after its round's other producers, and so is a
446                // gated one until that traversal.
447                counters.record(if state == components::State::Traversed {
448                    Event::SupportUnchangedVisit
449                } else {
450                    Event::SupportDeferredVisit
451                });
452                Ok(delta::traversal(rule, support, false))
453            }
454            Producer::Rich(reads) => {
455                // A gated producer reaches this arm only after the schedule
456                // traversed it in full, which follows its round's last visit
457                // here: never in the first round.
458                let traversed = first
459                    || every_round_traversed()
460                    || affected(&self.reads[reads.clone()], rule, support, limits, counters)?;
461                counters.record(if traversed {
462                    Event::SupportRichTraversal
463                } else {
464                    Event::SupportUnchangedVisit
465                });
466                Ok(delta::traversal(rule, support, traversed))
467            }
468            Producer::Constraint => Err(invalid(Failure::Owner, rule.location)),
469        }
470    }
471}

479/// A rule's kind, from its head and the body literals it holds at top level,
480/// one unit per literal. Under the test-only structural reference, a
481/// structural occurrence makes the rule rich.
482fn classify(
483    rule: &RuleIr,
484    limits: &FormulaLimits,
485    counters: &mut Counters,
486) -> Result<Kind, FormulaFailure> {
487    match &rule.head {
488        HeadIr::Normal(None) => return Ok(Kind::Constraint),
489        HeadIr::Normal(Some(_)) | HeadIr::Disjunction(_) => {}
490        // A choice element's condition is a join of its own: its rows can grow
491        // after a body binding was derived.
492        HeadIr::Choice(choice)
493            if choice
494                .elements
495                .iter()
496                .all(|element| element.condition.is_empty()) => {}
497        HeadIr::Choice(_) | HeadIr::ConditionalDisjunction { .. } => return Ok(Kind::Rich),
498    }
499    let mut inputs = 0;
500    for literal in &rule.body {
501        counters.work(limits, rule.location)?;
502        match literal {
503            LiteralIr::Atom(DefaultNegation::None, _) | LiteralIr::PatternAtom(_) => inputs += 1,
504            // Join::new collects only positive inputs. Flat negative atoms
505            // generate no bindings and are no join filters: a decided one is
506            // tested against each complete row after the join. Any argument
507            // evaluation is a separate, retained scalar operation in the IR.
508            LiteralIr::Atom(DefaultNegation::Not | DefaultNegation::NotNot, _)
509            | LiteralIr::Compare(..)
510            | LiteralIr::ArgumentCheck { .. }
511            | LiteralIr::TupleCompare(..)
512            | LiteralIr::Guard(_)
513            | LiteralIr::HeadGuard(_)
514            | LiteralIr::Bind { .. }
515            | LiteralIr::Range { .. } => {}
516            LiteralIr::ProjectedAtom(..) | LiteralIr::Conditional(_) | LiteralIr::Aggregate(_) => {
517                return Ok(Kind::Rich);
518            }
519        }
520    }
521    #[cfg(test)]
522    if !delta::structural_enabled()
523        && rule
524            .body
525            .iter()
526            .any(|literal| matches!(literal, LiteralIr::PatternAtom(_)))
527    {
528        return Ok(Kind::Rich);
529    }
530    Ok(Kind::Certified(inputs))
531}

533/// The pattern of a positive flat or structural body occurrence: an input of
534/// a certified rule, and what a definite rule reads.
535fn positive(literal: &LiteralIr) -> Option<Pattern> {
536    match literal {
537        LiteralIr::Atom(DefaultNegation::None, atom) => Some(*atom),
538        LiteralIr::PatternAtom(pattern) => Some(pattern.atom),
539        _ => None,
540    }
541}

543/// Collects a rich rule's occurrences, charging one unit per literal visited.
544struct Walk<'a, 'limits> {
545    found: &'a mut Vec<Occurrence>,
546    memory: &'a mut Memory<'limits>,
547    limits: &'a FormulaLimits,
548    counters: &'a mut Counters,
549    location: ProgramSite,
550}
551
552impl occurrences::Visitor for Walk<'_, '_> {
553    fn work(&mut self) -> Result<(), FormulaFailure> {
554        self.counters.work(self.limits, self.location)
555    }
556
557    fn occurrence(&mut self, occurrence: Occurrence) -> Result<(), FormulaFailure> {
558        self.memory.reserve(self.found, 1)?;
559        self.found.push(occurrence);
560        Ok(())
561    }
562}

564/// The signed predicate an occurrence names, through `view`: one unit.
565fn predicate<'view>(
566    occurrence: Occurrence,
567    view: TemplateComponentsRef<'view>,
568    limits: &FormulaLimits,
569    counters: &mut Counters,
570    location: ProgramSite,
571) -> Result<PredicateRef<'view>, FormulaFailure> {
572    Ok(match occurrence {
573        Occurrence::Atom(pattern) => pattern.get(view, limits, counters, location)?.predicate(),
574        Occurrence::Projected(predicate) => predicate.get(view, limits, counters, location)?,
575    })
576}

578/// Keep `occurrence` unless an entry already names its predicate. Entries stay
579/// sorted by predicate, which compares sign, name and arity; each comparison
580/// charges one unit and the two names' lengths.
581fn insert<'view>(
582    distinct: &mut Vec<(PredicateRef<'view>, Occurrence)>,
583    occurrence: Occurrence,
584    view: TemplateComponentsRef<'view>,
585    limits: &FormulaLimits,
586    counters: &mut Counters,
587    location: ProgramSite,
588) -> Result<(), FormulaFailure> {
589    let named = predicate(occurrence, view, limits, counters, location)?;
590    let mut start = 0;
591    let mut end = distinct.len();
592    while start < end {
593        let middle = start + (end - start) / 2;
594        let entry = distinct[middle].0;
595        counters.charge_work(
596            1 + entry.name().len() as u128 + named.name().len() as u128,
597            limits,
598            location,
599        )?;
600        match entry.cmp(&named) {
601            std::cmp::Ordering::Less => start = middle + 1,
602            std::cmp::Ordering::Greater => end = middle,
603            std::cmp::Ordering::Equal => return Ok(()),
604        }
605    }
606    distinct.insert(start, (named, occurrence));
607    Ok(())
608}

610/// Whether a read predicate gained rows at the last publication: `old_rows <
611/// row_count` in this round's snapshot, as the certified test reads it. One unit
612/// per entry tested, and the resolution's unit; it stops at the first changed
613/// predicate.
614fn affected(
615    reads: &[Occurrence],
616    rule: &RuleIr,
617    support: &Support<'_>,
618    limits: &FormulaLimits,
619    counters: &mut Counters,
620) -> Result<bool, FormulaFailure> {
621    if reads.is_empty() {
622        return Ok(false);
623    }
624    let view = support
625        .components()
626        .ok_or_else(|| super::super::components::missing(rule.location))?;
627    for &read in reads {
628        counters.work(limits, rule.location)?;
629        let predicate = predicate(read, view, limits, counters, rule.location)?;
630        if support.old_rows(predicate) < support.row_count(predicate) {
631            return Ok(true);
632        }
633    }
634    Ok(false)
635}

650/// The reference the skip is compared against: a scoped, thread-local switch
651/// that traverses every rich producer in every round. It is restored after an
652/// unwind and confined to its thread, as the structural switch is.
653#[cfg(test)]
654pub(crate) mod testing {
655    use std::cell::Cell;

657    thread_local! {
658        static EVERY_ROUND: Cell<bool> = const { Cell::new(false) };
659    }

661    pub(super) fn every_round_traversed() -> bool {
662        EVERY_ROUND.get()
663    }

665    /// Run `action` with every rich producer traversed in every round.
666    pub(crate) fn with_every_round_traversed<T>(action: impl FnOnce() -> T) -> T {
667        struct Restore(bool);
668        impl Drop for Restore {
669            fn drop(&mut self) {
670                EVERY_ROUND.set(self.0);
671            }
672        }
673        let _restore = Restore(EVERY_ROUND.replace(true));
674        action()
675    }
