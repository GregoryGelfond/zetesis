1//! The certain predicates of the registered rules.
2//!
3//! [`crate::formula_support::certain`] states what a certain predicate is and
4//! what its supported atoms establish. This is the analysis: one pass
5//! classifies every rule and resolves what it derives and reads, a second
6//! removes predicates until none reads a removed one, and a third lists the
7//! assignment aggregate elements whose conditions hold certain atoms and
8//! decided literals only, in the rules whose every traversal reads those
9//! elements' relations complete: a deferred producer, a gated producer for
10//! the elements of its gate, and a constraint. A fourth lists the awaited
11//! default-negated atoms of certain predicates, which those traversals
12//! decide.
13//!
14//! A deferred producer may itself be definite through an exact assignment
15//! aggregate. The schedule traverses it once, after every predicate it reads
16//! reached its fixed point, so when those predicates are certain each of its
17//! elements is listed, every tuple is required, and the one value it proposes
18//! is the aggregate's value in every interpretation. Its heads are then
19//! supported exactly when they hold, as a definite rule's are.
20//!
21//! *Certain* is a property of a predicate. It is unrelated to
22//! `Producer::Certified`, the registration's name for a rule whose body
23//! support completion can extend row by row.

44/// Whether `aggregate` is a plain assignment: one whose elements each have a
45/// tuple key and a condition of positive atoms and decided literals. Its
46/// value is then the measure of the tuples its elements yield over complete
47/// relations, the same in every interpretation that holds their atoms. One
48/// unit per element and per condition literal.
49fn plain(
50    aggregate: &AggregateIr,
51    limits: &FormulaLimits,
52    counters: &mut Counters,
53    location: ProgramSite,
54) -> Result<bool, FormulaFailure> {
55    if aggregate.binding.is_none() {
56        return Ok(false);
57    }
58    // A binding is granted only to an unnegated aggregate whose one guard is
59    // the equality with its target (`Compiler::assignment_targets`), so no
60    // other guard reads the value.
61    debug_assert!(aggregate.negation == DefaultNegation::None && aggregate.guards.len() == 1);
62    for element in &aggregate.elements {
63        counters.work(limits, location)?;
64        if !matches!(element.key, AggregateKey::Tuple(_)) {
65            return Ok(false);
66        }
67        for literal in &element.condition {
68            counters.work(limits, location)?;
69            if positive(literal).is_none() && !decided(literal) {
70                return Ok(false);
71            }
72        }
73    }
74    Ok(true)
75}

77/// Whether `rule` has one ordinary head atom over positive atoms, literals
78/// its join decides, default-negated atoms every traversal reads over
79/// complete relations (`awaited`, by body position) and, when the component
80/// schedule traverses it once over complete relations (`deferred`), plain
81/// assignment aggregates. An awaited negated atom is decided, and a plain
82/// assignment exact, once the predicates they read prove certain. One unit
83/// per body literal, with an aggregate's own.
84fn definite(
85    rule: &RuleIr,
86    deferred: bool,
87    awaited: impl Fn(usize) -> bool,
88    limits: &FormulaLimits,
89    counters: &mut Counters,
90) -> Result<bool, FormulaFailure> {
91    if !matches!(rule.head, HeadIr::Normal(Some(_))) {
92        return Ok(false);
93    }
94    for (occurrence, literal) in rule.body.iter().enumerate() {
95        counters.work(limits, rule.location)?;
96        let admitted = match literal {
97            LiteralIr::Aggregate(aggregate) => {
98                deferred && plain(aggregate, limits, counters, rule.location)?
99            }
100            LiteralIr::Atom(DefaultNegation::Not | DefaultNegation::NotNot, _) => {
101                awaited(occurrence)
102            }
103            other => positive(other).is_some() || decided(other),
104        };
105        if !admitted {
106            return Ok(false);
107        }
108    }
109    Ok(true)
110}

112/// The atoms a definite rule reads: its body's flat atoms of either sign,
113/// then the positive atoms of each plain assignment's element conditions.
114fn atoms_read(rule: &RuleIr) -> impl Iterator<Item = Pattern> + '_ {
115    rule.body.iter().flat_map(|literal| {
116        let (own, elements): (Option<Pattern>, &[AggregateElementIr]) = match literal {
117            LiteralIr::Aggregate(aggregate) => (None, aggregate.elements.as_slice()),
118            LiteralIr::Atom(_, atom) => (Some(*atom), &[]),
119            other => (positive(other), &[]),
120        };
121        own.into_iter().chain(
122            elements
123                .iter()
124                .flat_map(|element| element.condition.iter().filter_map(positive)),
125        )
126    })
127}

129/// Resolves an atom pattern's predicate to its node in the dependency graph.
130pub(super) struct Resolver<'a, 'source, 'view> {
131    pub(super) nodes: &'a [Node<'source>],
132    pub(super) view: TemplateComponentsRef<'view>,
133    pub(super) limits: &'a FormulaLimits,
134}
135
136impl Resolver<'_, '_, '_> {
137    /// The node of `pattern`'s predicate, or `None` when the graph does not
138    /// name it. The pattern's read and the lookup charge their own units.
139    pub(super) fn node(
140        &self,
141        pattern: Pattern,
142        counters: &mut Counters,
143        location: ProgramSite,
144    ) -> Result<Option<usize>, FormulaFailure> {
145        let predicate = pattern
146            .get(self.view, self.limits, counters, location)?
147            .predicate();
148        lookup(
149            self.nodes.len(),
150            |index| self.nodes[index].signature,
151            predicate,
152            self.limits,
153            counters,
154            location,
155        )
156    }
157}

159/// Per predicate node whether it is still certain, and per rule the head a
160/// definite rule derives with the range of the nodes it reads.
161struct Classified {
162    certain: Vec<bool>,
163    heads: Vec<Option<usize>>,
164    ranges: Vec<Range<usize>>,
165    reads: Vec<usize>,
166}
167
168impl<'source> Registration<'source> {
169    /// The certain predicates of the registered rules, and the assignment
170    /// aggregate elements they make required. Scratch is admitted beside the
171    /// registration's own bytes and `external` and released before returning;
172    /// the returned value's bytes are its owner's to admit. The schedule
173    /// then lets go of each certified producer it held back for negated
174    /// atoms none of which is decided
175    /// ([`components::Schedule::release_undecided`]), so this runs before
176    /// the first round.
177    pub(in crate::formula_support) fn certainty(
178        &self,
179        prepared: &'source Prepared,
180        view: TemplateComponentsRef<'_>,
181        external: u128,
182        (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
183    ) -> Result<Certainty, FormulaFailure> {
184        let mut memory = Memory::new(0, external + self.bytes as u128, limits, counters, location);
185        memory.add(0)?;
186        let nodes = graph_nodes(prepared, &mut memory, limits, counters, location)?;
187        let resolver = Resolver {
188            nodes: &nodes,
189            view,
190            limits,
191        };
192        let mut classified = self.classify(&resolver, &mut memory, counters)?;
193        remove_readers(&mut classified, limits, counters, location)?;
194        let elements =
195            self.required_elements(&resolver, &classified.certain, &mut memory, counters)?;
196        let negations =
197            self.decided_negations(&resolver, &classified.certain, &mut memory, counters)?;
198        // A certified producer held back for atoms none of which is decided
199        // waits for nothing it can use.
200        if let Some(schedule) = &self.schedule {
201            schedule.release_undecided(&negations, limits, counters, location)?;
202        }
203        let mut predicates = Vec::new();
204        for (index, _) in classified
205            .certain
206            .iter()
207            .enumerate()
208            .filter(|&(_, &certain)| certain)
209        {
210            counters.work(limits, location)?;
211            let signature = nodes[index].signature;
212            memory.reserve(&mut predicates, 1)?;
213            memory.add(signature.name.as_str().len())?;
214            predicates.push(Signature {
215                sign: match signature.sign {
216                    themelios_program::symbol::Sign::Positive => zetesis_core::Sign::Positive,
217                    themelios_program::symbol::Sign::Negative => zetesis_core::Sign::Negative,
218                },
219                name: signature.name.as_str().into(),
220                arity: usize::try_from(signature.arity).map_err(|_| {
221                    super::invalid(zetesis_core::relation::Failure::Overflow, location)
222                })?,
223            });
224        }
225        // The sort and deduplication of the three lists.
226        counters.charge_work(
227            1 + predicates.len() as u128 + elements.len() as u128 + negations.len() as u128,
228            limits,
229            location,
230        )?;
231        Ok(Certainty::new(predicates, elements, negations))
232    }

234    /// Classify every rule and resolve what it derives and reads. A rule that
235    /// is not definite removes each predicate it can derive; a definite rule
236    /// that reads a predicate the graph does not name removes its head.
237    fn classify(
238        &self,
239        resolver: &Resolver<'_, '_, '_>,
240        memory: &mut Memory<'_>,
241        counters: &mut Counters,
242    ) -> Result<Classified, FormulaFailure> {
243        let limits = resolver.limits;
244        let mut certain = Vec::new();
245        memory.reserve(&mut certain, resolver.nodes.len())?;
246        certain.resize(resolver.nodes.len(), true);
247        let mut heads: Vec<Option<usize>> = Vec::new();
248        let mut ranges: Vec<Range<usize>> = Vec::new();
249        let mut reads: Vec<usize> = Vec::new();
250        memory.reserve(&mut heads, self.rules.len())?;
251        memory.reserve(&mut ranges, self.rules.len())?;
252        for (index, rule) in self.rules.iter().enumerate() {
253            counters.work(limits, rule.location)?;
254            let start = reads.len();
255            let definite = definite(
256                rule,
257                self.deferred(index),
258                |occurrence| self.awaits(index, occurrence),
259                limits,
260                counters,
261            )?;
262            let mut head = None;
263            for pattern in components::rule_heads(rule) {
264                let found = resolver.node(pattern, counters, rule.location)?;
265                if definite {
266                    head = found;
267                } else if let Some(found) = found {
268                    certain[found] = false;
269                }
270            }
271            if let Some(head) = head {
272                for pattern in atoms_read(rule) {
273                    counters.work(limits, rule.location)?;
274                    match resolver.node(pattern, counters, rule.location)? {
275                        Some(read) => {
276                            memory.reserve(&mut reads, 1)?;
277                            reads.push(read);
278                        }
279                        // A predicate the dependency graph does not name.
280                        None => certain[head] = false,
281                    }
282                }
283            }
284            heads.push(head);
285            ranges.push(start..reads.len());
286        }
287        Ok(Classified {
288            certain,
289            heads,
290            ranges,
291            reads,
292        })
293    }

295    /// Whether the component schedule traverses the producer at `index` once,
296    /// after every predicate it reads reached its fixed point. A gated
297    /// producer is not: it is traversed again as its own component grows.
298    fn deferred(&self, index: usize) -> bool {
299        self.schedule
300            .as_ref()
301            .is_some_and(|schedule| schedule.state(index).is_some() && !schedule.resumes(index))
302    }
303
304    /// Whether every traversal of the producer at `index` reads the relation
305    /// of the default-negated atom at body position `occurrence` complete.
306    fn awaits(&self, index: usize, occurrence: usize) -> bool {
307        !crate::formula_support::certain::negation_left_out()
308            && self
309                .schedule
310                .as_ref()
311                .is_some_and(|schedule| schedule.awaits(index, occurrence))
312    }
313
314    /// The awaited default-negated atoms whose predicate is certain, as
315    /// `(rule, occurrence)` coordinates in the schedule's order: every
316    /// traversal of the rule's producer reads the atom's relation complete,
317    /// and a supported atom of that relation holds in every answer set, an
318    /// unsupported one in none. One unit per awaited atom, with its
319    /// resolution.
320    fn decided_negations(
321        &self,
322        resolver: &Resolver<'_, '_, '_>,
323        certain: &[bool],
324        memory: &mut Memory<'_>,
325        counters: &mut Counters,
326    ) -> Result<Vec<(usize, usize)>, FormulaFailure> {
327        let mut negations = Vec::new();
328        let Some(schedule) = &self.schedule else {
329            return Ok(negations);
330        };
331        if crate::formula_support::certain::negation_left_out() {
332            return Ok(negations);
333        }
334        for &(index, occurrence) in schedule.awaited() {
335            let rule = &self.rules[index];
336            counters.work(resolver.limits, rule.location)?;
337            let LiteralIr::Atom(_, pattern) = &rule.body[occurrence] else {
338                unreachable!("an awaited occurrence is a negated flat atom");
339            };
340            if resolver
341                .node(*pattern, counters, rule.location)?
342                .is_some_and(|read| certain[read])
343            {
344                memory.reserve(&mut negations, 1)?;
345                negations.push((index, occurrence));
346            }
347        }
348        Ok(negations)
349    }

351    /// Whether the producer at `index` is gated, and waits for the element at
352    /// `position` of the assignment aggregate `aggregate`: every traversal of
353    /// the producer reads that element's condition over complete relations.
354    fn waits_for(&self, index: usize, aggregate: usize, position: usize) -> bool {
355        self.schedule.as_ref().is_some_and(|schedule| {
356            schedule.resumes(index) && schedule.waits_for(aggregate, position)
357        })
358    }
359
360    /// Whether the rule at `index` reads its relations complete: a deferred
361    /// producer, or a constraint, which proposes no head and is grounded only
362    /// after support completion.
363    fn reads_complete_relations(&self, index: usize) -> bool {
364        self.deferred(index) || self.producers[index] == Producer::Constraint
365    }

367    /// The assignment aggregate elements whose conditions hold certain atoms
368    /// and decided literals only, where they are read over complete
369    /// relations, as `(aggregate, element)` coordinates: every such element
370    /// of a rule that reads its relations complete, and of a gated producer
371    /// the elements its gate waits for. Any other element of a producer
372    /// traversed in every round is read from relations that may still grow,
373    /// and is not listed.
374    fn required_elements(
375        &self,
376        resolver: &Resolver<'_, '_, '_>,
377        certain: &[bool],
378        memory: &mut Memory<'_>,
379        counters: &mut Counters,
380    ) -> Result<Vec<(usize, usize)>, FormulaFailure> {
381        let limits = resolver.limits;
382        let mut elements = Vec::new();
383        for (index, rule) in self.rules.iter().enumerate() {
384            counters.work(limits, rule.location)?;
385            let complete = self.reads_complete_relations(index);
386            let gated = self
387                .schedule
388                .as_ref()
389                .is_some_and(|schedule| schedule.resumes(index));
390            if !complete && !gated {
391                continue;
392            }
393            for literal in &rule.body {
394                counters.work(limits, rule.location)?;
395                let LiteralIr::Aggregate(aggregate) = literal else {
396                    continue;
397                };
398                if aggregate.binding.is_none() {
399                    continue;
400                }
401                for (position, element) in aggregate.elements.iter().enumerate() {
402                    counters.work(limits, rule.location)?;
403                    if element.required || !matches!(element.key, AggregateKey::Tuple(_)) {
404                        continue;
405                    }
406                    if !complete && !self.waits_for(index, aggregate.id, position) {
407                        continue;
408                    }
409                    let mut holds = true;
410                    for literal in &element.condition {
411                        counters.work(limits, rule.location)?;
412                        holds = match positive(literal) {
413                            Some(pattern) => resolver
414                                .node(pattern, counters, rule.location)?
415                                .is_some_and(|read| certain[read]),
416                            None => decided(literal),
417                        };
418                        if !holds {
419                            break;
420                        }
421                    }
422                    if holds {
423                        memory.reserve(&mut elements, 1)?;
424                        elements.push((aggregate.id, position));
425                    }
426                }
427            }
428        }
429        Ok(elements)
430    }
431}

433/// Remove predicates until no remaining one reads a removed one. Each round
434/// removes at least one predicate or ends the removal, and charges one unit
435/// per rule and per node it reads: at most the predicates and one more times
436/// the rules and their reads, when each round removes one predicate of a
437/// chain written head first.
438fn remove_readers(
439    classified: &mut Classified,
440    limits: &FormulaLimits,
441    counters: &mut Counters,
442    location: ProgramSite,
443) -> Result<(), FormulaFailure> {
444    loop {
445        let mut changed = false;
446        for (head, range) in classified.heads.iter().zip(&classified.ranges) {
447            counters.charge_work(1 + range.len() as u128, limits, location)?;
448            let Some(head) = *head else {
449                continue;
450            };
451            if classified.certain[head]
452                && classified.reads[range.clone()]
453                    .iter()
454                    .any(|&read| !classified.certain[read])
455            {
456                classified.certain[head] = false;
457                changed = true;
458            }
459        }
460        if !changed {
461            return Ok(());
462        }
463    }
