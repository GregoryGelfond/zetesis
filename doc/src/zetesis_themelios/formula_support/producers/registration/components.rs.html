1//! The component schedule of rich producers.
2//!
3//! A rich producer is traversed in full whenever a predicate it reads gains
4//! rows. When every predicate it reads lies in a dependency component before
5//! the components of its heads, those reads stop changing once their
6//! components reach their fixed point, and a single traversal then derives
7//! every head the per-change traversals would. Such a producer is *deferred*:
8//! it waits for that fixed point and is traversed once, in the round in which
9//! the components it reads reach it, over that round's snapshot. Rich
10//! producers that read their own component, or a later one, keep the
11//! per-change traversal.
12//!
13//! Components come from the dependency graph's dependency-before-dependent
14//! order. A component's dependencies, for this schedule, are the components of
15//! every positive input of a certified producer of one of its predicates and of
16//! every predicate a rich producer of one of its predicates reads, of either
17//! sign. A component is *decidable* when each dependency precedes it or is
18//! itself, and every earlier dependency is decidable; a producer reading an
19//! undecidable component is never deferred.
20//!
21//! After a round's other producers have run, a component has reached its fixed
22//! point when it gained no atom this round, every earlier dependency has
23//! reached its fixed point, and every deferred producer of its predicates has
24//! been traversed. Its producers then ran over final inputs and derived nothing
25//! new, so no later round can derive anything for it. The test runs in
26//! component order, so one pass decides every component; deferred producers
27//! whose reads all passed are traversed, and the test repeats until none is
28//! ready. Each deferred producer is traversed exactly once. When a round
29//! derives nothing, every component not waiting on an untraversed deferred
30//! producer passes, and the deferred producer of the earliest head component
31//! among those still waiting reads only passed components, so completion never
32//! ends with a deferred producer untraversed.
33//!
34//! Per round the test costs one visit per pending atom, a binary search for each
35//! run of atoms of one predicate, and one visit per component, dependency and
36//! deferred read. It is skipped once every deferred producer was traversed.

60/// Which deferred producers wait for which components, and the scratch the
61/// per-round test fills.
62pub(super) struct Schedule<'source> {
63    nodes: Vec<Node<'source>>,
64    decidable: Vec<bool>,
65    dependencies: Vec<usize>,
66    dependency_ranges: Vec<Range<usize>>,
67    producers_of: Vec<usize>,
68    producer_ranges: Vec<Range<usize>>,
69    reads: Vec<usize>,
70    /// Per registered producer: its read components, when it is deferred.
71    deferred: Vec<Option<Range<usize>>>,
72    state: Vec<Cell<State>>,
73    changed: RefCell<Vec<bool>>,
74    reached: RefCell<Vec<bool>>,
75}

77/// Scratch for one preparation: (head, input) and (head, producer) pairs, per
78/// producer its first head and whether every predicate it names has a
79/// component, per component whether all of its producers' predicates do, and
80/// one producer's resolved components.
81#[derive(Default)]
82struct Scratch {
83    edges: Vec<(usize, usize)>,
84    owners: Vec<(usize, usize)>,
85    first_heads: Vec<Option<usize>>,
86    known_producers: Vec<bool>,
87    known: Vec<bool>,
88    inputs: Vec<usize>,
89    heads: Vec<usize>,
90}
91
92impl Scratch {
93    fn bytes(&self) -> usize {
94        (self.edges.capacity() + self.owners.capacity()) * size_of::<(usize, usize)>()
95            + self.first_heads.capacity() * size_of::<Option<usize>>()
96            + self.known_producers.capacity()
97            + self.known.capacity()
98            + (self.inputs.capacity() + self.heads.capacity()) * size_of::<usize>()
99    }
100}

114impl<'source> Schedule<'source> {
115    /// The schedule of `registration`'s rich producers, or `None` when none
116    /// can be deferred. Retained vectors are admitted through `memory`; the
117    /// scratch is released before returning.
118    pub(super) fn prepare(
119        registration: &Registration<'source>,
120        prepared: &'source Prepared,
121        view: TemplateComponentsRef<'_>,
122        memory: &mut Memory<'_>,
123        work: (&FormulaLimits, &mut Counters, ProgramSite),
124    ) -> Result<Option<Self>, FormulaFailure> {
125        // The test-only reference traverses every rich producer every round.
126        if super::every_round_traversed() {
127            return Ok(None);
128        }
129        let (limits, counters, location) = work;
130        let nodes = super::super::graph_nodes(prepared, memory, limits, counters, location)?;
131        let count = nodes
132            .iter()
133            .filter_map(|node| node.component)
134            .max()
135            .map_or(0, |last| last + 1);
136        let mut scratch = Scratch::default();
137        let inputs = (registration, nodes.as_slice(), view);
138        resolve_all(
139            inputs,
140            count,
141            &mut scratch,
142            memory,
143            (limits, counters, location),
144        )?;
145        let dependencies = dependencies(&mut scratch, count, memory, (limits, counters, location))?;
146        let (deferred, reads) = classify(
147            inputs,
148            &dependencies.decidable,
149            &mut scratch,
150            memory,
151            (limits, counters, location),
152        )?;
153        let (producers_of, producer_ranges) = if deferred.iter().any(Option::is_some) {
154            by_component(
155                &mut scratch.owners,
156                count,
157                memory,
158                (limits, counters, location),
159            )?
160        } else {
161            (Vec::new(), Vec::new())
162        };
163        memory.release(scratch.bytes());
164        drop(scratch);
165        if deferred.iter().all(Option::is_none) {
166            return Ok(None);
167        }
168        let mut state = Vec::new();
169        memory.reserve(&mut state, deferred.len())?;
170        state.extend(deferred.iter().map(|_| Cell::new(State::Waiting)));
171        let mut changed = Vec::new();
172        let mut reached = Vec::new();
173        memory.reserve(&mut changed, count)?;
174        memory.reserve(&mut reached, count)?;
175        changed.resize(count, false);
176        reached.resize(count, false);
177        Ok(Some(Self {
178            nodes,
179            decidable: dependencies.decidable,
180            dependencies: dependencies.list,
181            dependency_ranges: dependencies.ranges,
182            producers_of,
183            producer_ranges,
184            reads,
185            deferred,
186            state,
187            changed: RefCell::new(changed),
188            reached: RefCell::new(reached),
189        }))
190    }
191
192    /// The producer's state, when it is deferred.
193    pub(super) fn state(&self, index: usize) -> Option<State> {
194        self.deferred
195            .get(index)
196            .and_then(Option::as_ref)
197            .map(|_| self.state[index].get())
198    }
199
200    /// Whether a deferred producer has not yet been traversed.
201    pub(super) fn pending(&self) -> bool {
202        self.state
203            .iter()
204            .zip(&self.deferred)
205            .any(|(state, deferred)| deferred.is_some() && state.get() != State::Traversed)
206    }

208    /// Mark ready every waiting producer whose read components reached their
209    /// fixed point, given this round's pending atoms in `delta`. Returns
210    /// whether any producer became ready.
211    pub(super) fn mark_ready(
212        &self,
213        delta: &SupportAppend<'_>,
214        limits: &FormulaLimits,
215        counters: &mut Counters,
216        location: ProgramSite,
217    ) -> Result<bool, FormulaFailure> {
218        if !self.state.iter().any(|state| state.get() == State::Waiting) {
219            return Ok(false);
220        }
221        let mut changed = self.changed.borrow_mut();
222        changed.fill(false);
223        let mut previous: Option<PredicateRef<'_>> = None;
224        for atom in delta.atoms() {
225            counters.work(limits, location)?;
226            let atom_predicate = atom.predicate();
227            if let Some(previous) = previous
228                && previous.equals_ref_with(atom_predicate, || counters.work(limits, location))?
229            {
230                continue;
231            }
232            previous = Some(atom_predicate);
233            let found = lookup(
234                self.nodes.len(),
235                |index| self.nodes[index].signature,
236                atom_predicate,
237                limits,
238                counters,
239                location,
240            )?;
241            if let Some(component) = found.and_then(|index| self.nodes[index].component) {
242                changed[component] = true;
243            }
244        }
245        let mut reached = self.reached.borrow_mut();
246        for component in 0..reached.len() {
247            let dependencies = &self.dependencies[self.dependency_ranges[component].clone()];
248            let producers = &self.producers_of[self.producer_ranges[component].clone()];
249            counters.charge_work(
250                1 + dependencies.len() as u128 + producers.len() as u128,
251                limits,
252                location,
253            )?;
254            reached[component] = !changed[component]
255                && self.decidable[component]
256                && dependencies
257                    .iter()
258                    .all(|&other| other == component || reached[other])
259                && producers
260                    .iter()
261                    .all(|&producer| self.state[producer].get() == State::Traversed);
262        }
263        let mut any = false;
264        for (index, reads) in self.deferred.iter().enumerate() {
265            let Some(reads) = reads else {
266                continue;
267            };
268            counters.charge_work(1 + reads.len() as u128, limits, location)?;
269            if self.state[index].get() == State::Waiting
270                && self.reads[reads.clone()].iter().all(|&read| reached[read])
271            {
272                self.state[index].set(State::Ready);
273                any = true;
274            }
275        }
276        Ok(any)
277    }
278
279    /// The next ready producer, which the caller traverses now.
280    pub(super) fn take_ready(&self) -> Option<usize> {
281        let index = self
282            .state
283            .iter()
284            .position(|state| state.get() == State::Ready)?;
285        self.state[index].set(State::Traversed);
286        Some(index)
287    }
288}

290/// Resolve every producer: its (head, input) edges, its first head, and
291/// whether every predicate it names has a component.
292fn resolve_all(
293    (registration, nodes, view): (&Registration<'_>, &[Node<'_>], TemplateComponentsRef<'_>),
294    count: usize,
295    scratch: &mut Scratch,
296    memory: &mut Memory<'_>,
297    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
298) -> Result<(), FormulaFailure> {
299    let producers = registration.producers.len();
300    memory.reserve(&mut scratch.first_heads, producers)?;
301    memory.reserve(&mut scratch.known_producers, producers)?;
302    memory.reserve(&mut scratch.known, count)?;
303    scratch.known.resize(count, true);
304    for (index, producer) in registration.producers.iter().enumerate() {
305        counters.work(limits, location)?;
306        let rule = &registration.rules[index];
307        scratch.inputs.clear();
308        scratch.heads.clear();
309        let complete = resolve(
310            registration,
311            producer,
312            rule,
313            nodes,
314            view,
315            (&mut scratch.inputs, &mut scratch.heads, memory),
316            (limits, counters, rule.location),
317        )?;
318        memory.reserve(
319            &mut scratch.edges,
320            scratch.heads.len() * scratch.inputs.len(),
321        )?;
322        for &head in &scratch.heads {
323            scratch.known[head] &= complete;
324            scratch
325                .edges
326                .extend(scratch.inputs.iter().map(|&input| (head, input)));
327        }
328        scratch
329            .first_heads
330            .push(scratch.heads.iter().copied().min());
331        scratch.known_producers.push(complete);
332    }
333    Ok(())
334}

336/// Each component's dependencies, and whether its fixed point is decidable:
337/// every dependency precedes it or is itself, and each earlier one is.
338fn dependencies(
339    scratch: &mut Scratch,
340    count: usize,
341    memory: &mut Memory<'_>,
342    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
343) -> Result<Dependencies, FormulaFailure> {
344    counters.charge_work(scratch.edges.len() as u128 + 1, limits, location)?;
345    scratch.edges.sort_unstable();
346    scratch.edges.dedup();
347    let mut list = Vec::new();
348    let mut ranges = Vec::new();
349    let mut decidable = Vec::new();
350    memory.reserve(&mut list, scratch.edges.len())?;
351    memory.reserve(&mut ranges, count)?;
352    memory.reserve(&mut decidable, count)?;
353    let mut edges = scratch.edges.iter().peekable();
354    for (component, &known) in scratch.known.iter().enumerate() {
355        counters.work(limits, location)?;
356        let start = list.len();
357        let mut earlier = true;
358        while let Some(&(_, other)) = edges.next_if(|&&(head, _)| head == component) {
359            earlier &= other == component || (other < component && decidable[other]);
360            list.push(other);
361        }
362        ranges.push(start..list.len());
363        decidable.push(known && earlier);
364    }
365    Ok(Dependencies {
366        list,
367        ranges,
368        decidable,
369    })
370}

372/// The deferred producers: rich, with every read decidable and earlier than
373/// every head. Returns each producer's read range and the reads; records each
374/// deferred producer's (head, producer) pairs in the scratch.
375fn classify(
376    (registration, nodes, view): (&Registration<'_>, &[Node<'_>], TemplateComponentsRef<'_>),
377    decidable: &[bool],
378    scratch: &mut Scratch,
379    memory: &mut Memory<'_>,
380    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
381) -> Result<Deferred, FormulaFailure> {
382    let mut deferred = Vec::new();
383    let mut reads = Vec::new();
384    memory.reserve(&mut deferred, registration.producers.len())?;
385    for (index, producer) in registration.producers.iter().enumerate() {
386        counters.work(limits, location)?;
387        let Producer::Rich(_) = producer else {
388            deferred.push(None);
389            continue;
390        };
391        let rule = &registration.rules[index];
392        scratch.inputs.clear();
393        scratch.heads.clear();
394        resolve(
395            registration,
396            producer,
397            rule,
398            nodes,
399            view,
400            (&mut scratch.inputs, &mut scratch.heads, memory),
401            (limits, counters, rule.location),
402        )?;
403        let waits = scratch.known_producers[index]
404            && scratch.first_heads[index].is_some_and(|first| {
405                scratch
406                    .inputs
407                    .iter()
408                    .all(|&read| read < first && decidable[read])
409            });
410        if !waits {
411            deferred.push(None);
412            continue;
413        }
414        let start = reads.len();
415        memory.reserve(&mut reads, scratch.inputs.len())?;
416        reads.extend_from_slice(&scratch.inputs);
417        deferred.push(Some(start..reads.len()));
418        memory.reserve(&mut scratch.owners, scratch.heads.len())?;
419        scratch
420            .owners
421            .extend(scratch.heads.iter().map(|&head| (head, index)));
422    }
423    Ok((deferred, reads))
424}

426/// The deferred producers of each component, as ranges into one list.
427fn by_component(
428    owners: &mut [(usize, usize)],
429    count: usize,
430    memory: &mut Memory<'_>,
431    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
432) -> Result<(Vec<usize>, Vec<Range<usize>>), FormulaFailure> {
433    counters.charge_work(owners.len() as u128 + 1, limits, location)?;
434    owners.sort_unstable();
435    let mut list = Vec::new();
436    let mut ranges = Vec::new();
437    memory.reserve(&mut list, owners.len())?;
438    memory.reserve(&mut ranges, count)?;
439    let mut entries = owners.iter().peekable();
440    for component in 0..count {
441        counters.work(limits, location)?;
442        let start = list.len();
443        while let Some(&(_, producer)) = entries.next_if(|&&(head, _)| head == component) {
444            if list.last() != Some(&producer) || list.len() == start {
445                list.push(producer);
446            }
447        }
448        ranges.push(start..list.len());
449    }
450    Ok((list, ranges))
451}

453/// A producer's head components into `heads` and the components its
454/// derivations read into `inputs`: a certified producer's positive inputs, a
455/// rich producer's reads. Returns whether every predicate it names has a
456/// component.
457fn resolve(
458    registration: &Registration<'_>,
459    producer: &Producer,
460    rule: &RuleIr,
461    nodes: &[Node<'_>],
462    view: TemplateComponentsRef<'_>,
463    (inputs, heads, memory): (&mut Vec<usize>, &mut Vec<usize>, &mut Memory<'_>),
464    (limits, counters, location): (&FormulaLimits, &mut Counters, ProgramSite),
465) -> Result<bool, FormulaFailure> {
466    let component = |predicate: PredicateRef<'_>,
467                     counters: &mut Counters|
468     -> Result<Option<usize>, FormulaFailure> {
469        Ok(lookup(
470            nodes.len(),
471            |index| nodes[index].signature,
472            predicate,
473            limits,
474            counters,
475            location,
476        )?
477        .and_then(|index| nodes[index].component))
478    };
479    let mut complete = true;
480    let mut record = |found: Option<usize>,
481                      into: &mut Vec<usize>,
482                      memory: &mut Memory<'_>|
483     -> Result<(), FormulaFailure> {
484        match found {
485            Some(found) => {
486                memory.reserve(into, 1)?;
487                into.push(found);
488            }
489            None => complete = false,
490        }
491        Ok(())
492    };
493    match producer {
494        Producer::Constraint => return Ok(true),
495        Producer::Certified(occurrences) => {
496            for &occurrence in &registration.inputs[occurrences.clone()] {
497                let atom = match &rule.body[occurrence] {
498                    LiteralIr::Atom(_, atom) => *atom,
499                    LiteralIr::PatternAtom(pattern) => pattern.atom,
500                    _ => unreachable!("a certified input is a positive occurrence"),
501                };
502                let predicate = atom.get(view, limits, counters, location)?.predicate();
503                record(component(predicate, counters)?, inputs, memory)?;
504            }
505        }
506        Producer::Rich(reads) => {
507            for &read in &registration.reads[reads.clone()] {
508                let read = predicate(read, view, limits, counters, location)?;
509                record(component(read, counters)?, inputs, memory)?;
510            }
511        }
512    }
513    for head in rule_heads(rule) {
514        let predicate = head.get(view, limits, counters, location)?.predicate();
515        record(component(predicate, counters)?, heads, memory)?;
516    }
517    Ok(complete)
518}

520/// Every atom a rule's head can derive: the normal head, each disjunct, each
521/// conditional head and each choice element.
522fn rule_heads(
523    rule: &RuleIr,
524) -> impl Iterator<Item = crate::formula_support::components::Pattern> + '_ {
525    let normal = match &rule.head {
526        HeadIr::Normal(head) => *head,
527        _ => None,
528    };
529    let elements = match &rule.head {
530        HeadIr::Choice(choice) => choice.elements.as_slice(),
531        _ => &[],
532    };
533    normal
534        .into_iter()
535        .chain(
536            rule.head
537                .disjuncts()
538                .filter_map(|literal| literal.atom().copied()),
539        )
540        .chain(
541            elements
542                .iter()
543                .filter_map(|element| element.head.atom().copied()),
544        )
