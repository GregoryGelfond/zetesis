1//! Optional totality over complete positive source-column domains.
2//!
3//! Final emission, hybrid capture and admitted hybrid checks share this gate
4//! over the same completed source carrier. Each expression in a flat constraint
5//! is either a leaf or reads one source variable covered by an ordinary positive
6//! atom column. Successful evaluation on that entire
7//! column domain covers every complete rule binding, including rows a scalar
8//! equality later excludes. An arithmetic failure declines the optimization;
9//! the original complete traversal remains responsible for its diagnostics.
10//! Hybrid capture first performs this preparation with append-capable terms;
11//! the first frozen check repeats it against the immutable rule and full source
12//! columns. Successful maps remain immutable across later cursors, independently
13//! of candidate selection. Missing frozen values remain errors.
14

30impl<'a> Projections<'a> {
31    /// The caller retains Complete coverage for source-family evidence. A
32    /// successful result permits only ordinary post-admission row selection.
33    pub(in crate::formula_support) fn total_constraint(
34        &mut self,
35        rule: &'a RuleIr,
36        support: &Support<'_>,
37        evaluation: &mut Evaluation,
38        mut context: Context<'_, &mut Computation<'_, '_>>,
39    ) -> Result<bool, FormulaFailure> {
40        tick(&mut context)?;
41        context
42            .computation
43            .allowance(&self.lease, context.work.limits, context.work.location)?;
44        if !eligible(rule, support, &mut context)? {
45            return Ok(false);
46        }
47        self.prepare(
48            &rule.body,
49            Context::new(
50                &*context.computation,
51                context.work.limits,
52                context.work.counters,
53                context.work.location,
54            ),
55        )?;
56        let mut binding = Binding::new(
57            context.computation,
58            context.work.limits,
59            context.work.counters,
60            context.work.location,
61        )?;
62        binding.extend_scope(
63            rule.body_variables,
64            context.computation,
65            context.work.limits,
66            context.work.counters,
67            context.work.location,
68        )?;
69        for (index, literal) in rule.body.iter().enumerate() {
70            tick(&mut context)?;
71            match literal {
72                LiteralIr::Compare(left, _, right) => {
73                    for (side, expression) in [left, right].into_iter().enumerate() {
74                        if !self.total_expression(
75                            (Some((index, side)), expression),
76                            &rule.body,
77                            support,
78                            &mut binding,
79                            evaluation,
80                            &mut context,
81                        )? {
82                            return Ok(false);
83                        }
84                    }
85                }
86                LiteralIr::TupleCompare(left, _, right) => {
87                    for expression in left.iter().chain(right) {
88                        if !self.total_expression(
89                            (None, expression),
90                            &rule.body,
91                            support,
92                            &mut binding,
93                            evaluation,
94                            &mut context,
95                        )? {
96                            return Ok(false);
97                        }
98                    }
99                }
100                LiteralIr::Guard(guard) | LiteralIr::HeadGuard(guard) => {
101                    for expression in guard.expressions() {
102                        if !self.total_expression(
103                            (None, expression),
104                            &rule.body,
105                            support,
106                            &mut binding,
107                            evaluation,
108                            &mut context,
109                        )? {
110                            return Ok(false);
111                        }
112                    }
113                }
114                LiteralIr::Atom(..) => {}
115                _ => unreachable!("eligible flat constraint checked above"),
116            }
117        }
118        let Self::Local(values) = self else {
119            unreachable!("totality preparation owns its projection values");
120        };
121        for projection in &mut values.values {
122            tick(&mut context)?;
123            projection.covered = Some(projection.results.len());
124        }
125        Ok(true)
126    }

128    fn total_expression(
129        &mut self,
130        (source, expression): (Option<(usize, usize)>, &Expression),
131        literals: &[LiteralIr],
132        support: &Support<'_>,
133        binding: &mut Binding<'static>,
134        evaluation: &mut Evaluation,
135        context: &mut Context<'_, &mut Computation<'_, '_>>,
136    ) -> Result<bool, FormulaFailure> {
137        context.work.counters.charge_work(
138            expression.nodes.len() as u128,
139            context.work.limits,
140            context.work.location,
141        )?;
142        // Constructor admission has its own failure variants. Its operational
143        // totality is not established by this numeric expression certificate.
144        if expression
145            .nodes
146            .iter()
147            .any(|node| matches!(node, Operation::Constructor(_)))
148        {
149            return Ok(false);
150        }
151        if let [Operation::Constant(_)] = expression.nodes.as_slice() {
152            return admitted(evaluation.source_expression(
153                expression,
154                |slot| binding.key(slot, context.work.location),
155                context.computation,
156                context.work.limits,
157                context.work.counters,
158                context.work.location,
159            ));
160        }
161        let mut inputs = expression.inputs();
162        let Some(slot) = inputs.next() else {
163            return Ok(false);
164        };
165        if !inputs.all(|input| input == slot) {
166            return Ok(false);
167        }
168        let Some(domain) = domain(slot, literals, support, context)? else {
169            return Ok(false);
170        };
171        if matches!(expression.nodes.as_slice(), [Operation::Variable(_)]) {
172            // Complete source columns already contain admitted canonical terms.
173            // A direct variable leaf performs no partial arithmetic.
174            return Ok(true);
175        }
176        for posting in domain.rows.columns[domain.column].values() {
177            tick(context)?;
178            let position = *posting.first().expect("a distinct input has a source row");
179            let row = domain
180                .rows
181                .row(position)
182                .expect("retained source row position");
183            let value = row.value(domain.column).expect("checked source column");
184            let input = context
185                .computation
186                .read()
187                .term_key(value)
188                .map_err(|error| {
189                    crate::formula_binding::assignment(error.into(), context.work.location)
190                })?;
191            binding.set(
192                slot,
193                &input,
194                context.work.limits,
195                context.work.counters,
196                context.work.location,
197            )?;
198            let result = if let Some(source) = source {
199                self.value(source, expression, binding, evaluation, context)
200            } else {
201                evaluation.source_expression(
202                    expression,
203                    |slot| binding.key(slot, context.work.location),
204                    context.computation,
205                    context.work.limits,
206                    context.work.counters,
207                    context.work.location,
208                )
209            };
210            if !admitted(result)? {
211                return Ok(false);
212            }
213        }
214        Ok(true)
215    }
216}

218/// The finite column argument covers only a flat constraint over ordinary
219/// positive source rows. Empty relations need no speculative arithmetic.
220fn eligible(
221    rule: &RuleIr,
222    support: &Support<'_>,
223    context: &mut Context<'_, &mut Computation<'_, '_>>,
224) -> Result<bool, FormulaFailure> {
225    if !matches!(rule.head, HeadIr::Normal(None)) || rule.variables != rule.body_variables {
226        return Ok(false);
227    }
228    for literal in &rule.body {
229        tick(context)?;
230        match literal {
231            LiteralIr::Atom(DefaultNegation::None, atom) => {
232                let pattern = context.computation.static_pattern(
233                    *atom,
234                    context.work.limits,
235                    context.work.counters,
236                    context.work.location,
237                )?;
238                let rows = support.resolve(
239                    pattern,
240                    context.work.limits,
241                    context.work.counters,
242                    context.work.location,
243                )?;
244                if rows.is_none_or(|rows| rows.row_count() == 0) {
245                    return Ok(false);
246                }
247            }
248            LiteralIr::Atom(..)
249            | LiteralIr::Compare(..)
250            | LiteralIr::TupleCompare(..)
251            | LiteralIr::Guard(_)
252            | LiteralIr::HeadGuard(_) => {}
253            // No generated bindings, structural captures, projections or
254            // nested families are certified by a flat column domain.
255            _ => return Ok(false),
256        }
257    }
258    Ok(true)
259}

261/// A complete input domain may overapproximate actual complete bindings.
262/// Arithmetic failure there is therefore not a source diagnostic. Resource,
263/// owner, prefix and other typed refusals remain failures of this operation.
264fn admitted<T>(result: Result<T, FormulaFailure>) -> Result<bool, FormulaFailure> {
265    match result {
266        Ok(_) => Ok(true),
267        Err(FormulaFailure::Expansion(ExpansionFailure::Evaluation { .. })) => Ok(false),
268        Err(error) => Err(error),
269    }
270}

272fn domain<'source>(
273    slot: usize,
274    literals: &[LiteralIr],
275    support: &Support<'source>,
276    context: &mut Context<'_, &mut Computation<'_, '_>>,
277) -> Result<Option<Domain<'source>>, FormulaFailure> {
278    let mut selected: Option<Domain<'source>> = None;
279    for literal in literals {
280        tick(context)?;
281        let LiteralIr::Atom(DefaultNegation::None, atom) = literal else {
282            continue;
283        };
284        let pattern = context.computation.static_pattern(
285            *atom,
286            context.work.limits,
287            context.work.counters,
288            context.work.location,
289        )?;
290        let terms = pattern.terms();
291        for column in 0..terms.len() {
292            tick(context)?;
293            if !matches!(terms.at(column), Some(TemplateTerm::Variable(input)) if input == slot) {
294                continue;
295            }
296            let Some(rows) = support.resolve(
297                pattern,
298                context.work.limits,
299                context.work.counters,
300                context.work.location,
301            )?
302            else {
303                return Ok(None);
304            };
305            if rows.row_count() == 0 {
306                return Ok(None);
307            }
308            if selected.as_ref().is_none_or(|previous| {
309                rows.columns[column].len() < previous.rows.columns[previous.column].len()
310            }) {
311                selected = Some(Domain { rows, column });
312            }
313        }
314    }
315    Ok(selected)
316}
