1//! Optional totality over complete positive source-column domains.
2//!
3//! Original-family admission, final emission, hybrid capture and admitted hybrid
4//! checks use this gate over completed support. Possible-head production may
5//! use the same body proof within its exact immutable support snapshot; the
6//! proof covers complete current columns, not later rounds or family evidence. A rule
7//! can alternatively receive static totality from the completed all-producer
8//! analysis, which permits support-round prefix selection without a projection
9//! map. That separate certificate never broadens this snapshot's authority.
10//! Each expression
11//! in an eligible flat normal body is either a leaf, reads one source variable
12//! covered by an ordinary positive atom column, or has checked arithmetic over
13//! several numeric column bounds. Deterministic body assignments propagate
14//! checked hulls in dependency order before comparisons are inspected. Every
15//! assignment is checked, including unused targets; no expression is expanded.
16//! Head-only generation remains a later phase. A body certificate does not
17//! establish totality of those head expressions or their operational limits.
18//! Products, quotients and remainders extend sums and differences;
19//! the bounds establish every intermediate and exclude zero divisors.
20//! Successful evaluation on that entire
21//! column domain covers every complete rule binding, including rows a scalar
22//! equality later excludes. An arithmetic failure declines the optimization;
23//! the original complete traversal remains responsible for its diagnostics.
24//! Hybrid capture first performs this preparation with append-capable terms;
25//! the first frozen check repeats it against the immutable rule and full source
26//! columns. Successful maps remain immutable across later cursors, independently
27//! of candidate selection. Missing frozen values remain errors.

42/// Preparation serves either partial scalar families or total arithmetic whose
43/// certificate admits incremental source checking. This does not redefine which
44/// source families require a defined witness: that remains `family::partial`.
45pub(in crate::formula_support) fn needed(literals: &[LiteralIr]) -> bool {
46    let arithmetic = |expression: &Expression| expression.nodes.iter().any(numeric::arithmetic);
47    crate::formula_support::family::partial(literals)
48        || literals.iter().any(|literal| match literal {
49            LiteralIr::Compare(comparison) => {
50                arithmetic(comparison.left()) || arithmetic(comparison.right())
51            }
52            LiteralIr::TupleCompare(left, _, right) => left.iter().chain(right).any(arithmetic),
53            LiteralIr::Bind { .. } => true,
54            _ => false,
55        })
56}

58/// Whether a flat rule uses the source-column certificate's arithmetic grammar.
59/// This gate changes no binding schedule and supplies no numeric or definedness
60/// premise. Only successful preparation over the completed source does that.
61pub(crate) fn supports_column_totality(
62    rule: &RuleIr,
63    limits: &FormulaLimits,
64    counters: &mut Counters,
65) -> Result<bool, FormulaFailure> {
66    for literal in &rule.body {
67        counters.work(limits, rule.location)?;
68        let expressions = match literal {
69            LiteralIr::Compare(comparison) => std::slice::from_ref(comparison.left())
70                .iter()
71                .chain(std::slice::from_ref(comparison.right())),
72            LiteralIr::TupleCompare(left, _, right) => left.iter().chain(right),
73            LiteralIr::Bind { target, value } if *target < rule.body_variables => {
74                std::slice::from_ref(value).iter().chain([].iter())
75            }
76            LiteralIr::Atom(..) | LiteralIr::Bind { .. } => continue,
77            _ => return Ok(false),
78        };
79        for expression in expressions {
80            for node in &expression.nodes {
81                counters.work(limits, rule.location)?;
82                if !numeric::supports(node) {
83                    return Ok(false);
84                }
85            }
86        }
87    }
88    Ok(true)
89}

96impl<'a> Projections<'a> {
97    /// Use full, candidate-independent columns of one immutable source snapshot.
98    /// Success establishes scalar totality for complete positive bindings within
99    /// that snapshot. This includes every scalar body assignment but excludes
100    /// the generated head suffix. A growing round may select only its current
101    /// snapshot. Completed-family admission additionally needs the full key and
102    /// head obligations in `total_family`, for every normalized sibling. This
103    /// proof supplies no defined witness and no guarantee about partial prefixes.
104    pub(in crate::formula_support) fn total_body(
105        &mut self,
106        rule: &'a RuleIr,
107        support: &Support<'_>,
108        evaluation: &mut Evaluation,
109        mut context: Context<'_, &mut Computation<'_, '_>>,
110    ) -> Result<bool, FormulaFailure> {
111        tick(&mut context)?;
112        context
113            .computation
114            .allowance(&self.lease, context.work.limits, context.work.location)?;
115        if !eligible(rule, support, &mut context)? {
116            return Ok(false);
117        }
118        self.prepare(
119            &rule.body,
120            Context::new(
121                &*context.computation,
122                context.work.limits,
123                context.work.counters,
124                context.work.location,
125            ),
126        )?;
127        let mut binding = Binding::new(
128            context.computation,
129            context.work.limits,
130            context.work.counters,
131            context.work.location,
132        )?;
133        binding.extend_scope(
134            rule.body_variables,
135            context.computation,
136            context.work.limits,
137            context.work.counters,
138            context.work.location,
139        )?;
140        let mut numeric = numeric::Preparation::new(rule, support);
141        if !numeric.bindings(&mut context)? {
142            return Ok(false);
143        }
144        for (index, literal) in rule.body.iter().enumerate() {
145            tick(&mut context)?;
146            match literal {
147                LiteralIr::Compare(comparison) => {
148                    let ([left, right], _) = comparison.operands();
149                    for (side, expression) in [left, right].into_iter().enumerate() {
150                        if !self.total_expression(
151                            (Some((index, side)), expression),
152                            &mut numeric,
153                            &mut binding,
154                            evaluation,
155                            &mut context,
156                        )? {
157                            return Ok(false);
158                        }
159                    }
160                }
161                LiteralIr::TupleCompare(left, _, right) => {
162                    for expression in left.iter().chain(right) {
163                        if !self.total_expression(
164                            (None, expression),
165                            &mut numeric,
166                            &mut binding,
167                            evaluation,
168                            &mut context,
169                        )? {
170                            return Ok(false);
171                        }
172                    }
173                }
174                LiteralIr::Guard(guard) | LiteralIr::HeadGuard(guard) => {
175                    for expression in guard.expressions() {
176                        if !self.total_expression(
177                            (None, expression),
178                            &mut numeric,
179                            &mut binding,
180                            evaluation,
181                            &mut context,
182                        )? {
183                            return Ok(false);
184                        }
185                    }
186                }
187                // Body bindings were certified in dependency order above.
188                // Head-only bindings retain their original later evaluation.
189                LiteralIr::Atom(..) | LiteralIr::Bind { .. } => {}
190                _ => unreachable!("eligible flat normal body checked above"),
191            }
192        }
193        let Self::Local(values) = self else {
194            unreachable!("totality preparation owns its projection values");
195        };
196        for projection in &mut values.values {
197            tick(&mut context)?;
198            // A generated slot has a checked hull, not an enumerated source
199            // column. Its empty memo must never become an empty-domain proof.
200            if numeric.domain(projection.slot, &mut context)?.is_some() {
201                projection.covered = Some(projection.results.len());
202            }
203        }
204        Ok(true)
205    }
206
207    /// Prove that a flat family's arithmetic evidence scan has nothing to
208    /// discover. The caller separately establishes key coverage, including
209    /// each generated input. Only constructors and leaves are admitted in the
210    /// head suffix; head arithmetic conservatively retains the original scan.
211    /// Constructor allocation, expanded size and depth are still checked by
212    /// the original head operation for each reached instance. Omitting this
213    /// duplicate scan changes work, never those construction limits.
214    pub(in crate::formula_support) fn total_family(
215        &mut self,
216        rule: &'a RuleIr,
217        support: &Support<'_>,
218        evaluation: &mut Evaluation,
219        mut context: Context<'_, &mut Computation<'_, '_>>,
220    ) -> Result<bool, FormulaFailure> {
221        if !self.total_body(
222            rule,
223            support,
224            evaluation,
225            Context::new(
226                &mut *context.computation,
227                context.work.limits,
228                context.work.counters,
229                context.work.location,
230            ),
231        )? {
232            return Ok(false);
233        }
234        for literal in &rule.body {
235            tick(&mut context)?;
236            let LiteralIr::Bind { target, value } = literal else {
237                continue;
238            };
239            if *target < rule.body_variables {
240                continue;
241            }
242            for operation in &value.nodes {
243                tick(&mut context)?;
244                if !matches!(
245                    operation,
246                    Operation::Constant(_) | Operation::Variable(_) | Operation::Constructor(_)
247                ) {
248                    return Ok(false);
249                }
250            }
251        }
252        Ok(true)
253    }

255    fn total_expression(
256        &mut self,
257        (source, expression): (Option<(usize, usize)>, &Expression),
258        numeric: &mut numeric::Preparation<'_, '_>,
259        binding: &mut Binding<'static>,
260        evaluation: &mut Evaluation,
261        context: &mut Context<'_, &mut Computation<'_, '_>>,
262    ) -> Result<bool, FormulaFailure> {
263        context.work.counters.charge_work(
264            expression.nodes.len() as u128,
265            context.work.limits,
266            context.work.location,
267        )?;
268        // Constructor admission has its own failure variants. Its operational
269        // totality is not established by this numeric expression certificate.
270        if expression
271            .nodes
272            .iter()
273            .any(|node| matches!(node, Operation::Constructor(_)))
274        {
275            return Ok(false);
276        }
277        if let [Operation::Constant(_)] = expression.nodes.as_slice() {
278            return admitted(evaluation.source_expression(
279                expression,
280                |slot| binding.key(slot, context.work.location),
281                context.computation,
282                context.work.limits,
283                context.work.counters,
284                context.work.location,
285            ));
286        }
287        let mut inputs = expression.inputs();
288        let Some(slot) = inputs.next() else {
289            return Ok(false);
290        };
291        if !inputs.all(|input| input == slot) {
292            return numeric.total(expression, context);
293        }
294        let Some(domain) = numeric.domain(slot, context)? else {
295            // A prior scalar assignment supplies a checked hull without a
296            // physical source column or an enumerable projection map.
297            return numeric.total(expression, context);
298        };
299        if matches!(expression.nodes.as_slice(), [Operation::Variable(_)]) {
300            // Complete source columns already contain admitted canonical terms.
301            // A direct variable leaf performs no partial arithmetic.
302            return Ok(true);
303        }
304        let Postings::Indexed(postings) = domain.rows.postings(domain.column) else {
305            unreachable!("a domain column keeps postings");
306        };
307        for posting in postings.values() {
308            tick(context)?;
309            let position = *posting.first().expect("a distinct input has a source row");
310            let row = domain
311                .rows
312                .row(position)
313                .expect("retained source row position");
314            let value = row.value(domain.column).expect("checked source column");
315            let input = context
316                .computation
317                .read()
318                .term_key(value)
319                .map_err(|error| {
320                    crate::formula_binding::assignment(error.into(), context.work.location)
321                })?;
322            binding.set(
323                slot,
324                &input,
325                context.work.limits,
326                context.work.counters,
327                context.work.location,
328            )?;
329            let result = if let Some(source) = source {
330                self.value(source, expression, binding, evaluation, context)
331            } else {
332                evaluation.source_expression(
333                    expression,
334                    |slot| binding.key(slot, context.work.location),
335                    context.computation,
336                    context.work.limits,
337                    context.work.counters,
338                    context.work.location,
339                )
340            };
341            if !admitted(result)? {
342                return Ok(false);
343            }
344        }
345        Ok(true)
346    }
347}

349/// The finite column argument covers only a flat normal body over ordinary
350/// positive source rows. Empty relations need no speculative arithmetic.
351fn eligible(
352    rule: &RuleIr,
353    support: &Support<'_>,
354    context: &mut Context<'_, &mut Computation<'_, '_>>,
355) -> Result<bool, FormulaFailure> {
356    if !matches!(rule.head, HeadIr::Normal(_)) {
357        return Ok(false);
358    }
359    for literal in &rule.body {
360        tick(context)?;
361        match literal {
362            LiteralIr::Atom(DefaultNegation::None, atom) => {
363                let pattern = context.computation.static_pattern(
364                    *atom,
365                    context.work.limits,
366                    context.work.counters,
367                    context.work.location,
368                )?;
369                let rows = support.resolve(
370                    pattern,
371                    context.work.limits,
372                    context.work.counters,
373                    context.work.location,
374                )?;
375                if rows.is_none_or(|rows| rows.row_count() == 0) {
376                    return Ok(false);
377                }
378            }
379            LiteralIr::Atom(..)
380            | LiteralIr::Compare(..)
381            | LiteralIr::TupleCompare(..)
382            | LiteralIr::Guard(_)
383            | LiteralIr::HeadGuard(_)
384            | LiteralIr::Bind { .. } => {}
385            // Range/aggregate generators, structural captures, projections and
386            // nested families need their separate coverage arguments.
387            _ => return Ok(false),
388        }
389    }
390    Ok(true)
391}

393/// A complete input domain may overapproximate actual complete bindings.
394/// Arithmetic failure there is therefore not a source diagnostic. Resource,
395/// owner, prefix and other typed refusals remain failures of this operation.
396fn admitted<T>(result: Result<T, FormulaFailure>) -> Result<bool, FormulaFailure> {
397    match result {
398        Ok(_) => Ok(true),
399        Err(FormulaFailure::Expansion(ExpansionFailure::Evaluation { .. })) => Ok(false),
400        Err(error) => Err(error),
401    }
402}

404fn domain<'source>(
405    slot: usize,
406    literals: &[LiteralIr],
407    support: &Support<'source>,
408    context: &mut Context<'_, &mut Computation<'_, '_>>,
409) -> Result<Option<Domain<'source>>, FormulaFailure> {
410    let mut selected: Option<Domain<'source>> = None;
411    for literal in literals {
412        tick(context)?;
413        let LiteralIr::Atom(DefaultNegation::None, atom) = literal else {
414            continue;
415        };
416        let pattern = context.computation.static_pattern(
417            *atom,
418            context.work.limits,
419            context.work.counters,
420            context.work.location,
421        )?;
422        let terms = pattern.terms();
423        for column in 0..terms.len() {
424            tick(context)?;
425            if !matches!(terms.at(column), Some(TemplateTerm::Variable(input)) if input == slot) {
426                continue;
427            }
428            let Some(rows) = support.resolve(
429                pattern,
430                context.work.limits,
431                context.work.counters,
432                context.work.location,
433            )?
434            else {
435                return Ok(None);
436            };
437            if rows.row_count() == 0 {
438                return Ok(None);
439            }
440            // A column without postings has no recorded distinct values: it is
441            // declined, never read as an empty domain.
442            let Postings::Indexed(values) = rows.postings(column) else {
443                continue;
444            };
445            if selected.as_ref().is_none_or(|previous| {
446                let Postings::Indexed(known) = previous.rows.postings(previous.column) else {
447                    unreachable!("a selected domain column keeps postings");
448                };
449                values.len() < known.len()
450            }) {
451                selected = Some(Domain { rows, column });
452            }
453        }
454    }
455    Ok(selected)
456}

458/// Prove scalar intermediates on the original range and full positive columns.
459/// The caller authenticates the one generated slot and flat positive profile.
460/// Unlike complete-column projection preparation, this populates no expression
461/// map and grants neither table coverage nor original-family admission.
462pub(in crate::formula_support) fn range_body(
463    rule: &RuleIr,
464    support: &Support<'_>,
465    target: usize,
466    bounds: (i32, i32),
467    divisor: Option<usize>,
468    mut context: Context<'_, &mut Computation<'_, '_>>,
469) -> Result<bool, FormulaFailure> {
470    let mut numeric = numeric::Preparation::new(rule, support);
471    numeric.seed_range(target, bounds, &mut context)?;
472    if let Some(slot) = divisor
473        && !numeric.positive_domain(slot, &mut context)?
474    {
475        return Ok(false);
476    }
477    for literal in &rule.body {
478        tick(&mut context)?;
479        if let LiteralIr::Compare(comparison) = literal {
480            for expression in [comparison.left(), comparison.right()] {
481                context.work.counters.charge_work(
482                    expression.nodes.len() as u128,
483                    context.work.limits,
484                    context.work.location,
485                )?;
486                if !numeric.total(expression, &mut context)? {
487                    return Ok(false);
488                }
489            }
490        }
491    }
492    Ok(true)
493}
