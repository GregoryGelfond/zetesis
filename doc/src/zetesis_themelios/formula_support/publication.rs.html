1//! Final views share one sealed source vocabulary and retain only occurrence maps.
2
3use crate::ProgramSite;
4use zetesis_core::AtomCatalog;
5
6use super::{CompletedCatalog, Counters, SourceSelection, StorageLease};
7use crate::{FormulaFailure, FormulaLimits, FormulaResource};
8
9/// Keep earlier output maps charged while subsequent views are constructed.
10/// This preparation lease ends when all finished outputs leave source admission;
11/// it neither mutates their shared payload nor becomes a checker's workspace.
12pub(crate) struct Publication<'a> {
13    source: &'a mut CompletedCatalog,
14    lease: StorageLease,
15}
16
17impl<'a> Publication<'a> {
18    /// Source identity and workspace ownership are independent. Authenticate
19    /// both receipts before using the shared total to admit any replacement.
20    pub(crate) fn check_lease(
21        &self,
22        lease: &StorageLease,
23        counters: &Counters,
24        location: ProgramSite,
25    ) -> Result<(), FormulaFailure> {
26        if !counters.accounting.workspace.owns(&self.lease)
27            || !counters.accounting.workspace.owns(lease)
28        {
29            return Err(crate::formula_binding::assignment(
30                zetesis_core::catalog::AssignmentError::Read(
31                    zetesis_core::catalog::ReadError::ForeignCatalog,
32                ),
33                location,
34            ));
35        }
36        Ok(())
37    }
38
39    fn current_bytes(
40        &self,
41        counters: &Counters,
42        location: ProgramSite,
43    ) -> Result<u128, FormulaFailure> {
44        self.check_lease(&self.lease, counters, location)?;
45        Ok(self.source.catalog.bytes(location)? as u128
46            + counters.accounting.workspace.bytes() as u128)
47    }
48
49    /// Canonical/support storage outside this account's complete workspace.
50    pub(crate) fn source_bytes(&self, location: ProgramSite) -> Result<usize, FormulaFailure> {
51        self.source.catalog.bytes(location)
52    }
53
54    pub(crate) fn remaining_bytes(
55        &self,
56        limits: &FormulaLimits,
57        counters: &Counters,
58        location: ProgramSite,
59    ) -> Result<usize, FormulaFailure> {
60        let bytes = self.current_bytes(counters, location)?;
61        crate::formula::ceiling(
62            FormulaResource::SupportBytes,
63            bytes,
64            limits.max_support_bytes as u128,
65            location,
66        )?;
67        Ok(limits.max_support_bytes - usize::try_from(bytes).expect("admitted usize ceiling"))
68    }
69
70    /// Record a new envelope while its input header and transferred buffer are
71    /// still retained. The constructor must have admitted this overlap first.
72    pub(crate) fn observe_peak(
73        &self,
74        extra: u128,
75        limits: &FormulaLimits,
76        counters: &Counters,
77        location: ProgramSite,
78    ) -> Result<(), FormulaFailure> {
79        let bytes = self.current_bytes(counters, location)? + extra;
80        counters.record(crate::grounding_observer::Event::SupportPeakBytes(bytes));
81        crate::formula::ceiling(
82            FormulaResource::SupportBytes,
83            bytes,
84            limits.max_support_bytes as u128,
85            location,
86        )
87    }
88
89    /// Keep already admitted output storage live until all source outputs finish.
90    /// Transferring its receipt neither copies payload nor invents growth overlap.
91    pub(crate) fn retain(
92        &mut self,
93        lease: StorageLease,
94        limits: &FormulaLimits,
95        counters: &Counters,
96        location: ProgramSite,
97    ) -> Result<(), FormulaFailure> {
98        self.check_lease(&lease, counters, location)?;
99        self.remaining_bytes(limits, counters, location)?;
100        lease.transfer_to(&mut self.lease, location)
101    }
102
103    pub(crate) fn new(
104        source: &'a mut CompletedCatalog,
105        counters: &Counters,
106        location: ProgramSite,
107    ) -> Result<Self, FormulaFailure> {
108        let mut lease = counters.accounting.workspace.lease();
109        lease.observe(size_of::<Self>(), location)?;
110        Ok(Self { source, lease })
111    }
112
113    /// Transfer only retained output receipts. This publication coordinator no
114    /// longer exists after the move; its caller admits its own fixed envelope.
115    pub(crate) fn into_lease(
116        mut self,
117        location: ProgramSite,
118    ) -> Result<StorageLease, FormulaFailure> {
119        self.lease
120            .observe(self.lease.bytes() - size_of::<Self>(), location)?;
121        Ok(self.lease)
122    }
123
124    /// Supplied occurrence order is preserved exactly. A selected view cannot
125    /// change support membership; its logical population has its own count bound.
126    pub(crate) fn atoms(
127        &mut self,
128        selection: SourceSelection,
129        limits: &FormulaLimits,
130        counters: &mut Counters,
131        location: ProgramSite,
132    ) -> Result<AtomCatalog, FormulaFailure> {
133        self.check_lease(selection.storage_lease(), counters, location)?;
134        let catalog = self.source.catalog.publish_selection(
135            selection.scope(),
136            selection.positions(),
137            counters.accounting.workspace.bytes(),
138            limits,
139            counters,
140            location,
141        )?;
142        let bytes = self.lease.bytes() as u128 + catalog.publication_bytes();
143        let bytes = usize::try_from(bytes).map_err(|_| FormulaFailure::Limit {
144            resource: FormulaResource::SupportBytes,
145            observed: bytes,
146            limit: limits.max_support_bytes as u128,
147            location,
148        })?;
149        self.lease.observe(bytes, location)?;
150        // The publication operation admitted this map while the selection was
151        // still live. Drop that selection only after recording its replacement.
152        drop(selection);
153        Ok(catalog)
154    }
155}
