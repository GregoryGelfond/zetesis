28/// Query state is separate from both the catalog and its immutable row views.
29/// A source builder or independent hybrid checker owns this workspace. Table
30/// indices are immutable after publication; every query owns its independent
31/// mask and admits work through its caller's accounting and control.

39}

87    pub(super) fn completed(
88        relations: &'source Relations<'source>,
89        strategy: JoinStrategy,
90        limits: &FormulaLimits,
91        counters: &Counters,
92        location: ProgramSite,
93    ) -> Result<Self, FormulaFailure> {
94        let bytes = relations
95            .bytes
96            .checked_add(size_of::<Self>())
97            .ok_or_else(|| allocation(Cause::Overflow, location))?;
98        let live = bytes
99            .checked_add(counters.accounting.workspace.bytes())
100            .ok_or_else(|| allocation(Cause::Overflow, location))?;
101        ceiling(
102            FormulaResource::SupportBytes,
103            live as u128,
104            limits.max_support_bytes as u128,
105            location,
106        )?;
107        counters.record(Event::SupportPeakBytes(live as u128));
108        Ok(Self {
109            relations,
110            workspace: counters.accounting.workspace.clone(),
111            tables: OnceCell::new(),
112            table_strategy: strategy == JoinStrategy::Table,
113            live: Cell::new(bytes),
114            entries: Cell::new(relations.entries),
115        })
116    }

225    fn indexed_failure(&self, error: FormulaFailure) -> FormulaFailure {
226        match error {
227            FormulaFailure::Limit {
228                resource: FormulaResource::SupportBytes,
229                observed,
230                limit,
231                location,
232            } => {
233                let outer = (self.live_bytes() - self.relations.current_bytes()) as u128;
234                FormulaFailure::Limit {
235                    resource: FormulaResource::SupportBytes,
236                    observed: observed + outer,
237                    limit: limit + outer,
238                    location,
239                }
240            }
241            other => other,
242        }
243    }

285    /// Select necessary finite domains with the same table and mask workspace
286    /// used by ordinary table joins, including when ordinary joins are indexed.
287    /// The caller establishes coverage before excluding source values. A finite
288    /// restriction names one unbound variable present in this flat pattern;
289    /// ordinary bound variables and constants retain their singleton domains.
290    /// Ranges are borrowed for this call only; returned rows keep source IDs.
291    pub(super) fn select_domains_at<'value>(
292        &self,
293        rows: Option<&'source RelationRows<'source>>,
294        pattern: PositivePattern<'value>,
295        values: BindingView<'value>,
296        finite: FiniteDomains<'value>,
297        work: GroundingWork<'_>,
298    ) -> Result<Option<Rows<'_, 'source>>, FormulaFailure> {
299        let GroundingWork {
300            limits,
301            counters,
302            location,
303        } = work;
304        let PositivePattern::Flat(pattern) = pattern else {
305            counters.record(Event::TableInapplicableProbe);
306            return Ok(None);
307        };
308        let Some(relation) = rows.map(|rows| &rows.relation) else {
309            counters.record(Event::TableInapplicableProbe);
310            return Ok(None);
311        };
312        Self::admit(self.live_bytes(), limits, counters, location)?;
313        let mut scratch = Scratch::new(self, limits, counters, location)?;
314        let bound = bind_domains(pattern, values, finite, &mut scratch, counters)?;
315        let workspace = self.tables.get_or_init(|| TableWorkspace {
316            indices: RefCell::new(Vec::new()),
317        });
318        let mut tables = workspace.indices.borrow_mut();
319        let found = find_table(&tables, pattern, &bound.scope, limits, counters, location)?;
320        let index = if let Some(index) = found {
321            counters.record(Event::TableReuse);
322            index
323        } else {
324            self.reserve_tables(&mut tables, limits, counters, location)?;
325            let outer = self.live_bytes() - relation.storage().retained_bytes;
326            let table_limits = self.table_limits(limits, counters, outer, location)?;
327            let base_work = counters.accounting.work;
328            let prepared = Table::prepare_with(relation, &bound.scope, table_limits, |amount| {
329                counters.charge_work(amount as u128, limits, location)
330            });
331            let table = self.result(
332                prepared,
333                true,
334                outer,
335                base_work,
336                GroundingWork::new(limits, counters, location),
337            )?;
338            let retained = table.statistics().retained_bytes - size_of::<Table<'_, '_>>();
339            self.live.set(self.live.get() + retained);
340            self.entries
341                .set(self.entries.get() + table.support_entries());
342            counters.record(Event::TableIndexBytes(retained));
343            let index = tables.len();
344            tables.push(table);
345            counters.record(Event::TablePreparation);
346            index
347        };
348        let table = &tables[index];
349        let outer = self.live_bytes()
350            - relation.storage().retained_bytes
351            - table.statistics().retained_bytes
352            + ROW_LEASE_BYTES;
353        let mut table_limits = self.table_limits(limits, counters, outer, location)?;
354        // The existing table already owns these admitted entries; selection
355        // checks the input's count, rather than adding entries to the cache.
356        table_limits.max_entries = table.support_entries();
357        counters.record(Event::JoinProbe);
358        counters.record(Event::TableProbe);
359        let base_work = counters.accounting.work;
360        let selected = table.select_with(&bound.domains, table_limits, |amount| {
361            counters.charge_work(amount as u128, limits, location)
362        });
363        let selection = self.result(
364            selected,
365            false,
366            outer,
367            base_work,
368            GroundingWork::new(limits, counters, location),
369        )?;
370        let bytes = selection.statistics().retained_bytes + ROW_LEASE_BYTES;
371        self.live.set(self.live.get() + bytes);
372        Ok(Some(Rows {
373            selection,
374            live: &self.live,
375            bytes,
376        }))
377    }

380        &self,
381        limits: &FormulaLimits,
382        counters: &Counters,

396    fn result<T: Receipt>(
397        &self,
398        result: Result<T, table::MeteredFailure<FormulaFailure>>,
399        preparing: bool,
400        outer: usize,
401        base_work: u64,
402        work: GroundingWork<'_>,
403    ) -> Result<T, FormulaFailure> {
404        let GroundingWork {
405            limits,
406            counters,
407            location,
408        } = work;
409        let (work, peak) = match &result {
410            Ok(value) => (value.receipt().work, value.receipt().peak_bytes),
411            Err(error) => (error.work, error.peak_bytes),
412        };
413        // The primitive admitted each operation through these counters before
414        // execution. Preserve its completed receipt even when caller admission
415        // refused; neither charge twice nor poll again over the original error.
416        counters.record(if preparing {
417            Event::TablePrepareWork(work)
418        } else {
419            Event::TableQueryWork(work)
420        });
421        counters.record(Event::SupportPeakBytes(outer as u128 + peak as u128));
422        result.map_err(|error| {
423            let cause = match error.cause {
424                table::MeteredCause::Stopped(error) => return error,
425                table::MeteredCause::Table(cause) => cause,
426            };
427            let error = table::Failure {
428                cause,
429                work: error.work,
430                peak_bytes: error.peak_bytes,
431            };
432            let mapped = match &error.cause {
433                Cause::Limit {
434                    resource: Resource::Work,
435                    observed,
436                    ..
437                } => Some((
438                    FormulaResource::Work,
439                    u128::from(base_work) + observed,
440                    u128::from(limits.max_work),
441                )),
442                Cause::Limit {
443                    resource: Resource::Bytes,
444                    observed,
445                    ..
446                } => Some((
447                    FormulaResource::SupportBytes,
448                    outer as u128 + observed,
449                    limits.max_support_bytes as u128,
450                )),
451                Cause::Limit {
452                    resource: Resource::Entries,
453                    observed,
454                    ..
455                } if preparing => Some((
456                    FormulaResource::SupportIndexEntries,
457                    self.entries.get() as u128 + observed,
458                    limits.max_support_index_entries as u128,
459                )),
460                _ => None,
461            };
462            mapped.map_or(
463                FormulaFailure::SupportTable { error, location },
464                |(resource, observed, limit)| FormulaFailure::Limit {
465                    resource,
466                    observed,
467                    limit,
468                    location,
469                },
470            )
471        })
472    }

476        limits: &FormulaLimits,
477        counters: &Counters,

490    fn reserve_tables(
491        &self,
492        tables: &mut Vec<Table<'source, 'source>>,
493        limits: &FormulaLimits,
494        counters: &mut Counters,
495        location: ProgramSite,
496    ) -> Result<(), FormulaFailure> {
497        if tables.len() < tables.capacity() {
498            return Ok(());
499        }
500        let old = tables.capacity() * size_of::<Table<'_, '_>>();
501        let desired = tables
502            .len()
503            .saturating_add(1)
504            .max(tables.capacity().saturating_mul(2));
505        let next = desired
506            .checked_mul(size_of::<Table<'_, '_>>())
507            .ok_or_else(|| allocation(Cause::Overflow, location))?;
508        let peak = self
509            .live_bytes()
510            .checked_add(next)
511            .ok_or_else(|| allocation(Cause::Overflow, location))?;
512        Self::admit(peak, limits, counters, location)?;
513        counters.charge_work(tables.len() as u128, limits, location)?;
514        tables
515            .try_reserve_exact(desired - tables.len())
516            .map_err(|_| allocation(Cause::Allocation, location))?;
517        let actual = tables.capacity() * size_of::<Table<'_, '_>>();
518        let peak = self
519            .live_bytes()
520            .checked_add(actual)
521            .ok_or_else(|| allocation(Cause::Overflow, location))?;
522        self.live.set(self.live.get() - old + actual);
523        counters.record(Event::SupportPeakBytes(peak as u128));
524        Self::admit(peak, limits, counters, location)
525    }
526}

528/// Borrowed necessary domains over source variable slots. One value buffer
529/// holds every range; an empty range excludes all values of that variable.
530/// The caller charges the buffers and retains their canonical value owners.
531#[derive(Clone, Copy, Default)]
532pub(super) struct FiniteDomains<'value> {
533    pub(super) values: &'value [TermRef<'value>],
534    pub(super) variables: &'value [(usize, Range<usize>)],
535}
536
537impl FiniteDomains<'_> {
538    fn validate(
539        &self,
540        pattern: PatternRef<'_>,
541        values: BindingView<'_>,
542        work: &mut GroundingWork<'_>,
543    ) -> Result<(), FormulaFailure> {
544        for (index, (variable, range)) in self.variables.iter().enumerate() {
545            work.counters.work(work.limits, work.location)?;
546            if *variable >= values.len() {
547                return Err(FormulaFailure::UnsafeVariable {
548                    variable: *variable,
549                    location: work.location,
550                });
551            }
552            if values.get(*variable).is_some() || self.values.get(range.clone()).is_none() {
553                return Err(allocation(Cause::Domains, work.location));
554            }
555            for (earlier, _) in &self.variables[..index] {
556                work.counters.work(work.limits, work.location)?;
557                if earlier == variable {
558                    return Err(allocation(Cause::Domains, work.location));
559                }
560            }
561            let terms = pattern.terms();
562            let mut present = false;
563            for column in 0..terms.len() {
564                work.counters.work(work.limits, work.location)?;
565                present |= terms.at(column) == Some(TemplateTerm::Variable(*variable));
566            }
567            if !present {
568                return Err(allocation(Cause::Domains, work.location));
569            }
570        }
571        Ok(())
572    }
573}
574
575impl<'value> FiniteDomains<'value> {
576    fn domain(
577        &self,
578        variable: usize,
579        work: &mut GroundingWork<'_>,
580    ) -> Result<Option<Domain<'value>>, FormulaFailure> {
581        for (slot, range) in self.variables {
582            work.counters.work(work.limits, work.location)?;
583            if *slot == variable {
584                let values = self
585                    .values
586                    .get(range.clone())
587                    .expect("validated finite domain");
588                return Ok(Some(Domain::Finite(values.into())));
589            }
590        }
591        Ok(None)
592    }
593}

600/// Canonical table labels are distinct from source variable slots. Each
601/// constant owns one label; repeated source slots share their first label.
602/// The descriptors borrow whole values only for this call. None means an
603/// unbound valid slot, whereas an out-of-range source slot is a typed failure.
604fn bind_domains<'value>(
605    pattern: PatternRef<'value>,
606    values: BindingView<'value>,
607    finite: FiniteDomains<'value>,
608    scratch: &mut Scratch<'_, '_>,
609    counters: &mut Counters,
610) -> Result<BoundDomains<'value>, FormulaFailure> {
611    let limits = scratch.limits;
612    let location = scratch.location;
613    finite.validate(
614        pattern,
615        values,
616        &mut GroundingWork::new(limits, counters, location),
617    )?;
618    let mut scope = Vec::new();
619    let mut domains = Vec::new();
620    scratch.reserve(&mut scope, pattern.terms().len(), counters)?;
621    scratch.reserve(&mut domains, pattern.terms().len(), counters)?;
622    let terms = pattern.terms();
623    for column in 0..terms.len() {
624        counters.work(limits, location)?;
625        let term = terms.at(column).expect("checked pattern arity");
626        let mut alias = None;
627        if let TemplateTerm::Variable(variable) = term {
628            for (previous, prior_scope) in scope.iter().enumerate().take(column) {
629                counters.work(limits, location)?;
630                let other = terms.at(previous).expect("checked pattern prefix");
631                if other == TemplateTerm::Variable(variable) {
632                    alias = Some(*prior_scope);
633                    break;
634                }
635            }
636        }
637        let variable = alias.unwrap_or(domains.len());
638        scope.push(variable);
639        if alias.is_none() {
640            let value = match term {
641                TemplateTerm::Constant(value) => Some(value),
642                TemplateTerm::Variable(variable) => {
643                    if variable >= values.len() {
644                        return Err(FormulaFailure::UnsafeVariable { variable, location });
645                    }
646                    values.get(variable)
647                }
648            };
649            let finite = if let TemplateTerm::Variable(variable) = term {
650                finite.domain(
651                    variable,
652                    &mut GroundingWork::new(limits, counters, location),
653                )?
654            } else {
655                None
656            };
657            domains.push(
658                finite.unwrap_or_else(|| value.map_or(Domain::Unrestricted, Domain::Singleton)),
659            );
660        }
661    }
662    Ok(BoundDomains { scope, domains })
663}

665fn find_table(
666    tables: &[Table<'_, '_>],
667    pattern: PatternRef<'_>,
668    scope: &[usize],
669    limits: &FormulaLimits,
670    counters: &mut Counters,
671    location: ProgramSite,
672) -> Result<Option<usize>, FormulaFailure> {
673    let mut found = None;
674    for (index, table) in tables.iter().enumerate() {
675        counters.charge_work(
676            1 + pattern.predicate().name().len() as u128 + scope.len() as u128,
677            limits,
678            location,
679        )?;
680        if table.relation().predicate() == pattern.predicate() && table.scope() == scope {
681            found = Some(index);
682            break;
683        }
684    }
685    Ok(found)
686}

729struct Scratch<'a, 'source> {
730    support: &'a Support<'source>,
731    limits: &'a FormulaLimits,
732    location: ProgramSite,
733    bytes: usize,
734}
735impl<'a, 'source> Scratch<'a, 'source> {
736    fn new(
737        support: &'a Support<'source>,
738        limits: &'a FormulaLimits,
739        counters: &Counters,
740        location: ProgramSite,
741    ) -> Result<Self, FormulaFailure> {
742        let bytes = 2 * size_of::<Vec<usize>>() + size_of::<Self>();
743        let total = support
744            .live
745            .get()
746            .checked_add(bytes)
747            .ok_or_else(|| allocation(Cause::Overflow, location))?;
748        Support::admit(
749            support.bytes_with_append(total, location)?,
750            limits,
751            counters,
752            location,
753        )?;
754        support.live.set(total);
755        Ok(Self {
756            support,
757            limits,
758            location,
759            bytes,
760        })
761    }
762    fn reserve<T>(
763        &mut self,
764        values: &mut Vec<T>,
765        count: usize,
766        counters: &mut Counters,
767    ) -> Result<(), FormulaFailure> {
768        let bytes = count
769            .checked_mul(size_of::<T>())
770            .ok_or_else(|| allocation(Cause::Overflow, self.location))?;
771        let proposed = self
772            .support
773            .live
774            .get()
775            .checked_add(bytes)
776            .ok_or_else(|| allocation(Cause::Overflow, self.location))?;
777        Support::admit(
778            self.support.bytes_with_append(proposed, self.location)?,
779            self.limits,
780            counters,
781            self.location,
782        )?;
783        values
784            .try_reserve_exact(count)
785            .map_err(|_| allocation(Cause::Allocation, self.location))?;
786        let actual = values
787            .capacity()
788            .checked_mul(size_of::<T>())
789            .ok_or_else(|| allocation(Cause::Overflow, self.location))?;
790        let owned = self
791            .bytes
792            .checked_add(actual)
793            .ok_or_else(|| allocation(Cause::Overflow, self.location))?;
794        let total = self
795            .support
796            .live
797            .get()
798            .checked_add(actual)
799            .ok_or_else(|| allocation(Cause::Overflow, self.location))?;
800        self.bytes = owned;
801        self.support.live.set(total);
802        counters.record(Event::SupportPeakBytes(
803            total as u128
804                + self.support.append_bytes() as u128
805                + self.support.workspace.bytes() as u128,
806        ));
807        Support::admit(
808            self.support.bytes_with_append(total, self.location)?,
809            self.limits,
810            counters,
811            self.location,
812        )
813    }
814}
815impl Drop for Scratch<'_, '_> {
816    fn drop(&mut self) {
817        self.support.live.set(self.support.live.get() - self.bytes);
818    }
819}

821fn allocation(cause: Cause, location: ProgramSite) -> FormulaFailure {
822    FormulaFailure::SupportTable {
823        error: table::Failure {
824            cause,
825            work: 0,
826            peak_bytes: 0,
827        },
828        location,
829    }
