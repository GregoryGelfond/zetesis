1//! Necessary multiples over one original closed range, after family admission.
2//!
3//! The full original interval and complete positive columns establish every
4//! scalar intermediate before a cursor may omit a value. No projection table,
5//! family evidence, support-round permission or incremental certificate follows.
6//! The scalar equality remains in the ordinary filter. Only positive flat body
7//! atoms are eligible: rejected rows can name no undiscovered negative atom.
8//!
9//! Preparation reuses the numeric totality workspace: O(variables + longest
10//! expression) leased scratch, plus charged distinct-column scans. The retained
11//! descriptor is inline in Join and Cursor; iteration uses constant space and
12//! the original ascending order. Every original range width is admitted before
13//! selection, and all emitted terms use the ordinary canonical number operation.
14

35/// Exact literal identity and divisor readiness belong to this rule's cursor.
36/// Construction remains private to successful full-domain preparation below.
37#[derive(Clone, Copy)]
38pub(crate) struct Prepared<'a> {
39    literal: &'a LiteralIr,
40    divisor: Divisor,
41}

43struct Candidate<'a> {

45    target: usize,

49fn tick<C>(context: &mut Context<'_, C>) -> Result<(), FormulaFailure> {
50    context
51        .work
52        .counters
53        .work(context.work.limits, context.work.location)
54}

56impl<'a> Join<'a, '_> {
57    /// Opt in only after original-family validation, immediately before ordinary
58    /// rule emission. The completed support, exact rule and full arithmetic hull
59    /// are authenticated here; no broad join/coverage flag changes on success.
60    pub(crate) fn select_range_multiples(
61        &mut self,
62        rule: &'a RuleIr,
63        computation: &mut Computation<'_, '_>,
64        limits: &FormulaLimits,
65        counters: &mut Counters,
66    ) -> Result<bool, FormulaFailure> {
67        #[cfg(test)]
68        if tests::disabled() {
69            return Ok(false);
70        }
71        if !self.generated || self.source_evidence || self.delta.is_some() || self.pending.is_some()
72        {
73            return Ok(false);
74        }
75        if !std::ptr::eq(self.literals.get(), rule.body.as_slice()) {
76            return Err(FormulaFailure::SupportRelation {
77                error: zetesis_core::relation::Failure::Owner,
78                location: rule.location,
79            });
80        }
81        let Some(candidate) = candidate(
82            rule,
83            &mut Context::new(&*computation, limits, counters, rule.location),
84        )?
85        else {
86            return Ok(false);
87        };
88        if !keys::covered_with_range(
89            rule,
90            &self.plan,
91            Some(candidate.target),
92            &mut Context::new(&*computation, limits, counters, rule.location),
93        )? {
94            return Ok(false);
95        }
96        let divisor = match candidate.prepared.divisor {
97            Divisor::Constant(_) => None,
98            Divisor::Variable(slot) => Some(slot),
99        };
100        if !projections::range_body(
101            rule,
102            self.support,
103            candidate.target,
104            candidate.bounds,
105            divisor,
106            Context::new(computation, limits, counters, rule.location),
107        )? {
108            return Ok(false);
109        }
110        // Cancellation or refusal before this final charged boundary publishes
111        // no selection. Temporary numeric buffers have already been released.
112        counters.work(limits, rule.location)?;
113        self.range_selection = Some(candidate.prepared);
114        #[cfg(test)]
115        tests::selected();
116        Ok(true)
117    }
118}

120fn candidate<'a>(
121    rule: &'a RuleIr,
122    context: &mut Context<'_, &Computation<'_, '_>>,
123) -> Result<Option<Candidate<'a>>, FormulaFailure> {
124    if !matches!(rule.head, HeadIr::Normal(_)) || rule.variables != rule.body_variables {
125        return Ok(None);
126    }
127    let mut range = None;
128    for literal in &rule.body {
129        tick(context)?;
130        match literal {
131            LiteralIr::Range {
132                target,
133                lower,
134                upper,
135                binder: true,
136            } if range.is_none() => {
137                let ([Operation::Constant(lower)], [Operation::Constant(upper)]) =
138                    (lower.nodes.as_slice(), upper.nodes.as_slice())
139                else {
140                    return Ok(None);
141                };
142                let (Some(lower), Some(upper)) =
143                    (number(*lower, context)?, number(*upper, context)?)
144                else {
145                    return Ok(None);
146                };
147                if lower <= 0 || upper < lower {
148                    return Ok(None);
149                }
150                range = Some((literal, *target, (lower, upper)));
151            }
152            LiteralIr::Atom(DefaultNegation::None, _) | LiteralIr::Compare(..) => {}
153            _ => return Ok(None),
154        }
155    }
156    let Some((literal, target, bounds)) = range else {
157        return Ok(None);
158    };
159    for comparison in &rule.body {
160        tick(context)?;
161        let LiteralIr::Compare(comparison) = comparison else {
162            continue;
163        };
164        let ([left, right], relation) = comparison.operands();
165        if relation != Relation::Eq {
166            continue;
167        }
168        for (expression, zero) in [(left, right), (right, left)] {
169            if let Some((dividend, divisor)) = remainder(expression, zero, context)?
170                && dividend == target
171                // A constant-one equality cannot remove any range value. Keep
172                // looking for a selective remainder before allocating evidence.
173                && !matches!(divisor, Divisor::Constant(1))
174                && !matches!(divisor, Divisor::Variable(slot) if slot == target)
175            {
176                return Ok(Some(Candidate {
177                    prepared: Prepared { literal, divisor },
178                    target,
179                    bounds,
180                }));
181            }
182        }
183    }
184    Ok(None)
185}

187impl Prepared<'_> {
188    /// The original endpoint evaluation and width admission precede this call.
189    /// A constant-one divisor keeps the existing cursor. Variable divisors are
190    /// re-read on every relational restart; no row's value survives backtracking.
191    pub(crate) fn factors(
192        self,
193        literal: &LiteralIr,
194        bounds: &RangeInclusive<i32>,
195        binding: &Binding<'_>,
196        mut context: Context<'_, &Computation<'_, '_>>,
197    ) -> Result<Option<(RangeInclusive<i64>, i32)>, FormulaFailure> {
198        tick(&mut context)?;
199        if !std::ptr::eq(self.literal, literal) {
200            return Ok(None);
201        }
202        let divisor = match self.divisor {
203            Divisor::Constant(value) => value,
204            Divisor::Variable(slot) => {
205                tick(&mut context)?;
206                let value =
207                    binding.read(slot, context.computation.read(), context.work.location)?;
208                let ValueNodeRef::Number(value) = value.descriptor() else {
209                    return Ok(None);
210                };
211                value
212            }
213        };
214        if divisor <= 1 {
215            return Ok(None);
216        }
217        #[cfg(test)]
218        tests::stepped();
219        Ok(Some((
220            positive_factors((*bounds.start(), *bounds.end()), divisor),
221            divisor,
222        )))
223    }
224}
