1//! Checked in-place sorting of private metadata slices.
2//!
3//! Each comparison admits a permit before reading its cells; the comparator
4//! admits any further payload navigation. Each swap admits its own permit.
5//! Refusal preserves the population but may leave a partial permutation.
6//! Heapsort allocates nothing and uses constant scratch and O(n log n)
7//! comparisons. Callers retain the slice's existing capacity receipt.
8//! Successful sortedness requires a consistent total comparator; permutation
9//! preservation does not. Private partially sorted metadata is not publication.
10
11use std::cmp::Ordering;
12
13use super::GroundingWork;
14use crate::FormulaFailure;
15
16pub(crate) fn by<T>(
17    values: &mut [T],
18    mut work: GroundingWork<'_>,
19    mut compare: impl FnMut(&T, &T, &mut GroundingWork<'_>) -> Result<Ordering, FormulaFailure>,
20) -> Result<(), FormulaFailure> {
21    let count = values.len();
22    for root in (0..count / 2).rev() {
23        sift(values, root, count, &mut work, &mut compare)?;
24    }
25    for end in (1..count).rev() {
26        work.counters.work(work.limits, work.location)?;
27        values.swap(0, end);
28        sift(values, 0, end, &mut work, &mut compare)?;
29    }
30    Ok(())
31}
32
33fn sift<T>(
34    values: &mut [T],
35    mut root: usize,
36    end: usize,
37    work: &mut GroundingWork<'_>,
38    compare: &mut impl FnMut(&T, &T, &mut GroundingWork<'_>) -> Result<Ordering, FormulaFailure>,
39) -> Result<(), FormulaFailure> {
40    while root < end / 2 {
41        // This guard bounds root * 2 + 1 below end, including at usize's limit.
42        let mut child = root * 2 + 1;
43        if child + 1 < end {
44            work.counters.work(work.limits, work.location)?;
45            if compare(&values[child], &values[child + 1], work)?.is_lt() {
46                child += 1;
47            }
48        }
49        work.counters.work(work.limits, work.location)?;
50        if !compare(&values[root], &values[child], work)?.is_lt() {
51            break;
52        }
53        work.counters.work(work.limits, work.location)?;
54        values.swap(root, child);
55        root = child;
56    }
57    Ok(())
58}
