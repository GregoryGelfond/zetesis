1//! Necessary whole-argument equalities for fully bound structural patterns.
2//!
3//! Reverse preorder resolves exact existing constructor identities. It admits
4//! no term and writes no incoming slot. A missing identity cannot occur in this
5//! support snapshot; partial or anonymous shapes keep the ordinary matcher.

25/// Only caller-owned ID metadata is retained between probes. Each component's
26/// existing lease accounts its header, capacity and replacement overlap.
27pub(super) struct Scratch<'a> {
28    stack: Binding<'static>,
29    children: Buffer<usize>,
30    arguments: Binding<'static>,
31    prepared: prepared::Prepared<'a>,
32}
33
34impl<'a> Scratch<'a> {
35    pub(super) const fn leased_header() -> usize {
36        2 * size_of::<Binding<'static>>()
37            + size_of::<Buffer<usize>>()
38            + prepared::Prepared::leased_header()
39    }
40
41    fn new(
42        patterns: &[super::PatternOccurrence<'a>],
43        context: &mut Context<'_, &Computation<'_, '_>>,
44    ) -> Result<Self, FormulaFailure> {
45        let prepared = prepared::Prepared::new(patterns, context)?;
46        let Context { computation, work } = context;
47        Ok(Self {
48            stack: Binding::new(computation, work.limits, work.counters, work.location)?,
49            children: Buffer::new(computation, work.limits, work.counters, work.location)?,
50            arguments: Binding::new(computation, work.limits, work.counters, work.location)?,
51            prepared,
52        })
53    }
54
55    fn clear(&mut self) {
56        // These provisional query IDs carry no binding or publication state.
57        self.stack.discard_suffix(0);
58        self.children.clear();
59        self.arguments.discard_suffix(0);
60    }
61
62    fn prepare(
63        &mut self,
64        pattern: Pattern<'_>,
65        incoming: &Binding<'_>,
66        context: &mut Context<'_, &mut Computation<'_, '_>>,
67    ) -> Result<bool, FormulaFailure> {
68        self.clear();
69        {
70            let Context { computation, work } = context;
71            let terms = pattern.atom().terms();
72            self.arguments.extend_scope(
73                terms.len(),
74                computation,
75                work.limits,
76                work.counters,
77                work.location,
78            )?;
79            for column in 0..terms.len() {
80                work.counters.work(work.limits, work.location)?;
81                match terms.at(column).expect("admitted pattern column") {
82                    TemplateTerm::Constant(value) => self.arguments.set_term_with(
83                        column,
84                        value,
85                        computation.read(),
86                        work.location,
87                        || work.counters.work(work.limits, work.location),
88                    )?,
89                    TemplateTerm::Variable(slot) if incoming.is_bound(slot, work.location)? => {
90                        self.arguments.copy_slot(column, incoming, slot, work)?;
91                    }
92                    TemplateTerm::Variable(_) => {}
93                }
94            }
95        }
96        for argument in pattern.arguments() {
97            if !eligible(argument, pattern, incoming, &mut context.work)? {
98                continue;
99            }
100            let key = if let Some(key) =
101                self.prepared
102                    .key(pattern, argument, incoming, &mut context.work)?
103            {
104                Some(key)
105            } else {
106                let key = self.resolve(argument, incoming, context)?;
107                if let Some(key) = &key {
108                    self.prepared
109                        .remember(pattern, argument, incoming, key, &mut context.work)?;
110                }
111                key
112            };
113            let Some(key) = key else {
114                return Ok(false);
115            };
116            self.arguments.set(
117                argument.position,
118                &key,
119                context.work.limits,
120                context.work.counters,
121                context.work.location,
122            )?;
123        }
124        Ok(true)
125    }
126
127    fn resolve(
128        &mut self,
129        argument: &ArgumentPattern,
130        incoming: &Binding<'_>,
131        context: &mut Context<'_, &mut Computation<'_, '_>>,
132    ) -> Result<Option<TermKey>, FormulaFailure> {
133        self.stack.discard_suffix(0);
134        for node in argument.nodes.iter().rev() {
135            let Context { computation, work } = context;
136            work.counters.work(work.limits, work.location)?;
137            let key = match node {
138                PatternNode::Slot(slot) => incoming.key(*slot, work.location)?,
139                PatternNode::Constant(scalar) => {
140                    computation.static_key(*scalar, work.limits, work.counters, work.location)?
141                }
142                PatternNode::Constructor(constructor) => {
143                    let descriptor = computation.static_constructor(
144                        *constructor,
145                        work.limits,
146                        work.counters,
147                        work.location,
148                    )?;
149                    let (ValueNodeRef::Function { arity, .. } | ValueNodeRef::Tuple { arity }) =
150                        descriptor
151                    else {
152                        unreachable!("admitted structural constructor")
153                    };
154                    let start = self
155                        .stack
156                        .len()
157                        .checked_sub(arity)
158                        .expect("admitted preorder supplies constructor children");
159                    self.children.clear();
160                    for child in (start..self.stack.len()).rev() {
161                        self.children.push(
162                            child,
163                            computation,
164                            work.limits,
165                            work.counters,
166                            work.location,
167                        )?;
168                    }
169                    let key = computation.find_constructed(
170                        descriptor,
171                        self.stack.slots(),
172                        self.children.slice(),
173                        GroundingWork::new(work.limits, work.counters, work.location),
174                    )?;
175                    let Some(key) = key else {
176                        return Ok(None);
177                    };
178                    self.stack
179                        .truncate(start, work.limits, work.counters, work.location)?;
180                    key
181                }
182                PatternNode::Wildcard => unreachable!("only determined arguments are resolved"),
183            };
184            let next = self.stack.len();
185            let end = next.checked_add(1).ok_or_else(|| {
186                crate::formula_binding::assignment(
187                    zetesis_core::catalog::AssignmentError::Storage(
188                        zetesis_core::catalog::Error::Overflow,
189                    ),
190                    work.location,
191                )
192            })?;
193            self.stack
194                .extend_scope(end, computation, work.limits, work.counters, work.location)?;
195            self.stack
196                .set(next, &key, work.limits, work.counters, work.location)?;
197        }
198        debug_assert_eq!(self.stack.len(), 1);
199        context
200            .work
201            .counters
202            .work(context.work.limits, context.work.location)?;
203        self.stack.key(0, context.work.location).map(Some)
204    }
205}

207fn eligible(
208    argument: &ArgumentPattern,
209    pattern: Pattern<'_>,
210    incoming: &Binding<'_>,
211    work: &mut GroundingWork<'_>,
212) -> Result<bool, FormulaFailure> {
213    work.counters.work(work.limits, work.location)?;
214    match pattern
215        .atom()
216        .terms()
217        .at(argument.position)
218        .expect("admitted argument column")
219    {
220        TemplateTerm::Constant(_) => return Ok(false),
221        TemplateTerm::Variable(slot) if incoming.is_bound(slot, work.location)? => {
222            return Ok(false);
223        }
224        TemplateTerm::Variable(_) => {}
225    }
226    for node in &argument.nodes {
227        work.counters.work(work.limits, work.location)?;
228        match node {
229            PatternNode::Wildcard => return Ok(false),
230            PatternNode::Slot(slot) if !incoming.is_bound(*slot, work.location)? => {
231                return Ok(false);
232            }
233            _ => {}
234        }
235    }
236    Ok(true)
237}

239impl<'source> Join<'_, 'source> {
240    pub(super) fn structural_selection(
241        &mut self,
242        pattern: PositivePattern<'_>,
243        source: Option<&'source super::relations::RelationRows<'source>>,
244        mut context: Context<'_, &mut Computation<'_, '_>>,
245    ) -> Result<Selection<'source>, FormulaFailure> {
246        #[cfg(test)]
247        if !tests::enabled() {
248            return Ok(Selection::Unchanged);
249        }
250        let PositivePattern::Structural(pattern) = pattern else {
251            return Ok(Selection::Unchanged);
252        };
253        // No constructor query is needed to prove an absent relation empty.
254        if source.is_none() {
255            return Ok(Selection::Unchanged);
256        }
257        let mut determined = false;
258        for argument in pattern.arguments() {
259            if eligible(argument, pattern, &self.values, &mut context.work)? {
260                determined = true;
261                break;
262            }
263        }
264        if !determined {
265            return Ok(Selection::Unchanged);
266        }
267        #[cfg(test)]
268        tests::record();
269        if self.structural_query.is_none() {
270            self.structural_query = Some(Scratch::new(
271                &self.plan.patterns,
272                &mut Context::new(
273                    &*context.computation,
274                    context.work.limits,
275                    context.work.counters,
276                    context.work.location,
277                ),
278            )?);
279            self.lease
280                .observe(self.storage_bytes(), context.work.location)?;
281        }
282        let scratch = self
283            .structural_query
284            .as_mut()
285            .expect("prepared structural query");
286        let result = (|| {
287            if !scratch.prepare(pattern, &self.values, &mut context)? {
288                return Ok(Selection::Posting(Some(&[])));
289            }
290            let arguments = scratch.arguments.view(
291                context.computation.read(),
292                context.work.limits,
293                context.work.counters,
294                context.work.location,
295            )?;
296            self.support
297                .probe_arguments_at(
298                    source,
299                    pattern.atom(),
300                    arguments,
301                    context.work.limits,
302                    context.work.counters,
303                    context.work.location,
304                )
305                .map(Selection::Posting)
306        })();
307        scratch.clear();
308        result
309    }
