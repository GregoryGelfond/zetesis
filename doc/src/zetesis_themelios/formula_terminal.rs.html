2//!
3//! The partition certificate excludes every semantic read of a deferred head.

5//! presence remains distinct from truth: reconstruction reads selected model
6//! rows, never the completed possible-support relation.

28/// Result of source materialization with terminal-definition analysis enabled.
29/// A terminal plan still requires base solving and answer reconstruction.
30#[derive(Debug)]
31pub enum FormulaMaterialization<T> {
32    /// The complete source theory was materialized.
33    Complete(T),
34    /// A certified base theory and its terminal positive definitions.
35    Terminal(TerminalFormula),
36}

38pub(crate) struct Extension {
39    original: partition::OriginalAnalysis,
40    deferred: Vec<RuleIr>,
41    closed: ClosedSource,
42    baseline: AccountingBaseline,
43    limits: FormulaLimits,
44    location: ProgramSite,
45    // Named noncanonical retained admission storage. Source AST/provenance
46    // follows the existing bounded source-copy policy, not a whole-heap claim.
47    metadata_bytes: u128,
48}

50struct Admitted {
51    base: Compiled,
52    extension: Extension,
53    source: Owner,
54    metadata: SourceMetadata,
55}
56
57/// One original source owner with a certified terminal-definition partition.
58///
59/// Its base theory alone does not denote the original answer sets. Each base
60/// answer must be extended by [`Self::reconstruction`]. Clones share immutable
61/// source, canonical payload and indexes; each reconstruction session owns its
62/// cumulative work history and private scratch. Objectives and explicit
63/// projection are outside this initial applicability class.
64#[derive(Clone)]
65pub struct TerminalFormula(Arc<Admitted>);
66
67impl fmt::Debug for TerminalFormula {
68    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69        f.debug_struct("TerminalFormula")
70            .field("base_theory", self.base_theory())
71            .field("deferred_templates", &self.deferred_templates())
72            .finish_non_exhaustive()
73    }
74}
