30/// Result of source materialization with terminal-definition analysis enabled.
31/// A terminal plan still requires base solving and answer reconstruction.
32#[derive(Debug)]
33pub enum FormulaMaterialization<T> {
34    /// The complete source theory was materialized.
35    Complete(T),
36    /// A certified base theory and its terminal positive definitions.
37    Terminal(TerminalFormula),
38}

53/// How a terminal owner's base was grounded.
54#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
55pub enum BaseKind {
56    /// Every base rule was instantiated: the base adaptive materialization
57    /// grounds, and the default of an empty receipt.
58    #[default]
59    Eager,
60    /// The base's producer core was instantiated and its eligible integrity
61    /// constraints (possibly none) are streamed: its stable models are
62    /// proposals until the constraints accept them.
63    Hybrid,
64}
65
66/// A terminal owner's base, by kind.
67#[derive(Clone, Copy, Debug)]
68pub enum TerminalBase<'a> {
69    /// The base theory, whose stable models are the base's answer sets.
70    Eager(&'a Theory),
71    /// The streamed core; only its checked stable models are base answers.
72    Hybrid(&'a crate::StreamedCore),
73}
74
75enum Base {
76    Eager(Box<Compiled>),
77    Hybrid(crate::StreamedCore),
78}
79
80impl Base {
81    fn compiled(&self) -> &Compiled {
82        match self {
83            Self::Eager(compiled) => compiled,
84            Self::Hybrid(core) => core.compiled(),
85        }
86    }
87}

89struct Admitted {
90    base: Base,
91    extension: Extension,
92    source: Arc<Owner>,
93    metadata: SourceMetadata,
94}

96/// One original source owner with a certified terminal-definition partition.
97///
98/// Its base theory alone does not denote the original answer sets. Each base
99/// answer must be extended by [`Self::reconstruction`]. Clones share immutable
100/// source, canonical payload and indexes; each reconstruction session owns its
101/// per-answer allowance, reported totals and private scratch. Objectives and explicit
102/// projection are outside this initial applicability class.
103#[derive(Clone)]
104pub struct TerminalFormula(Arc<Admitted>);
105
106impl fmt::Debug for TerminalFormula {
107    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
108        f.debug_struct("TerminalFormula")
109            .field("base_kind", &self.base_kind())
110            .field("deferred_templates", &self.deferred_templates())
111            .finish_non_exhaustive()
112    }
113}

115impl TerminalFormula {
116    pub(crate) fn new(
117        compiled: Compiled,
118        core: Core,
119        extension: Extension,
120        source: Owner,
121        metadata: SourceMetadata,
122    ) -> Self {
123        let source = Arc::new(source);
124        let base = match core {
125            Core::Eager => Base::Eager(Box::new(compiled)),
126            Core::Hybrid(constraints) => Base::Hybrid(crate::StreamedCore::new(
127                compiled,
128                constraints.map(|constraints| *constraints),
129                Arc::clone(&source),
130            )),
131        };
132        Self(Arc::new(Admitted {
133            base,
134            extension,
135            source,
136            metadata,
137        }))
138    }
139
140    /// How the base was grounded.
141    #[must_use]
142    pub fn base_kind(&self) -> BaseKind {
143        match self.0.base {
144            Base::Eager(_) => BaseKind::Eager,
145            Base::Hybrid(_) => BaseKind::Hybrid,
146        }
147    }
148
149    /// The base: its theory when eager, its streamed core when hybrid.
150    #[must_use]
151    pub fn base(&self) -> TerminalBase<'_> {
152        match &self.0.base {
153            Base::Eager(compiled) => TerminalBase::Eager(&compiled.theory),
154            Base::Hybrid(core) => TerminalBase::Hybrid(core),
155        }
156    }
157
158    /// Exact original-owner identity. Equal source text is insufficient.
159    #[must_use]
160    pub fn same_instance(&self, other: &Self) -> bool {
161        Arc::ptr_eq(&self.0, &other.0)
162    }
163
164    /// The eager base theory, whose verified answers require positive
165    /// extension; `None` for a hybrid base, whose core theory's stable models
166    /// are only proposals (see [`Self::base`]).
167    #[must_use]
168    pub fn base_theory(&self) -> Option<&Theory> {
169        match &self.0.base {
170            Base::Eager(compiled) => Some(&compiled.theory),
171            Base::Hybrid(_) => None,
172        }
173    }
