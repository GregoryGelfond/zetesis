1//! Terminal definitions indexed by head signature and parsed origin.
2//!
3//! Correspondence stays certified by whole-rule matching; the index only
4//! nominates candidates, so a shared origin never establishes a match. Each
5//! index holds borrowed keys and definition positions in leased scratch, never
6//! canonical payload. Building costs one step per definition, parsed origin and
7//! head plus a bulk charge for sorting, n·⌈log₂(n + 1)⌉ steps charged before
8//! the sort as a model of its comparisons rather than a count of them; each
9//! lookup costs one step per probe.
10
11use std::cmp::Ordering;
12
13use themelios_base::span::Location;
14use themelios_program::program::{Arguments, Statement};
15use themelios_program::provenance::WithProvenance;
16use zetesis_core::catalog::PredicateRef;
17
18use super::{
19    matching, reads,
20    workspace::{Context, Scratch},
21};
22use crate::FormulaFailure;

24pub(super) struct Defined<'d> {
25    pub all: &'d [&'d WithProvenance<Statement>],
26    /// Name and arity of each ordinary single-tuple head, ascending.
27    signatures: Scratch<(&'d str, usize)>,
28    /// Each parsed origin with the position of the definition carrying it,
29    /// ascending; the definitions of one origin keep their program order.
30    origins: Scratch<(Location, usize)>,
31}
32
33impl<'d> Defined<'d> {
34    pub fn index(
35        all: &'d [&'d WithProvenance<Statement>],
36        context: &mut Context<'_, '_>,
37    ) -> Result<Self, FormulaFailure> {
38        let mut signatures = Scratch::new(context)?;
39        signatures.reserve(all.len(), context)?;
40        let mut count = 0_usize;
41        for definition in all {
42            context.work()?;
43            count += crate::extended::parsed_origins(definition).len();
44        }
45        let mut origins = Scratch::new(context)?;
46        origins.reserve(count, context)?;
47        for (position, definition) in all.iter().enumerate() {
48            context.work()?;
49            if let Some(head) = matching::head(reads::source(definition))
50                && let Arguments::Single(arguments) = &head.arguments
51            {
52                signatures.push((head.name.as_str(), arguments.len()), context)?;
53            }
54            for origin in crate::extended::parsed_origins(definition) {
55                origins.push((origin, position), context)?;
56            }
57        }
58        sort(&mut signatures.values, context)?;
59        sort(&mut origins.values, context)?;
60        Ok(Self {
61            all,
62            signatures,
63            origins,
64        })
65    }
66
67    /// Whether a definition's head has the predicate's name and arity, as
68    /// `matching::signature` compares them with the sign ignored.
69    pub fn defines(
70        &self,
71        predicate: PredicateRef<'_>,
72        context: &mut Context<'_, '_>,
73    ) -> Result<bool, FormulaFailure> {
74        let key = (predicate.name(), predicate.arity());
75        let (mut low, mut high) = (0, self.signatures.values.len());
76        while low < high {
77            context.work()?;
78            let middle = low + (high - low) / 2;
79            match self.signatures.values[middle].cmp(&key) {
80                Ordering::Less => low = middle + 1,
81                Ordering::Greater => high = middle,
82                Ordering::Equal => return Ok(true),
83            }
84        }
85        Ok(false)
86    }
87
88    /// The index range of the definitions carrying `origin`.
89    pub fn origin(
90        &self,
91        origin: Location,
92        context: &mut Context<'_, '_>,
93    ) -> Result<std::ops::Range<usize>, FormulaFailure> {
94        let entries = &self.origins.values;
95        let start = partition(entries, context, |entry| entry.0 < origin)?;
96        let end = partition(entries, context, |entry| entry.0 <= origin)?;
97        Ok(start..end)
98    }
99
100    /// The definition position of the origin entry at `index`.
101    pub fn position(&self, index: usize) -> usize {
102        self.origins.values[index].1
103    }
104
105    /// Number of origin entries, for per-origin cursors.
106    pub fn origin_entries(&self) -> usize {
107        self.origins.values.len()
108    }
109}
110
111/// Sort `values`, charging n·⌈log₂(n + 1)⌉ steps first as a model of the
112/// comparisons.
113pub(super) fn sort<T: Ord>(
114    values: &mut [T],
115    context: &mut Context<'_, '_>,
116) -> Result<(), FormulaFailure> {
117    let count = values.len() as u128;
118    let depth = u128::from(usize::BITS - values.len().leading_zeros());
119    context.charge(count * depth)?;
120    values.sort_unstable();
121    Ok(())
122}
123
124/// The first index whose value is not `before`, one step per probe.
125pub(super) fn partition<T>(
126    values: &[T],
127    context: &mut Context<'_, '_>,
128    before: impl Fn(&T) -> bool,
129) -> Result<usize, FormulaFailure> {
130    let (mut low, mut high) = (0, values.len());
131    while low < high {
132        context.work()?;
133        let middle = low + (high - low) / 2;
134        if before(&values[middle]) {
135            low = middle + 1;
136        } else {
137            high = middle;
138        }
139    }
140    Ok(low)
141}
