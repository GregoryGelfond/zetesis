15/// Work and substitutions of one account of reconstruction.
16#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
17pub struct ReconstructionCharges {
18    /// Charged work.
19    pub work: u64,
20    /// Complete substitutions.
21    pub substitutions: u64,
22}
23
24impl ReconstructionCharges {
25    const fn saturating_sub(self, other: Self) -> Self {
26        Self {
27            work: self.work.saturating_sub(other.work),
28            substitutions: self.substitutions.saturating_sub(other.substitutions),
29        }
30    }
31    const fn saturating_add(self, other: Self) -> Self {
32        Self {
33            work: self.work.saturating_add(other.work),
34            substitutions: self.substitutions.saturating_add(other.substitutions),
35        }
36    }
37    fn max(self, other: Self) -> Self {
38        Self {
39            work: self.work.max(other.work),
40            substitutions: self.substitutions.max(other.substitutions),
41        }
42    }
43}

45/// Reconstruction history for one session. Each call is bounded by the
46/// per-answer `allowance`, fixed when the cursor is created; the totals are
47/// reported only.
48#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
49pub struct ReconstructionStatistics {
50    /// Calls begun, including a refused call.
51    pub attempts: u64,
52    /// Complete interpretations published.
53    pub completed: u64,
54    /// Accepted source and reconstruction work; includes admission history
55    /// and every call, for reporting.
56    pub work: u64,
57    /// Accepted source and reconstruction substitutions; includes admission
58    /// and every call, for reporting.
59    pub substitutions: u64,
60    /// Charged by source admission; every call starts after it.
61    pub admission: ReconstructionCharges,
62    /// What each call may charge: the headroom the formula ceilings left
63    /// after admission.
64    pub allowance: ReconstructionCharges,
65    /// The latest call's accepted charge: on a refused session, what the
66    /// refused call accepted before its refusal; otherwise the answer
67    /// delivered last.
68    pub latest: ReconstructionCharges,
69    /// The componentwise peak of accepted charges over every call, refused
70    /// calls included: how close an answer came to the allowance.
71    pub peak: ReconstructionCharges,
72}

74/// A reconstruction refused without publishing a partial interpretation.
75#[derive(Debug)]
76pub enum ReconstructionError {
77    /// The supplied interpretation belongs to a different base occurrence owner.
78    ForeignInput,
79    /// A prior call already refused; this reconstruction history is fused.
80    Failed,
81    /// Located source, work, storage, allocation or cancellation refusal.
82    Source(Box<FormulaFailure>),
83    /// The selected output could not be published.
84    Model(ModelFailure<Box<FormulaFailure>>),
85    /// A certified flat pattern did not match its retained frame shape.
86    Unification(UnificationError),
87    /// A certified head could not be instantiated by its complete body binding.
88    Head(zetesis_core::InstantiationError),
89}
90
91impl ReconstructionError {
92    /// State a work or substitution refusal of one call against its allowance:
93    /// both the limit and the observed count less admission's charge.
94    fn relative_to(self, admission: ReconstructionCharges) -> Self {
95        let relative = |failure: Box<FormulaFailure>| match *failure {
96            FormulaFailure::Limit {
97                resource,
98                limit,
99                observed,
100                location,
101            } if matches!(
102                resource,
103                crate::FormulaResource::Work | crate::FormulaResource::Substitutions
104            ) =>
105            {
106                let base = u128::from(if resource == crate::FormulaResource::Work {
107                    admission.work
108                } else {
109                    admission.substitutions
110                });
111                Box::new(FormulaFailure::Limit {
112                    resource,
113                    limit: limit.saturating_sub(base),
114                    observed: observed.saturating_sub(base),
115                    location,
116                })
117            }
118            other => Box::new(other),
119        };
120        match self {
121            Self::Source(error) => Self::Source(relative(error)),
122            Self::Model(ModelFailure::Stopped(error)) => {
123                Self::Model(ModelFailure::Stopped(relative(error)))
124            }
125            other => other,
126        }
127    }
128
129    fn retain_input(self, owner: &crate::formula_owner::Owner) -> Self {
130        match self {
131            Self::Source(error) => Self::Source(Box::new(owner.retain_failure(*error))),
132            Self::Model(ModelFailure::Stopped(error)) => Self::Model(ModelFailure::Stopped(
133                Box::new(owner.retain_failure(*error)),
134            )),
135            other => other,
136        }
137    }
138
139    /// Cancellation or deadline, distinct from resource or structural refusal.
140    #[must_use]
141    pub fn stop(&self) -> Option<Stop> {
142        match self {
143            Self::Source(error) | Self::Model(ModelFailure::Stopped(error)) => error.interruption(),
144            Self::ForeignInput
145            | Self::Failed
146            | Self::Model(ModelFailure::Model(_))
147            | Self::Unification(_)
148            | Self::Head(_) => None,
149        }
150    }
151}
152impl From<FormulaFailure> for ReconstructionError {
153    fn from(error: FormulaFailure) -> Self {
154        Self::Source(Box::new(error))
155    }
156}
157impl fmt::Display for ReconstructionError {
158    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
159        match self {
160            Self::ForeignInput => {
161                f.write_str("answer reconstruction received a different base program")
162            }
163            Self::Failed => f.write_str("answer reconstruction stopped after an earlier refusal"),
164            Self::Source(error) => error.fmt(f),
165            Self::Model(error) => error.fmt(f),
166            Self::Unification(error) => error.fmt(f),
167            Self::Head(error) => error.fmt(f),
168        }
169    }
170}
171impl std::error::Error for ReconstructionError {
172    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
173        match self {
174            Self::ForeignInput | Self::Failed => None,
175            Self::Source(error) => Some(error.as_ref()),
176            Self::Model(error) => Some(error),
177            Self::Unification(error) => Some(error),
178            Self::Head(error) => Some(error),
179        }
180    }
181}

183/// Independent extension cursor over an immutable admitted owner. Each call
184/// starts from admission's account under the cursor's per-answer allowance;
185/// only the reported totals accumulate (saturating). The first call prepares
186/// the immutable pattern and frame layout plan under that call's allowance.
187/// Later calls reuse it, counting its retained capacity against their storage
188/// ceiling. Bindings, cursors and selected derived rows are private to each
189/// call. Nothing from a previous answer becomes true through plan reuse.
190pub struct TerminalReconstruction<'a> {
191    owner: &'a TerminalFormula,
192    plan: Option<plan::Plan<'a>>,
193    admission: ReconstructionCharges,
194    allowance: ReconstructionCharges,
195    /// Admission and every call so far, for reporting.
196    total: ReconstructionCharges,
197    latest: ReconstructionCharges,
198    peak: ReconstructionCharges,
199    attempts: u64,
200    completed: u64,
201    failed: bool,
202}
203
204impl<'a> TerminalReconstruction<'a> {
205    pub(super) fn new(owner: &'a TerminalFormula) -> Result<Self, ReconstructionError> {
206        let prepared = &owner.0.extension;
207        if prepared.closed.components.is_none() {
208            return Err(owner
209                .0
210                .source
211                .retain_failure(components::missing(prepared.location))
212                .into());
213        }
214        let admission = ReconstructionCharges {
215            work: prepared.baseline.work,
216            substitutions: prepared.baseline.substitutions,
217        };
218        // The sizing: each answer may use the headroom admission left.
219        let ceilings = ReconstructionCharges {
220            work: prepared.limits.max_work,
221            substitutions: prepared.limits.max_substitutions,
222        };
223        Ok(Self {
224            owner,
225            plan: None,
226            admission,
227            allowance: ceilings.saturating_sub(admission),
228            total: admission,
229            latest: ReconstructionCharges::default(),
230            peak: ReconstructionCharges::default(),
231            attempts: 0,
232            completed: 0,
233            failed: false,
234        })
235    }
236
237    /// Charges accepted through the latest completed or refused call.
238    #[must_use]
239    pub fn statistics(&self) -> ReconstructionStatistics {
240        ReconstructionStatistics {
241            attempts: self.attempts,
242            completed: self.completed,
243            work: self.total.work,
244            substitutions: self.total.substitutions,
245            admission: self.admission,
246            allowance: self.allowance,
247            latest: self.latest,
248            peak: self.peak,
249        }
250    }
251
252    /// Extend exactly this base interpretation by all certified definitions.
253    /// The caller must separately establish that the input is a base answer set.
254    /// The partition theorem then establishes membership for the returned model.
255    /// Each call starts from admission's history and may charge the cursor's
256    /// `allowance` of work and substitutions, whatever earlier answers used;
257    /// a work or substitution refusal reports the allowance as its limit and
258    /// the call's requested charge (what the refused operation needed, past the
259    /// accepted one) as observed. Other ceilings are checked within
260    /// the call as during admission. No full possible-support relation is
261    /// enumerated during reconstruction. The first call also binds the closed
262    /// components and prepares the immutable rule plan; its actual preparation
263    /// charge is included in that call's statistics, never charged again.
264    ///
265    /// # Errors
266    /// Refuses a foreign owner, cancellation, a resource boundary or invalid
267    /// retained structure. Any refusal fuses this cursor. Earlier returned models
268    /// remain valid and their capacities are owned by the consuming session.
269    /// Logical source failures retain the original canonical program.
270    pub fn reconstruct(
271        &mut self,
272        model: &Model,
273        cancellation: &Cancellation,
274    ) -> Result<Model, ReconstructionError> {
275        if self.failed {
276            return Err(ReconstructionError::Failed);
277        }
278        self.failed = true;
279        self.attempts = self.attempts.checked_add(1).ok_or_else(|| {
280            self.owner
281                .0
282                .source
283                .retain_failure(super::storage::overflow(self.owner.0.extension.location))
284        })?;
285        let owner = self.owner;
286        let prepared = &owner.0.extension;
287        // Each answer starts from admission's history under its own allowance.
288        let limits = crate::FormulaLimits {
289            max_work: self.admission.work.saturating_add(self.allowance.work),
290            max_substitutions: self
291                .admission
292                .substitutions
293                .saturating_add(self.allowance.substitutions),
294            ..prepared.limits
295        };
296        let mut accounting = prepared.baseline.start();
297        let result = accounting.with_cancellation(cancellation, |counters| {
298            extend(owner, &mut self.plan, model, counters, &limits)
299        });
300        self.latest = ReconstructionCharges {
301            work: accounting.work,
302            substitutions: accounting.substitutions,
303        }
304        .saturating_sub(self.admission);
305        self.peak = self.peak.max(self.latest);
306        self.total = self.total.saturating_add(self.latest);
307        if result.is_ok() {
308            self.completed += 1; // bounded by the checked attempts count
309            self.failed = false;
310        }
311        result.map_err(|error| {
312            error
313                .relative_to(self.admission)
314                .retain_input(&owner.0.source)
315        })
316    }
317}

319fn extend<'a>(
320    owner: &'a TerminalFormula,
321    plan: &mut Option<plan::Plan<'a>>,
322    model: &Model,
323    counters: &mut Counters,
324    limits: &crate::FormulaLimits,
325) -> Result<Model, ReconstructionError> {
326    let prepared = &owner.0.extension;
327    counters.work(limits, prepared.location)?;
328    if !model.catalog().same_owner(owner.base_atom_catalog()) {
329        return Err(ReconstructionError::ForeignInput);
330    }
331    // The descendant writer counts the shared closed allocations. The separate
332    // closed handle, component metadata, base output maps and input selection
333    // remain external; no payload is charged twice merely because it is shared.
334    let external = prepared.metadata_bytes
335        + prepared.closed.metadata_bytes()
336        + size_of::<ClosedCatalog>() as u128
337        + model.selection_bytes()
338        + size_of::<TerminalReconstruction<'_>>() as u128
339        + plan.as_ref().map_or(0, plan::Plan::retained_bytes);
340    let mut work = Work {
341        limits,
342        counters,
343        location: prepared.location,
344        external,
345    };
346    let mut writer = work.writer(&prepared.closed.storage)?;
347    if plan.is_none() {
348        let components = prepared
349            .closed
350            .components
351            .as_ref()
352            .ok_or_else(|| components::missing(prepared.location))?
353            .bind_with(prepared.closed.storage.vocabulary_read(), || work.permit())
354            .map_err(|error| components::failure(error, limits, external, prepared.location))?;
355        let built = plan::Plan::new(&prepared.deferred, components, &writer, &mut work)?;
356        // The preparation's temporary lease is gone. Transfer only its vector
357        // capacities; its header already belongs to this cursor, and its
358        // borrowed components remain counted with the original closed owner.
359        work.external += built.retained_bytes();
360        work.observe(writer.storage_bytes())?;
361        *plan = Some(built);
362    }
363    work.location = prepared.location;
364    let location = work.location;
365    work.atoms(&mut writer, |writer, bound, counters| {
366        writer.discover_model_with(&prepared.closed.storage, model, bound, || {
367            counters.work(limits, location)
368        })
369    })?;
370    plan.as_ref()
371        .expect("a complete immutable plan was published")
372        .derive(model, &mut writer, &mut work)?;
373    work.location = prepared.location;
374    publish(&mut writer, &mut work)
375}

377fn publish(writer: &mut AtomInterner, work: &mut Work<'_>) -> Result<Model, ReconstructionError> {
378    let limits = work.limits;
379    let location = work.location;
380    work.atoms(writer, |writer, bound, counters| {
381        writer.commit_with(bound, || counters.work(limits, location))
382    })?;
383    let order = work.atoms(writer, |writer, bound, counters| {
384        writer.ordered_ids_with(bound, || counters.work(limits, location))
385    })?;
386    let mut order_lease = work.counters.lease();
387    order_lease.observe(
388        size_of_val(&order) + order.capacity() * size_of::<usize>(),
389        location,
390    )?;
391    work.observe(writer.storage_bytes())?;
392    let catalog = work.atoms(writer, |writer, bound, counters| {
393        writer.publish_selection_with(&order, bound, || counters.work(limits, location))
394    })?;
395    let mut output_lease = work.counters.lease();
396    output_lease.observe(
397        usize::try_from(catalog.publication_bytes())
398            .map_err(|_| super::storage::overflow(location))?,
399        location,
400    )?;
401    drop(order);
402    drop(order_lease);
403    let remaining = work.remaining(writer.storage_bytes())?;
404    let result =
405        Model::publish_ordered_catalog_with(catalog, remaining, || work.permit().map_err(Box::new));
406    let peak = match &result {
407        Ok(model) => model.selection_bytes(),
408        Err(error) => error.peak_bytes(),
409    };
410    let observed = work.observe(writer.storage_bytes() + peak);
411    let model = result.map_err(|error| ReconstructionError::Model(error.into_parts().0))?;
412    observed?;
413    work.permit()?;
414    Ok(model)
415}
