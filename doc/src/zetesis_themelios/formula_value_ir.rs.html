1//! Single-valued argument consumers share head binding and scalar evaluation.

3use themelios_program::program::{Arguments, DefaultNegation, Literal, LiteralInner};
4use themelios_program::term::Term;
5
6use crate::diagnostic::unsupported;
7use crate::formula_ir::{Compiler, LiteralIr, Operation, Variables};
8use crate::{ExpansionResource, FormulaFailure, ProfileFeature};

10impl Compiler<'_> {
11    /// Positive atoms may capture evaluated positions for a later equality check.
12    /// Negative atoms consume independently established bindings. Neither check
13    /// nor fresh data slot turns an expression input into a relational producer.
14    pub(super) fn literal_into(
15        &mut self,
16        literal: &Literal,
17        variables: &mut Variables,
18        body: &mut Vec<LiteralIr>,
19    ) -> Result<(), FormulaFailure> {
20        if self.range_comparison(literal, variables, body)? {
21            return Ok(());
22        }
23        if literal.negation == DefaultNegation::None
24            && let LiteralInner::Atom(atom) = &literal.inner
25            && self.positive_literal(atom.get(), variables, body)?
26        {
27            return Ok(());
28        }
29        if literal.negation != DefaultNegation::None
30            && let LiteralInner::Atom(atom) = &literal.inner
31            && let Arguments::Single(arguments) = &atom.get().arguments
32        {
33            // Pools have already been selected by the enclosing occurrence
34            // cursor. Ranges remain scoped value generators before truth checks.
35            for term in arguments.iter().flat_map(Term::subterms) {
36                self.budget
37                    .charge(ExpansionResource::TermWork, 1, self.location)?;
38                if matches!(term, Term::Pool(_)) {
39                    return Err(unsupported(ProfileFeature::Term, self.location).into());
40                }
41            }
42        }
43        if literal.negation != DefaultNegation::None
44            && let LiteralInner::Atom(atom) = &literal.inner
45            && let Some(projection) =
46                self.projected_atom(atom.get(), literal.negation, variables, body)?
47        {
48            body.push(projection);
49            return Ok(());
50        }
51        if literal.negation != DefaultNegation::None
52            && let LiteralInner::Atom(atom) = &literal.inner
53            && let Arguments::Single(arguments) = &atom.get().arguments
54            && arguments
55                .iter()
56                .any(|term| !matches!(term, Term::Variable(_) | Term::Symbolic(_)))
57        {
58            self.generated_arguments(arguments, variables)?;
59            let pattern = self.generated_atom(atom.get(), variables, body)?;
60            body.push(LiteralIr::Atom(literal.negation, pattern));
61        } else {
62            body.push(self.literal(literal, variables)?);
63        }
64        Ok(())
65    }

67    /// Bound the owned source copy and flat-plan payload before cloning the AST.
68    /// This accounting is separate from each evaluated value's construction cap.
69    pub(super) fn value_plan_preflight(&mut self, term: &Term) -> Result<(), FormulaFailure> {
70        // Scalar-only expressions keep their established charging/error order.
71        // The bounded source traversal only selects this newly admitted plan
72        // category; construction plans then charge their complete scan and copy.
73        if !term
74            .subterms()
75            .any(|node| matches!(node, Term::Function { .. } | Term::Tuple(_)))
76        {
77            return Ok(());
78        }
79        let mut bytes = 0_u128;
80        for node in term.subterms() {
81            self.budget
82                .charge(ExpansionResource::TermWork, 1, self.location)?;
83            bytes += (std::mem::size_of::<Term>() + std::mem::size_of::<Operation>()) as u128;
84            bytes += match node {
85                Term::Function { name, arguments } => {
86                    std::mem::size_of::<crate::formula_value::Constructor>() as u128
87                        + name.as_str().len() as u128 * 2
88                        + arguments.len() as u128 * std::mem::size_of::<usize>() as u128
89                }
90                Term::Tuple(arguments) => {
91                    std::mem::size_of::<crate::formula_value::Constructor>() as u128
92                        + arguments.len() as u128 * std::mem::size_of::<usize>() as u128
93                }
94                Term::Symbolic(symbol) => crate::structural_value::symbol_bytes(symbol),
95                Term::Variable(themelios_program::term::Variable::Named(name)) => {
96                    name.as_str().len() as u128
97                }
98                _ => 0,
99            };
100        }
101        self.budget
102            .charge(ExpansionResource::ScalarBytes, bytes, self.location)?;
103        Ok(())
104    }
