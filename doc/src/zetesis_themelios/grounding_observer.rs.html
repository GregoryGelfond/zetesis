10/// Outcome of the optional finite-domain attempt over the exact normalized
11/// source owner. This synchronous borrowed view cannot outlive that attempt.
12/// Completed analysis is an upper bound, not a source or answer-set certificate.
13#[derive(Clone, Copy, Debug)]
14pub enum DomainObservation<'a, 'source> {
15    /// No domain analysis was requested.
16    Disabled,
17    /// The complete normalized source/IR profile was not applicable.
18    Inapplicable,
19    /// Actual analysis, including its exact owner, status, context and logical
20    /// statistics. Unknown/Stopped supplies no narrowing. Retaining consumers
21    /// copy the specific owned status/statistics they need during the callback.
22    Analyzed(&'a zetesis_domain::Analysis<'source>),
23}

25/// Observe only complete formula materialization after source preparation.
26///
27/// Every `enter` has a matching `exit`, including a failed or unwinding attempt.
28/// A boundary event says nothing about semantic completion. Implementations must
29/// not panic or affect source semantics. No observer is called by the ordinary
30/// admission APIs, and this crate performs no timing itself.
31pub trait GroundingObserver {
32    /// Begin possible-support completion and finite formula instantiation.
33    fn enter(&self);
34    /// End the attempted materialization, whether it succeeded or failed.
35    fn exit(&self);
36
37    /// Opt into bounded work counters and coarse phase callbacks for this attempt.
38    ///
39    /// Called once after `enter`. The default retains boundary-only observation.
40    /// No clock is read by the frontend. Detailed observation allocates one fixed
41    /// counter bank; callback implementations own any further storage or timing.
42    fn details_enabled(&self) -> bool {
43        false
44    }
45
46    /// The certified source partition will ground a base formula and retain
47    /// terminal definitions for reconstruction after base membership checks.
48    /// Called before that base materialization attempt, including one that later
49    /// refuses. This identifies the attempted route, not completed grounding or
50    /// reconstructed answers; it introduces no measurement or work charge.
51    fn terminal_definitions(&self) {}
52
53    /// Observe the actual optional domain attempt before final rule guards.
54    /// Called synchronously, independently of detailed work-counter opt-in.
55    /// A prior grounding failure may prevent the attempt and this callback.
56    fn domain_analysis(&self, _observation: DomainObservation<'_, '_>) {}
57
58    /// Begin one coarse phase after its counter bank has been reset.
59    ///
60    /// Within one grounding attempt, phases are sequential and do not nest.
61    /// Rule instantiation emits one
62    /// pair per prepared IR rule; the site identifies the original statement and
63    /// any actual source coordinate, and may repeat after expansion. Joins,
64    /// filtering and formula
65    /// emission remain interleaved inside that phase.
66    /// Whole-program phases use `None` rather than claiming one program site.
67    fn phase_enter(&self, _phase: GroundingPhase, _site: Option<ProgramSite>) {}
68
69    /// End a phase with work performed before success, failure or unwind.
70    ///
71    /// Counts belong only to this phase, including its failed attempt. Counter
72    /// overflow makes that field unavailable and never changes admission or its
73    /// resource limits. Callback overhead is not removed from observed durations.
74    /// A completed phase does not establish completed grounding or stable models.
75    fn phase_exit(
76        &self,
77        _phase: GroundingPhase,
78        _site: Option<ProgramSite>,
79        _outcome: GroundingOutcome,
80        _work: GroundingWork,
81    ) {
82    }
83}

85pub(crate) fn observe<T>(
86    observer: Option<&dyn GroundingObserver>,
87    action: impl FnOnce() -> T,
88) -> T {
89    struct Exit<'a>(Option<&'a dyn GroundingObserver>);
90    impl Drop for Exit<'_> {
91        fn drop(&mut self) {
92            if let Some(observer) = self.0 {
93                observer.exit();
94            }
95        }
96    }
97    if let Some(observer) = observer {
98        observer.enter();
99    }
100    let _exit = Exit(observer);
101    action()
102}

104#[cfg(test)]
105mod tests {
106    use super::{GroundingObserver, observe};
107    use std::cell::Cell;
108
109    struct Observer(Cell<i32>);
110    impl GroundingObserver for Observer {
111        fn enter(&self) {
112            self.0.set(self.0.get() + 1);
113        }
114        fn exit(&self) {
115            self.0.set(self.0.get() - 1);
116        }
117    }
118
119    #[test]
120    fn unwind_restores_the_callers_boundary() {
121        let observer = Observer(Cell::new(0));
122        let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
123            observe(Some(&observer), || {
124                assert_eq!(observer.0.get(), 1);
125                panic!("controlled failure");
126            });
127        }));
128        assert!(result.is_err());
129        assert_eq!(observer.0.get(), 0);
130        assert_eq!(observe(None, || 7), 7);
131    }
