10/// A coarse operation inside eager formula materialization.
11///
12/// These phases exclude parsing and prepared-IR/observation compilation. Their
13/// callbacks are sequential; rule phases repeat in the existing prepared order.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15#[non_exhaustive]
16pub enum GroundingPhase {
17    /// Attempt optional domains over the exact normalized positive source.
18    DomainAnalysis,
19    /// Complete the finite possible-support relation and its indices, then
20    /// validate arithmetic families over that complete carrier.
21    SupportCompletion,
22    /// Determine active objective templates and construct their program.
23    ObjectiveActivation,
24    /// Initialize formula storage and the false/true nodes.
25    FormulaInitialization,
26    /// Join one prepared rule, evaluate filters and emit its formulas together.
27    RuleInstantiation,
28    /// Add constraints for present opposite classical signs.
29    Coherence,
30    /// Add necessary-support guards after every producer has been collected.
31    SupportGuards,
32    /// Validate and construct the final immutable formula theory.
33    TheoryValidation,
34}
35
36impl GroundingPhase {
37    /// Every currently defined phase, in materialization order.
38    ///
39    /// This supports fixed-size caller aggregation without enum discriminants
40    /// or retention of the individual per-rule callbacks.
41    pub const ALL: [Self; 8] = [
42        Self::DomainAnalysis,
43        Self::SupportCompletion,
44        Self::ObjectiveActivation,
45        Self::FormulaInitialization,
46        Self::RuleInstantiation,
47        Self::Coherence,
48        Self::SupportGuards,
49        Self::TheoryValidation,
50    ];
51
52    /// Stable machine-readable identifier; no allocation or clock access.
53    #[must_use]
54    pub const fn label(self) -> &'static str {
55        match self {
56            Self::DomainAnalysis => "domain_analysis",
57            Self::SupportCompletion => "support_completion",
58            Self::ObjectiveActivation => "objective_activation",
59            Self::FormulaInitialization => "formula_initialization",
60            Self::RuleInstantiation => "rule_instantiation",
61            Self::Coherence => "coherence",
62            Self::SupportGuards => "support_guards",
63            Self::TheoryValidation => "theory_validation",
64        }
65    }
66}

68/// How one phase returned; this is not a semantic membership verdict.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70#[non_exhaustive]
71pub enum GroundingOutcome {
72    /// The phase returned `Ok`; later phases can still fail.
73    Completed,
74    /// The phase returned an admission or resource error.
75    Failed,
76    /// The phase did not return because its execution unwound.
77    Unwound,
78}

95/// Exact selected work populations for one phase, including partial failures.
96///
97/// Every field starts at `Some(0)`. `None` means that field overflowed `u64`;
98/// other fields remain usable. Work counts and named capacity receipts are not
99/// durations and do not alter resource admission. A failed operation contributes only
100/// the events reached before its error. Recording uses constant space and one
101/// checked addition or maximum per event; event collection itself has diagnostic overhead.
102/// These fields do not cover every operation; for example, theory validation
103/// can take time while every counter remains zero. Combine phase-local work
104/// and publication counts by checked addition; combine capacity peaks by maximum.
105/// Propagate `None` whenever a constituent is unavailable.
106#[derive(Clone, Copy, Debug, PartialEq, Eq)]
107#[non_exhaustive]
108pub struct GroundingWork {
109    /// Actual accepted formula-work deltas, not estimates or durations.
110    /// Production, ordering, wake and publication are disjoint subsets of this total;
111    /// its remainder covers planning, initial scheduling, snapshot/query
112    /// preparation and round control. Preparation before construction and later
113    /// completed-support query setup are outside this total.
114    pub support_construction_work: Option<u64>,
115    /// Selected-rule traversal, variants, domain guards, head production and subsequent schedule selection.
116    /// Includes accepted work before refusal or unwind.
117    pub support_production_work: Option<u64>,
118    /// Advancing support-generation joins, including local joins, but excluding
119    /// join setup and keyed-group validation. A subset of production work,
120    /// disjoint from head work; includes accepted work before refusal or unwind.
121    pub support_join_work: Option<u64>,
122    /// Resolving, admitting and selecting a derived head. A subset of production
123    /// work, disjoint from join work; includes accepted work before refusal or unwind.
124    pub support_head_work: Option<u64>,
125    /// Sorting the pending support rows by typed identity.
126    /// Includes accepted work before refusal or unwind.
127    pub support_order_work: Option<u64>,
128    /// Preparing the next producer wake set from the ordered pending rows.
129    /// Includes accepted work before refusal or unwind.
130    pub support_wake_work: Option<u64>,
131    /// Initial, round and final support publication, including canonical commit and postings.
132    /// Includes accepted work before refusal or unwind.
133    pub support_publication_work: Option<u64>,
134    /// Support rounds admitted by the existing round ceiling.
135    pub support_rounds: Option<u64>,
136    /// Original producer occurrences entered for variant traversal, after an
137    /// admitted work tick. A later failure retains the entered occurrence.
138    pub support_producer_visits: Option<u64>,
139    /// Growing support/query snapshot attempts entered after an admitted work
140    /// tick. Counts attempts, including a later allocation or query refusal.
141    pub support_snapshot_preparations: Option<u64>,
142    /// New support atoms inserted while publishing a support round.
143    pub support_atoms: Option<u64>,
144    /// Column/row associations inserted into support indices, including rebuilds.
145    pub support_index_entries: Option<u64>,
146    /// Bound-column probe attempts, including absent relations or empty matches.
147    pub join_probes: Option<u64>,
148    /// Probes executed by the indexed positive-row matcher.
149    pub indexed_probes: Option<u64>,
150    /// Completed-support table probes using indexed matching for structural
151    /// patterns or absent relations. Support-growth phases remain indexed.
152    pub table_inapplicable_probes: Option<u64>,
153    /// Prepared table indices successfully retained by this grounding attempt.
154    pub table_preparations: Option<u64>,
155    /// Table probes reusing an already prepared predicate and alias scope.
156    pub table_reuses: Option<u64>,
157    /// Finite-table selections attempted over complete possible support.
158    pub table_probes: Option<u64>,
159    /// Table-selected row occurrences visited by the join, before any optional
160    /// necessary-domain guard and the unchanged full matcher.
161    pub table_rows: Option<u64>,
162    /// Completed charged work inside finite-table index preparation.
163    pub table_prepare_work: Option<u64>,
164    /// Completed charged work inside finite-table row selection.
165    pub table_query_work: Option<u64>,
166    /// Additional index capacity successfully retained, excluding cache slots.
167    pub table_index_bytes: Option<u64>,
168    /// Highest reported admitted support/query capacity, including live masks
169    /// and conservative operation peaks. Refused proposals do not contribute;
170    /// actual allocator slack can. Not process RSS or a complete stack measure.
171    pub support_peak_bytes: Option<u64>,
172    /// Charged optional applicability, analysis and guard-preparation work.
173    /// Includes completed logical work before a conservative analysis stop.
174    pub domain_prepare_work: Option<u64>,
175    /// Original rows inspected by a prepared final-rule domain guard.
176    pub domain_guard_rows: Option<u64>,
177    /// Domain equality-ID membership comparisons performed on those rows.
178    pub domain_guard_checks: Option<u64>,
179    /// Rows rejected before binding by a necessary-domain guard. These rows
180    /// have no complete positive continuation in the admitted pure profile.
181    pub domain_rejected_rows: Option<u64>,
182    /// Values a comparison over one variable alone excluded from that
183    /// variable's candidates before any row was read.
184    pub domain_excluded_values: Option<u64>,
185    /// Existing support rows selected for an attempted pattern match.
186    pub join_rows: Option<u64>,
187    /// Base relational binding snapshots successfully copied by the join cursor.
188    ///
189    /// These may contain placeholders for later generated assignments. This is
190    /// not the number of completed generated bindings or emitted ground rules.
191    /// Complete rows lent directly to their consumer are not owned snapshots.
192    pub binding_snapshots: Option<u64>,
193    /// Expression evaluations entered, including ones that subsequently fail.
194    pub expression_evaluations: Option<u64>,
195    /// Expression operations admitted by the work ceiling, including failed operations.
196    pub expression_nodes: Option<u64>,
197    /// Fully instantiated atom identities presented to the formula atom catalog.
198    pub atom_lookups: Option<u64>,
199    /// New atom identities inserted into the formula atom catalog.
200    pub atoms_inserted: Option<u64>,
201    /// Formula node identities presented to structural interning after the work check.
202    pub node_lookups: Option<u64>,
203    /// New structurally interned nodes, including imported aggregate nodes.
204    pub nodes_inserted: Option<u64>,
205    /// Formula roots and their origins successfully appended.
206    pub roots: Option<u64>,
207}

295    /// Combine snapshots in constant time and space: add work/capacity publication
296    /// counts and retain the maximum support capacity peak.
297    ///
298    /// A field is unavailable if either input is unavailable or their sum
299    /// overflows `u64`; other fields retain their exact sums. This operation
300    /// performs no timing, allocation, resource admission or semantic check.
301    #[must_use]
302    pub fn checked_sum(self, other: Self) -> Self {
303        let sum = |left: Option<u64>, right: Option<u64>| {
304            left.and_then(|left| right.and_then(|right| left.checked_add(right)))
305        };
306        Self {
307            support_construction_work: sum(
308                self.support_construction_work,
309                other.support_construction_work,
310            ),
311            support_production_work: sum(
312                self.support_production_work,
313                other.support_production_work,
314            ),
315            support_join_work: sum(self.support_join_work, other.support_join_work),
316            support_head_work: sum(self.support_head_work, other.support_head_work),
317            support_order_work: sum(self.support_order_work, other.support_order_work),
318            support_wake_work: sum(self.support_wake_work, other.support_wake_work),
319            support_publication_work: sum(
320                self.support_publication_work,
321                other.support_publication_work,
322            ),
323            support_rounds: sum(self.support_rounds, other.support_rounds),
324            support_producer_visits: sum(
325                self.support_producer_visits,
326                other.support_producer_visits,
327            ),
328            support_snapshot_preparations: sum(
329                self.support_snapshot_preparations,
330                other.support_snapshot_preparations,
331            ),
332            support_atoms: sum(self.support_atoms, other.support_atoms),
333            support_index_entries: sum(self.support_index_entries, other.support_index_entries),
334            join_probes: sum(self.join_probes, other.join_probes),

342            table_probes: sum(self.table_probes, other.table_probes),
343            table_rows: sum(self.table_rows, other.table_rows),
344            table_prepare_work: sum(self.table_prepare_work, other.table_prepare_work),
345            table_query_work: sum(self.table_query_work, other.table_query_work),

368    fn record(&mut self, event: Event) {
369        let amount = match event {
370            Event::SupportConstructionWork(work)
371            | Event::SupportProductionWork(work)
372            | Event::SupportJoinWork(work)
373            | Event::SupportHeadWork(work)
374            | Event::SupportOrderWork(work)
375            | Event::SupportWakeWork(work)
376            | Event::SupportPublicationWork(work)
377            | Event::TablePrepareWork(work)
378            | Event::TableQueryWork(work)
379            | Event::DomainPrepareWork(work) => Some(work),
380            Event::TableIndexBytes(bytes) => u64::try_from(bytes).ok(),
381            Event::SupportPeakBytes(bytes) => u64::try_from(bytes).ok(),
382            _ => Some(1),
383        };
384        if matches!(event, Event::SupportPeakBytes(_)) {
385            self.support_peak_bytes = self.support_peak_bytes.zip(amount).map(|(a, b)| a.max(b));
386            return;
387        }
388        let field = match event {
389            Event::SupportConstructionWork(_) => &mut self.support_construction_work,
390            Event::SupportProductionWork(_) => &mut self.support_production_work,
391            Event::SupportJoinWork(_) => &mut self.support_join_work,
392            Event::SupportHeadWork(_) => &mut self.support_head_work,
393            Event::SupportOrderWork(_) => &mut self.support_order_work,
394            Event::SupportWakeWork(_) => &mut self.support_wake_work,
395            Event::SupportPublicationWork(_) => &mut self.support_publication_work,
396            Event::SupportRound => &mut self.support_rounds,
397            Event::SupportProducerVisit => &mut self.support_producer_visits,
398            Event::SupportSnapshotPreparation => &mut self.support_snapshot_preparations,
399            Event::SupportAtom => &mut self.support_atoms,
400            Event::SupportIndexEntry => &mut self.support_index_entries,
401            Event::JoinProbe => &mut self.join_probes,
402            Event::IndexedProbe => &mut self.indexed_probes,
403            Event::TableInapplicableProbe => &mut self.table_inapplicable_probes,
404            Event::TablePreparation => &mut self.table_preparations,
405            Event::TableReuse => &mut self.table_reuses,
406            Event::TableProbe => &mut self.table_probes,
407            Event::TableRow => &mut self.table_rows,
408            Event::TablePrepareWork(_) => &mut self.table_prepare_work,
409            Event::TableQueryWork(_) => &mut self.table_query_work,
410            Event::TableIndexBytes(_) => &mut self.table_index_bytes,
411            Event::SupportPeakBytes(_) => &mut self.support_peak_bytes,
412            Event::DomainPrepareWork(_) => &mut self.domain_prepare_work,
413            Event::DomainGuardRow => &mut self.domain_guard_rows,
414            Event::DomainGuardCheck => &mut self.domain_guard_checks,
415            Event::DomainRejectedRow => &mut self.domain_rejected_rows,
416            Event::DomainExcludedValue => &mut self.domain_excluded_values,
417            Event::JoinRow => &mut self.join_rows,
418            Event::BindingSnapshot => &mut self.binding_snapshots,
419            Event::ExpressionEvaluation => &mut self.expression_evaluations,
420            Event::ExpressionNode => &mut self.expression_nodes,
421            Event::AtomLookup => &mut self.atom_lookups,
422            Event::AtomInserted => &mut self.atoms_inserted,
423            Event::NodeLookup => &mut self.node_lookups,
424            Event::NodeInserted => &mut self.nodes_inserted,
425            Event::Root => &mut self.roots,
426        };
427        *field = field.and_then(|value| amount.and_then(|amount| value.checked_add(amount)));
428    }

431/// Shared only between the phase guard and the existing mutable work counter.
432/// No allocation occurs unless detailed attribution was explicitly enabled.
433/// This serial-attempt bank makes its owning counter non-Send even when disabled.
434/// A future parallel grounder should retain worker-local value counters and merge
435/// their public snapshots, rather than add a global lock to these event sites.
436#[derive(Clone, Default)]
437pub(crate) struct Work(Option<Rc<RefCell<GroundingWork>>>);
438
439impl Work {
440    pub(crate) fn enabled(&self) -> bool {
441        self.0.is_some()
442    }
443
444    pub(crate) fn record(&self, event: Event) {
445        if let Some(work) = &self.0 {
446            work.borrow_mut().record(event);
447        }
448    }
449}

451pub(crate) struct Profile<'a> {
452    observer: Option<&'a dyn GroundingObserver>,
453    source_observer: Option<&'a dyn GroundingObserver>,
454    work: Work,
455}
456
457impl<'a> Profile<'a> {
458    pub(crate) fn new(observer: Option<&'a dyn GroundingObserver>) -> Self {
459        let source_observer = observer;
460        let observer = observer.filter(|observer| observer.details_enabled());
461        let work = Work(observer.map(|_| Rc::new(RefCell::new(GroundingWork::default()))));
462        Self {
463            observer,
464            source_observer,
465            work,
466        }
467    }
468
469    pub(crate) fn domain_analysis(&self, observation: super::DomainObservation<'_, '_>) {
470        if let Some(observer) = self.source_observer {
471            observer.domain_analysis(observation);
472        }
473    }
474
475    pub(crate) fn work(&self) -> Work {
476        self.work.clone()
477    }
478
479    pub(crate) fn phase<T, E>(
480        &self,
481        phase: GroundingPhase,
482        location: Option<ProgramSite>,
483        action: impl FnOnce() -> Result<T, E>,
484    ) -> Result<T, E> {
485        let Some(observer) = self.observer else {
486            return action();
487        };
488        *self
489            .work
490            .0
491            .as_ref()
492            .expect("enabled counter bank")
493            .borrow_mut() = GroundingWork::default();
494        observer.phase_enter(phase, location);
495        let mut exit = Exit {
496            profile: self,
497            phase,
498            location,
499            outcome: GroundingOutcome::Unwound,
500        };
501        let result = action();
502        exit.outcome = if result.is_ok() {
503            GroundingOutcome::Completed
504        } else {
505            GroundingOutcome::Failed
506        };
507        result
508    }
509}

511struct Exit<'a, 'b> {
512    profile: &'a Profile<'b>,
513    phase: GroundingPhase,
514    location: Option<ProgramSite>,
515    outcome: GroundingOutcome,
516}
517
518impl Drop for Exit<'_, '_> {
519    fn drop(&mut self) {
520        let work = *self
521            .profile
522            .work
523            .0
524            .as_ref()
525            .expect("enabled counter bank")
526            .borrow();
527        self.profile.observer.expect("enabled observer").phase_exit(
528            self.phase,
529            self.location,
530            self.outcome,
531            work,
532        );
533    }
534}
