1//! Admission of source and canonical logical programs through pinned themelios.
2//! [`admit_program`] borrows a canonical program for strict relational admission.
3//! [`prepare_program_relational`] retains an owned canonical program and display
4//! metadata through bounded scalar and fact expansion;
5//! [`prepare_program_formula`] retains its original program through an `Arc` and
6//! enters the shared formula compiler without rendering or reparsing.
7//! Strict S0 and extended scalar admission compile relational templates without
8//! grounding. The separate formula APIs compute a bounded possible-positive
9//! relation and ground complete joins into normal, conditional-choice, and finite-aggregate Ferraris
10//! formulas. Lifted minimization templates remain separate and score only supplied
11//! verified stable models. No source route invokes a solver or external engine.
12//!
13//! Source admission retains original bytes and parsed origins on success;
14//! bundle failures retain the complete source catalog. Typed formula input keeps
15//! original statement identity and any actual coordinates, without inventing
16//! source bytes. Independent source,
17//! syntax, scalar expansion, finite substitution, and formula storage limits
18//! refuse rather than return recovered or truncated programs. Finite count/sum
19//! assignment cursors may generate values absent from the source while retaining
20//! exact reduct equality formulas. Scalar equality and interval bindings follow
21//! dependency-ordered plans; evaluated normal heads use fresh value slots and
22//! bounded streamed cursors. Pinned upstream analysis reads a separately bounded
23//! pool-free owned projection; its facts are retained without claiming
24//! its narrower safety reading proves the clingo assignment extension.
25//! Syntax walks and
26//! term normalization are iterative. The formula route preserves recursive
27//! eligibility and uses constraints for choice bounds. Completed theories always
28//! include necessary support guards. These semantically redundant guards remain
29//! double-negated so they do not constrain reduct subsets; they are not a
30//! selectable constructor option.
31//! Source acceptance is not a parser-to-Lean or Rust implementation proof.
32#![forbid(unsafe_code)]

177/// Explicit host admission ceilings. Zero means that no resource of that kind
178/// may be consumed; limits never mean unlimited.
179#[derive(Clone, Copy, Debug)]
180pub struct AdmissionOptions {
181    /// Identity attached to every source location.
182    pub source_id: SourceId,
183    /// Maximum UTF-8 bytes accepted before parsing.
184    pub max_source_bytes: usize,
185    /// Maximum syntax nodes visited before raising.
186    pub max_syntax_nodes: usize,
187    /// Maximum syntax-node nesting, including every nested term.
188    pub max_syntax_depth: usize,
189    /// Maximum source body elements per rule, before set deduplication.
190    pub max_body_elements: usize,
191    /// Limits checked independently by the core template admission door.
192    pub core_limits: AdmissionLimits,
193}
194
195impl Default for AdmissionOptions {
196    fn default() -> Self {
197        Self {
198            source_id: SourceId::new(0),
199            max_source_bytes: 1_048_576,
200            max_syntax_nodes: 262_144,
201            max_syntax_depth: 128,
202            max_body_elements: 1_024,
203            core_limits: AdmissionLimits::default(),
204        }
205    }
206}

208/// An admitted program and the source evidence retained by its boundary.
209/// Template origins follow the core program's unchanged template order.
210#[derive(Debug)]
211pub struct Admitted {
212    program: Program,
213    source: Source,
214    template_origins: Vec<Vec<Location>>,
215    metadata: SourceMetadata,
216    expansion: ExpansionUsage,
217}
218
219impl Admitted {
220    /// The checked finite relational program.
221    #[must_use]
222    pub fn program(&self) -> &Program {
223        &self.program
224    }
225
226    /// Original bytes under the source identity used by every origin.
227    #[must_use]
228    pub fn source(&self) -> &Source {
229        &self.source
230    }
231
232    /// Parsed origins per template. Equal source rules may have merged several
233    /// origins during themelios canonicalization.
234    #[must_use]
235    pub fn template_origins(&self) -> &[Vec<Location>] {
236        &self.template_origins
237    }
238
239    /// Original source declarations and display selection. Display metadata
240    /// never changes the independently admitted program or model identity.
241    #[must_use]
242    pub fn metadata(&self) -> &SourceMetadata {
243        &self.metadata
244    }
245
246    /// Expansion charges this admission accepted, each under its ceiling.
247    #[must_use]
248    pub fn expansion_usage(&self) -> &ExpansionUsage {
249        &self.expansion
250    }
251
252    /// Consume the boundary value when source evidence is no longer required.
253    #[must_use]
254    pub fn into_program(self) -> Program {
255        self.program
256    }
257}

259/// Admit UTF-8 source under S0 using the clingo dialect. Refuses resource-limit
260/// violations, every parser/raiser diagnostic, unsupported source forms, and
261/// core validation failures. No recovered or truncated program is returned.
262/// Work before core admission is bounded by source bytes and syntax ceilings.
263///
264/// # Errors
265/// Returns the typed source, syntax, profile, construction, or core admission
266/// refusal at the first failing stage, preserving that stage's diagnostics.
267pub fn admit(text: String, options: AdmissionOptions) -> Result<Admitted, AdmissionFailure> {
268    let source = ParsedSource::new(text, options)?;
269    profile::check(source.parsed(), options)?;
270    let raised = raise(source.parsed());
271    if !raised.diagnostics().is_empty() {
272        return Err(AdmissionFailure::Raise(raised.diagnostics().to_vec()));
273    }
274    let source_location = Location {
275        source: source.source().id(),
276        span: source.source().span(),
277    };
278    let (mut templates, mut template_origins) =
279        compile::program(raised.program(), source_location)?;
280    coherence::append(
281        &mut templates,
282        &mut template_origins,
283        options.core_limits,
284        source_location,
285        |_, _, _| Ok::<(), AdmissionFailure>(()),
286    )?;
287    let program = Program::new(templates, options.core_limits).map_err(|error| {
288        let location = error
289            .template_index()
290            .and_then(|index| template_origins.get(index))
291            .and_then(|origins| origins.first())
292            .copied()
293            .unwrap_or(source_location);
294        AdmissionFailure::Core {
295            error,
296            location: location.into(),
297        }
298    })?;
299    Ok(Admitted {
300        program,
301        source: source.into_source(),
302        template_origins,
303        metadata: SourceMetadata::default(),
304        expansion: ExpansionUsage::default(),
305    })
306}
