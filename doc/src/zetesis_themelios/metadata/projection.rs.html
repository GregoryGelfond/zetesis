9/// Source declarations for explicit projected enumeration. This is not a
10/// completed projection domain and does not alter ordinary full-family APIs.
11#[derive(Clone, Debug, Default, PartialEq, Eq)]
12pub struct ProjectSelection {
13    pub(crate) explicit: bool,
14    signatures: SignatureSet,
15    pub(crate) conditional_atoms: bool,
16}
17impl ProjectSelection {
18    /// Whether any `#project` directive was authored, including an empty domain.
19    #[must_use]
20    pub fn is_explicit(&self) -> bool {
21        self.explicit
22    }
23    /// Sorted distinct signed predicate selectors.
24    #[must_use]
25    pub fn signatures(&self) -> Signatures<'_> {
26        self.signatures.view()
27    }
28    /// Whether atom/body declarations require complete source support grounding.
29    #[must_use]
30    pub fn has_conditional_atoms(&self) -> bool {
31        self.conditional_atoms
32    }
33}
34
35#[derive(Default)]
36pub(super) struct Builder {
37    explicit: bool,
38    signatures: Vec<Predicate>,
39    conditional_atoms: bool,
40}
41impl Builder {
42    pub(crate) fn signature(&mut self, predicate: Predicate) {
43        self.explicit = true;
44        self.signatures.push(predicate);
45    }
46    pub(crate) fn atom(&mut self) {
47        self.explicit = true;
48        self.conditional_atoms = true;
49    }
50    pub(super) fn finish(self, vocabulary: Option<Arc<MetadataVocabulary>>) -> ProjectSelection {
51        ProjectSelection {
52            explicit: self.explicit,
53            signatures: SignatureSet::publish(vocabulary, self.signatures),
54            conditional_atoms: self.conditional_atoms,
55        }
56    }
57}

59/// Fixed original-atom domain completed against one prepared source owner.
60/// Conditional declarations are grounded once; conditions are not reevaluated
61/// against individual answers. A projected result remains a full answer set.
62#[derive(Clone, Debug, Default)]
63pub struct PreparedProjection {
64    pub(crate) explicit: bool,
65    pub(crate) atoms: AtomCatalog,
66}
67impl PreparedProjection {
68    /// Whether source directives explicitly requested this domain.
69    #[must_use]
70    pub fn is_explicit(&self) -> bool {
71        self.explicit
72    }
73    /// Sorted unique full typed atoms defining projection identity. Positions
74    /// belong to this fixed domain, not to a solver's atom catalog.
75    #[must_use]
76    pub fn atoms(&self) -> Atoms<'_> {
77        self.atoms.atoms()
78    }
79    /// Whether this original atom belongs to the fixed domain.
80    #[must_use]
81    pub fn contains<'a>(&self, atom: impl Into<AtomRef<'a>>) -> bool {
82        self.atoms.atoms().binary_search(atom.into()).is_ok()
83    }
84}
85
86impl PartialEq for PreparedProjection {
87    fn eq(&self, other: &Self) -> bool {
88        self.explicit == other.explicit && self.atoms() == other.atoms()
89    }
90}
91impl Eq for PreparedProjection {}
