16/// Independent named capacity for one compiled metadata bundle. This is not
17/// cumulative source text, observation output, or process resident memory.
18#[derive(Clone, Copy, Debug)]
19pub struct MetadataStorageLimits {
20    /// Canonical vocabulary and component capacities, including their named
21    /// construction and publication overlap. Source topology, compiler maps and
22    /// temporary coordinate vectors retain their independent logical admission
23    /// limits; allocator bookkeeping and process resident memory are excluded.
24    pub max_bytes: usize,
25}
26impl Default for MetadataStorageLimits {
27    fn default() -> Self {
28        Self {
29            max_bytes: 64 * 1024 * 1024,
30        }
31    }
32}
33
34/// Typed refusal while constructing or publishing compiled metadata.
35#[derive(Clone, Debug, PartialEq, Eq)]
36pub enum MetadataStorageError {
37    /// Canonical component identity, capacity, allocation or failed transaction.
38    Components(TemplateCatalogFailure),
39    /// Scoped child/constant metadata or canonical term construction refused.
40    Assignment(AssignmentError),
41}
42impl fmt::Display for MetadataStorageError {
43    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
44        match self {
45            Self::Components(error) => error.fmt(formatter),
46            Self::Assignment(error) => error.fmt(formatter),
47        }
48    }
49}
50impl std::error::Error for MetadataStorageError {}
51impl From<TemplateCatalogFailure> for MetadataStorageError {
52    fn from(error: TemplateCatalogFailure) -> Self {
53        Self::Components(error)
54    }
55}
56impl From<StorageError> for MetadataStorageError {
57    fn from(error: StorageError) -> Self {
58        Self::Components(error.into())
59    }
60}

77impl MetadataVocabulary {
78    pub(crate) fn read_with<E>(
79        &self,
80        before: impl FnMut() -> Result<(), E>,
81    ) -> Result<Read<'_>, TemplateCatalogFailure<E>> {
82        let catalog = self.vocabulary.read();
83        self.components
84            .bind_with(catalog, before)
85            .map(|components| Read {
86                catalog,
87                components,
88            })
89    }

92/// A checked pairing of one immutable vocabulary prefix and its occurrences.
93#[derive(Clone, Copy)]
94pub(crate) struct Read<'a> {
95    catalog: CatalogRead<'a>,
96    components: TemplateComponentsRef<'a>,
97}
98impl<'a> Read<'a> {
99    pub(crate) fn catalog(self) -> CatalogRead<'a> {
100        self.catalog
101    }
102    pub(crate) fn scalar(self, scalar: Scalar) -> Option<TermRef<'a>> {
103        match self.components.term(scalar.0)? {
104            TemplateTerm::Constant(value) => Some(value),
105            TemplateTerm::Variable(_) => None,
106        }
107    }
108    pub(crate) fn constructor(self, constructor: Constructor) -> Option<ValueNodeRef<'a>> {
109        self.components.constructor(constructor.0)
110    }
111    pub(crate) fn constructor_declaration(
112        self,
113        constructor: Constructor,
114    ) -> Option<DeclaredConstructor> {
115        self.components.declared_constructor(constructor.0)
116    }
117    pub(crate) fn predicate(self, predicate: Predicate) -> Option<PredicateRef<'a>> {
118        self.components.predicate(predicate.0)
119    }
120}

122/// The bundle's sole unpublished vocabulary/component authority. Vocabulary and
123/// component capacities share one allowance; caller topology and temporary
124/// coordinate vectors retain their distinct logical limits. Complete nodes may
125/// remain after refusal; components
126/// retain their own poison state and are never published while incomplete.
127#[derive(Debug)]
128pub(crate) struct Authority {
129    builder: VocabularyBuilder,
130    components: TemplateComponents,
131    limits: MetadataStorageLimits,
132    peak: u128,
133}
134const HEADER: usize =
135    size_of::<Authority>() - size_of::<VocabularyBuilder>() - size_of::<TemplateComponents>();
136impl Authority {
137    pub(crate) fn new(
138        limits: MetadataStorageLimits,
139        external: u128,
140    ) -> Result<Self, MetadataStorageError> {
141        let outer = external + HEADER as u128;
142        let builder = VocabularyBuilder::new(remaining(limits, outer)?)
143            .map_err(|error| storage(error, outer, limits))?;
144        let components = TemplateComponents::new(
145            builder.read(),
146            remaining(limits, external + HEADER as u128 + builder.storage_bytes())?,
147        )
148        .map_err(|error| {
149            component(
150                error,
151                external + HEADER as u128 + builder.storage_bytes(),
152                limits,
153            )
154        })?;
155        let mut result = Self {
156            builder,
157            components,
158            limits,
159            peak: 0,
160        };
161        result.observe(external)?;
162        Ok(result)
163    }
164    pub(crate) fn bytes(&self) -> u128 {
165        HEADER as u128 + self.builder.storage_bytes() + self.components.storage_bytes()
166    }
167    pub(crate) fn read(&self) -> Result<Read<'_>, MetadataStorageError> {
168        let catalog = self.builder.read();
169        let components = self.components.bind_with(catalog, READY)?;
170        Ok(Read {
171            catalog,
172            components,
173        })
174    }
175    pub(crate) fn catalog(&self) -> CatalogRead<'_> {
176        self.builder.read()
177    }
178    pub(crate) fn observe(&mut self, external: u128) -> Result<(), MetadataStorageError> {
179        self.peak = self.peak.max(external + self.bytes());
180        check(self.limits, external + self.bytes())
181    }
182    fn writer(&mut self, external: u128) -> Result<u128, MetadataStorageError> {
183        let outer = external + HEADER as u128 + self.components.storage_bytes();
184        self.builder.restart_storage_peak();
185        self.builder
186            .ceiling(remaining(self.limits, outer)?)
187            .map_err(|error| storage(error, outer, self.limits))?;
188        Ok(outer)
189    }
190    fn written<T>(
191        &mut self,
192        result: Result<T, VocabularyFailure<Infallible>>,
193        outer: u128,
194    ) -> Result<T, MetadataStorageError> {
195        self.peak = self.peak.max(outer + self.builder.storage_peak_bytes());
196        result.map_err(|error| match error {
197            VocabularyFailure::Storage(error) => storage(error, outer, self.limits),
198            VocabularyFailure::Stopped(impossible) => match impossible {},
199        })
200    }
201    fn append<T>(
202        &mut self,
203        external: u128,
204        action: impl FnOnce(
205            &mut TemplateComponents,
206            CatalogRead<'_>,
207            usize,
208        ) -> Result<T, TemplateCatalogFailure>,
209    ) -> Result<T, MetadataStorageError> {
210        let outer = external + HEADER as u128 + self.builder.storage_bytes();
211        let available = remaining(self.limits, outer)?;
212        self.components.restart_storage_peak();
213        let result = action(&mut self.components, self.builder.read(), available);
214        self.peak = self.peak.max(outer + self.components.storage_peak_bytes());
215        let value = result.map_err(|error| component(error, outer, self.limits))?;
216        check(self.limits, outer + self.components.storage_peak_bytes())?;
217        Ok(value)
218    }
219    pub(crate) fn construct(
220        &mut self,
221        descriptor: ValueNodeRef<'_>,
222        values: AssignmentSlice<'_>,
223        children: &[usize],
224        logical: Limits,
225        external: u128,
226    ) -> Result<TermKey, MetadataStorageError> {
227        let outer = self.writer(external)?;
228        let result = self
229            .builder
230            .construct_term_with(descriptor, values, children, logical, READY);
231        self.peak = self.peak.max(outer + self.builder.storage_peak_bytes());
232        result.map_err(|error| match error {
233            AssignmentFailure::Assignment(AssignmentError::Storage(error)) => {
234                storage(error, outer, self.limits)
235            }
236            AssignmentFailure::Assignment(error) => MetadataStorageError::Assignment(error),
237            AssignmentFailure::Stopped(impossible) => match impossible {},
238        })
239    }
240    pub(crate) fn scalar(
241        &mut self,
242        key: &TermKey,
243        external: u128,
244    ) -> Result<Scalar, MetadataStorageError> {
245        self.append(external, |components, read, available| {
246            let term = read.term(key).map_err(TemplateCatalogFailure::Read)?;
247            components
248                .append_terms_with(read, [TemplateTerm::Constant(term)], available, READY)
249                .map(|range| Scalar(range.start))
250        })
251    }
252    pub(crate) fn predicate(
253        &mut self,
254        predicate: PredicateRef<'_>,
255        external: u128,
256    ) -> Result<Predicate, MetadataStorageError> {
257        let outer = self.writer(external)?;
258        let result = self.builder.declare_predicate_with(predicate, READY);
259        let predicate = self.written(result, outer)?;
260        self.append(external, |components, read, available| {
261            components
262                .append_predicate_with(read, &predicate, available, READY)
263                .map(Predicate)
264        })
265    }
266    pub(crate) fn constructor(
267        &mut self,
268        descriptor: ValueNodeRef<'_>,
269        external: u128,
270    ) -> Result<Constructor, MetadataStorageError> {
271        let outer = self.writer(external)?;
272        let result = self.builder.declare_constructor_with(descriptor, READY);
273        let constructor = self.written(result, outer)?;
274        self.declared_constructor(&constructor, external)
275    }
276    pub(crate) fn declared_constructor(
277        &mut self,
278        constructor: &DeclaredConstructor,
279        external: u128,
280    ) -> Result<Constructor, MetadataStorageError> {
281        self.append(external, |components, read, available| {
282            components
283                .append_constructor_with(read, constructor, available, READY)
284                .map(Constructor)
285        })
286    }
287    pub(crate) fn negate(
288        &mut self,
289        constructor: Constructor,
290    ) -> Result<Constructor, MetadataStorageError> {
291        let read = self.read()?;
292        let descriptor = read.constructor(constructor).ok_or(StorageError::Shape)?;
293        let ValueNodeRef::Function { sign, .. } = descriptor else {
294            return Err(StorageError::Shape.into());
295        };
296        let sign = match sign {
297            zetesis_core::Sign::Positive => zetesis_core::Sign::Negative,
298            zetesis_core::Sign::Negative => zetesis_core::Sign::Positive,
299        };
300        let declaration = read
301            .constructor_declaration(constructor)
302            .ok_or(StorageError::Shape)?;
303        let declaration = declaration.with_sign(sign).ok_or(StorageError::Shape)?;
304        self.declared_constructor(&declaration, 0)
305    }
306    pub(crate) fn finish(
307        mut self,
308        external: u128,
309    ) -> Result<Arc<MetadataVocabulary>, MetadataStorageError> {
310        self.observe(external)?;
311        // The writer had only its remaining allowance during admission. Its
312        // final operation checks the complete simultaneous publication owner.
313        self.builder.ceiling(self.limits.max_bytes)?;
314        let metadata = external
315            + HEADER as u128
316            + self.components.storage_bytes()
317            + (size_of::<MetadataVocabulary>()
318                - size_of::<Vocabulary>()
319                - size_of::<TemplateComponents>()) as u128;
320        let vocabulary =
321            self.builder
322                .finish_with(metadata, READY)
323                .map_err(|error| match error {
324                    VocabularyFailure::Storage(error) => MetadataStorageError::from(error),
325                    VocabularyFailure::Stopped(impossible) => match impossible {},
326                })?;
327        self.components.bind_with(vocabulary.read(), READY)?;
328        Ok(Arc::new(MetadataVocabulary {
329            vocabulary,
330            components: self.components,
331        }))
332    }
333}
334
335const READY: fn() -> Result<(), Infallible> = || Ok(());
336fn check(limits: MetadataStorageLimits, observed: u128) -> Result<(), MetadataStorageError> {
337    if observed > limits.max_bytes as u128 {
338        Err(StorageError::Storage {
339            required: observed,
340            limit: limits.max_bytes,
341        }
342        .into())
343    } else {
344        Ok(())
345    }
346}
347fn remaining(limits: MetadataStorageLimits, external: u128) -> Result<usize, MetadataStorageError> {
348    check(limits, external)?;
349    Ok(limits.max_bytes
350        - usize::try_from(external).expect("external storage checked against usize ceiling"))
351}
352fn storage(
353    error: StorageError,
354    outer: u128,
355    limits: MetadataStorageLimits,
356) -> MetadataStorageError {
357    match error {
358        StorageError::Storage { required, .. } => StorageError::Storage {
359            required: outer + required,
360            limit: limits.max_bytes,
361        }
362        .into(),
363        error => error.into(),
364    }
365}
366fn component(
367    error: TemplateCatalogFailure,
368    outer: u128,
369    limits: MetadataStorageLimits,
370) -> MetadataStorageError {
371    match error {
372        TemplateCatalogFailure::Storage(error) => storage(error, outer, limits),
373        error => error.into(),
374    }
