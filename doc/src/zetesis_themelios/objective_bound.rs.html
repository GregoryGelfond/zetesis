1//! Reusable objective eligibility, score reads and exact candidate bounds.
2//!
3//! A plan compiles complete positive objective joins over the supplied completed
4//! possible relation. Its atom order must match the original theory, and the
5//! caller establishes coverage of every stable model. Score reads reuse that
6//! eligibility independently of candidate pruning. Constructing an incumbent
7//! bound additionally requires a verified incumbent. Neither operation solves
8//! a source, establishes answer-set membership or changes its frozen reduct.

28/// Inclusive construction limits; zero is a real ceiling.
29#[derive(Clone, Copy, Debug)]
30pub struct ObjectivePlanLimits {
31    /// Supplied original semantic atoms. Also bounds the requested lengths of
32    /// two integer lookup orders and one reusable merge-scratch vector. Each
33    /// reservation is fallible; allocator slack and atom payload are not a
34    /// process memory ceiling.
35    pub max_atoms: usize,
36    /// Complete positive relational bindings before filters.
37    pub max_bindings: u64,
38    /// Distinct global priority/weight/tuple keys.
39    pub max_keys: usize,
40    /// Logical retained key payload, including scalar tags and string bytes.
41    pub max_key_bytes: usize,
42    /// Explicit scalar tuple width per objective key.
43    pub max_tuple_width: usize,
44    /// Eligibility DAG nodes, contribution rows and threshold descriptors, each.
45    pub max_nodes: usize,
46    /// Logical child occurrences, Boolean operands, conditions and weights, each.
47    pub max_operands: usize,
48    /// Local binding slots per lifted objective.
49    pub max_variables: usize,
50    /// Positive conditions per lifted objective.
51    pub max_body_atoms: usize,
52    /// Charged construction operations, including prepared-index comparisons
53    /// and writes, query probes and visited scalar descriptors/text bytes.
54    pub max_work: u64,
55}

132/// A refusal to construct an optional optimization, never a logical result.
133#[derive(Clone, Copy, Debug, PartialEq, Eq)]
134pub enum ObjectiveBoundErrorKind {
135    /// Cancellation or deadline.
136    Control(Stop),
137    /// Explicit construction ceiling.
138    Limit(ObjectiveBoundResource),
139    /// The atom catalog length differs from the original theory or contains duplicate atoms.
140    AtomCatalog,
141    /// Choice premises belong to a different admitted theory.
142    ChoiceOwner,
143    /// Optional source choice capture was incomplete. The complete source
144    /// receipt remains available through [`crate::RequiredChoices::capture_failure`].
145    ChoiceCapture,
146    /// An admitted objective variable unexpectedly remained unbound.
147    UnboundVariable,
148    /// A negated numeric contribution cannot be represented as i32.
149    WeightNormalizationOverflow,
150    /// A checked size cannot be represented.
151    Overflow,
152    /// Fallible allocation failed.
153    Allocation,
154    /// Exact aggregate lowering declined within its own explicit ceilings.
155    Aggregate(AggregateError),
156    /// Final candidate formula admission failed.
157    Theory(zetesis_ferraris::AdmissionError),
158}

171        self.kind
172    }

174    #[must_use]

176        self.template
177    }
178    /// Partial logical accounting without a completed constraint.
179    #[must_use]

181        self.statistics
182    }
183}

191    }

195/// Reusable exact positive eligibility formulas, independent of any incumbent.
196#[derive(Debug)]
197pub struct ObjectivePlan {
198    identity: PlanIdentity,
199    original: Theory,
200    objectives: ObjectiveProgram,
201    nodes: FormulaParts,
202    levels: BTreeMap<i32, Vec<AggregateElement>>,
203    choices: Option<choices::Costs>,
204    statistics: ObjectiveBoundStatistics,
205}
206impl ObjectivePlan {
207    /// Prepare necessary nonnegative choice costs without changing exact scores.
208    ///
209    /// A required group prepays its least certified objective weight under its
210    /// activation; allocated keys retain their excess over that weight. Groups
211    /// never share a coalesced objective key.
212    /// Unsupported bridges remain unproved, not false. The result is optional:
213    /// refusal preserves the exact plan and any preceding successful preparation.
214    /// Success and error statistics count this attempt separately from [`Self::statistics`].
215    /// `max_bytes` bounds source premises, retained group costs and temporary
216    /// proof/index capacities together. It is an independent allowance, not RSS.
217    ///
218    /// # Errors
219    /// Reports foreign premises, cancellation, allocation or preparation limits.
220    pub fn prepare_choice_bounds(
221        &mut self,
222        required: &crate::RequiredChoices,
223        limits: ObjectivePlanLimits,
224        max_bytes: u128,
225        cancellation: &Cancellation,
226    ) -> Result<ObjectiveBoundStatistics, ObjectiveBoundError> {
227        let (costs, statistics) =
228            choices::prepare(self, required, limits, max_bytes, cancellation)?;
229        self.choices = costs;
230        Ok(statistics)
231    }
232
233    /// Source groups whose costs strengthen the optional incumbent bound.
234    #[must_use]
235    pub fn choice_bound_groups(&self) -> usize {
236        self.choices.as_ref().map_or(0, |costs| costs.groups.len())
237    }
238    /// Compile all possible numeric objective keys and their exact eligibility.
239    /// `atoms` must be the original theory's complete, unique atom
240    /// catalog; each catalog index is reused verbatim. This API does not prove
241    /// that the supplied relation covers all stable models.
242    ///
243    /// Preparation borrows the atoms and builds an `AtomIndex` containing only
244    /// row positions. The atom ceiling is checked before its fallible linear
245    /// storage reservations. Both index construction and later binary searches
246    /// charge the same work owner; preparation is not free. Predicate ranges
247    /// preserve original catalog order within a predicate, while full-key
248    /// membership returns the original dense identity. The temporary index is
249    /// dropped after planning; the resulting plan retains its eligibility DAG.
250    ///
251    /// All comparisons use checked canonical typed identity, including sign,
252    /// arity and structural values. This is separate from ASP term ordering.
253    /// A missing closed-condition atom is false without extending the catalog.
254    /// Work cutoffs reflect actual preparation/traversal, so a changed algorithm
255    /// can refuse at a different numeric cutoff while preserving typed failure
256    /// and the performed prefix. No partly constructed plan is returned.
257    ///
258    /// # Errors
259    /// Refuses invalid catalogs, numeric normalization overflow, resource ceilings,
260    /// cancellation or allocation.
261    pub fn new(
262        original: &Theory,
263        atoms: Atoms<'_>,
264        objectives: &ObjectiveProgram,
265        limits: ObjectivePlanLimits,
266        cancellation: &Cancellation,
267    ) -> Result<Self, ObjectiveBoundError> {
268        join::compile(original, atoms, objectives, limits, cancellation)
269    }
270    /// Original theory instance, unchanged by planning.
271    #[must_use]
272    pub fn original(&self) -> &Theory {
273        &self.original
274    }
275    /// This plan's identity, fixed when it was compiled. Its bounds are
276    /// installed in a candidate stream under it, which compares their scores
277    /// only with other bounds of the same plan.
278    #[must_use]
279    pub const fn identity(&self) -> PlanIdentity {
280        self.identity
281    }
282    /// Complete planning accounting.
283    #[must_use]
284    pub const fn statistics(&self) -> ObjectiveBoundStatistics {
285        self.statistics
286    }
287    /// Compile lexicographic cost less than or equal to `incumbent`, retaining
288    /// all ties. Missing priority slots mean zero, as in `Score::compare_costs`.
289    /// The caller must independently verify the incumbent before using this to
290    /// prune a search. Apply the result only to classical candidates, never to
291    /// the original theory or its proper-subset reduct query.
292    /// Prepared choice consequences may exclude additional interpretations
293    /// outside the original theory. A better incumbent's bound implies an
294    /// earlier bound under that original theory, including when an optional
295    /// strengthening fails and returns only the exact cost condition.
296    ///
297    /// # Errors
298    /// Refuses bounded exact lowering, cancellation, storage or work limits.
299    pub fn bound(
300        &self,
301        incumbent: &Score,
302        limits: ObjectiveBoundLimits,
303        cancellation: &Cancellation,
304    ) -> Result<ObjectiveBound, ObjectiveBoundError> {
305        bound::compile(
306            self,
307            incumbent,
308            bound::Comparison::AtMost,
309            limits,
310            cancellation,
311        )
312    }
313
314    /// Compile lexicographic cost strictly less than `incumbent`, so that no
315    /// tie at the incumbent's cost survives, under the same conditions as
316    /// [`Self::bound`]. A better incumbent's strict bound implies an earlier
317    /// strict or non-strict bound under the original theory, as non-strict
318    /// bounds do. The optional choice strengthening compares certified lower
319    /// costs strictly too: a model's lower cost never exceeds its cost.
320    ///
321    /// # Errors
322    /// As [`Self::bound`].
323    pub fn strict_bound(
324        &self,
325        incumbent: &Score,
326        limits: ObjectiveBoundLimits,
327        cancellation: &Cancellation,
328    ) -> Result<ObjectiveBound, ObjectiveBoundError> {
329        bound::compile(
330            self,
331            incumbent,
332            bound::Comparison::Below,
333            limits,
334            cancellation,
335        )
336    }
337}

370struct Work<'a> {
371    cancellation: &'a Cancellation,
372    limits: ObjectivePlanLimits,
373    template: Option<usize>,
374    meter: Meter<'static>,
375    bindings: u64,
376    keys: usize,
377    key_bytes: usize,
378    nodes: usize,
379}
380impl<'a> Work<'a> {
381    fn new(limits: ObjectivePlanLimits, cancellation: &'a Cancellation) -> Self {
382        Self {
383            cancellation,
384            limits,
385            template: None,
386            meter: Meter::local(limits.max_work),
387            bindings: 0,
388            keys: 0,
389            key_bytes: 0,
390            nodes: 0,
391        }
392    }
393    fn statistics(&self) -> ObjectiveBoundStatistics {
394        ObjectiveBoundStatistics {
395            work: self.meter.charged(),
396            bindings: self.bindings,
397            keys: self.keys,
398            key_bytes: self.key_bytes,
399            nodes: self.nodes,
400        }
401    }
402    fn error(&self, kind: ObjectiveBoundErrorKind) -> ObjectiveBoundError {
403        ObjectiveBoundError {
404            kind,
405            template: self.template,
406            statistics: self.statistics(),
407        }
408    }
409    fn limit(&self, resource: ObjectiveBoundResource) -> ObjectiveBoundError {
410        self.error(ObjectiveBoundErrorKind::Limit(resource))
411    }
412    fn charge(&mut self, amount: u64) -> Result<(), ObjectiveBoundError> {
413        self.cancellation
414            .poll()
415            .map_err(|error| self.error(ObjectiveBoundErrorKind::Control(error)))?;
416        self.account(amount)
417    }
418    // Imported kernel work has already happened, including on cancellation.
419    // Record it without another cancellation check discarding that evidence.
420    fn account(&mut self, amount: u64) -> Result<(), ObjectiveBoundError> {
421        // The native contract distinguishes an unrepresentable total from an
422        // exhausted allowance; the shared meter owns the accepted count.
423        if self.meter.charged().checked_add(amount).is_none() {
424            return Err(self.error(ObjectiveBoundErrorKind::Overflow));
425        }
426        self.meter
427            .bounded(amount, || ((), amount))
428            .map_err(|failure| match failure.into_stop() {
429                Stop::WorkLimit => self.limit(ObjectiveBoundResource::Work),
430                stop => self.error(ObjectiveBoundErrorKind::Control(stop)),
431            })
432    }
433    fn tick(&mut self) -> Result<(), ObjectiveBoundError> {
434        self.charge(1)
435    }
436    fn reserve<T>(&self, count: usize) -> Result<Vec<T>, ObjectiveBoundError> {
437        let mut values = Vec::new();
438        values
439            .try_reserve_exact(count)
440            .map_err(|_| self.error(ObjectiveBoundErrorKind::Allocation))?;
441        Ok(values)
442    }
443    fn node(
444        &mut self,
445        nodes: &mut FormulaNodes,
446        node: NodeView<'_>,
447    ) -> Result<usize, ObjectiveBoundError> {
448        self.tick()?;
449        let (added_nodes, edges) = match node {
450            NodeView::And([]) => (2, 2),
451            NodeView::And([_]) | NodeView::Or([_]) => (0, 0),
452            NodeView::And(row) | NodeView::Or(row) => (1, row.len()),
453            NodeView::Threshold(row) => (1, row.conditions.len()),
454            NodeView::ExtremumCut(cut) => (1, cut.row.conditions.len()),
455            NodeView::Implies(_, _) => (1, 2),
456            NodeView::Atom(_) | NodeView::False => (1, 0),
457        };
458        Meter::check_count(nodes.view().len(), added_nodes, self.limits.max_nodes).map_err(
459            |error| match error {
460                MeterCountError::Overflow => self.error(ObjectiveBoundErrorKind::Overflow),
461                MeterCountError::Limit { .. } => self.limit(ObjectiveBoundResource::Nodes),
462            },
463        )?;
464        Meter::check_count(nodes.parts().occurrences(), edges, self.limits.max_operands).map_err(
465            |error| match error {
466                MeterCountError::Overflow => self.error(ObjectiveBoundErrorKind::Overflow),
467                MeterCountError::Limit { .. } => self.limit(ObjectiveBoundResource::Operands),
468            },
469        )?;
470        // Charge child validation and copied arena cells before publication.
471        // Numeric copies include both columns and row/threshold descriptors.
472        // The range-returning append allocates no root vector. The builder has
473        // no independent work owner.
474        let copies = match node {
475            NodeView::And([_]) | NodeView::Or([_]) => 1,
476            NodeView::And(row) | NodeView::Or(row) if row.len() >= 3 => row.len(),
477            NodeView::Threshold(row) => row
478                .conditions
479                .len()
480                .checked_mul(2)
481                .and_then(|cells| cells.checked_add(2))
482                .ok_or_else(|| self.error(ObjectiveBoundErrorKind::Overflow))?,
483            NodeView::ExtremumCut(cut) => cut
484                .row
485                .conditions
486                .len()
487                .checked_mul(2)
488                .and_then(|cells| cells.checked_add(2))
489                .ok_or_else(|| self.error(ObjectiveBoundErrorKind::Overflow))?,
490            _ => 0,
491        };
492        if matches!(node, NodeView::Threshold(_) | NodeView::ExtremumCut(_)) {
493            self.charge(
494                u64::try_from(nodes.parts().pending_numeric_validation_work())
495                    .map_err(|_| self.error(ObjectiveBoundErrorKind::Overflow))?,
496            )?;
497        }
498        // Ordinal publication also validates its independently supplied cut rule.
499        if matches!(node, NodeView::ExtremumCut(_)) {
500            self.tick()?;
501        }
502        for _ in 0..edges {
503            self.tick()?;
504        }
505        for _ in 0..copies {
506            self.tick()?;
507        }
508        let mut transaction = nodes.transaction();
509        let index = transaction
510            .push(node, self.limits.max_nodes, self.limits.max_operands)
511            .map_err(|error| {
512                self.error(match error {
513                    zetesis_ferraris::AdmissionError::Allocation => {
514                        ObjectiveBoundErrorKind::Allocation
515                    }
516                    _ => ObjectiveBoundErrorKind::Theory(error),
517                })
518            })?;
519        transaction.commit();
520        self.nodes = nodes.view().len();
521        Ok(index)
522    }
