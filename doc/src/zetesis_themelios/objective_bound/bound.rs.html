17pub(super) fn compile(
18    plan: &ObjectivePlan,
19    incumbent: &Score,
20    limits: ObjectiveBoundLimits,
21    cancellation: &Cancellation,
22) -> Result<ObjectiveBound, ObjectiveBoundError> {
23    let mut work = Work {
24        cancellation,
25        limits: ObjectivePlanLimits {
26            max_nodes: limits.aggregate.max_nodes,
27            max_work: limits.max_work,
28            ..ObjectivePlanLimits::default()
29        },
30        template: None,
31        statistics: ObjectiveBoundStatistics::default(),
32    };
33    work.tick()?;
34    let mut nodes = Vec::new();
35    for node in &plan.nodes {
36        work.node(&mut nodes, *node)?;
37    }
38    let falsum = work.node(&mut nodes, Node::False)?;
39    let mut root = work.node(&mut nodes, Node::Implies(falsum, falsum))?;
40    let mut costs = BTreeMap::new();
41    for priority in plan.levels.keys() {
42        work.tick()?;
43        costs.insert(*priority, 0);
44    }
45    for (priority, value) in incumbent.costs() {
46        work.tick()?;
47        costs.insert(*priority, *value);
48    }
49    for (priority, bound) in costs {
50        work.tick()?;
51        let (elements, bound) = normalize::prepare(
52            plan.levels.get(&priority).map_or(&[], Vec::as_slice),
53            bound,
54            falsum,
55            &mut nodes,
56            limits,
57            &mut work,
58        )?;
59        let build = family(&mut nodes, &elements, bound, limits, &mut work)?;
60        let suffix = work.node(&mut nodes, Node::And(build.roots()[1], root))?;
61        root = work.node(&mut nodes, Node::Or(build.roots()[0], suffix))?;
62    }
63    // Account for the final independent topology validation before admitting it.
64    for _ in &nodes {
65        work.tick()?;
66    }
67    let theory = Theory::new(
68        plan.original.atom_count(),
69        nodes,
70        vec![root],
71        AdmissionLimits {
72            max_atoms: plan.original.atom_count(),
73            max_nodes: limits.aggregate.max_nodes,
74            max_roots: 1,
75        },
76    )
77    .map_err(|error| work.error(Kind::Theory(error)))?;
78    Ok(ObjectiveBound {
79        original: plan.original.clone(),
80        theory,
81        statistics: work.statistics,
82    })
83}

85fn family(
86    nodes: &mut Vec<Node>,
87    elements: &[AggregateElement],
88    bound: i64,
89    limits: ObjectiveBoundLimits,
90    work: &mut Work<'_>,
91) -> Result<AggregateFamilyBuild, ObjectiveBoundError> {
92    let guards = [
93        AggregateGuard {
94            comparison: AggregateComparison::Lt,
95            bound,
96        },
97        AggregateGuard {
98            comparison: AggregateComparison::Eq,
99            bound,
100        },
101    ];
102    let mut aggregate = limits.aggregate;
103    aggregate.max_work = aggregate
104        .max_work
105        .min(limits.max_work - work.statistics.work);
106    let result = append_aggregate_family(
107        nodes,
108        elements,
109        &guards,
110        AggregateFamilyLimits {
111            aggregate,
112            max_guards: 2,
113        },
114        work.cancellation,
115    );
116    let build = match result {
117        Ok(build) => build,
118        Err(error) => {
119            work.account(error.statistics().work)?;
120            return Err(work.error(Kind::Aggregate(error)));
121        }
122    };
123    work.account(build.statistics().work)?;
124    work.statistics.nodes = nodes.len();
125    Ok(build)
126}
