1//! Classical candidate guards only: never rewrite the original reduct theory.
2
3use std::borrow::Cow;
4
5use zetesis_ferraris::{AggregateElement, FormulaNodes, NodeView};
6
7use super::{Kind, ObjectiveBoundError, ObjectiveBoundLimits, Work};
8
9#[derive(Default)]
10struct Shape {
11    offset: i128,
12    total: i128,
13    negative: usize,
14    nonzero: usize,
15    minimum: Option<i128>,
16}
17
18pub(in crate::objective_bound) fn prepare<'a>(
19    elements: &'a [AggregateElement],
20    bound: i64,
21    falsum: usize,
22    nodes: &mut FormulaNodes,
23    limits: ObjectiveBoundLimits,
24    work: &mut Work<'_>,
25) -> Result<(Cow<'a, [AggregateElement]>, i64), ObjectiveBoundError> {
26    let original = || (Cow::Borrowed(elements), bound);
27    work.tick()?;
28    // Leave the kernel's typed element refusal intact, before any temporary copy.
29    if elements.len() > limits.aggregate.max_elements {
30        return Ok(original());
31    }
32    let mut shape = Shape::default();
33    for element in elements {
34        work.tick()?;
35        let weight = i128::from(element.weight);
36        if element.weight == i32::MIN {
37            return Ok(original());
38        }
39        if weight < 0 {
40            shape.offset = shape
41                .offset
42                .checked_add(weight)
43                .ok_or_else(|| work.error(Kind::Overflow))?;
44            shape.negative += 1;
45        }
46        if weight != 0 {
47            shape.nonzero += 1;
48            shape.minimum = Some(shape.minimum.map_or(weight.abs(), |n| n.min(weight.abs())));
49        }
50        shape.total = shape
51            .total
52            .checked_add(weight.abs())
53            .ok_or_else(|| work.error(Kind::Overflow))?;
54    }
55    if shape.negative == 0 {
56        return Ok(original());
57    }
58    let shifted = i128::from(bound)
59        .checked_sub(shape.offset)
60        .ok_or_else(|| work.error(Kind::Overflow))?;
61    let Ok(shifted) = i64::try_from(shifted) else {
62        return Ok(original());
63    };
64    work.tick()?;
65    if !shape.fits(shifted, limits) {
66        return Ok(original());
67    }
68    let mut normalized = work.reserve(elements.len())?;
69    for element in elements {
70        work.tick()?;
71        normalized.push(if element.weight < 0 {
72            AggregateElement {
73                weight: element
74                    .weight
75                    .checked_neg()
76                    .ok_or_else(|| work.error(Kind::Overflow))?,
77                condition: work.node(nodes, NodeView::Implies(element.condition, falsum))?,
78            }
79        } else {
80            *element
81        });
82    }
83    // Each entry is a distinct original complete key, even if its transformed
84    // weight or eligibility matches another entry. Never deduplicate this list.
85    Ok((Cow::Owned(normalized), shifted))
86}
87
88impl Shape {
89    fn fits(&self, shifted: i64, limits: ObjectiveBoundLimits) -> bool {
90        // Lt(k), Eq(k) request G(k), G(k), G(k+1). A family with any
91        // interior threshold keeps all nonconstant guards on its numeric row
92        // so propagation can use their ordering. Pure Boolean families need
93        // no numeric state.
94        let minimum = self.minimum.unwrap_or(0);
95        let shifted = i128::from(shifted);
96        let bounds = [shifted, shifted, shifted + 1];
97        let numeric = bounds
98            .into_iter()
99            .any(|bound| bound > minimum && bound <= self.total - minimum);
100        let thresholds = bounds
101            .into_iter()
102            .filter(|&bound| bound > 0 && bound <= self.total)
103            .count();
104        let states = if numeric {
105            2 * self.nonzero as u128 + thresholds as u128
106        } else {
107            0
108        };
109        // Only representation compatibility decides this choice. Actual node,
110        // operand and work admission belongs to the shared builder, so an
111        // estimate cannot reject a construction at its exact inclusive limit.
112        states <= limits.aggregate.max_states as u128
113    }
