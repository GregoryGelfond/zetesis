1//! Classical candidate guards only: never rewrite the original reduct theory.
2
3use std::borrow::Cow;
4
5use zetesis_ferraris::{AggregateElement, FormulaNodes, NodeView};
6
7use super::{Kind, ObjectiveBoundError, ObjectiveBoundLimits, Work};
8
9#[derive(Default)]
10struct Shape {
11    offset: i128,
12    total: i128,
13    negative: usize,
14    nonzero: usize,
15}
16
17pub(super) fn prepare<'a>(
18    elements: &'a [AggregateElement],
19    bound: i64,
20    falsum: usize,
21    nodes: &mut FormulaNodes,
22    limits: ObjectiveBoundLimits,
23    work: &mut Work<'_>,
24) -> Result<(Cow<'a, [AggregateElement]>, i64), ObjectiveBoundError> {
25    let original = || (Cow::Borrowed(elements), bound);
26    work.tick()?;
27    // Leave the kernel's typed element refusal intact, before any temporary copy.
28    if elements.len() > limits.aggregate.max_elements {
29        return Ok(original());
30    }
31    let mut shape = Shape::default();
32    for element in elements {
33        work.tick()?;
34        let weight = i128::from(element.weight);
35        if element.weight == i32::MIN {
36            return Ok(original());
37        }
38        if weight < 0 {
39            shape.offset = shape
40                .offset
41                .checked_add(weight)
42                .ok_or_else(|| work.error(Kind::Overflow))?;
43            shape.negative += 1;
44        }
45        shape.nonzero += usize::from(weight != 0);
46        shape.total = shape
47            .total
48            .checked_add(weight.abs())
49            .ok_or_else(|| work.error(Kind::Overflow))?;
50    }
51    if shape.negative == 0 {
52        return Ok(original());
53    }
54    let shifted = i128::from(bound)
55        .checked_sub(shape.offset)
56        .ok_or_else(|| work.error(Kind::Overflow))?;
57    let Ok(shifted) = i64::try_from(shifted) else {
58        return Ok(original());
59    };
60    work.tick()?;
61    if !shape.fits(elements.len(), shifted, nodes, limits, work) {
62        return Ok(original());
63    }
64    let mut normalized = work.reserve(elements.len())?;
65    for element in elements {
66        work.tick()?;
67        normalized.push(if element.weight < 0 {
68            AggregateElement {
69                weight: element
70                    .weight
71                    .checked_neg()
72                    .ok_or_else(|| work.error(Kind::Overflow))?,
73                condition: work.node(nodes, NodeView::Implies(element.condition, falsum))?,
74            }
75        } else {
76            *element
77        });
78    }
79    // Each entry is a distinct original complete key, even if its transformed
80    // weight or eligibility matches another entry. Never deduplicate this list.
81    Ok((Cow::Owned(normalized), shifted))
82}
83
84impl Shape {
85    fn fits(
86        &self,
87        count: usize,
88        shifted: i64,
89        prefix: &FormulaNodes,
90        limits: ObjectiveBoundLimits,
91        work: &Work<'_>,
92    ) -> bool {
93        let shifted = i128::from(shifted);
94        // Exact maximum requested by this caller's Lt(k), Eq(k) pair. Widen
95        // before the successor; out-of-range thresholds are Boolean constants.
96        let maximum = if shifted >= 0 && shifted < self.total {
97            shifted + 1
98        } else if shifted > 0 && shifted == self.total {
99            shifted
100        } else {
101            0
102        };
103        let Ok(maximum) = u128::try_from(maximum) else {
104            return false;
105        };
106        let count = count as u128;
107        let negative = self.negative as u128;
108        let prefix_nodes = prefix.view().len() as u128;
109        let prefix_operands = prefix.parts().occurrences() as u128;
110        let cells = maximum * self.nonzero as u128;
111        let states = if maximum == 0 { 0 } else { 2 * (maximum + 1) };
112        // Reject before multiplying the work estimate; selected cell counts
113        // are bounded by the caller's addressable node capacity.
114        if states > limits.aggregate.max_states as u128
115            || cells > limits.aggregate.max_nodes as u128 / 2
116        {
117            return false;
118        }
119        // Two scans of 2^n subsets are mandatory on the signed path, independent
120        // of which subsets fail. Cell count is a scheduling heuristic, not a
121        // runtime guarantee; an unrepresentable subset count is already refused.
122        let subset_visits = u32::try_from(count)
123            .ok()
124            .and_then(|power| 1u128.checked_shl(power))
125            .and_then(|subsets| subsets.checked_mul(2))
126            .unwrap_or(u128::MAX);
127        let nodes = prefix_nodes + negative + 2 * cells + 7;
128        let operands = prefix_operands + 2 * negative + 4 * cells + 12;
129        let row_work = if maximum == 0 {
130            0
131        } else {
132            maximum + 1 + count + 7 * cells
133        };
134        // Checked construction retains prefix validation. Charge input visits,
135        // constants, both guard scans and their connective operand visits.
136        let family_work = 18 + count + row_work;
137        let normalize_work = count + 3 * negative;
138        let remaining = u128::from(limits.max_work - work.statistics.work);
139        cells <= subset_visits
140            && nodes <= limits.aggregate.max_nodes as u128
141            && operands <= limits.aggregate.max_operands as u128
142            && family_work <= u128::from(limits.aggregate.max_work)
143            // Two suffix appends; admission recounts nodes, checks topology
144            // and visits the sole root. Each edge occurrence remains charged.
145            && normalize_work + family_work + 6 + 2 * nodes + operands < remaining
146    }
147}
