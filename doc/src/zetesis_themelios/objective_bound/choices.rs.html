1//! Necessary costs from complete source choices and checked forward implications.
2//!
3//! A selected member must imply one allocated objective key. Every key is
4//! globally coalesced before this pass, and groups allocate disjoint key sets.
5//! Prepay each active group's minimum and retain every eligible key's excess
6//! over that minimum. The original cost covers this sum whenever an active
7//! group has at least one eligible allocated key.
8//! Cyclic or unsupported implication searches decline a proof; they never prove
9//! absence. The resulting restriction is only for original candidates.

49pub(super) fn prepare(
50    plan: &ObjectivePlan,
51    required: &crate::RequiredChoices,
52    limits: ObjectivePlanLimits,
53    max_bytes: u128,
54    cancellation: &Cancellation,
55) -> Result<(Option<Costs>, ObjectiveBoundStatistics), ObjectiveBoundError> {
56    let mut work = Work {
57        limits,
58        cancellation,
59        template: None,
60        statistics: ObjectiveBoundStatistics::default(),
61    };
62    work.tick()?;
63    if required.capture_failure().is_some() {
64        return Err(work.error(Kind::ChoiceCapture));
65    }
66    if !required.belongs_to(&plan.original) {
67        return Err(work.error(Kind::ChoiceOwner));
68    }
69    if required.groups.len() > limits.max_keys {
70        return Err(work.limit(Resource::Keys));
71    }
72    let mut required_member = false;
73    for group in &required.groups {
74        work.tick()?;
75        required_member |= group.lower > 0 && !group.members.is_empty();
76    }
77    let mut eligible_level = false;
78    for elements in plan.levels.values() {
79        let mut nonnegative = true;
80        let mut positive = false;
81        for element in elements {
82            work.tick()?;
83            nonnegative &= element.weight >= 0;
84            positive |= element.weight > 0;
85        }
86        eligible_level |= nonnegative && positive;
87    }
88    if !required_member || !eligible_level {
89        return Ok((None, work.statistics));
90    }
91    let mut memory = Storage::new(max_bytes, required, plan.choices.as_ref(), &work)?;
92    let mut bridges = Bridges::new(plan, required, &mut memory, &mut work)?;
93    let mut levels = memory.reserve(plan.levels.len(), &work)?;
94    for (&priority, elements) in &plan.levels {
95        work.tick()?;
96        let mut nonnegative = true;
97        for element in elements {
98            work.tick()?;
99            nonnegative &= element.weight >= 0;
100        }
101        if !nonnegative {
102            continue;
103        }
104        let mut allocated = memory.reserve(elements.len(), &work)?;
105        allocated.resize(elements.len(), false);
106        levels.push(Level {
107            priority,
108            elements,
109            allocated,
110        });
111    }
112    let mut groups = memory.reserve(required.groups.len(), &work)?;
113    for source in &required.groups {
114        work.tick()?;
115        if source.lower == 0 {
116            continue;
117        }
118        for level in &mut levels {
119            if let Some(group) = bridges.cost(source, level, &mut memory, &mut work)? {
120                memory.grow(&mut groups, 1, &work)?;
121                groups.push(group);
122            }
123        }
124    }
125    work.tick()?;
126    let costs = if groups.is_empty() {
127        None
128    } else {
129        let activations =
130            activations::prepare(&plan.original, &mut groups, &mut memory, &mut work)?;
131        Some(residuals::prepare(
132            plan,
133            &groups,
134            activations,
135            &mut memory,
136            &mut work,
137        )?)
138    };
139    Ok((costs, work.statistics))
140}

148struct Bridges<'a> {
149    graph: Implications<'a>,
150    proof: Proof,
151    targets: Vec<Vec<Key>>,
152}
153
154impl<'a> Bridges<'a> {
155    fn new(
156        plan: &'a ObjectivePlan,
157        required: &crate::RequiredChoices,
158        memory: &mut Storage,
159        work: &mut Work<'_>,
160    ) -> Result<Self, ObjectiveBoundError> {
161        let graph = Implications::new(plan, memory, work)?;
162        let mut proof = Proof::new(&graph, memory, work)?;
163        let targets = graph.targets(plan, required, &mut proof, memory, work)?;
164        Ok(Self {
165            graph,
166            proof,
167            targets,
168        })
169    }
170
171    /// Every member must imply an unused objective key. Keys used by this group
172    /// may coincide with each other, but never with a preceding group's keys.
173    fn cost(
174        &mut self,
175        source: &crate::formula_count_plan::RequiredGroup,
176        level: &mut Level<'_>,
177        memory: &mut Storage,
178        work: &mut Work<'_>,
179    ) -> Result<Option<Group>, ObjectiveBoundError> {
180        work.tick()?;
181        let mut keys = memory.reserve(source.members.len(), work)?;
182        let mut weight = i32::MAX;
183        for &member in &source.members {
184            work.tick()?;
185            self.proof.assume(&self.graph, source.body, member, work)?;
186            let mut chosen = None;
187            for key in &self.targets[member] {
188                work.tick()?;
189                if key.priority != level.priority || level.allocated[key.index] {
190                    continue;
191                }
192                let element = level.elements[key.index];
193                if element.weight > 0
194                    && self.proof.proves(
195                        &self.graph,
196                        self.graph.objective + element.condition,
197                        work,
198                    )?
199                {
200                    chosen = Some(key.index);
201                    weight = weight.min(element.weight);
202                    break;
203                }
204            }
205            let Some(index) = chosen else {
206                memory.retire(&keys);
207                return Ok(None);
208            };
209            keys.push(index);
210        }
211        if keys.is_empty() {
212            memory.retire(&keys);
213            return Ok(None);
214        }
215        // Repeated members may imply the same coalesced key; count it once.
216        let length = u64::try_from(keys.len()).map_err(|_| work.error(Kind::Overflow))?;
217        work.charge(
218            length
219                .checked_mul(length)
220                .ok_or_else(|| work.error(Kind::Overflow))?,
221        )?;
222        keys.sort_unstable();
223        keys.dedup();
224        for &index in &keys {
225            work.tick()?;
226            level.allocated[index] = true;
227        }
228        Ok(Some(Group {
229            priority: level.priority,
230            activation: source.body,
231            weight,
232            keys,
233        }))
234    }
235}

237/// Original normal-rule roots are sufficient forward implications. Other roots
238/// remain in the original theory but supply no edge to this optional proof.
239struct Implications<'a> {
240    plan: &'a ObjectivePlan,
241    bodies: Vec<Vec<usize>>,
242    facts: Vec<bool>,
243    objective: usize,
244    atoms: usize,
245    vertices: usize,
246}
247
248#[derive(Clone, Copy)]
249enum Reading {
250    Constant(bool),
251    All,
252    Any,
253}
254
255impl<'a> Implications<'a> {
256    fn new(
257        plan: &'a ObjectivePlan,
258        memory: &mut Storage,
259        work: &mut Work<'_>,
260    ) -> Result<Self, ObjectiveBoundError> {
261        let theory = &plan.original;
262        if theory.atom_count() > work.limits.max_atoms {
263            return Err(work.limit(Resource::Atoms));
264        }
265        if theory.nodes().len() > work.limits.max_nodes {
266            return Err(work.limit(Resource::Nodes));
267        }
268        let mut bodies = memory.reserve(theory.atom_count(), work)?;
269        let mut facts = memory.reserve(theory.atom_count(), work)?;
270        for _ in 0..theory.atom_count() {
271            work.tick()?;
272            bodies.push(Vec::new());
273            facts.push(false);
274        }
275        for &root in theory.roots() {
276            work.tick()?;
277            match theory
278                .view()
279                .node(root)
280                .map_err(|error| work.error(Kind::Theory(error)))?
281            {
282                NodeView::Atom(atom) => facts[atom] = true,
283                NodeView::Implies(body, head) => {
284                    work.tick()?;
285                    if let NodeView::Atom(atom) = theory
286                        .view()
287                        .node(head)
288                        .map_err(|error| work.error(Kind::Theory(error)))?
289                    {
290                        memory.grow(&mut bodies[atom], 1, work)?;
291                        bodies[atom].push(body);
292                    }
293                }
294                _ => {}
295            }
296        }
297        let objective = theory.nodes().len();
298        let atoms = objective
299            .checked_add(plan.nodes.view().len())
300            .ok_or_else(|| work.error(Kind::Overflow))?;
301        let vertices = atoms
302            .checked_add(theory.atom_count())
303            .ok_or_else(|| work.error(Kind::Overflow))?;
304        Ok(Self {
305            plan,
306            bodies,
307            facts,
308            objective,
309            atoms,
310            vertices,
311        })
312    }
313
314    fn node(&self, goal: usize) -> NodeView<'_> {
315        if goal < self.objective {
316            self.plan.original.view().node(goal).expect("original node")
317        } else {
318            self.plan
319                .nodes
320                .view()
321                .node(goal - self.objective)
322                .expect("objective node")
323        }
324    }
325
326    fn reading(&self, goal: usize) -> Reading {
327        if goal >= self.atoms {
328            return if self.facts[goal - self.atoms] {
329                Reading::Constant(true)
330            } else {
331                Reading::Any
332            };
333        }
334        match self.node(goal) {
335            NodeView::Atom(_) | NodeView::Or(_) => Reading::Any,
336            NodeView::And(_) => Reading::All,
337            NodeView::False => Reading::Constant(false),
338            NodeView::Implies(left, right) => {
339                let base = if goal < self.objective {
340                    0
341                } else {
342                    self.objective
343                };
344                Reading::Constant(
345                    self.node(base + left) == NodeView::False
346                        && self.node(base + right) == NodeView::False,
347                )
348            }
349        }
350    }
351
352    fn child(&self, goal: usize, position: usize) -> Option<usize> {
353        if goal >= self.atoms {
354            return self.bodies[goal - self.atoms].get(position).copied();
355        }
356        let base = if goal < self.objective {
357            0
358        } else {
359            self.objective
360        };
361        match self.node(goal) {
362            NodeView::Atom(atom) => (position == 0).then_some(self.atoms + atom),
363            NodeView::And(row) | NodeView::Or(row) => row.get(position).map(|node| base + node),
364            NodeView::False | NodeView::Implies(..) => None,
365        }
366    }
367
368    /// Reverse reachable objective keys avoid a member-by-all-keys proof scan.
369    /// This is only an overapproximation; `Proof` validates every used bridge.
370    fn targets(
371        &self,
372        plan: &ObjectivePlan,
373        required: &crate::RequiredChoices,
374        proof: &mut Proof,
375        memory: &mut Storage,
376        work: &mut Work<'_>,
377    ) -> Result<Vec<Vec<Key>>, ObjectiveBoundError> {
378        let mut members = memory.reserve(self.bodies.len(), work)?;
379        let mut output = memory.reserve(self.bodies.len(), work)?;
380        for _ in 0..self.bodies.len() {
381            work.tick()?;
382            members.push(false);
383            output.push(Vec::new());
384        }
385        for group in &required.groups {
386            for &member in &group.members {
387                work.tick()?;
388                members[member] = true;
389            }
390        }
391        let mut pending = memory.reserve(self.vertices, work)?;
392        for (&priority, elements) in &plan.levels {
393            for (index, element) in elements.iter().enumerate() {
394                work.tick()?;
395                if element.weight <= 0 {
396                    continue;
397                }
398                proof.clear(work)?;
399                let start = self.objective + element.condition;
400                proof.mark(start, 1);
401                pending.push(start);
402                while let Some(goal) = pending.pop() {
403                    work.tick()?;
404                    if goal >= self.atoms && members[goal - self.atoms] {
405                        let targets = &mut output[goal - self.atoms];
406                        memory.grow(targets, 1, work)?;
407                        targets.push(Key { priority, index });
408                    }
409                    if matches!(self.reading(goal), Reading::Constant(_)) {
410                        continue;
411                    }
412                    let mut position = 0;
413                    while let Some(child) = self.child(goal, position) {
414                        work.tick()?;
415                        if proof.states[child] == 0 {
416                            proof.mark(child, 1);
417                            pending.push(child);
418                        }
419                        position += 1;
420                    }
421                }
422            }
423        }
424        proof.clear(work)?;
425        memory.retire(&pending);
426        memory.retire(&members);
427        Ok(output)
428    }
429}

431/// The iterative proof visits each goal once under one group's assumptions.
432/// An active cycle yields no proof. A failed proof is never a negative fact.
433struct Proof {
434    states: Vec<u8>,
435    touched: Vec<usize>,
436    frames: Vec<(usize, usize)>,
437}
438
439impl Proof {
440    fn new(
441        graph: &Implications<'_>,
442        memory: &mut Storage,
443        work: &mut Work<'_>,
444    ) -> Result<Self, ObjectiveBoundError> {
445        let mut states = memory.reserve(graph.vertices, work)?;
446        for _ in 0..graph.vertices {
447            work.tick()?;
448            states.push(0);
449        }
450        Ok(Self {
451            states,
452            touched: memory.reserve(graph.vertices, work)?,
453            frames: memory.reserve(graph.vertices, work)?,
454        })
455    }
456
457    fn mark(&mut self, goal: usize, value: u8) {
458        if self.states[goal] == 0 {
459            self.touched.push(goal);
460        }
461        self.states[goal] = value;
462    }
463
464    fn clear(&mut self, work: &mut Work<'_>) -> Result<(), ObjectiveBoundError> {
465        for goal in self.touched.drain(..) {
466            work.tick()?;
467            self.states[goal] = 0;
468        }
469        self.frames.clear();
470        Ok(())
471    }
472
473    fn assume(
474        &mut self,
475        graph: &Implications<'_>,
476        activation: usize,
477        member: usize,
478        work: &mut Work<'_>,
479    ) -> Result<(), ObjectiveBoundError> {
480        self.clear(work)?;
481        self.mark(graph.atoms + member, 3);
482        self.mark(activation, 3);
483        self.frames.push((activation, 0));
484        while let Some((node, _)) = self.frames.pop() {
485            work.tick()?;
486            match graph.node(node) {
487                NodeView::Atom(atom) => self.mark(graph.atoms + atom, 3),
488                NodeView::And(row) => {
489                    for &child in row {
490                        work.tick()?;
491                        if self.states[child] != 3 {
492                            self.mark(child, 3);
493                            self.frames.push((child, 0));
494                        }
495                    }
496                }
497                _ => {}
498            }
499        }
500        Ok(())
501    }
502
503    fn proves(
504        &mut self,
505        graph: &Implications<'_>,
506        goal: usize,
507        work: &mut Work<'_>,
508    ) -> Result<bool, ObjectiveBoundError> {
509        if self.states[goal] != 0 {
510            return Ok(self.states[goal] == 3);
511        }
512        self.mark(goal, 1);
513        self.frames.push((goal, 0));
514        let mut result = None;
515        while let Some(&(current, position)) = self.frames.last() {
516            work.tick()?;
517            let reading = graph.reading(current);
518            let complete = match (reading, result.take()) {
519                (Reading::Constant(value), _) => Some(value),
520                (Reading::All, Some(false)) => Some(false),
521                (Reading::Any, Some(true)) => Some(true),
522                _ => None,
523            };
524            if let Some(value) = complete {
525                self.frames.pop();
526                self.mark(current, if value { 3 } else { 2 });
527                result = Some(value);
528                continue;
529            }
530            if let Some(child) = graph.child(current, position) {
531                self.frames.last_mut().expect("active proof frame").1 += 1;
532                if self.states[child] == 0 {
533                    self.mark(child, 1);
534                    self.frames.push((child, 0));
535                } else {
536                    result = Some(self.states[child] == 3);
537                }
538            } else {
539                let value = matches!(reading, Reading::All);
540                self.frames.pop();
541                self.mark(current, if value { 3 } else { 2 });
542                result = Some(value);
543            }
544        }
545        Ok(result == Some(true))
546    }
547}

549pub(super) fn strengthen(
550    plan: &ObjectivePlan,
551    mut exact: ObjectiveBound,
552    incumbent: &Score,
553    limits: ObjectiveBoundLimits,
554    cancellation: &Cancellation,
555) -> ObjectiveBound {
556    let Some(costs) = &plan.choices else {
557        return exact;
558    };
559    let mut work = Work {
560        limits: ObjectivePlanLimits {
561            max_nodes: limits.aggregate.max_nodes,
562            max_operands: limits.aggregate.max_operands,
563            max_work: limits.max_work,
564            ..ObjectivePlanLimits::default()
565        },
566        cancellation,
567        template: None,
568        statistics: exact.statistics,
569    };
570    match combined(plan, costs, &exact.theory, incumbent, limits, &mut work) {
571        Ok(theory) => exact.theory = theory,
572        Err(error) => exact.choice_failure = Some(error),
573    }
574    exact.statistics = work.statistics;
575    exact
576}
577
578fn combined(
579    plan: &ObjectivePlan,
580    costs: &Costs,
581    exact: &Theory,
582    incumbent: &Score,
583    limits: ObjectiveBoundLimits,
584    work: &mut Work<'_>,
585) -> Result<Theory, ObjectiveBoundError> {
586    let mut nodes = FormulaNodes::default();
587    for index in 0..exact.view().len() {
588        work.node(
589            &mut nodes,
590            exact
591                .view()
592                .node(index)
593                .map_err(|error| work.error(Kind::Theory(error)))?,
594        )?;
595    }
596    let copied = activations::append(&costs.activations, &mut nodes, work)?;
597    // Both priority lists ascend, and every prepared level belongs to the exact
598    // plan. Extra incumbent priorities leave this cursor in place.
599    let mut prepared = costs.levels.iter().peekable();
600    let root = super::bound::lexicographic(
601        &mut nodes,
602        plan.levels.keys().copied(),
603        incumbent,
604        limits,
605        work,
606        |priority, work| {
607            let Some(level) = prepared.peek().filter(|level| level.priority == priority) else {
608                return Ok(Cow::Borrowed(
609                    plan.levels.get(&priority).map_or(&[][..], Vec::as_slice),
610                ));
611            };
612            let count = level
613                .groups
614                .len()
615                .checked_add(level.residuals.len())
616                .ok_or_else(|| work.error(Kind::Overflow))?;
617            let mut elements = work.reserve(count)?;
618            for group in &costs.groups[level.groups.clone()] {
619                work.tick()?;
620                elements.push(AggregateElement {
621                    weight: group.weight,
622                    condition: copied[group.condition],
623                });
624            }
625            for &residual in &level.residuals {
626                work.tick()?;
627                elements.push(residual);
628            }
629            prepared.next();
630            Ok(Cow::Owned(elements))
631        },
632    )?;
633    let root = work.node(&mut nodes, NodeView::And(&[exact.roots()[0], root]))?;
634    let admission = (nodes.view().len() as u128) * 2 + nodes.parts().occurrences() as u128 + 1;
635    work.charge(u64::try_from(admission).map_err(|_| work.error(Kind::Overflow))?)?;
636    Theory::new(
637        plan.original.atom_count(),
638        nodes.into_parts(),
639        vec![root],
640        zetesis_ferraris::AdmissionLimits {
641            max_atoms: plan.original.atom_count(),
642            max_nodes: limits.aggregate.max_nodes,
643            max_roots: 1,
644            max_operands: limits.aggregate.max_operands,
645        },
646    )
647    .map_err(|error| work.error(Kind::Theory(error)))
648}

650/// Named vector capacities in optional proof preparation. Source premises are
651/// borrowed and charged once. Allocator overhead and unrelated owners are not
652/// represented; this is an explicit preparation allowance, not a process limit.
653struct Storage {
654    bytes: u128,
655    limit: u128,
656}
657
658impl Storage {
659    fn new(
660        limit: u128,
661        required: &crate::RequiredChoices,
662        previous: Option<&Costs>,
663        work: &Work<'_>,
664    ) -> Result<Self, ObjectiveBoundError> {
665        let mut bytes = required.retained_bytes() + size_of::<Costs>() as u128;
666        if let Some(previous) = previous {
667            bytes += Self::vector_bytes(&previous.groups);
668            bytes += previous.activations.node_capacity() as u128
669                * size_of::<zetesis_ferraris::Node>() as u128;
670            bytes += previous.activations.operand_capacity() as u128 * size_of::<usize>() as u128;
671            bytes += Self::vector_bytes(&previous.levels);
672            for level in &previous.levels {
673                bytes += Self::vector_bytes(&level.residuals);
674            }
675        }
676        let result = Self { bytes, limit };
677        result.check(0, work)?;
678        Ok(result)
679    }
680
681    fn vector_bytes<T>(values: &Vec<T>) -> u128 {
682        values.capacity() as u128 * size_of::<T>() as u128
683    }
684
685    fn check(&self, additional: u128, work: &Work<'_>) -> Result<(), ObjectiveBoundError> {
686        if self
687            .bytes
688            .checked_add(additional)
689            .is_none_or(|total| total > self.limit)
690        {
691            return Err(work.limit(Resource::ChoiceBytes));
692        }
693        Ok(())
694    }
695
696    fn reserve<T>(&mut self, count: usize, work: &Work<'_>) -> Result<Vec<T>, ObjectiveBoundError> {
697        self.check(count as u128 * size_of::<T>() as u128, work)?;
698        let values = work.reserve(count)?;
699        let bytes = Self::vector_bytes(&values);
700        self.check(bytes, work)?;
701        self.bytes += bytes;
702        Ok(values)
703    }
704
705    fn grow<T>(
706        &mut self,
707        values: &mut Vec<T>,
708        additional: usize,
709        work: &Work<'_>,
710    ) -> Result<(), ObjectiveBoundError> {
711        let length = values
712            .len()
713            .checked_add(additional)
714            .ok_or_else(|| work.error(Kind::Overflow))?;
715        if length <= values.capacity() {
716            return Ok(());
717        }
718        let before = Self::vector_bytes(values);
719        self.check(
720            (length - values.capacity()) as u128 * size_of::<T>() as u128,
721            work,
722        )?;
723        values
724            .try_reserve_exact(additional)
725            .map_err(|_| work.error(Kind::Allocation))?;
726        let delta = Self::vector_bytes(values) - before;
727        self.check(delta, work)?;
728        self.bytes += delta;
729        Ok(())
730    }
731
732    fn retire<T>(&mut self, values: &Vec<T>) {
733        self.bytes -= Self::vector_bytes(values);
734    }
