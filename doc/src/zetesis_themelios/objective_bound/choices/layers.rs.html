1//! A second necessary objective from the same checked disjoint key groups.
2//! Eligibility addresses the unchanged exact prefix; activation uses its one
3//! existing compact copy. Separate lower objectives are conjoined, never added.
4//!
5//! For each active group, cumulative cheaper-key disjunctions select nonnegative
6//! weight increments. `ChoiceMinimumLayers.group_lower_le` bounds that value by
7//! the group's exact cost under its required eligible-witness premise. Disjoint
8//! allocation, coordinate transport, finite-width operations and resource-aware
9//! execution are separate Rust correspondence obligations.

18#[derive(Clone, Copy)]
19pub(super) struct Request<'a> {
20    pub incumbent: &'a Score,
21    pub comparison: Comparison,
22    pub limits: ObjectiveBoundLimits,
23}

25pub(super) fn bound(
26    plan: &ObjectivePlan,
27    costs: &Costs,
28    nodes: &mut FormulaNodes,
29    copied: &[usize],
30    request: Request<'_>,
31    work: &mut Work<'_>,
32) -> Result<usize, ObjectiveBoundError> {
33    let mut levels = terms(plan, costs, nodes, copied, work)?
34        .into_iter()
35        .peekable();
36    bound::lexicographic(
37        nodes,
38        plan.levels.keys().copied(),
39        request.incumbent,
40        request.comparison,
41        request.limits,
42        work,
43        |priority, _| {
44            if levels
45                .peek()
46                .is_some_and(|level| level.priority == priority)
47            {
48                Ok(Cow::Owned(
49                    levels.next().expect("matched priority").elements,
50                ))
51            } else {
52                Ok(Cow::Borrowed(
53                    plan.levels.get(&priority).map_or(&[][..], Vec::as_slice),
54                ))
55            }
56        },
57    )
58}

65/// Use the existing checked quadratic sort-admission convention. The tariff
66/// bounds preparation work; it is not an instruction/comparison-count claim.
67pub(super) fn order(
68    plan: &ObjectivePlan,
69    groups: &mut [Group],
70    work: &mut Work<'_>,
71) -> Result<(), ObjectiveBoundError> {
72    for group in groups {
73        work.tick()?;
74        let count = u64::try_from(group.keys.len()).map_err(|_| work.error(Kind::Overflow))?;
75        work.charge(
76            count
77                .checked_mul(count)
78                .ok_or_else(|| work.error(Kind::Overflow))?,
79        )?;
80        let original = &plan.levels[&group.priority];
81        group
82            .keys
83            .sort_unstable_by_key(|&key| (original[key].weight, key));
84    }
85    Ok(())
86}

88/// At most one tier per allocated key plus the unallocated exact keys. Every
89/// temporary row is bounded by its original priority's admitted key population.
90pub(super) fn terms(
91    plan: &ObjectivePlan,
92    costs: &Costs,
93    nodes: &mut FormulaNodes,
94    copied: &[usize],
95    work: &mut Work<'_>,
96) -> Result<Vec<Level>, ObjectiveBoundError> {
97    let mut result = work.reserve(costs.levels.len())?;
98    let mut falsum = None;
99    for level in &costs.levels {
100        work.tick()?;
101        let original = &plan.levels[&level.priority];
102        let mut elements = work.reserve(original.len())?;
103        for &element in &level.layer_remainder {
104            work.tick()?;
105            elements.push(element);
106        }
107        for group in &costs.layered {
108            work.tick()?;
109            if group.priority == level.priority {
110                append_group(
111                    original,
112                    group,
113                    copied[group.activation],
114                    nodes,
115                    &mut elements,
116                    &mut falsum,
117                    work,
118                )?;
119            }
120        }
121        work.tick()?;
122        result.push(Level {
123            priority: level.priority,
124            elements,
125        });
126    }
127    Ok(result)
128}

130pub(super) fn append_group(
131    original: &[AggregateElement],
132    group: &Group,
133    activation: usize,
134    nodes: &mut FormulaNodes,
135    terms: &mut Vec<AggregateElement>,
136    falsum: &mut Option<usize>,
137    work: &mut Work<'_>,
138) -> Result<(), ObjectiveBoundError> {
139    work.tick()?;
140    let mut previous = group.weight;
141    assert!(!group.keys.is_empty(), "a checked group has a witness");
142    assert!(
143        previous >= 0,
144        "only nonnegative priorities supply checked groups"
145    );
146    if previous != 0 {
147        work.tick()?;
148        terms.push(AggregateElement {
149            weight: previous,
150            condition: activation,
151        });
152    }
153    let mut start = 0;
154    let mut prefix = None;
155    while start < group.keys.len() {
156        work.tick()?;
157        let mut end = start + 1;
158        while end < group.keys.len() {
159            work.tick()?;
160            if original[group.keys[end]].weight != previous {
161                break;
162            }
163            end += 1;
164        }
165        // The highest tier never belongs to a cheaper-key prefix.
166        if end == group.keys.len() {
167            break;
168        }
169        append_prefix(&group.keys[start..end], original, &mut prefix, nodes, work)?;
170        let bottom = if let Some(bottom) = *falsum {
171            bottom
172        } else {
173            let bottom = work.node(nodes, NodeView::False)?;
174            *falsum = Some(bottom);
175            bottom
176        };
177        let absent = work.node(
178            nodes,
179            NodeView::Implies(prefix.expect("a cheaper key exists"), bottom),
180        )?;
181        let condition = work.node(nodes, NodeView::And(&[activation, absent]))?;
182        work.tick()?;
183        let next = original[group.keys[end]].weight;
184        let weight = next
185            .checked_sub(previous)
186            .ok_or_else(|| work.error(Kind::Overflow))?;
187        work.tick()?;
188        terms.push(AggregateElement { weight, condition });
189        start = end;
190        previous = next;
191    }
192    Ok(())
193}

195fn append_prefix(
196    keys: &[usize],
197    original: &[AggregateElement],
198    prefix: &mut Option<usize>,
199    nodes: &mut FormulaNodes,
200    work: &mut Work<'_>,
201) -> Result<(), ObjectiveBoundError> {
202    for &key in keys {
203        work.tick()?;
204        let condition = original[key].condition;
205        *prefix = Some(match *prefix {
206            None => condition,
207            Some(before) => work.node(nodes, NodeView::Or(&[before, condition]))?,
208        });
209    }
210    Ok(())
211}
