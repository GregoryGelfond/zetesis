29pub(super) fn compile(
30    original: &Theory,
31    atoms: Atoms<'_>,
32    objectives: &ObjectiveProgram,
33    limits: ObjectivePlanLimits,
34    cancellation: &Cancellation,
35) -> Result<ObjectivePlan, ObjectiveBoundError> {
36    let mut work = Work {
37        cancellation,
38        limits,
39        template: None,
40        statistics: ObjectiveBoundStatistics::default(),
41    };
42    work.tick()?;
43    if atoms.len() != original.atom_count() {
44        return Err(work.error(Kind::AtomCatalog));
45    }
46    if atoms.len() > limits.max_atoms {
47        return Err(work.limit(Resource::Atoms));
48    }
49    let mut nodes = FormulaNodes::default();
50    // The existing atom ceiling admits both O(n) index orders and reusable
51    // merge scratch. Only integer row IDs are allocated; payload remains here.
52    let index =
53        AtomIndex::from_catalog_with(atoms, || work.tick()).map_err(|error| match error {
54            AtomIndexError::Stopped(error) => error,
55            AtomIndexError::Allocation => work.error(Kind::Allocation),
56            AtomIndexError::Duplicate { .. } => work.error(Kind::AtomCatalog),
57        })?;
58    for index in 0..atoms.len() {
59        work.node(&mut nodes, NodeView::Atom(index))?;
60    }
61    let falsum = work.node(&mut nodes, NodeView::False)?;
62    let truth = work.node(&mut nodes, NodeView::Implies(falsum, falsum))?;
63    let mut compiler = Compiler {
64        work,
65        index,
66        nodes,
67        keys: Vec::new(),
68        truth,
69    };
70    for (index, template) in objectives.templates().iter().enumerate() {
71        compiler.work.template = Some(index);
72        compiler.join(template)?;
73    }
74    compiler.work.template = None;
75    let mut levels = BTreeMap::new();
76    for priority in objectives.priorities() {
77        compiler.work.tick()?;
78        levels.insert(*priority, Vec::new());
79    }
80    for key in compiler.keys {
81        compiler.work.tick()?;
82        let elements = levels.entry(key.priority).or_default();
83        elements
84            .try_reserve(1)
85            .map_err(|_| compiler.work.error(Kind::Allocation))?;
86        elements.push(AggregateElement {
87            weight: key.weight,
88            condition: key.condition,
89        });
90    }
91    Ok(ObjectivePlan {
92        original: original.clone(),
93        objectives: objectives.clone(),
94        nodes: compiler.nodes.into_parts(),
95        levels,
96        choices: None,
97        statistics: compiler.work.statistics,
98    })
99}

101struct Frame<'index, 'source> {
102    rows: AtomRows<'index, 'source>,
103    trail_start: usize,
104}
105
106impl<'a> Compiler<'a> {
107    fn join(&mut self, template: ObjectiveTemplateRef<'a>) -> Result<(), ObjectiveBoundError> {
108        if template.positive().len() > self.work.limits.max_body_atoms {
109            return Err(self.work.limit(Resource::BodyAtoms));
110        }
111        if template.tuple().len() > self.work.limits.max_tuple_width {
112            return Err(self.work.limit(Resource::TupleWidth));
113        }
114        let count = self.variables(template)?;
115        let mut binding = self.work.reserve(count)?;
116        binding.resize(count, None);
117        let mut trail = self.work.reserve(count)?;
118        let condition = self.condition(template.condition())?;
119        if template.positive().is_empty() {
120            return self.active(template, &binding, condition.as_slice());
121        }
122        let mut frames = self.work.reserve(template.positive().len())?;
123        let mut chosen = self.work.reserve(template.positive().len())?;
124        chosen.resize(template.positive().len(), 0);
125        if let Some(condition) = condition {
126            chosen
127                .try_reserve(1)
128                .map_err(|_| self.work.error(Kind::Allocation))?;
129            chosen.push(condition);
130        }
131        let lookup = self.index.lookup();
132        frames.push(Frame {
133            rows: lookup.predicate_with(
134                template
135                    .positive()
136                    .at(0)
137                    .expect("nonempty positive body")
138                    .predicate(),
139                || self.work.tick(),
140            )?,
141            trail_start: 0,
142        });
143        while !frames.is_empty() {
144            self.work.tick()?;
145            let depth = frames.len() - 1;
146            let frame = &mut frames[depth];
147            while trail.len() > frame.trail_start {
148                self.work.tick()?;
149                let variable = trail
150                    .pop()
151                    .ok_or_else(|| self.work.error(Kind::UnboundVariable))?;
152                binding[variable] = None;
153            }
154            let Some(row) = frame.rows.next() else {
155                frames.pop();
156                continue;
157            };
158            let atom_index = row.position();
159            if !matches(
160                &mut self.work,
161                template.positive().at(depth).expect("join depth in body"),
162                row.atom(),
163                &mut binding,
164                &mut trail,
165            )? {
166                continue;
167            }
168            chosen[depth] = atom_index;
169            if depth + 1 == template.positive().len() {
170                active(
171                    &mut self.work,
172                    &mut self.nodes,
173                    &mut self.keys,
174                    self.truth,
175                    template,
176                    &binding,
177                    &chosen,
178                )?;
179            } else {
180                frames.push(Frame {
181                    rows: lookup.predicate_with(
182                        template
183                            .positive()
184                            .at(depth + 1)
185                            .expect("next join depth in body")
186                            .predicate(),
187                        || self.work.tick(),
188                    )?,
189                    trail_start: trail.len(),
190                });
191            }
192        }
193        Ok(())
194    }
195    fn active(
196        &mut self,
197        template: ObjectiveTemplateRef<'a>,
198        binding: &[Option<TermRef<'a>>],
199        chosen: &[usize],
200    ) -> Result<(), ObjectiveBoundError> {
201        active(
202            &mut self.work,
203            &mut self.nodes,
204            &mut self.keys,
205            self.truth,
206            template,
207            binding,
208            chosen,
209        )
210    }
211    fn variables(
212        &mut self,
213        template: ObjectiveTemplateRef<'_>,
214    ) -> Result<usize, ObjectiveBoundError> {
215        let mut count = 0;
216        for index in 0..template.positive().len() {
217            self.work.tick()?;
218            let pattern = template.positive().at(index).expect("bounded body pattern");
219            for column in 0..pattern.terms().len() {
220                self.work.tick()?;
221                if let TemplateTerm::Variable(variable) =
222                    pattern.terms().at(column).expect("bounded pattern column")
223                {
224                    count = count.max(
225                        variable
226                            .checked_add(1)
227                            .ok_or_else(|| self.work.error(Kind::Overflow))?,
228                    );
229                }
230            }
231        }
232        if count > self.work.limits.max_variables {
233            return Err(self.work.limit(Resource::Variables));
234        }
235        Ok(count)
236    }
237
238    fn condition(&mut self, condition: &Condition) -> Result<Option<usize>, ObjectiveBoundError> {
239        let mut nodes: Vec<usize> = self.work.reserve(condition.nodes().len())?;
240        for index in 0..condition.nodes().len() {
241            self.work.tick()?;
242            let operation = condition
243                .nodes()
244                .at(index)
245                .expect("bounded condition operation");
246            let node = match operation {
247                ConditionNode::Boolean(true) => self.truth,
248                ConditionNode::Boolean(false) => {
249                    self.work.node(&mut self.nodes, NodeView::False)?
250                }
251                ConditionNode::Atom(atom) => self.condition_atom(atom)?,
252                ConditionNode::Not(operand) => {
253                    let falsum = self.work.node(&mut self.nodes, NodeView::False)?;
254                    self.work
255                        .node(&mut self.nodes, NodeView::Implies(nodes[operand], falsum))?
256                }
257                ConditionNode::And(left, right) => self
258                    .work
259                    .node(&mut self.nodes, NodeView::And(&[nodes[left], nodes[right]]))?,
260                ConditionNode::Or(left, right) => self
261                    .work
262                    .node(&mut self.nodes, NodeView::Or(&[nodes[left], nodes[right]]))?,
263            };
264            nodes.push(node);
265        }
266        Ok(nodes.last().copied())
267    }
268
269    fn condition_atom(&mut self, query: AtomRef<'_>) -> Result<usize, ObjectiveBoundError> {
270        if let Some(row) = self.index.lookup().get_with(query, || self.work.tick())? {
271            return Ok(row.position());
272        }
273        // Atoms outside the caller's complete catalog are false in every
274        // represented candidate. Querying one must not enlarge that catalog.
275        self.work.node(&mut self.nodes, NodeView::False)
276    }
277}

279fn matches<'a>(
280    work: &mut Work<'_>,
281    pattern: PatternRef<'_>,
282    atom: AtomRef<'a>,
283    binding: &mut [Option<TermRef<'a>>],
284    trail: &mut Vec<usize>,
285) -> Result<bool, ObjectiveBoundError> {
286    for column in 0..pattern.terms().len() {
287        work.tick()?;
288        let term = pattern.terms().at(column).expect("bounded pattern column");
289        let value = atom.values().at(column).expect("matched predicate arity");
290        match term {
291            TemplateTerm::Constant(constant)
292                if !constant.equals_ref_with(value, || work.tick())? =>
293            {
294                return Ok(false);
295            }
296            TemplateTerm::Variable(variable) => {
297                if let Some(previous) = binding[variable] {
298                    if !previous.equals_ref_with(value, || work.tick())? {
299                        return Ok(false);
300                    }
301                } else {
302                    binding[variable] = Some(value);
303                    trail.push(variable);
304                }
305            }
306            TemplateTerm::Constant(_) => {}
307        }
308    }
309    Ok(true)
310}

312fn resolve<'a>(
313    work: &mut Work<'_>,
314    term: TemplateTerm<'a>,
315    binding: &[Option<TermRef<'a>>],
316) -> Result<TermRef<'a>, ObjectiveBoundError> {
317    work.tick()?;
318    match term {
319        TemplateTerm::Constant(value) => Ok(value),
320        TemplateTerm::Variable(variable) => binding
321            .get(variable)
322            .copied()
323            .flatten()
324            .ok_or_else(|| work.error(Kind::UnboundVariable)),
325    }
326}
327// Canonical typed identity order for contribution-key storage.
328// This is not ASP term order; use the shared checked comparator, not Value::Ord
329// behind a conservative payload-size estimate.
330fn compare_identity(
331    work: &mut Work<'_>,
332    left: TermRef<'_>,
333    right: TermRef<'_>,
334) -> Result<Ordering, ObjectiveBoundError> {
335    left.compare_ref_with(right, || work.tick())
336}

338fn active<'a>(
339    work: &mut Work<'_>,
340    nodes: &mut FormulaNodes,
341    keys: &mut Vec<Key<'a>>,
342    truth: usize,
343    template: ObjectiveTemplateRef<'a>,
344    binding: &[Option<TermRef<'a>>],
345    chosen: &[usize],
346) -> Result<(), ObjectiveBoundError> {
347    work.tick()?;
348    if work.statistics.bindings >= work.limits.max_bindings {
349        return Err(work.limit(Resource::Bindings));
350    }
351    work.statistics.bindings += 1;
352    for index in 0..template.filters().len() {
353        work.tick()?;
354        let filter = template.filters().at(index).expect("bounded filter index");
355        let (left, right) = filter.terms();
356        let left = resolve(work, left, binding)?;
357        let right = resolve(work, right, binding)?;
358        if left.equals_ref_with(right, || work.tick())? != filter.is_equality() {
359            return Ok(());
360        }
361    }
362    let weight = resolve(work, template.weight(), binding)?;
363    work.tick()?;
364    let ValueNodeRef::Number(weight) = weight.descriptor() else {
365        return Ok(());
366    };
367    let weight = template
368        .weight_polarity()
369        .normalize(weight)
370        .ok_or_else(|| work.error(Kind::WeightNormalizationOverflow))?;
371    let mut tuple = work.reserve(template.tuple().len())?;
372    for index in 0..template.tuple().len() {
373        work.tick()?;
374        let term = template.tuple().at(index).expect("bounded tuple field");
375        tuple.push(resolve(work, term, binding)?);
376    }
377    let condition = if chosen.is_empty() {
378        truth
379    } else {
380        work.node(nodes, NodeView::And(chosen))?
381    };
382    contribute(
383        work,
384        nodes,
385        keys,
386        Key {
387            priority: template.priority(),
388            weight,
389            tuple,
390            condition,
391        },
392    )
393}

395fn key_order(
396    work: &mut Work<'_>,
397    left: &Key<'_>,
398    right: &Key<'_>,
399) -> Result<Ordering, ObjectiveBoundError> {
400    work.tick()?;
401    let prefix = left
402        .priority
403        .cmp(&right.priority)
404        .then(left.weight.cmp(&right.weight));
405    if prefix != Ordering::Equal {
406        return Ok(prefix);
407    }
408    for (left, right) in left.tuple.iter().zip(&right.tuple) {
409        let order = compare_identity(work, *left, *right)?;
410        if order != Ordering::Equal {
411            return Ok(order);
412        }
413    }
414    Ok(left.tuple.len().cmp(&right.tuple.len()))
415}
416fn contribute<'a>(
417    work: &mut Work<'_>,
418    nodes: &mut FormulaNodes,
419    keys: &mut Vec<Key<'a>>,
420    key: Key<'a>,
421) -> Result<(), ObjectiveBoundError> {
422    let mut low = 0;
423    let mut high = keys.len();
424    while low < high {
425        let middle = low + (high - low) / 2;
426        match key_order(work, &keys[middle], &key)? {
427            Ordering::Less => low = middle + 1,
428            Ordering::Greater => high = middle,
429            Ordering::Equal => {
430                if keys[middle].condition != key.condition {
431                    keys[middle].condition = work.node(
432                        nodes,
433                        NodeView::Or(&[keys[middle].condition, key.condition]),
434                    )?;
435                }
436                return Ok(());
437            }
438        }
439    }
440    if keys.len() >= work.limits.max_keys {
441        return Err(work.limit(Resource::Keys));
442    }
443    let mut bytes: usize = 16;
444    for value in &key.tuple {
445        let length = value.canonical_bytes_with(|| work.tick())?;
446        bytes = bytes
447            .checked_add(length)
448            .ok_or_else(|| work.error(Kind::Overflow))?;
449    }
450    let bytes = work
451        .statistics
452        .key_bytes
453        .checked_add(bytes)
454        .ok_or_else(|| work.error(Kind::Overflow))?;
455    if bytes > work.limits.max_key_bytes {
456        return Err(work.limit(Resource::KeyBytes));
457    }
458    for _ in low..keys.len() {
459        work.tick()?;
460    }
461    keys.try_reserve(1)
462        .map_err(|_| work.error(Kind::Allocation))?;
463    keys.insert(low, key);
464    work.statistics.keys = keys.len();
465    work.statistics.key_bytes = bytes;
466    Ok(())
