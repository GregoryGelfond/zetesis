11/// A score-only read of the prepared eligibility, without tuple evidence.
12#[derive(Debug)]
13pub struct ObjectiveScore {
14    score: Score,
15    work: u64,
16}
17impl ObjectiveScore {
18    /// Transfer the complete score, retaining presence and inactive slots.
19    #[must_use]
20    pub fn into_score(self) -> Score {
21        self.score
22    }
23    /// Accepted DAG and numeric reduction work for this read.
24    #[must_use]
25    pub const fn work(&self) -> u64 {
26        self.work
27    }
28}

30/// The original typed cause of an incomplete prepared score.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum ObjectiveScoreErrorKind {
33    /// Equal atom counts do not authorize a different immutable theory.
34    WrongTheory,
35    /// Eligibility inspection, control or storage refused.
36    Eligibility(ObjectiveBoundError),
37    /// Numeric reduction refused without a complete score.
38    Reduction(zetesis_objective::Error),
39}
40
41/// Incomplete prepared scoring with its complete accepted work prefix.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub struct ObjectiveScoreError {
44    kind: ObjectiveScoreErrorKind,
45    work: u64,
46}
47impl ObjectiveScoreError {
48    /// Structured cause; no successful score accompanies this error.
49    #[must_use]
50    pub const fn kind(self) -> ObjectiveScoreErrorKind {
51        self.kind
52    }
53    /// Eligibility and reduction work accepted before the refusal.
54    #[must_use]
55    pub const fn work(self) -> u64 {
56        self.work
57    }
58}
59impl fmt::Display for ObjectiveScoreError {
60    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
61        match self.kind {
62            ObjectiveScoreErrorKind::WrongTheory => {
63                formatter.write_str("prepared objective score belongs to a different theory")
64            }
65            ObjectiveScoreErrorKind::Eligibility(error) => error.fmt(formatter),
66            ObjectiveScoreErrorKind::Reduction(error) => error.fmt(formatter),
67        }
68    }
69}
70impl std::error::Error for ObjectiveScoreError {}

72impl ObjectivePlan {
73    /// Score a candidate using the already coalesced keys and exact eligibility.
74    /// The candidate must retain this plan's original theory identity. Scoring
75    /// does not verify stability or the caller's completed-catalog coverage.
76    ///
77    /// `None` declines this path when the complete possible population cannot
78    /// establish the caller's binding/key/byte ceilings. The ordinary detailed
79    /// evaluator can then apply those ceilings to just the selected model. No
80    /// work is consumed by that decline; cancellation and identity are checked.
81    /// The retained plan supplies numeric keys only, so contribution evidence
82    /// always uses the ordinary evaluator.
83    ///
84    /// One unit is charged per DAG node and operand occurrence, followed by the
85    /// numeric reducer's work.
86    /// Priority lookup uses logarithmic map probes; the reducer charges one
87    /// logical priority visit, not each map comparison.
88    /// Temporary storage is linear in retained nodes and priority slots, bounded
89    /// by the successful plan's admission. There is no new key or tuple cache.
90    ///
91    /// # Errors
92    /// Returns exact-owner, control, work, allocation or cost overflow failures
93    /// with their complete accepted work prefix; no partial score is published.
94    pub fn score(
95        &self,
96        candidate: &Interpretation,
97        limits: Limits,
98        cancellation: &Cancellation,
99    ) -> Result<Option<ObjectiveScore>, ObjectiveScoreError> {
100        let mut work = Work {
101            cancellation,
102            limits: ObjectivePlanLimits {
103                max_work: limits.max_work,
104                ..ObjectivePlanLimits::default()
105            },
106            template: None,
107            statistics: ObjectiveBoundStatistics::default(),
108        };
109        work.charge(0).map_err(eligibility_error)?;
110        if !self.original.same_instance(candidate.theory()) {
111            return Err(ObjectiveScoreError {
112                kind: ObjectiveScoreErrorKind::WrongTheory,
113                work: 0,
114            });
115        }
116        if self.statistics.bindings > limits.max_bindings
117            || self.statistics.keys > limits.max_keys
118            || self.statistics.key_bytes > limits.max_key_bytes
119        {
120            return Ok(None);
121        }
122        let view = self.nodes.view();
123        let mut values = work.reserve(view.len()).map_err(eligibility_error)?;
124        for index in 0..view.len() {
125            work.tick().map_err(eligibility_error)?;
126            let node = view.node(index).map_err(|error| {
127                eligibility_error(work.error(super::ObjectiveBoundErrorKind::Theory(error)))
128            })?;
129            let value = match node {
130                NodeView::False => false,
131                NodeView::Atom(atom) => candidate.contains(atom),
132                NodeView::And(row) => evaluate_group(row, &values, true, &mut work)?,
133                NodeView::Or(row) => evaluate_group(row, &values, false, &mut work)?,
134                NodeView::Implies(left, right) => {
135                    work.tick().map_err(eligibility_error)?;
136                    let antecedent = values[left];
137                    work.tick().map_err(eligibility_error)?;
138                    !antecedent || values[right]
139                }
140            };
141            values.push(value);
142        }
143        let truth = values.as_slice();
144        let reduced = zetesis_objective::reduce_costs(
145            &self.objectives,
146            |priority| {
147                self.levels
148                    .get(&priority)
149                    .into_iter()
150                    .flatten()
151                    .map(move |element| truth[element.condition].then_some(element.weight))
152            },
153            limits.max_work - work.statistics.work,
154            cancellation,
155        )
156        .map_err(|error| ObjectiveScoreError {
157            work: work.statistics.work + error.statistics().work,
158            kind: ObjectiveScoreErrorKind::Reduction(error),
159        })?;
160        Ok(Some(ObjectiveScore {
161            work: work.statistics.work + reduced.work(),
162            score: reduced.into_score(),
163        }))
164    }
165}
166
167fn evaluate_group(
168    row: &[usize],
169    values: &[bool],
170    conjunction: bool,
171    work: &mut Work<'_>,
172) -> Result<bool, ObjectiveScoreError> {
173    let mut value = conjunction;
174    for &child in row {
175        work.tick().map_err(eligibility_error)?;
176        if conjunction {
177            value &= values[child];
178        } else {
179            value |= values[child];
180        }
181    }
182    Ok(value)
183}
184
185fn eligibility_error(error: ObjectiveBoundError) -> ObjectiveScoreError {
186    ObjectiveScoreError {
187        work: error.statistics().work,
188        kind: ObjectiveScoreErrorKind::Eligibility(error),
189    }
