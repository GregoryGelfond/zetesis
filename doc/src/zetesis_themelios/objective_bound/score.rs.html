11/// A score-only read of the prepared eligibility, without tuple evidence.
12#[derive(Debug)]
13pub struct ObjectiveScore {
14    score: Score,
15    work: u64,
16}
17impl ObjectiveScore {
18    /// Transfer the complete score, retaining presence and inactive slots.
19    #[must_use]
20    pub fn into_score(self) -> Score {
21        self.score
22    }
23    /// Accepted DAG and numeric reduction work for this read.
24    #[must_use]
25    pub const fn work(&self) -> u64 {
26        self.work
27    }
28}

30/// The original typed cause of an incomplete prepared score.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum ObjectiveScoreErrorKind {
33    /// Equal atom counts do not authorize a different immutable theory.
34    WrongTheory,
35    /// Eligibility inspection, control or storage refused.
36    Eligibility(ObjectiveBoundError),
37    /// Numeric reduction refused without a complete score.
38    Reduction(zetesis_objective::Error),
39}
40
41/// Incomplete prepared scoring with its complete accepted work prefix.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub struct ObjectiveScoreError {
44    kind: ObjectiveScoreErrorKind,
45    work: u64,
46}
47impl ObjectiveScoreError {
48    /// Structured cause; no successful score accompanies this error.
49    #[must_use]
50    pub const fn kind(self) -> ObjectiveScoreErrorKind {
51        self.kind
52    }
53    /// Eligibility and reduction work accepted before the refusal.
54    #[must_use]
55    pub const fn work(self) -> u64 {
56        self.work
57    }
58}
59impl fmt::Display for ObjectiveScoreError {
60    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
61        match self.kind {
62            ObjectiveScoreErrorKind::WrongTheory => {
63                formatter.write_str("prepared objective score belongs to a different theory")
64            }
65            ObjectiveScoreErrorKind::Eligibility(error) => error.fmt(formatter),
66            ObjectiveScoreErrorKind::Reduction(error) => error.fmt(formatter),
67        }
68    }
69}
70impl std::error::Error for ObjectiveScoreError {}

72impl ObjectivePlan {
73    /// Score a candidate using the already coalesced keys and exact eligibility.
74    /// The candidate must retain this plan's original theory identity. Scoring
75    /// does not verify stability or the caller's completed-catalog coverage.
76    ///
77    /// `None` declines this path when the complete possible population cannot
78    /// establish the caller's binding/key/byte ceilings. The ordinary detailed
79    /// evaluator can then apply those ceilings to just the selected model. No
80    /// work is consumed by that decline; cancellation and identity are checked.
81    /// The retained plan supplies numeric keys only, so contribution evidence
82    /// always uses the ordinary evaluator.
83    ///
84    /// One unit is charged per DAG node, followed by the numeric reducer's work.
85    /// Priority lookup uses logarithmic map probes; the reducer charges one
86    /// logical priority visit, not each map comparison.
87    /// Temporary storage is linear in retained nodes and priority slots, bounded
88    /// by the successful plan's admission. There is no new key or tuple cache.
89    ///
90    /// # Errors
91    /// Returns exact-owner, control, work, allocation or cost overflow failures
92    /// with their complete accepted work prefix; no partial score is published.
93    pub fn score(
94        &self,
95        candidate: &Interpretation,
96        limits: Limits,
97        cancellation: &Cancellation,
98    ) -> Result<Option<ObjectiveScore>, ObjectiveScoreError> {
99        let mut work = Work {
100            cancellation,
101            limits: ObjectivePlanLimits {
102                max_work: limits.max_work,
103                ..ObjectivePlanLimits::default()
104            },
105            template: None,
106            statistics: ObjectiveBoundStatistics::default(),
107        };
108        work.charge(0).map_err(eligibility_error)?;
109        if !self.original.same_instance(candidate.theory()) {
110            return Err(ObjectiveScoreError {
111                kind: ObjectiveScoreErrorKind::WrongTheory,
112                work: 0,
113            });
114        }
115        if self.statistics.bindings > limits.max_bindings
116            || self.statistics.keys > limits.max_keys
117            || self.statistics.key_bytes > limits.max_key_bytes
118        {
119            return Ok(None);
120        }
121        let mut values = work.reserve(self.nodes.len()).map_err(eligibility_error)?;
122        for node in &self.nodes {
123            work.tick().map_err(eligibility_error)?;
124            let value = match *node {
125                Node::False => false,
126                Node::Atom(atom) => candidate.contains(atom),
127                Node::And(left, right) => values[left] && values[right],
128                Node::Or(left, right) => values[left] || values[right],
129                Node::Implies(left, right) => !values[left] || values[right],
130            };
131            values.push(value);
132        }
133        let truth = values.as_slice();
134        let reduced = zetesis_objective::reduce_costs(
135            &self.objectives,
136            |priority| {
137                self.levels
138                    .get(&priority)
139                    .into_iter()
140                    .flatten()
141                    .map(move |element| truth[element.condition].then_some(element.weight))
142            },
143            limits.max_work - work.statistics.work,
144            cancellation,
145        )
146        .map_err(|error| ObjectiveScoreError {
147            work: work.statistics.work + error.statistics().work,
148            kind: ObjectiveScoreErrorKind::Reduction(error),
149        })?;
150        Ok(Some(ObjectiveScore {
151            work: work.statistics.work + reduced.work(),
152            score: reduced.into_score(),
153        }))
154    }
155}

157fn eligibility_error(error: ObjectiveBoundError) -> ObjectiveScoreError {
158    ObjectiveScoreError {
159        work: error.statistics().work,
160        kind: ObjectiveScoreErrorKind::Eligibility(error),
161    }
