1//! Invert one admitted arithmetic path, then check the authored expression.
2//!
3//! Inverse calculations are widened only to find an ordinary scalar candidate.
4//! Nonintegral/out-of-range candidates do not match. Authored arithmetic still
5//! uses the common checked evaluator after capture; no failed output escapes.
6
7use themelios_program::term::BinaryOp;
8
9use super::{Binding, Error, ErrorKind, Interpreter, Template, UnaryOp, Work};
10
11fn contains(term: &Template, slot: usize, work: &mut Work<'_>) -> Result<bool, Error> {
12    work.step(1)?;
13    Ok(match term {
14        Template::Variable(found) => *found == slot,
15        Template::Unary(_, inner) => contains(inner, slot, work)?,
16        Template::Binary(_, left, right) => {
17            contains(left, slot, work)? || contains(right, slot, work)?
18        }
19        _ => false,
20    })
21}
22fn candidate<'input>(
23    term: &Template,
24    slot: usize,
25    target: i128,
26    binding: &Binding<'input>,
27    context: &mut Interpreter<'input, '_, '_>,
28) -> Result<Option<i128>, Error> {
29    context.work.step(1)?;
30    match term {
31        Template::Variable(found) if *found == slot => Ok(Some(target)),
32        Template::Unary(UnaryOp::Negate, inner) => {
33            candidate(inner, slot, -target, binding, context)
34        }
35        Template::Binary(operator, left, right) => {
36            let unknown_left = contains(left, slot, context.work)?;
37            let (unknown, known) = if unknown_left {
38                (left, right)
39            } else {
40                (right, left)
41            };
42            let known = i128::from(context.numeric(known, binding)?);
43            let target = match operator {
44                BinaryOp::Add => target - known,
45                BinaryOp::Sub if unknown_left => target + known,
46                BinaryOp::Sub => known - target,
47                BinaryOp::Mul if known == 0 => {
48                    return Err(context.work.error(ErrorKind::Unsupported(
49                        super::super::Feature::UnsafeVariable,
50                    )));
51                }
52                BinaryOp::Mul if target % known != 0 => return Ok(None),
53                BinaryOp::Mul => target / known,
54                _ => unreachable!("compiler admits only single-candidate inverse operators"),
55            };
56            candidate(unknown, slot, target, binding, context)
57        }
58        _ => unreachable!("compiler admits exactly one unknown occurrence"),
59    }
60}
61pub(super) fn bind<'input>(
62    slot: usize,
63    expression: &Template,
64    target: i32,
65    binding: &mut Binding<'input>,
66    undo: &mut Vec<usize>,
67    context: &mut Interpreter<'input, '_, '_>,
68) -> Result<bool, Error> {
69    if binding.is_bound(slot) {
70        return Ok(context.numeric(expression, binding)? == target);
71    }
72    let Some(value) = candidate(expression, slot, i128::from(target), binding, context)? else {
73        return Ok(false);
74    };
75    let Ok(value) = i32::try_from(value) else {
76        return Ok(false);
77    };
78    let mut metric = super::Metric::default();
79    context.work.step(1)?;
80    context.work.node(&mut metric)?;
81    context.work.check(
82        super::Resource::Depth,
83        1,
84        context.work.limits.max_symbol_depth as u128,
85    )?;
86    context.work.construction_check(metric)?;
87    context.work.check(
88        super::Resource::LocalBytes,
89        context.work.local_bytes + metric.payload(),
90        context.work.limits.max_local_bytes as u128,
91    )?;
92    let key = context.number(value)?;
93    binding.bind_term(slot, &key, metric, context.work)?;
94    undo.push(slot);
95    Ok(context.numeric(expression, binding)? == target)
96}
