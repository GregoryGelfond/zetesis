1//! Finite query traversal with explicit ownership for each binding depth.
2//!
3//! Cursors enumerate relational alternatives and generated values. A depth owns
4//! only its new bindings and expanded alternatives; inherited slots are borrowed.
5//! Exhaustion, visitor stop and errors release this query's logical ownership.
6

12    Resource, Visitor, Work, anonymous, complete, matches, patterns, scopes, values,

15enum Alternatives {
16    Terms(values::Values),
17    Keys(anonymous::Alternatives),
18}
19impl Alternatives {
20    fn bytes(&self) -> u128 {
21        match self {
22            Self::Terms(values) => values.bytes,
23            Self::Keys(values) => values.bytes,
24        }
25    }
26}

28enum Choice {
29    Atom { alternative: usize, row: usize },
30    Value(usize),
31}
32impl Choice {
33    /// Every installed value belongs to the depth's undo owner, including when
34    /// later matching refuses or fails.
35    fn bind<'input>(
36        self,
37        binder: &Binder,
38        alternatives: Option<&Alternatives>,
39        atoms: &ModelRows<'input>,
40        binding: &mut Binding<'input>,
41        undo: &mut Vec<usize>,
42        context: &mut Interpreter<'input, '_, '_>,
43    ) -> Result<bool, Error> {
44        match (binder, self) {
45            (Binder::NumericMismatch(aggregate), Self::Value(_)) => {
46                scopes::aggregate(aggregate, atoms, binding, context)?;
47                Ok(false)
48            }
49            (Binder::Atom(alternatives), Self::Atom { alternative, row }) => matches(
50                &alternatives[alternative],
51                atoms.get(row),
52                binding,
53                undo,
54                true,
55                context,
56            ),
57            (Binder::AtomKey(slot, _), Self::Value(cursor)) => {
58                let Some(Alternatives::Keys(values)) = alternatives else {
59                    unreachable!("atom-key cursor prepares typed alternatives")
60                };
61                context.work.step(1)?;
62                let (key, metric) = values.values[cursor];
63                admit_local(metric, context.work)?;
64                binding.bind_pattern(*slot, key, metric, context.work);
65                undo.push(*slot);
66                Ok(true)
67            }
68            (
69                binder @ (Binder::Aggregate(_, _) | Binder::Assign(_, _) | Binder::Match { .. }),
70                Self::Value(cursor),
71            ) => {
72                let terms = match alternatives {
73                    Some(Alternatives::Terms(values)) => Some(values),
74                    None => None,
75                    Some(Alternatives::Keys(_)) => {
76                        unreachable!("scalar cursor retains term alternatives")
77                    }
78                };
79                let owned = owned_binding(binder, terms, cursor, atoms, binding, context)?;
80                retain_binding(binder, owned, cursor, binding, undo, context)
81            }
82            _ => unreachable!("query cursors preserve binder kind"),
83        }
84    }
85}

87enum Cursor {
88    Atoms(AtomChoices),
89    Values(usize),
90}
91impl Cursor {
92    fn next<'input>(
93        &mut self,
94        binder: &Binder,
95        binding: &Binding<'input>,
96        alternatives: &mut Option<Alternatives>,
97        context: &mut Interpreter<'input, '_, '_>,
98    ) -> Result<Option<Choice>, Error> {
99        match self {
100            Self::Atoms(rows) => Ok(rows
101                .next(context.work)?
102                .map(|(alternative, row)| Choice::Atom { alternative, row })),
103            Self::Values(cursor) => {
104                let count = choice_count(binder, binding, alternatives, context)?;
105                if *cursor == count {
106                    *cursor = 0;
107                    Ok(None)
108                } else {
109                    let choice = *cursor;
110                    *cursor += 1;
111                    Ok(Some(Choice::Value(choice)))
112                }
113            }
114        }
115    }
116}
117fn cursors<'input>(
118    query: &Query,
119    atoms: &ModelRows<'input>,
120    context: &mut Interpreter<'input, '_, '_>,
121) -> Result<Vec<Cursor>, Error> {
122    let mut cursors = context.work.reserve(query.binders.len())?;
123    for binder in &query.binders {
124        cursors.push(match binder {
125            Binder::Atom(patterns) => Cursor::Atoms(AtomChoices::new(
126                patterns,
127                atoms,
128                context.metadata,
129                context.work,
130            )?),
131            Binder::Assign(..)
132            | Binder::AtomKey(..)
133            | Binder::Match { .. }
134            | Binder::Aggregate(..)
135            | Binder::NumericMismatch(_) => Cursor::Values(0),
136        });
137    }
138    Ok(cursors)
139}
140fn undo_slots(query: &Query, work: &Work<'_>) -> Result<Vec<Vec<usize>>, Error> {
141    let mut undos = work.reserve(query.binders.len())?;
142    for binder in &query.binders {
143        let count = match binder {
144            Binder::Atom(alternatives) => alternatives
145                .iter()
146                .map(|pattern| {
147                    pattern.terms.iter().map(patterns::slots).sum::<usize>()
148                        + usize::from(pattern.key.is_some())
149                })
150                .max()
151                .unwrap_or(0),
152            Binder::Assign(..) | Binder::AtomKey(..) | Binder::Aggregate(..) => 1,
153            Binder::NumericMismatch(_) => 0,
154            Binder::Match {
155                patterns: alternatives,
156                ..
157            } => alternatives.iter().map(patterns::slots).max().unwrap_or(0) + 1,
158        };
159        undos.push(work.reserve(count)?);
160    }
161    Ok(undos)
162}
163fn admit_local(metric: Metric, work: &Work<'_>) -> Result<(), Error> {
164    work.construction_check(metric)?;
165    work.check(
166        Resource::LocalBytes,
167        work.local_bytes + metric.payload(),
168        work.limits.max_local_bytes as u128,
169    )
170}
171fn owned_binding<'input>(
172    binder: &Binder,
173    alternatives: Option<&values::Values>,
174    cursor: usize,
175    atoms: &ModelRows<'input>,
176    binding: &Binding<'input>,
177    context: &mut Interpreter<'input, '_, '_>,
178) -> Result<(usize, TermKey, Metric), Error> {
179    let cursor = if let Binder::Match { patterns, .. } = binder {
180        cursor / patterns.len()
181    } else {
182        cursor
183    };
184    match binder {
185        Binder::Assign(slot, expression)
186        | Binder::Match {
187            complete: slot,
188            value: expression,
189            ..
190        } => {
191            let (key, metric) = if let Some(values) = alternatives {
192                context.work.step(1)?;
193                let metric = values.metric(cursor);
194                admit_local(metric, context.work)?;
195                (values.key(cursor, context)?, metric)
196            } else {
197                context.own(expression, binding)?
198            };
199            Ok((*slot, key, metric))
200        }
201        Binder::Aggregate(slot, aggregate) => {
202            let (value, metric) = scopes::aggregate(aggregate, atoms, binding, context)?;
203            context.work.check(
204                Resource::Nodes,
205                metric.nodes as u128,
206                context.work.limits.max_symbol_nodes as u128,
207            )?;
208            context.work.check(
209                Resource::Depth,
210                1,
211                context.work.limits.max_symbol_depth as u128,
212            )?;
213            admit_local(metric, context.work)?;
214            // Guard arithmetic stays wide. A retained logical scalar crosses
215            // the pinned width boundary only when this binder constructs it.
216            let key = match value {
217                scopes::Value::Integer(value) => {
218                    context.number(i32::try_from(value).map_err(|_| {
219                        context
220                            .work
221                            .error(ErrorKind::Evaluation(EvaluationError::Overflow))
222                    })?)?
223                }
224                scopes::Value::Term(key) => key,
225            };
226            Ok((*slot, key, metric))
227        }
228        Binder::Atom(_) | Binder::AtomKey(..) | Binder::NumericMismatch(_) => {
229            unreachable!("this binder does not retain a scalar")
230        }
231    }
232}
233fn choice_count<'input>(
234    binder: &Binder,
235    binding: &Binding<'input>,
236    alternatives: &mut Option<Alternatives>,
237    context: &mut Interpreter<'input, '_, '_>,
238) -> Result<usize, Error> {
239    Ok(match binder {
240        Binder::Atom(_) => unreachable!("atom cursors enumerate predicate ranges"),
241        Binder::Aggregate(_, _) | Binder::NumericMismatch(_) => 1,
242        Binder::AtomKey(_, template) => {
243            if alternatives.is_none() {
244                *alternatives = Some(Alternatives::Keys(anonymous::collect(
245                    template, binding, context,
246                )?));
247            }
248            let Some(Alternatives::Keys(values)) = alternatives else {
249                unreachable!("typed atom-key alternatives")
250            };
251            values.values.len()
252        }
253        Binder::Assign(_, expression)
254        | Binder::Match {
255            value: expression, ..
256        } => {
257            if expression.multiple() && alternatives.is_none() {
258                *alternatives = Some(Alternatives::Terms(values::collect(
259                    expression, binding, context,
260                )?));
261            }
262            let values = match alternatives {
263                Some(Alternatives::Terms(values)) => values.len(),
264                None => 1,
265                Some(Alternatives::Keys(_)) => unreachable!("typed scalar alternatives"),
266            };
267            let patterns = if let Binder::Match { patterns, .. } = binder {
268                patterns.len()
269            } else {
270                1
271            };
272            values.checked_mul(patterns).ok_or_else(|| {
273                context.work.error(ErrorKind::Limit {
274                    resource: Resource::Bindings,
275                    observed: (values as u128) * (patterns as u128),
276                    limit: u128::from(context.work.limits.max_bindings),
277                })
278            })?
279        }
280    })
281}
282fn retain_binding<'input>(
283    binder: &Binder,
284    owned: (usize, TermKey, Metric),
285    cursor: usize,
286    binding: &mut Binding<'input>,
287    undo: &mut Vec<usize>,
288    context: &mut Interpreter<'input, '_, '_>,
289) -> Result<bool, Error> {
290    let (slot, key, metric) = owned;
291    // Match's complete slot is freshly allocated by the compiler; its pattern
292    // cannot refer to it. Install ownership before the matcher can fail, so the
293    // same unwind releases both the whole term and any partial captures.
294    binding.bind_term(slot, &key, metric, context.work)?;
295    undo.push(slot);
296    if let Binder::Match {
297        patterns: alternatives,
298        ..
299    } = binder
300    {
301        patterns::bind_key(
302            &alternatives[cursor % alternatives.len()],
303            &key,
304            binding,
305            undo,
306            context,
307        )
308    } else {
309        Ok(true)
310    }
311}

313pub(super) fn visit<'input>(
314    query: &Query,
315    atoms: &ModelRows<'input>,
316    outer: Option<&Binding<'input>>,
317    context: &mut Interpreter<'input, '_, '_>,
318    visitor: &mut Visitor<'input, '_>,
319) -> Result<bool, Error> {
320    context
321        .work
322        .step(1 + query.variables as u128 + query.binders.len() as u128)?;
323    let mut binding = Binding::new(
324        query.variables,
325        outer,
326        context.terms.read(),
327        context.work.limits.max_term_storage_bytes,
328        context.work,
329    )?;
330    let mut alternatives: Vec<Option<Alternatives>> = context.work.reserve(query.binders.len())?;
331    alternatives.resize_with(query.binders.len(), || None);
332    let result = (|| {
333        if query.binders.is_empty() {
334            return complete(query, atoms, &mut binding, context, visitor);
335        }
336        let mut cursors = cursors(query, atoms, context)?;
337        let mut undos = undo_slots(query, context.work)?;
338        let mut depth = 0;
339        loop {
340            context.work.step(1 + undos[depth].len() as u128)?;
341            for slot in undos[depth].drain(..) {
342                binding.clear(slot, context.work);
343            }
344            let choice = cursors[depth].next(
345                &query.binders[depth],
346                &binding,
347                &mut alternatives[depth],
348                context,
349            )?;
350            let Some(choice) = choice else {
351                if let Some(values) = alternatives[depth].take() {
352                    context.work.local_bytes -= values.bytes();
353                }
354                if depth == 0 {
355                    break;
356                }
357                depth -= 1;
358                continue;
359            };
360            if !choice.bind(
361                &query.binders[depth],
362                alternatives[depth].as_ref(),
363                atoms,
364                &mut binding,
365                &mut undos[depth],
366                context,
367            )? {
368                continue;
369            }
370            if depth + 1 == query.binders.len() {
371                if !complete(query, atoms, &mut binding, context, visitor)? {
372                    return Ok(false);
373                }
374            } else {
375                depth += 1;
376            }
377        }
378        Ok(true)
379    })();
380    // The visitor may retain aggregate keys in an enclosing scope. Release only
381    // this query's charges, on exhaustion, early termination and errors alike.
382    context.work.local_bytes -= alternatives
383        .iter()
384        .flatten()
385        .map(Alternatives::bytes)
386        .sum::<u128>();
387    binding.release(context.work);
388    result
