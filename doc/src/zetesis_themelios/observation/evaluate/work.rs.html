1//! One resource and cancellation account shared by evaluation and rendering.
2//! This account retains neither a query program nor a term authority.
3
4use themelios_base::span::Location;
5
6use super::{Cancellation, ConstructionLimits, Error, ErrorKind, Limits, Resource, Statistics};
7
8pub(in crate::observation) struct Work<'a> {
9    pub limits: Limits,
10    pub construction: ConstructionLimits,
11    pub cancellation: &'a Cancellation,
12    pub statistics: Statistics,
13    pub location: Option<Location>,
14    pub local_bytes: u128,
15}
16
17impl Work<'_> {
18    pub fn error(&self, kind: ErrorKind) -> Error {
19        Error::new(kind, self.location, self.statistics)
20    }
21
22    pub fn check(&self, resource: Resource, observed: u128, limit: u128) -> Result<(), Error> {
23        if observed > limit {
24            return Err(self.error(ErrorKind::Limit {
25                resource,
26                observed,
27                limit,
28            }));
29        }
30        Ok(())
31    }
32
33    pub fn step(&mut self, count: u128) -> Result<(), Error> {
34        self.cancellation
35            .poll()
36            .map_err(|stop| self.error(ErrorKind::Stopped(stop)))?;
37        let observed = u128::from(self.statistics.work) + count;
38        self.check(Resource::Work, observed, u128::from(self.limits.max_work))?;
39        self.statistics.work = u64::try_from(observed).expect("work checked against u64 ceiling");
40        Ok(())
41    }
42
43    pub fn reserve<T>(&self, count: usize) -> Result<Vec<T>, Error> {
44        let mut values = Vec::new();
45        values
46            .try_reserve_exact(count)
47            .map_err(|_| self.error(ErrorKind::Allocation))?;
48        Ok(values)
49    }
50}
