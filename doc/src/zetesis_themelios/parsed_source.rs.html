16/// One syntax-checked input with fixed source-admission options.
17///
18/// Construction takes ownership of the original UTF-8 bytes and parses once.
19/// The source-byte ceiling bounds that input; each later profile operation
20/// checks its syntax-node/depth and language restrictions in their existing
21/// order. Successful parsing therefore does not certify a language profile,
22/// finite grounding, or existence of an answer set.
23///
24/// Retaining this owner retains the source and upstream syntax tree, with
25/// O(source bytes + syntax nodes) space. Moving it between profile attempts
26/// neither clones the source nor reparses it. Each attempt still performs its
27/// own checked raising and normalization with the supplied expansion limits.
28///
29/// ```
30/// use zetesis_themelios::{AdmissionOptions, ExpansionLimits, FormulaLimits, ParsedSource};
31/// let source = ParsedSource::new("1{a;b}1.".into(), AdmissionOptions::default())?;
32/// let refused = source.admit_extended(ExpansionLimits::default()).unwrap_err();
33/// assert!(refused.error().needs_formula_admission());
34/// let prepared = refused.into_source()
35///     .prepare_formula(ExpansionLimits::default(), FormulaLimits::default())?;
36/// let admitted = prepared.ground()?;
37/// assert_eq!(admitted.atoms().len(), 2);
38/// # Ok::<(), Box<dyn std::error::Error>>(())
39/// ```
40#[derive(Debug)]
41pub struct ParsedSource {
42    source: Source,
43    parsed: Parse<ast::Program>,
44    options: AdmissionOptions,
45}
46
47impl ParsedSource {
48    /// Parse original bytes under the clingo dialect, retaining their identity.
49    /// Parsing is linear in the admitted input; no filesystem access occurs.
50    ///
51    /// # Errors
52    /// Returns the original source/byte refusal or complete parser diagnostics.
53    /// Syntax failure owns the rejected source; no recovered owner is returned.
54    pub fn new(text: String, options: AdmissionOptions) -> Result<Self, AdmissionFailure> {
55        let start = Location {
56            source: options.source_id,
57            span: Span::empty(ByteOffset::new(0)),
58        };
59        if text.len() > options.max_source_bytes {
60            return Err(AdmissionFailure::Limit {
61                resource: InputLimit::SourceBytes,
62                limit: options.max_source_bytes,
63                observed: text.len(),
64                location: start.into(),
65            });
66        }
67        let source =
68            Source::new(options.source_id, text).map_err(|error| AdmissionFailure::Source {
69                error,
70                location: start,
71            })?;
72        let parsed = parse(&source, Dialect::Clingo);
73        if !parsed.diagnostics().is_empty() {
74            let diagnostics = parsed.diagnostics().to_vec();
75            return Err(AdmissionFailure::Syntax(SyntaxFailure::new(
76                source,
77                diagnostics,
78            )));
79        }
80        Ok(Self {
81            source,
82            parsed,
83            options,
84        })
85    }
86
87    /// Original source identity and bytes, borrowed without reconstruction.
88    #[must_use]
89    pub fn source(&self) -> &Source {
90        &self.source
91    }
92
93    /// Original successful parse; locations refer to [`Self::source`].
94    #[must_use]
95    pub fn parsed(&self) -> &Parse<ast::Program> {
96        &self.parsed
97    }
98
99    /// The immutable source/profile limits used by each admission attempt.
100    #[must_use]
101    pub fn options(&self) -> AdmissionOptions {
102        self.options
103    }
104
105    /// Admit relational templates through the extended source profile.
106    /// Costs and refusals are those of [`crate::admit_extended`] after parsing.
107    /// This computes no answer sets and performs no static grounding.
108    ///
109    /// # Errors
110    /// Retains this exact owner beside the typed expansion refusal. A caller
111    /// may explicitly recover it to attempt formula preparation; recovery does
112    /// not classify the error as eligible for that alternative profile.
113    pub fn admit_extended(
114        self,
115        limits: ExpansionLimits,
116    ) -> Result<Admitted, SourceFailure<ExpansionFailure>> {
117        extended::admit_parsed(self, limits, None)
118    }
119
120    /// Admit relational templates while sharing cancellation and deadline control.
121    /// Parsing and upstream profile/raising calls are bounded cooperative intervals;
122    /// expansion polls the same token at its existing checked work boundaries.
123    ///
124    /// # Errors
125    /// Retains this owner with ordinary admission failures or a typed interruption.
126    pub fn admit_extended_with_cancellation(
127        self,
128        limits: ExpansionLimits,
129        cancellation: &zetesis_cpu::Cancellation,
130    ) -> Result<Admitted, SourceFailure<ExpansionFailure>> {
131        extended::admit_parsed(self, limits, Some(cancellation.clone()))
132    }
133
134    /// Prepare the finite formula profile without reparsing original bytes.
135    /// Costs and budgets are those of [`crate::prepare_formula`] after parsing.
136    /// The returned preparation retains the original source until grounding.
137    ///
138    /// # Errors
139    /// Retains this owner beside the complete typed preparation refusal. No
140    /// partial preparation or completed grounding is returned on failure.
141    pub fn prepare_formula(
142        self,
143        expansion: ExpansionLimits,
144        limits: FormulaLimits,
145    ) -> Result<PreparedFormula, SourceFailure<FormulaFailure>> {
146        formula::prepare_parsed(self, expansion, &limits, None)
147    }
148
149    /// Prepare formulas under shared cancellation and deadline control.
150    /// The returned preparation retains the token through later support completion
151    /// and eager, hybrid or terminal materialization. Counters are not refreshed.
152    /// Upstream profile/raising calls remain bounded cooperative intervals.
153    ///
154    /// # Errors
155    /// Retains this owner with ordinary formula failures or a typed interruption.
156    pub fn prepare_formula_with_cancellation(
157        self,
158        expansion: ExpansionLimits,
159        limits: FormulaLimits,
160        cancellation: &zetesis_cpu::Cancellation,
161    ) -> Result<PreparedFormula, SourceFailure<FormulaFailure>> {
162        formula::prepare_parsed(self, expansion, &limits, Some(cancellation.clone()))
163    }
164
165    /// Recover original source bytes and identity, discarding the syntax tree.
166    #[must_use]
167    pub fn into_source(self) -> Source {
168        self.source
169    }
170}

172/// A profile refusal retaining its exact parsed source for inspection or reuse.
173/// One allocation retains the original parsed owner and typed error together,
174/// without copying source bytes or the syntax tree. The public wrapper holds
175/// only that owner handle. Retained space includes the source and cause. No
176/// partial compiled program or preparation is retained as a successful result.
177pub struct SourceFailure<E> {
178    retained: Box<RetainedFailure<E>>,
179}
180
181struct RetainedFailure<E> {
182    source: ParsedSource,
183    error: E,
184}
185
186impl<E> SourceFailure<E> {
187    pub(crate) fn new(source: ParsedSource, error: E) -> Self {
188        Self {
189            retained: Box::new(RetainedFailure { source, error }),
190        }
191    }
192
193    /// The original source and syntax owner; no clone or reparse occurs.
194    #[must_use]
195    pub fn source(&self) -> &ParsedSource {
196        &self.retained.source
197    }
198
199    /// The unchanged typed refusal from the attempted profile.
200    #[must_use]
201    pub fn error(&self) -> &E {
202        &self.retained.error
203    }
204
205    /// Recover the source owner, discarding the refusal.
206    #[must_use]
207    pub fn into_source(self) -> ParsedSource {
208        self.retained.source
209    }
210
211    /// Recover the refusal, discarding the source and syntax tree.
212    #[must_use]
213    pub fn into_error(self) -> E {
214        self.retained.error
215    }
216
217    /// Recover both original owners without cloning either.
218    #[must_use]
219    pub fn into_parts(self) -> (ParsedSource, E) {
220        (self.retained.source, self.retained.error)
221    }
222}
223
224impl<E: fmt::Debug> fmt::Debug for SourceFailure<E> {
225    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
226        formatter
227            .debug_struct("SourceFailure")
228            .field("source", &self.retained.source)
229            .field("error", &self.retained.error)
230            .finish()
231    }
232}
233
234impl<E: fmt::Display> fmt::Display for SourceFailure<E> {
235    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
236        self.retained.error.fmt(formatter)
237    }
238}
239
240impl<E: std::error::Error + 'static> std::error::Error for SourceFailure<E> {
241    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
242        Some(&self.retained.error)
243    }
244}

250    #[test]
251    fn refusal_recovery_transfers_original_owners() {
252        struct Cause(Box<u8>);
253
254        let source = ParsedSource::new("p.".into(), AdmissionOptions::default()).unwrap();
255        let source_bytes = source.source().text().as_ptr();
256        let error = Cause(Box::new(7));
257        let error_payload = std::ptr::from_ref(error.0.as_ref());
258        let failure = SourceFailure::new(source, error);
259        assert_eq!(failure.source().source().text().as_ptr(), source_bytes);
260        assert_eq!(
261            std::ptr::from_ref(failure.error().0.as_ref()),
262            error_payload
263        );
264        let (source, error) = failure.into_parts();
265        assert_eq!(source.source().text().as_ptr(), source_bytes);
266        assert_eq!(std::ptr::from_ref(error.0.as_ref()), error_payload);
267        let error = SourceFailure::new(source, error).into_error();
268        assert_eq!(std::ptr::from_ref(error.0.as_ref()), error_payload);
269        assert_eq!(*error.0, 7);
270    }
271    #[test]
272    fn relational_control_retains_the_refused_source() {
273        let source = ParsedSource::new("p(1..3).".into(), AdmissionOptions::default()).unwrap();
274        let pointer = source.source().text().as_ptr();
275        let cancellation = zetesis_cpu::Cancellation::default();
276        cancellation.cancel();
277        let failure = source
278            .admit_extended_with_cancellation(ExpansionLimits::default(), &cancellation)
279            .unwrap_err();

281            failure.error(),
282            ExpansionFailure::Interrupted {
283                reason: zetesis_cpu::Stop::Cancelled,
284                ..
285            }
286        ));
287        assert_eq!(failure.source().source().text().as_ptr(), pointer);
288    }
