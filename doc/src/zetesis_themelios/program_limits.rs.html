1//! Bounded borrowed-term inspection shared by logical formula validation.
2//!
3//! The cursor retains one remaining-siblings slice per open term or symbol,
4//! using O(depth) space. Node, depth and text charges precede child traversal.
5
6use std::fmt;

13/// The logical resource whose configured ceiling was exceeded.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum Resource {
16    /// Visited logical nodes. Formula preparation also counts directive,
17    /// provenance and annotation structure.
18    Nodes,
19    /// Nesting of visited logical nodes, with each statement at depth one.
20    Depth,
21    /// Canonical elements in one body or, for formula preparation, condition.
22    BodyElements,
23    /// UTF-8 bytes in visited logical names and strings. Formula preparation
24    /// additionally counts annotation text and transformation tags.
25    TextBytes,
26}
27
28impl fmt::Display for Resource {
29    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
30        f.write_str(match self {
31            Self::Nodes => "logical nodes",
32            Self::Depth => "logical depth",
33            Self::BodyElements => "body elements",
34            Self::TextBytes => "logical text bytes",
35        })
36    }
37}

39/// A logical resource refusal before compilation of the offending statement.
40#[derive(Clone, Copy, Debug, PartialEq, Eq)]
41pub struct Limit {
42    /// The resource being counted.
43    pub resource: Resource,
44    /// The configured maximum.
45    pub limit: usize,
46    /// The count at refusal, saturated at `usize::MAX` if addition overflowed.
47    pub observed: usize,
48}
49
50impl fmt::Display for Limit {
51    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52        write!(
53            f,
54            "{} limit {} exceeded (observed {})",
55            self.resource, self.limit, self.observed
56        )
57    }
58}
59
60impl std::error::Error for Limit {}

62#[derive(Clone, Copy)]
63pub(crate) struct Limits {
64    pub(crate) nodes: usize,
65    pub(crate) depth: usize,
66    pub(crate) body_elements: usize,
67    pub(crate) text_bytes: usize,
68}
69
70/// Node and text counts are cumulative across calls; depth and body width are
71/// checked separately for each rule. A failed check must abort admission.
72pub(crate) struct Budget {
73    limits: Limits,
74    nodes: usize,
75    text_bytes: usize,
76}
77
78impl Budget {
79    pub(crate) const fn new(limits: Limits) -> Self {
80        Self {
81            limits,
82            nodes: 0,
83            text_bytes: 0,
84        }
85    }
86
87    pub(crate) fn node(&mut self, depth: usize) -> Result<(), Limit> {
88        check(Resource::Depth, self.limits.depth, depth)?;
89        charge(&mut self.nodes, 1, Resource::Nodes, self.limits.nodes)
90    }
91
92    pub(crate) fn text(&mut self, text: &str) -> Result<(), Limit> {
93        charge(
94            &mut self.text_bytes,
95            text.len(),
96            Resource::TextBytes,
97            self.limits.text_bytes,
98        )
99    }
100
101    /// Inspect bounded borrowed terms without a second recursive traversal.
102    /// The callback runs after each term's node/depth charge and before children.
103    pub(crate) fn inspect_term(
104        &mut self,
105        term: &Term,
106        depth: usize,
107        inspect: impl FnMut(&Term),
108    ) -> Result<(), Limit> {
109        self.values(Frame::Term(term, depth), inspect)
110    }
111
112    pub(crate) fn body_elements(&self, observed: usize) -> Result<(), Limit> {
113        check(Resource::BodyElements, self.limits.body_elements, observed)
114    }
115
116    fn values(&mut self, first: Frame<'_>, mut inspect: impl FnMut(&Term)) -> Result<(), Limit> {
117        let mut pending = vec![first];
118        while let Some(frame) = pending.pop() {
119            match frame {
120                Frame::Terms(terms, depth) => {
121                    if let Some((first, rest)) = terms.split_first() {
122                        pending.push(Frame::Terms(rest, depth));
123                        pending.push(Frame::Term(first, depth));
124                    }
125                }
126                Frame::Symbols(symbols, depth) => {
127                    if let Some((first, rest)) = symbols.split_first() {
128                        pending.push(Frame::Symbols(rest, depth));
129                        pending.push(Frame::Symbol(first, depth));
130                    }
131                }
132                Frame::Term(term, depth) => {
133                    self.node(depth)?;
134                    inspect(term);
135                    let child_depth = depth.saturating_add(1);
136                    match term {
137                        Term::Variable(Variable::Named(name)) => self.text(name.as_str())?,
138                        Term::Variable(Variable::Anonymous) => {}
139                        Term::Symbolic(symbol) => {
140                            pending.push(Frame::Symbol(symbol, child_depth));
141                        }
142                        Term::Function { name, arguments } | Term::External { name, arguments } => {
143                            self.text(name.as_str())?;
144                            pending.push(Frame::Terms(arguments, child_depth));
145                        }
146                        Term::Tuple(terms) | Term::Pool(terms) => {
147                            pending.push(Frame::Terms(terms, child_depth));
148                        }
149                        Term::UnaryOperation { argument, .. } | Term::Absolute(argument) => {
150                            pending.push(Frame::Term(argument, child_depth));
151                        }
152                        Term::BinaryOperation { left, right, .. }
153                        | Term::Interval {
154                            lower: left,
155                            upper: right,
156                        } => {
157                            pending.push(Frame::Term(right, child_depth));
158                            pending.push(Frame::Term(left, child_depth));
159                        }
160                    }
161                }
162                Frame::Symbol(symbol, depth) => {
163                    self.node(depth)?;
164                    let child_depth = depth.saturating_add(1);
165                    match symbol {
166                        Symbol::String(text) => self.text(text)?,
167                        Symbol::Function {
168                            name, arguments, ..
169                        } => {
170                            self.text(name.as_str())?;
171                            pending.push(Frame::Symbols(arguments, child_depth));
172                        }
173                        Symbol::Tuple(symbols) => {
174                            pending.push(Frame::Symbols(symbols, child_depth));
175                        }
176                        Symbol::Infimum | Symbol::Number(_) | Symbol::Supremum => {}
177                    }
178                }
179            }
180        }
181        Ok(())
182    }
183}

185/// A pending borrowed value, or the untouched suffix of one child sequence.
186/// Sibling frames do not count as logical nodes or increase logical depth.
187enum Frame<'a> {
188    Term(&'a Term, usize),
189    Symbol(&'a Symbol, usize),
190    Terms(&'a [Term], usize),
191    Symbols(&'a [Symbol], usize),
192}
193
194fn check(resource: Resource, limit: usize, observed: usize) -> Result<(), Limit> {
195    Meter::check_count(observed, 0, limit)
196        .map(|_| ())
197        .map_err(|_| Limit {
198            resource,
199            limit,
200            observed,
201        })
202}
203
204fn charge(used: &mut usize, amount: usize, resource: Resource, limit: usize) -> Result<(), Limit> {
205    *used = Meter::check_count(*used, amount, limit).map_err(|error| Limit {
206        resource,
207        limit,
208        observed: match error {
209            MeterCountError::Limit { observed, .. } => observed,
210            MeterCountError::Overflow => usize::MAX,
211        },
212    })?;
213    Ok(())
214}
