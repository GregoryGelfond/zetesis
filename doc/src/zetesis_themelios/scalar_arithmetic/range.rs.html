1//! Conservative intervals for checked scalar arithmetic.
2//!
3//! Endpoints and every admitted intermediate lie within i32. Transfers are
4//! calculated in i64 and decline when they cannot establish a defined i32
5//! result for every pair of operands. Callers establish the numeric source
6//! domains independently; this operation supplies neither bindings nor truth.
7
8use themelios_program::term::{BinaryOp, UnaryOp};
9
10#[derive(Clone, Copy)]
11pub(crate) struct Range {
12    low: i64,
13    high: i64,
14}
15
16impl Range {
17    pub(crate) fn point(value: i32) -> Self {
18        Self {
19            low: i64::from(value),
20            high: i64::from(value),
21        }
22    }
23
24    fn checked(low: i64, high: i64) -> Option<Self> {
25        (low >= i64::from(i32::MIN) && high <= i64::from(i32::MAX)).then_some(Self { low, high })
26    }
27
28    /// Enlarge a numeric bound by one admitted source value.
29    pub(crate) fn include(&mut self, value: i32) {
30        self.low = self.low.min(i64::from(value));
31        self.high = self.high.max(i64::from(value));
32    }
33
34    pub(crate) fn absolute(self) -> Option<Self> {
35        let low = if self.low <= 0 && self.high >= 0 {
36            0
37        } else {
38            self.low.abs().min(self.high.abs())
39        };
40        Self::checked(low, self.low.abs().max(self.high.abs()))
41    }
42
43    pub(crate) fn unary(self, operator: UnaryOp) -> Option<Self> {
44        match operator {
45            UnaryOp::Negate => Self::checked(-self.high, -self.low),
46            UnaryOp::BitwiseNot => Self::checked(!self.high, !self.low),
47        }
48    }
49
50    pub(crate) fn binary(self, operator: BinaryOp, right: Self) -> Option<Self> {
51        match operator {
52            BinaryOp::Add => Self::checked(self.low + right.low, self.high + right.high),
53            BinaryOp::Sub => Self::checked(self.low - right.high, self.high - right.low),
54            BinaryOp::Mul => self.corners(right, |left, right| Some(left * right)),
55            BinaryOp::Div | BinaryOp::Mod
56                if (right.low <= 0 && right.high >= 0)
57                    || (self.low == i64::from(i32::MIN) && right.low <= -1 && right.high >= -1) =>
58            {
59                None
60            }
61            BinaryOp::Div => self.corners(right, |left, right| Some(left / right)),
62            BinaryOp::Mod => {
63                let magnitude = right.low.abs().max(right.high.abs()) - 1;
64                Self::checked(
65                    self.low.min(0).max(-magnitude),
66                    self.high.max(0).min(magnitude),
67                )
68            }
69            BinaryOp::Pow if right.low == right.high && right.low >= 0 => {
70                let exponent = u32::try_from(right.low).ok()?;
71                let bound = self.corners(right, |left, _| left.checked_pow(exponent))?;
72                Some(Self {
73                    low: if exponent > 0 && self.low <= 0 && self.high >= 0 {
74                        bound.low.min(0)
75                    } else {
76                        bound.low
77                    },
78                    high: bound.high,
79                })
80            }
81            BinaryOp::Pow => None,
82            BinaryOp::BitAnd | BinaryOp::BitOr | BinaryOp::BitXor => {
83                Self::checked(i64::from(i32::MIN), i64::from(i32::MAX))
84            }
85        }
86    }
87
88    fn corners(self, right: Self, operation: impl Fn(i64, i64) -> Option<i64>) -> Option<Self> {
89        let corners = [
90            operation(self.low, right.low)?,
91            operation(self.low, right.high)?,
92            operation(self.high, right.low)?,
93            operation(self.high, right.high)?,
94        ];
95        Self::checked(*corners.iter().min()?, *corners.iter().max()?)
96    }
97}
