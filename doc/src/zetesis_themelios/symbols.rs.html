1//! Bounded export of borrowed logical values to themelios symbols.
2//!
3//! These operations construct the canonical upstream [`Symbol`] directly from
4//! [`TermRef`] and [`AtomRef`]. They do not render text, reparse it, or construct
5//! an intermediate owned core value. The input remains borrowed and unchanged.
6//! A caller exporting a complete interpretation visits every atom; source
7//! display selection does not belong to this conversion.
8//!
9//! The allowance covers named output and traversal capacities, including
10//! allocator overcapacity. It excludes allocator bookkeeping and the borrowed
11//! input. A callback can charge cumulative work and poll cancellation before
12//! navigation, reservation, text copying, and symbol assembly. Each failure
13//! returns no partial symbol. Collecting the returned symbols into themelios's
14//! `AnswerSet` uses the upstream `BTreeSet` allocation boundary; these operations
15//! do not make that collection's allocations fallible.
16
17use themelios_program::Symbol;
18use zetesis_core::catalog::{AtomRef, TermRef};
19
20pub use crate::structural_value::{BridgeError as Error, BridgeFailure as Failure};
21
22/// Export a borrowed closed term under one construction allowance.
23///
24/// Work is linear in the expanded logical nodes and text, plus canonical
25/// navigation and depth measurement. A borrowed ingress subtree requires an
26/// allocation-free depth scan with quadratic worst-case work; every scan step
27/// checks the callback. Complete ingress and canonical values use cached depth.
28/// Retained output is linear in the expansion; traversal scratch is proportional
29/// to its depth. Shared subterms become the tree required by the upstream symbol,
30/// without an intermediate owned core value.
31///
32/// # Errors
33/// Returns a typed storage, allocation, or name refusal, or the callback's
34/// original failure. No partial symbol is returned.
35pub fn term_with<E>(
36    value: TermRef<'_>,
37    max_bytes: u128,
38    before: impl FnMut() -> Result<(), E>,
39) -> Result<Symbol, Failure<E>> {
40    crate::structural_value::term_symbol_with(value, max_bytes, before)
41}
42
43/// Export a complete signed atom under one construction allowance.
44///
45/// The allowance covers the predicate, all arguments, and traversal scratch
46/// together; it is not reset for each argument. Work is linear in the expanded
47/// logical nodes and text, plus canonical navigation. Retained output is linear
48/// in that expansion; traversal scratch is proportional to the greatest depth.
49///
50/// ```
51/// use zetesis_core::{Atom, Predicate, Sign, Value};
52/// use zetesis_cpu::Cancellation;
53/// use zetesis_themelios::{logical::Symbol, symbols};
54///
55/// let atom = Atom::new(
56///     Predicate::with_sign("p", 1, Sign::Negative)?,
57///     vec![Value::Number(i32::MIN)],
58/// )?;
59/// let cancellation = Cancellation::default();
60/// let symbol = symbols::atom_with((&atom).into(), 4096, || cancellation.poll())?;
61/// assert!(matches!(&symbol, Symbol::Function { arguments, sign, .. }
62///     if arguments == &[Symbol::Number(i32::MIN)]
63///         && *sign == zetesis_themelios::logical::Sign::Negative));
64/// # Ok::<(), Box<dyn std::error::Error>>(())
65/// ```
66///
67/// # Errors
68/// Returns a typed storage, allocation, or name refusal, or the callback's
69/// original failure. No partial symbol is returned.
70pub fn atom_with<E>(
71    atom: AtomRef<'_>,
72    max_bytes: u128,
73    before: impl FnMut() -> Result<(), E>,
74) -> Result<Symbol, Failure<E>> {
75    crate::structural_value::atom_symbol_with(atom, max_bytes, before)
76}
