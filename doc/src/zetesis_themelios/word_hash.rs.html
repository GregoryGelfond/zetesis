1//! The fixed word hash the crate's indexes share.
2//!
3//! An index the crate keeps is placed by this hash rather than the
4//! standard library's randomized one, for a placement that is reproducible
5//! and a mixing that is a few instructions per word. The formula node
6//! index is keyed by identities the builder assigns, which no input drives
7//! into collisions. The JSON document's table of spelled atoms is keyed by
8//! what a program's author spells, and he could choose atoms that collide;
9//! that costs a lookup a scan of the table, bounded by its ceiling, and the
10//! solving his program commands already costs him more than any table
11//! could, while the randomized hasher was measured at 1.09 to 1.16 of the
12//! cell time on the series' large-model cells.
13
14use std::hash::Hasher;
15
16/// Folds the words of a key through [`zetesis_core::mix_word`], the mix
17/// zetesis-core's atom identity map also uses.
18#[derive(Default)]
19pub(crate) struct WordHasher(u64);
20
21impl Hasher for WordHasher {
22    fn finish(&self) -> u64 {
23        self.0
24    }
25    fn write(&mut self, bytes: &[u8]) {
26        for &byte in bytes {
27            self.write_u64(u64::from(byte));
28        }
29    }
30    fn write_u8(&mut self, word: u8) {
31        self.write_u64(u64::from(word));
32    }
33    fn write_usize(&mut self, word: usize) {
34        self.write_u64(word as u64);
35    }
36    fn write_u64(&mut self, word: u64) {
37        self.0 = zetesis_core::mix_word(self.0, word);
38    }
39}
