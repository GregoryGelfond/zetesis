33/// A theory over atoms `a` and `b` without roots, whose nodes are every
34/// connective the conditions use: falsity and truth, `a` and `b`, `not a`,
35/// `not not a` and `not b`, and `a` with `b` under conjunction, disjunction
36/// and implication.
37///
38/// # Panics
39/// Panics if the default admission limits refuse the theory; they admit it.
40#[must_use]
41pub fn theory() -> Theory {
42    let parts = FormulaParts::new(connectives(), vec![]).unwrap();
43    Theory::new(2, parts, vec![], AdmissionLimits::default()).unwrap()
44}
45
46/// The first nodes of every theory the Ferraris aggregate propositions
47/// build: [`theory`]'s, then `a or not a`, a choice of `a` under the Ferraris
48/// reduct.
49///
50/// # Panics
51/// Panics if the finite fixture's operand count overflows; it does not.
52#[must_use]
53pub fn ferraris_prefix() -> FormulaNodes {
54    let mut nodes = connectives();
55    nodes.push(Node::or_pair([2, 4]));
56    FormulaNodes::new(FormulaParts::new(nodes, vec![]).unwrap())
57}

59/// The theory over atoms `a` and `b` whose nodes are [`ferraris_prefix`]'s,
60/// without roots.
61///
62/// # Panics
63/// Panics if the default admission limits refuse the theory; they admit it.
64#[must_use]
65pub fn ferraris_theory() -> Theory {
66    Theory::new(
67        2,
68        ferraris_prefix().into_parts(),
69        vec![],
70        AdmissionLimits::default(),
71    )
72    .unwrap()
73}

75/// Falsity and truth, `a` and `b`, `not a`, `not not a` and `not b`, and `a`
76/// with `b` under conjunction, disjunction and implication.
77fn connectives() -> Vec<Node> {
78    vec![
79        Node::falsum(),
80        Node::implies(0, 0),
81        Node::atom(0),
82        Node::atom(1),
83        Node::implies(2, 0),
84        Node::implies(4, 0),
85        Node::implies(3, 0),
86        Node::and_pair([2, 3]),
87        Node::or_pair([2, 3]),
88        Node::implies(2, 3),
89    ]
90}

92/// The group of `function` over one tuple per entry of `first`, under
93/// `guards`. A tuple's key is its entry with its position, or empty for
94/// `None`, and its condition is the next of [`theory`]'s eight nodes past the
95/// constants, in turn.
96///
97/// # Panics
98/// Panics if the group is refused, as it is over a theory other than
99/// [`theory`].
100#[must_use]
101pub fn operation(
102    theory: &Theory,
103    function: Function,
104    first: Vec<Option<Value>>,
105    guards: Vec<Guard>,
106) -> Group {
107    let tuples = first
108        .into_iter()
109        .enumerate()
110        .map(|(index, value)| Tuple {
111            key: value.map_or_else(Vec::new, |value| {
112                vec![value, Value::Number(i32::try_from(index).unwrap())]
113            }),
114            condition: [2, 3, 4, 5, 6, 7, 8, 9][index % 8],
115        })
116        .collect::<Vec<_>>();
117    let limits = native::AdmissionLimits {
118        max_guards: guards.len(),
119        ..native::AdmissionLimits::default()
120    };
121    Group::new(
122        theory,
123        function,
124        tuples,
125        guards,
126        limits,
127        &Cancellation::default(),
128    )
129    .unwrap()
130}

132/// The group of `function` over `count` tuples whose values run through
133/// -3, 2, 0, 7 and -1 in turn, guarded by `≥ 0`.
134///
135/// # Panics
136/// As [`operation`].
137#[must_use]
138pub fn group(theory: &Theory, function: Function, count: usize) -> Group {
139    operation(
140        theory,
141        function,
142        (0..count)
143            .map(|index| Some(Value::Number([-3, 2, 0, 7, -1][index % 5])))
144            .collect(),
145        vec![Guard {
146            comparison: Comparison::Ge,
147            bound: Bound::Integer(0),
148        }],
149    )
150}

152/// The four interpretations of [`theory`]'s two atoms.
153///
154/// # Panics
155/// Panics if `theory` has fewer than two atoms.
156#[must_use]
157pub fn worlds(theory: &Theory) -> Vec<Interpretation> {
158    (0..4)
159        .map(|bits| {
160            Interpretation::new(theory, (0..2).filter(|atom| bits & (1 << atom) != 0)).unwrap()
161        })
162        .collect()
163}

165/// `count` eligibilities of `group`, five over each original world of
166/// `worlds` in turn: the first four freeze each world in turn, and the fifth
167/// is acquired without a frozen world.
168///
169/// # Panics
170/// Panics if `worlds` holds fewer than four interpretations or an
171/// eligibility is refused.
172#[must_use]
173pub fn observations<'a>(
174    group: impl Into<native::GroupRef<'a>>,
175    worlds: &'a [Interpretation],
176    count: usize,
177) -> Vec<Eligibility<'a>> {
178    let group = group.into();
179    (0..count)
180        .map(|index| {
181            group
182                .eligibility(
183                    &worlds[(index / 5) % 4],
184                    (index % 5 != 4).then(|| &worlds[index % 5]),
185                    native::EligibilityLimits::default(),
186                    &Cancellation::default(),
187                )
188                .unwrap()
189        })
190        .collect()
191}

193#[cfg(test)]
194mod tests {
195    use super::*;
196
197    #[test]
198    fn a_group_acquires_one_eligibility_per_observation() {
199        let theory = theory();
200        let worlds = worlds(&theory);
201        assert_eq!(worlds.len(), 4);
202        let group = group(&theory, Function::Sum, 5);
203        assert_eq!(observations(&group, &worlds, 7).len(), 7);
204    }
205
206    #[test]
207    fn the_ferraris_prefix_is_the_device_connectives_then_a_or_not_a() {
208        let prefix = ferraris_prefix();
209        let prefix = prefix.parts().nodes();
210        assert_eq!(prefix[..prefix.len() - 1], *theory().nodes());
211        assert_eq!(prefix.last(), Some(&Node::or_pair([2, 4])));
212        assert_eq!(ferraris_theory().nodes(), prefix);
213    }
