4pub(super) fn integers(text: &str, limits: Limits) -> Result<Vec<i64>, Error> {
5    check(
6        Resource::CostDimensions,
7        limits.max_cost_dimensions,
8        text.split_whitespace().count(),
9    )?;
10    text.split_whitespace()
11        .map(|value| {
12            value.parse().map_err(|_| {
13                invalid(
14                    Issue::MalformedField,
15                    "cost integer outside signed 64-bit report representation",
16                )
17            })
18        })
19        .collect()
20}

22pub(super) fn split_atoms(text: &str, comma_separated: bool) -> Result<Model, Error> {
23    let mut result = Model::new();
24    let mut atom = String::new();
25    let mut frame = Frame::default();
26    for character in text.chars() {
27        if frame.complete() && (character.is_whitespace() || (comma_separated && character == ','))
28        {
29            if !atom.is_empty() {
30                result.push(std::mem::take(&mut atom));
31            }
32        } else {
33            atom.push(character);
34        }
35        frame.advance(character)?;
36    }
37    if !frame.complete() {
38        return Err(invalid(
39            Issue::MalformedField,
40            "unterminated quoted atom or tuple",
41        ));
42    }
43    if !atom.is_empty() {
44        result.push(atom);
45    }
46    result.sort_unstable();
47    Ok(result)
48}

50/// Read one bounded display frame before interpreting later lines as metadata.
51/// Every character advances the balance once, then the completed frame is
52/// tokenized once. Multiline quoted symbols therefore remain linear in bytes.
53pub(super) fn model(lines: &mut std::str::Split<'_, char>) -> Result<Model, Error> {
54    let mut text = String::new();
55    let mut frame = Frame::default();
56    loop {
57        let line = lines
58            .next()
59            .ok_or_else(|| invalid(Issue::MissingField, "missing or unterminated native model"))?;
60        if !text.is_empty() {
61            text.push('\n');
62            frame.advance('\n')?;
63        }
64        for character in line.chars() {
65            frame.advance(character)?;
66        }
67        text.push_str(line);
68        if frame.complete() {
69            return split_atoms(&text, false);
70        }
71    }
72}

74#[derive(Default)]
75struct Frame {
76    depth: usize,
77    quoted: bool,
78    escaped: bool,
79}
80
81impl Frame {
82    const fn complete(&self) -> bool {
83        !self.quoted && self.depth == 0
84    }
85
86    fn advance(&mut self, character: char) -> Result<(), Error> {
87        if self.quoted {
88            if self.escaped {
89                self.escaped = false;
90            } else if character == '\\' {
91                self.escaped = true;
92            } else if character == '"' {
93                self.quoted = false;
94            }
95        } else {
96            match character {
97                '"' => self.quoted = true,
98                '(' => {
99                    self.depth = self
100                        .depth
101                        .checked_add(1)
102                        .ok_or_else(|| invalid(Issue::CountOverflow, "atom nesting overflow"))?;
103                }
104                ')' => {
105                    self.depth = self.depth.checked_sub(1).ok_or_else(|| {
106                        invalid(Issue::MalformedField, "unmatched atom parenthesis")
107                    })?;
108                }
109                _ => {}
110            }
111        }
112        Ok(())
113    }
114}
