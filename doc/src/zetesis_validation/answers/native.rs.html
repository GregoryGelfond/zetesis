7pub(super) fn parse(text: &str, optimized: bool, limits: Limits) -> Result<ReportedAnswers, Error> {
8    let mut witnesses: Vec<(Model, Option<Vec<i64>>)> = Vec::new();
9    let mut lines = text.split('\n');
10    let mut metadata = Vec::new();
11    let mut symbol_count = 0usize;
12    while let Some(line) = lines.next() {
13        let line = line.strip_suffix('\r').unwrap_or(line);
14        if let Some(label) = line.strip_prefix("Answer:") {
15            let label = label.trim();
16            if label.is_empty() || !label.bytes().all(|byte| byte.is_ascii_digit()) {
17                return Err(invalid(
18                    Issue::MalformedField,
19                    "invalid native answer identifier",
20                ));
21            }
22            check(
23                Resource::Witnesses,
24                limits.max_witnesses,
25                witnesses.len().saturating_add(1),
26            )?;
27            let model = model(&mut lines)?;
28            symbol_count = symbol_count
29                .checked_add(model.len())
30                .ok_or_else(|| invalid(Issue::CountOverflow, "symbol count overflow"))?;
31            check(Resource::Symbols, limits.max_symbols, symbol_count)?;
32            witnesses.push((model, None));
33        } else if let Some(raw) = line.strip_prefix("Optimization:") {
34            if !optimized {
35                return Err(invalid(
36                    Issue::Contradiction,
37                    "unexpected objective vector in nonoptimized native output",
38                ));
39            }
40            let cost = integers(raw, limits)?;
41            let witness = witnesses.last_mut().ok_or_else(|| {
42                invalid(Issue::MissingField, "cost without preceding native model")
43            })?;
44            if witness.1.replace(cost).is_some() {
45                return Err(invalid(
46                    Issue::Contradiction,
47                    "duplicate native cost record",
48                ));
49            }
50        } else {
51            metadata.push(line);
52        }
53    }
54    let (satisfiable, reported_count) = native_summary(&metadata, optimized)?;
55    if satisfiable == witnesses.is_empty() {
56        return Err(invalid(
57            Issue::Contradiction,
58            "native model/status mismatch",
59        ));
60    }
61    if u64::try_from(witnesses.len())
62        .map_err(|_| invalid(Issue::CountOverflow, "witness count exceeds u64"))?
63        != reported_count
64    {
65        return Err(invalid(
66            Issue::Contradiction,
67            "native answer count disagrees with Models summary",
68        ));
69    }
70    let cost = if optimized && satisfiable {
71        if witnesses.iter().any(|(_, cost)| cost.is_none()) {
72            return Err(invalid(
73                Issue::Contradiction,
74                "native optimized output requires an Optimization: vector for every model",
75            ));
76        }
77        let cost = witnesses[0].1.as_ref();
78        if witnesses.iter().any(|(_, actual)| actual.as_ref() != cost) {
79            return Err(invalid(
80                Issue::Contradiction,
81                "native final optimum contains different objective vectors",
82            ));
83        }
84        cost.cloned()
85    } else {
86        None
87    };
88    let model_multiplicities = multiplicities(witnesses.into_iter().map(|(model, _)| model))?;
89    Ok(ReportedAnswers {
90        satisfiable,
91        cost,
92        model_multiplicities,
93        model_count: reported_count,
94        solver: "zetesis native output".into(),
95    })
96}

98fn native_summary(metadata: &[&str], optimized: bool) -> Result<(bool, u64), Error> {
99    let coverage: Vec<_> = metadata
100        .iter()
101        .copied()
102        .filter(|line| line.starts_with("Coverage:"))
103        .collect();
104    if (!coverage.is_empty() && coverage != ["Coverage: exhausted"])
105        || metadata.iter().any(|line| line.starts_with("INCOMPLETE"))
106    {
107        return Err(invalid(
108            Issue::Incomplete,
109            "native output contains incomplete or conflicting coverage evidence",
110        ));
111    }
112    let statuses: Vec<_> = metadata
113        .iter()
114        .copied()
115        .filter(|line| matches!(*line, "SATISFIABLE" | "UNSATISFIABLE" | "OPTIMUM FOUND"))
116        .collect();
117    if statuses.len() != 1 || (!optimized && statuses[0] == "OPTIMUM FOUND") {
118        return Err(invalid(
119            Issue::Contradiction,
120            "native status is missing, contradictory, or unexpectedly optimized",
121        ));
122    }
123    if optimized && statuses[0] == "SATISFIABLE" {
124        return Err(invalid(
125            Issue::Incomplete,
126            "native optimized output lacks completed optimum evidence",
127        ));
128    }
129    let summaries: Vec<_> = metadata
130        .iter()
131        .copied()
132        .filter_map(|line| line.strip_prefix("Models:"))
133        .collect();
134    if summaries.len() != 1 {
135        return Err(invalid(
136            Issue::Contradiction,
137            "native output requires exactly one Models summary",
138        ));
139    }
140    let summary = summaries[0].trim();
141    let count = if coverage.is_empty() {
142        // Without legacy coverage evidence, only the complete compact spelling
143        // establishes enumeration. Requested limits and interruptions qualify it.
144        if !summary.bytes().all(|byte| byte.is_ascii_digit()) {
145            return Err(invalid(
146                Issue::MalformedField,
147                "invalid complete Models count",
148            ));
149        }
150        summary
151    } else {
152        summary
153            .split(';')
154            .next()
155            .ok_or_else(|| invalid(Issue::MissingField, "missing Models count"))?
156            .trim()
157    };
158    let count = count
159        .parse()
160        .map_err(|_| invalid(Issue::MalformedField, "invalid Models count"))?;
161    Ok((statuses[0] != "UNSATISFIABLE", count))
162}
