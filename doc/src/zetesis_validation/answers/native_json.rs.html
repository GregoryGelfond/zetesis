1//! Checked native full-model records, schema 2 and its schema-1 predecessor,
2//! separate from shown values.
3//!
4//! Success reconciles reported completion, publication, verification, costs and
5//! ties. It does not certify solving correctness or executable/source identity.
6//! The caller must independently require completed process capture and exit 0.
7//! Closed-value shape and native atom construction use the canonical core types;
8//! this decoder does not parse ASP or establish source-name lexical validity.
9//! The ASP spelling view separately validates every predicate, symbol and function
10//! name through the pinned themelios identifier lexer. Serde JSON object decoding
11//! retains the last occurrence of a duplicate key; duplicates are not rejected.
12//! The result checks that decoded document, not the uniqueness of its key spelling.
13

22/// Inclusive native-report acceptance and per-value construction ceilings.
23/// Initial JSON and temporary decoded node/text allocations are bounded by report
24/// input bytes and node ceilings. `value` applies the core constructor's acceptance
25/// and scratch accounting after those owned inputs exist; it is not a pre-allocation
26/// ceiling for the entire decoder. None of these measures is allocator RSS.
27#[derive(Clone, Copy, Debug)]
28pub struct Limits {
29    /// Serialized input, raw records, shown symbols, and cost dimensions.
30    pub report: AnswerLimits,
31    /// Full atom occurrences across all native model records.
32    pub max_atoms: usize,
33    /// Combined preorder nodes across full-model occurrences and shown terms.
34    /// Schema 2 charges reused atoms at every occurrence. Its spelled atom
35    /// table is independently bounded by this ceiling, including unused entries.
36    pub max_value_nodes: usize,
37    /// Core construction/scratch ceilings for each closed value.
38    pub value: ValueLimits,
39}
40impl Default for Limits {
41    fn default() -> Self {
42        Self {
43            report: AnswerLimits::default(),
44            max_atoms: 262_144,
45            max_value_nodes: 1_048_576,
46            value: ValueLimits::default(),
47        }
48    }
49}

51/// One producer record, retaining full identity separately from its shown view.
52#[derive(Debug)]
53pub struct ModelRecord {
54    atoms: Vec<Atom>,
55    shown_indices: Vec<usize>,
56    shown_terms: Vec<Value>,
57    costs: Option<Vec<(i32, i64)>>,
58}
59impl ModelRecord {
60    /// Full atom sequence as recorded; each atom occurs at most once.
61    #[must_use]
62    pub fn full_model(&self) -> &[Atom] {
63        &self.atoms
64    }
65    /// Strictly increasing selected positions in the recorded full atom sequence.
66    #[must_use]
67    pub fn shown_atom_indices(&self) -> &[usize] {
68        &self.shown_indices
69    }
70    /// Additional closed shown values; occurrences are preserved.
71    #[must_use]
72    pub fn shown_terms(&self) -> &[Value] {
73        &self.shown_terms
74    }
75    /// Reported costs in distinct descending priority slots, if present.
76    /// These are producer evidence, not a constructed objective evaluation.
77    #[must_use]
78    pub fn costs(&self) -> Option<&[(i32, i64)]> {
79        self.costs.as_deref()
80    }
81}

83/// Checked native claims with complete publication of the selected result set.
84#[derive(Debug)]
85pub struct NativeAnswers {
86    records: Vec<ModelRecord>,
87    verified_models: u64,
88    checked: u64,
89    costs: Option<Vec<(i32, i64)>>,
90}
91impl NativeAnswers {
92    /// Records in producer publication order; duplicate model records remain.
93    #[must_use]
94    pub fn records(&self) -> &[ModelRecord] {
95        &self.records
96    }
97    /// Reported verified models, including discarded optimization incumbents.
98    #[must_use]
99    pub const fn verified_models(&self) -> u64 {
100        self.verified_models
101    }
102    /// Schema-1 `checked` candidate-progress count, independently of publication.
103    /// It does not establish that every proposed candidate completed evaluation.
104    #[must_use]
105    pub const fn checked(&self) -> u64 {
106        self.checked
107    }
108    /// Final optimum vector when optimization was reported established.
109    #[must_use]
110    pub fn costs(&self) -> Option<&[(i32, i64)]> {
111        self.costs.as_deref()
112    }
113    /// Whether the completed producer report contains satisfying models.
114    #[must_use]
115    pub fn satisfiable(&self) -> bool {
116        !self.records.is_empty()
117    }
118    /// Canonical selected displays, preserving symbol and model multiplicities.
119    /// Full hidden interpretations are deliberately not part of this view.
120    /// `max_bytes` bounds the combined produced symbol spelling bytes; typed
121    /// records were already checked by [`parse`].
122    ///
123    /// # Errors
124    /// Refuses invalid identifier spellings, allocation failure or the spelling
125    /// byte ceiling before retaining each new symbol.
126    pub fn reported_displays(&self, max_bytes: usize) -> Result<super::ReportedAnswers, Error> {
127        let mut models = Vec::new();
128        models
129            .try_reserve_exact(self.records.len())
130            .map_err(|_| Error::Allocation)?;
131        let mut used = 0usize;
132        for record in &self.records {
133            let mut symbols = Vec::new();
134            symbols
135                .try_reserve_exact(record.shown_indices.len() + record.shown_terms.len())
136                .map_err(|_| Error::Allocation)?;
137            for &index in &record.shown_indices {
138                let atom = &record.atoms[index];
139                let bytes = spelling::atom_bytes(atom)?;
140                admit_spelling(&mut used, bytes, max_bytes)?;
141                symbols.push(spelling::atom(atom, bytes)?);
142            }
143            for value in &record.shown_terms {
144                let bytes = spelling::value_bytes(value)?;
145                admit_spelling(&mut used, bytes, max_bytes)?;
146                symbols.push(spelling::value(value, bytes)?);
147            }
148            symbols.sort_unstable();
149            models.push(symbols);
150        }
151        Ok(super::ReportedAnswers {
152            satisfiable: self.satisfiable(),
153            cost: self
154                .costs
155                .as_ref()
156                .map(|costs| costs.iter().map(|(_, value)| *value).collect()),
157            model_multiplicities: super::multiplicities(models.into_iter())?,
158            model_count: u64::try_from(self.records.len())
159                .map_err(|_| invalid(Issue::CountOverflow, "native display count"))?,
160            solver: "zetesis".into(),
161        })
162    }
163
164    /// Canonical ASP spellings of each full atom set, preserving model multiplicity.
165    ///
166    /// This is a consumer view of typed atoms, not a parser or an inference from
167    /// shown values. Atoms and records are sorted for comparison; equal records
168    /// remain repeated. `max_bytes` counts the combined atom spelling bytes.
169    /// Source identifiers are checked after each atom's byte preflight. The pinned
170    /// validator temporarily copies at most twice that atom's longest name bytes,
171    /// separately from retained spelling bytes and existing typed values.
172    ///
173    /// # Errors
174    /// Refuses spelling-byte overflow or the inclusive ceiling before retention,
175    /// invalid source identifiers, or fallible owned-buffer reservation.
176    pub fn full_model_symbols(&self, max_bytes: usize) -> Result<Vec<Vec<String>>, Error> {
177        let mut result = Vec::new();
178        let mut used = 0usize;
179        result
180            .try_reserve_exact(self.records.len())
181            .map_err(|_| Error::Allocation)?;
182        for record in &self.records {
183            let mut symbols = Vec::new();
184            symbols
185                .try_reserve_exact(record.atoms.len())
186                .map_err(|_| Error::Allocation)?;
187            for atom in &record.atoms {
188                let size = spelling::atom_bytes(atom)?;
189                used = used.checked_add(size).ok_or_else(|| {
190                    invalid(Issue::CountOverflow, "full-model spelling byte count")
191                })?;
192                check(Resource::SpellingBytes, max_bytes, used)?;
193                symbols.push(spelling::atom(atom, size)?);
194            }
195            symbols.sort_unstable();
196            result.push(symbols);
197        }
198        result.sort_unstable();
199        Ok(result)
200    }
201}

203/// Whether `document` is a native JSON document the decoder reads: schema 2,
204/// or schema 1 as the executables before it wrote it, in the `zetesis` format.
205pub(crate) fn is_native_document(document: &serde_json::Value) -> bool {
206    matches!(document["schema"].as_u64(), Some(1 | 2)) && document["format"] == "zetesis"
207}

209/// Decode native JSON schema 2, or schema 1 as the executables before it wrote
210/// it, with exhausted coverage and a successful outcome; the older form is
211/// read so that an earlier executable can stand as a baseline.
212/// Every full and shown value is validated, even when a consumer uses only full
213/// atoms. Failed/incomplete envelopes cannot qualify through retained optimum data.
214///
215/// # Errors
216/// Refuses input/value limits, unsupported schema, partial or contradictory
217/// outcomes, malformed typed values, duplicate atoms, invalid shown indices,
218/// or inconsistent counts, objective priorities, costs and final ties.
219pub fn parse(bytes: &[u8], limits: Limits) -> Result<NativeAnswers, Error> {
220    decode::parse(super::text(bytes, limits.report)?, limits)
221}

223fn admit_spelling(used: &mut usize, bytes: usize, maximum: usize) -> Result<(), Error> {
224    let attempted = used
225        .checked_add(bytes)
226        .ok_or_else(|| invalid(Issue::CountOverflow, "display spelling bytes"))?;
227    check(Resource::SpellingBytes, maximum, attempted)?;
228    *used = attempted;
229    Ok(())
230}
