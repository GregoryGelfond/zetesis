14/// Conformance disposition; failure never establishes an empty answer family.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
16#[serde(rename_all = "snake_case")]
17pub enum Decision {
18    /// Full families, known contracts and mandatory execution evidence agree.
19    Passed,
20    /// Spawn, capture, exit or cleanup could not establish completion.
21    ProcessFailure,
22    /// The process deadline expired.
23    Timeout,
24    /// The caller cancelled before or during this check.
25    Cancelled,
26    /// The process output ceiling was reached.
27    CaptureLimit,
28    /// Search or publication was incomplete.
29    Incomplete,
30    /// The structured answer report was malformed or contradictory, or the
31    /// executable its statistics name was refused: none (`null`, or a value
32    /// that is not a path), a third one, beside the invoked executable and the
33    /// one an earlier run named, or one that could not be sealed.
34    InvalidReport,
35    /// The completed full family or objective vector violated its known contract.
36    FamilyMismatch,
37    /// Actual backend, procedure or work evidence was absent or contradictory.
38    InvalidExecution,
39    /// The CPU reference did not pass, so the selected route was not launched.
40    ReferenceFailed,
41}
42impl Decision {
43    /// Stable human/structured status spelling.
44    #[must_use]
45    pub const fn label(self) -> &'static str {
46        match self {
47            Self::Passed => "pass",
48            Self::ProcessFailure => "process_failure",
49            Self::Timeout => "timeout",
50            Self::Cancelled => "cancelled",
51            Self::CaptureLimit => "capture_limit",
52            Self::Incomplete => "incomplete",
53            Self::InvalidReport => "invalid_report",
54            Self::FamilyMismatch => "family_mismatch",
55            Self::InvalidExecution => "invalid_execution",
56            Self::ReferenceFailed => "reference_failed",
57        }
58    }
59}

61/// One launched or refused child, with raw evidence separated from its decision.
62#[derive(Debug)]
63pub struct Attempt {
64    pub(super) arguments: Vec<String>,
65    pub(super) capture: Option<process::Capture>,
66    pub(super) decision: Decision,
67    pub(super) detail: Option<String>,
68    pub(super) family: Option<Family>,
69    pub(super) observation: Option<Observation>,
70    pub(super) cleanup: Option<Cleanup>,
71    /// What the run's report names as its executable: present exactly when
72    /// the run passed (its report validated and its child's cleanup
73    /// completed), and kept when the settlement then refuses the run, whose
74    /// refusal's detail also names it.
75    pub(super) executed: Option<selected::identity::Named>,
76}
77impl Attempt {
78    /// Checked disposition of this child.
79    #[must_use]
80    pub const fn decision(&self) -> Decision {
81        self.decision
82    }
83    /// Retained process evidence, absent when spawn was refused.
84    #[must_use]
85    pub const fn capture(&self) -> Option<&process::Capture> {
86        self.capture.as_ref()
87    }
88    /// Reconciled mandatory route and work evidence.
89    #[must_use]
90    pub const fn observation(&self) -> Option<&Observation> {
91        self.observation.as_ref()
92    }
93    /// Human explanation of a nonpass.
94    #[must_use]
95    pub fn detail(&self) -> Option<&str> {
96        self.detail.as_deref()
97    }
98    /// The executable this run's report names as having done its work,
99    /// resolved through links. Present exactly when the run's decision is
100    /// [`Decision::Passed`], or [`Decision::InvalidReport`] because of the
101    /// executable it named (a third one, beside the invoked executable and the
102    /// one an earlier run named, or one that could not be sealed), and in
103    /// either case its report named a path; so read it with
104    /// [`Self::decision`].
105    #[must_use]
106    pub fn executed(&self) -> Option<&Path> {
107        match &self.executed {
108            Some(selected::identity::Named::Path(path)) => Some(path),
109            _ => None,
110        }
111    }
112}

153/// Complete small-check report; unrun cases remain visible through expected count.
154#[derive(Debug, Serialize)]
155pub struct Report {
156    pub(super) schema: u8,
157    pub(super) format: &'static str,
158    pub(super) scope: &'static str,
159    pub(super) backend: selected::Backend,
160    pub(super) expected_cases: usize,
161    pub(super) executable: selected::FileSeal,
162    pub(super) executable_after: selected::Change,
163    /// The executable runs named as doing their work in place of the invoked
164    /// one, sealed at its first report, and its recheck when the report closed.
165    #[serde(skip_serializing_if = "Option::is_none")]
166    pub(super) executed: Option<selected::FileSeal>,
167    #[serde(skip_serializing_if = "Option::is_none")]
168    pub(super) executed_after: Option<selected::Change>,
169    pub(super) cases: Vec<CaseResult>,
170    pub(super) input_cleanup_failure: Option<String>,
171}
172impl Report {
173    /// Number of independently required fixture contracts.
174    #[must_use]
175    pub const fn required_cases(&self) -> usize {
176        self.expected_cases
177    }
178    /// All expected checks, the primary-executable recheck and the recheck of
179    /// any executable the runs named in its place passed.
180    #[must_use]
181    pub fn passed(&self) -> bool {
182        self.cases.len() == self.expected_cases
183            && self.executable_after.unchanged()
184            && self
185                .executed_after
186                .as_ref()
187                .is_none_or(selected::Change::unchanged)
188            && self.input_cleanup_failure.is_none()
189            && self
190                .cases
191                .iter()
192                .all(|case| case.decision == Decision::Passed)
193    }
194    /// The executable runs named as doing their work in place of the invoked
195    /// one, as sealed at its first report: a front's GPU runs name its sibling.
196    #[must_use]
197    pub const fn executed(&self) -> Option<&selected::FileSeal> {
198        self.executed.as_ref()
199    }
200    /// Cases in fixed order; unresolved cleanup can leave a shorter prefix.
201    #[must_use]
202    pub fn cases(&self) -> &[CaseResult] {
203        &self.cases
204    }
205    /// Render all evidence and explicit final acceptance without performing I/O.
206    ///
207    /// # Errors
208    /// Returns a serialization failure for retained evidence.
209    pub fn to_json(&self) -> Result<serde_json::Value, serde_json::Error> {
210        serde_json::to_value(self.json())
211    }
212    /// Borrowed structured view suitable for streaming without a JSON value tree.
213    #[must_use]
214    pub fn json(&self) -> impl Serialize + '_ {
215        #[derive(Serialize)]
216        struct View<'a> {
217            #[serde(flatten)]
218            report: &'a Report,
219            passed: bool,
220        }
221        View {
222            report: self,
223            passed: self.passed(),
224        }
225    }
226}

228impl Serialize for Attempt {
229    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
230        #[derive(Serialize)]
231        struct View<'a> {
232            arguments: &'a [String],
233            capture: Option<CaptureView<'a>>,
234            decision: Decision,
235            detail: Option<&'a str>,
236            family: Option<&'a Family>,
237            observation: Option<&'a Observation>,
238            cleanup: Option<&'a Cleanup>,
239            executed: Option<&'a Path>,
240        }
241        View {
242            arguments: &self.arguments,
243            capture: self.capture.as_ref().map(CaptureView::from),
244            decision: self.decision,
245            detail: self.detail.as_deref(),
246            family: self.family.as_ref(),
247            observation: self.observation.as_ref(),
248            cleanup: self.cleanup.as_ref(),
249            executed: self.executed(),
250        }
251        .serialize(serializer)
252    }
253}

255#[derive(Serialize)]
256struct CaptureView<'a> {
257    stop: process::Stop,
258    exit: Option<process::Exit>,
259    elapsed_ns: u128,
260    stdout: &'a [u8],
261    stderr: &'a [u8],
262    failure: Option<String>,
263    cleanup_failure: Option<String>,
264}
265impl<'a> From<&'a process::Capture> for CaptureView<'a> {
266    fn from(capture: &'a process::Capture) -> Self {
267        Self {
268            stop: capture.stop(),
269            exit: capture.exit(),
270            elapsed_ns: capture.elapsed().as_nanos(),
271            stdout: capture.stdout(),
272            stderr: capture.stderr(),
273            failure: capture.failure().map(ToString::to_string),
274            cleanup_failure: capture.cleanup_failure().map(ToString::to_string),
275        }
276    }
