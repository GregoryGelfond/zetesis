1//! Complete comparison of the pinned 94-case non-clingcon corpus.
2//!
3//! Loading verifies the selected source view before invoking trusted solvers.
4//! Reference and native complete displayed-model multisets, costs and source
5//! contracts are checked independently. Hidden atoms are not reconstructed.
6//! Requests for physical formula execution additionally check retained route
7//! telemetry; a requested backend alone never establishes physical execution.
8//!
9//! Execution is sequential. The clean source view uses the defaults of
10//! [`crate::examples::Limits`]. The original-source view caps its manifest at
11//! 4 MiB and each source at 1 MiB. Each child has a timeout and combined output
12//! allowance. At most
13//! 188 child captures are retained, so raw capture space is O(94 * allowance),
14//! plus parsed answers and JSON rendering. There is no campaign-wide deadline.
15//! Executable selection preserves per-invocation PATH lookup for bare names;
16//! absolute paths bypass lookup. Environment variables are inherited.
17//!
18//! Linux/macOS use the shared bounded process-group capture. Other platforms
19//! retain the command's direct-child adapter, which has weaker cleanup behavior
20//! and a smaller capture report. Neither path proves descendant quiescence.

38/// Failure before a corpus comparison can produce per-source outcomes.
39#[derive(Debug)]
40pub enum Error {
41    /// The per-child timeout or output allowance is zero.
42    InvalidLimits,
43    /// The source view could not establish the pinned corpus contract.
44    Corpus(String),
45}
46impl std::fmt::Display for Error {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        match self {
49            Self::InvalidLimits => f.write_str("timeout and output ceilings must be positive"),
50            Self::Corpus(error) => f.write_str(error),
51        }
52    }
53}
54impl std::error::Error for Error {}

56/// Load and compare the complete fixed corpus, retaining unsuccessful case results.
57///
58/// `on_case` observes each completed case in manifest order and performs no role
59/// in acceptance. It must return promptly and must not panic, mutate corpus
60/// inputs or the execution environment, or interfere with child processes. It runs outside each invocation's
61/// deadline. Presentation failures belong to the caller; they must not be
62/// represented as answer-set decisions. Pass `|_| {}` for silent operation.
63/// The library performs no global stdout/stderr writes or report publication.
64/// Pending cleanup stops further cases and receives one bounded retry; any
65/// unresolved direct-child ownership is explicitly recorded as abandoned.
66///
67/// # Errors
68/// Refuses zero process limits or a source view that fails corpus loading.
69/// Invocation/output failures are retained in [`Report`] instead.
70pub fn run(request: &Request, on_case: impl FnMut(&CaseResult)) -> Result<Report, Error> {
71    run_with_invocation(request, NativeInvocation::Legacy, on_case)
72}

74/// Compare the corpus using an explicitly selected native command protocol.
75///
76/// `Solve` requests structured full-model records and compares their selected
77/// displays with clingo. Hidden full atoms are validated by the native decoder,
78/// but are not inferred for clingo. Process, callback and cleanup contracts are
79/// identical to [`run`]. The legacy entry point retains its original protocol.
80/// Structured decoding keeps the native decoder's default atom/value ceilings;
81/// its input and selected-display spelling are bounded by the captured input
82/// and configured capture allowance, respectively.
83///
84/// # Errors
85/// Refuses invalid process limits or an unverified corpus before execution.
86pub fn run_with_invocation(
87    request: &Request,
88    invocation: NativeInvocation,
89    on_case: impl FnMut(&CaseResult),
90) -> Result<Report, Error> {
91    run_with_cancellation(
92        request,
93        invocation,
94        &std::sync::atomic::AtomicBool::new(false),
95        on_case,
96    )
97}

99/// Compare the corpus with cooperative cancellation of launches and captures.
100///
101/// The caller owns a monotone cancellation flag; this library installs no global
102/// signal handler. Source verification and callbacks are finite operations but
103/// are not preempted. Once cancellation is observed, no later solver starts.
104/// Active captures retain partial output and their normal cleanup evidence;
105/// completed earlier cases remain unchanged. [`Report::cancelled`] distinguishes
106/// the incomplete campaign even when no solver was launched. The platform's
107/// existing cleanup limitations still apply. Compatibility entry points use an
108/// uncancelled flag.
109///
110/// # Errors
111/// Refuses invalid process limits or an unverified corpus before execution.
112pub fn run_with_cancellation(
113    request: &Request,
114    invocation: NativeInvocation,
115    cancelled: &std::sync::atomic::AtomicBool,
116    on_case: impl FnMut(&CaseResult),
117) -> Result<Report, Error> {
118    if request.timeout_ms == 0 || request.max_output_bytes == 0 {
119        return Err(Error::InvalidLimits);
120    }
121    let corpus = corpus::load(request).map_err(Error::Corpus)?;
122    Ok(runner::run_with_cancellation(
123        request, corpus, invocation, cancelled, on_case,
124    ))
125}
