13#[derive(Debug, Serialize)]
14pub(crate) struct Capture {
15    pub(crate) status: CaptureStatus,
16    #[serde(flatten)]
17    pub(in crate::corpus_comparison) exit: ExitEvidence,
18    pub(crate) elapsed_ms: u128,
19    pub(crate) stdout: String,
20    pub(crate) stderr: String,
21    /// Explicitly lossy legacy views never qualify as completed text.
22    pub(crate) lossy_text: bool,
23    pub(crate) stdout_bytes: Vec<u8>,
24    pub(crate) stderr_bytes: Vec<u8>,
25    #[serde(rename = "capture_failure")]
26    pub(crate) failure: Option<String>,
27    pub(crate) cleanup_failure: Option<String>,
28    pub(crate) pending_child_id: Option<u32>,
29    #[serde(skip)]
30    pub(crate) pending: Option<PendingChild>,
31}
32
33impl Capture {
34    pub(crate) const fn cleanup_unresolved(&self) -> bool {
35        self.pending_child_id.is_some()
36    }
37}

57pub(in crate::corpus_comparison) fn invoke_with_cancellation(
58    executable: &Path,
59    arguments: &[OsString],
60    directory: &Path,
61    timeout: Duration,
62    output_limit: usize,
63    cancelled: &AtomicBool,
64) -> Result<Capture, InvocationFailure> {
65    if cancelled.load(Ordering::Relaxed) {
66        return Err(InvocationFailure::Cancelled);
67    }
68    let search_path = std::env::var_os("PATH");
69    let executable = process::resolve_executable(executable, search_path.as_deref())
70        .map_err(|error| error.to_string())?;
71    let directory = std::path::absolute(directory).map_err(|error| error.to_string())?;
72    let outcome = process::invoke_with_cancellation(
73        Invocation {
74            executable: &executable,
75            arguments,
76            directory: &directory,
77        },
78        Limits {
79            timeout,
80            max_output_bytes: output_limit,
81            cleanup_timeout: Duration::from_secs(1),
82        },
83        cancelled,
84    )
85    .map_err(|error| match error {
86        process::StartError::Cancelled => InvocationFailure::Cancelled,
87        error => InvocationFailure::Other(format!("{}: {error}", executable.display())),
88    })?;
89    let (capture, pending) = outcome.into_parts();
90    let utf8 = capture.stdout_text().and_then(|_| capture.stderr_text());
91    let lossy_text = utf8.is_err();
92    let status = match capture.stop() {
93        Stop::Completed if lossy_text => CaptureStatus::InvalidUtf8,
94        Stop::Completed => CaptureStatus::Completed,
95        Stop::Cancelled => CaptureStatus::Cancelled,
96        Stop::Deadline => CaptureStatus::Timeout,
97        Stop::OutputLimit => CaptureStatus::OutputLimit,
98        Stop::Failure => CaptureStatus::CaptureFailure,
99    };
100    Ok(Capture {
101        status,
102        exit: ExitEvidence(capture.exit()),
103        elapsed_ms: capture.elapsed().as_millis(),
104        stdout: String::from_utf8_lossy(capture.stdout()).into_owned(),
105        stderr: String::from_utf8_lossy(capture.stderr()).into_owned(),
106        lossy_text,
107        stdout_bytes: if lossy_text {
108            capture.stdout().to_vec()
109        } else {
110            Vec::new()
111        },
112        stderr_bytes: if lossy_text {
113            capture.stderr().to_vec()
114        } else {
115            Vec::new()
116        },
117        failure: capture.failure().map(ToString::to_string),
118        cleanup_failure: capture.cleanup_failure().map(ToString::to_string),
119        pending_child_id: pending.as_ref().map(PendingChild::id),
120        pending,
121    })
122}
