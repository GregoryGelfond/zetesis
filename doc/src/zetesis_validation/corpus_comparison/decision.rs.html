1//! Typed per-case decisions and retained capture classifications.

4/// Solver whose invocation or output is being checked.
5#[derive(Clone, Copy, Debug, PartialEq, Eq)]
6pub enum Producer {
7    /// Independent clingo reference.
8    Reference,
9    /// Native zetesis implementation.
10    Native,
11}

13/// Why a process capture cannot establish a completed solver result.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum CaptureFailure {
16    /// Caller cancellation stopped an active child capture.
17    Cancelled,
18    /// The invocation deadline was reached.
19    Timeout,
20    /// The combined output allowance was reached.
21    OutputLimit,
22    /// An OS or capture operation failed.
23    Capture,
24    /// Completed output was not UTF-8.
25    InvalidUtf8,
26}
27
28#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
29#[serde(rename_all = "snake_case")]
30pub(super) enum CaptureStatus {
31    Completed,
32    Cancelled,
33    Timeout,
34    OutputLimit,
35    #[serde(rename = "capture_failure")]
36    CaptureFailure,
37    #[cfg(any(target_os = "linux", target_os = "macos"))]
38    InvalidUtf8,
39}
40impl CaptureStatus {
41    pub(super) const fn failure(self) -> Option<CaptureFailure> {
42        match self {
43            Self::Completed => None,
44            Self::Cancelled => Some(CaptureFailure::Cancelled),
45            Self::Timeout => Some(CaptureFailure::Timeout),
46            Self::OutputLimit => Some(CaptureFailure::OutputLimit),
47            Self::CaptureFailure => Some(CaptureFailure::Capture),
48            #[cfg(any(target_os = "linux", target_os = "macos"))]
49            Self::InvalidUtf8 => Some(CaptureFailure::InvalidUtf8),
50        }
51    }
52}

54/// Result of checking one source. A reference pass does not establish native support.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub enum Decision {
57    /// Native and reference completed answers and required execution agree.
58    Passed,
59    /// Only the reference was requested and its contracts hold.
60    ReferencePassed,
61    /// Caller cancellation prevented this solver invocation from starting.
62    Cancelled(Producer),
63    /// A solver could not be invoked.
64    InvocationFailed(Producer, String),
65    /// Solver output capture did not complete under the requested contract.
66    CaptureFailed(Producer, CaptureFailure),
67    /// The reference returned an unexpected exit code.
68    ReferenceError,
69    /// The reference output failed its normalization contract.
70    ReferenceOutputError(String),
71    /// The reference failed the source's independent expectations.
72    ReferenceContractMismatch(String),
73    /// Native execution or output did not establish completed enumeration or optimality.
74    NativeIncomplete,
75    /// Native source admission explicitly refused the input.
76    NativeSourceRefused,
77    /// The native invocation returned another failure exit.
78    NativeError,
79    /// Native output failed its normalization contract.
80    NativeOutputUnsupported(String),
81    /// The completed answer families disagree.
82    Mismatch,
83    /// Native answers failed the source's independent expectations.
84    NativeContractMismatch(String),
85    /// Matching answers lacked the required physical-route evidence.
86    NativeExecutionUnqualified(String),
87}
88impl Decision {
89    /// Stable schema-1 status spelling, also used in command progress output.
90    #[must_use]
91    pub const fn label(&self) -> &'static str {
92        match self {
93            Self::Passed => "pass",
94            Self::ReferencePassed => "reference_pass",
95            Self::Cancelled(Producer::Reference)
96            | Self::CaptureFailed(Producer::Reference, CaptureFailure::Cancelled) => {
97                "reference_cancelled"
98            }
99            Self::Cancelled(Producer::Native)
100            | Self::CaptureFailed(Producer::Native, CaptureFailure::Cancelled) => {
101                "native_cancelled"
102            }
103            Self::InvocationFailed(Producer::Reference, _) => "reference_invocation_error",
104            Self::InvocationFailed(Producer::Native, _) => "native_invocation_error",
105            Self::CaptureFailed(Producer::Reference, CaptureFailure::Timeout) => {
106                "reference_timeout"
107            }
108            Self::CaptureFailed(Producer::Reference, CaptureFailure::OutputLimit) => {
109                "reference_output_limit"
110            }
111            Self::CaptureFailed(Producer::Reference, CaptureFailure::Capture) => {
112                "reference_capture_failure"
113            }
114            Self::CaptureFailed(Producer::Reference, CaptureFailure::InvalidUtf8) => {
115                "reference_invalid_utf8"
116            }
117            Self::CaptureFailed(Producer::Native, CaptureFailure::Timeout) => "native_timeout",
118            Self::CaptureFailed(Producer::Native, CaptureFailure::OutputLimit) => {
119                "native_output_limit"
120            }
121            Self::CaptureFailed(Producer::Native, CaptureFailure::Capture) => {
122                "native_capture_failure"
123            }
124            Self::CaptureFailed(Producer::Native, CaptureFailure::InvalidUtf8) => {
125                "native_invalid_utf8"
126            }
127            Self::ReferenceError => "reference_error",
128            Self::ReferenceOutputError(_) => "reference_output_error",
129            Self::ReferenceContractMismatch(_) => "reference_contract_mismatch",
130            Self::NativeIncomplete => "native_incomplete",
131            Self::NativeSourceRefused => "native_source_refused",
132            Self::NativeError => "native_error",
133            Self::NativeOutputUnsupported(_) => "native_output_unsupported",
134            Self::Mismatch => "mismatch",
135            Self::NativeContractMismatch(_) => "native_contract_mismatch",
136            Self::NativeExecutionUnqualified(_) => "native_execution_unqualified",
137        }
138    }
139    /// Diagnostic explanation, absent for the two successful decisions.
140    #[must_use]
141    pub fn detail(&self) -> Option<&str> {
142        match self {
143            Self::Passed | Self::ReferencePassed => None,
144            Self::Cancelled(_) => Some("caller cancellation prevented solver invocation"),
145            Self::InvocationFailed(_, error)
146            | Self::ReferenceOutputError(error)
147            | Self::ReferenceContractMismatch(error)
148            | Self::NativeOutputUnsupported(error)
149            | Self::NativeContractMismatch(error)
150            | Self::NativeExecutionUnqualified(error) => Some(error),
151            Self::CaptureFailed(Producer::Reference, _) => {
152                Some("reference invocation did not complete")
153            }
154            Self::CaptureFailed(Producer::Native, _) => Some("native invocation did not complete"),
155            Self::ReferenceError => Some("reference returned an unexpected exit code"),
156            Self::NativeIncomplete => {
157                Some("native solver did not establish completed enumeration or optimality")
158            }
159            Self::NativeSourceRefused | Self::NativeError => {
160                Some("native solver did not return a completed result")
161            }
162            Self::Mismatch => Some("native and reference completed results differ"),
163        }
164    }
165}

167/// Start refusal has no child capture; preserve cancellation as typed evidence.
168#[derive(Debug)]
169pub(super) enum InvocationFailure {
170    Cancelled,
171    Other(String),
172}
173
174impl From<String> for InvocationFailure {
175    fn from(value: String) -> Self {
176        Self::Other(value)
177    }
178}
179
180impl From<&str> for InvocationFailure {
181    fn from(value: &str) -> Self {
182        Self::Other(value.to_owned())
183    }
