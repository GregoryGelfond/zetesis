15/// Retained result for one independently checked source.
16/// Fields are immutable; callers inspect decisions without parsing report JSON.
17#[derive(Debug)]
18pub struct CaseResult {
19    pub(super) evidence: CaseEvidence,
20    pub(super) decision: Decision,
21}

23// Partial observations do not carry a provisional semantic decision.
24#[derive(Debug)]
25pub(super) struct CaseEvidence {
26    pub(super) source: corpus::Case,
27    pub(super) reference_process: Option<capture::Capture>,
28    pub(super) native_process: Option<capture::Capture>,
29    pub(super) reference_answer: Option<normalize::Answer>,
30    pub(super) native_answer: Option<normalize::Answer>,
31    pub(super) native_arguments: Option<Vec<OsString>>,
32    pub(super) phase: Observation<phase::PhaseTimings>,
33    pub(super) stage: Observation<stage::StageTimings>,
34    pub(super) execution: Option<execution::FormulaExecution>,
35    pub(super) answer_parity: bool,
36}
37impl CaseEvidence {
38    pub(super) fn new(source: &corpus::Case) -> Self {
39        Self {
40            source: source.clone(),
41            reference_process: None,
42            native_process: None,
43            reference_answer: None,
44            native_answer: None,
45            native_arguments: None,
46            phase: Observation::Absent,
47            stage: Observation::Absent,
48            execution: None,
49            answer_parity: false,
50        }
51    }
52}
53impl CaseResult {
54    /// Original corpus-relative source path.
55    #[must_use]
56    pub fn path(&self) -> &str {
57        &self.evidence.source.path
58    }
59    /// Source identity used by the corpus loader.
60    #[must_use]
61    pub fn source_sha256(&self) -> &str {
62        &self.evidence.source.sha256
63    }
64    /// Checked per-source outcome, independent of its rendered spelling.
65    #[must_use]
66    pub const fn decision(&self) -> &Decision {
67        &self.decision
68    }
69    /// Whether native and reference answer parity was established, even if a
70    /// subsequent physical-route obligation failed.
71    #[must_use]
72    pub const fn answer_parity_passed(&self) -> bool {
73        self.evidence.answer_parity
74    }
75    /// Completed reference answers, when normalization established them.
76    #[must_use]
77    pub fn reference_answers(&self) -> Option<&crate::answers::ReportedAnswers> {
78        self.evidence
79            .reference_answer
80            .as_ref()
81            .map(|answer| &answer.reported)
82    }
83    /// Completed native answers, when normalization established them.
84    #[must_use]
85    pub fn native_answers(&self) -> Option<&crate::answers::ReportedAnswers> {
86        self.evidence
87            .native_answer
88            .as_ref()
89            .map(|answer| &answer.reported)
90    }
91    /// Captured reference process elapsed milliseconds, not a benchmark sample.
92    #[must_use]
93    pub fn reference_elapsed_ms(&self) -> Option<u128> {
94        self.evidence
95            .reference_process
96            .as_ref()
97            .map(|capture| capture.elapsed_ms)
98    }
99    /// Captured native process elapsed milliseconds, not a benchmark sample.
100    #[must_use]
101    pub fn native_elapsed_ms(&self) -> Option<u128> {
102        self.evidence
103            .native_process
104            .as_ref()
105            .map(|capture| capture.elapsed_ms)
106    }
107
108    /// Observed reference exit, including its terminating signal on Unix.
109    /// Absence means no reaped exit was retained; it is not an exit code.
110    #[must_use]
111    pub fn reference_exit(&self) -> Option<crate::process::Exit> {
112        self.evidence
113            .reference_process
114            .as_ref()
115            .and_then(|capture| capture.exit.0)
116    }
117
118    /// Observed native exit, independent of capture stop and semantic outcome.
119    /// A normal exit alone does not establish completed answer enumeration.
120    #[must_use]
121    pub fn native_exit(&self) -> Option<crate::process::Exit> {
122        self.evidence
123            .native_process
124            .as_ref()
125            .and_then(|capture| capture.exit.0)
126    }
127}

137/// Whether the requested physical formula route was established for the full target.
138#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
139#[serde(rename_all = "snake_case")]
140pub enum PhysicalStatus {
141    /// No physical formula qualification was requested.
142    NotRequested,
143    /// Every case passed, but none exercised device membership.
144    NotExercised,
145    /// The complete target passed and device membership was exercised.
146    Qualified,
147    /// At least one requested condition remains unestablished.
148    Unqualified,
149}

151/// Owned comparison results and source identities. Constructed only by execution.
152#[derive(Debug)]
153pub struct Report {
154    pub(super) request: Request,
155    pub(super) invocation: super::NativeInvocation,
156    pub(super) required_cases: usize,
157    pub(super) corpus: corpus::Loaded,
158    pub(super) cases: Vec<CaseResult>,
159    pub(super) cleanup: Vec<Cleanup>,
160    pub(super) cancelled: bool,
161}
162impl Report {
163    /// Caller cancellation prevented a required launch or stopped an active
164    /// capture. Completed earlier cases remain valid; the requested corpus
165    /// comparison is incomplete. A late flag after all cases does not erase
166    /// already completed evidence.
167    #[must_use]
168    pub const fn cancelled(&self) -> bool {
169        self.cancelled
170    }
171    /// Number of independently required sources in the pinned target.
172    #[must_use]
173    pub const fn required_cases(&self) -> usize {
174        self.required_cases
175    }
176    /// Results in source-manifest order, stopping on cancellation or unresolved cleanup.
177    #[must_use]
178    pub fn cases(&self) -> &[CaseResult] {
179        &self.cases
180    }
181    /// Whether all 94 cases satisfy the requested mode and its route requirements.
182    #[must_use]
183    pub fn passed(&self) -> bool {
184        !self.cancelled
185            && self.cases.len() == self.required_cases()
186            && self.cases.iter().all(|case| {
187                if self.request.reference_only {
188                    case.decision == Decision::ReferencePassed
189                } else {
190                    case.decision == Decision::Passed
191                }
192            })
193            && (!self.request.physical_formula() || self.gpu_exercised() > 0)
194    }
195    /// Full native answer parity; reference-only execution cannot establish it.
196    #[must_use]
197    pub fn answer_parity_passed(&self) -> bool {
198        !self.cancelled
199            && !self.request.reference_only
200            && self.cases.len() == self.required_cases()
201            && self.cases.iter().all(|case| case.evidence.answer_parity)
202    }
203    pub(super) fn gpu_exercised(&self) -> usize {
204        self.cases
205            .iter()
206            .filter(|case| {
207                case.evidence
208                    .execution
209                    .as_ref()
210                    .is_some_and(|e| e.status == execution::Status::GpuExercised)
211            })
212            .count()
213    }
214    pub(super) fn outer_unsat(&self) -> usize {
215        self.cases
216            .iter()
217            .filter(|case| {
218                case.evidence
219                    .execution
220                    .as_ref()
221                    .is_some_and(|e| e.status == execution::Status::OuterUnsatWithoutMembership)
222            })
223            .count()
224    }
225    /// Requested physical-route result, separate from answer parity.
226    #[must_use]
227    pub fn physical_status(&self) -> PhysicalStatus {
228        if !self.request.physical_formula() {
229            PhysicalStatus::NotRequested
230        } else if !self.cancelled
231            && self.cases.len() == self.required_cases()
232            && self
233                .cases
234                .iter()
235                .all(|case| case.decision == Decision::Passed)
236            && self.gpu_exercised() == 0
237        {
238            PhysicalStatus::NotExercised
239        } else if self.passed() {
240            PhysicalStatus::Qualified
241        } else {
242            PhysicalStatus::Unqualified
243        }
244    }
