4//! files. Preserved assertion excerpts independently decode to those source
5//! bytes and helper expectations. Verification never reads C++ files or runs a solver.

28/// Inclusive serialized-input and retained-source ceilings. Zero means zero.
29/// JSON allocations are bounded by the manifest byte ceiling; these measures
30/// do not claim allocator overhead or RSS. Assertion decoding uses temporary
31/// storage linear in each excerpt, bounded by the manifest byte ceiling.
32#[derive(Clone, Copy, Debug)]
33pub struct Limits {
34    /// Maximum serialized manifest bytes, including assertion excerpts.
35    pub manifest_bytes: usize,
36    /// Maximum bytes in one decoded source file.
37    pub source_bytes: usize,
38    /// Maximum combined bytes of retained decoded source files.
39    pub total_source_bytes: usize,
40    /// Maximum bytes in the retained MIT license.
41    pub license_bytes: usize,
42    /// Maximum cases before processing their source files.
43    pub cases: usize,
44    /// Maximum recorded full-model occurrences across the corpus.
45    pub models: usize,
46    /// Maximum atom occurrences across all recorded full models.
47    pub atoms: usize,
48}
49impl Default for Limits {
50    fn default() -> Self {
51        Self {
52            manifest_bytes: 1_048_576,
53            source_bytes: 65_536,
54            total_source_bytes: 1_048_576,
55            license_bytes: 1_048_576,
56            cases: 24,
57            models: 4_096,
58            atoms: 65_536,
59        }
60    }
61}

63/// A precisely scoped corpus integrity resource.
64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub enum Resource {
66    /// Serialized manifest bytes.
67    ManifestBytes,
68    /// Bytes of one decoded source.
69    SourceBytes,
70    /// Combined retained source bytes.
71    TotalSourceBytes,
72    /// Bytes of the retained license.
73    LicenseBytes,

78    /// Atom occurrences in full-model records.

82/// A refusal to establish integrity, never a semantic solver verdict.
83#[derive(Debug)]
84pub enum Error {
85    /// Filesystem operation failed at the named path.
86    Io {
87        /// Relevant path.
88        path: PathBuf,
89        /// Underlying filesystem failure.
90        source: std::io::Error,
91    },
92    /// Serialized data is outside the declared schema.
93    Json(serde_json::Error),
94    /// An inclusive resource ceiling was exceeded.
95    Limit {
96        /// Measured resource.
97        resource: Resource,
98        /// Requested amount.
99        observed: u128,
100        /// Inclusive allowance.
101        limit: usize,
102    },
103    /// Source, license, or manifest bytes differ from their seal.
104    Digest {
105        /// Named input.
106        path: String,
107        /// Required SHA-256.
108        expected: String,
109        /// Observed SHA-256.
110        actual: String,
111    },
112    /// A relative path is not confined to the chosen root.
113    Path(String),
114    /// A case, provenance, or model contract is inconsistent.
115    Contract(String),
116    /// A preserved assertion uses spelling outside the declared literal subset.
117    Literal(String),
118}
119impl fmt::Display for Error {
120    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
121        match self {
122            Self::Io { path, source } => write!(formatter, "{}: {source}", path.display()),
123            Self::Json(error) => write!(formatter, "corpus document: {error}"),
124            Self::Limit {
125                resource,
126                observed,
127                limit,
128            } => write!(formatter, "corpus {resource:?} {observed} exceeds {limit}"),
129            Self::Digest {
130                path,
131                expected,
132                actual,
133            } => write!(
134                formatter,
135                "{path}: SHA-256 {actual} differs from {expected}"
136            ),
137            Self::Path(path) => write!(formatter, "corpus path is not confined: {path}"),
138            Self::Contract(reason) => write!(formatter, "corpus contract: {reason}"),
139            Self::Literal(reason) => {
140                write!(formatter, "unsupported C++ literal contract: {reason}")
141            }
142        }
143    }
144}
145impl std::error::Error for Error {
146    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
147        match self {
148            Self::Io { source, .. } => Some(source),
149            Self::Json(error) => Some(error),
150            _ => None,
151        }
152    }
153}
154impl From<serde_json::Error> for Error {
155    fn from(error: serde_json::Error) -> Self {
156        Self::Json(error)
157    }
158}
159pub(super) fn ceiling(resource: Resource, observed: u128, limit: usize) -> Result<(), Error> {
160    if observed > limit as u128 {
161        Err(Error::Limit {
162            resource,
163            observed,
164            limit,
165        })
166    } else {
167        Ok(())
168    }
169}

171/// Verify the pinned curated target and retain its exact UTF-8 source bytes.
172/// Returned paths describe verified inputs; later filesystem changes do not
173/// change the retained sources. No original C++ files or subprocesses are read.
174/// Original-file spans are recorded metadata, not freshly authenticated by this
175/// operation. Retained assertion excerpts are decoded and checked against sources.
176///
177/// # Errors
178/// Refuses changed seals, malformed/duplicate contracts, unconfined paths,
179/// invalid UTF-8, missing files, or exceeded inclusive ceilings.
180pub fn open(root: &Path, limits: Limits) -> Result<Corpus, Error> {
181    let root = files::canonical(root)?;
182    let bytes = files::read(
183        &files::confined(&root, "manifest.json")?,
184        limits.manifest_bytes,
185        Resource::ManifestBytes,
186    )?;
187    files::digest("manifest.json", &bytes, MANIFEST_SHA256)?;
188    let document: document::Document = serde_json::from_slice(&bytes)?;
189    document::load(root, document, limits)
190}
