1//! Restricted decoding of preserved assertion excerpts, not C++ source files.
2//!
3//! Only the pinned REQUIRE/solve literal forms and helper output are accepted.
4//! Every cursor advances within its manifest-bounded excerpt; balanced delimiters
5//! use an explicit stack. Decoding retains at most linear excerpt-sized storage.
6
7use super::Error;
8
9pub(super) fn trivia(text: &str, mut offset: usize) -> Result<usize, Error> {
10    let bytes = text.as_bytes();
11    loop {
12        while bytes.get(offset).is_some_and(u8::is_ascii_whitespace) {
13            offset += 1;
14        }
15        if text[offset..].starts_with("//") {
16            offset = text[offset..]
17                .find('\n')
18                .map_or(text.len(), |end| offset + end + 1);
19        } else if text[offset..].starts_with("/*") {
20            offset += 2;
21            offset += text[offset..]
22                .find("*/")
23                .ok_or_else(|| Error::Literal("unterminated comment".into()))?
24                + 2;
25        } else {
26            return Ok(offset);
27        }
28    }
29}
30fn string_end(text: &str, start: usize) -> Result<usize, Error> {
31    let bytes = text.as_bytes();
32    if bytes.get(start) != Some(&b'"') {
33        return Err(Error::Literal("expected ordinary string literal".into()));
34    }
35    let mut offset = start + 1;
36    while let Some(&byte) = bytes.get(offset) {
37        match byte {
38            b'"' => return Ok(offset + 1),
39            b'\\' => {
40                offset += 1;
41                if !matches!(bytes.get(offset), Some(b'\\' | b'"' | b'n' | b'r' | b't')) {
42                    return Err(Error::Literal("escape outside declared subset".into()));
43                }
44            }
45            b'\n' | b'\r' => {
46                return Err(Error::Literal(
47                    "literal line break outside declared subset".into(),
48                ));
49            }
50            _ => {}
51        }
52        offset += 1;
53    }
54    Err(Error::Literal("unterminated string".into()))
55}
56fn literal_prefix(text: &str) -> Result<(String, usize), Error> {
57    let mut offset = trivia(text, 0)?;
58    let mut output = String::new();
59    let mut found = false;
60    while text.as_bytes().get(offset) == Some(&b'"') {
61        let end = string_end(text, offset)?;
62        let value: String = serde_json::from_str(&text[offset..end])?;
63        output.push_str(&value);
64        found = true;
65        offset = trivia(text, end)?;
66    }
67    if !found {
68        return Err(Error::Literal("empty literal sequence".into()));
69    }
70    Ok((output, offset))
71}
72pub(super) fn literals(text: &str) -> Result<String, Error> {
73    let (output, end) = literal_prefix(text)?;
74    if end != text.len() {
75        return Err(Error::Literal("trailing literal expression".into()));
76    }
77    Ok(output)
78}
79/// Balanced delimiters and top-level commas, excluding strings/comments.
80pub(super) fn balanced(text: &str, opening: usize) -> Result<(usize, Vec<usize>), Error> {
81    if !matches!(text.as_bytes().get(opening), Some(b'(' | b'[' | b'{')) {
82        return Err(Error::Literal("expected opening delimiter".into()));
83    }
84    let mut stack = Vec::new();
85    let mut separators = Vec::new();
86    let mut offset = opening;
87    let bytes = text.as_bytes();
88    while offset < bytes.len() {
89        offset = trivia(text, offset)?;
90        let Some(&byte) = bytes.get(offset) else {
91            break;
92        };
93        match byte {
94            b'"' => {
95                offset = string_end(text, offset)?;
96                continue;
97            }
98            b'(' => stack.push(b')'),
99            b'[' => stack.push(b']'),
100            b'{' => stack.push(b'}'),
101            b')' | b']' | b'}' => {
102                if stack.pop() != Some(byte) {
103                    return Err(Error::Literal("mismatched delimiters".into()));
104                }
105                if stack.is_empty() {
106                    return Ok((offset, separators));
107                }
108            }
109            b',' if stack.len() == 1 => separators.push(offset),
110            b'\'' => {
111                return Err(Error::Literal(
112                    "character literal outside declared subset".into(),
113                ));
114            }
115            byte if !byte.is_ascii() => {
116                return Err(Error::Literal("non-ASCII token outside a string".into()));
117            }
118            _ => {}
119        }
120        offset += 1;
121    }
122    Err(Error::Literal("unterminated expression".into()))
123}
124pub(super) fn assertion(text: &str) -> Result<(String, Vec<String>, String), Error> {
125    let text = text
126        .strip_prefix("REQUIRE")
127        .ok_or_else(|| Error::Literal("expected REQUIRE assertion".into()))?;
128    let opening = trivia(text, 0)?;
129    if text.as_bytes().get(opening) != Some(&b'(') {
130        return Err(Error::Literal("expected REQUIRE opening".into()));
131    }
132    let (end, _) = balanced(text, opening)?;
133    if trivia(text, end + 1)? != text.len() {
134        return Err(Error::Literal("trailing assertion text".into()));
135    }
136    let inner = &text[opening + 1..end];
137    let (expected, end) = literal_prefix(inner)?;
138    let right = inner[end..]
139        .strip_prefix("==")
140        .ok_or_else(|| Error::Literal("expected equality assertion".into()))?;
141    let right = right
142        .trim()
143        .strip_prefix("IO::to_string")
144        .ok_or_else(|| Error::Literal("expected IO::to_string".into()))?;
145    let outer = trivia(right, 0)?;
146    let (outer_end, _) = balanced(right, outer)?;
147    if trivia(right, outer_end + 1)? != right.len() {
148        return Err(Error::Literal("trailing helper text".into()));
149    }
150    let call = right[outer + 1..outer_end].trim();
151    let call = call
152        .strip_prefix("solve")
153        .ok_or_else(|| Error::Literal("expected solve helper".into()))?;
154    let opening = trivia(call, 0)?;
155    let (end, separators) = balanced(call, opening)?;
156    if trivia(call, end + 1)? != call.len() {
157        return Err(Error::Literal("trailing solve expression".into()));
158    }
159    let mut edges = vec![opening];
160    edges.extend(separators);
161    edges.push(end);
162    let args: Vec<_> = edges
163        .windows(2)
164        .map(|pair| call[pair[0] + 1..pair[1]].trim().to_owned())
165        .collect();
166    if args.is_empty() || args.len() > 2 {
167        return Err(Error::Literal(
168            "expected source plus optional prefix set".into(),
169        ));
170    }
171    let source = literals(&args[0])?;
172    Ok((source, args[1..].to_vec(), expected))
173}
174pub(super) fn prefixes(arguments: &[String]) -> Result<Vec<String>, Error> {
175    let [text] = arguments else {
176        return if arguments.is_empty() {
177            Ok(vec![String::new()])
178        } else {
179            Err(Error::Literal(
180                "objective-bound helper is outside selected contract".into(),
181            ))
182        };
183    };
184    let text = text.trim();
185    if !text.starts_with('{') {
186        return Err(Error::Literal("expected prefix initializer".into()));
187    }
188    let (end, separators) = balanced(text, 0)?;
189    if end + 1 != text.len() {
190        return Err(Error::Literal("trailing prefix initializer".into()));
191    }
192    if text[1..end].trim().is_empty() {
193        return Ok(Vec::new());
194    }
195    let mut edges = vec![0];
196    edges.extend(separators);
197    edges.push(end);
198    edges
199        .windows(2)
200        .map(|pair| literals(&text[pair[0] + 1..pair[1]]))
201        .collect()
202}
203pub(super) fn helper_models(text: &str) -> Result<Vec<Vec<String>>, Error> {
204    if text.contains('"') || !text.starts_with("([") {
205        return Err(Error::Literal(
206            "helper output outside selected subset".into(),
207        ));
208    }
209    let (end, _) = balanced(text, 1)?;
210    let mut offset = 2;
211    let mut models = Vec::new();
212    while offset < end {
213        if text.as_bytes().get(offset) != Some(&b'[') {
214            return Err(Error::Literal("expected helper model".into()));
215        }
216        let (stop, separators) = balanced(text, offset)?;
217        let mut edges = vec![offset];
218        edges.extend(separators);
219        edges.push(stop);
220        let mut model: Vec<_> = edges
221            .windows(2)
222            .filter_map(|pair| {
223                let atom = text[pair[0] + 1..pair[1]].trim();
224                (!atom.is_empty()).then(|| atom.to_owned())
225            })
226            .collect();
227        model.sort();
228        models.push(model);
229        offset = stop + 1;
230        if offset < end {
231            if text.as_bytes().get(offset) != Some(&b',') {
232                return Err(Error::Literal("expected model separator".into()));
233            }
234            offset += 1;
235        }
236    }
237    if !text[end + 1..].starts_with(',') || !text.ends_with(')') {
238        return Err(Error::Literal(
239            "expected retained helper diagnostics".into(),
240        ));
241    }
242    models.sort();
243    Ok(models)
244}
