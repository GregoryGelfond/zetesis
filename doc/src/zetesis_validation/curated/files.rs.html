1//! Confined, bounded corpus reads and explicit SHA-256 identities.
2
3use super::{Error, Resource, ceiling};
4use sha2::{Digest, Sha256};
5use std::io::Read;
6use std::path::{Component, Path, PathBuf};
7
8pub(super) fn io(path: &Path, source: std::io::Error) -> Error {
9    Error::Io {
10        path: path.to_owned(),
11        source,
12    }
13}
14pub(super) fn canonical(path: &Path) -> Result<PathBuf, Error> {
15    path.canonicalize().map_err(|source| io(path, source))
16}
17pub(super) fn relative(text: &str) -> Result<&Path, Error> {
18    let path = Path::new(text);
19    // Backslashes and drive separators must not acquire platform-specific meaning.
20    if text.is_empty()
21        || text.contains(['\\', ':'])
22        || text
23            .split('/')
24            .any(|part| part.is_empty() || matches!(part, "." | ".."))
25        || path
26            .components()
27            .any(|component| !matches!(component, Component::Normal(_)))
28    {
29        return Err(Error::Path(text.to_owned()));
30    }
31    Ok(path)
32}
33pub(super) fn confined(root: &Path, text: &str) -> Result<PathBuf, Error> {
34    let actual = canonical(&root.join(relative(text)?))?;
35    if !actual.starts_with(root) {
36        return Err(Error::Path(text.to_owned()));
37    }
38    Ok(actual)
39}
40pub(super) fn read(path: &Path, maximum: usize, resource: Resource) -> Result<Vec<u8>, Error> {
41    let file = std::fs::File::open(path).map_err(|source| io(path, source))?;
42    let metadata = file.metadata().map_err(|source| io(path, source))?;
43    if !metadata.is_file() {
44        return Err(Error::Path(path.display().to_string()));
45    }
46    ceiling(resource, u128::from(metadata.len()), maximum)?;
47    let cap = u64::try_from(maximum).unwrap_or(u64::MAX).saturating_add(1);
48    let mut bytes = Vec::new();
49    file.take(cap)
50        .read_to_end(&mut bytes)
51        .map_err(|source| io(path, source))?;
52    ceiling(resource, bytes.len() as u128, maximum)?;
53    Ok(bytes)
54}
55pub(super) fn hash(bytes: &[u8]) -> String {
56    format!("{:x}", Sha256::digest(bytes))
57}
58pub(super) fn digest(path: &str, bytes: &[u8], expected: &str) -> Result<(), Error> {
59    let actual = hash(bytes);
60    if actual != expected {
61        return Err(Error::Digest {
62            path: path.to_owned(),
63            expected: expected.to_owned(),
64            actual,
65        });
66    }
67    Ok(())
