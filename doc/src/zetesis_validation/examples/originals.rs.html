11/// Check every original hash, exact comment deletion, recorded edit, include coordinate and
12/// typed-contract translation against the retained cleaned sources.
13/// This reads the preserved original tree only when explicitly requested.
14/// Only `limits.source_bytes` applies: the supplied `Corpus` has already passed
15/// its manifest, source-count, retained-byte and contract-symbol admission limits.
16///
17/// # Errors
18/// Refuses changed originals, deletion spans outside ordinary leading comments,
19/// changed includes, mismatched contracts, unconfined paths or source-byte limits.
20pub fn verify_originals(
21    corpus: &Corpus,
22    original_root: &Path,
23    limits: Limits,
24) -> Result<(), Error> {
25    let root = files::canonical(original_root)?;
26    let license = files::read(
27        &files::confined(&root, "LICENSE")?,
28        limits.source_bytes,
29        Resource::SourceBytes,
30    )?;
31    files::digest("LICENSE", &license, corpus.license_sha256())?;
32    for source in corpus.files() {
33        let bytes = files::read(
34            &files::confined(&root, source.path())?,
35            limits.source_bytes,
36            Resource::SourceBytes,
37        )?;
38        files::digest(source.path(), &bytes, source.original_sha256())?;
39        let original = String::from_utf8(bytes).map_err(Error::Utf8)?;
40        let stripped = remove_annotations(&original, source.removed_annotations())?;
41        let cleaned = derive_source(&stripped, source.edits(), limits.source_bytes)?;
42        if cleaned != source.source() {
43            return Err(Error::Contract(format!(
44                "recorded source derivation differs for {}",
45                source.path()
46            )));
47        }
48        for include in source.includes() {
49            let expected = format!("#include \"{}\".", include.spelling());
50            if include
51                .line()
52                .checked_sub(1)
53                .and_then(|index| original.lines().nth(index))
54                .map(str::trim)
55                != Some(expected.as_str())
56            {
57                return Err(Error::Contract(format!(
58                    "original include coordinate differs for {}",
59                    source.path()
60                )));
61            }
62        }
63        if let Some(case) = corpus
64            .cases()
65            .iter()
66            .find(|case| case.path() == source.path())
67        {
68            let translated = translate(source.removed_annotations())?;
69            if &translated != case.contract() {
70                return Err(Error::Contract(format!(
71                    "original display contract differs for {}",
72                    source.path()
73                )));
74            }
75        } else if !source.removed_annotations().is_empty() {
76            return Err(Error::Contract(format!(
77                "unowned annotations in {}",
78                source.path()
79            )));
80        }
81    }
82    Ok(())
83}

85/// Apply exact ordered replacements without changing unselected source bytes.
86///
87/// Coordinates refer to the supplied UTF-8 source before any replacement.
88/// The caller bounds the supplied source and metadata allocations. The output
89/// content-byte ceiling is checked before reservation; it does not measure RSS.
90/// This operation proves byte derivation only, not semantic equivalence.
91///
92/// # Errors
93/// Refuses invalid, overlapping or mismatched spans, exceeded output bytes and
94/// allocation failure without returning a partial source.
95pub fn derive_source(source: &str, edits: &[Edit], limit: usize) -> Result<String, Error> {
96    let mut previous_end = 0;
97    let mut length = source.len() as u128;
98    for edit in edits {
99        if edit.start_byte < previous_end
100            || edit.start_byte >= edit.end_byte
101            || source.get(edit.start_byte..edit.end_byte) != Some(edit.before.as_str())
102        {
103            return Err(invalid(
104                "source edits overlap or differ from the recorded bytes",
105            ));
106        }
107        length = length - edit.before.len() as u128 + edit.after.len() as u128;
108        previous_end = edit.end_byte;
109    }
110    ceiling(Resource::SourceBytes, length, limit)?;
111    let capacity = usize::try_from(length)
112        .map_err(|_| invalid("source derivation exceeds addressable storage"))?;
113    let mut result = String::new();
114    result
115        .try_reserve_exact(capacity)
116        .map_err(Error::Allocation)?;
117    previous_end = 0;
118    for edit in edits {
119        // The validation pass established ordered, disjoint UTF-8 boundaries.
120        result.push_str(&source[previous_end..edit.start_byte]);
121        result.push_str(&edit.after);
122        previous_end = edit.end_byte;
123    }
124    result.push_str(&source[previous_end..]);
125    Ok(result)
126}

128pub(super) fn remove_annotations(
129    original: &str,
130    annotations: &[Annotation],
131) -> Result<String, Error> {
132    let mut result = String::with_capacity(original.len());
133    let mut previous_end = 0;
134    for annotation in annotations {
135        let before = original
136            .get(..annotation.start_byte)
137            .ok_or_else(|| invalid("annotation start is outside source"))?;
138        let removed = original
139            .get(annotation.start_byte..annotation.end_byte)
140            .ok_or_else(|| invalid("annotation span is outside source"))?;
141        // This pinned corpus places all contracts in a leading ordinary-comment
142        // header. Requiring that shape excludes strings and block comments without
143        // introducing a second ASP lexer into the provenance tooling.
144        let leading_comments = before.lines().all(|line| {
145            let line = line.trim();
146            line.is_empty() || line.starts_with('%') && !line.starts_with("%*")
147        });
148        let line_boundary = before.is_empty() || before.ends_with('\n');
149        let line = before.bytes().filter(|byte| *byte == b'\n').count() + 1;
150        let without_newline = removed.strip_suffix('\n').unwrap_or(removed);
151        let spelling = without_newline
152            .strip_suffix('\r')
153            .unwrap_or(without_newline);
154        if annotation.start_byte < previous_end
155            || annotation.start_byte >= annotation.end_byte
156            || !leading_comments
157            || !line_boundary
158            || line != annotation.line
159            || !removed.ends_with('\n') && annotation.end_byte != original.len()
160            || spelling.contains(['\n', '\r'])
161            || spelling != annotation.source
162            || !spelling.trim_start().starts_with("% @")
163        {
164            return Err(invalid(
165                "annotation coordinate is not an ordinary leading comment line",
166            ));
167        }
168        result.push_str(
169            original
170                .get(previous_end..annotation.start_byte)
171                .ok_or_else(|| invalid("annotation spans overlap"))?,
172        );
173        previous_end = annotation.end_byte;
174    }
175    result.push_str(
176        original
177            .get(previous_end..)
178            .ok_or_else(|| invalid("annotation end is outside source"))?,
179    );
180    Ok(result)
181}

183fn invalid(detail: &str) -> Error {
184    Error::Contract(detail.into())
185}
186fn braces(text: &str) -> Result<&str, Error> {
187    text.strip_prefix('{')
188        .and_then(|value| value.strip_suffix('}'))
189        .map(str::trim)
190        .ok_or_else(|| invalid("annotation requires braces"))
191}
192fn symbols(text: &str) -> Result<Vec<String>, Error> {
193    answers::split_display(braces(text)?, true, answers::Limits::for_bytes(text.len()))
194        .map_err(|error| Error::Contract(error.to_string()))
195}
196fn set<T: PartialEq>(slot: &mut Option<T>, value: T) -> Result<(), Error> {
197    if slot.as_ref().is_some_and(|previous| previous != &value) {
198        return Err(invalid("conflicting annotation contracts"));
199    }
200    *slot = Some(value);
201    Ok(())
202}

204pub(super) fn translate(annotations: &[Annotation]) -> Result<Contract, Error> {
205    let mut satisfiability = None;
206    let mut family = None;
207    let mut model_count = None;
208    let mut cost = None;
209    let mut witnesses = Vec::new();
210    let mut required_symbols = Vec::new();
211    let mut notes = Vec::new();
212    for annotation in annotations {
213        let text = annotation
214            .source
215            .trim_start()
216            .strip_prefix("% @")
217            .ok_or_else(|| invalid("unsupported annotation comment"))?;
218        let (tag, argument) = text
219            .split_once(char::is_whitespace)
220            .ok_or_else(|| invalid("annotation argument is missing"))?;
221        let argument = argument.trim();
222        match tag {
223            "expect" => set(
224                &mut satisfiability,
225                match argument {
226                    "sat" => Satisfiability::Sat,
227                    "unsat" => Satisfiability::Unsat,
228                    _ => return Err(invalid("unsupported satisfiability annotation")),
229                },
230            )?,
231            "count" => {
232                let (scope, count) = argument
233                    .strip_prefix("optimal ")
234                    .map_or((Family::All, argument), |count| {
235                        (Family::Optimal, count.trim())
236                    });
237                set(&mut family, scope)?;
238                set(
239                    &mut model_count,
240                    count
241                        .parse::<u64>()
242                        .map_err(|error| Error::Contract(error.to_string()))?,
243                )?;
244            }
245            "cost" => {
246                set(&mut family, Family::Optimal)?;
247                set(
248                    &mut cost,
249                    answers::parse_costs(
250                        braces(argument)?,
251                        answers::Limits::for_bytes(argument.len()),
252                    )
253                    .map_err(|error| Error::Contract(error.to_string()))?,
254                )?;
255            }
256            "model" | "optimal" => {
257                set(
258                    &mut family,
259                    if tag == "model" {
260                        Family::All
261                    } else {
262                        Family::Optimal
263                    },
264                )?;
265                witnesses.push(symbols(argument)?);
266            }
267            "cautious" => {
268                set(&mut family, Family::Optimal)?;
269                let argument = argument
270                    .strip_prefix("optimal ")
271                    .ok_or_else(|| invalid("unsupported required-symbol annotation scope"))?;
272                required_symbols.extend(symbols(argument.trim())?);
273            }
274            "note" => notes.push(argument.to_owned()),
275            _ => return Err(invalid("unsupported original annotation")),
276        }
277    }
278    required_symbols.sort();
279    let contract = Contract {
280        satisfiability: satisfiability
281            .ok_or_else(|| invalid("missing original satisfiability contract"))?,
282        family: family.unwrap_or(Family::All),
283        model_count,
284        cost,
285        witnesses,
286        required_symbols,
287        notes,
288    };
289    document::validate_contract(&contract)?;
290    Ok(contract)
291}
