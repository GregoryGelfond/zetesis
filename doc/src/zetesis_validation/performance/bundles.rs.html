1//! Bounded JSON loading for unchanged external source workloads.
2//!
3//! This adapter owns specification I/O only. [`matrix::Workload::bundle`] owns
4//! graph admission; the existing matrix owns qualification, captures and timing.
5
6use std::path::{Path, PathBuf};
7
8use serde::Deserialize;
9
10use super::{Error, matrix};
11use crate::{examples, selected::identity};
12
13#[derive(Deserialize)]
14#[serde(deny_unknown_fields)]
15struct Manifest {
16    schema_version: u32,
17    root: PathBuf,
18    cases: Vec<Case>,
19}
20
21#[derive(Deserialize)]
22#[serde(deny_unknown_fields)]
23struct Case {
24    entry: String,
25    contract: examples::Contract,
26    files: Vec<matrix::BundleFile>,
27}
28
29/// Load 1..=94 explicitly sealed external workloads from a bounded JSON manifest.
30///
31/// Schema version 1 has `root` and `cases`; each case has `entry`, the existing
32/// typed `contract`, and `files`. Each file has `path`, lowercase `sha256` and
33/// ordered `includes`, each with `spelling` and root-relative `target`. Unknown
34/// fields are refused. Relative `root` resolves against the manifest's canonical
35/// parent; absolute roots are also admitted. Members never escape that root.
36///
37/// Reads at most `metadata_bytes` of manifest before deserialization. The same
38/// inclusive metadata and combined source ceilings bound the aggregate admitted
39/// workloads; individual source reads use `source_bytes`. Retains each original
40/// member seal and the manifest seal for the matrix's before/after checks, with
41/// no retained source payload. Work is the sum of bundle admission and bounded
42/// manifest decoding. This function neither launches a solver nor grants rights
43/// to redistribute caller-owned sources.
44///
45/// # Errors
46/// Refuses invalid JSON/schema, empty or oversized selections, duplicate workload
47/// identities, changed files, invalid graphs/contracts and exceeded byte bounds.
48pub fn load(path: &Path, limits: matrix::WorkloadLimits) -> Result<Vec<matrix::Workload>, Error> {
49    let seal = identity::seal(path, limits.metadata_bytes)?;
50    let bytes = examples::Files::read(
51        seal.requested(),
52        limits.metadata_bytes,
53        examples::Resource::ManifestBytes,
54    )
55    .map_err(Error::Corpus)?;
56    examples::Files::digest("external workload manifest", &bytes, seal.sha256())
57        .map_err(Error::Corpus)?;
58    let manifest: Manifest = serde_json::from_slice(&bytes)
59        .map_err(|error| Error::Corpus(examples::Error::Json(error)))?;
60    if manifest.schema_version != 1 || manifest.cases.is_empty() || manifest.cases.len() > 94 {
61        return Err(Error::Configuration(
62            "external workload manifest requires schema 1 and 1..=94 cases",
63        ));
64    }
65    let parent = seal.canonical().parent().ok_or(Error::Configuration(
66        "external workload manifest has no parent",
67    ))?;
68    let root = parent.join(&manifest.root);
69    let mut workloads = Vec::<matrix::Workload>::new();
70    let mut metadata_bytes = 0usize;
71    let mut source_bytes = 0usize;
72    for case in &manifest.cases {
73        let mut workload = matrix::Workload::bundle(
74            matrix::BundleProgram {
75                root: &root,
76                entry: &case.entry,
77                files: &case.files,
78                contract: &case.contract,
79            },
80            limits,
81        )?;
82        workload.attach_bundle_manifest(&seal, limits.metadata_bytes)?;
83        if workloads
84            .iter()
85            .any(|prior| prior.identity() == workload.identity())
86        {
87            return Err(Error::Configuration(
88                "external workload identities are repeated",
89            ));
90        }
91        metadata_bytes = metadata_bytes
92            .checked_add(workload.metadata_bytes)
93            .ok_or(Error::Configuration("external workload metadata overflow"))?;
94        source_bytes =
95            source_bytes
96                .checked_add(workload.source_bytes)
97                .ok_or(Error::Configuration(
98                    "external workload source bytes overflow",
99                ))?;
100        if metadata_bytes > limits.metadata_bytes || source_bytes > limits.total_source_bytes {
101            return Err(Error::Configuration(
102                "external workload selection exceeds aggregate ceilings",
103            ));
104        }
105        workloads.push(workload);
106    }
107    Ok(workloads)
108}
