1//! Reusable campaign selection for application commands.
2//!
3//! Command adapters resolve paths and map arguments to the existing typed plan.
4//! This module owns the maintained series expansion; measurement, qualification,
5//! resource limits and report publication remain the matrix library's contracts.
6
7use super::{Error, matrix, series};
8
9/// Run a matrix, expanding the maintained generated/constant series when selected.
10///
11/// Limits are used exactly as supplied. A caller choosing the maintained series
12/// can explicitly apply [`series::limits`] and [`series::native_answers`] before
13/// constructing its request. This operation never silently raises a ceiling.
14/// The returned report is not published; the caller chooses when to publish it.
15/// A series case selection names workload entries, including generated paths;
16/// each entry expands to all its cells in series order, with entries visited in
17/// the caller's order. The report retains the original selection and each
18/// expanded workload's identity.
19///
20/// # Errors
21/// Returns corpus, workload preparation or campaign configuration failures.
22pub fn run(
23    request: &matrix::Request<'_>,
24    invocation: matrix::NativeInvocation,
25) -> Result<matrix::Report, Error> {
26    run_with_cancellation(
27        request,
28        invocation,
29        &std::sync::atomic::AtomicBool::new(false),
30    )
31}
32
33/// Select and run a campaign with explicit caller-owned cancellation.
34///
35/// Cancellation and partial work are retained by the matrix owner; no global
36/// signal handlers are installed and the returned report remains unpublished.
37///
38/// # Errors
39/// Returns the same preparation refusals as [`run`].
40pub fn run_with_cancellation(
41    request: &matrix::Request<'_>,
42    invocation: matrix::NativeInvocation,
43    cancelled: &std::sync::atomic::AtomicBool,
44) -> Result<matrix::Report, Error> {
45    if request.plan.suite() == matrix::Suite::Series {
46        let corpus =
47            crate::examples::load(request.corpus, request.limits.corpus).map_err(Error::Corpus)?;
48        let workloads = series::workloads(&corpus, matrix::WorkloadLimits::default())?;
49        matrix::run_series_with_cancellation(request, &workloads, invocation, cancelled)
50    } else {
51        matrix::run_with_cancellation(request, invocation, cancelled)
52    }
53}
