1//! Generated one-shape programs whose complete families have closed forms.
2//!
3//! Each family is a few lines of source with one size parameter. The bytes are
4//! a pure function of the family and the size, so an observation can name its
5//! input exactly without retaining a source file; the size range of each family
6//! keeps the source bounded and its contract representable. These programs
7//! exercise routes the pinned example corpus does not reach: the closure route,
8//! deep derivation, cyclic and stratified negation, and large answer families.
9//! Nothing here runs a solver or claims a family beyond the stated counts.
10

19/// A generated program shape with one integer size.
20#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Serialize)]
21#[serde(rename_all = "kebab-case")]
22pub enum Family {
23    /// Independent sets of a path of `n` nodes, choice form: Fib(n+2) answers.
24    IndependentChoice,
25    /// The same independent sets in the `in/out` negation idiom.
26    IndependentNegation,
27    /// The negation idiom plus one aggregate constraint, which selects the
28    /// formula route by admission.
29    IndependentNegationAggregate,
30    /// `p(X) | q(X)` over `n` values: 2ⁿ answers.
31    Disjunction,
32    /// Choose exactly two of `n` and pay one per chosen atom: C(n,2) optimal
33    /// answers, every one at cost two.
34    Ties,
35    /// Transitive closure of a path of `n` nodes given as edge facts.
36    TransitivePath,
37    /// Transitive closure of the strict order on `n` values.
38    TransitiveDense,
39    /// A derivation chain of depth `n` over edge facts.
40    Chain,
41    /// A derivation chain of depth `n` by head arithmetic.
42    ChainArithmetic,
43    /// Stratified negation over derived atoms: one answer, decided entirely by
44    /// the well-founded model.
45    Stratified,
46    /// `n` facts and `n − 1` two-literal rules over distinct predicates.
47    ProducerChain,
48    /// Latin squares of order `n` with the first row fixed to `1..n`:
49    /// `(n − 1)!` times the number of reduced Latin squares, a constraint
50    /// problem in the shape of Sudoku, every cell a choice of one value.
51    LatinSquare,
52    /// A line walked for `n` steps, one action a step, move or stay,
53    /// ending at position `n / 2`: `C(n, n / 2)` plans, the inertia of the
54    /// position carried by frame rules from step to step.
55    Planning,
56    /// One `p/q` choice per position, with no `p` before a later `q`: `n + 1`
57    /// answers. The producer core alone has 2ⁿ answers, exposing lost pruning
58    /// when its ordinary constraints are streamed instead of retained.
59    MonotoneChoices,
60    /// A complete binary relation plus an independent `a/b` choice: two answers.
61    /// A connected transitivity constraint has n³ possible body substitutions
62    /// but rejects none. This separates constraint storage from rejection.
63    RedundantTransitivity,
64}

66/// A refused generation request.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68pub enum Error {
69    /// The size lies outside the family's admitted range.
70    Size {
71        /// Requested family.
72        family: Family,
73        /// Requested size.
74        size: u32,
75    },
76}
77impl fmt::Display for Error {
78    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
79        match self {
80            Self::Size { family, size } => write!(
81                f,
82                "family {} admits sizes {:?}, not {size}",
83                family.label(),
84                family.sizes()
85            ),
86        }
87    }
88}
89impl std::error::Error for Error {}

134    /// Admitted sizes. The upper bounds keep every count below 2⁶⁴, every
135    /// source below a mebibyte, and the stratified pattern well formed; the
136    /// Latin squares stop where their count is known in closed form. The two
137    /// constraint controls additionally cap their core choices at 2¹⁶ and their
138    /// connected substitutions at 64³. These bounds do not promise admission or
139    /// completion under any particular solver's resource settings.
140    #[must_use]
141    pub const fn sizes(self) -> RangeInclusive<u32> {
142        match self {
143            Self::IndependentChoice
144            | Self::IndependentNegation
145            | Self::IndependentNegationAggregate => 1..=64,
146            Self::Disjunction => 1..=63,
147            Self::Ties | Self::TransitivePath | Self::TransitiveDense => 2..=1024,
148            Self::Chain | Self::ChainArithmetic => 1..=8192,
149            Self::Stratified => 8..=4096,
150            Self::ProducerChain => 2..=4096,
151            Self::LatinSquare => 1..=5,
152            Self::Planning => 1..=20,
153            Self::MonotoneChoices => 1..=16,
154            Self::RedundantTransitivity => 2..=64,
155        }
156    }

158    /// The program text at `size`, byte-exact and deterministic.
159    ///
160    /// # Errors
161    /// Refuses a size outside [`Self::sizes`].
162    pub fn source(self, size: u32) -> Result<String, Error> {
163        self.admit(size)?;
164        Ok(match self {
165            Self::IndependentChoice => format!(
166                "node(1..{size}).\n{}{{ in(X) }} :- node(X).\n:- edge(X,Y), in(X), in(Y).\n",
167                path_edges("edge", 1, size)
168            ),
169            Self::IndependentNegation => independent_negation(size),
170            Self::IndependentNegationAggregate => {
171                format!("{}:- #count{{X:in(X)}} < 0.\n", independent_negation(size))
172            }
173            Self::Disjunction => format!("d(1..{size}).\np(X) | q(X) :- d(X).\n"),
174            Self::Ties => format!(
175                "n(1..{size}).\n{{ p(X) }} :- n(X).\n:- #count{{X:p(X)}} != 2.\n#minimize{{ 1,X : p(X) }}.\n"
176            ),
177            Self::TransitivePath => format!(
178                "{}reach(X,Y) :- e(X,Y).\nreach(X,Z) :- reach(X,Y), e(Y,Z).\n",
179                path_edges("e", 1, size)
180            ),
181            Self::TransitiveDense => format!(
182                "v(1..{size}).\ne(X,Y) :- v(X), v(Y), X < Y.\nreach(X,Y) :- e(X,Y).\nreach(X,Z) :- reach(X,Y), e(Y,Z).\n"
183            ),
184            Self::Chain => format!("{}r(0).\nr(Y) :- r(X), e(X,Y).\n", path_edges("e", 0, size)),
185            Self::ChainArithmetic => format!("p(0).\np(X+1) :- p(X), X < {size}.\n"),
186            Self::Stratified => stratified(size),
187            Self::ProducerChain => producer_chain(size),
188            Self::LatinSquare => latin_square(size),
189            Self::Planning => planning(size),
190            Self::MonotoneChoices => {
191                format!("d(1..{size}).\np(X) | q(X) :- d(X).\n:- p(X), q(Y), X < Y.\n")
192            }
193            Self::RedundantTransitivity => format!(
194                "d(1..{size}).\np(X,Y) :- d(X), d(Y).\na | b.\n:- p(X,Y), p(Y,Z), not p(X,Z).\n"
195            ),
196        })
197    }

199    /// The complete selected family the program has at `size`. Every admitted
200    /// size has at least one answer, so the count is never zero.
201    ///
202    /// # Errors
203    /// Refuses a size outside [`Self::sizes`].
204    pub fn contract(self, size: u32) -> Result<Contract, Error> {
205        self.admit(size)?;
206        let count = match self {
207            Self::IndependentChoice
208            | Self::IndependentNegation
209            | Self::IndependentNegationAggregate => fibonacci(size + 2),
210            Self::Disjunction => 1u64 << size,
211            Self::Ties => u64::from(size) * u64::from(size - 1) / 2,
212            Self::TransitivePath
213            | Self::TransitiveDense
214            | Self::Chain
215            | Self::ChainArithmetic
216            | Self::Stratified
217            | Self::ProducerChain => 1,
218            Self::LatinSquare => latin_squares(size),
219            Self::Planning => binomial(size, size / 2),
220            // Each cut chooses a q-prefix and a p-suffix, including either
221            // empty prefix/suffix. All domain facts occur in every answer.
222            Self::MonotoneChoices => u64::from(size) + 1,
223            // Every d/p atom is forced, so transitivity always holds. Exactly
224            // one of a or b remains in each subset-minimal interpretation.
225            Self::RedundantTransitivity => 2,
226        };
227        let count = NonZeroU64::new(count).ok_or(Error::Size { family: self, size })?;
228        Ok(match self {
229            Self::Ties => Contract::optimal_family(count, vec![2]),
230            _ => Contract::complete_family(count),
231        })
232    }
233
234    const fn admit(self, size: u32) -> Result<(), Error> {
235        if *self.sizes().start() <= size && size <= *self.sizes().end() {
236            Ok(())
237        } else {
238            Err(Error::Size { family: self, size })
239        }
240    }
241}

243// Edge facts `name(i,i+1)` for `i` in `first..last`, one line.
244fn path_edges(name: &str, first: u32, last: u32) -> String {
245    let mut text = String::new();
246    for (position, i) in (first..last).enumerate() {
247        if position > 0 {
248            text.push(' ');
249        }
250        // Writing to a `String` cannot fail.
251        let _ = write!(text, "{name}({i},{}).", i + 1);
252    }
253    text.push('\n');
254    text
255}

257fn independent_negation(size: u32) -> String {
258    format!(
259        "node(1..{size}).\n{}in(X) :- node(X), not out(X).\nout(X) :- node(X), not in(X).\n:- edge(X,Y), in(X), in(Y).\n",
260        path_edges("edge", 1, size)
261    )
262}

264// Every seventh node from 3 is bad and blocks its successor; a skip edge
265// around each node keeps the last node reachable, so the program is
266// consistent and its unique answer is fixed by the well-founded model.
267fn stratified(size: u32) -> String {
268    let bad: Vec<String> = (3..size - 2)
269        .step_by(7)
270        .map(|node| format!("bad({node})."))
271        .collect();
272    let skip: Vec<String> = (1..size - 1)
273        .map(|node| format!("e({node},{}).", node + 2))
274        .collect();
275    format!(
276        "node(1..{size}).\n{}{}\n{}{}\nblocked(Y) :- bad(X), next(X,Y).\nreach(1).\nreach(Y) :- reach(X), e(X,Y), not blocked(Y).\n:- not reach({size}).\n",
277        path_edges("e", 1, size),
278        skip.join(" "),
279        path_edges("next", 1, size),
280        bad.join(" ")
281    )
282}

284fn producer_chain(size: u32) -> String {
285    let facts: Vec<String> = (1..=size).map(|i| format!("p{i}.")).collect();
286    let rules: Vec<String> = (1..size)
287        .map(|i| format!("q{i} :- p{i}, p{}.\n", i + 1))
288        .collect();
289    format!("{}\n{}", facts.join(" "), rules.concat())
290}

292// One value in every cell, no value twice in a row or a column, and the
293// first row in order; the count is the reduced count times `(n − 1)!`.
294fn latin_square(size: u32) -> String {
295    format!(
296        "n(1..{size}).\n1 {{ cell(R,C,V) : n(V) }} 1 :- n(R), n(C).\n:- cell(R,C1,V), cell(R,C2,V), C1 < C2.\n:- cell(R1,C,V), cell(R2,C,V), R1 < R2.\n:- n(C), not cell(1,C,C).\n#show cell/3.\n"
297    )
298}

300// The reduced Latin squares of orders one through five, those with the
301// first row and the first column in order; the squares with the first row
302// fixed are `(n − 1)!` times as many, one for each permutation of the first
303// column below its first cell.
304fn latin_squares(size: u32) -> u64 {
305    const REDUCED: [u64; 5] = [1, 1, 1, 4, 56];
306    let factorial: u64 = (1..u64::from(size)).product();
307    factorial * REDUCED[size as usize - 1]
308}

310// Move or stay at each step; the position carries by inertia when the
311// walker stays and advances when it moves; the goal fixes the end.
312fn planning(size: u32) -> String {
313    let goal = size / 2;
314    format!(
315        "step(0..{}).\ntime(0..{size}).\npos(0..{size}).\nat(0,0).\n1 {{ move(T); stay(T) }} 1 :- step(T).\nat(P+1,T+1) :- at(P,T), move(T), pos(P+1).\nat(P,T+1) :- at(P,T), stay(T).\n:- not at({goal},{size}).\n#show move/1.\n",
316        size - 1
317    )
318}

320fn binomial(n: u32, k: u32) -> u64 {
321    (1..=u64::from(k)).fold(1, |acc, i| acc * (u64::from(n) - i + 1) / i)
322}

324fn fibonacci(index: u32) -> u64 {
325    let (mut previous, mut current) = (0u64, 1u64);
326    for _ in 0..index {
327        (previous, current) = (current, previous + current);
328    }
329    previous
