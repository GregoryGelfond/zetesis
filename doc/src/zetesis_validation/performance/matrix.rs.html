1//! Instrumented end-to-end profile comparisons over clean source workloads.
2//!
3//! Every native invocation uses JSON and statistics. Wall time includes their
4//! overhead, setup, grounding, solving and captured output. This is distinct from
5//! the legacy uninstrumented CPU protocol and the eventual full uninstrumented
6//! corpus matrix. Selected displays/counts/costs are compared with clingo when
7//! it takes part, and with each workload's recorded contract when the campaign
8//! is clingo-free; a workload without a contract then needs clingo and is not
9//! launched. Native full families also agree across profiles and repeats,
10//! including hidden atoms, shown values and multiplicities. Hidden clingo
11//! interpretations are unavailable. A qualification-only reference policy leaves
12//! all later measurements native.
13//!
14//! First-observed refusals and failures disable only their cell's later launches;
15//! every fixed schedule position remains recorded. No failed sample is replaced.
16//! Callers arrange a qualified executable and quiet physical-device window.
17//! The top-level manifest is the pinned catalog context for corpus selections.
18//! Generated and authored workloads authenticate their own inputs in per-workload
19//! provenance; the catalog does not claim to contain or qualify those sources.

57/// Owned complete matrix evidence, including unlaunched positions and refusals.
58#[derive(Debug, Serialize)]
59pub struct Report {
60    schema: u32,
61    protocol: &'static str,
62    tool: Tool,
63    manifest_sha256: &'static str,
64    manifest_scope: &'static str,
65    plan: RecordedPlan,
66    limits: super::Limits,
67    native_normalization_limits: serde_json::Value,
68    cases: Vec<String>,
69    #[serde(skip_serializing_if = "Option::is_none")]
70    workloads: Option<Vec<Workload>>,
71    started_unix_ns: u128,
72    finished_unix_ns: Option<u128>,
73    wall_scope: &'static str,
74    comparison_scope: &'static str,
75    peak_rss: &'static str,
76    before: Vec<FileSeal>,
77    after: Vec<Change>,
78    metadata: Vec<serialization::MatrixCapture>,
79    samples: Vec<Sample>,
80    total_capture_bytes: usize,
81    faults: Vec<Fault>,
82    unresolved_children: Vec<u32>,
83    #[serde(skip)]
84    destination: publication::Destination,
85}
86impl Report {
87    /// Derive compact typed counts, successful timed distributions and separate
88    /// memory observations without copying or parsing raw answer captures.
89    /// It scans all positions once per source/producer cell and retains only
90    /// one cell's interval scratch plus the returned compact rows.
91    #[must_use]
92    pub fn summary(&self) -> Summary<'_> {
93        summary::summarize(self)
94    }
95
96    /// Every planned position is represented; this does not require solver parity.
97    /// The schedule is rebuilt from the policy the report records.
98    #[must_use]
99    pub fn accounted(&self) -> bool {
100        self.plan
101            .plan
102            .slots(self.cases.len(), self.plan.reference_policy.reference())
103            .is_ok_and(|slots| slots.len() == self.samples.len())
104    }
105    /// Every requested invocation passed parity/telemetry with unchanged inputs,
106    /// and every metadata capture the run owed completed: zetesis's version and
107    /// help, and clingo's version when clingo took part. Refused, skipped and
108    /// needs-clingo cells make this false even in a fully accounted report.
109    #[must_use]
110    pub fn passed(&self) -> bool {
111        let owed = 2 + usize::from(self.plan.reference_policy.reference().is_some());
112        self.accounted()
113            && self.samples.iter().all(|s| s.decision == Decision::Pass)
114            && self.metadata.len() == owed
115            && self.metadata.iter().all(|c| c.0.complete(false))
116            && self.after.iter().all(Change::unchanged)
117            && self.faults.is_empty()
118            && self.unresolved_children.is_empty()
119            && self.finished_unix_ns.is_some()
120    }
121    /// Exact requested and actual disposition at each schedule position.
122    #[must_use]
123    pub fn samples(&self) -> &[Sample] {
124        &self.samples
125    }
126    /// Entry paths in schedule order; explicit workloads may repeat a path.
127    #[must_use]
128    pub fn cases(&self) -> &[String] {
129        &self.cases
130    }
131    /// Explicit workload identities in schedule order; absent for default suites.
132    #[must_use]
133    pub fn workloads(&self) -> Option<&[Workload]> {
134        self.workloads.as_deref()
135    }
136    /// Setup/capture scheduling faults, separate from solver conclusions.
137    #[must_use]
138    pub fn faults(&self) -> &[Fault] {
139        &self.faults
140    }
141    /// Immutable requested profile and schedule configuration.
142    #[must_use]
143    pub const fn plan(&self) -> &Plan {
144        &self.plan.plan
145    }
146    /// The reference policy the run used; none for a clingo-free campaign.
147    #[must_use]
148    pub const fn reference_policy(&self) -> Option<ReferencePolicy> {
149        self.plan.reference_policy.reference()
150    }
151    /// The tool that ran the campaign and wrote this report.
152    #[must_use]
153    pub const fn tool(&self) -> &Tool {
154        &self.tool
155    }
156    /// Pre-run primary and private-copy source/executable seals.
157    #[must_use]
158    pub fn before(&self) -> &[FileSeal] {
159        &self.before
160    }
161    /// Post-run seal checks, including failed reads.
162    #[must_use]
163    pub fn after(&self) -> &[Change] {
164        &self.after
165    }
166    /// Retained executable version/help captures outside solve observations.
167    #[must_use]
168    pub fn metadata(&self) -> impl ExactSizeIterator<Item = &Capture> {
169        self.metadata.iter().map(|capture| &capture.0)
170    }
171    /// Direct child IDs explicitly abandoned after bounded cleanup failed.
172    #[must_use]
173    pub fn unresolved_children(&self) -> &[u32] {
174        &self.unresolved_children
175    }
176    /// Raw retained capture bytes, excluding JSON serialization and allocator overhead.
177    #[must_use]
178    pub const fn total_capture_bytes(&self) -> usize {
179        self.total_capture_bytes
180    }
181    /// Publish evidence without replacing inputs, executables or existing files.
182    /// The destination parent must remain exclusively controlled by the caller.
183    ///
184    /// # Errors
185    /// Refuses aliases, changed destination identity, byte ceilings or I/O failures.
186    pub fn publish(&self) -> Result<(), Error> {
187        #[derive(Serialize)]
188        struct Published<'a> {
189            passed: bool,
190            accounted: bool,
191            report: &'a Report,
192        }
193        publication::write(
194            &Published {
195                passed: self.passed(),
196                accounted: self.accounted(),
197                report: self,
198            },
199            &self.destination,
200            self.limits.max_report_bytes,
201        )
202        .map_err(Error::Boundary)
203    }
204}
205/// Seal the verified source/include closure and binaries, then run the bounded
206/// fixed matrix. Sources retain exact bytes in a private directory. Refusals,
207/// timeouts and mismatches remain evidence rather than replacing samples.
208/// Direct-child cleanup failure prevents further process launches. Bounds cover
209/// requested capture/storage quantities, not allocator RSS or OS response latency.
210///
211/// # Errors
212/// Returns configuration/source/identity failures before any solver is launched.
213pub fn run(request: &Request<'_>) -> Result<Report, Error> {
214    run_with_invocation(request, NativeInvocation::Legacy)
215}

217/// Run explicit constant variants drawn from the plan's allowed source suite.
218///
219/// The existing profile schedule, process capture and complete-answer decoder
220/// are shared with `run`. Each instance has a private source closure and a
221/// distinct content identity, even when several instances share an entry path.
222/// Default contracts remain provenance for amended workloads; those workloads
223/// require a complete clingo family instead of the default model count.
224/// Reports use schema 2. No first-answer samples are added; memory rounds
225/// follow the plan.
226///
227/// # Errors
228/// Refuses a plan with a case selection, which chooses suite cases rather than
229/// explicit workloads, empty/oversized populations, repeated content
230/// identities, foreign corpus identities, sources outside the allowed suite and
231/// resource excess. Materialization failures are retained in the returned
232/// report before any solver invocation.
233pub fn run_workloads(request: &Request<'_>, workloads: &[Workload]) -> Result<Report, Error> {
234    run_workloads_with_invocation(request, workloads, NativeInvocation::Legacy)
235}

240/// Run the fixed matrix using an explicit native command interface.
241///
242/// The sealed executable must support the selected interface. Both interfaces
243/// request the same machine answers and statistics; all qualification, bounds
244/// and failure retention are shared with [`run`].
245///
246/// # Errors
247/// Returns the same pre-launch failures as [`run`].
248pub fn run_with_invocation(
249    request: &Request<'_>,
250    invocation: NativeInvocation,
251) -> Result<Report, Error> {
252    run_with_cancellation(
253        request,
254        invocation,
255        &std::sync::atomic::AtomicBool::new(false),
256    )
257}

259/// Run a fixed matrix with caller-owned cancellation and no signal handlers.
260///
261/// The active child receives bounded ownership-checked cleanup. No subsequent
262/// child starts after cancellation is observed; the fixed remaining schedule is
263/// retained as unattempted. Partial capture and cancellation are nonpassing
264/// evidence, not an empty answer family or a completed measurement.
265/// The cancellation flag must remain set once requested.
266///
267/// # Errors
268/// Returns the same pre-launch refusals as [`run_with_invocation`].
269pub fn run_with_cancellation(
270    request: &Request<'_>,
271    invocation: NativeInvocation,
272    cancelled: &std::sync::atomic::AtomicBool,
273) -> Result<Report, Error> {
274    run::campaign(request, None, invocation, cancelled)
275}

307/// Expand a selection over the maintained series' workload entries before
308/// invoking the explicit-workload runner. One entry can name several amended
309/// cells; every match keeps its identity and its order within that entry.
310/// Only this command-owned expansion consumes a selection: the public
311/// explicit-workload operations still refuse one. The returned report records
312/// both the original request and the actual expanded population.
313pub(super) fn run_series_with_cancellation(
314    request: &Request<'_>,
315    workloads: &[Workload],
316    invocation: NativeInvocation,
317    cancelled: &std::sync::atomic::AtomicBool,
318) -> Result<Report, Error> {
319    let Some(selection) = request.plan.selection() else {
320        return run_workloads_with_cancellation(request, workloads, invocation, cancelled);
321    };
322    let mut selected = Vec::with_capacity(workloads.len());
323    for path in selection {
324        let previous = selected.len();
325        selected.extend(
326            workloads
327                .iter()
328                .filter(|workload| workload.entry() == path.as_str())
329                .cloned(),
330        );
331        if selected.len() == previous {
332            return Err(Error::Configuration(
333                "a selected case is not a workload of the series",
334            ));
335        }
336    }
337    let expanded = Request {
338        tool: request.tool.clone(),
339        plan: Plan {
340            selection: None,
341            ..request.plan.clone()
342        },
343        ..*request
344    };
345    let mut report = run::campaign(&expanded, Some(&selected), invocation, cancelled)?;
346    report.plan.plan = request.plan.clone();
347    Ok(report)
348}
