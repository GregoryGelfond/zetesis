13#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]

15pub enum Suite {

17    Corpus,
18    /// Established SEND, queens02 and task-allocation cases.
19    Baseline,
20    /// All six curated queens encodings at their default N=8.
21    Queens,
22    /// The fixed cell set of [`super::super::series`]: generated families,
23    /// amended queens boards and two unchanged entries, meant for
24    /// `run_workloads` with the series' workloads. Its corpus entries are

27    Series,
28    /// Authored queens/pigeonhole sizes and the three baseline corpus entries,
29    /// with optional Einstein. Requires explicit workloads, normally supplied by
30    /// [`super::super::scalability::run_with_cancellation`].
31    Scalability,

34/// Which populations include the independent reference solver when one takes part.
35#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
36#[serde(rename_all = "snake_case")]
37pub enum ReferencePolicy {
38    /// Qualify and measure the reference alongside every native profile.
39    AllPhases,
40    /// Establish one complete reference census per case; measure native profiles only.
41    QualificationOnly,
42}

44/// The reference policy a report records: the one the campaign ran under, or
45/// `clingo_free` when no reference took part.
46#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
47#[serde(rename_all = "snake_case")]
48pub enum RecordedPolicy {
49    /// The reference was qualified and measured alongside every native profile.
50    AllPhases,
51    /// The reference established one complete census per case; only native
52    /// profiles were measured.
53    QualificationOnly,
54    /// No reference took part: each native family was qualified against its
55    /// workload's recorded contract.
56    ClingoFree,
57}
58impl RecordedPolicy {
59    /// The policy the campaign ran under; none for a clingo-free campaign.
60    #[must_use]
61    pub const fn reference(self) -> Option<ReferencePolicy> {
62        match self {
63            Self::AllPhases => Some(ReferencePolicy::AllPhases),
64            Self::QualificationOnly => Some(ReferencePolicy::QualificationOnly),
65            Self::ClingoFree => None,
66        }
67    }
68}
69impl From<Option<ReferencePolicy>> for RecordedPolicy {
70    fn from(policy: Option<ReferencePolicy>) -> Self {
71        match policy {
72            Some(ReferencePolicy::AllPhases) => Self::AllPhases,
73            Some(ReferencePolicy::QualificationOnly) => Self::QualificationOnly,
74            None => Self::ClingoFree,
75        }
76    }
77}

90/// Validated finite campaign configuration; requested profiles never imply execution.
91/// Whether a reference takes part is the request's, not the plan's.
92#[derive(Clone, Debug, Serialize)]
93pub struct Plan {
94    pub(super) suite: Suite,
95    pub(super) profiles: Vec<NativeExecution>,
96    pub(super) reference_workers: NonZeroUsize,
97    pub(super) warmups: usize,
98    pub(super) repetitions: usize,
99    /// Separate child-resource rounds per producer and case, after the
100    /// timed rounds; zero unless requested.
101    pub(super) memory_runs: usize,
102    /// Relative entry paths of the suite to run, in caller order; the
103    /// whole suite when absent.
104    #[serde(skip_serializing_if = "Option::is_none")]
105    pub(super) selection: Option<Vec<String>>,
106}
107impl Plan {
108    /// Construct up to eight profiles, on any backend. Each worker count is bounded
109    /// by 256; every profile retains its batch/scratch ceilings. Zero through
110    /// five warmups and one through 41 timed rounds are admitted. An automatic
111    /// grounding request is admitted; its observations retain the mode taken.
112    ///
113    /// # Errors
114    /// Refuses empty/oversized profile families or counts outside these bounds.
115    /// Repeated profiles are permitted as authored controls.
116    pub fn new(
117        suite: Suite,
118        profiles: Vec<NativeExecution>,
119        reference_workers: NonZeroUsize,
120        warmups: usize,
121        repetitions: usize,
122    ) -> Result<Self, Error> {
123        super::super::Schedule::new(warmups, repetitions)?;
124        validate_profiles(&profiles, reference_workers)?;
125        Ok(Self {
126            suite,
127            profiles,
128            reference_workers,
129            warmups,
130            repetitions,
131            memory_runs: 0,
132            selection: None,
133        })
134    }
135
136    /// Qualify each complete family once with each native profile, and with
137    /// the reference when one takes part. No warmup, timed or memory positions
138    /// are scheduled. Profile and worker bounds are identical to [`Self::new`].
139    ///
140    /// # Errors
141    /// Refuses empty/oversized profile families or worker counts above 256.
142    pub fn qualification(
143        suite: Suite,
144        profiles: Vec<NativeExecution>,
145        reference_workers: NonZeroUsize,
146    ) -> Result<Self, Error> {
147        validate_profiles(&profiles, reference_workers)?;
148        Ok(Self {
149            suite,
150            profiles,
151            reference_workers,
152            warmups: 0,
153            repetitions: 0,
154            memory_runs: 0,
155            selection: None,
156        })
157    }
158    /// Request zero through 41 memory rounds per producer and case: each a
159    /// separate invocation through a fresh helper that reports the child's
160    /// peak resident set, excluded from the timed population.
161    ///
162    /// # Errors
163    /// Refuses more than 41 rounds, or nonzero rounds on a qualification plan.
164    pub fn with_memory(mut self, rounds: usize) -> Result<Self, Error> {
165        if rounds > 41 || (self.repetitions == 0 && rounds != 0) {
166            return Err(Error::Configuration(
167                "memory rounds must be 0..=41 and require a measurement plan",
168            ));
169        }
170        self.memory_runs = rounds;
171        Ok(self)
172    }
173    /// Memory rounds per producer and case.
174    #[must_use]
175    pub const fn memory_runs(&self) -> usize {
176        self.memory_runs
177    }
178    /// Run exactly `paths`, one through 94 distinct relative entry paths of
179    /// the plan's suite, in the order given, instead of the whole suite.
180    /// The ordinary matrix runner selects corpus cases. When
181    /// [`super::super::command::run`] expands the series, paths name workload
182    /// entries, including generated paths, and one entry can select several
183    /// amended cells. An unknown path is refused before launching anything.
184    ///
185    /// # Errors
186    /// Refuses empty or oversized selections, duplicates and escaping or empty paths.
187    pub fn with_cases(mut self, paths: Vec<String>) -> Result<Self, Error> {
188        super::super::config::selection(&paths)?;
189        self.selection = Some(paths);
190        Ok(self)
191    }
192    /// The selected cases, in order; none when the whole suite runs.
193    #[must_use]
194    pub fn selection(&self) -> Option<&[String]> {
195        self.selection.as_deref()
196    }
197    /// Ordered requested native profiles, indexed by [`Producer::Native`].
198    #[must_use]
199    pub fn profiles(&self) -> &[NativeExecution] {
200        &self.profiles
201    }
202    /// Selected base family; explicit workloads must originate in this family.
203    #[must_use]
204    pub const fn suite(&self) -> Suite {
205        self.suite
206    }
207    /// Complete schedule under `reference`, the policy of the reference that
208    /// takes part, or none for a clingo-free campaign. Qualification visits the
209    /// reference first. Later rounds rotate both case and producer positions,
210    /// without compacting refused cells. The memory rounds follow the timed rounds.
211    ///
212    /// # Errors
213    /// Refuses a case count outside the sealed corpus maximum.
214    pub fn slots(
215        &self,
216        cases: usize,
217        reference: Option<ReferencePolicy>,
218    ) -> Result<Vec<Slot>, Error> {
219        if !(1..=MAX_CASES).contains(&cases) {
220            return Err(Error::Configuration("matrix cases must be 1..=94"));
221        }
222        let width = self.profiles.len() + 1;
223        let mut slots = Vec::with_capacity(
224            cases * width * (1 + self.warmups + self.repetitions + self.memory_runs),
225        );
226        for (phase, rounds) in [
227            (Phase::Qualification, 1),
228            (Phase::Warmup, self.warmups),
229            (Phase::Timed, self.repetitions),
230            (Phase::Memory, self.memory_runs),
231        ] {
232            let reference = reference.is_some_and(|policy| {
233                phase == Phase::Qualification || policy == ReferencePolicy::AllPhases
234            });
235            let width = self.profiles.len() + usize::from(reference);
236            for round in 0..rounds {
237                for position in 0..cases {
238                    let case = (position + round) % cases;
239                    for position in 0..width {
240                        let index = if phase == Phase::Qualification {
241                            position
242                        } else {
243                            (position + round + case) % width
244                        };
245                        slots.push(Slot {
246                            case,
247                            phase,
248                            round,
249                            producer: if reference && index == 0 {
250                                Producer::Reference
251                            } else {
252                                Producer::Native {
253                                    profile: index - usize::from(reference),
254                                }
255                            },
256                        });
257                    }
258                }
259            }
260        }
261        Ok(slots)
262    }
263}

265fn validate_profiles(
266    profiles: &[NativeExecution],
267    reference_workers: NonZeroUsize,
268) -> Result<(), Error> {
269    if profiles.is_empty()
270        || profiles.len() > 8
271        || reference_workers.get() > 256
272        || profiles
273            .iter()
274            .any(|profile| profile.workers.get() > 256 || profile.completion_workers.get() > 256)
275    {
276        return Err(Error::Configuration(
277            "matrix requires 1..=8 profiles and workers 1..=256",
278        ));
279    }
280    Ok(())
281}
282/// Producer at a fixed configuration position.
283#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
284#[serde(tag = "solver", rename_all = "snake_case")]
285pub enum Producer {
286    /// Independent clingo, with the configured reference worker count.
287    Reference,
288    /// Native profile at its zero-based position in [`Plan::profiles`].
289    Native {
290        /// Profile index.
291        profile: usize,
292    },
293}
294impl Producer {
295    pub(super) const fn index(self) -> usize {
296        match self {
297            Self::Reference => 0,
298            Self::Native { profile } => profile + 1,
299        }
300    }
301}
302/// A requested invocation, retained even when an earlier failure prevents launch.
303#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
304pub struct Slot {
305    /// Zero-based case index in the report's input order.
306    pub case: usize,

309    /// Round within the population.
310    pub round: usize,
311    /// Requested solver/profile.
312    pub producer: Producer,
313}

325/// Library-owned experiment request, independent of clap and global I/O.
326#[derive(Debug)]
327pub struct Request<'a> {
328    /// The tool that runs the campaign; its report names it.
329    pub tool: Tool,
330    /// Verified clean examples/correctness root.
331    pub corpus: &'a Path,
332    /// Absolute native executable.
333    pub native: &'a Path,
334    /// The independent reference solver when one takes part; absent, the
335    /// campaign is clingo-free.
336    pub reference: Option<Reference<'a>>,
337    /// New no-clobber evidence path in an exclusively owned parent.
338    pub report: &'a Path,
339    /// Immutable profile/schedule configuration.
340    pub plan: Plan,
341    /// Existing process, corpus, cumulative capture and publication bounds.
342    pub limits: super::super::Limits,
343    /// Native typed model/value acceptance bounds.
344    pub native_answers: crate::answers::native_json::Limits,
345    /// Combined selected-symbol spelling bytes per native report.
346    pub max_spelling_bytes: usize,
347    /// Absolute helper executable that runs each memory round's solver as
348    /// its child and records the child's peak resident set; required when
349    /// the plan has memory rounds, and sealed with the other executables.
350    pub helper: Option<&'a Path>,
351}
