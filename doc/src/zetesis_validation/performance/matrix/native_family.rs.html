1//! Exact native families, independent of publication and catalog ordering.
2//!
3//! One decoded qualification family is retained per workload. Comparisons borrow
4//! atom/value payloads and sort only pointer vectors outside the measured interval.
5//! Duplicate records and shown terms remain occurrences, never set entries.
6
7use crate::answers::native_json::{ModelRecord, NativeAnswers};
8use zetesis_core::{Atom, Value};
9
10#[derive(PartialEq, Eq, PartialOrd, Ord)]
11struct Record<'a> {
12    atoms: Vec<&'a Atom>,
13    shown: Vec<&'a Atom>,
14    terms: Vec<&'a Value>,
15    costs: Option<&'a [(i32, i64)]>,
16}
17
18impl<'a> From<&'a ModelRecord> for Record<'a> {
19    fn from(model: &'a ModelRecord) -> Self {
20        let mut atoms: Vec<_> = model.full_model().iter().collect();
21        let mut shown: Vec<_> = model
22            .shown_atom_indices()
23            .iter()
24            .map(|&index| &model.full_model()[index])
25            .collect();
26        let mut terms: Vec<_> = model.shown_terms().iter().collect();
27        atoms.sort_unstable();
28        shown.sort_unstable();
29        terms.sort_unstable();
30        Self {
31            atoms,
32            shown,
33            terms,
34            costs: model.costs(),
35        }
36    }
37}
38
39fn records(family: &NativeAnswers) -> Vec<Record<'_>> {
40    let mut records: Vec<_> = family.records().iter().map(Record::from).collect();
41    records.sort_unstable();
42    records
43}
44
45/// Record the first successful native census, then require exact native parity.
46/// The caller supplies only completely decoded, reference-qualified answers.
47/// Retained population is bounded by the existing per-report decoder/capture
48/// limits and finite case count; it is not an allocator-RSS bound.
49pub(super) fn accept(
50    baseline: &mut Option<NativeAnswers>,
51    observed: NativeAnswers,
52) -> Result<(), (super::Decision, String)> {
53    if let Some(expected) = baseline {
54        if expected.costs() != observed.costs() || records(expected) != records(&observed) {
55            return Err((super::Decision::ParityMismatch,
56                "complete native atoms, shown values, multiplicities or costs differ from the native census".into()));
57        }
58    } else {
59        *baseline = Some(observed);
60    }
61    Ok(())
62}
