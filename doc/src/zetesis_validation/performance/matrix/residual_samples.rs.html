1//! Wire measurements for the two populations of committed GPU handovers.

4/// Whether the published ranks cover the complete eligible population.
5#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
6#[serde(rename_all = "snake_case")]
7pub enum ResidualQuantiles {
8    /// Measurement coverage or sample size was insufficient.
9    Unavailable,
10    /// All eligible values were retained.
11    Exact,
12    /// Ranks were computed from a smaller uniform sample.
13    Estimated,
14}
15
16/// One population's total and its uniform reservoir.
17#[derive(Debug, PartialEq, Eq, Serialize, Deserialize)]
18pub struct ResidualSamplePopulation {
19    /// Total members, when known.
20    #[serde(deserialize_with = "super::handovers::nullable")]
21    pub seen: Option<u64>,
22    /// Maximum number of retained members.
23    pub capacity: usize,
24    /// Occupied positions, including missing measurements.
25    pub retained: usize,
26}

28/// Coverage, exact reductions and ranks for one measurement.
29#[derive(Debug, PartialEq, Eq, Serialize, Deserialize)]
30pub struct ResidualSampleMetric {
31    /// Members to which this measurement applies, when known.
32    #[serde(deserialize_with = "super::handovers::nullable")]
33    pub eligible: Option<u64>,
34    /// Eligible members with a known value, absent when the count overflowed.
35    #[serde(deserialize_with = "super::handovers::nullable")]
36    pub known: Option<u64>,
37    /// Usable values from the retained sample.
38    pub retained: usize,
39    /// Exact sum when representable with complete coverage.
40    #[serde(deserialize_with = "super::handovers::nullable")]
41    pub sum: Option<u128>,
42    /// Population maximum when known.
43    #[serde(deserialize_with = "super::handovers::nullable")]
44    pub maximum: Option<u128>,
45    /// Midpoint median rounded down in the metric's integer units.
46    #[serde(deserialize_with = "super::handovers::nullable")]
47    pub median: Option<u128>,
48    /// Nearest-rank ninetieth percentile.
49    #[serde(deserialize_with = "super::handovers::nullable")]
50    pub p90: Option<u128>,
51    /// Provenance of the published ranks.
52    pub quantiles: ResidualQuantiles,
53}

93/// Compact versioned run report. Raw sampled records stay with their run owner.
94#[derive(Debug, PartialEq, Eq, Serialize, Deserialize)]
95pub struct ResidualRunSamples {
96    /// Wire format version.
97    pub version: u32,
98    /// Random-stream and sampling protocol.
99    pub algorithm: String,
100    /// Initial run seed.
101    pub seed: u64,
102    /// Independent observation byte allowance.
103    pub byte_limit: u64,
104    /// Currently retained accounted bytes.
105    pub retained_bytes: u64,
106    /// Largest observed storage demand.
107    pub peak_bytes: u64,
108    /// Population of residual worlds.
109    pub worlds: ResidualSamplePopulation,
110    /// Population of committed batches.
111    pub batches: ResidualSamplePopulation,
112    /// Settled groups when finalization and its storage were available.
113    #[serde(deserialize_with = "super::handovers::nullable")]
114    pub groups: Option<Box<ResidualSampleGroups>>,
115}

117impl ResidualRunSamples {
118    pub(crate) fn check(&self, committed: Option<u64>) -> Result<(), String> {
119        if self.version != 1 || self.algorithm != "splitmix64-reject-v1" {
120            return Err("unsupported GPU residual sampling protocol".into());
121        }
122        if self.retained_bytes > self.byte_limit || self.peak_bytes < self.retained_bytes {
123            return Err("inconsistent GPU residual sampling bytes".into());
124        }
125        for pool in [&self.worlds, &self.batches] {
126            if pool.retained > pool.capacity
127                || pool
128                    .seen
129                    .is_some_and(|seen| pool.retained as u128 > u128::from(seen))
130            {
131                return Err("inconsistent GPU residual sample population".into());
132            }
133        }
134        if committed
135            .zip(self.batches.seen)
136            .is_some_and(|(a, b)| a != b)
137        {
138            return Err("sampled GPU batch population disagrees with commits".into());
139        }
140        if let Some(groups) = &self.groups {
141            if groups
142                .all
143                .residuals
144                .sum
145                .zip(self.worlds.seen)
146                .is_some_and(|(sum, seen)| sum != u128::from(seen))
147            {
148                return Err("residual batch counts disagree with the world population".into());
149            }
150            for group in [
151                &groups.all,
152                &groups.fixed_point,
153                &groups.round_limit,
154                &groups.work_limit,
155            ] {
156                for metric in [
157                    &group.open_atoms,
158                    &group.visited,
159                    &group.attempted_splits,
160                    &group.published_splits,
161                    &group.depth,
162                    &group.work,
163                    &group.query_ns,
164                ] {
165                    metric.check(&self.worlds)?;
166                }
167                for metric in [&group.residuals, &group.width, &group.batch_ns] {
168                    metric.check(&self.batches)?;
169                    if metric
170                        .eligible
171                        .zip(self.batches.seen)
172                        .is_some_and(|(a, b)| a != b)
173                    {
174                        return Err("residual batch metric omitted committed batches".into());
175                    }
176                }
177            }
178        }
179        Ok(())
180    }
181}

183impl ResidualSampleMetric {
184    fn check(&self, pool: &ResidualSamplePopulation) -> Result<(), String> {
185        let sample = self.retained as u128;
186        if self.retained > pool.retained
187            || self.known.is_some_and(|n| sample > u128::from(n))
188            || self
189                .known
190                .zip(self.eligible)
191                .is_some_and(|(known, n)| known > n)
192            || self
193                .known
194                .zip(pool.seen)
195                .is_some_and(|(known, n)| known > n)
196            || self
197                .eligible
198                .zip(pool.seen)
199                .is_some_and(|(eligible, seen)| eligible > seen)
200        {
201            return Err("inconsistent residual metric coverage".into());
202        }
203        let complete = self.known.is_some() && self.eligible == self.known;
204        let known = self.known.map(u128::from);
205        let ranks = self.median.zip(self.p90);
206        let valid = match self.quantiles {
207            ResidualQuantiles::Unavailable => self.median.is_none() && self.p90.is_none(),
208            ResidualQuantiles::Exact => {
209                complete && sample > 0 && Some(sample) == known && ranks.is_some()
210            }
211            ResidualQuantiles::Estimated => {
212                complete && sample > 0 && known.is_some_and(|n| sample < n) && ranks.is_some()
213            }
214        };
215        if !valid
216            || !complete && self.sum.is_some()
217            || ranks.is_some_and(|(m, p)| m > p)
218            || self.known == Some(0) && self.maximum.is_some()
219            || self
220                .maximum
221                .zip(self.sum)
222                .is_some_and(|(max, sum)| max > sum)
223            || self.maximum.zip(self.p90).is_some_and(|(max, p)| p > max)
224        {
225            return Err("inconsistent residual metric result".into());
226        }
227        Ok(())
228    }
