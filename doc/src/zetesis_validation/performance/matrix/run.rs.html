17/// Scheduling limits shared by metadata, timed and supervised memory children.
18#[derive(Clone, Copy)]
19struct Schedule<'a> {
20    deadline: Instant,
21    cancelled: &'a AtomicBool,
22}
23impl Schedule<'_> {
24    fn stopped(self, report: &mut Report) -> bool {
25        if self.cancelled.load(Ordering::Relaxed) {
26            record_cancellation(report);
27            true
28        } else {
29            !report.unresolved_children.is_empty()
30        }
31    }
32}

34pub(super) fn campaign(
35    request: &Request<'_>,
36    workloads: Option<&[Workload]>,
37    invocation: NativeInvocation,
38    cancelled: &AtomicBool,
39) -> Result<Report, Error> {
40    if !cfg!(any(target_os = "linux", target_os = "macos")) {
41        return Err(Error::Configuration(
42            "bounded process capture requires Linux or macOS",
43        ));
44    }
45    let deadline = Instant::now()
46        .checked_add(request.limits.campaign_timeout)
47        .ok_or(Error::Configuration(
48            "campaign deadline is not representable",
49        ))?;
50    let started =
51        capture::unix_ns().ok_or(Error::Configuration("UTC metadata precedes Unix epoch"))?;
52    if request.plan.memory_runs() > 0 && !request.helper.is_some_and(Path::is_absolute) {
53        return Err(Error::Configuration(
54            "memory rounds require an absolute helper executable",
55        ));
56    }
57    // A selection chooses cases of the suite; explicit workloads bring their
58    // own, so a selection beside them would be silently ignored.
59    if workloads.is_some() && request.plan.selection().is_some() {
60        return Err(Error::Configuration(
61            "a case selection applies to suite cases, not to explicit workloads",
62        ));
63    }
64    let corpus = examples::load(request.corpus, request.limits.corpus).map_err(Error::Corpus)?;
65    let cases = prepare(&corpus, request, workloads)?;
66    let sources: BTreeSet<_> = cases
67        .iter()
68        .flat_map(|case| case.input.corpus_source_paths())
69        .collect();
70    let before = seal_inputs(&corpus, &sources, request, workloads)?;
71    let destination = publication::prepare(request.report, corpus.root(), &before)?;
72    let directory = tempfile::tempdir()
73        .map_err(|e| super::super::io(Path::new("private matrix sources"), e))?;
74    let mut report = Report {
75        schema: if workloads.is_some() { 2 } else { 1 },
76        protocol: if workloads.is_some() {
77            "instrumented_derived_workload_matrix_v2"
78        } else {
79            "instrumented_explicit_profile_matrix_v1"
80        },
81        tool: request.tool.clone(),
82        manifest_sha256: examples::MANIFEST_SHA256,
83        manifest_scope: "correctness_catalog_context; authored_and_generated_workloads_retain_independent_source_and_contract_identities",
84        plan: RecordedPlan {
85            plan: request.plan.clone(),
86            reference_policy: request.reference.map(|reference| reference.policy).into(),
87        },
88        limits: request.limits,
89        native_normalization_limits: normalization_limits(request),
90        cases: cases
91            .iter()
92            .map(|case| case.input.path().to_owned())
93            .collect(),
94        workloads: workloads.map(<[Workload]>::to_vec),
95        started_unix_ns: started,
96        finished_unix_ns: None,
97        wall_scope: wall_scope(request.reference.is_some()),
98        comparison_scope: comparison_scope(request.reference.is_some()),
99        peak_rss: if request.plan.memory_runs() > 0 {
100            "memory_rounds: separate_fresh_helper_RUSAGE_CHILDREN; excludes_helper; may_include_usage_propagated_by_waited_descendants; not_simultaneous_tree_RSS_or_device_memory; macOS_bytes_Linux_KiB_converted_to_bytes"
101        } else {
102            "unavailable: safe direct-child capture does not retain per-child rusage; no inference from logical allocation counters"
103        },
104        before,
105        after: Vec::new(),
106        metadata: Vec::new(),
107        samples: Vec::new(),
108        total_capture_bytes: 0,
109        faults: Vec::new(),
110        unresolved_children: Vec::new(),
111        destination,
112    };
113    match materialize(&corpus, &sources, &cases, directory.path(), request) {
114        Ok(sealed) => {
115            report.before.extend(sealed);
116            execute(
117                request,
118                &cases,
119                directory.path(),
120                Schedule {
121                    deadline,
122                    cancelled,
123                },
124                &mut report,
125                invocation,
126            )?;
127        }
128        Err(error) => {
129            report.faults.push(Fault::InputWrite(error.to_string()));
130            fill_unattempted(&mut report)?;
131        }
132    }
133    report.after = report.before.iter().map(identity::recheck).collect();
134    if let Err(error) = directory.close() {
135        report.faults.push(Fault::InputCleanup(error.to_string()));
136    }
137    report.finished_unix_ns = capture::unix_ns();
138    if report.finished_unix_ns.is_none() {
139        report.faults.push(Fault::Clock);
140    }
141    Ok(report)
142}

163/// Preserve the native, reference (when one takes part) and catalog prefix used
164/// by retained-report readers, then seal optional helpers and independent
165/// authored sources in the same campaign.
166fn seal_inputs(
167    corpus: &examples::Corpus,
168    sources: &BTreeSet<&str>,
169    request: &Request<'_>,
170    workloads: Option<&[Workload]>,
171) -> Result<Vec<crate::selected::FileSeal>, Error> {
172    let mut sealed = super::super::run::seals(
173        corpus,
174        sources,
175        request.corpus,
176        request.native,
177        request.reference.map(|reference| reference.executable),
178        request.limits,
179    )?;
180    if let Some(helper) = request.helper.filter(|_| request.plan.memory_runs() > 0) {
181        sealed.push(identity::seal(helper, request.limits.max_executable_bytes)?);
182    }
183    if let Some(workloads) = workloads {
184        sealed.extend(
185            workloads
186                .iter()
187                .filter_map(Workload::authored_seal)
188                .cloned(),
189        );
190    }
191    Ok(sealed)
192}

194struct Prepared<'a> {
195    input: Input<'a>,
196    directory: PathBuf,
197}
198
199/// Where a cell's program comes from: a sealed corpus entry, unchanged or
200/// through a workload that amends it, or an independently sealed authored/generated program.
201enum Input<'a> {
202    Corpus {
203        case: &'a examples::Case,
204        workload: Option<&'a Workload>,
205    },
206    Independent(&'a Workload),
207}
208impl<'a> Input<'a> {
209    fn path(&self) -> &str {
210        match self {
211            Self::Corpus { case, .. } => case.path(),
212            Self::Independent(workload) => workload.entry(),
213        }
214    }
215    /// Corpus files this input reads; a generated program reads none.
216    fn corpus_source_paths(&self) -> impl Iterator<Item = &str> {
217        match self {
218            Self::Corpus { case, .. } => case.transitive_source_paths().iter(),
219            Self::Independent(_) => [].iter(),
220        }
221        .map(String::as_str)
222    }
223    /// The workload the input runs through, when it runs through one.
224    fn workload(&self) -> Option<&'a Workload> {
225        match self {
226            Self::Corpus { workload, .. } => *workload,
227            Self::Independent(workload) => Some(workload),
228        }
229    }
230    /// The contract the campaign checks: the corpus contract of an
231    /// unchanged entry, and the workload's otherwise.
232    fn contract(&self) -> Option<&'a examples::Contract> {
233        match self {
234            Self::Corpus {
235                case,
236                workload: None,
237            } => Some(case.contract()),
238            Self::Corpus {
239                workload: Some(workload),
240                ..
241            }
242            | Self::Independent(workload) => workload.contract(),
243        }
244    }
245}

247fn prepare<'a>(
248    corpus: &'a examples::Corpus,
249    request: &Request<'_>,
250    workloads: Option<&'a [Workload]>,
251) -> Result<Vec<Prepared<'a>>, Error> {
252    if request.plan.suite() == Suite::Scalability && workloads.is_none() {
253        return Err(Error::Configuration(
254            "scalability suite requires explicit authored workloads",
255        ));
256    }
257    let allowed = cases(corpus, &request.plan)?;
258    let Some(workloads) = workloads else {
259        return Ok(allowed
260            .into_iter()
261            .map(|original| Prepared {
262                input: Input::Corpus {
263                    case: original,
264                    workload: None,
265                },
266                directory: PathBuf::new(),
267            })
268            .collect());
269    };
270    if workloads.is_empty()
271        || workloads.len() > request.limits.corpus.cases.min(super::config::MAX_CASES)
272    {
273        return Err(Error::Configuration(
274            "explicit workload population must fit 1..=94",
275        ));
276    }
277    let mut metadata = 0usize;
278    let mut sources = 0usize;
279    for (position, workload) in workloads.iter().enumerate() {
280        if workloads[..position]
281            .iter()
282            .any(|prior| prior.identity() == workload.identity())
283        {
284            return Err(Error::Configuration(
285                "explicit workloads repeat a content identity",
286            ));
287        }
288        workload.validate(corpus, request.limits)?;
289        metadata = metadata
290            .checked_add(workload.metadata_bytes)
291            .ok_or(Error::Configuration("workload metadata sum overflow"))?;
292        sources = sources
293            .checked_add(workload.source_bytes)
294            .ok_or(Error::Configuration("workload source sum overflow"))?;
295    }
296    if metadata > request.limits.corpus.manifest_bytes
297        || sources > request.limits.corpus.total_source_bytes
298    {
299        return Err(Error::Configuration(
300            "combined workloads exceed retained input ceilings",
301        ));
302    }
303    let mut prepared = Vec::new();
304    prepared
305        .try_reserve_exact(workloads.len())
306        .map_err(|_| Error::Configuration("workload population allocation failed"))?;
307    for (position, workload) in workloads.iter().enumerate() {
308        let input = if workload.is_generated() || workload.is_authored() {
309            Input::Independent(workload)
310        } else {
311            Input::Corpus {
312                case: allowed
313                    .iter()
314                    .find(|case| case.path() == workload.entry())
315                    .ok_or(Error::Configuration(
316                        "workload is outside the plan's allowed suite",
317                    ))?,
318                workload: Some(workload),
319            }
320        };
321        prepared.push(Prepared {
322            input,
323            directory: format!("workload-{position:02}").into(),
324        });
325    }
326    Ok(prepared)
327}

329fn materialize(
330    corpus: &examples::Corpus,
331    sources: &BTreeSet<&str>,
332    cases: &[Prepared<'_>],
333    directory: &Path,
334    request: &Request<'_>,
335) -> Result<Vec<crate::selected::FileSeal>, Error> {
336    if cases.iter().all(|case| case.input.workload().is_none()) {
337        return super::super::run::copy_sources(
338            corpus,
339            sources,
340            directory,
341            request.limits.corpus.source_bytes,
342        );
343    }
344    let mut sealed = Vec::new();
345    for case in cases {
346        let workload = case
347            .input
348            .workload()
349            .ok_or(Error::Configuration("mixed workload preparation"))?;
350        sealed.extend(workload.materialize(
351            corpus,
352            &directory.join(&case.directory),
353            request.limits.corpus.source_bytes,
354        )?);
355    }
356    Ok(sealed)
357}
358/// The plan's cases: its selection when it has one, each a case of its suite,
359/// and the whole suite otherwise.
360fn cases<'a>(corpus: &'a examples::Corpus, plan: &Plan) -> Result<Vec<&'a examples::Case>, Error> {
361    let suite: Vec<&examples::Case> = match plan.suite {
362        Suite::Corpus => corpus.cases().iter().collect(),
363        Suite::Baseline | Suite::Scalability => {
364            members(corpus, super::super::Suite::Baseline.cases())?
365        }
366        Suite::Queens => members(corpus, super::super::Suite::Queens.cases())?,
367        Suite::Series => members(corpus, &super::super::series::CORPUS_CASES)?,
368    };
369    let Some(selection) = &plan.selection else {
370        return Ok(suite);
371    };
372    selection
373        .iter()
374        .map(|path| {
375            suite
376                .iter()
377                .copied()
378                .find(|case| case.path() == path)
379                .ok_or(Error::Configuration(
380                    "a selected case is not a case of the plan's suite",
381                ))
382        })
383        .collect()
384}

386fn members<'a>(
387    corpus: &'a examples::Corpus,
388    cases: &[super::super::Case],
389) -> Result<Vec<&'a examples::Case>, Error> {
390    cases
391        .iter()
392        .map(|selected| {
393            corpus
394                .cases()
395                .iter()
396                .find(|case| case.path() == selected.path())
397                .ok_or(Error::Configuration(
398                    "selected case is missing from sealed corpus",
399                ))
400        })
401        .collect()
402}

428fn fill_unattempted(report: &mut Report) -> Result<(), Error> {
429    report.samples = report
430        .plan
431        .plan
432        .slots(report.cases.len(), report.plan.reference_policy.reference())?
433        .into_iter()
434        .map(|slot| unattempted(slot, None, "campaign setup prevented execution"))
435        .collect();
436    Ok(())

438fn capture_metadata(
439    request: &Request<'_>,
440    directory: &Path,
441    schedule: Schedule<'_>,
442    report: &mut Report,
443    invocation: NativeInvocation,
444) -> bool {
445    let reference = request
446        .reference
447        .map(|reference| (reference.executable, vec!["--version".into()]));
448    for (executable, arguments) in [
449        (request.native, vec!["--version".into()]),
450        (request.native, invocation.help_arguments()),
451    ]
452    .into_iter()
453    .chain(reference)
454    {
455        let Some((observed, _)) = invoke(executable, arguments, false, directory, schedule, report)
456        else {
457            return false;
458        };
459        let completed = observed.complete(false);
460        report.metadata.push(MatrixCapture(observed));
461        if !completed {
462            report.faults.push(Fault::Metadata);
463            return false;
464        }
465    }
466    true
467}
468fn execute(
469    request: &Request<'_>,
470    cases: &[Prepared<'_>],
471    directory: &Path,
472    schedule: Schedule<'_>,
473    report: &mut Report,
474    invocation: NativeInvocation,
475) -> Result<(), Error> {
476    if !capture_metadata(request, directory, schedule, report, invocation) {
477        fill_unattempted(report)?;
478        return Ok(());
479    }
480    let mut censuses: Vec<_> = (0..cases.len()).map(|_| Census::default()).collect();
481    let width = request.plan.profiles.len() + 1;
482    let mut blocked: Vec<Option<usize>> = vec![None; cases.len() * width];
483    let mut stopped = false;
484    let policy = request.reference.map(|reference| reference.policy);
485    for slot in request.plan.slots(cases.len(), policy)? {
486        stopped |= schedule.stopped(report);
487        let cell = slot.case * width + slot.producer.index();
488        let skipped = if stopped {
489            Some((None, "campaign scheduling stopped; no replacement launches"))
490        } else if let Some(previous) = blocked[cell] {
491            Some((
492                Some(previous),
493                "cell disabled by its first non-pass observation",
494            ))
495        } else if request.reference.is_some()
496            && slot.phase != Phase::Qualification
497            && censuses[slot.case].reference.is_none()
498        {
499            Some((
500                blocked[slot.case * width],
501                "reference census did not establish a complete family",
502            ))
503        } else {
504            None
505        };
506        if let Some((blocker, detail)) = skipped {
507            report.samples.push(unattempted(slot, blocker, detail));
508            continue;
509        }
510        let selected = &cases[slot.case];
511        if request.reference.is_none() && selected.input.contract().is_none() {
512            blocked[cell] = Some(report.samples.len());
513            report.samples.push(needs_clingo(slot));
514            continue;
515        }
516        let case_directory = directory.join(&selected.directory);
517        let (executable, arguments) = arguments(
518            request,
519            &case_directory,
520            selected.input.path(),
521            slot.producer,
522            invocation,
523        );
524        let record = case_directory.join("child-rss.json");
525        let launched = launch(
526            request,
527            slot.phase,
528            (executable, arguments),
529            &record,
530            &case_directory,
531            schedule,
532            report,
533        );
534        let Some((capture, curtailed)) = launched else {
535            stopped = true;
536            report.samples.push(unattempted(
537                slot,
538                None,
539                "campaign scheduling budget or cleanup prevented launch",
540            ));
541            continue;
542        };
543        let mut sample = Sample {
544            slot,
545            capture: Some(MatrixCapture(capture)),
546            decision: Decision::Pass,
547            detail: None,
548            blocked_by: None,
549            selected_models: None,
550            cost: None,
551            observation: None,
552            memory: None,
553        };
554        if slot.phase == Phase::Memory
555            && !cancelled_capture(sample.capture().unwrap())
556            && !measured(&mut sample, &record, report)
557        {
558            continue;
559        }
560        let contract = selected.input.contract();
561        if let Err((decision, detail)) =
562            censuses[slot.case].check(&mut sample, contract, request, curtailed)
563        {
564            sample.decision = decision;
565            sample.detail = Some(detail);
566            blocked[cell] = Some(report.samples.len());
567        }
568        stopped = !report.unresolved_children.is_empty();
569        report.samples.push(sample);
570    }
571    Ok(())
572}
573fn arguments<'a>(
574    request: &Request<'a>,
575    directory: &Path,
576    path: &str,
577    producer: Producer,
578    invocation: NativeInvocation,
579) -> (&'a Path, Vec<OsString>) {
580    let (executable, mut arguments): (_, Vec<OsString>) = match producer {
581        Producer::Reference => (
582            request
583                .reference
584                .expect("the schedule has reference positions only when a reference takes part")
585                .executable,
586            vec![
587                "--models=0".into(),
588                "--outf=2".into(),
589                "--opt-mode=optN".into(),
590                format!("--parallel-mode={}", request.plan.reference_workers).into(),
591                "--warn=none".into(),
592            ],
593        ),
594        Producer::Native { profile } => {
595            let profile = request.plan.profiles[profile];
596            (
597                request.native,
598                invocation
599                    .arguments(&profile)
600                    .into_iter()
601                    .chain(["--color".into(), "never".into()])
602                    .chain(["--json".into(), "--stats".into()])
603                    .collect(),
604            )
605        }
606    };
607    arguments.push(directory.join(path).into_os_string());
608    (executable, arguments)
609}

616/// One workload's completed answer baselines. Neither a failed reference nor a
617/// mismatching native sample can replace the first accepted census it violates.
618#[derive(Default)]
619struct Census {
620    reference: Option<answers::ReportedAnswers>,
621    native: Option<answers::native_json::NativeAnswers>,
622}
623
624impl Census {
625    fn check(
626        &mut self,
627        sample: &mut Sample,
628        contract: Option<&examples::Contract>,
629        request: &Request<'_>,
630        curtailed: Curtailed,
631    ) -> Result<(), (Decision, String)> {
632        let answer = qualify(
633            sample,
634            contract,
635            self.reference.as_ref(),
636            request,
637            curtailed,
638        )?;
639        if let Some(native) = answer.native {
640            super::native_family::accept(&mut self.native, native)?;
641        }
642        if sample.slot.phase == Phase::Qualification && sample.slot.producer == Producer::Reference
643        {
644            self.reference = Some(answer.display);
645        }
646        Ok(())
647    }
648}

650/// The decision a capture that stopped at a limit earns, or `None` for one
651/// that did not. A stop at a limit the campaign lowered is the campaign's;
652/// only a stop at the authored limit says something of the workload.
653fn limit_stop(capture: &Capture, curtailed: Curtailed) -> Option<(Decision, String)> {
654    match capture.stop() {
655        Some(process::Stop::Deadline) if curtailed.time => {
656            Some((Decision::CampaignDeadline, "campaign deadline".into()))
657        }
658        Some(process::Stop::Deadline) => Some((Decision::Timeout, "process deadline".into())),
659        Some(process::Stop::OutputLimit) if curtailed.bytes => Some((
660            Decision::CampaignCaptureBudget,
661            "campaign capture budget".into(),
662        )),
663        Some(process::Stop::OutputLimit) => {
664            Some((Decision::CaptureLimit, "capture byte ceiling".into()))
665        }
666        _ => None,
667    }
668}

670fn qualify(
671    sample: &mut Sample,
672    contract: Option<&examples::Contract>,
673    reference: Option<&answers::ReportedAnswers>,
674    request: &Request<'_>,
675    curtailed: Curtailed,
676) -> Result<Qualified, (Decision, String)> {
677    let capture = sample
678        .capture()
679        .expect("qualification follows a launched capture");
680    if cancelled_capture(capture) {
681        return Err((Decision::Cancelled, "campaign cancelled".into()));
682    }
683    if let Some(stop) = limit_stop(capture, curtailed) {
684        return Err(stop);
685    }
686    if capture.stop() != Some(process::Stop::Completed)
687        || capture.failure().is_some()
688        || capture.cleanup_failure().is_some()
689    {
690        return Err((
691            Decision::InvocationFailure,
692            "capture/start/cleanup did not complete".into(),
693        ));
694    }
695    let exit = solver_exit(sample)?;
696    let (parsed, native) = match sample.slot.producer {
697        Producer::Reference => {
698            if !exit.is_some_and(|exit| {
699                exit.signal.is_none() && matches!(exit.code, Some(0 | 10 | 20 | 30))
700            }) {
701                return Err((
702                    Decision::InvocationFailure,
703                    "reference exit did not complete".into(),
704                ));
705            }
706            (
707                answers::clingo_json(capture.stdout(), request.limits.answers)
708                    .map_err(|error| invalid(&error))?,
709                None,
710            )
711        }
712        Producer::Native { profile } => {
713            let document: Value = serde_json::from_slice(capture.stdout())
714                .map_err(|e| (Decision::InvalidReport, e.to_string()))?;
715            outcome::check(&document, exit)?;
716            let native = answers::native_json::parse(capture.stdout(), request.native_answers)
717                .map_err(|error| invalid(&error))?;
718            let display = native
719                .reported_displays(request.max_spelling_bytes)
720                .map_err(|error| invalid(&error))?;
721            sample.observation = Some(
722                super::Observation::from_statistics(
723                    &document,
724                    capture.stderr(),
725                    request.plan.profiles[profile],
726                )
727                .map_err(|e| (Decision::InvalidTelemetry, e))?,
728            );
729            (display, Some(native))
730        }
731    };
732    sample.selected_models = Some(parsed.model_count());
733    sample.cost = parsed.cost().map(<[i64]>::to_vec);
734    if let Some(contract) = contract {
735        contract
736            .check(&parsed)
737            .map_err(|e| (Decision::ParityMismatch, e.to_string()))?;
738    }
739    // With clingo taking part, its census is the authority; without it, the
740    // recorded contract checked above is, and a family without one is refused.
741    if let Some(reference) = reference {
742        if !answers::same_displays(reference, &parsed) {
743            return Err((
744                Decision::ParityMismatch,
745                "complete selected displays/counts/costs differ from qualified reference".into(),
746            ));
747        }
748    } else if request.reference.is_none() {
749        if contract.is_none() {
750            return Err((
751                Decision::NeedsClingo,
752                "no recorded contract; only clingo establishes this workload's family".into(),
753            ));
754        }
755    } else if sample.slot.producer != Producer::Reference
756        || sample.slot.phase != Phase::Qualification
757    {
758        return Err((
759            Decision::ReferenceUnavailable,
760            "no complete qualified reference family".into(),
761        ));
762    }
763    Ok(Qualified {
764        display: parsed,
765        native,
766    })
767}

769/// A memory capture belongs to the helper. Require its success separately,
770/// then use the measured solver exit for the producer's outcome contract.
771fn solver_exit(sample: &Sample) -> Result<Option<process::Exit>, (Decision, String)> {
772    let capture = sample.capture().expect("a launched sample has a capture");
773    if sample.slot.phase != Phase::Memory {
774        return Ok(capture.exit());
775    }
776    if !capture.complete(false) {
777        return Err((
778            Decision::InvocationFailure,
779            "memory helper did not complete successfully".into(),
780        ));
781    }
782    let memory = sample
783        .memory
784        .filter(|memory| memory.valid())
785        .ok_or_else(|| {
786            (
787                Decision::InvalidMemory,
788                "memory round lacks a valid solver resource record".into(),
789            )
790        })?;
791    Ok(Some(process::Exit {
792        code: memory.exit_code,
793        signal: memory.signal,
794    }))
795}
796fn invalid(error: &answers::Error) -> (Decision, String) {
797    (Decision::InvalidReport, error.to_string())
798}
799/// Launch a slot's solver: directly, or on a memory round as the helper's
800/// child, the helper writing the child's resource record to `record`.
801fn launch(
802    request: &Request<'_>,
803    phase: Phase,
804    (executable, arguments): (&Path, Vec<OsString>),
805    record: &Path,
806    directory: &Path,
807    schedule: Schedule<'_>,
808    report: &mut Report,
809) -> Option<(Capture, Curtailed)> {
810    if phase != Phase::Memory {
811        return invoke(executable, arguments, false, directory, schedule, report);
812    }
813    let helper = request
814        .helper
815        .expect("memory rounds admitted with a helper");
816    let mut supervised: Vec<OsString> = vec![
817        crate::process::memory::HELPER_COMMAND.into(),
818        record.as_os_str().to_owned(),
819        executable.as_os_str().to_owned(),
820    ];
821    supervised.extend(arguments);
822    invoke(helper, supervised, true, directory, schedule, report)
823}

825/// Read a memory round's resource record into its sample. A missing or
826/// contradictory record decides the sample and is retained with a fault;
827/// the campaign goes on, since the rounds are independent invocations.
828fn measured(sample: &mut Sample, record: &Path, report: &mut Report) -> bool {
829    let helper_child = sample.capture().and_then(Capture::helper_child_id);
830    match super::super::run::memory::read(record, helper_child).1 {
831        Ok(measurement) => {
832            sample.memory = Some(measurement);
833            true
834        }
835        Err(detail) => {
836            sample.decision = Decision::InvalidMemory;
837            sample.detail = Some(detail);
838            report.faults.push(Fault::Observation);
839            report.samples.push(std::mem::replace(
840                sample,
841                unattempted(sample.slot, None, ""),
842            ));
843            false
844        }
845    }
846}

848/// Which of a launch's limits the campaign's remainder set below the authored
849/// one, so that a stop at that limit is the campaign's, not the workload's.
850#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
851struct Curtailed {
852    /// The campaign's remaining time was shorter than the process timeout.
853    time: bool,
854    /// The campaign's remaining capture budget was smaller than the ceiling.
855    bytes: bool,
856}
857
858/// The limits of one launch: each authored limit, lowered to the `time` and
859/// `bytes` the campaign has left, with which of them the campaign lowered.
860fn within_campaign(
861    authored: process::Limits,
862    time: Duration,
863    bytes: usize,
864) -> (process::Limits, Curtailed) {
865    let curtailed = Curtailed {
866        time: time < authored.timeout,
867        bytes: bytes < authored.max_output_bytes,
868    };
869    let limits = process::Limits {
870        timeout: authored.timeout.min(time),
871        max_output_bytes: authored.max_output_bytes.min(bytes),
872        ..authored
873    };
874    (limits, curtailed)
875}
876
877/// Launch one bounded process within the campaign's remaining time and
878/// capture, reporting which limits the campaign lowered; a supervised launch
879/// is the memory helper's, whose child is reaped separately.
880fn invoke(
881    executable: &Path,
882    arguments: Vec<OsString>,
883    supervised: bool,
884    directory: &Path,
885    schedule: Schedule<'_>,
886    report: &mut Report,
887) -> Option<(Capture, Curtailed)> {
888    if schedule.stopped(report) {
889        return None;
890    }
891    let time = schedule.deadline.saturating_duration_since(Instant::now());
892    if time.is_zero() {
893        report.faults.push(Fault::Deadline);
894        return None;
895    }
896    let bytes = report
897        .limits
898        .max_total_capture_bytes
899        .saturating_sub(report.total_capture_bytes);
900    if bytes == 0 {
901        report.faults.push(Fault::CaptureBudget);
902        return None;
903    }
904    let (limits, curtailed) = within_campaign(report.limits.process, time, bytes);
905    let (capture, fault) = if supervised {
906        capture::supervised_with_cancellation(
907            executable,
908            arguments,
909            directory,
910            limits,
911            schedule.cancelled,
912        )
913    } else {
914        capture::invoke_with_cancellation(
915            executable,
916            arguments,
917            directory,
918            limits,
919            schedule.cancelled,
920        )
921    };
922    if cancelled_capture(&capture) {
923        record_cancellation(report);
924    }
925    report.total_capture_bytes += capture.stdout().len() + capture.stderr().len();
926    if let Some(fault) = fault {
927        report.faults.push(Fault::ChildCleanup(fault));
928    }
929    if let Some(id) = capture.unresolved_child {
930        report.unresolved_children.push(id);
931    }
932    Some((capture, curtailed))
933}
934
935fn cancelled_capture(capture: &Capture) -> bool {
936    capture.stop() == Some(process::Stop::Cancelled)
937        || capture
938            .failure()
939            .is_some_and(|failure| failure.kind() == crate::selected::InvocationFault::Cancelled)
940}
941
942fn record_cancellation(report: &mut Report) {
943    if !report
944        .faults
945        .iter()
946        .any(|fault| matches!(fault, Fault::Cancelled))
947    {
948        report.faults.push(Fault::Cancelled);
949    }
950}
