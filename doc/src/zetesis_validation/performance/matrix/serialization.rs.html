1//! Borrowed lossless stream views keep matrix publication outside measured solves.
2//!
3//! UTF-8 validation scans each stream once per serialization pass, without copying
4//! its bytes. Bounded publication preflights and then writes the same view.
5//! Serde writes the borrowed text or byte sequence through the existing bounded
6//! publication writer. The legacy capture serializer remains the ordinary
7//! campaign's.
8use std::ffi::OsString;
9use std::path::Path;
10
11use serde::{Serialize, Serializer};
12
13use super::super::Capture;
14use crate::{process, selected::InvocationFailure};
15
16#[derive(Serialize)]
17#[serde(tag = "encoding", content = "data", rename_all = "snake_case")]
18enum Stream<'a> {
19    Utf8(&'a str),
20    Bytes(&'a [u8]),
21}
22impl<'a> Stream<'a> {
23    fn new(bytes: &'a [u8]) -> Self {
24        match std::str::from_utf8(bytes) {
25            Ok(text) => Self::Utf8(text),
26            Err(_) => Self::Bytes(bytes),
27        }
28    }
29}
30
31#[derive(Serialize)]
32struct View<'a> {
33    executable: &'a Path,
34    arguments: &'a [OsString],
35    directory: &'a Path,
36    started_unix_ns: Option<u128>,
37    elapsed_ns: Option<u128>,
38    stop: Option<process::Stop>,
39    exit: Option<process::Exit>,
40    stdout: Stream<'a>,
41    stderr: Stream<'a>,
42    failure: Option<&'a InvocationFailure>,
43    cleanup_failure: Option<&'a InvocationFailure>,
44    unresolved_child: Option<u32>,
45    #[serde(skip_serializing_if = "Option::is_none")]
46    helper_child_id: Option<u32>,
47}
48impl<'a> View<'a> {
49    fn new(capture: &'a Capture) -> Self {
50        Self {
51            executable: capture.executable(),
52            arguments: capture.arguments(),
53            directory: capture.directory(),
54            started_unix_ns: capture.started_unix_ns(),
55            elapsed_ns: capture.elapsed_ns(),
56            stop: capture.stop(),
57            exit: capture.exit(),
58            stdout: Stream::new(capture.stdout()),
59            stderr: Stream::new(capture.stderr()),
60            failure: capture.failure(),
61            cleanup_failure: capture.cleanup_failure(),
62            unresolved_child: capture.unresolved_child,
63            helper_child_id: capture.helper_child_id(),
64        }
65    }
66}
67
68/// A launched process's capture as a matrix report records it: serialized as
69/// the lossless stream view, each stream as its UTF-8 text or, failing that,
70/// its bytes.
71#[derive(Debug)]
72pub(super) struct MatrixCapture(pub(super) Capture);
73impl Serialize for MatrixCapture {
74    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
75        View::new(&self.0).serialize(serializer)
76    }
77}
