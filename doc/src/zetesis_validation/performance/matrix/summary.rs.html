9/// Count of one actual disposition, including qualifications and skipped slots.

11pub struct DecisionCount {
12    /// Actual outcome; failures never enter successful timing populations.
13    pub decision: Decision,
14    /// Fixed schedule positions with this outcome.
15    pub positions: usize,

18/// What a cell's answer family was qualified against.
19#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
20#[serde(rename_all = "snake_case")]
21pub enum Qualification {
22    /// Clingo's complete census, which took part in the campaign.
23    Clingo,
24    /// The workload's recorded contract, in a clingo-free campaign.
25    Contract,
26    /// Nothing: a clingo-free campaign cannot qualify a workload without a
27    /// recorded contract, so its cells were not launched.
28    NeedsClingo,
29}

31/// One source/producer cell's compact observations, without raw answer streams.
32#[derive(Clone, Debug, Serialize)]
33pub struct CellSummary {
34    /// Index into the report's unchanged source-entry list.
35    pub case: usize,
36    /// Reference solver or requested native profile index.
37    pub producer: Producer,
38    /// What the cell's answer family was qualified against.
39    pub qualification: Qualification,
40    /// Counts over every requested phase, including unlaunched positions.
41    pub decisions: Vec<DecisionCount>,
42    /// Non-pass explanations grouped by schedule phase and original blocker.
43    /// Details come from retained samples; missing legacy reasons stay explicit.
44    pub reasons: BTreeMap<String, usize>,
45    /// Available only when every timed position passed and its duration fits u64.
46    pub timing: Option<Timing>,
47    /// Median of successfully validated separate memory rounds, or unavailable.
48    pub peak_rss_bytes: Option<u64>,
49}

51/// Compact typed campaign view; input identities and profiles remain borrowed.
52/// Computing it scans retained positions and copies durations, counts and
53/// failure details. It never serializes or decodes raw captured answer streams.
54#[derive(Debug, Serialize)]
55pub struct Summary<'a> {
56    /// Version of this compact view, distinct from full campaign report schemas.
57    pub schema: u32,
58    /// Stable machine-view identifier.
59    pub format: &'static str,
60    /// Every campaign requirement qualified.
61    pub passed: bool,
62    /// Every fixed schedule position has a retained disposition.
63    pub accounted: bool,
64    /// Source entry paths in fixed case order.
65    pub cases: &'a [String],
66    /// Exact generated or constant-amended workload identities, when selected.
67    pub workloads: Option<&'a [super::Workload]>,
68    /// Exact requested native profiles.
69    pub profiles: &'a [crate::selected::NativeExecution],
70    /// Whether the reference was measured or used only for its census, or
71    /// took no part (`clingo_free`).
72    pub reference_policy: super::RecordedPolicy,
73    /// Sealed executable and source identities.
74    pub before: &'a [crate::selected::FileSeal],
75    /// Source-major, reference-first compact cells; a clingo-free campaign has
76    /// no reference cells.
77    pub cells: Vec<CellSummary>,
78}

80/// What a case's cells were qualified against: clingo's census when it took
81/// part, the recorded contract otherwise, and nothing for a workload without one.
82fn qualification(report: &Report, case: usize) -> Qualification {
83    if report.plan.reference_policy.reference().is_some() {
84        return Qualification::Clingo;
85    }
86    match report.workloads.as_deref() {
87        Some(workloads) if workloads[case].contract().is_none() => Qualification::NeedsClingo,
88        _ => Qualification::Contract,
89    }
90}

92pub(super) fn summarize(report: &Report) -> Summary<'_> {
93    let mut cells = Vec::new();
94    let reference = report
95        .plan
96        .reference_policy
97        .reference()
98        .map(|_| Producer::Reference);
99    for case in 0..report.cases.len() {
100        let qualification = qualification(report, case);
101        for producer in reference
102            .into_iter()
103            .chain((0..report.plan.plan.profiles.len()).map(|profile| Producer::Native { profile }))
104        {
105            let mut decisions: Vec<DecisionCount> = Vec::new();
106            let mut reasons = BTreeMap::new();
107            let mut intervals = Some(Vec::new());
108            let mut memory = Vec::new();
109            for sample in report
110                .samples
111                .iter()
112                .filter(|sample| sample.slot.case == case && sample.slot.producer == producer)
113            {
114                if let Some(count) = decisions
115                    .iter_mut()
116                    .find(|count| count.decision == sample.decision)
117                {
118                    count.positions += 1;
119                } else {
120                    decisions.push(DecisionCount {
121                        decision: sample.decision,
122                        positions: 1,
123                    });
124                }
125                if sample.decision != Decision::Pass {
126                    *reasons.entry(reason(sample, &report.samples)).or_insert(0) += 1;
127                }
128                if sample.slot.phase == Phase::Timed {
129                    let duration = (sample.decision == Decision::Pass)
130                        .then(|| sample.capture()?.elapsed_ns()?.try_into().ok())
131                        .flatten();
132                    match (&mut intervals, duration) {
133                        (Some(intervals), Some(duration)) => intervals.push(duration),
134                        _ => intervals = None,
135                    }
136                }
137                if sample.slot.phase == Phase::Memory && sample.decision == Decision::Pass {
138                    memory.extend(sample.memory.map(|memory| memory.peak_rss_bytes));
139                }
140            }
141            let timing = intervals.and_then(|mut values| {
142                let median_ns = median(&mut values)?;
143                Some(Timing {
144                    samples: values.len(),
145                    minimum_ns: values[0],
146                    median_ns,
147                    maximum_ns: values[values.len() - 1],
148                })
149            });
150            cells.push(CellSummary {
151                case,
152                producer,
153                qualification,
154                decisions,
155                reasons,
156                timing,
157                peak_rss_bytes: median(&mut memory),
158            });
159        }
160    }
161    Summary {
162        schema: 1,
163        format: "zetesis_benchmark_summary",
164        passed: report.passed(),
165        accounted: report.accounted(),
166        cases: &report.cases,
167        workloads: report.workloads.as_deref(),
168        profiles: &report.plan.plan.profiles,
169        reference_policy: report.plan.reference_policy,
170        before: &report.before,
171        cells,
172    }
173}

175fn reason(sample: &super::Sample, samples: &[super::Sample]) -> String {
176    let detail = sample
177        .detail
178        .as_deref()
179        .unwrap_or("reason unavailable in retained sample");
180    let mut reason = format!(
181        "{}: {}: {detail}",
182        sample.slot.phase.label(),
183        sample.decision.label()
184    );
185    if let Some(index) = sample.blocked_by {
186        use std::fmt::Write;
187        if let Some(blocker) = samples.get(index) {
188            write!(
189                reason,
190                "; blocked by sample {index} ({}, {}): {}",
191                blocker.slot.phase.label(),
192                blocker.decision.label(),
193                blocker
194                    .detail
195                    .as_deref()
196                    .unwrap_or("reason unavailable in retained sample"),
197            )
198            .expect("writing to a String cannot fail");
199        } else {
200            write!(reason, "; blocking sample {index} unavailable")
201                .expect("writing to a String cannot fail");
202        }
203    }
204    reason
205}
