10pub(super) fn observe(
11    document: &Value,
12    stderr: &[u8],
13    request: NativeExecution,
14) -> Result<Observation, String> {
15    let timing = super::super::timing::parse_any(stderr)?;
16    let hybrid = hybrid::read(document)?;
17    let terminal = terminal::read(document)?;
18    let hybrid_terminal = terminal_route(terminal.as_ref(), hybrid.is_some(), request)?;
19    // An explicit request names the mode the cell must have taken; an
20    // automatic request accepts either mode and retains the one observed.
21    let taken = match timing.grounding_mode.as_str() {
22        "eager" => Grounder::Eager,
23        "lazy_interleaved" => Grounder::Lazy,
24        "mixed" if hybrid.is_some() => Grounder::Lazy,
25        "eager_base_terminal_definitions" if terminal.is_some() && !hybrid_terminal => {
26            Grounder::Eager
27        }
28        "hybrid_base_terminal_definitions" if hybrid_terminal && hybrid.is_some() => Grounder::Lazy,
29        _ => return Err("unsupported reported grounding mode".into()),
30    };
31    if request.grounder != Grounder::Auto && taken != request.grounder {
32        return Err("reported grounding differs from requested matrix cell".into());
33    }
34    let statistics = &document["statistics"];
35    if !statistics.is_object()
36        || statistics["stage_timings"]["grounding_mode"] != timing.grounding_mode.as_str()
37    {
38        return Err("typed statistics and authored stage view disagree".into());
39    }
40    consistent_timings(statistics, &timing)?;
41    let text = std::str::from_utf8(stderr).map_err(|e| e.to_string())?;
42    let routes: Vec<_> = text
43        .lines()
44        .filter_map(|line| line.strip_prefix("Backend: "))
45        .collect();
46    let [route] = routes.as_slice() else {
47        return Err("missing or ambiguous actual backend metadata".into());
48    };
49    // `gpu` asks for the platform's native API; what ran is the concrete one.
50    let (backend, adapter) = reported_backend(route)?;
51    if backend.resolved_api() != request.backend.resolved_api() {
52        return Err("actual backend differs from requested matrix cell".into());
53    }
54    let effective = one(
55        text.lines()
56            .filter_map(|line| line.trim_start().strip_prefix("effective execution: ")),
57        "effective execution",
58    )?;
59    let effective_backend = field(effective, "backend")?;
60    let grounding = if hybrid_terminal {
61        "hybrid_base_terminal_definitions"
62    } else if terminal.is_some() {
63        "eager_base_terminal_definitions"
64    } else if hybrid.is_some() {
65        "hybrid"
66    } else {
67        taken.label()
68    };
69    if field(effective, "grounder")? != grounding {
70        return Err("effective execution and measured grounding disagree".into());
71    }
72    let procedure = procedure(effective, request.oracle)?;
73    let hybrid_mode = if hybrid_terminal {
74        "hybrid_base_terminal_definitions"
75    } else {
76        "mixed"
77    };
78    if hybrid.is_some() && (timing.grounding_mode != hybrid_mode || procedure == Procedure::Closure)
79    {
80        return Err(
81            "hybrid source checking requires mixed grounding and formula membership".into(),
82        );
83    }
84    if let Some(receipt) = &terminal {
85        terminal::execution(receipt, &timing, procedure)?;
86    }
87    if matches!(
88        procedure,
89        Procedure::PositiveConsequences | Procedure::StratifiedConsequences
90    ) && (backend != Backend::Cpu || (taken != Grounder::Eager && hybrid.is_none()))
91    {
92        return Err("direct consequences require the eager CPU formula route".into());
93    }
94    let device = device(
95        statistics,
96        route,
97        effective_backend,
98        // An authenticated hybrid receipt establishes eager core membership
99        // followed by host acceptance. Device counters describe that core, not
100        // the original answer population checked by `hybrid::read`.
101        if hybrid.is_some() {
102            Grounder::Eager
103        } else {
104            taken
105        },
106        backend,
107        procedure,
108        adapter.as_deref(),
109    )?;
110    Ok(Observation {
111        timing,
112        hybrid,
113        terminal,
114        execution: Execution {
115            backend,
116            procedure,
117            adapter,
118            device,
119        },
120    })
121}

123fn procedure(effective: &str, request: Oracle) -> Result<Procedure, String> {
124    let procedure = match field(effective, "oracle")? {
125        "closure" => Procedure::Closure,
126        "countermodel" => Procedure::Countermodel,
127        "tight-support" => Procedure::TightSupport,
128        "positive-consequences" => Procedure::PositiveConsequences,
129        "stratified-consequences" => Procedure::StratifiedConsequences,
130        _ => return Err("unsupported actual oracle metadata".into()),
131    };
132    let matches_request = match request {
133        Oracle::Auto => true,
134        Oracle::Closure => procedure == Procedure::Closure,
135        Oracle::Countermodel => procedure == Procedure::Countermodel,
136    };
137    if !matches_request {
138        return Err("actual oracle differs from explicit requested procedure".into());
139    }
140    Ok(procedure)
141}

143/// Decode the hardware actually named by the unique backend record. Requested
144/// policy and effective execution are reconciled separately by the caller.
145fn reported_backend(route: &str) -> Result<(Backend, Option<String>), String> {
146    if route.starts_with("cpu") {
147        return Ok((Backend::Cpu, None));
148    }
149    let prefix = route
150        .strip_prefix("gpu (")
151        .or_else(|| route.strip_prefix("hybrid GPU propagation + exact CPU residual search ("))
152        .or_else(|| route.strip_prefix("GPU tight support ("))
153        .ok_or("unsupported actual backend metadata")?;
154    let (identity, _) = prefix
155        .split_once("; vendor=")
156        .ok_or("missing actual device vendor")?;
157    let (adapter, api) = identity
158        .rsplit_once(", ")
159        .ok_or("missing actual device API")?;
160    let api = match api {
161        "Metal" => GpuApi::Metal,
162        "Vulkan" => GpuApi::Vulkan,
163        _ => return Err("actual device API is neither Metal nor Vulkan".into()),
164    };
165    if adapter.is_empty() {
166        return Err("empty actual device name".into());
167    }
168    Ok((Backend::Gpu(Some(api)), Some(adapter.to_owned())))
169}

171fn one<'a>(mut values: impl Iterator<Item = &'a str>, label: &str) -> Result<&'a str, String> {
172    let value = values.next().ok_or_else(|| format!("missing {label}"))?;
173    if values.next().is_some() {
174        return Err(format!("ambiguous {label}"));
175    }
176    Ok(value)
177}
178fn field<'a>(line: &'a str, name: &str) -> Result<&'a str, String> {
179    let prefix = format!("{name}=");
180    one(
181        line.split("; ")
182            .filter_map(|field| field.strip_prefix(&prefix)),
183        name,
184    )
185}
186fn device(
187    statistics: &Value,
188    route: &str,
189    effective: &str,
190    grounder: Grounder,
191    backend: Backend,
192    procedure: Procedure,
193    adapter: Option<&str>,
194) -> Result<DeviceWork, String> {
195    let lazy_stats = statistics
196        .get("lazy_execution")
197        .ok_or("missing lazy activity field")?;
198    let formula_stats = statistics
199        .get("execution")
200        .ok_or("missing formula activity field")?;
201    if backend == Backend::Cpu {
202        let expected = if formula_stats.is_null() {
203            "cpu"
204        } else {
205            "cpu batched exact completion"
206        };
207        if effective != expected {
208            return Err("primary and effective CPU routes disagree".into());
209        }
210        return cpu(statistics);
211    }
212    let api = backend.resolved_api().ok_or("missing actual device API")?;
213    if procedure == Procedure::Closure {
214        if effective != "requested GPU policy"
215            || !route.starts_with("gpu (")
216            || !formula_stats.is_null()
217        {
218            return Err("primary/effective closure route or activity disagrees".into());
219        }
220        if grounder == Grounder::Lazy {
221            if !route.ends_with("; lazy immutable reduct rounds)") || !lazy_stats.is_object() {
222                return Err("lazy GPU route requires its own activity record and route".into());
223            }
224            return lazy(lazy_stats, api, adapter);
225        }
226        if !route.contains("; static atoms=") || !lazy_stats.is_null() {
227            return Err("static closure route has conflicting lazy activity".into());
228        }
229        return Ok(DeviceWork::Unavailable {
230            reason: "static closure driver does not accumulate dispatch/transfer counters",
231        });
232    }
233    if route.starts_with("GPU tight support (") || effective == "GPU tight support" {
234        if grounder != Grounder::Eager
235            || procedure != Procedure::TightSupport
236            || effective != "GPU tight support"
237            || !route.starts_with("GPU tight support (")
238            || !formula_stats.is_object()
239            || !lazy_stats.is_null()
240        {
241            return Err("tight GPU route requires its own activity record and route".into());
242        }
243        return tight(formula_stats, api, adapter);
244    }
245    if grounder != Grounder::Eager
246        || effective != "hybrid GPU propagation + exact CPU residual search"
247        || !route.starts_with("hybrid GPU propagation + exact CPU residual search (")
248        || !formula_stats.is_object()
249        || !lazy_stats.is_null()
250    {
251        return Err("hybrid GPU route requires its own activity record and route".into());
252    }
253    formula(formula_stats, api, adapter)
254}

256fn number(value: &Value, key: &str) -> Result<u64, String> {
257    value[key]
258        .as_u64()
259        .ok_or_else(|| format!("missing or non-u64 execution counter: {key}"))
260}
261fn lazy(value: &Value, api: GpuApi, adapter: Option<&str>) -> Result<DeviceWork, String> {
262    let submitted = number(value, "submitted_candidates")?;
263    let completed = number(value, "completed_candidates")?;
264    if value["backend"] != api.name()
265        || value["adapter"].as_str() != adapter
266        || number(value, "stopped_candidates")? != 0
267        || number(value, "queued_results")? != 0
268        || completed != submitted
269    {
270        return Err("lazy device completion/identity contradicts exhausted coverage".into());
271    }
272    let dispatches = number(value, "dispatches")?;
273    let world_instances = number(value, "world_instances")?;
274    let uploaded_bytes = number(value, "uploaded_bytes")?;
275    let downloaded_bytes = number(value, "downloaded_bytes")?;
276    if (dispatches == 0 && (world_instances != 0 || uploaded_bytes != 0 || downloaded_bytes != 0))
277        || (dispatches > 0
278            && (world_instances == 0 || uploaded_bytes == 0 || downloaded_bytes == 0))
279    {
280        return Err("lazy dispatch and transfer counts disagree".into());
281    }
282    Ok(DeviceWork::Lazy {
283        dispatches,
284        world_instances,
285        uploaded_bytes,
286        downloaded_bytes,
287        completed_candidates: completed,
288    })
289}
290fn formula(value: &Value, api: GpuApi, adapter: Option<&str>) -> Result<DeviceWork, String> {
291    if !value["tight_work_per_candidate"].is_null() || !value["gpu_scheduled_work"].is_null() {
292        return Err("propagation route reports tight-support work".into());
293    }
294    let activity = formula_activity(value, api, adapter)?;
295    Ok(DeviceWork::Formula {
296        batches: activity.batches,
297        candidates: activity.candidates,
298        work: activity.work,
299        cpu_residuals: activity.residuals,
300        gpu_residuals: residual_reasons(value, activity.residuals)?,
301        peak_accounted_bytes: activity.peak_bytes,
302    })
303}

305fn residual_reasons(
306    value: &Value,
307    expected: u64,
308) -> Result<Option<FormulaResidualStatistics>, String> {
309    let reasons = &value["gpu_residuals"];
310    if reasons.is_null() {
311        return Ok(None);
312    }
313    let reasons = FormulaResidualStatistics {
314        fixed_point: number(reasons, "fixed_point")?,
315        round_limit: number(reasons, "round_limit")?,
316        work_limit: number(reasons, "work_limit")?,
317    };
318    // Complete observations already reconcile every decoded candidate with a
319    // committed device or CPU verdict. Completion attempts can include retries
320    // and are deliberately not the population counted by this device receipt.
321    if reasons
322        .fixed_point
323        .checked_add(reasons.round_limit)
324        .and_then(|sum| sum.checked_add(reasons.work_limit))
325        != Some(expected)
326    {
327        return Err("decoded residual reasons contradict complete formula work".into());
328    }
329    Ok(Some(reasons))
330}

340fn formula_activity(
341    value: &Value,
342    api: GpuApi,
343    adapter: Option<&str>,
344) -> Result<FormulaActivity, String> {
345    let adapter = adapter.ok_or("missing actual formula adapter")?;
346    if !value["adapter"].as_str().is_some_and(|reported| {
347        reported.starts_with(&format!("{adapter}, {}; vendor=", api.name()))
348    }) {
349        return Err("formula adapter disagrees with actual route".into());
350    }
351    let candidates = number(value, "gpu_candidates")?;
352    let residuals = number(value, "cpu_residuals")?;
353    let batches = number(value, "gpu_batches")?;
354    if number(value, "gpu_decided")?.checked_add(residuals) != Some(candidates)
355        || number(value, "pending_candidates")? != 0
356        || number(value, "queued_models")? != 0
357        || (batches == 0) != (candidates == 0)
358        || batches > candidates
359    {
360        return Err("formula device accounting contradicts exhausted coverage".into());
361    }
362    completion(&value["completion"], candidates, residuals)?;
363    Ok(FormulaActivity {
364        batches,
365        candidates,
366        work: number(value, "gpu_work")?,
367        residuals,
368        peak_bytes: number(value, "peak_accounted_bytes")?,
369    })
370}

372fn tight(value: &Value, api: GpuApi, adapter: Option<&str>) -> Result<DeviceWork, String> {
373    let activity = formula_activity(value, api, adapter)?;
374    let submitted_batches = number(value, "gpu_submitted_batches")?;
375    let submitted_candidates = number(value, "gpu_submitted_candidates")?;
376    let scheduled_work = number(value, "gpu_scheduled_work")?;
377    let work_per_candidate_limit = number(value, "tight_work_per_candidate")?;
378    if !value["gpu_residuals"].is_null()
379        || value.get("gpu_limits") != Some(&Value::Null)
380        || number(value, "gpu_rounds")? != 0
381        || activity.residuals != 0
382        || number(&value["completion"], "residuals")? != 0
383        || submitted_batches < activity.batches
384        || submitted_candidates < activity.candidates
385        || (submitted_batches == 0) != (submitted_candidates == 0)
386        || submitted_batches > submitted_candidates
387        || activity.work > scheduled_work
388        || u128::from(scheduled_work)
389            > u128::from(submitted_candidates) * u128::from(work_per_candidate_limit)
390        || u128::from(activity.work)
391            > u128::from(activity.candidates) * u128::from(work_per_candidate_limit)
392    {
393        return Err("tight support counters or limits contradict the reported primitive".into());
394    }
395    // One immutable plan performs the same complete scan in every candidate.
396    // With no decoded candidate, submitted work remains a scheduling receipt;
397    // it must not be relabelled as completed work or propagation sweeps.
398    if activity.candidates > 0
399        && (activity.work % activity.candidates != 0
400            || u128::from(scheduled_work) * u128::from(activity.candidates)
401                != u128::from(activity.work) * u128::from(submitted_candidates))
402    {
403        return Err("tight support scan work disagrees across candidate counts".into());
404    }
405    Ok(DeviceWork::TightSupport {
406        batches: activity.batches,
407        candidates: activity.candidates,
408        work: activity.work,
409        submitted_batches,
410        submitted_candidates,
411        scheduled_work,
412        work_per_candidate_limit,
413        peak_accounted_bytes: activity.peak_bytes,
414    })
415}

417/// Attempt counters may include uncommitted work and retries. Their `complete`
418/// field certifies only that diagnostic aggregation did not overflow.
419fn completion(value: &Value, candidates: u64, committed_residuals: u64) -> Result<(), String> {
420    let entered = number(value, "entered")?;
421    let completed = number(value, "completed")?;
422    let failed = number(value, "failed")?;
423    let residuals = number(value, "residuals")?;
424    let residual_completed = number(value, "residual_completed")?;
425    let residual_failed = number(value, "residual_failed")?;
426    if value["complete"] != true
427        || completed.checked_add(failed) != Some(entered)
428        || residual_completed.checked_add(residual_failed) != Some(residuals)
429        || residuals > entered
430        || residual_completed > completed
431        || residual_failed > failed
432        || completed < candidates
433        || residual_completed < committed_residuals
434        || completed - residual_completed < candidates - committed_residuals
435    {
436        return Err("completion attempts contradict committed formula work".into());
437    }
438    Ok(())
439}

441fn consistent_timings(
442    statistics: &Value,
443    timing: &super::super::Diagnostics,
444) -> Result<(), String> {
445    let phases = &statistics["phase_timings"];
446    // The original unversioned text profile also admitted typed records without
447    // a schema field. Newer profiles require their exact explicit version.
448    let schema_matches = match phases.get("schema") {
449        None => timing.phase_schema == 1,
450        Some(schema) => schema.as_u64() == Some(u64::from(timing.phase_schema)),
451    };
452    if !schema_matches
453        || phases["measurements"].as_object().map(serde_json::Map::len) != Some(timing.phases.len())
454    {
455        return Err("typed and text phase schemas disagree".into());
456    }
457    for (section, measurements) in [
458        ("stage_timings", &timing.stages),
459        ("phase_timings", &timing.phases),
460    ] {
461        let section = &statistics[section];
462        if section["driver_elapsed_ns"].as_u64() != Some(timing.driver_elapsed_ns) {
463            return Err("typed and text driver clocks disagree".into());
464        }
465        for (name, expected) in measurements {
466            let actual = section["measurements"]
467                .get(name)
468                .ok_or("missing typed phase/stage measurement")?;
469            match expected {
470                None if actual.is_null() => {}
471                Some(expected)
472                    if actual["calls"].as_u64() == Some(expected.calls)
473                        && actual["elapsed_ns"].as_u64() == Some(expected.elapsed_ns)
474                        && actual["complete"] == true => {}
475                _ => return Err("typed and text phase/stage measurements disagree".into()),
476            }
477        }
478    }
479    if statistics["stage_timings"]["complete"] != true
480        || statistics["stage_timings"]["unattributed_elapsed_ns"].as_u64()
481            != Some(timing.unattributed_elapsed_ns)
482    {
483        return Err("typed and text stage completeness disagree".into());
484    }
485    Ok(())
486}

491fn cpu(statistics: &Value) -> Result<DeviceWork, String> {
492    if !statistics["lazy_execution"].is_null() {
493        return Err("CPU route contradicts lazy device evidence".into());
494    }
495    let execution = &statistics["execution"];
496    if !execution.is_null() {
497        if !execution["adapter"].is_null()
498            || !execution["gpu_residuals"].is_null()
499            || !execution["tight_work_per_candidate"].is_null()
500            || !execution["gpu_scheduled_work"].is_null()
501        {
502            return Err(
503                "CPU route reports device identity, residual reasons or tight-support work".into(),
504            );
505        }
506        for counter in [
507            "gpu_batches",
508            "gpu_candidates",
509            "gpu_work",
510            "gpu_rounds",
511            "gpu_decided",
512            "peak_accounted_bytes",
513            "pending_candidates",
514            "queued_models",
515        ] {
516            if number(execution, counter)? != 0 {
517                return Err("CPU route reports GPU work".into());
518            }
519        }
520        let residuals = number(execution, "cpu_residuals")?;
521        completion(&execution["completion"], residuals, residuals)?;
522    }
523    Ok(DeviceWork::Cpu)
524}

526/// Whether a terminal receipt is of a hybrid base, once its route agrees with
527/// the request: an eager base is automatic grounding's, with no source
528/// checking; a hybrid one is lazy (or automatic) grounding's, with both receipts.
529fn terminal_route(
530    terminal: Option<&super::TerminalStatistics>,
531    hybrid: bool,
532    request: NativeExecution,
533) -> Result<bool, String> {
534    match terminal {
535        Some(receipt) if receipt.hybrid_base => {
536            if !hybrid || request.grounder == Grounder::Eager {
537                return Err(
538                    "a hybrid terminal base requires lazy grounding and its checking receipt"
539                        .into(),
540                );
541            }
542            Ok(true)
543        }
544        Some(_) if hybrid || request.grounder != Grounder::Auto => {
545            Err("terminal definitions require automatic grounding and a distinct route".into())
546        }
547        Some(_) | None => Ok(false),
548    }
