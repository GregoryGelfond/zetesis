1//! A complete source-checking receipt authenticates the mixed grounding route.
2
3use super::number;
4use crate::performance::matrix::{HybridStatistics, RegionChecks};
5use serde_json::Value;
6
7pub(super) fn read(document: &Value) -> Result<Option<HybridStatistics>, String> {
8    let statistics = &document["statistics"];
9    let receipt = &statistics["hybrid_execution"];
10    if receipt.is_null() {
11        if statistics["search"]["scope"] == "retained_core" {
12            return Err("retained-core search lacks its source-checking receipt".into());
13        }
14        return Ok(None);
15    }
16    let constraints = &receipt["constraints"];
17    let parsed = HybridStatistics {
18        regions: region_checks(&statistics["search"]["region_filter"])?,
19        core_answers: number(receipt, "core_answers")?,
20        accepted: number(receipt, "accepted")?,
21        rejected: number(receipt, "rejected")?,
22        pending: number(receipt, "pending")?,
23        work: number(constraints, "work")?,
24        substitutions: number(constraints, "substitutions")?,
25        scalar_bytes: number(constraints, "scalar_bytes")?,
26    };
27    if parsed.pending != 0
28        || parsed.accepted.checked_add(parsed.rejected) != Some(parsed.core_answers)
29        || statistics["search"]["scope"] != "retained_core"
30        || number(&statistics["search"], "stable_models")? != parsed.core_answers
31        || number(&document["outcome"], "verified_models")? != parsed.accepted
32    {
33        return Err(
34            "hybrid core/checker counts contradict complete original-program acceptance".into(),
35        );
36    }
37    Ok(Some(parsed))
38}
39
40fn region_checks(receipt: &Value) -> Result<Option<RegionChecks>, String> {
41    if receipt.is_null() {
42        return Ok(None);
43    }
44    let parsed = RegionChecks {
45        preparations: number(receipt, "preparations")?,
46        checks: number(receipt, "checks")?,
47        refuted: number(receipt, "refuted")?,
48    };
49    if receipt["overflowed"].as_bool() != Some(false)
50        || number(receipt, "failed")? != 0
51        || parsed.refuted > parsed.checks
52        || (parsed.checks != 0 && parsed.preparations == 0)
53    {
54        return Err("region restriction receipt does not establish completed checking".into());
55    }
56    Ok(Some(parsed))
57}
