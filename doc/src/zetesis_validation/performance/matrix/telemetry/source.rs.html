3use super::{field, number, one};
4use crate::performance::Diagnostics;
5use crate::performance::matrix::{
6    DeviceWork, Execution, Observation, Procedure, SourceEvaluationStatistics,
7};
8use crate::selected::{Backend, Grounder, NativeExecution, Oracle};
9use serde_json::Value;

11const ROUTE: &str =
12    "cpu (dependency-ordered source evaluation; constraints checked before publication)";
13const OPERATION: &str = "direct source evaluation";
14
15pub(super) fn read(
16    statistics: &Value,
17    stderr: &[u8],
18    request: NativeExecution,
19    timing: &Diagnostics,
20) -> Result<Option<SourceEvaluationStatistics>, String> {
21    let text = std::str::from_utf8(stderr).map_err(|error| error.to_string())?;
22    let effective = one(
23        text.lines()
24            .filter_map(|line| line.trim_start().strip_prefix("effective execution: ")),
25        "effective execution",
26    )?;
27    let receipt = &statistics["source_evaluation"];
28    if receipt.is_null() {
29        // Older records have no field or a null one. An operation-labelled
30        // route is new evidence and cannot silently acquire an old oracle.
31        if has_field(effective, "operation")
32            || text
33                .lines()
34                .any(|line| line.strip_prefix("Backend: ") == Some(ROUTE))
35        {
36            return Err("source operation lacks its typed admission receipt".into());
37        }
38        return Ok(None);
39    }
40    let parsed = receipt_statistics(receipt)?;
41    let route = one(
42        text.lines()
43            .filter_map(|line| line.strip_prefix("Backend: ")),
44        "actual backend metadata",
45    )?;
46    if request.backend != Backend::Cpu
47        || request.grounder != Grounder::Auto
48        || request.oracle != Oracle::Auto
49        || route != ROUTE
50        || field(effective, "backend")? != "cpu"
51        || field(effective, "operation")? != OPERATION
52        || effective.split("; ").count() != 2
53    {
54        return Err("source evaluation requires its automatic CPU operation metadata".into());
55    }
56    activity(statistics, timing)?;
57    let reported = one(
58        text.lines()
59            .filter_map(|line| line.trim_start().strip_prefix("source evaluation: ")),
60        "source evaluation work",
61    )?;
62    if field(reported, "scope")? != "source_admission"
63        || field(reported, "work")?.parse::<u64>().ok() != Some(parsed.work)
64        || field(reported, "substitutions")?.parse::<u64>().ok() != Some(parsed.substitutions)
65    {
66        return Err("typed and text source admission receipts disagree".into());
67    }
68    Ok(Some(parsed))
69}
70
71fn receipt_statistics(value: &Value) -> Result<SourceEvaluationStatistics, String> {
72    if value.as_object().map(serde_json::Map::len) != Some(3)
73        || value["scope"] != "source_admission"
74    {
75        return Err("unknown source evaluation receipt shape or scope".into());
76    }
77    Ok(SourceEvaluationStatistics {
78        work: number(value, "work")?,
79        substitutions: number(value, "substitutions")?,
80    })
81}

83fn activity(statistics: &Value, timing: &Diagnostics) -> Result<(), String> {
84    for field in [
85        "execution",
86        "lazy_execution",
87        "shared_execution",
88        "closure_execution",
89        "query_execution",
90        "hybrid_execution",
91        "terminal_execution",
92        "search",
93        "candidate_restrictions",
94    ] {
95        if statistics.get(field) != Some(&Value::Null) {
96            return Err(format!(
97                "source evaluation has conflicting or missing {field} evidence"
98            ));
99        }
100    }
101    if number(statistics, "objective_work")? != 0
102        || timing.grounding_mode != "eager"
103        || !timing.stages.get("grounding").is_some_and(|value| {
104            value
105                .as_ref()
106                .is_some_and(|measurement| measurement.calls > 0)
107        })
108        || !timing
109            .phases
110            .get("admission_materialization")
111            .is_some_and(|value| {
112                value
113                    .as_ref()
114                    .is_some_and(|measurement| measurement.calls > 0)
115            })
116    {
117        return Err("source evaluation lacks eager admission or reports objective work".into());
118    }
119    for (name, value) in &timing.phases {
120        if value.is_some()
121            && !matches!(
122                name.as_str(),
123                "admission_materialization"
124                    | "execution_setup"
125                    | "model_construction"
126                    | "observation_output"
127            )
128        {
129            return Err("source evaluation reports conflicting formula or search phases".into());
130        }
131    }
132    Ok(())
133}
134
135fn has_field(line: &str, name: &str) -> bool {
136    line.split("; ")
137        .any(|field| field.split_once('=').is_some_and(|(key, _)| key == name))
138}

140pub(super) fn observation(receipt: SourceEvaluationStatistics, timing: Diagnostics) -> Observation {
141    Observation {
142        timing,
143        execution: Execution {
144            backend: Backend::Cpu,
145            procedure: Procedure::AcyclicSource,
146            adapter: None,
147            device: DeviceWork::Cpu,
148        },
149        hybrid: None,
150        terminal: None,
151        source_evaluation: Some(receipt),
152    }
153}
