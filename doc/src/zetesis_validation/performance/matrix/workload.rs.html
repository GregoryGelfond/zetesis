18/// One requested integer constant replacement; borrowed metadata is caller-owned.
19#[derive(Clone, Copy, Debug)]
20pub struct ConstantAmendment<'a> {
21    /// Entry/include path relative to the admitted corpus.
22    pub source_path: &'a str,
23    /// Exact declared constant name.
24    pub name: &'a str,
25    /// Required original nonnegative decimal i32 literal.
26    pub expected: i32,
27    /// Replacement i32, rendered as decimal without inserting other syntax.
28    pub replacement: i32,
29}

31/// Inclusive admission bounds for one source workload.
32///
33/// Source bytes bound parser input and derived content, not syntax-tree
34/// allocation or RSS. Metadata bytes count retained provenance text and cost
35/// values; vector/object counts are bounded by source and amendment populations.
36#[derive(Clone, Copy, Debug)]
37pub struct WorkloadLimits {
38    /// Input or derived bytes in one source.
39    pub source_bytes: usize,
40    /// Combined input plus derived closure bytes inspected during admission.
41    pub total_source_bytes: usize,
42    /// Retained path, digest, edit and default-contract payload bytes.
43    pub metadata_bytes: usize,
44    /// Requested replacements, also bounded by 128.
45    pub amendments: usize,
46}
47
48impl Default for WorkloadLimits {
49    fn default() -> Self {
50        Self {
51            source_bytes: 1_048_576,
52            total_source_bytes: 4_194_304,
53            metadata_bytes: 65_536,
54            amendments: MAX_AMENDMENTS,
55        }
56    }
57}

59/// Validated constant-only derivation of one complete admitted source closure,
60/// an authored singleton, or one generated program.
61///
62/// No source payload is retained. Materialization repeats the checked derivation
63/// or generation and seals the bytes actually launched. Authored sources retain
64/// their own file seal, independent of the correctness catalog. The unchanged corpus
65/// contract remains provenance for amended workloads; a generated workload's
66/// contract is its family's closed-form count.
67#[derive(Clone, Debug, Serialize)]
68pub struct Workload {
69    entry: String,
70    identity: String,
71    /// Corpus identity of a corpus workload; absent for independent inputs.
72    #[serde(skip_serializing_if = "Option::is_none")]
73    manifest_sha256: Option<&'static str>,
74    sources: Vec<Source>,
75    default_contract: examples::Contract,
76    amended: bool,
77    #[serde(skip_serializing_if = "Option::is_none")]
78    generated: Option<Generated>,
79    /// An authored singleton source is sealed independently of the campaign's
80    /// correctness catalog. It has no upstream-cleaning or generated-family claim.
81    #[serde(skip_serializing_if = "Option::is_none")]
82    authored: Option<authored::Authored>,
83    pub(super) metadata_bytes: usize,
84    pub(super) source_bytes: usize,
85}

87/// Provenance of a generated program: the family, the size and the digest of
88/// the bytes the generator produced when the workload was admitted.
89#[derive(Clone, Debug, Serialize)]
90struct Generated {
91    family: Family,
92    size: u32,
93    sha256: String,
94    bytes: usize,
95}
96
97impl Generated {
98    /// The entry path of a generated program of the family and size.
99    fn entry_of(family: Family, size: u32) -> String {
100        format!("generated/{}-{size}.lp", family.label())
101    }
102
103    /// The entry path of the generated program.
104    fn entry(&self) -> String {
105        Self::entry_of(self.family, self.size)
106    }
107
108    /// The program the generator produces now, when it is the one admitted:
109    /// the same length and digest.
110    ///
111    /// # Errors
112    /// Refuses a size the family no longer admits, or a program other than
113    /// the admitted one.
114    fn regenerate(&self) -> Result<String, Error> {
115        let source = self
116            .family
117            .source(self.size)
118            .map_err(|_| Error::Configuration("generated workload size is not admitted"))?;
119        if source.len() != self.bytes
120            || format!("{:x}", Sha256::digest(source.as_bytes())) != self.sha256
121        {
122            return Err(Error::Configuration(
123                "generated program differs from its admitted program",
124            ));
125        }
126        Ok(source)
127    }
128}

139impl Workload {
140    /// Validate an unchanged entry and its parsed include closure.
141    ///
142    /// # Errors
143    /// Refuses unknown entries, syntax/include discrepancies or admission bounds.
144    pub fn original(
145        corpus: &examples::Corpus,
146        entry: &str,
147        limits: WorkloadLimits,
148    ) -> Result<Self, Error> {
149        Self::amended(corpus, entry, &[], limits)
150    }
151
152    /// Validate literal constant replacements through the pinned syntax parser.
153    ///
154    /// Only unannotated nonnegative decimal i32 source literals are admitted.
155    /// Each named declaration must occur once in its specified source; repeated
156    /// requests for a declaration are refused. Replacements cannot alter includes.
157    /// Parser input and all derived content are bounded before retention.
158    /// No-op edits retain their provenance but keep the unchanged contract.
159    ///
160    /// # Errors
161    /// Refuses unknown paths/names, duplicate requests, unsupported declarations,
162    /// unexpected literal values, syntax/include discrepancies and resource bounds.
163    pub fn amended(
164        corpus: &examples::Corpus,
165        entry: &str,
166        amendments: &[ConstantAmendment<'_>],
167        limits: WorkloadLimits,
168    ) -> Result<Self, Error> {
169        let case = corpus
170            .cases()
171            .iter()
172            .find(|case| case.path() == entry)
173            .ok_or(Error::Configuration(
174                "workload entry is not in the admitted corpus",
175            ))?;
176        if amendments.len() > limits.amendments.min(MAX_AMENDMENTS) {
177            return Err(Error::Configuration(
178                "workload amendment count exceeds its ceiling",
179            ));
180        }
181        let mut requested = 0usize;
182        for (position, amendment) in amendments.iter().enumerate() {
183            requested = add(requested, amendment.source_path.len())?;
184            requested = add(requested, amendment.name.len())?;
185            if requested > limits.metadata_bytes
186                || !case
187                    .transitive_source_paths()
188                    .iter()
189                    .any(|path| path == amendment.source_path)
190                || amendments[..position].iter().any(|prior| {
191                    prior.source_path == amendment.source_path && prior.name == amendment.name
192                })
193            {
194                return Err(Error::Configuration(
195                    "workload amendment metadata is invalid or exceeds its ceiling",
196                ));
197            }
198        }
199        let mut budget = Budget {
200            limits,
201            metadata: 0,
202            sources: 0,
203        };
204        budget.metadata(std::mem::size_of_val(
205            case.contract().cost().unwrap_or_default(),
206        ))?;
207        for text in case
208            .contract()
209            .witnesses()
210            .iter()
211            .flatten()
212            .chain(case.contract().required_symbols())
213            .chain(case.contract().notes())
214        {
215            budget.metadata(text.len())?;
216        }
217        let default_contract = case.contract().clone();
218        let entry = budget.text(entry)?;
219        let mut sources = reserve(case.transitive_source_paths().len())?;
220        for path in case.transitive_source_paths() {
221            let source = corpus
222                .files()
223                .iter()
224                .find(|source| source.path() == path)
225                .ok_or(Error::Configuration(
226                    "workload source closure is incomplete",
227                ))?;
228            budget.source(source.source().len())?;
229            let edits = constants::edits(source, amendments, &mut budget)?;
230            let remaining = limits.total_source_bytes - budget.sources;
231            let derived = examples::derive_source(
232                source.source(),
233                &edits,
234                limits.source_bytes.min(remaining),
235            )
236            .map_err(Error::Corpus)?;
237            budget.source(derived.len())?;
238            let digest = format!("{:x}", Sha256::digest(derived.as_bytes()));
239            sources.push(Source {
240                path: budget.text(path)?,
241                base_sha256: budget.text(source.source_sha256())?,
242                derived_sha256: budget.text(&digest)?,
243                derived_bytes: derived.len(),
244                edits,
245            });
246        }
247        let identity = budget.text(&corpus_identity(&entry, corpus.manifest_sha256(), &sources))?;
248        let amended = sources
249            .iter()
250            .any(|source| source.base_sha256 != source.derived_sha256);
251        Ok(Self {
252            entry,
253            identity,
254            manifest_sha256: Some(corpus.manifest_sha256()),
255            amended,
256            sources,
257            default_contract,
258            generated: None,
259            authored: None,
260            metadata_bytes: budget.metadata,
261            source_bytes: budget.sources,
262        })
263    }

265    /// Admit the generated program of `family` at `size` as a workload.
266    ///
267    /// The entry path is `generated/<family>-<size>.lp`; the program's bytes
268    /// are bounded by the source ceiling and its digest is retained, so that
269    /// materialization can check that the generator still produces them.
270    ///
271    /// # Errors
272    /// Refuses a size outside the family's range or a program above the
273    /// source ceiling.
274    pub fn generated(family: Family, size: u32, limits: WorkloadLimits) -> Result<Self, Error> {
275        let source = family
276            .source(size)
277            .map_err(|_| Error::Configuration("generated workload size is not admitted"))?;
278        let contract = family
279            .contract(size)
280            .map_err(|_| Error::Configuration("generated workload size is not admitted"))?;
281        let mut budget = Budget {
282            limits,
283            metadata: 0,
284            sources: 0,
285        };
286        budget.source(source.len())?;
287        let digest = format!("{:x}", Sha256::digest(source.as_bytes()));
288        let entry = budget.text(&Generated::entry_of(family, size))?;
289        let mut hash = Sha256::new();
290        hash.update(b"zetesis-generated-workload-v1\0");
291        field(&mut hash, family.label().as_bytes());
292        field(&mut hash, &size.to_le_bytes());
293        field(&mut hash, digest.as_bytes());
294        let identity = budget.text(&format!("{:x}", hash.finalize()))?;
295        let sha256 = budget.text(&digest)?;
296        Ok(Self {
297            entry,
298            identity,
299            manifest_sha256: None,
300            sources: Vec::new(),
301            default_contract: contract,
302            amended: false,
303            generated: Some(Generated {
304                family,
305                size,
306                sha256,
307                bytes: source.len(),
308            }),
309            authored: None,
310            metadata_bytes: budget.metadata,
311            source_bytes: budget.sources,
312        })
313    }

373    pub(super) fn validate(
374        &self,
375        corpus: &examples::Corpus,
376        limits: super::super::Limits,
377    ) -> Result<(), Error> {
378        if self.is_authored() {
379            return self.validate_authored(limits);
380        }
381        if let Some(generated) = &self.generated {
382            generated.regenerate()?;
383            if self.entry != generated.entry()
384                || generated.bytes > limits.corpus.source_bytes
385                || self.metadata_bytes > limits.corpus.manifest_bytes
386            {
387                return Err(Error::Configuration(
388                    "generated workload differs from its admitted program",
389                ));
390            }
391            return Ok(());
392        }
393        let case = corpus
394            .cases()
395            .iter()
396            .find(|case| case.path() == self.entry)
397            .ok_or(Error::Configuration(
398                "workload entry differs from the request corpus",
399            ))?;
400        if self.manifest_sha256 != Some(corpus.manifest_sha256())
401            || self.default_contract != *case.contract()
402            || self.sources.len() != case.transitive_source_paths().len()
403            || self.metadata_bytes > limits.corpus.manifest_bytes
404        {
405            return Err(Error::Configuration(
406                "workload provenance differs from the request corpus",
407            ));
408        }
409        for (source, path) in self.sources.iter().zip(case.transitive_source_paths()) {
410            let original = corpus
411                .files()
412                .iter()
413                .find(|original| original.path() == path)
414                .ok_or(Error::Configuration("workload base source is missing"))?;
415            if source.path != *path
416                || source.base_sha256 != original.source_sha256()
417                || source.derived_bytes > limits.corpus.source_bytes
418            {
419                return Err(Error::Configuration(
420                    "workload source identity or size differs",
421                ));
422            }
423        }
424        Ok(())
425    }

427    pub(super) fn materialize(
428        &self,
429        corpus: &examples::Corpus,
430        directory: &Path,
431        limit: usize,
432    ) -> Result<Vec<crate::selected::FileSeal>, Error> {
433        if self.is_authored() {
434            return self.materialize_authored(directory, limit);
435        }
436        if let Some(generated) = &self.generated {
437            let source = generated.regenerate()?;
438            let path = directory.join(&self.entry);
439            let parent = path
440                .parent()
441                .ok_or(Error::Configuration("generated workload has no parent"))?;
442            std::fs::create_dir_all(parent).map_err(|error| super::super::io(parent, error))?;
443            std::fs::write(&path, source).map_err(|error| super::super::io(&path, error))?;
444            return Ok(vec![super::super::run::checked_seal(
445                &path,
446                limit,
447                &generated.sha256,
448            )?]);
449        }
450        let mut sealed = reserve(self.sources.len())?;
451        for source in &self.sources {
452            let original = corpus
453                .files()
454                .iter()
455                .find(|original| original.path() == source.path)
456                .ok_or(Error::Configuration("workload base source is missing"))?;
457            let derived = examples::derive_source(original.source(), &source.edits, limit)
458                .map_err(Error::Corpus)?;
459            if derived.len() != source.derived_bytes
460                || format!("{:x}", Sha256::digest(derived.as_bytes())) != source.derived_sha256
461            {
462                return Err(Error::Configuration(
463                    "workload derivation differs from its admitted hash",
464                ));
465            }
466            let path = directory.join(&source.path);
467            let parent = path
468                .parent()
469                .ok_or(Error::Configuration("workload source has no parent"))?;
470            std::fs::create_dir_all(parent).map_err(|error| super::super::io(parent, error))?;
471            std::fs::write(&path, derived).map_err(|error| super::super::io(&path, error))?;
472            sealed.push(super::super::run::checked_seal(
473                &path,
474                limit,
475                &source.derived_sha256,
476            )?);
477        }
478        Ok(sealed)
479    }

482// Both typed campaign summaries and retained-report views use the same label
483// format. The latter supplies decoded edits without rebuilding a Workload.
484pub(crate) fn workload_label<'a>(
485    entry: &str,
486    generated: bool,
487    edits: impl IntoIterator<Item = (&'a str, &'a str)>,
488) -> String {
489    let path = entry.strip_suffix(".lp").unwrap_or(entry);
490    if generated {
491        return path.rsplit('/').next().unwrap_or(path).to_owned();
492    }
493    let components: Vec<&str> = path.rsplit('/').take(2).collect();
494    let mut label = components.into_iter().rev().collect::<Vec<_>>().join("/");
495    for (before, after) in edits {
496        label.push(' ');
497        label.push_str(before);
498        label.push('→');
499        label.push_str(after);
500    }
501    label
502}

504fn corpus_identity(entry: &str, manifest_sha256: &str, sources: &[Source]) -> String {
505    let mut hash = Sha256::new();
506    hash.update(b"zetesis-workload-v1\0");
507    field(&mut hash, entry.as_bytes());
508    field(&mut hash, manifest_sha256.as_bytes());
509    for source in sources {
510        field(&mut hash, source.path.as_bytes());
511        field(&mut hash, source.base_sha256.as_bytes());
512        field(&mut hash, source.derived_sha256.as_bytes());
513    }
514    format!("{:x}", hash.finalize())
515}
516
517fn field(hash: &mut Sha256, value: &[u8]) {
518    hash.update((value.len() as u128).to_le_bytes());
519    hash.update(value);
520}
521
522fn add(left: usize, right: usize) -> Result<usize, Error> {
523    left.checked_add(right)
524        .ok_or(Error::Configuration("workload byte count overflow"))
525}
526
527fn reserve<T>(count: usize) -> Result<Vec<T>, Error> {
528    let mut result = Vec::new();
529    result
530        .try_reserve_exact(count)
531        .map_err(|_| Error::Configuration("workload allocation failed"))?;
532    Ok(result)
533}
534
535struct Budget {
536    limits: WorkloadLimits,
537    metadata: usize,
538    sources: usize,
539}
540impl Budget {
541    fn metadata(&mut self, bytes: usize) -> Result<(), Error> {
542        let next = add(self.metadata, bytes)?;
543        if next > self.limits.metadata_bytes {
544            return Err(Error::Configuration(
545                "workload metadata exceeds its byte ceiling",
546            ));
547        }
548        self.metadata = next;
549        Ok(())
550    }
551    fn text(&mut self, text: &str) -> Result<String, Error> {
552        self.metadata(text.len())?;
553        let mut result = String::new();
554        result
555            .try_reserve_exact(text.len())
556            .map_err(|_| Error::Configuration("workload text allocation failed"))?;
557        result.push_str(text);
558        Ok(result)
559    }
560    fn source(&mut self, bytes: usize) -> Result<(), Error> {
561        let next = add(self.sources, bytes)?;
562        if bytes > self.limits.source_bytes || next > self.limits.total_source_bytes {
563            return Err(Error::Configuration(
564                "workload source exceeds its byte ceiling",
565            ));
566        }
567        self.sources = next;
568        Ok(())
569    }
