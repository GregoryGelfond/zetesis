7pub(super) fn validate(entry: &str, files: &[BundleFile]) -> Result<(), Error> {
8    examples::Files::relative(entry).map_err(Error::Corpus)?;
9    if files.is_empty() || files.len() > 256 {
10        return Err(Error::Configuration(
11            "bundle requires 1..=256 source members",
12        ));
13    }
14    let mut edges = 0usize;
15    for (position, file) in files.iter().enumerate() {
16        examples::Files::relative(&file.path).map_err(Error::Corpus)?;
17        if files[..position]
18            .iter()
19            .any(|prior| prior.path == file.path)
20            || file.sha256.len() != 64
21            || !file
22                .sha256
23                .bytes()
24                .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
25        {
26            return Err(Error::Configuration(
27                "bundle member is repeated or its digest is invalid",
28            ));
29        }
30        edges = super::super::add(edges, file.includes.len())?;
31        if edges > 1024 {
32            return Err(Error::Configuration(
33                "bundle exceeds 1024 include occurrences",
34            ));
35        }
36        for include in &file.includes {
37            examples::Files::relative(&include.target).map_err(Error::Corpus)?;
38            if target(&file.path, &include.spelling)? != Path::new(&include.target) {
39                return Err(Error::Configuration(
40                    "bundle include differs from its declared relative target",
41                ));
42            }
43        }
44    }
45    let mut active = Vec::new();
46    let mut heights = vec![0; files.len()];
47    visit(entry, files, &mut active, &mut heights)?;
48    if heights.contains(&0) {
49        return Err(Error::Configuration(
50            "bundle has unreachable source members",
51        ));
52    }
53    Ok(())
54}

56// Every recursive call consumes one of 32 path positions; active membership
57// forbids cycles. Repeated edges still resolve, while completed DAG branches
58// reuse their established subtree height, preserving the longest-path bound.
59fn visit(
60    name: &str,
61    files: &[BundleFile],
62    active: &mut Vec<usize>,
63    heights: &mut [usize],
64) -> Result<usize, Error> {
65    let index = files
66        .iter()
67        .position(|file| file.path == name)
68        .ok_or(Error::Configuration(
69            "bundle target is not a declared member",
70        ))?;
71    if active.contains(&index) {
72        return Err(Error::Configuration("bundle include cycle"));
73    }
74    if active.len() >= 32 {
75        return Err(Error::Configuration("bundle include depth exceeds 32"));
76    }
77    if heights[index] > 0 {
78        return Ok(heights[index]);
79    }
80    active.push(index);
81    let mut height = 1;
82    for include in &files[index].includes {
83        height = height.max(1 + visit(&include.target, files, active, heights)?);
84    }
85    active.pop();
86    if height > 32 {
87        return Err(Error::Configuration("bundle include depth exceeds 32"));
88    }
89    heights[index] = height;
90    Ok(height)
91}

93fn target(including: &str, spelling: &str) -> Result<PathBuf, Error> {
94    let path = Path::new(spelling);
95    if spelling.is_empty()
96        || spelling.contains(['\\', ':'])
97        || spelling.split('/').any(str::is_empty)
98    {
99        return Err(Error::Configuration(
100            "bundle include spelling is not portable",
101        ));
102    }
103    let mut target = Path::new(including)
104        .parent()
105        .unwrap_or(Path::new(""))
106        .to_path_buf();
107    for part in path.components() {
108        match part {
109            Component::Normal(value) => target.push(value),
110            Component::CurDir => {}
111            Component::ParentDir if target.pop() => {}
112            _ => return Err(Error::Configuration("bundle include escapes its root")),
113        }
114    }
115    Ok(target)
116}

118pub(super) fn resolve(root: &Path, files: &[BundleFile]) -> Result<(), Error> {
119    let root = examples::Files::canonical(root).map_err(Error::Corpus)?;
120    for file in files {
121        let including = root.join(&file.path);
122        let parent = including
123            .parent()
124            .ok_or(Error::Configuration("bundle member has no parent"))?;
125        for include in &file.includes {
126            let expected =
127                examples::Files::confined(&root, &include.target).map_err(Error::Corpus)?;
128            let local = examples::Files::canonical(&parent.join(&include.spelling))
129                .map_err(Error::Corpus)?;
130            if local != expected {
131                return Err(Error::Configuration(
132                    "bundle include aliases an undeclared target",
133                ));
134            }
135            let cwd = root.join(&include.spelling);
136            match cwd.canonicalize() {
137                Ok(actual) if actual != expected => {
138                    return Err(Error::Configuration(
139                        "cwd include shadows the declared bundle target",
140                    ));
141                }
142                Ok(_) => {}
143                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
144                Err(error) => return Err(crate::performance::io(&cwd, error)),
145            }
146        }
147    }
148    Ok(())
