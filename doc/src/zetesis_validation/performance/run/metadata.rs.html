1//! Bounded producer configuration evidence, outside measured solves.
2use super::{Fault, Report, Request, invoke};
3use std::path::Path;
4use std::time::Instant;
5
6pub(super) fn collect(
7    request: &Request<'_>,
8    directory: &Path,
9    deadline: Instant,
10    report: &mut Report,
11) -> bool {
12    let native_help = if request.schedule.extended() {
13        "--help"
14    } else {
15        "--help-all"
16    };
17    for (executable, argument) in [
18        (request.native, "--version"),
19        (request.native, native_help),
20        (request.reference, "--version"),
21    ] {
22        if !query(executable, argument, directory, deadline, report) {
23            return false;
24        }
25        if argument == "--help" {
26            let capture = report.metadata.last().expect("successful query retained");
27            let advertised = full_help(capture.stdout())
28                .and_then(|stdout| full_help(capture.stderr()).map(|stderr| stdout || stderr));
29            match advertised {
30                Ok(true) if !query(executable, "--help-all", directory, deadline, report) => {
31                    return false;
32                }
33                Err(_) => {
34                    report.faults.push(Fault::Metadata);
35                    return false;
36                }
37                Ok(_) => {}
38            }
39        }
40    }
41    if request.schedule.extended()
42        && !query(request.reference, "--help", directory, deadline, report)
43    {
44        return false;
45    }
46    report.metadata_complete = true;
47    true
48}
49
50fn query(
51    executable: &Path,
52    argument: &str,
53    directory: &Path,
54    deadline: Instant,
55    report: &mut Report,
56) -> bool {
57    let Some(capture) = invoke(
58        executable,
59        vec![argument.into()],
60        directory,
61        deadline,
62        report,
63    ) else {
64        return false;
65    };
66    let complete = capture.complete(false);
67    report.metadata.push(capture);
68    if !complete {
69        report.faults.push(Fault::Metadata);
70    }
71    complete
72}

74fn full_help(bytes: &[u8]) -> Result<bool, std::str::Utf8Error> {
75    let text = std::str::from_utf8(bytes)?;
76    let mut clean = String::new();
77    let mut characters = text.chars().peekable();
78    while let Some(character) = characters.next() {
79        if character == '\u{1b}' && characters.peek() == Some(&'[') {
80            characters.next();
81            for character in characters.by_ref() {
82                if ('@'..='~').contains(&character) {
83                    break;
84                }
85            }
86        } else {
87            clean.push(character);
88        }
89    }
90    Ok(clean.split_whitespace().any(|word| word == "--help-all"))
91}
