24/// Reviewed metadata for the two default scalability fixtures.
25pub const MANIFEST_SHA256: &str =
26    "eabb3ca0161beca0816f7438439818aa4920465b979c9618c764f3c3b3bb1870";
27const EINSTEIN_SHA256: &str = "d142a0b2f515954e6d4bbceeebac7f25f87f040ebc72897049c666dd9db1652b";
28
29#[derive(Deserialize)]
30#[serde(deny_unknown_fields)]
31struct Manifest {
32    schema_version: u32,
33    description: String,
34    reference_toolchain: String,
35    cases: Vec<Fixture>,
36}
37
38#[derive(Deserialize)]
39#[serde(deny_unknown_fields)]
40struct Fixture {
41    path: String,
42    scaling_constant: String,
43    default_value: i32,
44    contract: examples::Contract,
45    source_sha256: String,
46}
47
48fn fixtures(root: &Path, limits: WorkloadLimits) -> Result<Vec<Fixture>, Error> {
49    let root = examples::files::canonical(root).map_err(Error::Corpus)?;
50    let path =
51        examples::files::confined(&root, "scalability/manifest.json").map_err(Error::Corpus)?;
52    let bytes = examples::files::read(
53        &path,
54        limits.metadata_bytes,
55        examples::Resource::ManifestBytes,
56    )
57    .map_err(Error::Corpus)?;
58    examples::files::digest("scalability/manifest.json", &bytes, MANIFEST_SHA256)
59        .map_err(Error::Corpus)?;
60    let manifest: Manifest = serde_json::from_slice(&bytes)
61        .map_err(|error| Error::Corpus(examples::Error::Json(error)))?;
62    if manifest.schema_version != 1
63        || manifest.description.is_empty()
64        || manifest.reference_toolchain != "clingo version 5.8.2"
65        || manifest.cases.len() != 2
66    {
67        return Err(Error::Configuration("unsupported scalability manifest"));
68    }
69    Ok(manifest.cases)
70}

72fn workload(
73    root: &Path,
74    fixture: &Fixture,
75    size: i32,
76    limits: WorkloadLimits,
77) -> Result<Workload, Error> {
78    let entry = format!("scalability/{}", fixture.path);
79    let amendment = ConstantAmendment {
80        source_path: &entry,
81        name: &fixture.scaling_constant,
82        expected: fixture.default_value,
83        replacement: size,
84    };
85    Workload::authored(
86        AuthoredProgram {
87            root,
88            entry: &entry,
89            sha256: &fixture.source_sha256,
90            contract: &fixture.contract,
91        },
92        if size == fixture.default_value {
93            &[]
94        } else {
95            std::slice::from_ref(&amendment)
96        },
97        limits,
98    )
99}

101/// Admit queens at n=8 and pigeonhole at h=7, with their original contracts.
102/// `root` is the repository's `examples` directory.
103///
104/// # Errors
105/// Refuses changed metadata/sources, unsupported syntax or admission limits.
106pub fn defaults(root: &Path, limits: WorkloadLimits) -> Result<Vec<Workload>, Error> {
107    fixtures(root, limits)?
108        .iter()
109        .map(|fixture| workload(root, fixture, fixture.default_value, limits))
110        .collect()
111}

113/// Admit the unchanged Einstein riddle with its unique five-house display.
114///
115/// # Errors
116/// Refuses changed source bytes, unsupported syntax or admission limits.
117pub fn einstein(root: &Path, limits: WorkloadLimits) -> Result<Workload, Error> {
118    let witness = [
119        r#"solution(1,"Yellow","The Norwegian","Fox","Water","Kools")"#,
120        r#"solution(2,"Blue","The Ukrainian","Horse","Tea","Chesterfields")"#,
121        r#"solution(3,"Red","The Englishman","Snail","Milk","Old Gold")"#,
122        r#"solution(4,"Ivory","The Spaniard","Dog","Orange Juice","Lucky Strikes")"#,
123        r#"solution(5,"Green","The Japanese","Zebra","Coffee","Parliaments")"#,
124    ]
125    .map(str::to_owned)
126    .to_vec();
127    let contract = examples::Contract::complete_family(NonZeroU64::MIN)
128        .with_witnesses(vec![witness])
129        .map_err(Error::Corpus)?;
130    Workload::authored(
131        AuthoredProgram {
132            root,
133            entry: "einstein-riddle.lp",
134            sha256: EINSTEIN_SHA256,
135            contract: &contract,
136        },
137        &[],
138        limits,
139    )
140}

142/// Queens at n=8/9/10 and pigeonhole at h=5/6/7, followed by unchanged queens
143/// variant 02, SEND+MORE=MONEY and task allocation; Einstein is an optional tenth
144/// case. Amended sizes are qualified against a complete reference enumeration.
145///
146/// # Errors
147/// Refuses fixture/corpus identity changes and checked-amendment limits.
148pub fn workloads(
149    corpus: &examples::Corpus,
150    root: &Path,
151    include_einstein: bool,
152    limits: WorkloadLimits,
153) -> Result<Vec<Workload>, Error> {
154    let fixtures = fixtures(root, limits)?;
155    let mut result = Vec::with_capacity(10);
156    for (fixture, sizes) in fixtures.iter().zip([[8, 9, 10], [5, 6, 7]]) {
157        for size in sizes {
158            result.push(workload(root, fixture, size, limits)?);
159        }
160    }
161    for case in [
162        super::Case::Queens02,
163        super::Case::Send,
164        super::Case::TaskAllocation,
165    ] {
166        result.push(Workload::original(corpus, case.path(), limits)?);
167    }
168    if include_einstein {
169        result.push(einstein(root, limits)?);
170    }
171    Ok(result)
172}

174/// CPU eager/indexed region search at 1, 2, 4, 8 and 14 workers, with one
175/// completion worker. An optional expansion ceiling is retained in every profile.
176#[must_use]
177pub fn profiles(max_expansion_work: Option<usize>) -> Vec<NativeExecution> {
178    const WORKERS: [NonZeroUsize; 5] = [
179        NonZeroUsize::MIN,
180        NonZeroUsize::new(2).unwrap(),
181        NonZeroUsize::new(4).unwrap(),
182        NonZeroUsize::new(8).unwrap(),
183        NonZeroUsize::new(14).unwrap(),
184    ];
185    WORKERS
186        .map(|workers| NativeExecution {
187            workers,
188            formula_joins: Some(FormulaJoins::Indexed),
189            search: Some(SearchMethod::Regions),
190            max_expansion_work,
191            ..NativeExecution::default()
192        })
193        .to_vec()
194}

196/// Run the maintained scalability population through the shared matrix owner.
197/// The request must select [`matrix::Suite::Scalability`]. Qualification-only
198/// and measurement plans use the same workloads, provenance and complete-family
199/// checks. All request limits are honored without raising them. The caller owns
200/// cancellation and publication of the returned report.
201///
202/// # Errors
203/// Refuses a different suite, changed input identities, workload admission
204/// limits, or the matrix's pre-launch configuration and source failures.
205pub fn run_with_cancellation(
206    request: &matrix::Request<'_>,
207    root: &Path,
208    include_einstein: bool,
209    invocation: matrix::NativeInvocation,
210    cancelled: &std::sync::atomic::AtomicBool,
211) -> Result<matrix::Report, Error> {
212    if request.plan.suite() != matrix::Suite::Scalability {
213        return Err(Error::Configuration(
214            "authored scalability requires the scalability suite",
215        ));
216    }
217    let corpus = examples::load(request.corpus, request.limits.corpus).map_err(Error::Corpus)?;
218    let limits = WorkloadLimits {
219        source_bytes: request.limits.corpus.source_bytes,
220        total_source_bytes: request.limits.corpus.total_source_bytes,
221        metadata_bytes: request.limits.corpus.manifest_bytes,
222        ..WorkloadLimits::default()
223    };
224    let workloads = workloads(&corpus, root, include_einstein, limits)?;
225    matrix::run_workloads_with_cancellation(request, &workloads, invocation, cancelled)
226}
