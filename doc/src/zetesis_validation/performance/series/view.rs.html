1//! Derived comparison of published matrix reports over the same cells.
2//!
3//! The view reads the retained records and computes exact integer medians of
4//! the timed native and reference intervals per cell and profile, and the
5//! counters the native records carry; its tables divide those medians between
6//! reports in the order given and against the reference, exactly, and its JSON
7//! publishes the medians alone. For each report and profile it also keeps a
8//! scoreboard against the reference solver: the cells where both passed,
9//! which of them the native solver decided faster, and each cell's time
10//! split into grounding, candidate proposal and membership on the native
11//! side and grounding and solving on the reference's, with the peak
12//! resident set of each when the campaign ran memory rounds. It never pools
13//! reports, never averages a cell that did not pass, and retains each
14//! report's native executable seal so that a published comparison names
15//! what it compared.
16

27/// A published report under the name it will carry in the comparison.
28#[derive(Clone, Copy, Debug)]
29pub struct Labelled<'a> {
30    /// Column name, for example `main`, `before` or `after`.
31    pub label: &'a str,
32    /// The published matrix report document (`{"passed", "accounted", "report"}`).
33    pub report: &'a Value,
34}

36/// A refused comparison.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub enum ViewError {
39    /// No report was given.
40    Empty,
41    /// Two reports carry the same label.
42    Label {
43        /// The repeated label.
44        label: String,
45    },
46    /// The reports do not measure the same cells in the same order.
47    Cells {
48        /// The label whose cells differ from the first report's.
49        label: String,
50    },
51    /// The reports do not request the same profiles in the same order.
52    Profiles {
53        /// The label whose profiles differ from the first report's.
54        label: String,
55    },
56    /// A report's profiles do not request the same search method, so it
57    /// has no one method to stand on its scoreboards.
58    Methods {
59        /// The label of the report.
60        label: String,
61    },
62    /// Qualification-only evidence has no native timed population to compare.
63    NoTimedPopulation {
64        /// The label of the report.
65        label: String,
66    },
67    /// A report lacks a field the view reads.
68    Malformed {
69        /// The label of the report.
70        label: String,
71        /// The missing or ill-typed field.
72        field: &'static str,
73    },
74}
75impl fmt::Display for ViewError {
76    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
77        match self {
78            Self::Empty => write!(f, "no reports to compare"),
79            Self::Label { label } => write!(f, "report label {label:?} is repeated"),
80            Self::Cells { label } => write!(f, "report {label:?} measures different cells"),
81            Self::Profiles { label } => {
82                write!(f, "report {label:?} requests different profiles")
83            }
84            Self::Methods { label } => {
85                write!(f, "report {label:?} requests different search methods")
86            }
87            Self::NoTimedPopulation { label } => {
88                write!(
89                    f,
90                    "report {label:?} has no native timed population to compare"
91                )
92            }
93            Self::Malformed { label, field } => {
94                write!(f, "report {label:?} lacks a readable {field}")
95            }
96        }
97    }
98}

329/// The derived comparison, serializable as a retained observation.
330#[derive(Clone, Debug, Serialize)]
331pub struct Comparison {
332    /// Report labels in the order given.
333    pub labels: Vec<String>,
334    /// Cells in schedule order.
335    pub cells: Vec<Cell>,
336    /// Per-report provenance.
337    pub provenance: BTreeMap<String, Provenance>,
338    /// The formula search method each report requested, by label, where a
339    /// report requested one; the executable's default otherwise. Reports may
340    /// differ in this field alone, since the methods are compared on the
341    /// same cells by design. A report written before the `search` field
342    /// carries the method as `candidates`, and one campaign's reports name
343    /// the region workers beside it as `region_workers`; both are read as
344    /// the method and left out of the profile comparison.
345    pub methods: BTreeMap<String, String>,
346    /// Each report's and profile's standing against the reference, in report
347    /// order and then profile order; a report whose campaign ran without
348    /// clingo has none.
349    pub scoreboards: Vec<Scoreboard>,
350}

352/// Compare published reports over the same cells and profiles. The formula
353/// search method is the one profile field the reports may differ in, under
354/// any of its spellings ([`Comparison::methods`]). Cells must have the same
355/// ordered entry paths, sealed corpus manifest and explicit workload content
356/// identities. A derived report must retain one workload identity per cell;
357/// a corpus report without workloads compares only with other corpus reports.
358///
359/// # Errors
360/// Refuses an empty list, repeated labels, reports whose cells or profiles
361/// differ, a report whose profiles disagree on the search method, and
362/// reports lacking the fields the view reads, or qualification evidence without
363/// a native timed population.
364pub fn compare(reports: &[Labelled<'_>]) -> Result<Comparison, ViewError> {
365    let first = reports.first().ok_or(ViewError::Empty)?;
366    let entries = cases(first)?;
367    let workloads = workload_identities(first, &entries)?;
368    let manifest = provenance(first)?.manifest_sha256;
369    let requested = profiles(first)?;
370    let mut labels = Vec::with_capacity(reports.len());
371    let mut provenance = BTreeMap::new();
372    let mut methods = BTreeMap::new();
373    for labelled in reports {
374        if !samples(labelled)?.iter().any(|sample| {
375            sample["slot"]["phase"] == "timed" && sample["slot"]["producer"]["solver"] == "native"
376        }) {
377            return Err(ViewError::NoTimedPopulation {
378                label: labelled.label.into(),
379            });
380        }
381        if labels.contains(&labelled.label.to_owned()) {
382            return Err(ViewError::Label {
383                label: labelled.label.into(),
384            });
385        }
386        let recorded = self::provenance(labelled)?;
387        if cases(labelled)? != entries
388            || workload_identities(labelled, &entries)? != workloads
389            || recorded.manifest_sha256 != manifest
390        {
391            return Err(ViewError::Cells {
392                label: labelled.label.into(),
393            });
394        }
395        if profiles(labelled)? != requested {
396            return Err(ViewError::Profiles {
397                label: labelled.label.into(),
398            });
399        }
400        labels.push(labelled.label.to_owned());
401        provenance.insert(labelled.label.to_owned(), recorded);
402        methods.insert(labelled.label.to_owned(), method(labelled)?);
403    }
404    let workloads = first.report["report"]["workloads"].as_array();
405    let mut cells = Vec::with_capacity(entries.len());
406    for (index, entry) in entries.iter().enumerate() {
407        let label = label(entry, workloads.and_then(|workloads| workloads.get(index)));
408        let mut reference = BTreeMap::new();
409        let mut qualification = BTreeMap::new();
410        let mut failure_reasons = BTreeMap::new();
411        for labelled in reports {
412            if let Some(record) = self::reference(labelled, index)? {
413                reference.insert(labelled.label.to_owned(), record);
414            }
415            qualification.insert(
416                labelled.label.to_owned(),
417                self::qualification(labelled, index)?,
418            );
419            let reasons = self::failure_reasons(labelled, index)?;
420            if !reasons.is_empty() {
421                failure_reasons.insert(labelled.label.to_owned(), reasons);
422            }
423        }
424        let mut rows = Vec::with_capacity(requested.len());
425        for (profile, request) in requested.iter().enumerate() {
426            let mut records = BTreeMap::new();
427            for labelled in reports {
428                records.insert(labelled.label.to_owned(), native(labelled, index, profile)?);
429            }
430            rows.push(ProfileRow {
431                profile: request.clone(),
432                reports: records,
433            });
434        }
435        cells.push(Cell {
436            entry: entry.clone(),
437            label,
438            profiles: rows,
439            reference,
440            qualification,
441            failure_reasons,
442        });
443    }
444    // A scoreboard stands a report against clingo's times, so a report that
445    // did not time clingo has none.
446    let with_clingo: Vec<String> = labels
447        .iter()
448        .filter(|label| provenance[*label].reference_timed)
449        .cloned()
450        .collect();
451    let scoreboards = scoreboards(&with_clingo, &methods, &cells);
452    Ok(Comparison {
453        labels,
454        cells,
455        provenance,
456        methods,
457        scoreboards,
458    })
459}

461/// One scoreboard per report in `labels` and per profile, over the cells
462/// where both passed.
463fn scoreboards(
464    labels: &[String],
465    methods: &BTreeMap<String, String>,
466    cells: &[Cell],
467) -> Vec<Scoreboard> {
468    let profiles = cells.first().map_or(0, |cell| cell.profiles.len());
469    let mut scoreboards = Vec::with_capacity(labels.len() * profiles);
470    for label in labels {
471        for profile in 0..profiles {
472            let mut verdicts: Vec<Verdict> = cells
473                .iter()
474                .filter_map(|cell| {
475                    let Some(Native::Passed(passed)) = cell.profiles[profile].reports.get(label)
476                    else {
477                        return None;
478                    };
479                    let reference = cell.reference.get(label)?;
480                    Some(verdict(&cell.label, passed, reference))
481                })
482                .collect();
483            verdicts.sort_by_key(|verdict| {
484                Ratio::against_reference(verdict.native_ns, verdict.reference_ns)
485            });
486            scoreboards.push(Scoreboard {
487                report: label.clone(),
488                profile,
489                method: methods
490                    .get(label)
491                    .cloned()
492                    .unwrap_or_else(|| "default".to_owned()),
493                compared: verdicts.len(),
494                wins: verdicts
495                    .iter()
496                    .filter(|verdict| verdict.native_ns < verdict.reference_ns)
497                    .count(),
498                verdicts,
499            });
500        }
501    }
502    scoreboards
503}

505/// One median interval over another, exact: ordered by cross-multiplication
506/// and shown to three decimals, rounded half up by integer arithmetic, as
507/// `milliseconds` shows an interval.
508#[derive(Clone, Copy, Debug)]
509struct Ratio {
510    numerator: u64,
511    denominator: u64,
512}
513
514impl Ratio {
515    /// A native median over the reference's; a reference median of zero reads
516    /// as one nanosecond, so the ratio is always a number.
517    fn against_reference(native_ns: u64, reference_ns: u64) -> Self {
518        Self {
519            numerator: native_ns,
520            denominator: reference_ns.max(1),
521        }
522    }
523    /// A later report's median over an earlier one's, when the earlier is positive.
524    fn between(later_ns: u64, earlier_ns: u64) -> Option<Self> {
525        (earlier_ns > 0).then_some(Self {
526            numerator: later_ns,
527            denominator: earlier_ns,
528        })
529    }
530}
531
532impl Ord for Ratio {
533    fn cmp(&self, other: &Self) -> Ordering {
534        // Each product of two u64 values fits in a u128.
535        (u128::from(self.numerator) * u128::from(other.denominator))
536            .cmp(&(u128::from(other.numerator) * u128::from(self.denominator)))
537    }
538}
539
540impl PartialOrd for Ratio {
541    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
542        Some(self.cmp(other))
543    }
544}
545
546impl PartialEq for Ratio {
547    fn eq(&self, other: &Self) -> bool {
548        self.cmp(other) == Ordering::Equal
549    }
550}
551
552impl Eq for Ratio {}
553
554impl fmt::Display for Ratio {
555    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
556        // Thousandths rounded half up, floor((2000 n + d) / 2d), exact in u128.
557        let numerator = u128::from(self.numerator);
558        let denominator = u128::from(self.denominator);
559        let thousandths = (2_000 * numerator + denominator) / (2 * denominator);
560        write!(f, "{}.{:03}", thousandths / 1_000, thousandths % 1_000)
561    }
562}

593/// The profile line of a table, and the search method of each report when
594/// any report requested one.
595fn heading(f: &mut fmt::Formatter<'_>, comparison: &Comparison, profile: &Value) -> fmt::Result {
596    writeln!(
597        f,
598        "Native wall time, ms: median [minimum, maximum] of the timed intervals; profile {}.\n",
599        Profile(profile)
600    )?;
601    if comparison
602        .methods
603        .values()
604        .any(|method| method != "default")
605    {
606        write!(f, "Formula search method by report:")?;
607        for label in &comparison.labels {
608            let method = comparison
609                .methods
610                .get(label)
611                .map_or("default", String::as_str);
612            write!(f, " {label}={method};")?;
613        }
614        writeln!(f, "\n")?;
615    }
616    Ok(())
617}

619impl fmt::Display for Markdown<'_> {
620    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
621        let comparison = self.0;
622        let last = comparison
623            .labels
624            .last()
625            .map(String::as_str)
626            .unwrap_or_default();
627        let profile_count = comparison
628            .cells
629            .first()
630            .map_or(0, |cell| cell.profiles.len());
631        for profile in 0..profile_count {
632            let first = &comparison.cells[0].profiles[profile];
633            heading(f, comparison, &first.profile)?;
634            let pairs = report_pairs(&comparison.labels);
635            write!(f, "| Cell |")?;
636            for label in &comparison.labels {
637                write!(f, " {label} |")?;
638            }
639            for (earlier, later) in &pairs {
640                write!(f, " {later}/{earlier} |")?;
641            }
642            for label in &comparison.labels {
643                write!(f, " {label}/reference |")?;
644            }
645            write!(f, "\n|---|")?;
646            for _ in 0..2 * comparison.labels.len() + pairs.len() {
647                write!(f, "---:|")?;
648            }
649            writeln!(f)?;
650            for cell in &comparison.cells {
651                let row = &cell.profiles[profile];
652                write!(f, "| {} |", cell.label)?;
653                for label in &comparison.labels {
654                    write!(f, " {} |", native_cell(row.reports.get(label)))?;
655                }
656                for (earlier, later) in &pairs {
657                    let ratio = passed_median(row.reports.get(*later))
658                        .zip(passed_median(row.reports.get(*earlier)))
659                        .and_then(|(later, earlier)| Ratio::between(later, earlier));
660                    match ratio {
661                        Some(ratio) => write!(f, " {ratio} |")?,
662                        None => write!(f, " n/a |")?,
663                    }
664                }
665                for label in &comparison.labels {
666                    let ratio = passed_median(row.reports.get(label))
667                        .zip(cell.reference.get(label))
668                        .map(|(native, reference)| {
669                            Ratio::against_reference(native, reference.timing.median_ns)
670                        });
671                    match ratio {
672                        Some(ratio) => write!(f, " {ratio} |")?,
673                        None if !clingo_ran(comparison, label) => write!(f, " not run |")?,
674                        None if !clingo_timed(comparison, label) => write!(f, " not timed |")?,
675                        None => write!(f, " n/a |")?,
676                    }
677                }
678                writeln!(f)?;
679            }
680            writeln!(f)?;
681        }
682        reference_table(f, comparison)?;
683        writeln!(
684            f,
685            "\nCounters of report {last}: published models, candidates examined, charged search work, driver median ms.\n\n| Cell | profile | models | candidates | work | driver ms |\n|---|---|---:|---:|---:|---:|"
686        )?;
687        for cell in &comparison.cells {
688            for (index, row) in cell.profiles.iter().enumerate() {
689                write!(f, "| {} | {index} |", cell.label)?;
690                match row.reports.get(last) {
691                    Some(Native::Passed(passed)) => writeln!(
692                        f,
693                        " {} | {} | {} | {} |",
694                        optional(passed.published_models),
695                        optional(passed.candidates_examined),
696                        optional(passed.search_work),
697                        passed
698                            .driver_median_ns
699                            .map_or_else(|| "n/a".to_owned(), milliseconds)
700                    )?,
701                    Some(Native::NotPassed { decisions, reasons }) => {
702                        writeln!(f, " {} | | | |", non_pass_text(decisions, reasons))?;
703                    }
704                    None => writeln!(f, " n/a | | | |")?,
705                }
706            }
707        }
708        for scoreboard in &comparison.scoreboards {
709            scoreboard_tables(f, comparison, scoreboard)?;
710        }
711        failure_table(f, comparison)?;
712        Ok(())
713    }
714}

716/// Clingo's wall time per cell and report: "not passed" where it failed,
717/// "not timed" where the report's campaign ran it only to qualify the cells,
718/// and "not run" where the campaign ran without it.
719fn reference_table(f: &mut fmt::Formatter<'_>, comparison: &Comparison) -> fmt::Result {
720    write!(f, "Reference wall time, ms, same notation.\n\n| Cell |")?;
721    for label in &comparison.labels {
722        write!(f, " {label} |")?;
723    }
724    write!(f, "\n|---|")?;
725    for _ in &comparison.labels {
726        write!(f, "---:|")?;
727    }
728    writeln!(f)?;
729    for cell in &comparison.cells {
730        write!(f, "| {} |", cell.label)?;
731        for label in &comparison.labels {
732            match cell.reference.get(label) {
733                Some(record) => write!(f, " {} |", timing_cell(&record.timing))?,
734                None if !clingo_ran(comparison, label) => write!(f, " not run |")?,
735                None if !clingo_timed(comparison, label) => write!(f, " not timed |")?,
736                None => write!(f, " not passed |")?,
737            }
738        }
739        writeln!(f)?;
740    }
741    Ok(())
742}

791/// One scoreboard: its sentence, its wins, its losses and its peak memory.
792fn scoreboard_tables(
793    f: &mut fmt::Formatter<'_>,
794    comparison: &Comparison,
795    scoreboard: &Scoreboard,
796) -> fmt::Result {
797    let profile = comparison
798        .cells
799        .first()
800        .map(|cell| Profile(&cell.profiles[scoreboard.profile].profile).to_string())
801        .unwrap_or_default();
802    write!(
803        f,
804        "\nAgainst the reference: report {}, profile {} ({profile}; search {}): ",
805        scoreboard.report, scoreboard.profile, scoreboard.method
806    )?;
807    if scoreboard.compared == 0 {
808        writeln!(f, "no cell where both passed.")?;
809        return Ok(());
810    }
811    let tenths = scoreboard.wins * 1000 / scoreboard.compared;
812    writeln!(
813        f,
814        "faster on {} of {} cells where both passed ({}.{}%).",
815        scoreboard.wins,
816        scoreboard.compared,
817        tenths / 10,
818        tenths % 10
819    )?;
820    let (wins, losses): (Vec<&Verdict>, Vec<&Verdict>) = scoreboard
821        .verdicts
822        .iter()
823        .partition(|verdict| verdict.native_ns < verdict.reference_ns);
824    for (name, verdicts) in [
825        ("Wins, fastest first", wins),
826        ("Losses, closest first", losses),
827    ] {
828        if verdicts.is_empty() {
829            writeln!(f, "\n{name}: none.")?;
830            continue;
831        }
832        writeln!(
833            f,
834            "\n{name}. Milliseconds: the native and the reference medians and their ratio; the native split into grounding, candidate proposal and membership, each summed over its intervals, which can overlap within or across threads, so the parts need not add up to the native median; the reference's into grounding and solving from its own report.\n\n| Cell | native | reference | native/reference | grounding | proposal | membership | reference grounding | reference solving |\n|---|---:|---:|---:|---:|---:|---:|---:|---:|"
835        )?;
836        for verdict in verdicts {
837            writeln!(
838                f,
839                "| {} | {} | {} | {} | {} | {} | {} | {} | {} |",
840                verdict.cell,
841                milliseconds(verdict.native_ns),
842                milliseconds(verdict.reference_ns),
843                Ratio::against_reference(verdict.native_ns, verdict.reference_ns),
844                optional_ms(verdict.native.grounding),
845                optional_ms(verdict.native.proposal),
846                optional_ms(verdict.native.membership),
847                optional_ms(verdict.reference_grounding_ns),
848                optional_ms(verdict.reference_solving_ns),
849            )?;
850        }
851    }
852    if scoreboard.verdicts.iter().any(|verdict| {
853        verdict.native_peak_rss_bytes.is_some()
854            || verdict.reference_peak_rss_bytes.is_some()
855            || verdict.device_bytes.is_some()
856    }) {
857        writeln!(
858            f,
859            "\nPeak memory, MiB: the resident set of the native and of the reference over the memory rounds, and the device memory the native accounted.\n\n| Cell | native | reference | device |\n|---|---:|---:|---:|"
860        )?;
861        for verdict in &scoreboard.verdicts {
862            writeln!(
863                f,
864                "| {} | {} | {} | {} |",
865                verdict.cell,
866                optional_mib(verdict.native_peak_rss_bytes),
867                optional_mib(verdict.reference_peak_rss_bytes),
868                optional_mib(verdict.device_bytes),
869            )?;
870        }
871    }
872    Ok(())
873}

893impl fmt::Display for Profile<'_> {
894    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
895        let field = |name: &str| match self.0[name].as_str() {
896            Some(text) => text.to_owned(),
897            None => self.0[name].to_string(),

930            write!(f, ", time limit={seconds} s")?;

932        if let Some(work) = self.0["max_expansion_work"].as_u64() {

936    }

978/// A retained diagnostic is text, never Markdown or a new table row.
979fn markdown_text(text: &str) -> String {
980    let mut escaped = String::new();
981    for character in text.chars() {
982        match character {
983            '&' => escaped.push_str("&amp;"),
984            '<' => escaped.push_str("&lt;"),
985            '>' => escaped.push_str("&gt;"),
986            '\\' | '|' | '`' | '*' | '_' | '[' | ']' => {
987                escaped.push('\\');
988                escaped.push(character);
989            }
990            character if character.is_control() => escaped.extend(character.escape_default()),
991            character => escaped.push(character),
992        }
993    }
994    escaped
995}

1017/// The reports the tables divide, as `(earlier, later)`: each report after the

1021    let mut pairs: Vec<(&String, &String)> = labels.windows(2).map(|w| (&w[0], &w[1])).collect();

1051/// Content identities distinguish amendments that deliberately share an
1052/// entry path. Schema 2 always carries them; older corpus reports do not.
1053fn workload_identities<'a>(
1054    labelled: &Labelled<'a>,
1055    entries: &[String],
1056) -> Result<Option<Vec<&'a str>>, ViewError> {
1057    let malformed = |field| ViewError::Malformed {
1058        label: labelled.label.into(),
1059        field,
1060    };
1061    let Some(raw) = labelled.report["report"].get("workloads") else {
1062        return if labelled.report["report"]["schema"] == 2 {
1063            Err(malformed("report.workloads"))
1064        } else {
1065            Ok(None)
1066        };
1067    };
1068    let workloads = raw
1069        .as_array()
1070        .ok_or_else(|| malformed("report.workloads"))?;
1071    if workloads.len() != entries.len() {
1072        return Err(malformed("one workload per report.cases entry"));
1073    }
1074    workloads
1075        .iter()
1076        .zip(entries)
1077        .map(|(workload, entry)| {
1078            if workload["entry"].as_str() != Some(entry.as_str()) {
1079                return Err(malformed("workload.entry matching report.cases"));
1080            }
1081            workload["identity"]
1082                .as_str()
1083                .filter(|identity| {
1084                    identity.len() == 64 && identity.bytes().all(|byte| byte.is_ascii_hexdigit())
1085                })
1086                .ok_or_else(|| malformed("workload.identity SHA-256"))
1087        })
1088        .collect::<Result<Vec<_>, _>>()
1089        .map(Some)
1090}

1092/// The fields that spell a profile's search method, left out of the profile

1156/// What can qualify a case in one report: clingo where it took part; without
1157/// it, the workload's contract, which amendments invalidate even when no
1158/// position ran. Older corpus reports without workload metadata retain the
1159/// reading from their recorded decisions.
1160fn qualification(labelled: &Labelled<'_>, case: usize) -> Result<Qualification, ViewError> {
1161    if !clingo_free(labelled) {
1162        return Ok(Qualification::Clingo);
1163    }
1164    if let Some(workloads) = labelled.report["report"].get("workloads") {
1165        let amended = workloads
1166            .get(case)
1167            .and_then(|workload| workload["amended"].as_bool())
1168            .ok_or(ViewError::Malformed {
1169                label: labelled.label.into(),
1170                field: "workload.amended",
1171            })?;
1172        return Ok(if amended {
1173            Qualification::NeedsClingo
1174        } else {
1175            Qualification::Contract
1176        });
1177    }
1178    let needs_clingo = samples(labelled)?
1179        .iter()
1180        .any(|sample| sample["slot"]["case"] == case && sample["decision"] == "needs_clingo");
1181    Ok(if needs_clingo {
1182        Qualification::NeedsClingo
1183    } else {
1184        Qualification::Contract
1185    })
1186}

1188fn provenance(labelled: &Labelled<'_>) -> Result<Provenance, ViewError> {
1189    let malformed = |field| ViewError::Malformed {
1190        label: labelled.label.into(),
1191        field,
1192    };
1193    // The campaign seals the native executable, the reference executable when
1194    // clingo takes part, and the manifest first, in that order, before the
1195    // sources. The recorded policy says whether clingo took part.
1196    let seal = |index: usize, field| {
1197        labelled.report["report"]["before"]
1198            .as_array()
1199            .and_then(|seals| seals.get(index))
1200            .and_then(|seal| seal["sha256"].as_str())
1201            .map(str::to_owned)
1202            .ok_or(malformed(field))
1203    };
1204    let clingo_free = clingo_free(labelled);
1205    let reference_timed = samples(labelled)?.iter().any(|sample| {
1206        sample["slot"]["phase"] == "timed" && sample["slot"]["producer"]["solver"] == "reference"
1207    });
1208    Ok(Provenance {
1209        native_sha256: seal(0, "report.before[0].sha256")?,
1210        reference_sha256: if clingo_free {
1211            None
1212        } else {
1213            Some(seal(1, "report.before[1].sha256")?)
1214        },
1215        reference_timed,
1216        manifest_sha256: if clingo_free {
1217            seal(1, "report.before[1].sha256")?
1218        } else {
1219            seal(2, "report.before[2].sha256")?
1220        },
1221        started_unix_ns: labelled.report["report"]["started_unix_ns"]
1222            .as_u64()
1223            .ok_or(malformed("report.started_unix_ns"))?,
1224        finished_unix_ns: labelled.report["report"]["finished_unix_ns"].as_u64(),
1225        passed: labelled.report["passed"]
1226            .as_bool()
1227            .ok_or(malformed("passed"))?,
1228        accounted: labelled.report["accounted"]
1229            .as_bool()
1230            .ok_or(malformed("accounted"))?,
1231    })
1232}

1268                })
1269        })
1270        .collect::<Result<Vec<_>, _>>()?;
1271    let median_ns = median(&mut intervals).ok_or(ViewError::Malformed {
1272        label: labelled.label.into(),
1273        field: "report.samples",
1274    })?;
1275    Ok(Timing {
1276        samples: intervals.len(),
1277        minimum_ns: intervals[0],
1278        median_ns,
1279        maximum_ns: intervals[intervals.len() - 1],
1280    })
1281}

1283fn native(labelled: &Labelled<'_>, case: usize, profile: usize) -> Result<Native, ViewError> {
1284    let records = timed(labelled, case, |producer| {
1285        producer["solver"] == "native" && producer["profile"].as_u64() == Some(profile as u64)
1286    })?;
1287    let all = samples(labelled)?;
1288    let mut decisions = BTreeMap::new();
1289    let mut reasons = BTreeMap::new();
1290    for record in &records {
1291        let decision = record["decision"].as_str().unwrap_or("unknown");
1292        if decision == "pass" {
1293            continue;
1294        }
1295        // A position the campaign never launched is described by the
1296        // decision that disabled its cell, when the record names it.
1297        let name = match record["blocked_by"]
1298            .as_u64()
1299            .and_then(|index| usize::try_from(index).ok())
1300            .and_then(|index| all.get(index))
1301            .and_then(|blocker| blocker["decision"].as_str())
1302        {
1303            Some(blocker) if decision == "not_attempted" => format!("blocked by {blocker}"),
1304            _ => decision.to_owned(),
1305        };
1306        *decisions.entry(name).or_insert(0) += 1;
1307        *reasons.entry(sample_reason(record, all)).or_insert(0) += 1;
1308    }
1309    if records.is_empty() {
1310        return Err(ViewError::Malformed {
1311            label: labelled.label.into(),
1312            field: "timed native samples for the cell",
1313        });
1314    }
1315    if !decisions.is_empty() {
1316        return Ok(Native::NotPassed { decisions, reasons });
1317    }
1318    let timing = timing(labelled, &records)?;
1319    let first = records[0];
1320    let stdout: Option<Value> = first["capture"]["stdout"]["data"]
1321        .as_str()
1322        .and_then(|text| serde_json::from_str(text).ok());
1323    let stderr = first["capture"]["stderr"]["data"].as_str().unwrap_or("");
1324    let mut drivers: Vec<u64> = records
1325        .iter()
1326        .filter_map(|record| record["observation"]["timing"]["driver_elapsed_ns"].as_u64())
1327        .collect();
1328    let driver_median_ns = median(&mut drivers);
1329    let mut phases: BTreeMap<String, Vec<u64>> = BTreeMap::new();
1330    for record in &records {
1331        if let Some(measured) = record["observation"]["timing"]["phases"].as_object() {
1332            for (name, value) in measured {
1333                if let Some(elapsed) = value["elapsed_ns"].as_u64() {
1334                    phases.entry(name.clone()).or_default().push(elapsed);
1335                }
1336            }
1337        }
1338    }
1339    let phases = phases
1340        .into_iter()
1341        .filter_map(|(name, mut values)| {
1342            median(&mut values).map(|median_ns| (name, PhaseTiming { median_ns }))
1343        })
1344        .collect();
1345    let peak_rss_bytes = memory(labelled, case, |producer| {
1346        producer["solver"] == "native" && producer["profile"].as_u64() == Some(profile as u64)
1347    })?;
1348    Ok(Native::Passed(Passed {
1349        timing,
1350        driver_median_ns,
1351        breakdown: breakdown(&records),
1352        peak_rss_bytes,
1353        device_bytes: stdout.as_ref().and_then(|document| {
1354            document["statistics"]["execution"]["peak_accounted_bytes"].as_u64()
1355        }),
1356        published_models: stdout
1357            .as_ref()
1358            .and_then(|document| document["outcome"]["published_models"].as_u64()),
1359        // The formula route types the count as `statistics.search.candidates`;
1360        // the closure route reports it only on the human results line.
1361        candidates_examined: stdout
1362            .as_ref()
1363            .and_then(|document| document["statistics"]["search"]["candidates"].as_u64())
1364            .or_else(|| candidates_examined(stderr)),
1365        search_work: stdout.as_ref().and_then(|document| {
1366            document["statistics"]["search"]["work"]
1367                .as_u64()
1368                .or_else(|| document["statistics"]["closure_execution"]["work"].as_u64())
1369        }),
1370        phases,
1371    }))
1372}

1374fn failure_reasons(
1375    labelled: &Labelled<'_>,
1376    case: usize,
1377) -> Result<BTreeMap<String, usize>, ViewError> {
1378    let all = samples(labelled)?;
1379    let mut reasons = BTreeMap::new();
1380    for record in all.iter().filter(|record| {
1381        record["slot"]["case"].as_u64() == Some(case as u64) && record["decision"] != "pass"
1382    }) {
1383        let producer = &record["slot"]["producer"];
1384        let producer = match producer["solver"].as_str() {
1385            Some("native") => producer["profile"].as_u64().map_or_else(
1386                || "native profile unavailable".to_owned(),
1387                |profile| format!("native profile index {profile}"),
1388            ),
1389            Some(solver) => solver.to_owned(),
1390            None => "producer unavailable".to_owned(),
1391        };
1392        let reason = format!("{producer}: {}", sample_reason(record, all));
1393        *reasons.entry(reason).or_insert(0) += 1;
1394    }
1395    Ok(reasons)
1396}

1398fn sample_reason(record: &Value, samples: &[Value]) -> String {
1399    let phase = record["slot"]["phase"]
1400        .as_str()
1401        .unwrap_or("phase unavailable");
1402    let decision = record["decision"].as_str().unwrap_or("unknown");
1403    let detail = record["detail"]
1404        .as_str()
1405        .unwrap_or("reason unavailable in retained sample");
1406    let mut reason = format!("{phase}: {decision}: {detail}");
1407    if let Some(index) = record["blocked_by"].as_u64() {
1408        use std::fmt::Write;
1409        if let Some(blocker) = usize::try_from(index)
1410            .ok()
1411            .and_then(|index| samples.get(index))
1412        {
1413            write!(
1414                reason,
1415                "; blocked by sample {index} ({}, {}): {}",
1416                blocker["slot"]["phase"]
1417                    .as_str()
1418                    .unwrap_or("phase unavailable"),
1419                blocker["decision"].as_str().unwrap_or("unknown"),
1420                blocker["detail"]
1421                    .as_str()
1422                    .unwrap_or("reason unavailable in retained sample"),
1423            )
1424            .expect("writing to a String cannot fail");
1425        } else {
1426            write!(reason, "; blocking sample {index} unavailable")
1427                .expect("writing to a String cannot fail");
1428        }
1429    }
1430    reason
1431}

1443fn reference(labelled: &Labelled<'_>, case: usize) -> Result<Option<Reference>, ViewError> {
1444    let records = timed(labelled, case, |producer| producer["solver"] == "reference")?;
1445    if records.is_empty() || records.iter().any(|record| record["decision"] != "pass") {
1446        return Ok(None);
1447    }
1448    let timing = timing(labelled, &records)?;
1449    // The reference prints its own times in seconds to the millisecond; its
1450    // total less its solving time is grounding and preprocessing.
1451    let mut grounding = Vec::new();
1452    let mut solving = Vec::new();
1453    for record in &records {
1454        let Some(times) = record["capture"]["stdout"]["data"]
1455            .as_str()
1456            .and_then(|text| serde_json::from_str::<ReportedTimes<'_>>(text).ok())
1457        else {
1458            continue;
1459        };
1460        if let (Some(total), Some(solve)) = (
1461            decimal_seconds_ns(times.time.total.get()),
1462            decimal_seconds_ns(times.time.solve.get()),
1463        ) {
1464            grounding.push(total.saturating_sub(solve));
1465            solving.push(solve);
1466        }
1467    }
1468    Ok(Some(Reference {
1469        timing,
1470        grounding_ns: median(&mut grounding),
1471        solving_ns: median(&mut solving),
1472        peak_rss_bytes: memory(labelled, case, |producer| producer["solver"] == "reference")?,
1473    }))
1474}

1476/// The sums of each part's phases per record, and their medians.
1477fn breakdown(records: &[&Value]) -> Breakdown {
1478    let mut grounding = Vec::new();
1479    let mut proposal = Vec::new();
1480    let mut membership = Vec::new();
1481    for record in records {
1482        let timing = &record["observation"]["timing"];
1483        if let Some(elapsed) = timing["stages"]["grounding"]["elapsed_ns"].as_u64() {
1484            grounding.push(elapsed);
1485        }
1486        let sum = |names: &[&str]| {
1487            let measured: Vec<u64> = names
1488                .iter()
1489                .filter_map(|name| timing["phases"][*name]["elapsed_ns"].as_u64())
1490                .collect();
1491            (!measured.is_empty()).then(|| measured.iter().sum::<u64>())
1492        };
1493        proposal.extend(sum(&PROPOSAL_PHASES));
1494        membership.extend(sum(&MEMBERSHIP_PHASES));
1495    }
1496    Breakdown {
1497        grounding: median(&mut grounding),
1498        proposal: median(&mut proposal),
1499        membership: median(&mut membership),
1500    }
1501}

1503/// The median peak resident set over the passed memory rounds of one
1504/// producer on one cell; absent when the campaign ran none.
1505fn memory(
1506    labelled: &Labelled<'_>,
1507    case: usize,
1508    producer: impl Fn(&Value) -> bool,
1509) -> Result<Option<u64>, ViewError> {
1510    let mut peaks: Vec<u64> = samples(labelled)?
1511        .iter()
1512        .filter(|sample| {
1513            sample["slot"]["case"].as_u64() == Some(case as u64)
1514                && sample["slot"]["phase"] == "memory"
1515                && sample["decision"] == "pass"
1516                && producer(&sample["slot"]["producer"])
1517        })
1518        .filter_map(|sample| sample["memory"]["peak_rss_bytes"].as_u64())
1519        .collect();
1520    Ok(median(&mut peaks))
1521}

1523/// The one median of the file: the middle value, or the midpoint of the two
1524/// middle values for an even count; `None` of nothing.
1525pub(crate) fn median(values: &mut [u64]) -> Option<u64> {
1526    values.sort_unstable();
1527    let count = values.len();
1528    match count {
1529        0 => None,
1530        _ if count.is_multiple_of(2) => {
1531            Some(u64::midpoint(values[count / 2 - 1], values[count / 2]))
1532        }
1533        _ => Some(values[count / 2]),
1534    }
1535}

1537/// The times in the reference's saved report, as the text it wrote them in.
1538#[derive(Deserialize)]
1539struct ReportedTimes<'a> {
1540    #[serde(rename = "Time", borrow)]
1541    time: Times<'a>,
1542}
1543
1544#[derive(Deserialize)]
1545struct Times<'a> {
1546    #[serde(rename = "Total", borrow)]
1547    total: &'a RawValue,
1548    #[serde(rename = "Solve", borrow)]
1549    solve: &'a RawValue,
1550}

1552/// Seconds written as the reference writes them, a plain decimal such as
1553/// `0.012`, as whole nanoseconds, exactly. Any other spelling, such as a sign
1554/// or an exponent, or a nonzero digit beyond the ninth fractional one, reads as
1555/// no time rather than an approximation.
1556fn decimal_seconds_ns(text: &str) -> Option<u64> {
1557    let (whole, fraction) = text.split_once('.').unwrap_or((text, ""));
1558    let digits = |part: &str| part.bytes().all(|byte| byte.is_ascii_digit());
1559    if whole.is_empty() || !digits(whole) || !digits(fraction) {
1560        return None;
1561    }
1562    if fraction.bytes().skip(9).any(|byte| byte != b'0') {
1563        return None;
1564    }
1565    let nanoseconds = (0..9).fold(0, |value: u64, position| {
1566        let digit = fraction
1567            .as_bytes()
1568            .get(position)
1569            .map_or(0, |byte| byte - b'0');
1570        value * 10 + u64::from(digit)
1571    });
1572    whole
1573        .parse::<u64>()
1574        .ok()?
1575        .checked_mul(1_000_000_000)?
1576        .checked_add(nanoseconds)
1577}
