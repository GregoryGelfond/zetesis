1//! Serialized limits retain their distinct units and measurement boundaries.
2use super::{Limits, Report};
3use serde::{Serialize, Serializer};
4use serde_json::json;
5#[derive(Serialize)]
6pub(super) struct Published<'a> {
7    pub passed: bool,
8    #[serde(flatten)]
9    pub report: &'a Report,
10    #[serde(skip_serializing_if = "Option::is_none")]
11    pub summary: Option<Vec<super::Summary>>,
12}
13impl Serialize for Limits {
14    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
15        json!({ "corpus": { "manifest_bytes":self.corpus.manifest_bytes,
16            "source_bytes":self.corpus.source_bytes,"total_source_bytes":self.corpus.total_source_bytes,
17            "files":self.corpus.files,"cases":self.corpus.cases,"contract_symbols":self.corpus.contract_symbols },
18            "process": { "timeout":self.process.timeout,"cleanup_timeout":self.process.cleanup_timeout,"max_output_bytes":self.process.max_output_bytes },
19            "campaign_timeout":self.campaign_timeout,"max_total_capture_bytes":self.max_total_capture_bytes,
20            "max_executable_bytes":self.max_executable_bytes,"max_report_bytes":self.max_report_bytes,
21            "answers":{"max_input_bytes":self.answers.max_input_bytes,"max_witnesses":self.answers.max_witnesses,
22            "max_symbols":self.answers.max_symbols,"max_cost_dimensions":self.answers.max_cost_dimensions} }).serialize(serializer)
23    }
24}
