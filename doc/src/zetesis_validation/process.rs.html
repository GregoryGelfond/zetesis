1//! Raw child-process evidence with bounded capture and cleanup polling.
2//!
3//! The strong backend currently supports Linux and macOS. It starts a fresh
4//! process group, drains both pipes without reader threads, and retains at most
5//! `max_output_bytes` bytes across the two streams. Completion means that the
6//! direct child was reaped and both pipes reached EOF. It proves no descendant
7//! termination property: even a same-group descendant can survive after closing
8//! both inherited pipes. Process-group termination is attempted only for stopped
9//! or faulted capture. Completion says nothing about a solver's enumeration,
10//! answer correctness, or system quiescence. Calls require exclusive ownership
11//! of the child wait status.
12//!
13//! Deadlines bound authored polling, not the latency of OS calls or allocation.
14//! Retained vectors use fallible amortized growth. Capture bytes exclude vector
15//! capacity and OS pipe storage. A stopped capture
16//! is partial evidence. UTF-8 conversion is a separate, fallible view.
17

38/// Borrowed process arguments. Both paths must be absolute.
39#[derive(Clone, Copy, Debug)]
40pub struct Invocation<'a> {
41    /// Executable selected and resolved by the caller; no shell is inserted.
42    pub executable: &'a Path,
43    /// Arguments excluding the executable name.
44    pub arguments: &'a [OsString],
45    /// Working directory, independent of executable resolution.
46    pub directory: &'a Path,
47}

49/// Inclusive raw capture ceiling and independent polling intervals.
50#[derive(Clone, Copy, Debug)]
51pub struct Limits {
52    /// Invocation deadline, starting before spawn. Zero permits no polling work.
53    pub timeout: Duration,
54    /// Retained stdout plus stderr bytes. Zero still permits empty output.
55    pub max_output_bytes: usize,
56    /// Additional interval for group termination and direct-child reaping.
57    pub cleanup_timeout: Duration,
58}
59
60impl Default for Limits {
61    fn default() -> Self {
62        Self {
63            timeout: Duration::from_secs(30),
64            max_output_bytes: 8 * 1024 * 1024,
65            cleanup_timeout: Duration::from_secs(1),
66        }
67    }
68}

70/// Why capture polling ended. This is not a solver result.
71#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
72#[serde(rename_all = "snake_case")]
73pub enum Stop {
74    /// Direct child reaped and both streams closed without a capture fault.
75    /// This establishes no descendant-termination or system-quiescence property.
76    Completed,
77    /// The invocation deadline was reached.
78    Deadline,
79    /// The caller requested cancellation; partial capture and cleanup are retained.
80    Cancelled,
81    /// A byte beyond the combined retained-output ceiling was observed.
82    OutputLimit,
83    /// A capture or child-lifetime operation failed; inspect the failure fields.
84    Failure,
85}

87/// Direct-child exit evidence, independent of the reason capture stopped.
88#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
89pub struct Exit {
90    /// Numeric normal-exit code, absent for signal termination.
91    pub code: Option<i32>,
92    /// Terminating POSIX signal, absent for normal exit.
93    pub signal: Option<i32>,
94}
95
96impl From<ExitStatus> for Exit {
97    fn from(value: ExitStatus) -> Self {
98        #[cfg(unix)]
99        let signal = {
100            use std::os::unix::process::ExitStatusExt;
101            value.signal()
102        };
103        #[cfg(not(unix))]
104        let signal = None;
105        Self {
106            code: value.code(),
107            signal,
108        }
109    }
110}

112/// Failed operation; error text is diagnostic rather than a classification.
113#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize)]
114#[serde(rename_all = "snake_case")]
115pub enum Operation {
116    /// Making a child pipe nonblocking.
117    ConfigurePipe,
118    /// Reading the standard-output pipe.
119    ReadStdout,
120    /// Reading the standard-error pipe.
121    ReadStderr,
122    /// Reserving retained byte storage before copying output.
123    RetainBytes,
124    /// Observing child termination without consuming its wait status.
125    ObserveExit,
126    /// Sending a termination signal to the still-owned process group.
127    TerminateGroup,
128    /// Reaping the direct child without a blocking wait.
129    ReapChild,
130}

132/// An OS or allocation failure associated with a specific capture operation.
133#[derive(Debug)]
134pub struct Failure {
135    operation: Operation,
136    cause: io::Error,
137}
138
139impl Failure {
140    /// Operation that failed.
141    #[must_use]
142    pub const fn operation(&self) -> Operation {
143        self.operation
144    }
145    /// Original diagnostic error.
146    #[must_use]
147    pub const fn cause(&self) -> &io::Error {
148        &self.cause
149    }
150    #[cfg(any(target_os = "linux", target_os = "macos"))]
151    pub(super) fn new(operation: Operation, cause: io::Error) -> Self {
152        Self { operation, cause }
153    }
154}
155
156impl fmt::Display for Failure {
157    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
158        write!(f, "{:?}: {}", self.operation, self.cause)
159    }
160}
161impl std::error::Error for Failure {
162    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
163        Some(&self.cause)
164    }
165}

167/// Refusal before a child is started.
168#[derive(Debug)]
169pub enum StartError {
170    /// The caller requested cancellation before the child was started.
171    Cancelled,
172    /// The strong process-group backend is unavailable on this platform.
173    UnsupportedPlatform,
174    /// Executable or working-directory path is relative.
175    RelativePath,
176    /// An invocation or cleanup deadline cannot be represented by `Instant`.
177    DeadlineOverflow,
178    /// Starting the requested executable failed.
179    Spawn(io::Error),
180}
181impl fmt::Display for StartError {
182    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
183        match self {
184            Self::Cancelled => f.write_str("process invocation cancelled before spawn"),
185            Self::UnsupportedPlatform => {
186                f.write_str("bounded process groups require Linux or macOS")
187            }
188            Self::RelativePath => f.write_str("process executable and directory must be absolute"),
189            Self::DeadlineOverflow => f.write_str("process deadline is not representable"),
190            Self::Spawn(error) => write!(f, "spawn: {error}"),
191        }
192    }
193}
194impl std::error::Error for StartError {
195    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
196        if let Self::Spawn(error) = self {
197            Some(error)
198        } else {
199            None
200        }
201    }
202}

204/// Immutable retained bytes and direct-child evidence.
205#[derive(Debug)]
206pub struct Capture {
207    child_id: u32,
208    stop: Stop,
209    exit: Option<Exit>,
210    elapsed: Duration,
211    stdout: Vec<u8>,
212    stderr: Vec<u8>,
213    failure: Option<Failure>,
214    cleanup_failure: Option<Failure>,
215}
216impl Capture {
217    /// Direct child created by this invocation; identity is not a liveness claim.
218    #[must_use]
219    pub const fn child_id(&self) -> u32 {
220        self.child_id
221    }
222    /// Reason capture polling stopped.
223    #[must_use]
224    pub const fn stop(&self) -> Stop {
225        self.stop
226    }
227    /// Direct-child exit if successfully reaped.
228    #[must_use]
229    pub const fn exit(&self) -> Option<Exit> {
230        self.exit
231    }
232    /// Host elapsed time including the initial cleanup attempt.
233    #[must_use]
234    pub const fn elapsed(&self) -> Duration {
235        self.elapsed
236    }
237    /// Exact retained standard-output prefix.
238    #[must_use]
239    pub fn stdout(&self) -> &[u8] {
240        &self.stdout
241    }
242    /// Exact retained standard-error prefix.
243    #[must_use]
244    pub fn stderr(&self) -> &[u8] {
245        &self.stderr
246    }
247    /// Strict UTF-8 view; no replacement characters are inserted.
248    ///
249    /// # Errors
250    /// Returns the first invalid UTF-8 sequence, including in a stopped prefix.
251    pub fn stdout_text(&self) -> Result<&str, std::str::Utf8Error> {
252        std::str::from_utf8(&self.stdout)
253    }
254    /// Strict UTF-8 view of retained standard error.
255    ///
256    /// # Errors
257    /// Returns the first invalid UTF-8 sequence.
258    pub fn stderr_text(&self) -> Result<&str, std::str::Utf8Error> {
259        std::str::from_utf8(&self.stderr)
260    }
261    /// Primary capture fault, without replacing a preceding limit/deadline stop.
262    #[must_use]
263    pub const fn failure(&self) -> Option<&Failure> {
264        self.failure.as_ref()
265    }
266    /// First fault during the bounded cleanup attempt.
267    #[must_use]
268    pub const fn cleanup_failure(&self) -> Option<&Failure> {
269        self.cleanup_failure.as_ref()
270    }
271}

273/// Capture plus any direct child whose cleanup still requires caller action.
274#[derive(Debug)]
275#[must_use = "inspect capture and handle pending cleanup explicitly"]
276pub struct Outcome {
277    capture: Capture,
278    pending: Option<PendingChild>,
279}
280impl Outcome {
281    /// Evidence retained even if cleanup remains unresolved.
282    #[must_use]
283    pub const fn capture(&self) -> &Capture {
284        &self.capture
285    }
286    /// Whether bounded cleanup left an owned direct child.
287    #[must_use]
288    pub const fn cleanup_pending(&self) -> bool {
289        self.pending.is_some()
290    }
291    /// Separate evidence from the explicit cleanup obligation.
292    #[must_use]
293    pub fn into_parts(self) -> (Capture, Option<PendingChild>) {
294        (self.capture, self.pending)
295    }

298/// Exclusive ownership of a direct child not yet successfully reaped.
299///
300/// `Drop` performs no wait, signal, or thread creation. Call [`Self::retry`] or
301/// explicitly [`Self::abandon`] at the application's failure boundary. Neither
302/// the handle nor its numeric ID proves that escaped descendants have stopped.
303#[derive(Debug)]
304#[must_use = "retry or explicitly report abandoned child ownership"]
305pub struct PendingChild {
306    child: Child,
307    #[cfg(any(target_os = "linux", target_os = "macos"))]
308    group_owned: bool,
309}
310impl PendingChild {
311    /// Numeric direct-child identity retained for failure reporting.
312    #[must_use]
313    pub fn id(&self) -> u32 {
314        self.child.id()
315    }
316    /// Attempt group termination and nonblocking reaping for one more interval.
317    /// A failed attempt returns ownership again, with the first cleanup fault.
318    pub fn retry(self, timeout: Duration) -> Cleanup {
319        #[cfg(any(target_os = "linux", target_os = "macos"))]
320        {
321            posix::cleanup(self, timeout)
322        }
323        #[cfg(not(any(target_os = "linux", target_os = "macos")))]
324        {
325            let _ = timeout;
326            Cleanup {
327                exit: None,
328                pending: Some(self),
329                failure: None,
330            }
331        }
332    }
333    /// Explicitly stop owning cleanup and return the unreaped child ID.
334    ///
335    /// This is a failure disposition, never successful termination. It performs
336    /// no hidden wait and must be retained in the application's failure report.
337    #[must_use]
338    pub fn abandon(self) -> u32 {
339        self.child.id()
340    }
341}

344#[derive(Debug)]
345#[must_use = "retain cleanup failure and handle any returned child"]
346pub struct Cleanup {
347    /// Reaped direct-child exit, when established.
348    pub exit: Option<Exit>,
349    /// Ownership returned if reaping was not established.
350    pub pending: Option<PendingChild>,
351    /// First termination or reap failure from this attempt.
352    pub failure: Option<Failure>,

355/// Start and capture one invocation under the strong process-group contract.
356///
357/// Environment variables are inherited by `Command`; the caller chooses the
358/// executable and working directory explicitly. No library stdout/stderr writes
359/// occur. Memory retained for pipe bytes is linear in the combined byte ceiling.
360///
361/// # Errors
362/// Returns a typed refusal only before spawning. Every post-spawn failure
363/// returns an [`Outcome`] with partial capture and explicit cleanup ownership.
364pub fn invoke(invocation: Invocation<'_>, limits: Limits) -> Result<Outcome, StartError> {
365    invoke_with_cancellation(invocation, limits, &AtomicBool::new(false))
366}
367
368/// Capture a child while observing a caller-owned cancellation flag.
369///
370/// The caller may set the flag from another thread or its own signal handler.
371/// This library installs no handlers. A set flag refuses a new launch; after
372/// spawn it stops polling and performs the same bounded, ownership-checked group
373/// cleanup as a deadline. Cancellation never shortens the cleanup obligation.
374/// The flag must remain set once cancellation is requested.
375///
376/// # Errors
377/// Returns [`StartError::Cancelled`] before spawn, or the refusals of [`invoke`].
378pub fn invoke_with_cancellation(
379    invocation: Invocation<'_>,
380    limits: Limits,
381    cancelled: &AtomicBool,
382) -> Result<Outcome, StartError> {
383    start(invocation, limits, false, cancelled)
384}

386/// Capture a helper whose descendants inherit its process group.
387///
388/// Unlike [`invoke`], nonzero or signalled helper completion also attempts group
389/// termination while its waitable leader reserves its group ID, before reaping it.
390/// This covers failed helpers whose solver closed both output pipes. It does
391/// not certify escaped-descendant cleanup or provide their wait status. The
392/// trusted helper must return zero only after waiting for its sole solver and
393/// publishing the solver's separate exit record. Zero is not a general process
394/// tree completion guarantee; the caller must validate that additional record.
395/// This extra supervision is excluded from the ordinary timing protocol.
396///
397/// # Errors
398/// Returns the same pre-spawn refusals as [`invoke`]; post-spawn faults retain
399/// bounded capture and explicit helper cleanup ownership.
400pub fn invoke_supervised(
401    invocation: Invocation<'_>,
402    limits: Limits,
403) -> Result<Outcome, StartError> {
404    invoke_supervised_with_cancellation(invocation, limits, &AtomicBool::new(false))
405}

407/// Supervised helper capture with the cancellation contract of
408/// [`invoke_with_cancellation`] and the group ownership of [`invoke_supervised`].
409///
410/// # Errors
411/// Returns the same pre-spawn refusals as [`invoke_with_cancellation`].
412pub fn invoke_supervised_with_cancellation(
413    invocation: Invocation<'_>,
414    limits: Limits,
415    cancelled: &AtomicBool,
416) -> Result<Outcome, StartError> {
417    start(invocation, limits, true, cancelled)
418}
419
420fn start(
421    invocation: Invocation<'_>,
422    limits: Limits,
423    supervised: bool,
424    cancelled: &AtomicBool,
425) -> Result<Outcome, StartError> {
426    if cancelled.load(Ordering::Relaxed) {
427        return Err(StartError::Cancelled);
428    }
429    if !invocation.executable.is_absolute() || !invocation.directory.is_absolute() {
430        return Err(StartError::RelativePath);
431    }
432    let started = Instant::now();
433    let deadline = started
434        .checked_add(limits.timeout)
435        .ok_or(StartError::DeadlineOverflow)?;
436    deadline
437        .checked_add(limits.cleanup_timeout)
438        .ok_or(StartError::DeadlineOverflow)?;
439    #[cfg(any(target_os = "linux", target_os = "macos"))]
440    {
441        posix::invoke(invocation, limits, started, deadline, supervised, cancelled)
442    }
443    #[cfg(not(any(target_os = "linux", target_os = "macos")))]
444    {
445        let _ = (deadline, supervised);
446        Err(StartError::UnsupportedPlatform)
447    }
