1//! Interrupt registration for a process that supervises bounded children.
2//!
3//! The campaigns receive only a cancellation flag; this guard makes SIGINT and
4//! SIGTERM set it. A handler performs no allocation, reporting, waits or group
5//! signalling; the supervised capture's normal cleanup keeps those
6//! responsibilities and its ownership checks.
7
8use std::io;
9use std::sync::{Arc, atomic::AtomicBool};
10
11use signal_hook::{
12    SigId,
13    consts::signal::{SIGINT, SIGTERM},
14};
15
16/// SIGINT and SIGTERM registered to set a cancellation flag, until dropped.
17pub struct Interrupts {
18    registrations: Vec<SigId>,
19}
20
21impl Interrupts {
22    /// Register SIGINT and SIGTERM to set `cancelled`.
23    ///
24    /// # Errors
25    /// Returns the operating system's refusal to register either signal; any
26    /// registration already made is removed when the partial guard drops.
27    pub fn install(cancelled: &Arc<AtomicBool>) -> io::Result<Self> {
28        let mut installed = Self {
29            registrations: Vec::new(),
30        };
31        for signal in [SIGINT, SIGTERM] {
32            installed
33                .registrations
34                .push(signal_hook::flag::register(signal, Arc::clone(cancelled))?);
35        }
36        Ok(installed)
37    }
38}
39
40impl Drop for Interrupts {
41    fn drop(&mut self) {
42        // Unregistering removes these actions, not the operating system's
43        // handler; the guard lives to process-entry return, so this is a
44        // shutdown action, not a way to replace signal dispositions temporarily.
45        for registration in self.registrations.drain(..) {
46            signal_hook::low_level::unregister(registration);
47        }
48    }
49}
