1//! Separate fresh-helper observations of solver-child peak resident memory.
2//!
3//! The parent uses [`super::invoke_supervised`] to capture a fresh helper. The helper calls
4//! [`measure`] exactly once, inheriting its parent's process group, stdout and
5//! stderr. Thus the parent's deadline/output stop signals the helper and solver
6//! together. The helper exclusively waits for the solver; the parent exclusively
7//! waits for the helper. Neither uses another reaper or cumulative runner usage.
8//! As with ordinary capture, this does not certify escaped descendant cleanup.
9

18/// The first argument that makes a campaign's helper executable measure one
19/// child: `HELPER RECORD EXECUTABLE [ARGUMENT…]`, dispatched by [`run_helper`].
20pub const HELPER_COMMAND: &str = "__measure-child";
21
22/// The measurement helper's whole behaviour, for the executable a campaign
23/// launches as its helper.
24///
25/// `arguments` follow [`HELPER_COMMAND`]: the new record's path, then the
26/// absolute executable to measure and its unmodified arguments. Measures that
27/// one child from the current directory with [`measure_to_file`], in this
28/// fresh process; the child's own output passes through. A missing argument or
29/// a failed measurement is reported to `diagnostics` and exits with status 2.
30pub fn run_helper(
31    arguments: impl IntoIterator<Item = OsString>,
32    diagnostics: &mut impl Write,
33) -> ExitCode {
34    let mut arguments = arguments.into_iter();
35    let (Some(record), Some(executable)) = (arguments.next(), arguments.next()) else {
36        let _ = writeln!(
37            diagnostics,
38            "usage: {HELPER_COMMAND} RECORD EXECUTABLE [ARGUMENT...]"
39        );
40        return ExitCode::from(2);
41    };
42    let child_arguments: Vec<OsString> = arguments.collect();
43    let result = std::env::current_dir()
44        .map_err(RecordError::Io)
45        .and_then(|directory| {
46            measure_to_file(
47                super::Invocation {
48                    executable: std::path::Path::new(&executable),
49                    arguments: &child_arguments,
50                    directory: &directory,
51                },
52                std::path::Path::new(&record),
53            )
54        });
55    match result {
56        Ok(()) => ExitCode::SUCCESS,
57        Err(error) => {
58            let _ = writeln!(diagnostics, "measurement helper: {error}");
59            ExitCode::from(2)
60        }
61    }
62}

66/// Native `ru_maxrss` unit, before the explicit conversion to bytes.
67#[derive(Clone, Copy, Debug, PartialEq, Eq, Deserialize, Serialize)]
68#[serde(rename_all = "snake_case")]
69pub enum Unit {
70    /// macOS reports bytes.
71    Bytes,
72    /// Linux reports kibibytes.
73    Kibibytes,
74}
75
76impl Unit {
77    /// Native reporting unit on the current supported platform.
78    #[must_use]
79    pub const fn current() -> Option<Self> {
80        if cfg!(target_os = "macos") {
81            Some(Self::Bytes)
82        } else if cfg!(target_os = "linux") {
83            Some(Self::Kibibytes)
84        } else {
85            None
86        }
87    }
88}

90/// Resource evidence from one fresh helper's waited-for solver child.
91///
92/// `RUSAGE_CHILDREN` excludes the helper itself. The operating system may include
93/// usage propagated from descendants reaped by the solver; this is neither a
94/// simultaneous process-tree RSS sum nor device-memory measurement.
95#[derive(Clone, Copy, Debug, PartialEq, Eq, Deserialize, Serialize)]
96#[serde(deny_unknown_fields)]
97pub struct Measurement {
98    /// Format version for the private helper record.
99    pub schema: u32,
100    /// Solver process ID, distinct from the captured helper process.
101    pub child: u32,
102    /// Solver normal-exit code; absent on signal termination.
103    pub exit_code: Option<i32>,
104    /// Solver termination signal; absent on normal exit.
105    pub signal: Option<i32>,
106    /// Unconverted nonnegative `ru_maxrss` value.
107    pub raw_max_rss: u64,
108    /// Operating-system unit of the raw observation.
109    pub raw_unit: Unit,
110    /// Checked conversion of the raw observation to bytes.
111    pub peak_rss_bytes: u64,
112}
113
114impl Measurement {
115    /// Validate the record's version, exit evidence and unit conversion.
116    #[must_use]
117    pub fn valid(self) -> bool {
118        self.schema == 1
119            && self.child > 1
120            && (self.exit_code.is_some() != self.signal.is_some())
121            && self.exit_code.is_none_or(|code| (0..=255).contains(&code))
122            && self.signal.is_none_or(|signal| signal > 0)
123            && bytes(self.raw_max_rss, self.raw_unit) == Some(self.peak_rss_bytes)
124    }
125}
126
127fn bytes(raw: u64, unit: Unit) -> Option<u64> {
128    match unit {
129        Unit::Bytes => Some(raw),
130        Unit::Kibibytes => raw.checked_mul(1024),
131    }
132}

134/// Failure to obtain one complete child usage observation.
135#[derive(Debug)]
136pub enum Error {
137    /// Native units have not been qualified on this platform.
138    UnsupportedPlatform,
139    /// Executable or directory was not absolute.
140    RelativePath,
141    /// The supposedly fresh helper has already accumulated child usage.
142    NotFresh,
143    /// The OS could not start or wait for the child, or obtain its usage.
144    Operation(io::Error),
145    /// The OS returned unrepresentable usage or exit evidence.
146    InvalidUsage,
147}
148impl fmt::Display for Error {
149    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
150        match self {
151            Self::UnsupportedPlatform => f.write_str("child RSS units require macOS or Linux"),
152            Self::RelativePath => {
153                f.write_str("child RSS executable and directory must be absolute")
154            }
155            Self::NotFresh => f.write_str("child RSS helper has previous waited-for child usage"),
156            Self::Operation(error) => write!(f, "child RSS operation: {error}"),
157            Self::InvalidUsage => f.write_str("invalid child RSS or exit evidence"),
158        }
159    }
160}
161impl std::error::Error for Error {
162    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
163        match self {
164            Self::Operation(error) => Some(error),
165            _ => None,
166        }
167    }
168}

170/// Run and wait for one solver in a fresh, separately bounded helper process.
171///
172/// Call only once in a fresh helper that starts no other children. This operation
173/// deliberately inherits its process group and stdout/stderr. The supervising
174/// process must apply [`super::invoke_supervised`]'s deadline and output bounds; calling this
175/// directly in the campaign would mix cumulative usage and remove that bound.
176/// The blocking wait has exactly one owner. No process tree or device RSS is
177/// inferred. The returned elapsed helper execution must never enter timed runs.
178///
179/// # Errors
180/// Returns unsupported platform, path, freshness, spawn/wait/usage or conversion
181/// failures. A wait error requires the supervising process to terminate the
182/// still-owned group; the helper cannot establish successful solver completion.
183pub fn measure(invocation: super::Invocation<'_>) -> Result<Measurement, Error> {
184    if !invocation.executable.is_absolute() || !invocation.directory.is_absolute() {
185        return Err(Error::RelativePath);
186    }
187    #[cfg(any(target_os = "linux", target_os = "macos"))]
188    {
189        use nix::sys::resource::{UsageWho, getrusage};
190        use std::os::unix::process::ExitStatusExt;
191        use std::process::{Command, Stdio};
192
193        let usage = || {
194            getrusage(UsageWho::RUSAGE_CHILDREN)
195                .map_err(|error| Error::Operation(io::Error::from(error)))
196        };
197        if usage()?.max_rss() != 0 {
198            return Err(Error::NotFresh);
199        }
200        let mut child = Command::new(invocation.executable)
201            .args(invocation.arguments)
202            .current_dir(invocation.directory)
203            .stdin(Stdio::null())
204            .spawn()
205            .map_err(Error::Operation)?;
206        let id = child.id();
207        let exit = child.wait().map_err(Error::Operation)?;
208        let raw = u64::try_from(usage()?.max_rss()).map_err(|_| Error::InvalidUsage)?;
209        let unit = if cfg!(target_os = "linux") {
210            Unit::Kibibytes
211        } else {
212            Unit::Bytes
213        };
214        let measurement = Measurement {
215            schema: 1,
216            child: id,
217            exit_code: exit.code(),
218            signal: exit.signal(),
219            raw_max_rss: raw,
220            raw_unit: unit,
221            peak_rss_bytes: bytes(raw, unit).ok_or(Error::InvalidUsage)?,
222        };
223        if !measurement.valid() {
224            return Err(Error::InvalidUsage);
225        }
226        Ok(measurement)
227    }
228    #[cfg(not(any(target_os = "linux", target_os = "macos")))]
229    {
230        Err(Error::UnsupportedPlatform)
231    }
