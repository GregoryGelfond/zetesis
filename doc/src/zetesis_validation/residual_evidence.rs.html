19/// Fixed input filename in the invocation's private directory.
20pub const REQUEST_NAME: &str = "request.json";
21/// Fixed final artifact filename in the same directory.
22pub const ARTIFACT_NAME: &str = "samples.bin";
23const REQUEST_LIMIT: u64 = 1024;

25fn invalid(message: &'static str) -> io::Error {
26    io::Error::new(io::ErrorKind::InvalidData, message)
27}

29/// Explicit request bound to one invocation. The destination is always a sibling of the request.
30#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
31#[serde(deny_unknown_fields)]
32pub struct Request {
33    /// Request schema.
34    pub version: u32,
35    /// Unique identifier for this qualification slot.
36    pub nonce: [u8; 16],
37    /// Maximum bytes the child may publish for this optional artifact.
38    pub max_bytes: u64,
39}
40
41impl Request {
42    fn validate(&self) -> io::Result<()> {
43        if self.version != 1 || self.max_bytes > MAX_BYTES {
44            return Err(invalid("unsupported residual evidence request"));
45        }
46        Ok(())
47    }
48
49    /// Read a bounded qualification request.
50    ///
51    /// # Errors
52    /// Returns file, schema and size errors without affecting a solve.
53    pub fn read(path: &Path) -> io::Result<Self> {
54        let file = regular_file(path)?;
55        let mut bytes = Vec::new();
56        file.take(REQUEST_LIMIT + 1).read_to_end(&mut bytes)?;
57        if bytes.len() as u64 > REQUEST_LIMIT {
58            return Err(invalid("residual evidence request exceeds its byte limit"));
59        }
60        let request: Self = serde_json::from_slice(&bytes).map_err(io::Error::other)?;
61        request.validate()?;
62        Ok(request)
63    }
64
65    /// Create a request without replacing any earlier one.
66    ///
67    /// # Errors
68    /// Returns schema, creation or write failures.
69    pub fn write(&self, path: &Path) -> io::Result<()> {
70        self.validate()?;
71        let mut file = OpenOptions::new().create_new(true).write(true).open(path)?;
72        serde_json::to_writer(&mut file, self).map_err(io::Error::other)?;
73        file.flush()
74    }
75}

77fn regular_file(path: &Path) -> io::Result<File> {
78    if !path.symlink_metadata()?.file_type().is_file() {
79        return Err(invalid("residual evidence requires a regular file"));
80    }
81    let mut options = OpenOptions::new();
82    options.read(true);
83    #[cfg(unix)]
84    {
85        use rustix::fs::OFlags;
86        use std::os::unix::fs::OpenOptionsExt;
87        options.custom_flags((OFlags::NOFOLLOW | OFlags::NONBLOCK).bits().cast_signed());
88    }
89    let file = options.open(path)?;
90    if !file.metadata()?.is_file() {
91        return Err(invalid("residual evidence file changed kind"));
92    }
93    Ok(file)
94}

96/// Revalidate the retained bytes immediately before publishing their report.
97/// Use the same regular-file boundary as the initial decode, and never read
98/// past the offered length plus the single byte needed to detect growth.
99pub(crate) fn verify_digest(path: &Path, bytes: u64, expected: &str) -> io::Result<()> {
100    let file = regular_file(path)?;
101    if bytes > MAX_BYTES || file.metadata()?.len() != bytes {
102        return Err(invalid("retained residual evidence changed length"));
103    }
104    let mut hash = Sha256::new();
105    let actual = io::copy(&mut file.take(bytes + 1), &mut hash)?;
106    if actual != bytes || crate::digest::spelled(&hash.finalize()) != expected {
107        return Err(invalid(
108            "retained residual evidence changed after validation",
109        ));
110    }
111    Ok(())
112}

114/// Validated record values and seal for one invocation.
115#[derive(Debug)]
116pub struct Artifact {
117    header: Header,
118    worlds: Vec<World>,
119    batches: Vec<Batch>,
120    totals: metrics::Totals,
121    digest: String,
122    bytes: u64,
123}
124
125impl Artifact {
126    /// SHA-256 of the exact bytes validated.
127    #[must_use]
128    pub fn digest(&self) -> &str {
129        &self.digest
130    }
131    /// Exact size of the sealed artifact.
132    #[must_use]
133    pub const fn byte_len(&self) -> u64 {
134        self.bytes
135    }
136}

138/// Authenticate the file against its request and the summary from this same process.
139///
140/// # Errors
141/// Refuses a truncated or oversized file, a mismatched identity, any inconsistent
142/// record, or disagreement between the retained observations and the summary.
143pub fn read(
144    path: &Path,
145    request: &Request,
146    summary: &ResidualRunSamples,
147    complete: bool,
148) -> io::Result<Artifact> {
149    request.validate()?;
150    summary
151        .check(None)
152        .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
153    for pool in [&summary.worlds, &summary.batches] {
154        // Version one admits each fixed reservoir once. It loses the entire
155        // frame on ordinal overflow and never silently thins a retained set.
156        let expected = pool.seen.map_or(0, |n| n.min(pool.capacity as u64));
157        if pool.retained as u64 != expected || pool.seen.is_none() && pool.capacity != 0 {
158            return Err(invalid(
159                "residual evidence has a thinned or lost population",
160            ));
161        }
162    }
163    let groups = summary
164        .groups
165        .as_ref()
166        .ok_or_else(|| invalid("terminal residual sample coverage unavailable"))?;
167    let file = regular_file(path)?;
168    let bytes = file.metadata()?.len();
169    if bytes > request.max_bytes {
170        return Err(invalid("residual evidence exceeds its offered bytes"));
171    }
172    let mut source = BufReader::new(file);
173    let mut hash = Sha256::new();
174    let header = Header::decode(&block(&mut source, &mut hash)?)?;
175    if header.byte_len() != Some(bytes)
176        || header.nonce != request.nonce
177        || header.complete != complete
178        || header.seed != summary.seed
179        || header.worlds != summary.worlds.seen
180        || header.batches != summary.batches.seen
181        || header.retained_worlds as usize != summary.worlds.retained
182        || header.retained_batches as usize != summary.batches.retained
183    {
184        return Err(invalid(
185            "residual evidence header disagrees with request or native result",
186        ));
187    }
188    let worlds = records(
189        header.retained_worlds,
190        &mut source,
191        &mut hash,
192        World::decode,
193    )?;
194    let batches = records(
195        header.retained_batches,
196        &mut source,
197        &mut hash,
198        Batch::decode,
199    )?;
200    if source.read(&mut [0; 1])? != 0 {
201        return Err(invalid("trailing residual evidence"));
202    }
203    let artifact = Artifact {
204        header,
205        worlds,
206        batches,
207        totals: metrics::Totals::from_groups(groups),
208        bytes,
209        digest: crate::digest::spelled(&hash.finalize()),
210    };
211    metrics::check(&artifact, groups)?;
212    Ok(artifact)
213}

215fn block<const N: usize>(source: &mut impl Read, hash: &mut Sha256) -> io::Result<[u8; N]> {
216    let mut bytes = [0; N];
217    source.read_exact(&mut bytes)?;
218    hash.update(bytes);
219    Ok(bytes)
220}
221
222fn records<T, const N: usize>(
223    n: u32,
224    source: &mut impl Read,
225    hash: &mut Sha256,
226    decode: impl Fn(&[u8; N]) -> io::Result<T>,
227) -> io::Result<Vec<T>> {
228    let mut rows = Vec::new();
229    rows.try_reserve_exact(n as usize)
230        .map_err(io::Error::other)?;
231    for _ in 0..n {
232        rows.push(decode(&block(source, hash)?)?);
233    }
234    Ok(rows)
235}
