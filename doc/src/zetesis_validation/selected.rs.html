1//! Complete selected-curated comparison, composed from shared validation APIs.
2//!
3//! The immutable 24-case target has no show projection or external source roots.
4//! Its clingo displays can therefore be compared against the recorded full-model
5//! contracts; that implication does not apply to arbitrary ASP sources. Native
6//! full atoms come from typed JSON records, independently of shown values.
7//!
8//! Requests select named executables and a bounded execution configuration;
9//! source paths, all-model enumeration and output controls belong to the campaign.
10//! Primary executables and every curated input are sealed before and after runs.
11//! Seals do not cover dynamic libraries, the inherited environment, hardware or
12//! changes that are restored between checks. Completed children do not establish
13//! descendant termination or system quiescence. Solvers must be trusted.
14

100/// Execution requests only; no extra source, constants, stdin or output options.
101/// Unspecified solver budgets retain the sealed executable's defaults.
102#[derive(Clone, Copy, Debug, Serialize)]
103pub struct NativeExecution {
104    /// Requested execution hardware.
105    pub backend: Backend,
106    /// Requested reduct procedure.
107    pub oracle: Oracle,
108    /// Requested materialization policy.
109    pub grounder: Grounder,
110    /// Optional eager-formula join strategy. Omission preserves the sealed
111    /// executable's default and permits comparison with versions before this flag.
112    #[serde(skip_serializing_if = "Option::is_none")]
113    pub formula_joins: Option<FormulaJoins>,
114    /// Optional formula search method. Omission preserves the sealed
115    /// executable's default and permits comparison with versions before this flag.
116    #[serde(skip_serializing_if = "Option::is_none")]
117    pub search: Option<SearchMethod>,
118    /// Closure worker request.
119    pub workers: NonZeroUsize,
120    /// Formula completion worker request.
121    pub completion_workers: NonZeroUsize,
122    /// Native candidate batch ceiling.
123    pub batch_size: NonZeroUsize,
124    /// Logical completion scratch allowance; zero is a valid requested ceiling.
125    pub max_completion_scratch_bytes: u64,
126    /// Cooperative deadline passed as `--time-limit`, in whole seconds. Absent
127    /// means no deadline; a deadline changes what the solver polls at every
128    /// charged unit, so it is part of the profile's identity.
129    #[serde(skip_serializing_if = "Option::is_none")]
130    pub time_limit_seconds: Option<std::num::NonZeroU64>,
131    /// Explicit formula expansion-work ceiling. Omission preserves the sealed
132    /// executable's default; an override is part of the measured profile.
133    #[serde(skip_serializing_if = "Option::is_none")]
134    pub max_expansion_work: Option<usize>,
135}
136impl NativeExecution {
137    /// The command-line arguments that request this execution of the native
138    /// solver, in one fixed order: the backend, oracle, grounder, workers,
139    /// completion workers, batch size, completion scratch and model count,
140    /// then the join strategy, search method, deadline and expansion ceiling when the
141    /// profile names them. A campaign appends its output flags and the input.
142    pub(crate) fn arguments(&self) -> Vec<std::ffi::OsString> {
143        let values = [
144            ("--backend", self.backend.label().into()),
145            ("--oracle", self.oracle.label().into()),
146            ("--grounder", self.grounder.label().into()),
147            ("--workers", self.workers.to_string()),
148            ("--completion-workers", self.completion_workers.to_string()),
149            ("--batch-size", self.batch_size.to_string()),
150            (
151                "--max-completion-scratch-bytes",
152                self.max_completion_scratch_bytes.to_string(),
153            ),
154            ("--models", "0".into()),
155        ];
156        values
157            .into_iter()
158            .flat_map(|(flag, value)| [flag.into(), value.into()])
159            .chain(self.formula_joins.into_iter().flat_map(|joins| {
160                [
161                    std::ffi::OsString::from("--formula-joins"),
162                    std::ffi::OsString::from(joins.label()),
163                ]
164            }))
165            .chain(self.search.into_iter().flat_map(|search| {
166                [
167                    std::ffi::OsString::from("--search"),
168                    std::ffi::OsString::from(search.label()),
169                ]
170            }))
171            .chain(self.time_limit_seconds.into_iter().flat_map(|seconds| {
172                [
173                    std::ffi::OsString::from("--time-limit"),
174                    std::ffi::OsString::from(seconds.to_string()),
175                ]
176            }))
177            .chain(self.max_expansion_work.into_iter().flat_map(|work| {
178                [
179                    std::ffi::OsString::from("--max-expansion-work"),
180                    std::ffi::OsString::from(work.to_string()),
181                ]
182            }))
183            .collect()
184    }
185}

302/// A setup, integrity or report-publication failure, never semantic UNSAT.
303#[derive(Debug)]
304pub enum Error {
305    /// The selected corpus did not establish its pinned integrity contract.
306    Corpus(curated::Error),
307    /// The strong process backend is not supported on this platform.
308    UnsupportedPlatform,
309    /// A path or input-closure policy was refused before execution/publication.
310    Path {
311        /// Relevant requested path.
312        path: PathBuf,
313        /// Diagnostic policy explanation; not an interchange discriminator.
314        detail: &'static str,
315    },
316    /// A filesystem operation failed.
317    Io {
318        /// Relevant path.
319        path: PathBuf,
320        /// Original failure.
321        source: std::io::Error,
322    },
323    /// A bounded file or serialized report exceeded its inclusive byte allowance.
324    Bytes {
325        /// Named file or report.
326        path: PathBuf,
327        /// Inclusive byte ceiling.
328        limit: usize,
329    },
330    /// A final report could not be serialized.
331    Json(serde_json::Error),
332    /// Primary publication failed and scoped temporary-file cleanup also failed.
333    Cleanup {
334        /// Primary publication failure, retained without replacement.
335        primary: Box<Error>,
336        /// Secondary cleanup failure.
337        cleanup: std::io::Error,
338    },
339}
340impl fmt::Display for Error {
341    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
342        match self {
343            Self::Corpus(error) => write!(f, "{error}"),
344            Self::UnsupportedPlatform => {
345                f.write_str("selected campaign requires the Linux/macOS process backend")
346            }
347            Self::Path { path, detail } => write!(f, "{}: {detail}", path.display()),
348            Self::Io { path, source } => write!(f, "{}: {source}", path.display()),
349            Self::Bytes { path, limit } => {
350                write!(f, "{} exceeds byte ceiling {limit}", path.display())
351            }
352            Self::Json(error) => write!(f, "report JSON: {error}"),
353            Self::Cleanup { primary, cleanup } => {
354                write!(f, "{primary}; temporary cleanup: {cleanup}")
355            }
356        }
357    }
358}
359impl std::error::Error for Error {
360    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
361        match self {
362            Self::Corpus(error) => Some(error),
363            Self::Io { source, .. } => Some(source),
364            Self::Json(error) => Some(error),
365            Self::Cleanup { primary, .. } => Some(primary.as_ref()),
366            _ => None,
367        }
368    }
369}

371/// Complete retained campaign evidence. Only [`run`] constructs this owner.
372#[derive(Debug, Serialize)]
373pub struct Report {
374    schema: u32,
375    target: &'static str,
376    manifest_sha256: &'static str,
377    requested_execution: NativeExecution,
378    requested_limits: Limits,
379    full_model_comparison_scope: &'static str,
380    physical_execution_qualified: bool,
381    before: Vec<FileSeal>,
382    after: Vec<Change>,
383    cases: Vec<CaseResult>,
384    total_capture_bytes: usize,
385    unresolved_children: Vec<u32>,
386    faults: Vec<CampaignFault>,
387    #[serde(skip)]
388    destination: publication::Destination,
389    #[serde(skip)]
390    max_report_bytes: usize,
391}
392impl Report {
393    /// True only for all 24 complete exact comparisons and unchanged inputs.
394    #[must_use]
395    pub fn passed(&self) -> bool {
396        self.cases.len() == CASE_COUNT
397            && self
398                .cases
399                .iter()
400                .all(|case| case.decision() == Decision::Pass)
401            && self.after.iter().all(Change::unchanged)
402            && self.unresolved_children.is_empty()
403            && self.faults.is_empty()
404    }
405    /// Every attempted case, including all retained failures.
406    #[must_use]
407    pub fn cases(&self) -> &[CaseResult] {
408        &self.cases
409    }
410    /// Complete primary-input seals established before any solver starts.
411    #[must_use]
412    pub fn before(&self) -> &[FileSeal] {
413        &self.before
414    }
415    /// Rechecks performed after the last invocation, including read failures.
416    #[must_use]
417    pub fn after(&self) -> &[Change] {
418        &self.after
419    }
420    /// Effect failures retained independently of semantic comparison.
421    #[must_use]
422    pub fn faults(&self) -> &[CampaignFault] {
423        &self.faults
424    }
425    /// Child IDs explicitly abandoned after the bounded cleanup retry failed.
426    /// No further launches occur after this condition. This is diagnostic
427    /// evidence, not a live process handle or a descendant-termination claim.
428    #[must_use]
429    pub fn unresolved_children(&self) -> &[u32] {
430        &self.unresolved_children
431    }
432    /// Publish this evidence without overwriting an existing path.
433    /// The destination is revalidated before publication under the documented
434    /// exclusive-parent assumption; this is not an adversarial filesystem lease.
435    ///
436    /// # Errors
437    /// Refuses report aliases, existing destinations, serialization/byte ceilings
438    /// and filesystem failures. Temporary cleanup failures retain the primary cause.
439    pub fn publish(&self) -> Result<(), Error> {
440        publication::write(
441            &view::Published {
442                passed: self.passed(),
443                report: self,
444            },
445            &self.destination,
446            self.max_report_bytes,
447        )
448    }
449}

451/// Verify all inputs and the output path, then execute the complete selected target.
452/// Sources are copied byte-for-byte from the owned verified corpus into a private
453/// temporary directory. Every started invocation uses the shared process runner.
454/// Failure records are retained; no retry silently turns an incomplete solve into
455/// a pass. Unresolved direct-child cleanup stops further launches.
456///
457/// # Errors
458/// Returns setup/integrity errors before any solver launch. Individual invocation
459/// and semantic failures remain in the returned report. A temporary-input cleanup
460/// failure is also retained as campaign failure evidence.
461pub fn run(request: &Request<'_>) -> Result<Report, Error> {
462    run::campaign(request)
