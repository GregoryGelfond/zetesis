1//! Complete selected-curated comparison, composed from shared validation APIs.
2//!
3//! The immutable 24-case target has no show projection or external source roots.
4//! Its clingo displays can therefore be compared against the recorded full-model
5//! contracts; that implication does not apply to arbitrary ASP sources. Native
6//! full atoms come from typed JSON records, independently of shown values.
7//!
8//! Requests select named executables and a bounded execution configuration;
9//! source paths, all-model enumeration and output controls belong to the campaign.
10//! Primary executables and every curated input are sealed before and after runs.
11//! Seals do not cover dynamic libraries, the inherited environment, hardware or
12//! changes that are restored between checks. Completed children do not establish
13//! descendant termination or system quiescence. Solvers must be trusted.
14

100/// Execution requests only; no extra source, constants, stdin or output options.
101/// The ordinary public resource policy is requested; effective behavior belongs
102/// to the sealed executable. Historical executables may retain bounded defaults.
103#[derive(Clone, Copy, Debug, Serialize)]
104#[serde(tag = "resource_policy", rename = "ordinary")]
105pub struct NativeExecution {
106    /// Requested execution hardware.
107    pub backend: Backend,
108    /// Requested reduct procedure.
109    pub oracle: Oracle,
110    /// Requested materialization policy.
111    pub grounder: Grounder,
112    /// Optional eager-formula join strategy. Omission preserves the sealed
113    /// executable's default and permits comparison with versions before this flag.
114    #[serde(skip_serializing_if = "Option::is_none")]
115    pub formula_joins: Option<FormulaJoins>,
116    /// Optional formula search method. Omission preserves the sealed
117    /// executable's default and permits comparison with versions before this flag.
118    #[serde(skip_serializing_if = "Option::is_none")]
119    pub search: Option<SearchMethod>,
120    /// Requested worker threads.
121    pub threads: NonZeroUsize,
122    /// Explicit named-memory allowance in bytes. Omission requests the sealed
123    /// executable's ordinary default; neither form is a process RSS limit.
124    #[serde(skip_serializing_if = "Option::is_none")]
125    pub memory_bytes: Option<u64>,
126    /// Cooperative deadline passed as `--time-limit`, in whole seconds. Absent
127    /// means no deadline. A deadline can stop an otherwise completed run, so
128    /// it is part of the requested profile's identity.
129    #[serde(skip_serializing_if = "Option::is_none")]
130    pub time_limit_seconds: Option<std::num::NonZeroU64>,
131}
132impl NativeExecution {
133    /// Arguments for the ordinary public policy: algorithm selection, threads,
134    /// complete enumeration, then optional join/search, deadline and memory.
135    /// A campaign appends its output flags and input. No internal work, batch
136    /// or completion ceiling is requested.
137    pub(crate) fn arguments(&self) -> Vec<std::ffi::OsString> {
138        let values = [
139            ("--backend", self.backend.label().into()),
140            ("--oracle", self.oracle.label().into()),
141            ("--grounder", self.grounder.label().into()),
142            ("--threads", self.threads.to_string()),
143            ("--models", "0".into()),
144        ];
145        values
146            .into_iter()
147            .flat_map(|(flag, value)| [flag.into(), value.into()])
148            .chain(self.formula_joins.into_iter().flat_map(|joins| {
149                [
150                    std::ffi::OsString::from("--formula-joins"),
151                    std::ffi::OsString::from(joins.label()),
152                ]
153            }))
154            .chain(self.search.into_iter().flat_map(|search| {
155                [
156                    std::ffi::OsString::from("--search"),
157                    std::ffi::OsString::from(search.label()),
158                ]
159            }))
160            .chain(self.time_limit_seconds.into_iter().flat_map(|seconds| {
161                [
162                    std::ffi::OsString::from("--time-limit"),
163                    std::ffi::OsString::from(seconds.to_string()),
164                ]
165            }))
166            .chain(self.memory_bytes.into_iter().flat_map(|bytes| {
167                [
168                    std::ffi::OsString::from("--memory"),
169                    std::ffi::OsString::from(bytes.to_string()),
170                ]
171            }))
172            .collect()
173    }
174}

288/// A setup, integrity or report-publication failure, never semantic UNSAT.
289#[derive(Debug)]
290pub enum Error {
291    /// The selected corpus did not establish its pinned integrity contract.
292    Corpus(curated::Error),
293    /// The strong process backend is not supported on this platform.
294    UnsupportedPlatform,
295    /// A path or input-closure policy was refused before execution/publication.
296    Path {
297        /// Relevant requested path.
298        path: PathBuf,
299        /// Diagnostic policy explanation; not an interchange discriminator.
300        detail: &'static str,
301    },
302    /// A filesystem operation failed.
303    Io {
304        /// Relevant path.
305        path: PathBuf,
306        /// Original failure.
307        source: std::io::Error,
308    },
309    /// A bounded file or serialized report exceeded its inclusive byte allowance.
310    Bytes {
311        /// Named file or report.
312        path: PathBuf,
313        /// Inclusive byte ceiling.
314        limit: usize,
315    },
316    /// A final report could not be serialized.
317    Json(serde_json::Error),
318    /// Primary publication failed and scoped temporary-file cleanup also failed.
319    Cleanup {
320        /// Primary publication failure, retained without replacement.
321        primary: Box<Error>,
322        /// Secondary cleanup failure.
323        cleanup: std::io::Error,
324    },
325}
326impl fmt::Display for Error {
327    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
328        match self {
329            Self::Corpus(error) => write!(f, "{error}"),
330            Self::UnsupportedPlatform => {
331                f.write_str("selected campaign requires the Linux/macOS process backend")
332            }
333            Self::Path { path, detail } => write!(f, "{}: {detail}", path.display()),
334            Self::Io { path, source } => write!(f, "{}: {source}", path.display()),
335            Self::Bytes { path, limit } => {
336                write!(f, "{} exceeds byte ceiling {limit}", path.display())
337            }
338            Self::Json(error) => write!(f, "report JSON: {error}"),
339            Self::Cleanup { primary, cleanup } => {
340                write!(f, "{primary}; temporary cleanup: {cleanup}")
341            }
342        }
343    }
344}
345impl std::error::Error for Error {
346    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
347        match self {
348            Self::Corpus(error) => Some(error),
349            Self::Io { source, .. } => Some(source),
350            Self::Json(error) => Some(error),
351            Self::Cleanup { primary, .. } => Some(primary.as_ref()),
352            _ => None,
353        }
354    }
355}

357/// Complete retained campaign evidence. Only [`run`] constructs this owner.
358#[derive(Debug, Serialize)]
359pub struct Report {
360    schema: u32,
361    target: &'static str,
362    manifest_sha256: &'static str,
363    requested_execution: NativeExecution,
364    requested_limits: Limits,
365    full_model_comparison_scope: &'static str,
366    physical_execution_qualified: bool,
367    before: Vec<FileSeal>,
368    after: Vec<Change>,
369    cases: Vec<CaseResult>,
370    total_capture_bytes: usize,
371    unresolved_children: Vec<u32>,
372    faults: Vec<CampaignFault>,
373    #[serde(skip)]
374    destination: publication::Destination,
375    #[serde(skip)]
376    max_report_bytes: usize,
377}
378impl Report {
379    /// True only for all 24 complete exact comparisons and unchanged inputs.
380    #[must_use]
381    pub fn passed(&self) -> bool {
382        self.cases.len() == CASE_COUNT
383            && self
384                .cases
385                .iter()
386                .all(|case| case.decision() == Decision::Pass)
387            && self.after.iter().all(Change::unchanged)
388            && self.unresolved_children.is_empty()
389            && self.faults.is_empty()
390    }
391    /// Every attempted case, including all retained failures.
392    #[must_use]
393    pub fn cases(&self) -> &[CaseResult] {
394        &self.cases
395    }
396    /// Complete primary-input seals established before any solver starts.
397    #[must_use]
398    pub fn before(&self) -> &[FileSeal] {
399        &self.before
400    }
401    /// Rechecks performed after the last invocation, including read failures.
402    #[must_use]
403    pub fn after(&self) -> &[Change] {
404        &self.after
405    }
406    /// Effect failures retained independently of semantic comparison.
407    #[must_use]
408    pub fn faults(&self) -> &[CampaignFault] {
409        &self.faults
410    }
411    /// Child IDs explicitly abandoned after the bounded cleanup retry failed.
412    /// No further launches occur after this condition. This is diagnostic
413    /// evidence, not a live process handle or a descendant-termination claim.
414    #[must_use]
415    pub fn unresolved_children(&self) -> &[u32] {
416        &self.unresolved_children
417    }
418    /// Publish this evidence without overwriting an existing path.
419    /// The destination is revalidated before publication under the documented
420    /// exclusive-parent assumption; this is not an adversarial filesystem lease.
421    ///
422    /// # Errors
423    /// Refuses report aliases, existing destinations, serialization/byte ceilings
424    /// and filesystem failures. Temporary cleanup failures retain the primary cause.
425    pub fn publish(&self) -> Result<(), Error> {
426        publication::write(
427            &view::Published {
428                passed: self.passed(),
429                report: self,
430            },
431            &self.destination,
432            self.max_report_bytes,
433        )
434    }
435}
