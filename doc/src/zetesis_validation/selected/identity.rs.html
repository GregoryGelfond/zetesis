12/// A primary input's requested path, resolved file identity and exact byte seal.
13/// The bounded read must agree with the opened file's length before and after
14/// reading. This consistency check does not establish an immutable snapshot.
15#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
16pub struct FileSeal {
17    requested: PathBuf,
18    canonical: PathBuf,
19    bytes: u64,
20    sha256: String,
21    device: Option<u64>,
22    inode: Option<u64>,
23    #[serde(skip)]
24    limit: usize,
25}
26impl FileSeal {
27    /// Absolute path requested for this input, preserving symlink traversal.
28    #[must_use]
29    pub fn requested(&self) -> &Path {
30        &self.requested
31    }
32    /// Canonical path resolved at this check.
33    #[must_use]
34    pub fn canonical(&self) -> &Path {
35        &self.canonical
36    }
37    /// SHA-256 of the primary file's exact contents.
38    #[must_use]
39    pub fn sha256(&self) -> &str {
40        &self.sha256
41    }
42    /// Number of bytes hashed.
43    #[must_use]
44    pub const fn bytes(&self) -> u64 {
45        self.bytes
46    }
47
48    /// Compare observed file identity independently of each read's byte ceiling.
49    pub(crate) fn same_identity(&self, other: &Self) -> bool {
50        self.requested == other.requested
51            && self.canonical == other.canonical
52            && self.bytes == other.bytes
53            && self.sha256 == other.sha256
54            && self.device == other.device
55            && self.inode == other.inode
56    }
57}

59/// One after-run input check, retaining failure instead of inventing an identity.
60#[derive(Debug, Serialize)]
61pub struct Change {
62    path: PathBuf,
63    unchanged: bool,
64    after: Option<FileSeal>,
65    error: Option<String>,
66}
67impl Change {
68    /// Whether requested/resolved path, file identity and byte seal agree.
69    #[must_use]
70    pub const fn unchanged(&self) -> bool {
71        self.unchanged
72    }
73    /// Observed post-run seal, absent when the recheck failed.
74    #[must_use]
75    pub const fn after(&self) -> Option<&FileSeal> {
76        self.after.as_ref()
77    }
78    /// Original diagnostic if the post-run input could not be checked.
79    #[must_use]
80    pub fn error(&self) -> Option<&str> {
81        self.error.as_deref()
82    }
83}

85pub(crate) fn absolute(path: &Path) -> Result<PathBuf, Error> {
86    let path = std::path::absolute(path).map_err(|source| io(path, source))?;
87    if path.to_str().is_none() {
88        return Err(Error::Path {
89            path,
90            detail: "report paths require lossless UTF-8 representation",
91        });
92    }
93    Ok(path)
94}

96pub(crate) fn seal(path: &Path, limit: usize) -> Result<FileSeal, Error> {
97    let requested = absolute(path)?;
98    let canonical = std::fs::canonicalize(&requested).map_err(|source| io(&requested, source))?;
99    if canonical.to_str().is_none() {
100        return Err(Error::Path {
101            path: canonical,
102            detail: "canonical path requires lossless UTF-8 representation",
103        });
104    }
105    let mut file = File::open(&canonical).map_err(|source| io(&canonical, source))?;
106    let metadata = file.metadata().map_err(|source| io(&canonical, source))?;
107    if !metadata.is_file() {
108        return Err(Error::Path {
109            path: canonical,
110            detail: "sealed input must be a regular file",
111        });
112    }
113    if u128::from(metadata.len()) > limit as u128 {
114        return Err(Error::Bytes {
115            path: canonical,
116            limit,
117        });
118    }
119    let (bytes, sha256) = content_seal(&mut file, &canonical, limit, metadata.len(), |file| {
120        file.metadata().map(|metadata| metadata.len())
121    })?;
122    #[cfg(unix)]
123    let (device, inode) = {
124        use std::os::unix::fs::MetadataExt;
125        (Some(metadata.dev()), Some(metadata.ino()))
126    };
127    #[cfg(not(unix))]
128    let (device, inode) = (None, None);
129    Ok(FileSeal {
130        requested,
131        canonical,
132        bytes,
133        sha256,
134        device,
135        inode,
136        limit,
137    })
138}

140/// Hash a bounded stream, then require its length to agree with both metadata
141/// observations. Production obtains both lengths from the same open file.
142fn content_seal<R: Read>(
143    reader: &mut R,
144    path: &Path,
145    limit: usize,
146    before: u64,
147    after_length: impl FnOnce(&R) -> std::io::Result<u64>,
148) -> Result<(u64, String), Error> {
149    let mut digest = Sha256::new();
150    let mut bytes = 0usize;
151    let mut buffer = [0_u8; 8192];
152    loop {
153        let room = limit
154            .saturating_sub(bytes)
155            .saturating_add(1)
156            .min(buffer.len());
157        let count = reader
158            .read(&mut buffer[..room])
159            .map_err(|source| io(path, source))?;
160        if count == 0 {
161            break;
162        }
163        bytes = bytes.checked_add(count).ok_or_else(|| Error::Bytes {
164            path: path.to_owned(),
165            limit,
166        })?;
167        if bytes > limit {
168            return Err(Error::Bytes {
169                path: path.to_owned(),
170                limit,
171            });
172        }
173        digest.update(&buffer[..count]);
174    }
175    let bytes = u64::try_from(bytes).map_err(|_| Error::Bytes {
176        path: path.to_owned(),
177        limit,
178    })?;
179    let after = after_length(reader).map_err(|source| io(path, source))?;
180    if bytes != before || bytes != after {
181        let kind = if bytes < before {
182            std::io::ErrorKind::UnexpectedEof
183        } else {
184            std::io::ErrorKind::InvalidData
185        };
186        return Err(io(
187            path,
188            std::io::Error::new(
189                kind,
190                format!(
191                    "sealed input length disagrees: before {before} bytes, read {bytes} bytes, after {after} bytes"
192                ),
193            ),
194        ));
195    }
196    Ok((bytes, spelled(&digest.finalize())))
197}

199pub(crate) fn recheck(before: &FileSeal) -> Change {
200    match seal(&before.requested, before.limit) {
201        Ok(after) => Change {
202            path: before.requested.clone(),
203            unchanged: &after == before,
204            after: Some(after),
205            error: None,
206        },
207        Err(error) => Change {
208            path: before.requested.clone(),
209            unchanged: false,
210            after: None,
211            error: Some(error.to_string()),
212        },
213    }
214}

216/// What a native run's report says about the executable that did its work.
217#[derive(Clone, Debug, PartialEq, Eq)]
218pub(crate) enum Named {
219    /// The report carries no statistics object, so the rule has no
220    /// `executable` key to read.
221    NoStatistics,
222    /// No `executable` key: a build that predates the key, which arrived with
223    /// the default install's front, so it cannot have handed the run off.
224    Omitted,
225    /// `null`, any other value that is not a path, or a path that is not
226    /// absolute: the run did not name its executable's resolved path.
227    Unresolved,
228    /// The absolute path a run names as its executable. A zetesis build writes
229    /// it resolved through links; [`Executables::settle`] resolves it again,
230    /// so a path through a link names the executable the link leads to.
231    Path(PathBuf),
232}
233
234/// What a native run's report names as its executable, read from the report's
235/// `statistics` value: absent or not an object, the report carries no
236/// statistics. A relative path is unresolved: it would resolve against this
237/// process's working directory, not the run's.
238pub(crate) fn named(statistics: &serde_json::Value) -> Named {
239    let Some(statistics) = statistics.as_object() else {
240        return Named::NoStatistics;
241    };
242    match statistics.get("executable") {
243        None => Named::Omitted,
244        Some(value) => value
245            .as_str()
246            .map(PathBuf::from)
247            .filter(|path| path.is_absolute())
248            .map_or(Named::Unresolved, Named::Path),
249    }
250}

252/// The executables a report's native runs name, settled against the native
253/// executable it sealed and invoked. Each named path is resolved through links
254/// before it is compared, so runs that name one executable by different paths
255/// name it once. A run naming the native executable, or naming none because it
256/// predates the key, changes nothing. The first other
257/// executable a run names, as when the default install's front hands a GPU run
258/// to its sibling, is sealed at that report; later runs may name the native
259/// executable or that one.
260#[derive(Debug)]
261pub(crate) struct Executables {
262    native: PathBuf,
263    other: Option<FileSeal>,
264    limit: usize,
265}
266
267impl Executables {
268    /// The rule for runs of the native executable sealed as `native`, sealing
269    /// another within `limit` bytes.
270    pub(crate) fn new(native: &FileSeal, limit: usize) -> Self {
271        Self {
272            native: native.canonical().to_owned(),
273            other: None,
274            limit,
275        }
276    }
277
278    /// Settle what one native run's report names.
279    ///
280    /// # Errors
281    /// Returns why the run's evidence is refused: its report carries no
282    /// statistics object to name its executable; it did not name its
283    /// executable's resolved path, so after a hand-off it may be the sibling;
284    /// the executable it names could not be resolved or sealed; or it names a
285    /// third executable, beside the native one and the other one sealed
286    /// earlier.
287    pub(crate) fn settle(&mut self, named: &Named) -> Result<(), String> {
288        let named = match named {
289            Named::NoStatistics => {
290                return Err(
291                    "the native report carries no statistics to name its executable".to_owned(),
292                );
293            }
294            Named::Omitted => return Ok(()),
295            Named::Unresolved => {
296                return Err("this run could not name the executable that did its work".to_owned());
297            }
298            Named::Path(path) => path,
299        };
300        let unsealed = |error: Error| {
301            format!(
302                "the executable this run names, {}, could not be sealed: {error}",
303                named.display()
304            )
305        };
306        let resolved =
307            std::fs::canonicalize(named).map_err(|source| unsealed(io(named, source)))?;
308        if resolved == self.native {
309            return Ok(());
310        }
311        match &self.other {
312            None => {
313                self.other = Some(seal(named, self.limit).map_err(unsealed)?);
314                Ok(())
315            }
316            Some(seal) if seal.canonical() == resolved => Ok(()),
317            Some(seal) => Err(format!(
318                "this run names {}, but an earlier run named {} besides the native {}",
319                named.display(),
320                seal.canonical().display(),
321                self.native.display()
322            )),
323        }
324    }
325
326    /// The other executable runs named, as sealed at its first report, and
327    /// its recheck now; both absent when no run named one. The value names no
328    /// other executable afterwards.
329    pub(crate) fn close(&mut self) -> (Option<FileSeal>, Option<Change>) {
330        let other = self.other.take();
331        let after = other.as_ref().map(recheck);
332        (other, after)
333    }
334}

336pub(crate) fn aliases(left: &FileSeal, right: &FileSeal) -> bool {
337    left.canonical == right.canonical
338        || (left.device.is_some() && left.device == right.device && left.inode == right.inode)
339}
