3use std::fmt;

5/// An observed compute API. Unlike a preference, an observation cannot be Auto.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum AdapterBackend {
8    /// The nonexecuting test backend.
9    Noop,
10    /// Vulkan.
11    Vulkan,
12    /// Metal.
13    Metal,
14    /// Direct3D 12.
15    Dx12,
16    /// OpenGL or OpenGL ES.
17    Gl,
18    /// Browser WebGPU.
19    BrowserWebGpu,
20}
21impl AdapterBackend {
22    /// Stable legacy display label; classification does not require parsing it.
23    #[must_use]
24    pub const fn label(self) -> &'static str {
25        match self {
26            Self::Noop => "Noop",
27            Self::Vulkan => "Vulkan",
28            Self::Metal => "Metal",
29            Self::Dx12 => "Dx12",
30            Self::Gl => "Gl",
31            Self::BrowserWebGpu => "BrowserWebGpu",
32        }
33    }
34}
35impl fmt::Display for AdapterBackend {
36    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
37        formatter.write_str(self.label())
38    }
39}

41/// The device category reported by the backend, not an execution qualification.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub enum AdapterCategory {
44    /// The backend did not provide a more specific category.
45    Other,
46    /// An integrated GPU.
47    IntegratedGpu,
48    /// A discrete GPU.
49    DiscreteGpu,
50    /// A virtual GPU.
51    VirtualGpu,
52    /// A CPU adapter.
53    Cpu,
54}
55impl AdapterCategory {
56    /// Stable legacy display label.
57    #[must_use]
58    pub const fn label(self) -> &'static str {
59        match self {
60            Self::Other => "Other",
61            Self::IntegratedGpu => "IntegratedGpu",
62            Self::DiscreteGpu => "DiscreteGpu",
63            Self::VirtualGpu => "VirtualGpu",
64            Self::Cpu => "Cpu",
65        }
66    }
67}
68impl fmt::Display for AdapterCategory {
69    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
70        formatter.write_str(self.label())
71    }
72}

74/// A borrowed data view of an adapter's reported metadata.
75///
76/// Empty optional text means absence in the backend contract and becomes None.
77/// Numeric IDs retain their exact reported values; zero is not universally
78/// interpreted as absence. These fields do not certify device execution or a
79/// unique physical device. Constructing/copying this view performs no discovery
80/// or allocation and has fixed cost.
81#[derive(Clone, Copy, Debug, PartialEq, Eq)]
82pub struct AdapterMetadata<'a> {
83    /// Reported adapter name, without normalization.
84    pub name: &'a str,
85    /// Observed API, independent of the requested selection policy.
86    pub backend: AdapterBackend,
87    /// Reported device category.
88    pub category: AdapterCategory,
89    /// Exact backend-reported vendor ID; zero can mean unknown.
90    pub vendor_id: u32,
91    /// Exact backend-reported device ID; not a physical-device identifier.
92    pub device_id: u32,
93    /// PCI bus identifier where reported.
94    pub pci_bus_id: Option<&'a str>,
95    /// Driver name where reported.
96    pub driver: Option<&'a str>,
97    /// Driver version/details where reported.
98    pub driver_info: Option<&'a str>,
99}
