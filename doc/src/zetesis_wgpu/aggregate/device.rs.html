16/// Exact device reduction of already-acquired aggregate eligibility.
17///
18/// Retains one prepared numeric group and an exact occurrence-count transport.
19/// Every successful nonempty batch executes the selected device. This is
20/// an aggregate primitive: mask acquisition, head permission, source coverage,
21/// candidate generation and stable-model minimality remain outside it.
22pub struct GpuAggregateOracle {
23    runtime: Runtime,
24    resident: Option<Resident>,
25    epoch: u32,
26    last: Option<AggregateGpuBatchStats>,
27    activity: AggregateGpuActivity,
28}
29impl GpuAggregateOracle {
30    /// Create an independently owned device/pipeline under the hard API filter.
31    /// Initialization is outside batch resource/time bounds; no CPU fallback occurs.
32    ///
33    /// # Errors
34    /// Returns typed adapter, capability, allocation, validation or device failure.
35    pub fn new_selected(options: GpuOptions, selection: GpuSelection) -> Result<Self, GpuError> {
36        let runtime = pollster::block_on(Runtime::new(options, selection, Self::profile()))?;
37        Ok(Self::with_runtime(runtime))
38    }
39
40    /// Compile this primitive on an existing device context.
41    ///
42    /// Prepared subjects belong to this primitive; device health and execution
43    /// serialization are shared with all primitives using the context.
44    ///
45    /// # Errors
46    /// Refuses an active or invalidated context, unsupported granted limits,
47    /// allocation and shader validation failure. No adapter selection occurs.
48    pub fn from_context(context: &crate::GpuContext) -> Result<Self, GpuError> {
49        let runtime = pollster::block_on(Runtime::from_context(context, Self::profile()))?;
50        Ok(Self::with_runtime(runtime))
51    }
52
53    /// Exact device context retained by this primitive.
54    #[must_use]
55    pub fn context(&self) -> &crate::GpuContext {
56        &self.runtime.context
57    }
58
59    fn with_runtime(runtime: Runtime) -> Self {
60        Self {
61            runtime,
62            resident: None,
63            epoch: 0,
64            last: None,
65            activity: AggregateGpuActivity::default(),
66        }
67    }
68
69    fn profile() -> DeviceProfile {
70        DeviceProfile {
71            device_label: "zetesis numeric aggregate device",
72            shader_label: "exact integer aggregate reduction",
73            pipeline_label: "original and frozen aggregate guards",
74            shader: SHADER.into(),
75            entry_point: "reduce",
76            validate_limits: check_limits,
77        }
78    }
79    /// Actual selected adapter identity; no device operation.
80    #[must_use]
81    pub fn info(&self) -> &GpuInfo {
82        self.runtime.context.info()
83    }
84    /// Latest successfully completed nonempty batch; cleared at every attempt.
85    #[must_use]
86    pub const fn last_batch_stats(&self) -> Option<&AggregateGpuBatchStats> {
87        self.last.as_ref()
88    }
89    /// Actual submitted/validated work retained after a later failure.
90    #[must_use]
91    pub const fn activity(&self) -> AggregateGpuActivity {
92        self.activity
93    }
94    /// Drop owned residency and successful statistics. This does not repair an
95    /// invalidated device, reset submission identity or clear attempted activity.
96    pub fn clear_residency(&mut self) {
97        self.resident = None;
98        self.last = None;
99    }

101    /// Reduce every acquired occurrence in input order, preserving repetitions.
102    /// Records must reference the plan's exact Group. Original-only and paired
103    /// records can share one batch; an unrequested frozen result remains absent.
104    /// The kernel reserves both phase scans for each occurrence.
105    ///
106    /// Pure identity/resource/control refusals preserve prior residency. Planned
107    /// incompatible owners are evicted before packing or device allocation, so
108    /// old cache slack never rejects an admissible exact new shape. Incoming
109    /// owned payload is reported separately from the prospective active ceiling.
110    /// Work is linear in tuples and guards per occurrence; device reduction uses
111    /// one 64-lane workgroup per occurrence with fixed shared scratch.
112    ///
113    /// # Errors
114    /// Returns no partial truth vector on identity, resource, control, allocation,
115    /// validation, device or readback failure. Host preparation failures leave the
116    /// device reusable; execution-stage failures drop this residency and invalidate
117    /// the entire shared context. Busy contention is a reusable preflight refusal.
118    /// Empty calls still check health/control, then perform no work or allocation.
119    pub fn check_batch(
120        &mut self,
121        group: &AggregateGpuPlan<'_>,
122        records: &[Eligibility<'_>],
123        limits: AggregateGpuLimits,
124        cancellation: &Cancellation,
125    ) -> Result<Vec<AggregateGpuReduction>, AggregateGpuError> {
126        self.last = None;
127        self.activity = AggregateGpuActivity::default();
128        self.check(group, records, limits, cancellation)
129            .map_err(Into::into)
130    }
131
132    fn check(
133        &mut self,
134        group: &AggregateGpuPlan<'_>,
135        records: &[Eligibility<'_>],
136        limits: AggregateGpuLimits,
137        cancellation: &Cancellation,
138    ) -> Result<Vec<AggregateGpuReduction>, GpuError> {
139        let context = self.runtime.context.clone();
140        let _lease = context.lease()?;
141        poll(cancellation)?;
142        self.runtime.check_health()?;
143        if records.is_empty() {
144            return Ok(Vec::new());
145        }
146        let epoch = self
147            .epoch
148            .checked_add(1)
149            .ok_or_else(|| capacity("aggregate submission identity exhausted"))?;
150        let plan = Plan::new(
151            group,
152            records,
153            limits,
154            self.runtime.limits(),
155            epoch,
156            cancellation,
157        )?;
158        let fresh = self
159            .resident
160            .as_ref()
161            .is_none_or(|value| !Arc::ptr_eq(&value.numeric, &group.numeric));
162        let allocated = fresh
163            || self
164                .resident
165                .as_ref()
166                .is_none_or(|value| !value.matches_count(records.len()));
167        let stats = AggregateGpuBatchStats {
168            group_uploaded: fresh,
169            transport_allocated: allocated,
170            incoming_resident_bytes: self
171                .resident
172                .as_ref()
173                .map(Resident::bytes)
174                .transpose()?
175                .unwrap_or(0),
176            resident_group_bytes: group.numeric.bytes,
177            resident_transport_bytes: plan.transport,
178            accounted_bytes: plan.accounted,
179            host_work: plan.host_work,
180            occurrences: u64::from(plan.worlds),
181            device_work: plan.total_work,
182        };
183        poll(cancellation)?;
184        if fresh {
185            self.resident = None;
186        } else if allocated && let Some(resident) = self.resident.as_mut() {
187            resident.transport = None;
188        }
189        let masks = plan.pack(records, cancellation)?;
190        poll(cancellation)?;
191        let scopes = ErrorScopes::new(self.runtime.device());
192        if fresh {
193            self.activity.uploaded_bytes = group.numeric.bytes;
194        }
195        let resident = self
196            .resident
197            .get_or_insert_with(|| Resident::new(&self.runtime, Arc::clone(&group.numeric)));
198        self.epoch = epoch;
199        let outcome = resident.dispatch(
200            &self.runtime,
201            &masks,
202            &plan,
203            limits.timeout,
204            cancellation,
205            &mut self.activity,
206        );
207        let result = self.runtime.complete(scopes, outcome).and_then(|results| {
208            poll(cancellation)?;
209            Ok(results)
210        });
211        if result.is_ok() {
212            self.last = Some(stats);
213        } else {
214            self.resident = None;
215            // Runtime settlement already invalidates uncertain/failed device
216            // state. A later caller stop must not discard a healthy context.
217        }
218        result
219    }
220}

222fn check_limits(limits: &wgpu::Limits) -> Result<(), GpuError> {
223    for (label, required, available) in [
224        (
225            "workgroup invocations",
226            LANES,
227            limits.max_compute_invocations_per_workgroup,
228        ),
229        (
230            "workgroup width",
231            LANES,
232            limits.max_compute_workgroup_size_x,
233        ),
234        (
235            "workgroup scratch",
236            4 * LANES * 4,
237            limits.max_compute_workgroup_storage_size,
238        ),
239        (
240            "storage bindings",
241            4,
242            limits.max_storage_buffers_per_shader_stage,
243        ),
244        (
245            "uniform bindings",
246            1,
247            limits.max_uniform_buffers_per_shader_stage,
248        ),
249        ("bindings per group", 5, limits.max_bindings_per_bind_group),
250    ] {
251        if available < required {
252            return Err(capacity(&format!(
253                "aggregate {label}: need {required}, device provides {available}"
254            )));
255        }
256    }
257    Ok(())
258}
