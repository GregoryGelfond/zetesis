16pub(super) struct Resident {
17    pub(super) numeric: Arc<Numeric>,
18    tuples: wgpu::Buffer,
19    guards: wgpu::Buffer,
20    pub(super) transport: Option<Transport>,
21}
22impl Resident {
23    pub(super) fn new(runtime: &Runtime, numeric: Arc<Numeric>) -> Self {
24        Self {
25            tuples: runtime::initialized(
26                runtime.device(),
27                "aggregate contributions",
28                &numeric.tuples,
29                wgpu::BufferUsages::STORAGE,
30            ),
31            guards: runtime::initialized(
32                runtime.device(),
33                "aggregate guards",
34                &numeric.guards,
35                wgpu::BufferUsages::STORAGE,
36            ),
37            numeric,
38            transport: None,
39        }
40    }
41    pub(super) fn bytes(&self) -> Result<u64, GpuError> {
42        self.numeric
43            .bytes
44            .checked_mul(2)
45            .and_then(|bytes| {
46                bytes.checked_add(self.transport.as_ref().map_or(0, |value| value.bytes))
47            })
48            .ok_or_else(|| capacity("aggregate incoming residency bytes overflow"))
49    }
50    pub(super) fn matches_count(&self, count: usize) -> bool {
51        self.transport
52            .as_ref()
53            .is_some_and(|value| value.worlds as usize == count)
54    }
55
56    pub(super) fn dispatch(
57        &mut self,
58        runtime: &Runtime,
59        masks: &[u32],
60        plan: &Plan,
61        timeout: Duration,
62        cancellation: &Cancellation,
63        activity: &mut AggregateGpuActivity,
64    ) -> Result<runtime::Completion<Vec<AggregateGpuReduction>>, GpuError> {
65        let uploaded = activity
66            .uploaded_bytes
67            .checked_add(PARAM_BYTES)
68            .and_then(|bytes| bytes.checked_add(plan.masks))
69            .ok_or_else(|| capacity("aggregate upload accounting overflow"))?;
70        let transport = self
71            .transport
72            .get_or_insert_with(|| Transport::new(runtime, &self.tuples, &self.guards, plan));
73        poll(cancellation)?;
74        runtime.queue().write_buffer(
75            &transport.params,
76            0,
77            bytemuck::cast_slice(&plan.params(&self.numeric)),
78        );
79        runtime
80            .queue()
81            .write_buffer(&transport.masks, 0, bytemuck::cast_slice(masks));
82        activity.uploaded_bytes = uploaded;
83        let submission = runtime::submit(
84            runtime.device(),
85            runtime.queue(),
86            &runtime::Dispatch {
87                command_label: "aggregate batch",
88                pass_label: "original and frozen reductions",
89                pipeline: &runtime.pipeline,
90                group: &transport.group,
91                workgroups: [plan.worlds, 1, 1],
92                result: &transport.output,
93                readback: &transport.readback,
94                result_bytes: plan.results,
95            },
96        );
97        activity.submissions = 1;
98        activity.submitted_occurrences = u64::from(plan.worlds);
99        activity.scheduled_work = plan.total_work;
100        let result = runtime::read_polled(
101            runtime.device(),
102            &transport.readback,
103            submission,
104            timeout,
105            || poll(cancellation),
106            |words| packing::decode(words, &self.numeric, plan, masks, cancellation),
107        );
108        if result.is_ok() {
109            activity.completed_occurrences = u64::from(plan.worlds);
110            activity.completed_work = plan.total_work;
111            activity.downloaded_bytes = plan.results;
112        }
113        Ok(result)
114    }
115}

117pub(super) struct Transport {
118    worlds: u32,
119    bytes: u64,
120    params: wgpu::Buffer,
121    masks: wgpu::Buffer,
122    output: wgpu::Buffer,
123    readback: wgpu::Buffer,
124    group: wgpu::BindGroup,
125}
126impl Transport {
127    fn new(runtime: &Runtime, tuples: &wgpu::Buffer, guards: &wgpu::Buffer, plan: &Plan) -> Self {
128        let params = runtime::buffer(
129            runtime.device(),
130            "aggregate dimensions",
131            PARAM_BYTES,
132            wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
133        );
134        let masks = runtime::buffer(
135            runtime.device(),
136            "aggregate eligibility",
137            plan.masks,
138            wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
139        );
140        let output = runtime::buffer(
141            runtime.device(),
142            "aggregate measures",
143            plan.results,
144            wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_SRC,
145        );
146        let readback = runtime::buffer(
147            runtime.device(),
148            "aggregate readback",
149            plan.results,
150            wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
151        );
152        let group = runtime
153            .device()
154            .create_bind_group(&wgpu::BindGroupDescriptor {
155                label: Some("numeric aggregate batch"),
156                layout: &runtime.pipeline.get_bind_group_layout(0),
157                entries: &[
158                    runtime::entry(0, &params),
159                    runtime::entry(1, tuples),
160                    runtime::entry(2, guards),
161                    runtime::entry(3, &masks),
162                    runtime::entry(4, &output),
163                ],
164            });
165        Self {
166            worlds: plan.worlds,
167            bytes: plan.transport,
168            params,
169            masks,
170            output,
171            readback,
172            group,
173        }
174    }
