1//! Ordered results bound to the exact subject of a completed device operation.
2
3use zetesis_ferraris::{Interpretation, Theory};
4
5use crate::{GpuError, GpuErrorKind};
6
7/// A completely decoded device batch, retaining its original theory and the
8/// exact borrowed candidate slice. Only a successful oracle operation creates
9/// this receipt; callers cannot reconstruct it from detached verdict values.
10///
11/// Each non-`NotModel` formula or tight result establishes original satisfaction,
12/// including a residual that still needs exact reduct completion. This receipt
13/// certifies the operation and subject, not mechanical verification of shaders.
14#[derive(Debug)]
15pub struct CheckedBatch<'a, T> {
16    theory: &'a Theory,
17    candidates: &'a [Interpretation],
18    checks: Vec<T>,
19}
20
21impl<'a, T> CheckedBatch<'a, T> {
22    pub(crate) fn new(
23        theory: &'a Theory,
24        candidates: &'a [Interpretation],
25        checks: Vec<T>,
26    ) -> Self {
27        Self {
28            theory,
29            candidates,
30            checks,
31        }
32    }
33
34    /// Original theory checked by the operation.
35    #[must_use]
36    pub const fn theory(&self) -> &'a Theory {
37        self.theory
38    }
39
40    /// Exact submitted candidates, in result order.
41    #[must_use]
42    pub const fn candidates(&self) -> &'a [Interpretation] {
43        self.candidates
44    }
45
46    /// Ordered results, without detaching them from this receipt.
47    #[must_use]
48    pub fn checks(&self) -> &[T] {
49        &self.checks
50    }
51
52    /// Consume the results only for their exact original subject and slice.
53    ///
54    /// # Errors
55    /// Refuses a foreign theory, copied/reordered slice, or inconsistent count.
56    pub fn into_checks(
57        self,
58        theory: &Theory,
59        candidates: &[Interpretation],
60    ) -> Result<Vec<T>, GpuError> {
61        if !self.theory.same_instance(theory)
62            || !std::ptr::eq(self.candidates, candidates)
63            || self.checks.len() != candidates.len()
64        {
65            return Err(GpuError::new(
66                GpuErrorKind::Readback,
67                "device receipt does not belong to this exact candidate batch",
68            ));
69        }
70        Ok(self.checks)
71    }
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77    use zetesis_ferraris::{AdmissionLimits, FormulaParts};
78
79    fn theory() -> Theory {
80        Theory::new(
81            1,
82            FormulaParts::default(),
83            vec![],
84            AdmissionLimits::default(),
85        )
86        .unwrap()
87    }
88
89    #[test]
90    fn receipt_refuses_equal_but_foreign_subjects() {
91        let owner = theory();
92        let foreign = theory();
93        let candidates = [Interpretation::new(&owner, []).unwrap()];
94        let copies = candidates.clone();
95        for (subject, slice) in [(&foreign, &candidates[..]), (&owner, &copies[..])] {
96            let receipt = CheckedBatch::new(&owner, &candidates, vec![true]);
97            assert!(receipt.into_checks(subject, slice).is_err());
98        }
99    }
100
101    #[test]
102    fn receipt_preserves_exact_order_and_count() {
103        let owner = theory();
104        let candidates = [
105            Interpretation::new(&owner, []).unwrap(),
106            Interpretation::new(&owner, [0]).unwrap(),
107        ];
108        let receipt = CheckedBatch::new(&owner, &candidates, vec![1, 2]);
109        assert_eq!(
110            receipt.into_checks(&owner, &candidates).unwrap(),
111            vec![1, 2]
112        );
113        let short = CheckedBatch::new(&owner, &candidates, vec![1]);
114        assert!(short.into_checks(&owner, &candidates).is_err());
115    }
116}
