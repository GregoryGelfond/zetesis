1//! Where a GPU primitive obtains its device.
2
3use crate::{GpuContext, GpuDiscovery, GpuOptions, GpuSelection};
4
5/// Where a GPU primitive obtains its device: a selection of its own, the same
6/// selection over an adapter enumeration a [`GpuDiscovery`] began earlier, or
7/// an existing context.
8///
9/// A selected or discovered source chooses, among the adapters of the API that
10/// the policy admits, the best-ranked one advertising compute shaders and
11/// limits the constructing primitive accepts, then requests a device of its
12/// own from that adapter. A discovered source selects over the adapters present
13/// when its discovery began, so it chooses the adapter and reports the failures
14/// a selected source would have then, at the same steps; [`GpuDiscovery`]
15/// states what that snapshot's age means. A discovery saves only the adapter
16/// enumeration's wait. A context source selects nothing: the primitive compiles
17/// on that exact context, which keeps the timestamp request it was created with.
18///
19/// `timing` requests compute-pass timestamps for the primitives that measure
20/// compute time, the formula and tight oracles; the other primitives request
21/// none. Unsupported timestamps leave solving available and are reported by
22/// [`GpuContext::compute_timing`].
23#[derive(Clone, Copy)]
24#[non_exhaustive]
25pub enum GpuDeviceSource<'a> {
26    /// Enumerate `selection`'s adapters, select one under `options` and
27    /// request its device.
28    Selected {
29        /// The hard adapter policy.
30        options: GpuOptions,
31        /// The API to enumerate; never replaced by another.
32        selection: GpuSelection,
33        /// Request compute-pass timestamps.
34        timing: bool,
35    },
36    /// Select under `options` as [`Self::Selected`] does for the discovery's
37    /// API, over the adapter enumeration `discovery` began. A discovery serves one
38    /// primitive: once taken, or when its thread failed, the primitive
39    /// enumerates as [`Self::Selected`] does.
40    Discovered {
41        /// The adapter enumeration begun earlier.
42        discovery: &'a GpuDiscovery,
43        /// The hard adapter policy.
44        options: GpuOptions,
45        /// Request compute-pass timestamps.
46        timing: bool,
47    },
48    /// Compile on this exact context: no selection and no device request.
49    Context(&'a GpuContext),
50}
