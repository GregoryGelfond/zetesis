1//! Adapter enumeration begun before the GPU primitive that will use it.
2//!
3//! A discovery performs, on one named thread, the instance creation and adapter
4//! enumeration an independent constructor would perform, and creates no device.
5//! A primitive constructed from it reruns its own selection over that
6//! enumeration, with its own policy and capability check, so it chooses the
7//! adapter and reports the error an independent constructor would have when
8//! the discovery began, at the same step; it then requests its device from the
9//! chosen adapter, whose handle nothing else has used. The enumeration is a
10//! snapshot: [`GpuDiscovery`] states what its age means.
11//!
12//! Ownership: the thread holds only its result, which it returns through its
13//! `JoinHandle`. The join handle lives in the state the consumers' handles
14//! share and the thread never references, so the last handle dropped is always
15//! a consumer's. Taking the result replaces that state, so it is taken once.
16//! Dropping the last handle while the thread runs joins it: no discovery work
17//! outlives its owner, at the cost of waiting for the rest of the enumeration,
18//! which wgpu does not let anyone cancel.
19
20use std::sync::{Arc, Mutex, PoisonError};
21use std::thread::JoinHandle;

23use crate::selection::{self, Candidate};
24use crate::{GpuContext, GpuError, GpuOptions, GpuSelection};

26/// Adapter enumeration begun before the GPU primitive that will use it; it
27/// creates no device. A primitive takes it through
28/// [`crate::GpuDeviceSource::Discovered`].
29///
30/// The enumeration is a snapshot of the adapters present when the discovery
31/// began. A primitive constructed from it chooses the adapter, and reports the
32/// selection error, that an independent constructor would have at that moment.
33/// An adapter added since is not seen; one removed since can still be chosen,
34/// and then fails its device request with the device error. The snapshot's age
35/// is the time from [`Self::spawn`] to the primitive's construction, which the
36/// caller controls. The command line begins one after validating its
37/// configuration and consumes it when the solve constructs its primitive: on
38/// the formula routes once the program is read, admitted and grounded, and on
39/// the relational routes once it is read and admitted, before any grounding.
40/// The age therefore grows with the time to read, admit and, on the formula
41/// routes, ground the program. A caller that keeps a discovery across a
42/// possible device change should spawn a new one, or construct without it.
43///
44/// Cloning shares the discovery. It serves one primitive: the first primitive
45/// constructed from it takes its enumeration, and any later one enumerates as
46/// an independent constructor does. Dropping the last handle waits for the
47/// discovery thread, so no discovery work outlives its owner; that wait is
48/// bounded by wgpu's adapter enumeration, which cannot be cancelled.
49#[derive(Clone)]
50#[must_use]
51pub struct GpuDiscovery {
52    shared: Arc<Shared>,
53}
54
55struct Shared {
56    selection: GpuSelection,
57    state: Mutex<State>,
58    /// Test seam: whether a consumer selected over the prefetched enumeration.
59    #[cfg(test)]
60    served: std::sync::atomic::AtomicBool,
61}
62
63enum State {
64    Running(JoinHandle<Result<Vec<Candidate>, GpuError>>),
65    /// Taken by a consumer, or never started: consumers enumerate as
66    /// independent constructors do.
67    Spent,
68}
69
70impl GpuDiscovery {
71    /// Start, on one named thread, the instance creation and adapter
72    /// enumeration an independent constructor would perform for `selection`.
73    ///
74    /// Never fails: a thread that cannot be started leaves nothing prefetched,
75    /// and consumers then enumerate as independent constructors do.
76    pub fn spawn(selection: GpuSelection) -> Self {
77        let state = std::thread::Builder::new()
78            .name("zetesis-gpu-discovery".into())
79            .spawn(move || pollster::block_on(selection::enumerate_requested(selection)))
80            .map_or(State::Spent, State::Running);
81        Self::with_state(selection, state)
82    }
83
84    fn with_state(selection: GpuSelection, state: State) -> Self {
85        Self {
86            shared: Arc::new(Shared {
87                selection,
88                state: Mutex::new(state),
89                #[cfg(test)]
90                served: std::sync::atomic::AtomicBool::new(false),
91            }),
92        }
93    }
94
95    /// The API this discovery enumerates.
96    #[must_use]
97    pub fn selection(&self) -> GpuSelection {
98        self.shared.selection
99    }
100
101    /// The enumeration, waiting for the thread if it is still running; `None`
102    /// once taken, or when the thread never started or panicked.
103    fn take(&self) -> Option<Result<Vec<Candidate>, GpuError>> {
104        let state = {
105            let mut state = self
106                .shared
107                .state
108                .lock()
109                .unwrap_or_else(PoisonError::into_inner);
110            std::mem::replace(&mut *state, State::Spent)
111        };
112        // The lock is released before the join, so a waiting consumer never
113        // holds it while the enumeration finishes.
114        match state {
115            State::Running(thread) => thread.join().ok(),
116            State::Spent => None,
117        }
118    }
119
120    /// The context a primitive with device `label`, capability check `validate`
121    /// and timestamp request `timing` opens under `options`: its own selection
122    /// over the enumeration, then its device requested from the chosen adapter.
123    /// Without an enumeration to take this is an independent constructor's
124    /// selection.
125    ///
126    /// # Errors
127    /// The independent constructors' typed failures, at the same step: the
128    /// enumeration's, the selection's, the device request's, or the granted
129    /// limits' refusal by `validate`.
130    pub(crate) async fn context(
131        &self,
132        options: GpuOptions,
133        label: &'static str,
134        validate: fn(&wgpu::Limits) -> Result<(), GpuError>,
135        timing: bool,
136    ) -> Result<GpuContext, GpuError> {
137        let Some(enumeration) = self.take() else {
138            return GpuContext::create_with_timing(
139                options,
140                self.selection(),
141                label,
142                validate,
143                timing,
144            )
145            .await;
146        };
147        #[cfg(test)]
148        self.shared
149            .served
150            .store(true, std::sync::atomic::Ordering::Relaxed);
151        let candidates = enumeration?;
152        let index = selection::select_from(&candidates, options, self.selection(), validate)?;
153        let candidate = &candidates[index];
154        GpuContext::open(
155            &candidate.adapter,
156            candidate.info.clone(),
157            label,
158            validate,
159            timing,
160        )
161        .await
162    }
163
164    /// Whether a consumer selected over this discovery's prefetched enumeration.
165    #[cfg(test)]
166    pub(crate) fn served(&self) -> bool {
167        self.shared
168            .served
169            .load(std::sync::atomic::Ordering::Relaxed)
170    }
171}
172
173impl Drop for Shared {
174    fn drop(&mut self) {
175        let state = std::mem::replace(
176            self.state.get_mut().unwrap_or_else(PoisonError::into_inner),
177            State::Spent,
178        );
179        if let State::Running(thread) = state {
180            // The enumeration, with its adapter handles, is dropped with it.
181            let _ = thread.join();
182        }
183    }
184}
