15/// GPU original-truth evaluation and sound frozen-query propagation.
16///
17/// The immutable transport retains native threshold and extremum-cut rows as
18/// shared condition/value columns. Portable u32 limbs preserve all admitted
19/// weights and bounds without optional device integer capabilities. Original
20/// truth is exact; frozen numeric projection uses conservative linear bounds,
21/// which can leave more residual work than expanded Boolean formulas.
22///
23/// This is a partial reduct primitive; exact residual search remains necessary.
24/// A residual carries the semantic atom region retained by the completed device sweeps. No source grounding, objective scoring, candidate
25/// enumeration or CPU oracle fallback occurs here.
26pub struct GpuFormulaOracle {
27    profile: GpuFormulaProfile,
28    resident: Option<Resident>,
29    candidates: crate::candidates::Staging,
30    epoch: u32,
31    last: Option<FormulaBatchStats>,
32    last_submission_candidates: Option<usize>,
33    // Timestamp storage and pair ledger, from the first timed batch until the
34    // residency is cleared; transports and theories come and go beneath it.
35    timer: Option<ComputeTimer>,
36    reported: Option<BatchComputeTime>,
37}

203    /// Resident formula buffers, actual host seed capacity and owned timer
204    /// storage after a submission. The returned checks own their own regions.
205    ///
206    /// # Errors
207    /// Returns a capacity error if the sum is outside the byte counter.
208    pub fn retained_batch_bytes(&self) -> Result<u64, GpuError> {
209        let device = self
210            .resident
211            .as_ref()
212            .map_or(Ok(0), Resident::retained_bytes)?;
213        let seeds = self.candidates.retained_bytes()?;
214        let timer = if self.timer.is_some() {
215            crate::timing::OVERHEAD_BYTES - crate::timing::READBACK_BYTES
216        } else {
217            0
218        };
219        device
220            .checked_add(seeds)
221            .and_then(|bytes| bytes.checked_add(timer))
222            .ok_or_else(|| packing::capacity("retained formula batch exceeds u64"))
223    }

232    /// The compute-pass time of the batch this oracle's latest call consumed.

241    /// batch unavailable. Under unsupported timing each completing call
242    /// reports its own batch unavailable; disabled timing reports nothing.
243    /// `None` when the latest call consumed no batch: every call clears it at its
244    /// start, including empty and refused calls. A measured duration is never

256    /// Resolve the unresolved batch's interval with one resolve-only submission.
257    ///
258    /// Returns at once, with no report, when no batch is unresolved. Otherwise a
259    /// zero `timeout` is refused; the context lease is taken, and `Busy` submits
260    /// nothing; `cancellation` and context health are then checked once, and a
261    /// stop or an unhealthy context submits nothing. Each refusal leaves the
262    /// batch unresolved and a healthy context reusable. The flush then waits at most
263    /// `timeout` for its submission, polling no control. On observed completion
264    /// the batch is reported, measured or unavailable, and the context stays
265    /// reusable. A timeout or a mapping, device or scope failure reports it
266    /// unavailable, retires the timer and invalidates the context, as for any
267    /// submission whose completion is unknown or faulted. The flush allocates nothing and
268    /// downloads 16 bytes; batch statistics and submission receipts exclude it.
269    ///
270    /// # Errors
271    /// Returns the refusals and failures above.
272    pub fn flush_compute_time(
273        &mut self,
274        timeout: Duration,
275        cancellation: &Cancellation,
276    ) -> Result<(), GpuError> {
277        self.reported = None;
278        let Some(timer) = self.timer.as_mut().filter(|timer| timer.unresolved()) else {
279            return Ok(());
280        };
281        crate::runtime::positive_timeout(timeout)?;
282        let runtime = &self.profile.runtime;
283        let _lease = runtime.context.lease()?;
284        cancellation.poll().map_err(GpuError::interrupted)?;
285        runtime.check_health()?;
286        let scopes = ErrorScopes::new(runtime.device());
287        let completion = timer.flush(runtime.device(), runtime.queue(), timeout);
288        match runtime.complete(scopes, Ok(completion)) {
289            Ok(tail) => {
290                self.reported = timer.flushed(tail, runtime.queue().get_timestamp_period());
291                Ok(())
292            }
293            Err(error) => {
294                self.reported = crate::timing::retire(&mut self.timer);
295                Err(error)
296            }
297        }
298    }
299    /// Release authored resident handles and the compute timer. This neither
300    /// cancels GPU work nor repairs an invalidated device; driver retirement can
301    /// occur later. Clears the last batch and submission receipts. A batch still
302    /// unresolved is reported unavailable, since it can never be resolved; the
303    /// next timed batch starts a fresh timer.
304    pub fn clear_residency(&mut self) {
305        self.resident = None;
306        self.candidates.release();
307        self.last = None;
308        self.last_submission_candidates = None;
309        self.reported = crate::timing::retire(&mut self.timer);
310    }
311    /// Evaluate original roots, freeze M, and cooperatively narrow the proper-
312    /// subset query. Results retain input order and no partial batch is returned.
313    /// Clones of one Theory reuse its graph; independent equal theories do not.
314    /// Exact candidate-count transport is replaced when that count changes.
315    /// Host candidate words reuse initialized scratch for the same subject/count;
316    /// replacement releases the old scratch before cold preparation. Actual
317    /// capacity is charged on every call, and a smaller allowance can replace an
318    /// oversized optional allocation. Explicit clearing releases this scratch too.
319    /// A cold subject is prepared in O(nodes + edges + roots + levels) host work, outside
320    /// the per-candidate device-work limit. One buffer retains fixed headers and
321    /// complete native operand rows; no binary expansion or per-candidate graph
322    /// copy is made. Its upload buffers are released after
323    /// device copies are created. Pure chains retain serial truth; other DAGs
324    /// distribute independent nodes by dependency level. Narrow levels can incur
325    /// synchronization cost without useful parallelism; no crossover is assumed.
326    ///
327    /// # Errors
328    /// Refuses foreign interpretations, insufficient setup/work/storage limits,
329    /// u32 address overflow, allocation, device/poll, validation or readback
330    /// failures. Execution/readback failures invalidate the entire context.
331    /// Minimum shape, input and Busy refusals preserve residency. Later cold
332    /// preparation refusal releases the old resident handles and leaves a
333    /// healthy, cold context without advancing its epoch. Driver retirement is
334    /// excluded from authored byte limits. A propagation limit is an explicit
335    /// residual, never a proof of stability. Under fitted sizing, the first call
336    /// that dispatches a size compiles its pipeline. A failure to compile it is
337    /// reported before any dispatch and invalidates the context; the size stays
338    /// uncompiled, and no theory stays resident, because only a freshly
339    /// prepared theory can need a size first and its cold preparation released
340    /// the previous one.
341    pub fn propagate_batch(
342        &mut self,
343        theory: &Theory,
344        candidates: &[Interpretation],
345        limits: FormulaLimits,
346    ) -> Result<Vec<FormulaCheck>, GpuError> {
347        self.propagate_batch_with_cancellation(theory, candidates, limits, &Cancellation::default())
348    }

350    /// Propagate with cooperative caller cancellation and deadline observation.
351    /// Busy admission precedes control; control precedes device health, including
352    /// empty batches. Further polls follow host packing and bracket waits of at
353    /// most 50 ms. Host work, driver calls and scope drains are not preempted.
354    ///
355    /// # Errors
356    /// Preserves [`Self::propagate_batch`]'s failures. An interruption returns its
357    /// exact stop and no partial checks. Pre-submission control refusal leaves
358    /// health reusable; an interrupted submitted operation invalidates it.
359    /// Scope/device faults retain priority. This is not a hard wall-clock deadline.
360    pub fn propagate_batch_with_cancellation(
361        &mut self,
362        theory: &Theory,
363        candidates: &[Interpretation],
364        limits: FormulaLimits,
365        cancellation: &Cancellation,
366    ) -> Result<Vec<FormulaCheck>, GpuError> {
367        self.propagate(
368            theory,
369            candidates,
370            limits,
371            cancellation,
372            crate::candidates::BatchSize::Exact(candidates.len()),
373        )
374    }

376    /// Check the largest nonempty prefix admitted by the prepared device layout.
377    ///
378    /// The prefix is bounded by both the supplied slice and `max_candidates`.
379    /// Cold graph packing occurs once; admission then selects a count using the
380    /// same storage, address and work checks as the exact batch operation. The
381    /// receipt borrows exactly the checked prefix and retains input order.
382    /// An empty slice still observes control and context health before returning.
383    /// No completed prefix is returned after an interrupted or failed operation.
384    ///
385    /// # Errors
386    /// Preserves [`Self::propagate_batch_with_cancellation`]'s typed failures.
387    /// If one candidate cannot fit, returns that refusal without dispatch. Only
388    /// capacity limits reduce the selected count; allocation and device failures
389    /// do not select a fallback. The chosen schedule is not changed to enlarge
390    /// the prefix. Subsequent calls may reuse the prepared subject.
391    pub fn propagate_prefix_with_cancellation<'a>(
392        &mut self,
393        theory: &'a Theory,
394        candidates: &'a [Interpretation],
395        limits: FormulaLimits,
396        cancellation: &Cancellation,
397    ) -> Result<crate::CheckedBatch<'a, FormulaCheck>, GpuError> {
398        let checks = self.propagate(
399            theory,
400            candidates,
401            limits,
402            cancellation,
403            crate::candidates::BatchSize::Prefix(candidates.len()),
404        )?;
405        Ok(crate::CheckedBatch::new(
406            theory,
407            &candidates[..checks.len()],
408            checks,
409        ))
410    }

412    /// Prepare the next submission of `slate`: admit the largest nonempty
413    /// prefix of its candidates from `offset` on, as
414    /// [`Self::propagate_prefix_with_cancellation`] admits a prefix, pack it
415    /// into this oracle's reused upload storage, and plan it at the next epoch.
416    /// Nothing is submitted, and the epoch is committed only by the check.
417    ///
418    /// The returned submission holds this oracle's mutable borrow until a
419    /// check consumes it, so no other operation on the oracle, and no second
420    /// preparation, can run between the plan and its check. Dropping it submits
421    /// nothing and releases the borrow; any retained state its preparation
422    /// replaced stays released, and the next check prepares again. That can be
423    /// the host staging, the resident graph on a fresh admission, and the
424    /// transport when the preparation allocates one. The context's lease is
425    /// taken by the check, not here; this polls control and context health
426    /// before preparing.
427    ///
428    /// # Errors
429    /// Refuses an offset that leaves no candidate, and every refusal and
430    /// failure of [`Self::propagate_prefix_with_cancellation`]'s preparation,
431    /// with its rules for residency.
432    pub fn prepare<'o>(
433        &'o mut self,
434        slate: SlateRef<'_>,
435        offset: usize,
436        limits: FormulaLimits,
437        cancellation: &Cancellation,
438    ) -> Result<PreparedFormulaSubmission<'o>, GpuError> {
439        self.last = None;
440        self.last_submission_candidates = None;
441        self.reported = None;
442        let candidates = slate
443            .candidates()
444            .get(offset..)
445            .filter(|rest| !rest.is_empty())
446            .ok_or_else(|| {
447                GpuError::new(
448                    GpuErrorKind::Capacity,
449                    "no candidate of the slate remains at the offset",
450                )
451            })?;
452        cancellation.poll().map_err(GpuError::interrupted)?;
453        self.profile.runtime.check_health()?;
454        let call = self.prepare_epoch(
455            slate.theory(),
456            candidates,
457            limits,
458            cancellation,
459            crate::candidates::BatchSize::Prefix(candidates.len()),
460        )?;
461        cancellation.poll().map_err(GpuError::interrupted)?;
462        Ok(PreparedFormulaSubmission {
463            count: call.plan.worlds as usize,
464            oracle: self,
465            call,
466            theory: slate.theory().clone(),
467            slate: slate.id(),
468            offset,
469            timeout: limits.timeout,
470        })
471    }

473    fn propagate(
474        &mut self,
475        theory: &Theory,
476        candidates: &[Interpretation],
477        limits: FormulaLimits,
478        cancellation: &Cancellation,
479        size: crate::candidates::BatchSize,
480    ) -> Result<Vec<FormulaCheck>, GpuError> {
481        self.last = None;
482        self.last_submission_candidates = None;
483        self.reported = None;
484        let context = self.profile.runtime.context.clone();
485        let _lease = context.lease()?;
486        cancellation.poll().map_err(GpuError::interrupted)?;
487        self.profile.runtime.check_health()?;
488        if candidates.is_empty() {
489            return Ok(Vec::new());
490        }
491        let call = self.prepare_epoch(theory, candidates, limits, cancellation, size)?;
492        cancellation.poll().map_err(GpuError::interrupted)?;
493        self.run(call, limits.timeout, cancellation, None)
494            .map(|(checks, _)| checks)
495    }

497    /// Prepare a batch at the next epoch, which it commits only when it is
498    /// submitted.
499    fn prepare_epoch(
500        &mut self,
501        theory: &Theory,
502        candidates: &[Interpretation],
503        limits: FormulaLimits,
504        cancellation: &Cancellation,
505        size: crate::candidates::BatchSize,
506    ) -> Result<PreparedCall, GpuError> {
507        let epoch = self.epoch.checked_add(1).ok_or_else(|| {
508            GpuError::new(GpuErrorKind::Capacity, "formula epoch counter exhausted")
509        })?;
510        self.prepare_call(theory, candidates, limits, cancellation, epoch, size)
511    }

513    /// Run a prepared batch under the caller's lease: compile its pipeline at
514    /// first need, install a fresh graph, submit, run `during` while the
515    /// submission is in flight, finish and settle.
516    ///
517    /// Without `during`, the error scopes close after the readback and the
518    /// wait polls the control, as every synchronous query does. With it, they
519    /// close right after the submission, so they capture none of `during`'s
520    /// device calls; a submission that fails before it is made runs no
521    /// `during`. When the control has fired by the time `during` returns, the
522    /// finish observes the submission's completion without polling the
523    /// control, within the batch's timeout, and reports the stop without
524    /// decoding: a completed, released readback leaves the context reusable.
525    /// Beside the checks it returns the finish's wait after `during`: from
526    /// `during`'s return to the observed completion, before mapping and
527    /// decoding; absent without `during`.
528    fn run(
529        &mut self,
530        call: PreparedCall,
531        timeout: Duration,
532        cancellation: &Cancellation,
533        during: Option<&mut dyn FnMut()>,
534    ) -> Result<(Vec<FormulaCheck>, Option<Duration>), GpuError> {
535        let PreparedCall {
536            fresh,
537            plan,
538            stats,
539            output,
540        } = call;
541        // The dispatched size's pipeline is compiled at first need, before this
542        // call's scopes and before the fresh graph becomes resident. Only a
543        // freshly prepared graph can need a size first, and its preparation has
544        // released any previous resident, so a refusal leaves none, as a
545        // cold-preparation failure does.
546        let workgroup_size = fresh
547            .as_ref()
548            .map(|prepared| prepared.graph.workgroup_size)
549            .or_else(|| {
550                self.resident
551                    .as_ref()
552                    .map(|resident| resident.graph.workgroup_size)
553            })
554            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing resident formula"))?;
555        self.profile
556            .runtime
557            .ensure_pipeline(workgroup_size, crate::runtime::Variant::Full)?;
558        let scopes = ErrorScopes::new(self.profile.runtime.device());
559        if let Some(prepared) = fresh {
560            self.resident = Some(Resident::new(self.profile.runtime.device(), prepared));
561        }
562        let status = self.profile.runtime.context.compute_timing();
563        if status == ComputeTimingStatus::Enabled && self.timer.is_none() {
564            self.timer = Some(ComputeTimer::new(self.profile.runtime.device()));
565        }
566        self.epoch = plan.epoch;
567        let runtime = &self.profile.runtime;
568        let mut scopes = Scopes::Open(scopes);
569        let submitted = self
570            .resident
571            .as_mut()
572            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing resident formula"))
573            .and_then(|resident| {
574                resident.submit(runtime, self.candidates.words(), &plan, self.timer.as_ref())
575            });
576        let mut finish_wait = None;
577        let (completion, timing) = match submitted {
578            Err(error) => (Err(error), BatchTiming::default()),
579            Ok(submitted) => {
580                self.last_submission_candidates = Some(submitted.candidates);
581                let mut during_ended = None;
582                let observation = match during {
583                    None => Observation::Polled,
584                    Some(during) => {
585                        scopes.settle();
586                        during();
587                        during_ended = Some(Instant::now());
588                        cancellation.poll().map_or_else(
589                            |stop| Observation::AfterStop(GpuError::interrupted(stop)),
590                            |()| Observation::Polled,
591                        )
592                    }
593                };
594                match self
595                    .resident
596                    .as_ref()
597                    .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing resident formula"))
598                    .and_then(|resident| {
599                        resident.finish(
600                            runtime,
601                            submitted,
602                            packing::Readback {
603                                plan: &plan,
604                                seeds: self.candidates.words(),
605                                output,
606                            },
607                            observation,
608                            timeout,
609                            cancellation,
610                        )
611                    }) {
612                    Ok(dispatched) => {
613                        finish_wait = during_ended
614                            .and_then(|ended| dispatched.completion.window_since(ended));
615                        (Ok(dispatched.completion), dispatched.timing)
616                    }
617                    Err(error) => (Err(error), BatchTiming::default()),
618                }
619            }
620        };
621        let outcome = runtime.complete_scopes(scopes, completion);
622        if self.last_submission_candidates.is_some() {
623            self.reported = crate::timing::settle_call(
624                status,
625                &mut self.timer,
626                outcome.is_ok(),
627                &timing,
628                self.profile.runtime.queue().get_timestamp_period(),
629            );
630        }
631        let checks = outcome?;
632        self.last = Some(stats);
633        Ok((checks, finish_wait))
634    }

636    // All cold staging is owned and admitted before device effects. Minimum
637    // shape/identity refusals preserve residency; later cold failures leave a
638    // healthy context without a resident subject or a committed epoch.
639    fn prepare_call(
640        &mut self,
641        theory: &Theory,
642        candidates: &[Interpretation],
643        mut limits: FormulaLimits,
644        cancellation: &Cancellation,
645        epoch: u32,
646        size: crate::candidates::BatchSize,
647    ) -> Result<PreparedCall, GpuError> {
648        let timed =
649            self.context().compute_timing() == zetesis_backend::ComputeTimingStatus::Enabled;
650        let timing_bytes = if timed {
651            crate::timing::OVERHEAD_BYTES
652        } else {
653            0
654        };
655        limits.max_batch_bytes = limits
656            .max_batch_bytes
657            .checked_sub(timing_bytes)
658            .ok_or_else(|| {
659                GpuError::new(
660                    GpuErrorKind::Capacity,
661                    "formula timestamp storage exceeds the batch byte allowance",
662                )
663            })?;
664        // Old scratch must not coexist with a cold graph's temporary directory.
665        // Releasing it does not change the resident device subject on refusal.
666        if self.resident.as_ref().is_none_or(|resident| {
667            !resident.graph.shape.theory.same_instance(theory)
668                || (size.is_exact() && !resident.matches_count(candidates.len()))
669        }) {
670            self.candidates.release();
671        }
672        let retained = self.retained_plan(theory, size, limits, epoch, timed, cancellation)?;
673        let (fresh, plan) = if let Some(plan) = retained {
674            identities(theory, candidates)?;
675            (None, plan)
676        } else {
677            self.candidates.release();
678            let preparation = Preparation::with_control(
679                theory,
680                size.initial(),
681                limits,
682                self.profile.runtime.limits(),
683                epoch,
684                cancellation,
685            )?
686            .with_propagation(self.profile.kernel_options().propagation)
687            .with_workgroup_sizing(self.profile.kernel_options().workgroup_sizing);
688            identities(theory, candidates)?;
689            self.resident = None;
690            let (prepared, exact) =
691                preparation.finish(self.profile.runtime.limits(), cancellation)?;
692            let plan = if size.is_exact() {
693                exact
694            } else {
695                size.select(limits.max_candidates, cancellation, |count| {
696                    let plan =
697                        prepared.plan(count, limits, self.profile.runtime.limits(), epoch)?;
698                    crate::timing::readback_size(
699                        plan.results,
700                        timed,
701                        self.profile.runtime.limits().max_buffer_size,
702                    )?;
703                    Ok(plan)
704                })?
705            };
706            (Some(prepared), plan)
707        };
708        crate::timing::readback_size(
709            plan.results,
710            timed,
711            self.profile.runtime.limits().max_buffer_size,
712        )?;
713        self.stage_candidates(fresh, plan, candidates, limits, timing_bytes, cancellation)
714    }
715
716    fn stage_candidates(
717        &mut self,
718        fresh: Option<PreparedGraph>,
719        mut plan: Plan,
720        candidates: &[Interpretation],
721        limits: FormulaLimits,
722        timing_bytes: u64,
723        cancellation: &Cancellation,
724    ) -> Result<PreparedCall, GpuError> {
725        let graph = fresh
726            .as_ref()
727            .map(|prepared| &prepared.graph)
728            .or_else(|| self.resident.as_ref().map(|resident| &resident.graph))
729            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing formula graph"))?;
730        let mut stats = FormulaBatchStats {
731            propagation: graph.schedule.propagation(),
732            theory_uploaded: fresh.is_some(),
733            transport_allocated: fresh.is_some()
734                || self
735                    .resident
736                    .as_ref()
737                    .is_none_or(|resident| !resident.matches(&plan)),
738            resident_theory_bytes: graph.schedule.bytes,
739            // The timer is the oracle's; only the readback tail is transport.
740            resident_transport_bytes: plan.transport
741                + if timing_bytes == 0 {
742                    0
743                } else {
744                    crate::timing::READBACK_BYTES
745                },
746            accounted_bytes: plan.accounted + timing_bytes,
747            workgroup_size: graph.workgroup_size.invocations(),
748        };
749        if stats.transport_allocated {
750            self.candidates.release();
751            if let Some(resident) = self.resident.as_mut() {
752                resident.transport = None;
753            }
754        }
755        let graph = fresh
756            .as_ref()
757            .map(|prepared| &prepared.graph)
758            .or_else(|| self.resident.as_ref().map(|resident| &resident.graph))
759            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing formula packing graph"))?;
760        plan.accounted = plan.pack(
761            graph,
762            &candidates[..plan.worlds as usize],
763            &mut self.candidates,
764            limits.max_batch_bytes,
765            cancellation,
766        )?;
767        let output = packing::Output::reserve(&plan)?;
768        stats.accounted_bytes = output.accounted + timing_bytes;
769        Ok(PreparedCall {
770            fresh,
771            plan,
772            stats,
773            output,
774        })
775    }

777    // Optional indexing must not turn a smaller caller allowance into a new
778    // mandatory representation. A capacity refusal re-enters cold admission;
779    // only after its minimum and identities pass is the old resident retired.
780    fn retained_plan(
781        &self,
782        theory: &Theory,
783        size: crate::candidates::BatchSize,
784        limits: FormulaLimits,
785        epoch: u32,
786        timed: bool,
787        cancellation: &Cancellation,
788    ) -> Result<Option<Plan>, GpuError> {
789        let Some(resident) = self
790            .resident
791            .as_ref()
792            .filter(|resident| resident.graph.shape.theory.same_instance(theory))
793        else {
794            return Ok(None);
795        };
796        match size.select(limits.max_candidates, cancellation, |count| {
797            let plan = Plan::new(
798                &resident.graph,
799                count,
800                limits,
801                self.profile.runtime.limits(),
802                false,
803                epoch,
804            )?;
805            crate::timing::readback_size(
806                plan.results,
807                timed,
808                self.profile.runtime.limits().max_buffer_size,
809            )?;
810            Ok(plan)
811        }) {
812            Ok(plan) => Ok(Some(plan)),
813            Err(error)
814                if resident.graph.schedule.incidence.is_some()
815                    && error.kind() == GpuErrorKind::Capacity =>
816            {
817                Ok(None)
818            }
819            Err(error) => Err(error),
820        }
821    }
822}

824/// One submission of a slate that a [`GpuFormulaOracle`] admitted, packed and
825/// planned but has not submitted ([`GpuFormulaOracle::prepare`]).
826///
827/// It holds its oracle's mutable borrow until [`Self::check`] or
828/// [`Self::check_overlapping`] consumes it, so what is resident and the epoch
829/// are those its check runs on. It carries copies of its slate's identity, its
830/// offset and its count, and no borrow of the slate's candidates, so the
831/// slate's stream can be borrowed again while it runs. A second preparation
832/// cannot begin while one exists:
833///
834/// ```compile_fail,E0499
835/// # use zetesis_wgpu::{FormulaLimits, GpuFormulaOracle};
836/// # fn second(oracle: &mut GpuFormulaOracle, slate: zetesis_ferraris::SlateRef<'_>,
837/// #     control: &zetesis_algebra::Cancellation) {
838/// let first = oracle.prepare(slate, 0, FormulaLimits::default(), control);
839/// let second = oracle.prepare(slate, 0, FormulaLimits::default(), control);
840/// drop((first, second));
841/// # }
842/// ```
843///
844/// Nor can any other operation on the oracle run between a preparation and its
845/// check:
846///
847/// ```compile_fail,E0499
848/// # use zetesis_wgpu::{FormulaLimits, GpuFormulaOracle};
849/// # fn between(oracle: &mut GpuFormulaOracle, slate: zetesis_ferraris::SlateRef<'_>,
850/// #     control: &zetesis_algebra::Cancellation) {
851/// let prepared = oracle.prepare(slate, 0, FormulaLimits::default(), control).unwrap();
852/// oracle.clear_residency();
853/// let _ = prepared.check(control);
854/// # }
855/// ```
856///
857/// And a prepared submission is checked only by its own oracle: the oracle has
858/// no public operation that takes a prepared submission, so one prepared
859/// elsewhere has nowhere to go.
860///
861/// ```compile_fail,E0624
862/// # use zetesis_wgpu::{FormulaLimits, GpuFormulaOracle};
863/// # fn elsewhere(one: &mut GpuFormulaOracle, other: &mut GpuFormulaOracle,
864/// #     slate: zetesis_ferraris::SlateRef<'_>, control: &zetesis_algebra::Cancellation) {
865/// let prepared = one.prepare(slate, 0, FormulaLimits::default(), control).unwrap();
866/// let _ = other.propagate(prepared, control);
867/// # }
868/// ```
869///
870/// One after the other, they compile:
871///
872/// ```no_run
873/// # use zetesis_wgpu::{FormulaLimits, GpuFormulaOracle};
874/// # fn in_order(oracle: &mut GpuFormulaOracle, slate: zetesis_ferraris::SlateRef<'_>,
875/// #     control: &zetesis_algebra::Cancellation) {
876/// let first = oracle.prepare(slate, 0, FormulaLimits::default(), control).unwrap();
877/// let _ = first.check(control);
878/// let second = oracle.prepare(slate, 0, FormulaLimits::default(), control).unwrap();
879/// let (_, during) = second.check_overlapping(control, || 1);
880/// assert_eq!(during, Some(1));
881/// # }
882/// ```
883#[must_use = "a prepared submission submits nothing until it is checked"]
884pub struct PreparedFormulaSubmission<'o> {
885    oracle: &'o mut GpuFormulaOracle,
886    call: PreparedCall,
887    theory: Theory,
888    slate: SlateId,
889    offset: usize,
890    count: usize,
891    timeout: Duration,
892}
893
894impl PreparedFormulaSubmission<'_> {
895    /// Peak bytes admitted for this prepared dispatch, including output
896    /// capacity and timing storage. Use this while admitting overlapping work.
897    #[must_use]
898    pub const fn accounted_bytes(&self) -> u64 {
899        self.call.stats.accounted_bytes
900    }
901
902    /// The identity of the slate this submission belongs to.
903    #[must_use]
904    pub const fn slate(&self) -> SlateId {
905        self.slate
906    }
907
908    /// The position of the submission's first candidate in its slate.
909    #[must_use]
910    pub const fn offset(&self) -> usize {
911        self.offset
912    }
913
914    /// The candidates the submission admitted: at least one, and at most the
915    /// rest of the slate from its offset.
916    #[must_use]
917    pub const fn count(&self) -> usize {
918        self.count
919    }
920
921    /// Submit the prepared batch and wait for it, as
922    /// [`GpuFormulaOracle::propagate_prefix_with_cancellation`] checks a
923    /// prefix: the context's lease is taken, control and health are polled,
924    /// and the wait polls control between waits of at most 50 ms.
925    ///
926    /// # Errors
927    /// Has [`GpuFormulaOracle::propagate_prefix_with_cancellation`]'s typed
928    /// failures after preparation, with its rules for context reuse; `Busy`
929    /// contention submits nothing.
930    pub fn check(
931        self,
932        cancellation: &Cancellation,
933    ) -> Result<CheckedSlate<FormulaCheck>, GpuError> {
934        self.run(cancellation, None)
935    }
936
937    /// Submit the prepared batch, run `during` while the device checks it,
938    /// then finish: wait for the submission, map, decode and settle it. One
939    /// submission is in flight while `during` runs, and none when this
940    /// returns; nothing this returns names the submission.
941    ///
942    /// The context's lease is held from before the submission to the finish,
943    /// including `during`, so another operation on the shared context is
944    /// refused with `Busy` meanwhile, and an operation on the same context
945    /// inside `during` is refused too. The error scopes close right after the
946    /// submission, before `during`. When the control has fired by the time
947    /// `during` returns, the finish observes the submission's completion
948    /// without polling the control, within the batch's timeout, so a completed
949    /// and released readback leaves the context reusable, and the stop is
950    /// reported without decoding. A panic in `during` unwinds through this
951    /// operation and invalidates the context, since its submission's
952    /// completion is then unobserved.
953    ///
954    /// Returns the check's outcome and `during`'s result. A successful
955    /// outcome's receipt carries the finish's wait after `during`
956    /// ([`CheckedSlate::finish_wait`]). The result is absent only when the
957    /// check failed before submitting, in which case `during` did not run.
958    pub fn check_overlapping<R>(
959        self,
960        cancellation: &Cancellation,
961        during: impl FnOnce() -> R,
962    ) -> (Result<CheckedSlate<FormulaCheck>, GpuError>, Option<R>) {
963        crate::runtime::overlapping(during, |run| self.run(cancellation, Some(run)))
964    }
965
966    fn run(
967        self,
968        cancellation: &Cancellation,
969        during: Option<&mut dyn FnMut()>,
970    ) -> Result<CheckedSlate<FormulaCheck>, GpuError> {
971        let Self {
972            oracle,
973            call,
974            theory,
975            slate,
976            offset,
977            count,
978            timeout,
979        } = self;
980        let context = oracle.profile.runtime.context.clone();
981        let (checks, finish_wait) = context.lease().and_then(|_lease| {
982            cancellation.poll().map_err(GpuError::interrupted)?;
983            oracle.profile.runtime.check_health()?;
984            oracle.run(call, timeout, cancellation, during)
985        })?;
986        Ok(CheckedSlate::new(theory, slate, offset, count, checks).with_finish_wait(finish_wait))
987    }
988}
989
990fn identities(theory: &Theory, candidates: &[Interpretation]) -> Result<(), GpuError> {
991    if candidates
992        .iter()
993        .any(|candidate| !theory.same_instance(candidate.theory()))
994    {
995        return Err(GpuError::new(
996            GpuErrorKind::Seed,
997            "candidate belongs to another Theory",
998        ));
999    }
1000    Ok(())
1001}
