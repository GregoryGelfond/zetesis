17/// Bounds for one resident finite-formula propagation dispatch.
18#[derive(Clone, Copy, Debug)]
19pub struct FormulaLimits {
20    /// Maximum candidate worlds, additionally bounded by device dispatch limits.
21    pub max_candidates: usize,
22    /// Conservative authored graph, transport, host packing and result bytes.
23    /// Cold preparation bounds its numeric-row directory before allocating it,
24    /// then admits a minimum envelope, followed by the complete
25    /// schedule and actual retained node/root upload-vector element capacities.
26    /// These staging buffers coexist with device copies during upload and are
27    /// released afterward. The directory is charged by capacity and released once
28    /// node packing finishes. Candidate upload scratch is retained only for an
29    /// unchanged subject and batch shape, and charged by actual element capacity.
30    /// Before submission, the host admits the decoded result vector's actual
31    /// capacity, mask storage for every candidate and space to construct one
32    /// region at a time.
33    /// Caller-owned inputs, allocator overhead and driver-private allocations
34    /// or deferred retirement are excluded; this is not an RSS bound.
35    pub max_batch_bytes: u64,
36    /// Maximum complete propagation sweeps per candidate. At zero, a candidate
37    /// that satisfies the original theory returns a round-limit residual.
38    pub max_rounds: u32,
39    /// Maximum charged device work per candidate. Formula setup reserves
40    /// `2*nodes+edges+atoms+roots+levels` for two node walks, all child occurrences, semantic-atom
41    /// initialization, root visits and dependency-level synchronization.
42    /// Result projection reserves another `atoms + 2*ceil(atoms/32)` units for
43    /// one classification per atom and two mask stores per word. A terminal
44    /// result skips classification and reports only the stores from this allowance.
45    /// Pure chains and empty graphs select serial truth and charge zero levels;
46    /// other DAGs charge their number of occupied dependency levels.
47    /// Leaf nodes are visited but need no separate domain initialization store.
48    /// A sweep reserves `9*nodes+2*edges+atoms+65+levels` units, including eight possible gate
49    /// truth-table rows per binary node, two native operand walks, the dependency-level walk and
50    /// a constant 65 for one strict-subset reduction, merging the invocations' summaries and
51    /// applying one projection, whatever the workgroup size. Both setup and each sweep additionally reserve `48*n+32` for
52    /// each variable threshold and `24*n+16` for each variable extremum cut,
53    /// where `n` is its number of condition occurrences. These cover linear
54    /// condition visits and portable limb arithmetic; constant guards carry no
55    /// numeric row charge. These are policy units, not raw GPU instructions.
56    /// Inactive nodes still consume their allowance.
57    /// Opt-in changed-dependency execution additionally charges `2*W` setup
58    /// and `W+nodes+4*variables+6*incidences` per sweep, where `W` is the
59    /// ceiling of `nodes/32`. This bounds two-bit domain shrink events, their
60    /// incident gate visits, active-word reads and generation clears. Each node
61    /// is visited at most once per sweep; skipped gates and dependency levels
62    /// do not earn a smaller charge. A smaller allowance can select dense execution.
63    /// Incremental group reduction additionally reserves `4*nodes+6*group_edges+3*groups`
64    /// setup units and `14*incidences+4*nodes` per sweep, covering initialized
65    /// occurrence counts and uniquely removed-bit notifications. It retains
66    /// the scanned-group allowance; it does not report actual active work.
67    /// If this optional envelope cannot fit, groups retain direct scans.
68    /// A budget below mandatory setup is refused before dispatch.
69    pub max_work_per_candidate: u32,
70    /// Maximum host wait for submitted work; not GPU preemption or a deadline
71    /// for device initialization, compilation, allocation or error-scope drains.
72    /// Must be positive for a nonempty batch; empty batches perform no wait.
73    pub timeout: Duration,
74}
75impl Default for FormulaLimits {
76    fn default() -> Self {
77        Self {
78            max_candidates: 1024,
79            max_batch_bytes: 128 * 1024 * 1024,
80            max_rounds: 64,
81            max_work_per_candidate: 100_000_000,
82            timeout: Duration::from_secs(30),
83        }
84    }
85}

87/// Why propagation leaves the exact proper-subset query unresolved.
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub enum ResidualReason {
90    /// A full no-change sweep left unknowns or a satisfiable query. Quiescence
91    /// never proves stability or the existence of a proper-subset witness.
92    FixedPoint,
93    /// The configured complete-sweep ceiling was reached.
94    RoundLimit,
95    /// The next full sweep would exceed charged work.
96    WorkLimit,
97}

99/// A partial decision about the original candidate and its frozen reduct.
100#[derive(Clone, Debug, PartialEq, Eq)]
101pub enum FormulaVerdict {
102    /// The candidate violates an original asserted root.
103    NotModel,
104    /// Original roots hold, and sound propagation refutes every proper subset.
105    NoProperSubset,
106    /// Original roots hold. The returned region contains the remaining proper-subset queries.
107    Residual(ResidualReason, zetesis_algebra::region::Region),
108}

119/// Validated result for one candidate, preserving input order.
120#[derive(Clone, Debug, PartialEq, Eq)]
121pub struct FormulaCheck {
122    verdict: FormulaVerdict,
123    statistics: FormulaStatistics,
124}
125impl FormulaCheck {
126    /// Logical or residual result. No fixed-point result implies stability.
127    #[must_use]
128    pub const fn verdict(&self) -> &FormulaVerdict {
129        &self.verdict
130    }
131    /// Move the authenticated outcome and any residual root into host completion.
132    #[must_use]
133    pub fn into_verdict(self) -> FormulaVerdict {
134        self.verdict
135    }
136    /// Per-candidate propagation accounting.
137    #[must_use]
138    pub const fn statistics(&self) -> FormulaStatistics {
139        self.statistics
140    }
141}

143/// Residency and allocation diagnostics for a successful nonempty batch.
144#[derive(Clone, Copy, Debug, PartialEq, Eq)]
145#[non_exhaustive]
146pub struct FormulaBatchStats {
147    /// Actual gate scheduling after optional storage/work admission. This can

149    pub propagation: crate::GpuPropagation,
150    /// Whether this call uploaded a different immutable Theory instance.
151    pub theory_uploaded: bool,
152    /// Whether this call allocated its exact candidate-count transport shape.
153    pub transport_allocated: bool,
154    /// Requested immutable node and root buffer bytes, including padding.
155    /// The root buffer also holds the dependency schedule when levels are used.
156    pub resident_theory_bytes: u64,
157    /// Requested parameters, candidates, masks, domains, results and readback.
158    pub resident_transport_bytes: u64,
159    /// Conservative authored allocation sum checked against the batch ceiling.
160    pub accounted_bytes: u64,
161    /// The workgroup size dispatched, in invocations per candidate: the fixed
162    /// size, or the size the theory's dependency schedule fits.
163    pub workgroup_size: u32,
