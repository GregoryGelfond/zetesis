3use crate::{GpuError, GpuErrorKind};
4use std::borrow::Cow;

6/// How an enabled frozen binary gate projects supported Boolean values.
7///
8/// Both implementations use the same three atomic observations and intersections,
9/// including physical-slot aliases and observations taken at different instants.
10/// Native groups use the shared operand reduction, which preserves satisfying
11/// completions without claiming exact projection of racing aliased observations.
12/// Both policies retain the frozen query, work bounds and result protocol.
13/// Selecting an alternative establishes no physical qualification or speedup.
14#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
15pub enum GateProjection {
16    /// Enumerate all eight Boolean relation rows; the existing default shader.
17    #[default]
18    Enumerated,
19    /// Intersect eight-bit relation masks before projecting position supports.
20    /// This opt-in alternative still requires physical device qualification.
21    Bitwise,
22}
23
24#[cfg(test)]
25mod tests;
26
27impl GateProjection {
28    /// Both implementations, with the existing baseline first.
29    pub const ALL: [Self; 2] = [Self::Enumerated, Self::Bitwise];
30
31    /// Stable diagnostic identifier. Constant time and no allocation.
32    #[must_use]
33    pub const fn label(self) -> &'static str {
34        match self {
35            Self::Enumerated => "enumerated",
36            Self::Bitwise => "bitwise",
37        }
38    }
39
40    // The alternative replaces only the local transfer in the immutable baseline.
41    // A fixed-size, fallible reservation precedes construction. No user text or
42    // repeated replacement occurs; both variants therefore share the scaffold.
43    pub(super) fn shader(self) -> Result<Cow<'static, str>, GpuError> {
44        const BASELINE: &str = include_str!("../formula.wgsl");
45        match self {
46            Self::Enumerated => Ok(Cow::Borrowed(BASELINE)),
47            Self::Bitwise => {
48                let invalid = || {
49                    GpuError::new(
50                        GpuErrorKind::Validation,
51                        "formula shader gate boundary is missing",
52                    )
53                };
54                let (prefix, gate_and_suffix) =
55                    BASELINE.split_once("fn gate(").ok_or_else(invalid)?;
56                let (_, suffix) = gate_and_suffix
57                    .split_once("fn finish(")
58                    .ok_or_else(invalid)?;
59                let parts = [prefix, include_str!("bitwise.wgsl"), "fn finish(", suffix];
60                let capacity = parts
61                    .iter()
62                    .try_fold(0usize, |total, part| total.checked_add(part.len()))
63                    .ok_or_else(|| {
64                        GpuError::new(
65                            GpuErrorKind::Capacity,
66                            "formula shader source length overflow",
67                        )
68                    })?;
69                let mut source = String::new();
70                source
71                    .try_reserve_exact(capacity)
72                    .map_err(|error| GpuError::new(GpuErrorKind::Allocation, error.to_string()))?;
73                for part in parts {
74                    source.push_str(part);
75                }
76                Ok(Cow::Owned(source))
77            }
78        }
79    }
