1//! One immutable GPU transport view of an admitted formula. The fixed headers
2//! and complete wide operand rows share one storage binding. Unused source arena
3//! cells are omitted; overlapping source spans are copied once per logical row.
4
5use crate::{GpuError, GpuErrorKind};
6use zetesis_cpu::Cancellation;
7use zetesis_ferraris::{NodeView, Theory};
8
9pub(crate) struct Shape {
10    pub(crate) bytes: u64,
11    pub(crate) edges: u32,
12    pub(crate) wide_words: u32,
13    pub(crate) auxiliary: u32,
14}
15
16fn capacity(detail: &str) -> GpuError {
17    GpuError::new(GpuErrorKind::Capacity, detail)
18}
19
20fn address(value: usize) -> Result<u32, GpuError> {
21    u32::try_from(value).map_err(|_| capacity("formula transport address exceeds u32"))
22}
23
24fn bytes(nodes: u32, wide_words: u32) -> Result<u64, GpuError> {
25    let words = nodes
26        .checked_mul(4)
27        .and_then(|n| n.checked_add(wide_words))
28        .ok_or_else(|| capacity("formula header and operand indexing exceeds u32"))?;
29    Ok(u64::from(words.max(4)) * 4)
30}
31
32impl Shape {
33    pub(crate) fn new(theory: &Theory, device: &wgpu::Limits) -> Result<Self, GpuError> {
34        let nodes = address(theory.view().len())?;
35        let edges = address(theory.parts().occurrences())?;
36        let mut wide_words = 0u32;
37        let mut auxiliary = 0u32;
38        for index in 0..theory.view().len() {
39            let node = theory.view().node(index).expect("admitted formula node");
40            if !matches!(node, NodeView::Atom(_)) {
41                auxiliary += 1;
42            }
43            if let NodeView::And(children) | NodeView::Or(children) = node
44                && children.len() > 2
45            {
46                wide_words = wide_words
47                    .checked_add(address(children.len())?)
48                    .ok_or_else(|| capacity("formula wide operand count exceeds u32"))?;
49            }
50        }
51        let bytes = bytes(nodes, wide_words)?;
52        if bytes > device.max_buffer_size || bytes > device.max_storage_buffer_binding_size {
53            return Err(capacity("formula transport exceeds granted device limits"));
54        }
55        Ok(Self {
56            bytes,
57            edges,
58            wide_words,
59            auxiliary,
60        })
61    }
62}

64/// Encode one admitted logical row. Its groups have at least two operands:
65/// pairs are inline, wider rows use offsets relative to the appended tail.
66/// This does not revalidate topology. The caller traverses the same Theory for
67/// headers and operands and checks the final tail count against Shape.
68pub(crate) fn header(node: NodeView<'_>, tail: &mut u32) -> Result<[u32; 4], GpuError> {
69    Ok(match node {
70        NodeView::False => [0, 0, 0, 0],
71        NodeView::Atom(atom) => [1, address(atom)?, 0, 0],
72        NodeView::Implies(left, right) => [4, address(left)?, address(right)?, 0],
73        NodeView::And(children) | NodeView::Or(children) => {
74            let conjunction = matches!(node, NodeView::And(_));
75            if children.len() == 2 {
76                [
77                    if conjunction { 2 } else { 3 },
78                    address(children[0])?,
79                    address(children[1])?,
80                    0,
81                ]
82            } else {
83                let length = address(children.len())?;
84                let start = *tail;
85                *tail = tail
86                    .checked_add(length)
87                    .ok_or_else(|| capacity("formula operand offset exceeds u32"))?;
88                [if conjunction { 5 } else { 6 }, start, length, 0]
89            }
90        }
91    })
92}

94/// Append complete wide rows after all 4*N header words, in the same node order.
95/// The caller reserves `Shape::bytes` and retains or drops this whole transport;
96/// no individual tail row is an independently published formula owner.
97pub(crate) fn append_operands(
98    theory: &Theory,
99    words: &mut Vec<u32>,
100    cancellation: &Cancellation,
101) -> Result<(), GpuError> {
102    for index in 0..theory.view().len() {
103        cancellation.poll().map_err(GpuError::interrupted)?;
104        if let NodeView::And(children) | NodeView::Or(children) =
105            theory.view().node(index).expect("admitted formula node")
106            && children.len() > 2
107        {
108            for &child in children {
109                cancellation.poll().map_err(GpuError::interrupted)?;
110                words.push(address(child)?);
111            }
112        }
113    }
114    if words.is_empty() {
115        words.extend([0; 4]);
116    }
117    Ok(())
118}
