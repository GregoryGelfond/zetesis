8pub(crate) fn translated(map: &[u32], index: usize) -> Result<u32, GpuError> {
9    if map.is_empty() {
10        address(index)
11    } else {
12        map.get(index)
13            .copied()
14            .ok_or_else(|| capacity("missing semantic formula address"))
15    }
16}

18/// The caller byte-admits and reserves all output, mapping and DP capacity.
19/// An empty map reservation selects the unchanged all-Boolean transport path.
20pub(crate) fn pack(
21    theory: &Theory,
22    words: &mut Vec<u32>,
23    map: &mut Vec<u32>,
24    scratch: &mut Vec<u32>,
25    cancellation: &Cancellation,
26) -> Result<(), GpuError> {
27    if map.capacity() == 0 {
28        let mut tail = 0;
29        for index in 0..theory.view().len() {
30            cancellation.poll().map_err(GpuError::interrupted)?;
31            words.extend(super::header(
32                theory.view().node(index).expect("admitted node"),
33                &mut tail,
34            )?);
35        }
36        return super::append_operands(theory, words, cancellation);
37    }
38    let mut tail = 0_u32;
39    for index in 0..theory.view().len() {
40        cancellation.poll().map_err(GpuError::interrupted)?;
41        let node = theory.view().node(index).expect("admitted formula node");
42        let output = if let Some(output) = lower_numeric(node, words, map, scratch, cancellation)? {
43            output
44        } else {
45            let output = address(words.len() / 4)?;
46            let header = match node {
47                NodeView::False => [0, 0, 0, 0],
48                NodeView::Atom(atom) => [1, address(atom)?, 0, 0],
49                NodeView::Implies(left, right) => {
50                    [4, translated(map, left)?, translated(map, right)?, 0]
51                }
52                NodeView::And(children) | NodeView::Or(children) => {
53                    let conjunction = matches!(node, NodeView::And(_));
54                    if children.len() == 2 {
55                        [
56                            if conjunction { 2 } else { 3 },
57                            translated(map, children[0])?,
58                            translated(map, children[1])?,
59                            0,
60                        ]
61                    } else {
62                        let length = address(children.len())?;
63                        let start = tail;
64                        tail = tail
65                            .checked_add(length)
66                            .ok_or_else(|| capacity("formula operand tail overflows"))?;
67                        [if conjunction { 5 } else { 6 }, start, length, 0]
68                    }
69                }
70                NodeView::Threshold(_) | NodeView::ExtremumCut(_) => {
71                    unreachable!("numeric node handled above")
72                }
73            };
74            push(words, header)?;
75            output
76        };
77        if map.len() == map.capacity() {
78            return Err(capacity("formula map exceeds admitted capacity"));
79        }
80        map.push(output);
81    }
82    for index in 0..theory.view().len() {
83        cancellation.poll().map_err(GpuError::interrupted)?;
84        if let NodeView::And(children) | NodeView::Or(children) =
85            theory.view().node(index).expect("admitted node")
86            && children.len() > 2
87        {
88            for &child in children {
89                cancellation.poll().map_err(GpuError::interrupted)?;
90                if words.len() == words.capacity() {
91                    return Err(capacity("formula operands exceed admitted capacity"));
92                }
93                words.push(translated(map, child)?);
94            }
95        }
96    }
97    Ok(())
98}

100/// Both native domains publish through the same physical sink and address map.
101fn lower_numeric(
102    node: NodeView<'_>,
103    words: &mut Vec<u32>,
104    map: &[u32],
105    scratch: &mut Vec<u32>,
106    cancellation: &Cancellation,
107) -> Result<Option<u32>, GpuError> {
108    let mut sink = Sink {
109        words,
110        map,
111        cancellation,
112        falsum: None,
113        truth: None,
114    };
115    let output = match node {
116        NodeView::Threshold(threshold) => {
117            let shape = threshold_lowering::Shape::new(threshold, || {
118                cancellation.poll().map_err(GpuError::interrupted)
119            })?;
120            if shape.scratch > scratch.capacity() {
121                return Err(capacity("threshold scratch exceeds admitted capacity"));
122            }
123            while scratch.len() < shape.scratch {
124                cancellation.poll().map_err(GpuError::interrupted)?;
125                scratch.push(0);
126            }
127            threshold_lowering::lower(threshold, scratch, &mut sink)?
128        }
129        NodeView::ExtremumCut(cut) => extremum_lowering::lower(cut, &mut sink)?,
130        _ => return Ok(None),
131    };
132    Ok(Some(output))
133}

135fn push(words: &mut Vec<u32>, header: [u32; 4]) -> Result<(), GpuError> {
136    if words.capacity().saturating_sub(words.len()) < 4 {
137        return Err(capacity("physical formula exceeds admitted capacity"));
138    }
139    words.extend(header);
140    Ok(())
141}

143struct Sink<'a> {
144    words: &'a mut Vec<u32>,
145    map: &'a [u32],
146    cancellation: &'a Cancellation,
147    falsum: Option<u32>,
148    truth: Option<u32>,
149}
150
151impl threshold_lowering::Sink for Sink<'_> {
152    type Value = u32;
153    type Error = GpuError;
154
155    fn tick(&mut self) -> Result<(), GpuError> {
156        self.cancellation.poll().map_err(GpuError::interrupted)
157    }
158
159    fn constant(&mut self, value: bool) -> Result<u32, GpuError> {
160        let falsum = if let Some(index) = self.falsum {
161            index
162        } else {
163            let index = address(self.words.len() / 4)?;
164            push(self.words, [0, 0, 0, 0])?;
165            self.falsum = Some(index);
166            index
167        };
168        if !value {
169            return Ok(falsum);
170        }
171        if let Some(index) = self.truth {
172            return Ok(index);
173        }
174        let truth = address(self.words.len() / 4)?;
175        push(self.words, [4, falsum, falsum, 0])?;
176        self.truth = Some(truth);
177        Ok(truth)
178    }
179
180    fn condition(&mut self, index: usize) -> Result<u32, GpuError> {
181        translated(self.map, index)
182    }
183
184    fn gate(&mut self, disjunction: bool, left: u32, right: u32) -> Result<u32, GpuError> {
185        self.tick()?;
186        let index = address(self.words.len() / 4)?;
187        push(
188            self.words,
189            [if disjunction { 3 } else { 2 }, left, right, 0],
190        )?;
191        Ok(index)
192    }
193}
