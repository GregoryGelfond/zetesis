1//! Useful per-world device inference over bounded lazy source chunks.
2//!
3//! The CPU source coordinator offers instances from the union snapshot. The
4//! kernel independently checks each world's positives and frozen gates and
5//! derives its head delta. No completed CPU closure or complete ground graph is
6//! uploaded. One batch reuses bounded transport capacity across chunks/rounds;
7//! immutable input prefixes are reused only within their batch/round and layout.
8//! Every active output is cleared before dispatch. Automatic
9//! performance selection remains a separate policy.
10

35/// Actual submitted device work for the latest attempted lazy batch, including
36/// submissions before a later source/device failure. No timing is a GPU-only
37/// kernel duration; waits include submission completion and mapped readback.
38#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
39pub struct LazyGpuStatistics {
40    /// Submitted nonempty source chunks.
41    pub dispatches: u64,
42    /// Sum of submitted source instances times candidate occurrences.
43    pub world_instances: u64,
44    /// Bytes uploaded in uniforms, offsets, records, snapshots and seeds.
45    pub uploaded_bytes: u64,
46    /// Successfully decoded readback bytes.
47    pub downloaded_bytes: u64,
48    /// Submitted chunks requesting at least one new transport buffer.
49    /// This counts allocation requests, including a later device failure.
50    pub transport_allocations: u64,
51    /// Submitted chunks reusing every transport buffer from an earlier chunk.
52    pub transport_reuses: u64,
53    /// Largest requested GPU buffer payload for a submitted chunk, including
54    /// retained inactive capacity. Excludes host payload and driver allocations.
55    pub peak_transport_bytes: u64,
56    /// Overlapping reasons why submitted chunks required fresh transport.
57    pub transport_replacements: LazyTransportReplacements,
58    /// Per-buffer allocation/reuse requests and prospective slack releases.
59    pub transport_usage: LazyTransportUsage,
60    /// Completed host time waiting for submissions and decoding readback.
61    pub host_wait: Duration,
62}

133    /// Check the exact seed occurrences using host source joins and useful
134    /// per-world device consequences. Source limits are shared across the batch;
135    /// transport limits apply to every nonempty chunk. `max_batch_bytes` counts
136    /// all retained GPU buffer capacities plus active host packing and decoded
137    /// words; coordinator storage is separately bounded by
138    /// `source_limits.max_host_bytes`. Rejected bindings and buffers are dropped
139    /// before allocating replacements; fitting inputs, the uniform and exactly
140    /// matching result buffers can survive. Input slack is released when needed
141    /// to preserve admission of the exact shape. All buffers are dropped on each
142    /// batch exit; driver-private deferred retirement is not counted.
143    /// Frozen seeds are uploaded once per retained layout; snapshots once per
144    /// retained layout and immutable source round. Replaced input buffers need
145    /// fresh writes. Chunk records, offsets, uniform and cleared output remain
146    /// per dispatch. Round indices have meaning only in this batch-owned scope.
147    ///
148    /// # Errors
149    /// Returns no completed check on interrupted source coverage, capacity,
150    /// device, timeout, readback or malformed-output failure. Charged source
151    /// progress is retained in the error, device work in [`Self::statistics`].
152    /// The whole batch leases the shared context, including source validation.
153    /// Busy contention returns before source work; execution-stage failures
154    /// invalidate the context for every primitive sharing it.
155    /// After leasing, cancellation is checked before context health. A known
156    /// invalidated context refuses before source admission, including empty calls.
157    pub fn check_batch(
158        &mut self,
159        program: &Program,
160        seeds: &[Seed],
161        source_limits: lazy::Limits,
162        limits: GpuLimits,
163        cancellation: &Cancellation,
164    ) -> Result<lazy::Batch, lazy::Failure<GpuError>> {
165        self.check_batch_with_source(
166            program,
167            seeds,
168            source_limits,
169            limits,
170            lazy::SourceSelection::Union,
171            cancellation,
172        )
173    }

175    /// Check the same seed occurrences with explicit host source selection.
176    /// [`lazy::SourceSelection::Worlds`] removes positive prefixes with no
177    /// current-world witness before transport. The device still checks every
178    /// positive antecedent and frozen gate; its ABI and execution are unchanged.
179    /// Source mask storage/work share the supplied source budgets, and their
180    /// progress is distinct from actual submitted device work. Host join storage
181    /// is reused only within this batch; each round rebuilds membership truth.
182    ///
183    /// # Errors
184    /// Preserves [`Self::check_batch`]'s failure and incomplete-coverage contract.
185    /// This opt-in path requires its own physical-device qualification; portable
186    /// semantic and transport tests do not establish hardware execution.
187    pub fn check_batch_with_source(
188        &mut self,
189        program: &Program,
190        seeds: &[Seed],
191        source_limits: lazy::Limits,
192        limits: GpuLimits,
193        selection: lazy::SourceSelection,
194        cancellation: &Cancellation,
195    ) -> Result<lazy::Batch, lazy::Failure<GpuError>> {
196        self.check_batch_with_source_views(
197            program,
198            seeds.iter().map(Seed::view),
199            source_limits,
200            limits,
201            selection,
202            cancellation,
203        )
204    }

206    /// Check borrowed candidate occurrences through the same lazy source rounds.
207    /// No owned seed or intermediate view vector is materialized. The iterator
208    /// preserves exact input order and must be replayable for source admission
209    /// and final seed/closure agreement. Device work starts only for emitted
210    /// source chunks; constructing a view does not compile a static graph.
211    ///
212    /// # Errors
213    /// Preserves [`Self::check_batch`]'s failure and incomplete-coverage contract.
214    pub fn check_batch_views<'seed>(
215        &mut self,
216        program: &Program,
217        seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
218        source_limits: lazy::Limits,
219        limits: GpuLimits,
220        cancellation: &Cancellation,
221    ) -> Result<lazy::Batch, lazy::Failure<GpuError>> {
222        self.check_batch_with_source_views(
223            program,
224            seeds,
225            source_limits,
226            limits,
227            lazy::SourceSelection::Union,
228            cancellation,
229        )
230    }

232    /// Check borrowed candidate occurrences using explicit host source selection.
233    /// The implementation and device protocol are shared with
234    /// [`Self::check_batch_with_source`]; views change input ownership only.
235    ///
236    /// # Errors
237    /// Preserves [`Self::check_batch_with_source`]'s failure and coverage contract.
238    pub fn check_batch_with_source_views<'seed>(
239        &mut self,
240        program: &Program,
241        seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
242        source_limits: lazy::Limits,
243        limits: GpuLimits,
244        selection: lazy::SourceSelection,
245        cancellation: &Cancellation,
246    ) -> Result<lazy::Batch, lazy::Failure<GpuError>> {
247        self.statistics = LazyGpuStatistics::default();
248        let context = self.runtime.context.clone();
249        let _lease = context.lease().map_err(|error| lazy::Failure {
250            cause: lazy::Cause::Execution(error),
251            progress: lazy::Progress::default(),
252        })?;
253        cancellation.poll().map_err(|stop| lazy::Failure {
254            cause: lazy::Cause::Source(stop),
255            progress: lazy::Progress::default(),
256        })?;
257        context.check_health().map_err(|error| lazy::Failure {
258            cause: lazy::Cause::Execution(error),
259            progress: lazy::Progress::default(),
260        })?;
261        let mut transport = None;
262        let result = lazy::check_with_source_views(
263            program,
264            seeds,
265            source_limits,
266            selection,
267            cancellation,
268            |chunk| self.execute(chunk, limits, cancellation, &mut transport),
269        );
270        if matches!(
271            &result,
272            Err(lazy::Failure {
273                cause: lazy::Cause::InvalidOutput,
274                ..
275            })
276        ) {
277            self.runtime.invalidate();
278        }
279        result.map_err(|mut failure| {
280            if let lazy::Cause::Execution(error) = &failure.cause
281                && let Some(stop) = error.interruption
282            {
283                failure.cause = lazy::Cause::Source(stop);
284            }
285            failure
286        })
287    }

289    // The complete public batch retains the context lease, including source
290    // validation after this chunk returns. Direct transport tests own an isolated
291    // executor and exercise this internal step without another context client.
292    fn execute(
293        &mut self,
294        chunk: &lazy::Chunk<'_>,
295        limits: GpuLimits,
296        cancellation: &Cancellation,
297        transport: &mut Option<Transport>,
298    ) -> Result<Vec<u32>, GpuError> {
299        self.runtime.check_health()?;
300        let epoch = crate::packing::next_epoch(self.epoch)?;
301        let plan = Plan::new(epoch, chunk, limits, self.runtime.limits())?;
302        self.epoch = epoch.get();
303        let scopes = ErrorScopes::new(self.runtime.device());
304        let outcome = self.dispatch(chunk, limits, &plan, cancellation, transport);
305        let result = self.runtime.complete(scopes, outcome);
306        if result.is_err() {
307            // Discard this transport after any failed read. A pending wait also
308            // invalidates the shared context; a settled caller stop need not.
309            *transport = None;
310        }
311        result
312    }

314    fn dispatch(
315        &mut self,
316        chunk: &lazy::Chunk<'_>,
317        limits: GpuLimits,
318        plan: &Plan,
319        cancellation: &Cancellation,
320        cached: &mut Option<Transport>,
321    ) -> Result<runtime::Completion<Vec<u32>>, GpuError> {
322        let device = self.runtime.device();
323        let mut selection = Selection::new(
324            cached.as_ref().map(|transport| transport.capacity),
325            plan,
326            limits.max_batch_bytes,
327        );
328        selection.uploads = upload::Uploads::needed(
329            cached.as_ref().and_then(|transport| transport.upload),
330            chunk.into(),
331            selection.retention,
332        );
333        let counters = self.statistics.submitted(plan, selection)?;
334        if !selection.transition.is_reuse() {
335            *cached = Some(cached.take().map_or_else(
336                || Transport::new(&self.runtime, selection.capacity),
337                |previous| previous.replace(&self.runtime, selection),
338            ));
339        }
340        let transport = cached.as_mut().ok_or_else(|| {
341            GpuError::new(GpuErrorKind::Device, "lazy transport is not initialized")
342        })?;
343        let submission = transport.submit(&self.runtime, chunk, plan, selection.uploads);
344        self.statistics = counters;
345        let start = std::time::Instant::now();
346        let result = runtime::read_polled(
347            device,
348            transport.readback(),
349            submission,
350            limits.timeout,
351            || cancellation.poll().map_err(GpuError::interrupted),
352            |words| plan.decode(words),
353        );
354        self.statistics.host_wait = self
355            .statistics
356            .host_wait
357            .checked_add(start.elapsed())
358            .ok_or_else(|| {
359                GpuError::new(GpuErrorKind::Capacity, "lazy host-wait duration overflow")
360            })?;
361        if result.is_ok() {
362            self.statistics.downloaded_bytes = self
363                .statistics
364                .downloaded_bytes
365                .checked_add(plan.result_bytes)
366                .ok_or_else(|| {
367                    GpuError::new(
368                        GpuErrorKind::Capacity,
369                        "lazy readback byte counter overflow",
370                    )
371                })?;
372        }
373        Ok(result)
374    }
375}

377impl LazyGpuStatistics {
378    fn submitted(self, plan: &Plan, selection: Selection) -> Result<Self, GpuError> {
379        let capacity = || GpuError::new(GpuErrorKind::Capacity, "lazy submission counter overflow");
380        Ok(Self {
381            dispatches: self.dispatches.checked_add(1).ok_or_else(capacity)?,
382            world_instances: self
383                .world_instances
384                .checked_add(u64::from(plan.dimensions[1]) * u64::from(plan.dimensions[2]))
385                .ok_or_else(capacity)?,
386            uploaded_bytes: self
387                .uploaded_bytes
388                .checked_add(selection.uploads.bytes(plan)?)
389                .ok_or_else(capacity)?,
390            transport_allocations: self
391                .transport_allocations
392                .checked_add(u64::from(!selection.transition.is_reuse()))
393                .ok_or_else(capacity)?,
394            transport_reuses: self
395                .transport_reuses
396                .checked_add(u64::from(selection.transition.is_reuse()))
397                .ok_or_else(capacity)?,
398            peak_transport_bytes: self
399                .peak_transport_bytes
400                .max(selection.capacity.bytes().ok_or_else(capacity)?),
401            transport_replacements: self.transport_replacements.record(selection.transition)?,
402            transport_usage: self.transport_usage.record(selection)?,
403            ..self
404        })
405    }
406}

408fn check_limits(limits: &wgpu::Limits) -> Result<(), GpuError> {
409    // The immutable-round shader has no workgroup variables. Its five storage
410    // buffers and uniform are independent of the static closure kernel's scratch.
411    for (name, required, available) in [
412        (
413            "workgroup invocations",
414            64,
415            limits.max_compute_invocations_per_workgroup,
416        ),
417        ("workgroup width", 64, limits.max_compute_workgroup_size_x),
418        (
419            "storage bindings",
420            5,
421            limits.max_storage_buffers_per_shader_stage,
422        ),
423        (
424            "uniform bindings",
425            1,
426            limits.max_uniform_buffers_per_shader_stage,
427        ),
428        ("bindings per group", 6, limits.max_bindings_per_bind_group),
429    ] {
430        if available < required {
431            return Err(GpuError::new(
432                GpuErrorKind::Capacity,
433                format!("lazy {name}: need {required}, device provides {available}"),
434            ));
435        }
436    }
437    Ok(())
438}
