1//! Exact batched reduct checking with static and lazy source executors.
2//!
3//! Static workgroups own complete candidate closures. [`GpuLazyOracle`] instead
4//! evaluates bounded source instances over immutable per-world snapshots, with
5//! host joins and round barriers. Both use integer truth and frozen seed gates.
6//! A returned rejection is logical; adapter, capacity, timeout, validation and
7//! device errors remain distinct failures.
8//!
9//! [`GpuContext`] allows distinct primitives to retain prepared subjects on one
10//! device. Existing constructors create independent contexts; `from_context`
11//! constructors share one context and its failure boundary. Calls are serialized
12//! without waiting for a competing call: overlap returns [`GpuErrorKind::Busy`].
13//! Each primitive's resource limit keeps its documented local scope.
14//!
15//! The shader implements the monotone event/iteration schedules modeled in the
16//! Lean specification. The Rust-to-WGSL packing and device implementation are
17//! tested refinements, not mechanically verified implementations.

94/// Per-dispatch limits, checked before packing or creating GPU buffers.
95#[derive(Clone, Copy, Debug)]
96pub struct GpuLimits {
97    /// Maximum candidates in one dispatch, additionally bounded by the device.
98    pub max_candidates: usize,
99    /// Maximum requested bytes for resident GPU buffers, new host packing,
100    /// returned closure arrays, and their metadata. Every nonempty call counts its
101    /// resident graph and transport, even when reused. Existing caller-owned
102    /// objects, allocator rounding, and wgpu/driver-private allocations and
103    /// deferred resource retirement are outside this sum.
104    pub max_batch_bytes: u64,
105    /// Maximum host wait for the submitted GPU work. A timeout invalidates the
106    /// shared device context and returns an error, never a candidate rejection.
107    /// Must be positive for a nonempty batch; empty batches perform no wait.
108    pub timeout: Duration,
109}
110
111impl Default for GpuLimits {
112    fn default() -> Self {
113        Self {
114            max_candidates: 1024,
115            max_batch_bytes: 128 * 1024 * 1024,
116            timeout: Duration::from_secs(30),
117        }
118    }
119}

121/// A failure outside the logical accept/reject relation.
122#[derive(Clone, Copy, Debug, PartialEq, Eq)]
123pub enum GpuErrorKind {
124    /// No compatible adapter could be obtained.
125    AdapterUnavailable,
126    /// The adapter did not satisfy the caller's hardware policy.
127    AdapterRefused,
128    /// A static dimension or caller/device resource limit was exceeded.
129    Capacity,
130    /// Host or device memory could not be allocated.
131    Allocation,
132    /// Another operation holds this shared context; retry after it completes.
133    /// No work was submitted and context health remains unchanged.
134    Busy,
135    /// A candidate does not belong to this compiled program.
136    Seed,
137    /// Shader or command validation failed.
138    Validation,
139    /// Device creation, execution, or a device callback reported failure.
140    Device,
141    /// The bounded host wait expired.
142    Timeout,
143    /// Caller cancellation or deadline was observed; no complete batch is returned.
144    Interrupted,
145    /// Readback mapping, shape, status, or tail-bit validation failed.
146    Readback,
147}

149/// Typed backend failure with explanatory detail; never a logical rejection.
150#[derive(Clone, Debug, PartialEq, Eq)]
151pub struct GpuError {
152    kind: GpuErrorKind,
153    detail: String,
154    interruption: Option<zetesis_cpu::Stop>,
155}
156
157impl GpuError {
158    fn new(kind: GpuErrorKind, detail: impl Into<String>) -> Self {
159        Self {
160            kind,
161            detail: detail.into(),
162            interruption: None,
163        }
164    }
165
166    fn interrupted(stop: zetesis_cpu::Stop) -> Self {
167        Self {
168            kind: GpuErrorKind::Interrupted,
169            detail: stop.to_string(),
170            interruption: Some(stop),
171        }
172    }
173
174    /// Machine-readable failure class.
175    #[must_use]
176    pub fn kind(&self) -> GpuErrorKind {
177        self.kind
178    }
179
180    /// Context identifying the failed operation or dimension.
181    #[must_use]
182    pub fn detail(&self) -> &str {
183        &self.detail
184    }
185
186    /// Exact control stop, when this error reports an interrupted operation.
187    /// A preceding scope/device fault replaces the stop and returns `None`.
188    #[must_use]
189    pub const fn interruption(&self) -> Option<zetesis_cpu::Stop> {
190        self.interruption
191    }
192}

194impl fmt::Display for GpuError {
195    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
196        write!(formatter, "{:?}: {}", self.kind, self.detail)
197    }
198}

202/// One complete static reduct closure and its exact gate/constraint verdict.
203#[derive(Clone, Debug, PartialEq, Eq)]
204pub struct GpuCheck {
205    closure_words: Vec<u32>,
206    status: u32,
207}
208
209impl GpuCheck {
210    /// True exactly when the closure's gate-carrier projection equals the
211    /// seed and every enabled constraint is false.
212    #[must_use]
213    pub fn accepted(&self) -> bool {
214        self.status == 0
215    }
216
217    /// Whether an enabled constraint's positive body holds in the closure.
218    #[must_use]
219    pub fn constraint_violated(&self) -> bool {
220        self.status & 1 != 0
221    }
222
223    /// Whether the closure's projection differs from the frozen seed.
224    #[must_use]
225    pub fn seed_mismatch(&self) -> bool {
226        self.status & 2 != 0
227    }
228
229    /// Dense closure words in this `GroundProgram`'s atom order. The caller
230    /// may reconstruct a symbolic model through `GroundProgram::model_from_words`.
231    #[must_use]
232    pub fn closure_words(&self) -> &[u32] {
233        &self.closure_words
234    }
235}

360    /// Validate, pack, dispatch, and read back a bounded static candidate batch.
361    /// Results preserve the input seed order. No source grounding occurs here.
362    /// One immutable Program instance remains resident between calls; changing
363    /// Program identity replaces it. Transport is reused for equal batch sizes
364    /// and replaced for a changed size. Completed reads are unmapped before reuse.
365    /// Every returned record must identify its submission and input position,
366    /// carry a nonzero completion marker, and agree with the seed projection.
367    /// Per candidate, host validation visits the closure words and gate atoms;
368    /// the original gate IDs are borrowed without allocating another mask.
369    /// An empty batch validates device health and graph admission, then returns
370    /// without checking dispatch limits, allocating transport or advancing the
371    /// submission epoch. It leaves existing residency intact.
372    ///
373    /// # Errors
374    /// Reports seed identity, capacity, allocation, device, timeout, validation,
375    /// or readback failures. A failing dispatch returns no partial batch.
376    /// Device/readback failures invalidate the entire context; retry requires a
377    /// fresh context. Capacity, seed and Busy refusals leave the context reusable.
378    /// The checked 32-bit submission sequence never wraps; exhausting it requires
379    /// a new oracle, even after clearing residency.
380    pub fn check_batch(
381        &mut self,
382        program: &GroundProgram,
383        seeds: &[Seed],
384        limits: GpuLimits,
385    ) -> Result<Vec<GpuCheck>, GpuError> {
386        self.check_batch_views(program, seeds.iter().map(Seed::view), limits)
387    }
388

408    /// Check the same static candidate occurrences through borrowed seed views.
409    /// The cloneable exact-size iterator preserves input order across admission
410    /// and packing. Wrapping owned seeds or shared selections allocates no view

412    /// Residency, submission identity and independent projection validation are

417    pub fn check_batch_views<'seed>(
418        &mut self,
419        program: &GroundProgram,
420        seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
421        limits: GpuLimits,
422    ) -> Result<Vec<GpuCheck>, GpuError> {
423        self.check_batch_views_with_cancellation(program, seeds, limits, &Cancellation::default())
424    }

426    /// Check borrowed seeds with cooperative caller control. Busy admission
427    /// precedes control; control precedes health and graph admission, even for
428    /// empty batches. Polls also follow host packing and occur between device
429    /// waits of at most 50 ms. Host operations, drivers and scope settlement are
430    /// not preempted, so this is not a hard wall-clock deadline.
431    ///
432    /// # Errors
433    /// Returns [`GpuErrorKind::Interrupted`] with the exact stop and no partial
434    /// batch. Pre-submission stops leave health reusable; an interrupted wait
435    /// invalidates it. Scope/device faults retain priority over interruption.
436    /// Other failures follow [`Self::check_batch`].
437    pub fn check_batch_views_with_cancellation<'seed>(
438        &mut self,
439        program: &GroundProgram,
440        seeds: impl ExactSizeIterator<Item = SeedView<'seed>> + Clone,
441        limits: GpuLimits,
442        cancellation: &Cancellation,
443    ) -> Result<Vec<GpuCheck>, GpuError> {
444        self.last_batch_stats = None;
445        let context = self.runtime.context.clone();
446        let _lease = context.lease()?;
447        cancellation.poll().map_err(GpuError::interrupted)?;
448        self.runtime.check_health()?;
449        let fresh_graph = if self
450            .resident
451            .as_ref()
452            .is_some_and(|resident| resident.plan.matches(program))
453        {
454            None
455        } else {
456            Some(GraphPlan::new(program, self.runtime.limits())?)
457        };
458        let graph = fresh_graph
459            .as_ref()
460            .or_else(|| self.resident.as_ref().map(|resident| &resident.plan))
461            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing graph plan"))?;
462        let candidate_count = seeds.len();
463        if candidate_count == 0 {
464            return Ok(Vec::new());
465        }
466        let epoch = packing::next_epoch(self.epoch)?;
467        let plan = BatchPlan::for_graph(
468            graph,
469            candidate_count,
470            limits,
471            self.runtime.limits(),
472            fresh_graph.is_some(),
473            epoch,
474        )?;
475        // Refuse foreign seeds before replacing resident resources or packing.
476        if seeds
477            .clone()
478            .any(|seed| !program.program().same_instance(seed.program()))
479        {
480            return Err(GpuError::new(
481                GpuErrorKind::Seed,
482                "candidate belongs to another Program",
483            ));
484        }
485        let stats = GpuBatchStats {
486            graph_uploaded: fresh_graph.is_some(),
487            transport_allocated: fresh_graph.is_some()
488                || self
489                    .resident
490                    .as_ref()
491                    .is_none_or(|resident| !resident.has_transport(&plan)),
492            resident_graph_bytes: graph.resident_bytes,
493            resident_transport_bytes: plan.transport_bytes,
494            accounted_batch_bytes: plan.accounted_bytes,
495        };
496        // Evict before new host/device allocations so the old authored buffers
497        // cannot silently enlarge this call's accounted resident working set.
498        if fresh_graph.is_some() {
499            self.resident = None;
500        } else if stats.transport_allocated
501            && let Some(resident) = self.resident.as_mut()
502        {
503            resident.clear_transport();
504        }
505        let packed = PackedSeeds::new(program, seeds, &plan)?;
506        let packed_graph = fresh_graph
507            .as_ref()
508            .map(|graph| PackedGraph::new(program, graph))
509            .transpose()?;
510        cancellation.poll().map_err(GpuError::interrupted)?;
511        let scopes = ErrorScopes::new(self.runtime.device());
512        if let Some((graph, packed)) = fresh_graph.zip(packed_graph) {
513            self.resident = Some(ResidentGraph::new(self.runtime.device(), graph, &packed));
514        }
515        // Never reuse an identity after a dispatch attempt, including one whose
516        // readback fails. Clearing residency does not reset this sequence.
517        self.epoch = epoch.get();
518        let outcome = self
519            .resident
520            .as_mut()
521            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing resident graph"))
522            .and_then(|resident| {
523                resident.dispatch(&self.runtime, &packed, &plan, limits.timeout, cancellation)
524            });
525        let result = self.runtime.complete(scopes, outcome);
526        if result.is_ok() {
527            self.last_batch_stats = Some(stats);
528        }
529        result
530    }
531}

533fn check_adapter_limits(limits: &wgpu::Limits) -> Result<(), GpuError> {
534    for (name, required, available) in [
535        (
536            "workgroup invocations",
537            WORKGROUP_SIZE,
538            limits.max_compute_invocations_per_workgroup,
539        ),
540        (
541            "workgroup width",
542            WORKGROUP_SIZE,
543            limits.max_compute_workgroup_size_x,
544        ),
545        (
546            "workgroup bytes",
547            WORKGROUP_BYTES,
548            limits.max_compute_workgroup_storage_size,
549        ),
550        (
551            "storage bindings",
552            5,
553            limits.max_storage_buffers_per_shader_stage,
554        ),
555        (
556            "uniform bindings",
557            1,
558            limits.max_uniform_buffers_per_shader_stage,
559        ),
560        ("bindings per group", 6, limits.max_bindings_per_bind_group),
561    ] {
562        if available < required {
563            return Err(GpuError::new(
564                GpuErrorKind::Capacity,
565                format!("{name}: need {required}, device provides {available}"),
566            ));
567        }
568    }
569    Ok(())
570}
