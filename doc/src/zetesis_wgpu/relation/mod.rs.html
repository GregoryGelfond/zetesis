1//! Bounded equality filtering over a borrowed immutable relation snapshot.
2//!
3//! Uploaded u32 columns use the core Relation's local equality dictionary.
4//! They are derived coordinates, not canonical term IDs or arithmetic values.
5//! The Relation resolves typed query terms (including foreign-catalog terms)
6//! into that dictionary; preparation and filtering require the same Relation
7//! instance. Canonical payload remains with the borrowed source catalog, and
8//! returned mask bits name original row occurrences, including repetitions.

31/// Inclusive bounds for a prepared view and one equality-filter invocation.
32#[derive(Clone, Copy, Debug)]
33pub struct RelationGpuLimits {
34    /// Queries in one batch, preserving repeated query occurrences.
35    pub max_queries: usize,
36    /// Authored resident columns, transport, host packing and returned masks.
37    /// Preflight checks minimum payload; before device effects, host reservations
38    /// additionally charge all retained element capacity, including spare slots.
39    /// Borrowed source atoms, relation storage and caller query allocations are
40    /// excluded and must be charged by the enclosing operation. Driver-private
41    /// storage, allocator metadata and delayed retirement are also excluded.
42    pub max_bytes: u64,
43    /// Scheduled row flags, equality comparisons, bit-fold steps and receipts.
44    /// For R rows, Q queries, T=ceil(R/64), W=ceil(R/32), and E total query
45    /// equalities, the nonempty schedule charges Q*(64*T+32*W+T)+R*E.
46    /// Each tile has one fixed-size completion/identity receipt unit. These are
47    /// policy units, not a measured GPU instruction count.
48    pub max_work: u64,
49    /// Bounded wait for submitted work; control is polled between short waits.
50    /// Device creation, shader compilation and driver allocations are outside it.
51    /// Must be positive when rows and queries are nonempty; empty calls do not wait.
52    pub timeout: Duration,
53}
54
55impl Default for RelationGpuLimits {
56    fn default() -> Self {
57        Self {
58            max_queries: 256,
59            max_bytes: 128 * 1024 * 1024,
60            max_work: 100_000_000,
61            timeout: Duration::from_secs(30),
62        }
63    }
64}

107/// Ordered equality masks tied to their original immutable relation owner.
108///
109/// Mask position is the input query position; bit position is the original row
110/// occurrence, not its external catalog index. A mask is an equality filter,
111/// not a complete pattern match or an answer-set membership certificate.
112pub struct RelationGpuMasks<'owner, 'source> {
113    relation: &'owner Relation<'source>,
114    queries: usize,
115    words_per_query: usize,
116    words: Vec<u32>,
117}
118
119impl<'owner, 'source> RelationGpuMasks<'owner, 'source> {
120    /// Immutable owner supplying dictionary and original catalog mapping.
121    #[must_use]
122    pub const fn relation(&self) -> &'owner Relation<'source> {
123        self.relation
124    }
125
126    /// Complete query occurrences in original input order.
127    #[must_use]
128    pub const fn query_count(&self) -> usize {
129        self.queries
130    }
131
132    /// Packed equality selection for one query; unused tail bits are zero.
133    #[must_use]
134    pub fn words(&self, query: usize) -> Option<&[u32]> {
135        if query >= self.queries {
136            return None;
137        }
138        let start = query * self.words_per_query;
139        Some(&self.words[start..start + self.words_per_query])
140    }
141
142    /// Reconstruct one checked ordered row selection without retaining all
143    /// query selections simultaneously. The caller still performs the complete
144    /// term-pattern match. The limit includes this mask object, its full retained
145    /// capacity, the relation and one returned position vector. Other live
146    /// selections remain caller owned. Shared core reconstruction charges two
147    /// scans of the mask words and two operations per selected row; the returned
148    /// Selection's work counter covers that complete reconstruction.
149    ///
150    /// # Errors
151    /// Refuses an absent query, exceeded limits or allocation failure. Structural
152    /// row validation alone does not establish that a GPU mask is complete; that
153    /// correspondence is the shader/decoder contract and device qualification.
154    pub fn selection(
155        &self,
156        query: usize,
157        limits: relation::Limits,
158    ) -> Result<Selection<'owner, 'source>, RelationGpuError> {
159        let words = self
160            .words(query)
161            .ok_or_else(|| capacity("relation query is out of range"))?;
162        let mask_bytes = size_of::<Self>() + self.words.capacity() * size_of::<u32>();
163        // Core charges this borrowed query slice. This wrapper charges the
164        // remaining batch capacity and object, keeping the relation counted once.
165        let external = mask_bytes - size_of_val(words);
166        let available = limits
167            .max_bytes
168            .checked_sub(external)
169            .ok_or_else(|| capacity("relation masks exceed selection byte ceiling"))?;
170        self.relation
171            .selection_from_mask(
172                words,
173                relation::Limits {
174                    max_bytes: available,
175                    ..limits
176                },
177            )
178            .map_err(RelationGpuError::Relation)
179    }
180}

182/// An incomplete filtering operation; never an empty-result substitution.
183#[derive(Clone, Debug, PartialEq, Eq)]
184pub enum RelationGpuError {
185    /// Shared cancellation or deadline was observed.
186    Stopped(Stop),
187    /// Relation ownership, checked selection or resource failure.
188    Relation(relation::Failure),
189    /// Adapter, capacity, execution or readback failure.
190    Gpu(GpuError),
191}
192
193impl fmt::Display for RelationGpuError {
194    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
195        match self {
196            Self::Stopped(stop) => stop.fmt(f),
197            Self::Relation(error) => error.fmt(f),
198            Self::Gpu(error) => error.fmt(f),
199        }
200    }
201}
202impl std::error::Error for RelationGpuError {
203    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
204        Some(match self {
205            Self::Stopped(stop) => stop,
206            Self::Relation(error) => error,
207            Self::Gpu(error) => error,
208        })
209    }
210}
211impl From<GpuError> for RelationGpuError {
212    fn from(error: GpuError) -> Self {
213        match error.interruption {
214            Some(stop) => Self::Stopped(stop),
215            None => Self::Gpu(error),
216        }
217    }
218}
219
220fn capacity(detail: &str) -> GpuError {
221    GpuError::new(GpuErrorKind::Capacity, detail)
222}
223fn poll(cancellation: &Cancellation) -> Result<(), GpuError> {
224    cancellation.poll().map_err(|stop| GpuError {
225        kind: GpuErrorKind::Device,
226        detail: "relation equality filtering interrupted".to_owned(),
227        interruption: Some(stop),
228    })
229}
