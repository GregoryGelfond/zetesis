1//! Device resources for one immutable program and one exact transport shape.

3use std::time::Duration;
4
5use crate::packing::{self, BatchPlan, GraphPlan, PackedGraph, PackedSeeds};
6use crate::runtime::{self, Dispatch, buffer, entry, initialized};
7use crate::{GpuCheck, GpuError, GpuErrorKind};

9pub(crate) struct ResidentGraph {
10    pub(crate) plan: GraphPlan,
11    rules: wgpu::Buffer,
12    antecedents: wgpu::Buffer,
13    carrier: wgpu::Buffer,
14    transport: Option<Transport>,
15}
16
17impl ResidentGraph {
18    pub(crate) fn new(device: &wgpu::Device, plan: GraphPlan, packed: &PackedGraph) -> Self {
19        Self {
20            plan,
21            rules: initialized(
22                device,
23                "resident rules",
24                &packed.rules,
25                wgpu::BufferUsages::STORAGE,
26            ),
27            antecedents: initialized(
28                device,
29                "resident antecedents",
30                &packed.antecedents,
31                wgpu::BufferUsages::STORAGE,
32            ),
33            carrier: initialized(
34                device,
35                "resident carrier",
36                &packed.carrier,
37                wgpu::BufferUsages::STORAGE,
38            ),
39            transport: None,
40        }
41    }
42
43    pub(crate) fn has_transport(&self, plan: &BatchPlan) -> bool {
44        self.transport
45            .as_ref()
46            .is_some_and(|transport| transport.world_count == plan.world_count)
47    }
48
49    pub(crate) fn clear_transport(&mut self) {
50        self.transport = None;
51    }
52
53    pub(crate) fn dispatch(
54        &mut self,
55        runtime: &runtime::Runtime,
56        packed: &PackedSeeds<'_>,
57        plan: &BatchPlan,
58        timeout: Duration,
59        cancellation: &zetesis_cpu::Cancellation,
60    ) -> Result<runtime::Completion<Vec<GpuCheck>>, GpuError> {
61        let device = runtime.device();
62        let queue = runtime.queue();
63        let pipeline = &runtime.pipeline;
64        if self.transport.is_none() {
65            self.transport = Some(Transport::new(device, pipeline, self, packed, plan));
66        }
67        let transport = self
68            .transport
69            .as_ref()
70            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing batch transport"))?;
71        queue.write_buffer(&transport.params, 0, bytemuck::cast_slice(&packed.params));
72        queue.write_buffer(&transport.seeds, 0, bytemuck::cast_slice(&packed.seeds));
73        let submission = runtime::submit(
74            device,
75            queue,
76            &Dispatch {
77                command_label: "static reduct batch",
78                pass_label: "independent candidate worlds",
79                pipeline,
80                group: &transport.group,
81                workgroups: [plan.world_count, 1, 1],
82                result: &transport.result,
83                readback: &transport.readback,
84                result_bytes: plan.result_bytes,
85            },
86        );
87        Ok(runtime::read_polled(
88            device,
89            &transport.readback,
90            submission,
91            timeout,
92            || cancellation.poll().map_err(GpuError::interrupted),
93            |words| packing::decode(words, plan, packed),
94        ))
95    }
96}

98struct Transport {
99    world_count: u32,
100    params: wgpu::Buffer,
101    seeds: wgpu::Buffer,
102    result: wgpu::Buffer,
103    readback: wgpu::Buffer,
104    group: wgpu::BindGroup,
105}
106
107impl Transport {
108    fn new(
109        device: &wgpu::Device,
110        pipeline: &wgpu::ComputePipeline,
111        graph: &ResidentGraph,
112        packed: &PackedSeeds<'_>,
113        plan: &BatchPlan,
114    ) -> Self {
115        // Dimensions remain fixed for this transport shape; every dispatch
116        // rewrites its checked epoch as well as the candidate bits.
117        let params = initialized(
118            device,
119            "parameters",
120            &packed.params,
121            wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
122        );
123        let seeds = buffer(
124            device,
125            "frozen seeds",
126            plan.seed_bytes,
127            wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
128        );
129        let result = buffer(
130            device,
131            "reduct closures and verdicts",
132            plan.result_bytes,
133            wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_SRC,
134        );
135        let readback = buffer(
136            device,
137            "reduct readback",
138            plan.result_bytes,
139            wgpu::BufferUsages::MAP_READ | wgpu::BufferUsages::COPY_DST,
140        );
141        let group = device.create_bind_group(&wgpu::BindGroupDescriptor {
142            label: Some("resident static epoch"),
143            layout: &pipeline.get_bind_group_layout(0),
144            entries: &[
145                entry(0, &params),
146                entry(1, &graph.rules),
147                entry(2, &graph.antecedents),
148                entry(3, &seeds),
149                entry(4, &graph.carrier),
150                entry(5, &result),
151            ],
152        });
153        Self {
154            world_count: plan.world_count,
155            params,
156            seeds,
157            result,
158            readback,
159            group,
160        }
161    }
