33// Compiled immutable resources, plus a fitted kernel's write-once pipeline
34// slots. Formula profiles share this owner while keeping residency, epochs and
35// result accounting in each fresh oracle.
36pub(crate) struct Runtime {
37    pub(crate) context: GpuContext,
38    pub(crate) kernel: crate::kernel::Kernel,
39    pipelines: Pipelines,
40}

42/// A runtime's pipelines, by kernel kind.
43enum Pipelines {
44    /// A static kernel's one pipeline.
45    Static(wgpu::ComputePipeline),
46    /// A formula or tight kernel, compiled by workgroup size.
47    Sized(Sized),
48}
49
50/// A sized kernel's shader module, created and validated with the kernel, and
51/// one slot for each size it can dispatch: its fixed size, whose full pipeline
52/// is compiled when the kernel is created, or each fitted size the context
53/// grants. Every other pipeline is compiled by the first check that dispatches
54/// it, under the context lease, which is exclusive across every primitive on
55/// the context; reads take no lock.
56struct Sized {
57    shader: wgpu::ShaderModule,
58    entry_point: &'static str,
59    dense_entry_point: Option<&'static str>,
60    label: &'static str,
61    slots: Vec<(crate::GpuWorkgroupSize, Slot)>,
62    /// Test seam: the invocation count compiled for the next first need,
63    /// standing in for a compilation the device refuses.
64    #[cfg(test)]
65    refused: std::sync::Mutex<Option<u32>>,
66}

68/// One size's write-once pipelines, one per variant.
69#[derive(Default)]
70struct Slot {
71    full: std::sync::OnceLock<wgpu::ComputePipeline>,
72    dense: std::sync::OnceLock<wgpu::ComputePipeline>,
73}

75impl Slot {
76    const fn of(&self, variant: Variant) -> &std::sync::OnceLock<wgpu::ComputePipeline> {
77        match variant {
78            Variant::Full => &self.full,
79            Variant::Dense => &self.dense,
80        }
81    }
82}

84impl Sized {
85    /// The slot of `size` and `variant`'s entry point, or a validation failure
86    /// for a size the kernel cannot dispatch or a variant it lacks.
87    fn slot(
88        &self,
89        size: crate::GpuWorkgroupSize,
90        variant: Variant,
91    ) -> Result<(&std::sync::OnceLock<wgpu::ComputePipeline>, &'static str), GpuError> {
92        let entry_point = match variant {
93            Variant::Full => self.entry_point,
94            Variant::Dense => self.dense_entry_point.ok_or_else(|| unavailable(size))?,
95        };
96        self.slots
97            .iter()
98            .find(|(fitted, _)| *fitted == size)
99            .map(|(_, slot)| (slot.of(variant), entry_point))
100            .ok_or_else(|| unavailable(size))
101    }
102}

104/// A size this runtime cannot provide: a static kernel, another fixed size, a
105/// fitted size the context does not grant or has not compiled, or a dense
106/// variant of a kernel without one.
107fn unavailable(size: crate::GpuWorkgroupSize) -> GpuError {
108    GpuError::new(
109        GpuErrorKind::Validation,
110        format!(
111            "no {}-invocation pipeline was compiled for this kernel",
112            size.invocations()
113        ),
114    )
115}

117impl Runtime {
118    /// Open `profile`'s runtime on the device `source` provides.
119    ///
120    /// A selected or discovered source checks the profile's limits against the
121    /// adapters' advertised limits during selection and against the new
122    /// device's granted limits before any shader is created. A context source
123    /// leases the context, then checks its health and the profile's limits
124    /// against its granted limits. Only formula and tight kernels act on the
125    /// source's timestamp request.
126    pub(crate) async fn open(
127        source: GpuDeviceSource<'_>,
128        profile: DeviceProfile,
129    ) -> Result<Self, GpuError> {
130        let measures = profile.kernel.measures_compute_time();
131        let context = match source {
132            GpuDeviceSource::Selected {
133                options,
134                selection,
135                timing,
136            } => {
137                GpuContext::create_with_timing(
138                    options,
139                    selection,
140                    profile.device_label,
141                    profile.validate_limits,
142                    timing && measures,
143                )
144                .await?
145            }
146            GpuDeviceSource::Discovered {
147                discovery,
148                options,
149                timing,
150            } => {
151                discovery
152                    .context(
153                        options,
154                        profile.device_label,
155                        profile.validate_limits,
156                        timing && measures,
157                    )
158                    .await?
159            }
160            GpuDeviceSource::Context(context) => {
161                let _lease = context.lease()?;
162                context.check_health()?;
163                (profile.validate_limits)(context.limits())?;
164                return Self::compile(context, profile).await;
165            }
166        };
167        let _lease = context.lease()?;
168        Self::compile(&context, profile).await
169    }
170
171    // Every source retains its context lease through all scopes, and its
172    // profile limits are already checked. A static or fixed kernel compiles its
173    // one pipeline here; a fitted kernel creates and validates its shader
174    // module and compiles each size at the first check that dispatches it
175    // (`ensure_pipeline`).
176    async fn compile(context: &GpuContext, profile: DeviceProfile) -> Result<Self, GpuError> {
177        profile.kernel.validate(context.limits())?;
178        let source = profile.kernel.source(profile.shader, context.features())?;
179        let device = context.device();
180        let scopes = ErrorScopes::new(device);
181        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
182            label: Some(profile.shader_label),
183            source: wgpu::ShaderSource::Wgsl(source),
184        });
185        let compile = |invocations| {
186            create_pipeline(
187                device,
188                &shader,
189                profile.entry_point,
190                profile.pipeline_label,
191                invocations,
192            )
193        };
194        let pipelines = match profile
195            .kernel
196            .options()
197            .map(|options| options.workgroup_sizing)
198        {
199            None => Pipelines::Static(compile(None)),
200            Some(sizing) => {
201                let slots = match sizing {
202                    crate::GpuWorkgroupSizing::Fixed(size) => {
203                        let slot = Slot::default();
204                        let _ = slot.full.set(compile(Some(size.invocations())));
205                        vec![(size, slot)]
206                    }
207                    crate::GpuWorkgroupSizing::Fitted => {
208                        crate::kernel::candidate_sizes(sizing, context.limits())
209                            .map(|size| (size, Slot::default()))
210                            .collect()
211                    }
212                };
213                Pipelines::Sized(Sized {
214                    slots,
215                    shader,
216                    entry_point: profile.entry_point,
217                    dense_entry_point: profile.dense_entry_point,
218                    label: profile.pipeline_label,
219                    #[cfg(test)]
220                    refused: std::sync::Mutex::new(None),
221                })
222            }
223        };
224        context.complete(scopes.finish().await, Ok(()), Effects::MayBeLive)?;
225        Ok(Self {
226            context: context.clone(),
227            kernel: profile.kernel,
228            pipelines,
229        })
230    }

232    /// The pipeline of a static kernel, which has no workgroup sizing.
233    ///
234    /// # Errors
235    /// A formula or tight kernel's pipelines are chosen by size: a validation
236    /// failure.
237    pub(crate) fn static_pipeline(&self) -> Result<&wgpu::ComputePipeline, GpuError> {
238        match &self.pipelines {
239            Pipelines::Static(pipeline) => Ok(pipeline),
240            Pipelines::Sized(_) => Err(GpuError::new(
241                GpuErrorKind::Validation,
242                "this kernel's pipelines are chosen by workgroup size",
243            )),
244        }
245    }

247    /// Make the pipeline of `size` and `variant` available to the check about
248    /// to dispatch it. A pipeline already compiled, a fixed kernel's full
249    /// pipeline included, returns at once. Otherwise it is compiled now, inside
250    /// its own error scopes settled as at construction, and its slot written.
251    /// The caller holds the context lease, has finished preparing its batch and
252    /// has not opened its dispatch scopes. A freshly prepared graph can need a
253    /// size's first pipeline: for a tight theory the dense twin, since a fresh
254    /// theory's first batch never allows the update route. A resident tight
255    /// theory can need only the full kernel first, at its first batch allowing
256    /// the route, and only under fitted sizing: a fixed size compiles its full
257    /// pipeline at construction. A fresh graph's cold preparation has already
258    /// released any previous resident; a resident theory stays, and its oracle
259    /// invalidates its retained truth as after every failed call. Either way a
260    /// failure leaves the slot unwritten and the context invalidated.
261    ///
262    /// # Errors
263    /// A size or variant this runtime cannot provide is a validation failure. A
264    /// compilation failure is the scope's typed error and invalidates the
265    /// context, as the same failure at construction did; the slot stays
266    /// unwritten and is never retried.
267    pub(crate) fn ensure_pipeline(
268        &self,
269        size: crate::GpuWorkgroupSize,
270        variant: Variant,
271    ) -> Result<(), GpuError> {
272        match &self.pipelines {
273            Pipelines::Sized(sized) => {
274                let (slot, entry_point) = sized.slot(size, variant)?;
275                if slot.get().is_some() {
276                    return Ok(());
277                }
278                #[cfg(test)]
279                let invocations = sized
280                    .refused
281                    .lock()
282                    .unwrap_or_else(std::sync::PoisonError::into_inner)
283                    .take()
284                    .unwrap_or(size.invocations());
285                #[cfg(not(test))]
286                let invocations = size.invocations();
287                let pipeline = self.compile_sized(sized, entry_point, invocations)?;
288                // Every write happens under the context lease, so the slot is
289                // still empty here; a lost race could only store an equal pipeline.
290                let _ = slot.set(pipeline);
291                Ok(())
292            }
293            Pipelines::Static(_) => Err(unavailable(size)),
294        }
295    }

297    /// Compile `sized`'s `entry_point` with `invocations` per workgroup,
298    /// settling its own error scopes through the context's completion, as
299    /// construction does.
300    fn compile_sized(
301        &self,
302        sized: &Sized,
303        entry_point: &'static str,
304        invocations: u32,
305    ) -> Result<wgpu::ComputePipeline, GpuError> {
306        let device = self.device();
307        let scopes = ErrorScopes::new(device);
308        let pipeline = create_pipeline(
309            device,
310            &sized.shader,
311            entry_point,
312            sized.label,
313            Some(invocations),
314        );
315        self.context.complete(
316            pollster::block_on(scopes.finish()),
317            Ok(()),
318            Effects::MayBeLive,
319        )?;
320        Ok(pipeline)
321    }

323    /// The pipeline of `size` and `variant`, compiled at construction for a
324    /// fixed kernel's full pipeline or by `ensure_pipeline` otherwise.
325    ///
326    /// # Errors
327    /// A pipeline this runtime has not compiled, including any size of a
328    /// static kernel, is a validation failure.
329    pub(crate) fn pipeline_for(
330        &self,
331        size: crate::GpuWorkgroupSize,
332        variant: Variant,
333    ) -> Result<&wgpu::ComputePipeline, GpuError> {
334        match &self.pipelines {
335            Pipelines::Sized(sized) => sized
336                .slot(size, variant)?
337                .0
338                .get()
339                .ok_or_else(|| unavailable(size)),
340            Pipelines::Static(_) => Err(unavailable(size)),
341        }
342    }

344    /// The sizes with a compiled pipeline of either variant, ascending.
345    #[cfg(test)]
346    pub(crate) fn compiled_sizes(&self) -> Vec<crate::GpuWorkgroupSize> {
347        self.compiled()
348            .into_iter()
349            .map(|(size, _)| size)
350            .fold(Vec::new(), |mut sizes, size| {
351                if sizes.last() != Some(&size) {
352                    sizes.push(size);
353                }
354                sizes
355            })

358    /// The compiled pipelines, by size ascending and then variant, full first.
359    #[cfg(test)]
360    pub(crate) fn compiled(&self) -> Vec<(crate::GpuWorkgroupSize, Variant)> {
361        match &self.pipelines {
362            Pipelines::Static(_) => Vec::new(),
363            Pipelines::Sized(sized) => sized
364                .slots
365                .iter()
366                .flat_map(|(size, slot)| {
367                    [Variant::Full, Variant::Dense]
368                        .into_iter()
369                        .filter(|variant| slot.of(*variant).get().is_some())
370                        .map(|variant| (*size, variant))
371                })
372                .collect(),
373        }
374    }

401    // The caller retains its context lease through scope and health completion.
402    pub(crate) fn complete<T>(
403        &self,
404        scopes: ErrorScopes,
405        outcome: Result<Completion<T>, GpuError>,
406    ) -> Result<T, GpuError> {
407        self.complete_settled(pollster::block_on(scopes.finish()), outcome)
408    }
409
410    /// Complete an operation whose error scopes are in either state.
411    pub(crate) fn complete_scopes<T>(
412        &self,
413        scopes: Scopes,
414        outcome: Result<Completion<T>, GpuError>,
415    ) -> Result<T, GpuError> {
416        match scopes {
417            Scopes::Open(scopes) => self.complete(scopes, outcome),
418            Scopes::Settled(validation) => self.complete_settled(validation, outcome),
419        }
420    }
421
422    /// Complete an operation whose error scopes were already popped, as an
423    /// overlapping check pops them right after its submission, before the
424    /// caller's work runs. `validation` is their settled outcome.
425    fn complete_settled<T>(
426        &self,
427        validation: Result<(), GpuError>,
428        outcome: Result<Completion<T>, GpuError>,
429    ) -> Result<T, GpuError> {
430        let Completion {
431            outcome, effects, ..
432        } = outcome.unwrap_or_else(|error| Completion {
433            outcome: Err(error),
434            effects: Effects::MayBeLive,
435            observed: None,
436        });
437        self.context.complete(validation, outcome, effects)
438    }

440    /// For operations that submitted no queue work and released mapped access.
441    /// Scopes are fully drained and health checked before a caller interruption
442    /// may leave the context reusable. Other failures retain strict invalidation.
443    pub(crate) fn complete_unsubmitted<T>(
444        &self,
445        scopes: ErrorScopes,
446        outcome: Result<T, GpuError>,
447    ) -> Result<T, GpuError> {
448        self.context.complete(
449            pollster::block_on(scopes.finish()),
450            outcome,
451            Effects::NoSubmission,
452        )
453    }

456/// Readback outcome plus private evidence about released device effects.
457/// Only the shared readback boundary can construct this envelope. Successful
458/// queue polling alone does not establish mapping success or released access.
459#[derive(Debug)]
460pub(crate) struct Completion<T> {
461    outcome: Result<T, GpuError>,
462    effects: Effects,
463    // The instant at which the wait observed the submission's completion.
464    observed: Option<std::time::Instant>,
465}
466impl<T> Completion<T> {
467    pub(crate) fn is_ok(&self) -> bool {
468        self.outcome.is_ok()
469    }
470
471    /// The host window from `submitted` to the observed completion; absent when
472    /// the wait did not observe completion.
473    pub(crate) fn window_since(&self, submitted: std::time::Instant) -> Option<Duration> {
474        self.observed
475            .map(|observed| observed.saturating_duration_since(submitted))
476    }

500/// A compute pipeline over `shader`'s entry point, overriding `WORKGROUP_SIZE`
501/// with `invocations` when the kernel is compiled by size.
502fn create_pipeline(
503    device: &wgpu::Device,
504    shader: &wgpu::ShaderModule,
505    entry_point: &'static str,
506    label: &'static str,
507    invocations: Option<u32>,
508) -> wgpu::ComputePipeline {
509    let constants = invocations.map(|invocations| [("WORKGROUP_SIZE", f64::from(invocations))]);
510    device.create_compute_pipeline(&wgpu::ComputePipelineDescriptor {
511        label: Some(label),
512        layout: None,
513        module: shader,
514        entry_point: Some(entry_point),
515        compilation_options: compilation_options(
516            constants.as_ref().map_or(&[], |values| values.as_slice()),
517        ),
518        cache: None,
519    })
520}
521
522/// Whether every pipeline keeps WebGPU's zero-fill of workgroup memory.
523///
524/// It is `true`, wgpu's default; CONTRIBUTING ("Carry the semantic contract
525/// into Lean and Rust") records each kernel's fill status and what removing the
526/// fill requires. Every kernel satisfies the rule that makes `false` sound: each
527/// workgroup location a kernel reads was written earlier in its workgroup, by
528/// the same invocation or by another ordered before the read by a barrier or
529/// uniform load, and every workgroup array index is within its array, since the
530/// bounds policy would otherwise substitute some element, possibly one nothing
531/// wrote. A decision that differs between kernels belongs in a field of each
532/// kernel's `DeviceProfile`, which `compilation_options` would then take in place
533/// of this constant.
534const ZERO_FILL_WORKGROUP_MEMORY: bool = true;
535
536/// The compilation options of every pipeline in the crate: `constants` for the
537/// overridable constants, and WebGPU's zero-fill of workgroup memory, which
538/// every pipeline keeps while `ZERO_FILL_WORKGROUP_MEMORY` is `true`.
539fn compilation_options<'a>(
540    constants: &'a [(&'a str, f64)],
541) -> wgpu::PipelineCompilationOptions<'a> {
542    wgpu::PipelineCompilationOptions {
543        constants,
544        zero_initialize_workgroup_memory: ZERO_FILL_WORKGROUP_MEMORY,
545    }
546}

548pub(crate) fn submit(
549    device: &wgpu::Device,
550    queue: &wgpu::Queue,
551    dispatch: &Dispatch<'_>,
552) -> Submission {
553    let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
554        label: Some(dispatch.command_label),
555    });
556    if dispatch.clear_result {
557        encoder.clear_buffer(dispatch.result, 0, Some(dispatch.result_bytes));
558    }
559    {
560        let mut pass = encoder.begin_compute_pass(&wgpu::ComputePassDescriptor {
561            label: Some(dispatch.pass_label),
562            timestamp_writes: dispatch.timing.map(crate::timing::ComputeTimer::writes),
563        });
564        pass.set_pipeline(dispatch.pipeline);
565        pass.set_bind_group(0, dispatch.group, &[]);
566        let [x, y, z] = dispatch.workgroups;
567        pass.dispatch_workgroups(x, y, z);
568    }
569    encoder.copy_buffer_to_buffer(
570        dispatch.result,
571        0,
572        dispatch.readback,
573        0,
574        dispatch.result_bytes,
575    );
576    if let Some(timing) = dispatch.timing {
577        timing.resolve(&mut encoder, dispatch.readback, dispatch.result_bytes);
578    }
579    let command = encoder.finish();
580    let at = std::time::Instant::now();
581    Submission {
582        index: queue.submit([command]),
583        at,
584    }
585}

587/// Admit a dispatch's wait policy before any device allocation or submission.
588/// Empty operations that submit no work do not need a wait allowance.
589pub(crate) fn positive_timeout(timeout: Duration) -> Result<(), GpuError> {
590    if timeout.is_zero() {
591        Err(GpuError::new(
592            GpuErrorKind::Capacity,
593            "the GPU wait timeout must be positive",

595    } else {
596        Ok(())
597    }

600// Poll control between bounded waits. Keep settled effects distinct from the
601// caller outcome: cancellation is never converted to success for reuse. Resident
602// buffer handles may remain owned until explicit release or owner destruction.
603pub(crate) fn read_polled<T>(
604    device: &wgpu::Device,
605    readback: &wgpu::Buffer,
606    submission: wgpu::SubmissionIndex,
607    timeout: Duration,
608    cancellation: impl FnMut() -> Result<(), GpuError>,
609    decode: impl FnOnce(&[u32]) -> Result<T, GpuError>,
610) -> Completion<T> {
611    let pending = request_read(readback, submission);
612    finish_pending(
613        device,
614        readback,
615        pending,
616        timeout,
617        Observation::Polled,
618        cancellation,
619        decode,
620    )
621}

623/// A readback whose mapping was requested at its submission, and whose wait
624/// follows whatever the caller runs while the submission is in flight.
625pub(crate) struct PendingRead {
626    receiver: mpsc::Receiver<Result<(), wgpu::BufferAsyncError>>,
627    submission: wgpu::SubmissionIndex,
628    // The submission's timeout runs from here, just after the submit.
629    requested: std::time::Instant,
630}
631
632/// Request `readback`'s mapping for `submission`. The mapping resolves when
633/// the submission completes; [`finish_pending`] waits for it.
634pub(crate) fn request_read(
635    readback: &wgpu::Buffer,
636    submission: wgpu::SubmissionIndex,
637) -> PendingRead {
638    let (sender, receiver) = mpsc::sync_channel(1);
639    readback
640        .slice(..)
641        .map_async(wgpu::MapMode::Read, move |mapped| {
642            let _ = sender.try_send(mapped);
643        });
644    PendingRead {
645        receiver,
646        submission,
647        requested: std::time::Instant::now(),
648    }
649}

651/// How a pending read's wait observes the caller's control.
652pub(crate) enum Observation {
653    /// Between bounded waits, as every synchronous read does.
654    Polled,
655    /// Not at all: the control fired while the caller's work ran beside the
656    /// submission. The wait then only establishes the submission's completion,
657    /// within its timeout, and the read reports this stop after it without
658    /// decoding, so a completed and released readback leaves the context
659    /// reusable.
660    AfterStop(GpuError),
661}

663/// An overlapping check's shape, factored so that controls cover it without a
664/// device: `check` submits, runs the work it is handed beside the submission
665/// at most once, and finishes. The work's result is returned beside the
666/// check's outcome whenever the work ran, whether or not the finish then
667/// fails, and is absent only when the check failed before running it.
668pub(crate) fn overlapping<T, R>(
669    during: impl FnOnce() -> R,
670    check: impl FnOnce(&mut dyn FnMut()) -> T,
671) -> (T, Option<R>) {
672    let mut during = Some(during);
673    let mut result = None;
674    let outcome = check(&mut || {
675        if let Some(during) = during.take() {
676            result = Some(during());
677        }
678    });
679    (outcome, result)
680}

682/// Wait for a pending read's submission, then map, decode and release its
683/// readback, observing the control as `observation` says.
684pub(crate) fn finish_pending<T>(
685    device: &wgpu::Device,
686    readback: &wgpu::Buffer,
687    pending: PendingRead,
688    timeout: Duration,
689    observation: Observation,
690    mut cancellation: impl FnMut() -> Result<(), GpuError>,
691    decode: impl FnOnce(&[u32]) -> Result<T, GpuError>,
692) -> Completion<T> {
693    let PendingRead {
694        receiver,
695        submission,
696        requested,
697    } = pending;
698    let submission = Some(submission);
699    let wait = wait_observing(
700        timeout,
701        observation,
702        &mut cancellation,
703        |wait| {
704            device.poll(wgpu::PollType::Wait {
705                submission_index: submission.clone(),
706                timeout: Some(wait),
707            })
708        },
709        || requested.elapsed(),
710    );
711    let observed = wait.completed.then(std::time::Instant::now);
712    let mut completion = finish_read(
713        wait,
714        || {
715            receiver
716                .try_recv()
717                .map_err(|error| {
718                    GpuError::new(
719                        GpuErrorKind::Readback,
720                        format!("mapping callback absent after completed poll: {error}"),
721                    )
722                })?
723                .map_err(|error| GpuError::new(GpuErrorKind::Readback, error.to_string()))
724        },
725        || {
726            readback
727                .slice(..)
728                .get_mapped_range()
729                .map_err(|error| GpuError::new(GpuErrorKind::Readback, error.to_string()))
730                .and_then(|mapped| {
731                    bytemuck::try_cast_slice::<u8, u32>(&mapped)
732                        .map_err(|error| GpuError::new(GpuErrorKind::Readback, error.to_string()))
733                        .and_then(decode)
734                })
735        },
736        // Also terminate a pending map after an interrupted wait. Unmapping
737        // alone does not establish queue completion or permit context reuse.
738        || readback.unmap(),
739        &mut cancellation,
740    );
741    completion.observed = observed;
742    completion
743}
744
745struct SubmissionWait {
746    outcome: Result<(), GpuError>,
747    completed: bool,
748}
749
750// The actual production cleanup sequence, factored so deterministic controls
751// exercise mapping/decoding/control/release ordering without a physical device.
752fn finish_read<T>(
753    wait: SubmissionWait,
754    mapping: impl FnOnce() -> Result<(), GpuError>,
755    decode: impl FnOnce() -> Result<T, GpuError>,
756    release: impl FnOnce(),
757    mut cancellation: impl FnMut() -> Result<(), GpuError>,
758) -> Completion<T> {
759    // A completed wait can have returned a late control stop. Inspect its map
760    // callback for health evidence, but preserve that original stop and skip
761    // decoding. An absent/failed callback prevents reuse even in that case.
762    let mapped = if wait.completed { mapping() } else { Ok(()) };
763    let mapped_successfully = wait.completed && mapped.is_ok();
764    let outcome = wait
765        .outcome
766        .and(mapped)
767        .and_then(|()| decode())
768        .and_then(|value| {
769            cancellation()?;
770            Ok(value)
771        });
772    release(); // Every mapped view has left the decode closure before this call.
773    Completion {
774        outcome,
775        effects: if mapped_successfully {
776            Effects::SubmittedAndReleased
777        } else {
778            Effects::MayBeLive
779        },
780        observed: None,
781    }
782}
783
784// The wait's observation of the control, factored so deterministic controls
785// cover it without a device. After a stop the caller's work observed, the wait
786// polls no control: it reports that stop once completion is observed, which
787// skips decoding, and otherwise the wait's own failure, which says why
788// completion was not observed.
789fn wait_observing(
790    timeout: Duration,
791    observation: Observation,
792    cancellation: impl FnMut() -> Result<(), GpuError>,
793    poll: impl FnMut(Duration) -> Result<wgpu::PollStatus, wgpu::PollError>,
794    elapsed: impl FnMut() -> Duration,
795) -> SubmissionWait {
796    match observation {
797        Observation::Polled => wait_for_submission(timeout, cancellation, poll, elapsed),
798        Observation::AfterStop(stop) => {
799            let wait = wait_for_submission(timeout, || Ok(()), poll, elapsed);
800            SubmissionWait {
801                outcome: if wait.completed {
802                    Err(stop)
803                } else {
804                    wait.outcome
805                },
806                completed: wait.completed,
807            }
808        }
809    }
810}
811
812// The production wait loop; injected poll/elapsed operations let controls cover
813// timing boundaries deterministically without invoking a driver or sleeping.
814fn wait_for_submission(
815    timeout: Duration,
816    mut cancellation: impl FnMut() -> Result<(), GpuError>,
817    mut poll: impl FnMut(Duration) -> Result<wgpu::PollStatus, wgpu::PollError>,
818    mut elapsed: impl FnMut() -> Duration,
819) -> SubmissionWait {
820    loop {
821        if let Err(error) = cancellation() {
822            return SubmissionWait {
823                outcome: Err(error),
824                completed: false,
825            };
826        }
827        let remaining = timeout.saturating_sub(elapsed());
828        match poll(remaining.min(Duration::from_millis(50))) {
829            Ok(status) if status.wait_finished() => {
830                return SubmissionWait {
831                    outcome: cancellation(),
832                    completed: true,
833                };
834            }
835            Ok(_) => {
836                return SubmissionWait {
837                    outcome: Err(GpuError::new(
838                        GpuErrorKind::Device,
839                        "device wait did not establish submission completion",
840                    )),
841                    completed: false,
842                };
843            }
844            Err(wgpu::PollError::Timeout) if elapsed() < timeout => {}
845            Err(error) => {
846                return SubmissionWait {
847                    outcome: Err(GpuError::new(
848                        if matches!(error, wgpu::PollError::Timeout) {
849                            GpuErrorKind::Timeout
850                        } else {
851                            GpuErrorKind::Device
852                        },
853                        error.to_string(),
854                    )),
855                    completed: false,
856                };
857            }
858        }
859    }
860}

894/// An operation's error scopes: still open, as a synchronous operation keeps
895/// them until its readback settles, or already popped with their outcome, as an
896/// overlapping check pops them right after its submission.
897pub(crate) enum Scopes {
898    Open(ErrorScopes),
899    Settled(Result<(), GpuError>),
900}
901
902impl Scopes {
903    /// Pop open scopes now, keeping their outcome for the settlement.
904    pub(crate) fn settle(&mut self) {
905        if let Self::Open(_) = self
906            && let Self::Open(scopes) = std::mem::replace(self, Self::Settled(Ok(())))
907        {
908            *self = Self::Settled(pollster::block_on(scopes.finish()));
909        }
910    }
911}
912
913pub(crate) struct ErrorScopes {
914    // Field drop order is the reverse of the push order, including unwinding.
915    allocation: wgpu::ErrorScopeGuard,
916    internal: wgpu::ErrorScopeGuard,
917    validation: wgpu::ErrorScopeGuard,
918}
919
920impl ErrorScopes {
921    pub(crate) fn new(device: &wgpu::Device) -> Self {
922        Self {
923            validation: device.push_error_scope(wgpu::ErrorFilter::Validation),
924            internal: device.push_error_scope(wgpu::ErrorFilter::Internal),
925            allocation: device.push_error_scope(wgpu::ErrorFilter::OutOfMemory),
926        }
927    }
928
929    pub(crate) async fn finish(self) -> Result<(), GpuError> {
930        let mut failure = None;
931        for scope in [self.allocation, self.internal, self.validation] {
932            if let Some(error) = scope.pop().await {
933                let kind = match error {
934                    wgpu::Error::OutOfMemory { .. } => GpuErrorKind::Allocation,
935                    wgpu::Error::Internal { .. } => GpuErrorKind::Device,
936                    wgpu::Error::Validation { .. } => GpuErrorKind::Validation,
937                };
938                failure.get_or_insert_with(|| GpuError::new(kind, error.to_string()));
939            }
940        }
941        failure.map_or(Ok(()), Err)
942    }
943}
