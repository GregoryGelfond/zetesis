1//! Decide identity, complete-scan bounds and cache replacement before effects.
2
3use super::packing::{Graph, PARAM_BYTES, Packed, Packing, Plan};
4use super::{TightGpuBatchStats, TightGpuLimits, poll};
5use crate::{GpuError, GpuErrorKind};
6use zetesis_algebra::Cancellation;
7use zetesis_ferraris::{Interpretation, TightPlan};
8
9pub(super) struct Admission {
10    pub(super) fresh: Option<Graph>,
11    pub(super) plan: Plan,
12    pub(super) stats: TightGpuBatchStats,
13}
14
15impl Admission {
16    // No device handles are mutated here. Refusal preserves existing residency;
17    // an accepted replacement must evict its old buffers before host packing.
18    pub(super) fn new(
19        certificate: &TightPlan,
20        candidates: &[Interpretation],
21        limits: TightGpuLimits,
22        packing: Packing<'_>,
23        cached: Option<(&Graph, bool)>,
24        prior_epoch: u32,
25        cancellation: &Cancellation,
26    ) -> Result<Self, GpuError> {
27        poll(cancellation)?;
28        let epoch = prior_epoch
29            .checked_add(1)
30            .ok_or_else(|| GpuError::new(GpuErrorKind::Capacity, "tight epoch exhausted"))?;
31        let fresh =
32            if cached.is_some_and(|(graph, _)| graph.theory.same_instance(certificate.theory())) {
33                None
34            } else {
35                Some(Graph::with_control(
36                    certificate,
37                    packing,
38                    limits.max_batch_bytes,
39                    cancellation,
40                )?)
41            };
42        let graph = fresh
43            .as_ref()
44            .or_else(|| cached.map(|(graph, _)| graph))
45            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing tight graph"))?;
46        let plan = Plan::new(
47            graph,
48            candidates.len(),
49            limits,
50            packing.device,
51            fresh.is_some(),
52            epoch,
53        )?;
54        for candidate in candidates {
55            poll(cancellation)?;
56            if !certificate.theory().same_instance(candidate.theory()) {
57                return Err(GpuError::new(
58                    GpuErrorKind::Seed,
59                    "candidate belongs to another Theory",
60                ));
61            }
62        }
63        let stats = batch_stats(
64            graph,
65            &plan,
66            fresh.is_some(),
67            cached.is_none_or(|(_, matches)| !matches),
68        );
69        Ok(Self { fresh, plan, stats })
70    }
71    pub(super) fn pack(
72        &mut self,
73        certificate: &TightPlan,
74        limits: TightGpuLimits,
75        device: &wgpu::Limits,
76        cancellation: &Cancellation,
77    ) -> Result<Option<Packed>, GpuError> {
78        let Some(graph) = self.fresh.as_mut() else {
79            return Ok(None);
80        };
81        let packed = graph.prepare(certificate, limits, device, &mut self.plan, cancellation)?;
82        self.stats = batch_stats(graph, &self.plan, true, true);
83        Ok(Some(packed))
84    }
85}
86
87pub(super) fn batch_stats(
88    graph: &Graph,
89    plan: &Plan,
90    fresh: bool,
91    transport_allocated: bool,
92) -> TightGpuBatchStats {
93    TightGpuBatchStats {
94        theory_uploaded: fresh,
95        transport_allocated: fresh || transport_allocated,
96        resident_theory_bytes: graph.bytes,
97        resident_transport_bytes: plan.transport,
98        accounted_bytes: plan.accounted,
99        candidates: u64::from(plan.worlds),
100        work: u64::from(plan.worlds) * u64::from(plan.work),
101        dispatches: 1,
102        truth_levels: plan.levels,
103        workgroup_size: graph.workgroup_size.invocations(),
104        original_truth_reused: plan.original_truth_reused,
105        routes: super::TightGpuRoutes::default(),
106        nodes_evaluated: 0,
107        update_critical_path: 0,
108        dense_critical_path: graph.dense_path,
109        updates: super::TightGpuUpdates::Pending,
110        uploaded_bytes: PARAM_BYTES + plan.seeds + if fresh { graph.bytes } else { 0 },
111        downloaded_bytes: plan.results,
112    }
113}
