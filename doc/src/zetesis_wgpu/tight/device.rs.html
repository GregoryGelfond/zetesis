16/// Real-device original satisfaction and ranked producer support checking.
17///
18/// Takes a complete opaque [`TightPlan`]; never constructs an unchecked rank or
19/// relies on an incomplete lazy registry. Candidate generation, objectives and
20/// any witness-producing residual completion remain the caller's responsibilities.
21/// A complete failed support scan also permits proof-based rejection by the
22/// support-necessity theorem; it publishes no concrete deletion witness.
23/// Every nonempty successful call executes the support kernel on the selected
24/// adapter. Ordinary solver dispatch selects this independent primitive only
25/// after preparing its complete certificate; no CPU fallback occurs here.
26///
27/// Work is linear in candidates times nodes, roots, producers and atoms. Each
28/// workgroup evaluates its DAG sequentially, then shares root/support scans
29/// across 64 invocations. Truth uses one word per node; candidate membership and
30/// producer support use one bit per atom, in separate world-major arrays. Heads
31/// sharing a support word use atomic OR by default. Grouped support instead
32/// visits each word's producers sequentially in one invocation; this trades
33/// contention for an index and potentially uneven parallel work.
34/// Shared graph and transport storage have explicit byte bounds. The oracle
35/// retains one shared theory handle and one exact transport
36/// shape, without cloning the caller's certificate or storing its ranks.
37pub struct GpuTightOracle {
38    runtime: Runtime,
39    support: TightSupport,
40    resident: Option<Resident>,
41    epoch: u32,
42    last: Option<TightGpuBatchStats>,
43    activity: TightGpuActivity,
44}

63    /// Create a device with an explicit producer-support construction policy.
64    /// The policy remains fixed for the oracle's lifetime. Grouped construction
65    /// retains one offset per support word plus an end offset; its temporary
66    /// packing cursors are included in fresh-batch byte admission.
67    /// Each complete device record identifies the construction branch. The
68    /// decoder refuses a marker for the other policy, even if its verdict agrees.
69    /// This is protocol validation, not a proof of shader or hardware correctness.
70    ///
71    /// # Errors
72    /// Returns the same typed construction failures as [`Self::new_selected`].
73    pub fn new_with_support(
74        options: GpuOptions,
75        selection: GpuSelection,
76        support: TightSupport,
77    ) -> Result<Self, GpuError> {
78        let runtime = pollster::block_on(Runtime::new(options, selection, Self::profile(support)))?;
79        Ok(Self::with_runtime(runtime, support))
80    }

166    /// Check complete original truth and support for every ordered occurrence.
167    /// Repeated interpretations remain repeated results. The first false root
168    /// follows original root-list order; unsupported atoms use ascending order.
169    ///
170    /// Checks control and health even for an empty batch, then returns no results
171    /// without charging storage/work or dispatching. On nonempty calls, clones
172    /// of one Theory reuse immutable storage; independent equal theories do not.
173    /// Every admitted `TightPlan` canonically extracts producers from all original
174    /// roots, independently of the particular valid ranks. Thus theory identity
175    /// is sufficient for this producer cache; arbitrary supplied producers are
176    /// never accepted. Transport is replaced when candidate count changes.
177    ///
178    /// # Errors
179    /// Returns no partial result vector on foreign identity, capacity, work,
180    /// allocation, cancellation, deadline, validation, device or readback failure.
181    /// Host preparation refusals leave the oracle reusable. Failures after device
182    /// allocation/submission invalidate the entire context; a fresh context is
183    /// required before retrying. Busy contention submits no work and permits retry.
184    /// Cancellation is polled during packing, waiting, decoding and before returning.
185    pub fn check_batch(
186        &mut self,
187        certificate: &TightPlan,
188        candidates: &[Interpretation],
189        limits: TightGpuLimits,
190        cancellation: &Cancellation,
191    ) -> Result<Vec<TightGpuCheck>, TightGpuError> {
192        self.last = None;
193        self.activity = TightGpuActivity::default();
194        self.check(certificate, candidates, limits, cancellation)
195            .map_err(Into::into)
196    }
197
198    fn check(
199        &mut self,
200        certificate: &TightPlan,
201        candidates: &[Interpretation],
202        limits: TightGpuLimits,
203        cancellation: &Cancellation,
204    ) -> Result<Vec<TightGpuCheck>, GpuError> {
205        let context = self.runtime.context.clone();
206        let _lease = context.lease()?;
207        poll(cancellation)?;
208        self.runtime.check_health()?;
209        if candidates.is_empty() {
210            return Ok(Vec::new());
211        }
212        let Admission { fresh, plan, stats } = Admission::new(
213            certificate,
214            candidates,
215            limits,
216            Packing {
217                device: self.runtime.limits(),
218                support: self.support,
219            },
220            self.resident
221                .as_ref()
222                .map(|resident| (&resident.graph, resident.matches_count(candidates.len()))),
223            self.epoch,
224            cancellation,
225        )?;
226        // Evict authored buffers before replacing them; deferred driver retirement
227        // is explicitly outside the logical payload budget.
228        if fresh.is_some() {
229            self.resident = None;
230        } else if stats.transport_allocated
231            && let Some(resident) = self.resident.as_mut()
232        {
233            resident.transport = None;
234        }
235        let graph = fresh
236            .as_ref()
237            .or_else(|| self.resident.as_ref().map(|r| &r.graph))
238            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing tight packing graph"))?;
239        let seeds = plan.pack(graph, candidates, cancellation)?;
240        let packed = fresh
241            .as_ref()
242            .map(|g| g.pack(certificate, cancellation))
243            .transpose()?;
244        poll(cancellation)?;
245        let scopes = ErrorScopes::new(self.runtime.device());
246        if let Some((graph, packed)) = fresh.zip(packed) {
247            self.activity.uploaded_bytes = graph.bytes;
248            self.resident = Some(Resident::new(self.runtime.device(), graph, &packed));
249        }
250        self.epoch = plan.epoch;
251        let outcome = self
252            .resident
253            .as_mut()
254            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing resident tight graph"))
255            .and_then(|resident| {
256                resident.dispatch(
257                    &Execution {
258                        device: self.runtime.device(),
259                        queue: self.runtime.queue(),
260                        pipeline: &self.runtime.pipeline,
261                        timeout: limits.timeout,
262                        cancellation,
263                    },
264                    &seeds,
265                    &plan,
266                    &mut self.activity,
267                )
268            });
269        let result = self.runtime.complete(scopes, outcome).and_then(|checks| {
270            poll(cancellation)?;
271            Ok(checks)
272        });
273        if result.is_ok() {
274            self.last = Some(stats);
275        }
276        result
277    }
278}

280fn check_limits(limits: &wgpu::Limits) -> Result<(), GpuError> {
281    for (label, required, available) in [
282        (
283            "workgroup invocations",
284            64,
285            limits.max_compute_invocations_per_workgroup,
286        ),
287        ("workgroup width", 64, limits.max_compute_workgroup_size_x),
288        (
289            "workgroup storage",
290            8,
291            limits.max_compute_workgroup_storage_size,
292        ),
293        (
294            "storage bindings",
295            7,
296            limits.max_storage_buffers_per_shader_stage,
297        ),
298        (
299            "uniform bindings",
300            1,
301            limits.max_uniform_buffers_per_shader_stage,
302        ),
303        ("bindings per group", 8, limits.max_bindings_per_bind_group),
304    ] {
305        if available < required {
306            return Err(GpuError::new(
307                GpuErrorKind::Capacity,
308                format!("tight {label}: need {required}, device provides {available}"),
309            ));
310        }
311    }
312    Ok(())
313}
