23/// Physical construction of the same complete producer-support relation.
24/// This selects a library primitive; it does not change answer-set semantics or
25/// ordinary solver dispatch policy.
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
27pub enum TightSupport {
28    /// Producers independently set enabled head bits through atomic OR.
29    #[default]
30    Atomic,
31    /// One invocation reduces every producer belonging to a support word.
32    /// This avoids contended ORs but can serialize large or uneven groups.
33    Grouped,
34}

36/// Bounds for one complete support-checking batch, independent of search.
37#[derive(Clone, Copy, Debug)]
38pub struct TightGpuLimits {
39    /// Candidate occurrences, also bounded by granted device dispatch limits.
40    pub max_candidates: usize,
41    /// Conservative authored graph, transport, packing and result payload bytes.
42    /// Candidate upload scratch is charged by actual element capacity and retained
43    /// only while the subject and candidate count stay unchanged.
44    /// Shared caller-owned theory/certificate/candidates, allocator overhead,
45    /// driver-private allocations and deferred retirement are excluded. Not RSS.
46    pub max_batch_bytes: u64,
47    /// Each candidate reserves `nodes + edges + roots + producers + 2 * words`
48    /// operations, plus `48*n+32` for each variable threshold and `24*n+16`
49    /// for each variable extremum cut with `n` condition occurrences. These
50    /// conservative units cover portable limb arithmetic; constant guards carry
51    /// no numeric row charge. A cooperative truth schedule adds one unit per
52    /// dependency level. Serial execution remains available when that schedule
53    /// cannot fit the work or cold-storage allowance. Here `edges` counts child occurrences and
54    /// `words = ceil(atoms / 32)`. One word scan initializes
55    /// or constructs packed support under the selected [`TightSupport`] policy;
56    /// the other finds unsupported candidate bits by words. This replaces the
57    /// former accounting contract; native operand reads are included.
58    /// Every scan completes, including initialization and scans after an original
59    /// failure; this differs from scalar early-exit work. Insufficient work is
60    /// refused before dispatch, never converted into a logical result.
61    ///
62    /// Reusing a wholly successful batch's original truth adds
63    /// `2 * nodes + edges + 4` for change detection; it runs only when that
64    /// fits. The update route (see [`Self::min_update_candidates`]) is
65    /// allowed only when its derived worst case and dense reuse after its
66    /// change count also fit; while it is allowed, the complete and dense
67    /// routes add the retained candidate's writes, `words + 1`, and dense reuse
68    /// after the change count adds the count's `2 * words`. Each check then
69    /// reports the work its route performed, never more than was reserved.
70    pub max_work_per_candidate: u64,
71    /// The fewest candidates whose per-slot update state a batch must admit
72    /// before the update route is used.
73    ///
74    /// Once a theory's batches of one count hold reuse authority, the oracle
75    /// can retain, per candidate slot, the candidate it last evaluated and that
76    /// evaluation's support summaries, and then re-evaluate only the nodes a
77    /// new candidate changes. That state is counted in each batch's
78    /// [`Self::max_batch_bytes`]. A prefix call admits it only when the
79    /// state-inclusive layout still admits at least this many candidates; an
80    /// exact call admits it only when that layout admits its whole count. The
81    /// state never turns an admitted batch into a refusal: without it the dense
82    /// routes run as before. Zero declines the update route.
83    ///
84    /// Independently of this limit, the route engages only on a schedule whose
85    /// dense evaluation takes at least
86    /// [`GpuTightOracle::UPDATE_ENGAGEMENT_STRIDES`](crate::GpuTightOracle::UPDATE_ENGAGEMENT_STRIDES)
87    /// workgroup strides per dependency level. Narrower schedules report
88    /// [`TightGpuUpdateUnavailable::NarrowSchedule`]. Once engaged, the oracle's
89    /// runtime guard can decline the route for a few batches at a time after a
90    /// batch whose slowest update's critical path exceeded dense evaluation's
91    /// ([`TightGpuUpdateUnavailable::GuardDeclined`]).
92    pub min_update_candidates: usize,
93    /// Bounded submission wait, with control polling at most 50 ms apart while
94    /// waiting. Not GPU preemption or a bound on device creation, allocation,
95    /// shader compilation or error-scope drainage.
96    /// Must be positive for a nonempty batch; empty batches perform no wait.
97    pub timeout: Duration,
98}
99
100impl Default for TightGpuLimits {
101    fn default() -> Self {
102        Self {
103            max_candidates: 1024,
104            max_batch_bytes: 128 * 1024 * 1024,
105            max_work_per_candidate: 100_000_000,
106            min_update_candidates: 1,
107            timeout: Duration::from_secs(30),
108        }
109    }
110}

112/// A complete device support check for one input occurrence.
113#[derive(Clone, Copy, Debug, PartialEq, Eq)]
114pub struct TightGpuCheck {
115    verdict: TightVerdict,
116    work: u64,
117    critical_path: u64,
118}
119
120impl TightGpuCheck {
121    /// Exact scalar certificate verdict, including its ordered witness.
122    /// A residual names an unsupported present atom. A caller requiring a
123    /// concrete reduct witness must complete it; a proof-based protocol may
124    /// reject it using the complete certificate's support-necessity theorem.
125    #[must_use]
126    pub const fn verdict(&self) -> TightVerdict {
127        self.verdict
128    }
129
130    /// Operations the check's route performed, under the units of
131    /// [`TightGpuLimits::max_work_per_candidate`]: the complete and dense
132    /// routes charge their fixed allowance, the update route what it ran.
133    #[must_use]
134    pub const fn work(&self) -> u64 {
135        self.work
136    }
137
138    /// The update route's critical path for this check, by the runtime
139    /// guard's cost table, which counts what [`Self::work`] counts save that a
140    /// wide conjunction or disjunction costs the operands it read: the costliest
141    /// invocation of the candidate's workgroup in each phase after the
142    /// summary, and in each phase of the summary when the slot took dense reuse
143    /// beyond the change threshold in its previous batch, summed, with each
144    /// invocation's share of the update's uniform scans and of the record. A
145    /// summary after a batch that seeded every slot is left out. Zero for every
146    /// other route. Deterministic for the theory, the schedule and the
147    /// candidates.
148    #[must_use]
149    pub const fn critical_path(&self) -> u64 {
150        self.critical_path
151    }
152}

154/// Resource evidence for the last successful nonempty batch.
155#[derive(Clone, Copy, Debug, PartialEq, Eq)]
156#[non_exhaustive]
157pub struct TightGpuBatchStats {
158    /// A different immutable theory required graph upload.
159    pub theory_uploaded: bool,
160    /// Exact candidate-count transport was allocated or replaced.
161    pub transport_allocated: bool,
162    /// Requested immutable header/operand, root and producer buffer bytes, with
163    /// padding; the producer buffer includes a built readers index.
164    pub resident_theory_bytes: u64,
165    /// Requested uniform, candidate, truth, support, result and readback
166    /// bytes, and the per-slot update state once admitted: bound when the
167    /// update route is allowed, and otherwise allocated but unbound, on the
168    /// kernel without the update route.
169    pub resident_transport_bytes: u64,
170    /// Conservative authored payload checked against the batch byte ceiling.
171    pub accounted_bytes: u64,
172    /// Fully checked input occurrences, preserving repetitions.
173    pub candidates: u64,
174    /// Sum of the operations each candidate's route performed.
175    pub work: u64,
176    /// Actual compute submissions; one for this nonempty successful batch.
177    pub dispatches: u64,
178    /// Dependency levels evaluated cooperatively; zero selects serial original truth.
179    pub truth_levels: u32,
180    /// The workgroup size dispatched, in invocations per candidate: the fixed
181    /// size, or the size the theory's dependency schedule fits.
182    pub workgroup_size: u32,
183    /// A wholly successful same-subject/count batch supplied original truth.
184    /// This selects change detection; it does not assert that any gate was skipped.
185    pub original_truth_reused: bool,
186    /// Candidate slots per route.
187    pub routes: TightGpuRoutes,
188    /// Nodes re-evaluated by update-route slots, summed over the batch.
189    pub nodes_evaluated: u64,
190    /// The largest critical path among the batch's update slots (see
191    /// [`TightGpuCheck::critical_path`]), or zero without update slots: the
192    /// runtime guard's measure of the update route.
193    pub update_critical_path: u64,
194    /// Dense reuse's least critical path for one candidate at the dispatched
195    /// workgroup size, by the same cost table as an update's: each dependency
196    /// level's and each scan's busiest invocation, summed, where every
197    /// candidate-dependent cost takes its least. The runtime guard declines
198    /// the route after a batch whose `update_critical_path` exceeds it. Zero
199    /// for a schedule the update route does not engage.
200    pub dense_critical_path: u64,
201    /// Whether the update route was available to this batch, and why not.
202    pub updates: TightGpuUpdates,
203    /// Authored initialized/write-buffer payload, not measured bus traffic,
204    /// including the rebuilt producer buffer of a batch that builds the index.
205    pub uploaded_bytes: u64,
206    /// Successfully decoded readback payload, not measured bus traffic.
207    pub downloaded_bytes: u64,
208}

210/// Candidate slots of one batch by the route that evaluated them.
211///
212/// Each slot takes exactly one route. The complete route evaluates everything;
213/// dense reuse re-evaluates every node against a retained evaluation; the
214/// update route re-evaluates only the nodes a candidate's change reaches.
215#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
216#[non_exhaustive]
217pub struct TightGpuRoutes {
218    /// Slots without reuse authority.
219    pub complete: u64,
220    /// Slots that reused a retained evaluation and re-evaluated every node.
221    pub dense_reuse: u64,
222    /// Dense-reuse slots that first counted their change and found it beyond
223    /// the update route's threshold; included in `dense_reuse`.
224    pub beyond_threshold: u64,
225    /// Slots that brought a retained evaluation to their new candidate.
226    pub update: u64,
227}

229/// The update route's availability to one batch.
230#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
231#[non_exhaustive]
232pub enum TightGpuUpdates {
233    /// The batch held no reuse authority at its stateless count, which the
234    /// route needs before it builds the theory's readers index. A fresh
235    /// theory's first batch reports this unless its schedule or limits
236    /// already decline the route. The batch ran the kernel without the update
237    /// route.
238    #[default]
239    Pending,
240    /// Per-slot state was bound and the route was allowed: the batch ran the
241    /// kernel with the update route.
242    Admitted {
243        /// Bits of each per-head support count: 8, 16 or 32.
244        count_bits: u32,
245        /// This batch uploaded the theory's readers index.
246        index_built: bool,
247    },
248    /// The route did not run in this batch, for the given reason; the batch
249    /// ran the kernel without the update route.
250    Unavailable(TightGpuUpdateUnavailable),
251}
252
253/// Why the update route was unavailable to a batch.
254#[derive(Clone, Copy, Debug, PartialEq, Eq)]
255#[non_exhaustive]
256pub enum TightGpuUpdateUnavailable {
257    /// The batch evaluates original truth serially, which has no levels.
258    SerialSchedule,
259    /// Dense evaluation of the resident schedule takes fewer than
260    /// [`GpuTightOracle::UPDATE_ENGAGEMENT_STRIDES`](crate::GpuTightOracle::UPDATE_ENGAGEMENT_STRIDES)
261    /// workgroup strides per dependency level on average, below which the
262    /// route is not expected to pay. Decided once per resident theory and
263    /// workgroup size.
264    NarrowSchedule,
265    /// A prefix call admits fewer candidates than
266    /// [`TightGpuLimits::min_update_candidates`] before any state is
267    /// considered: its input or its `max_candidates` is shorter.
268    CandidateCeiling,
269    /// The theory exceeds the readers index's intrinsic bounds; this holds
270    /// for as long as the theory stays resident.
271    IndexBounds,
272    /// The state-inclusive layout did not admit enough candidates within the
273    /// batch's byte allowance.
274    Bytes,
275    /// The route's worst case did not fit the work allowance.
276    Work,
277    /// [`TightGpuLimits::min_update_candidates`] was zero.
278    Declined,
279    /// The oracle's runtime guard declined the route for this batch, which ran
280    /// the complete or dense routes on the kernel without the update route,
281    /// its state allocated and accounted. The guard decides which route
282    /// evaluates a batch, never a verdict.
283    GuardDeclined,
284}

310/// An incomplete invocation or device failure; never a membership verdict.
311#[derive(Clone, Debug, PartialEq, Eq)]
312pub enum TightGpuError {
313    /// Shared cancellation or deadline was observed.
314    Stopped(Stop),
315    /// Adapter, identity, capacity, allocation, execution or readback failure.
316    Gpu(GpuError),
317}
318
319impl fmt::Display for TightGpuError {
320    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
321        match self {
322            Self::Stopped(stop) => stop.fmt(f),
323            Self::Gpu(error) => error.fmt(f),
324        }
325    }
326}
327impl std::error::Error for TightGpuError {
328    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
329        Some(match self {
330            Self::Stopped(stop) => stop,
331            Self::Gpu(error) => error,
332        })
333    }
334}
335impl From<GpuError> for TightGpuError {
336    fn from(error: GpuError) -> Self {
337        match error.interruption {
338            Some(stop) => Self::Stopped(stop),
339            None => Self::Gpu(error),
340        }
341    }
342}
343
344fn poll(cancellation: &Cancellation) -> Result<(), GpuError> {
345    cancellation.poll().map_err(|stop| GpuError {
346        kind: crate::GpuErrorKind::Device,
347        detail: "tight-support operation interrupted".to_owned(),
348        interruption: Some(stop),
349    })
350}
