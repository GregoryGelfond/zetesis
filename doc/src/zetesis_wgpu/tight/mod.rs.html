16/// Physical construction of the same complete producer-support relation.
17/// This selects a library primitive; it does not change answer-set semantics or
18/// ordinary solver dispatch policy.
19#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
20pub enum TightSupport {
21    /// Producers independently set enabled head bits through atomic OR.
22    #[default]
23    Atomic,
24    /// One invocation reduces every producer belonging to a support word.
25    /// This avoids contended ORs but can serialize large or uneven groups.
26    Grouped,
27}
28
29/// Bounds for one complete support-checking batch, independent of search.
30#[derive(Clone, Copy, Debug)]
31pub struct TightGpuLimits {
32    /// Candidate occurrences, also bounded by granted device dispatch limits.
33    pub max_candidates: usize,
34    /// Conservative authored graph, transport, packing and result payload bytes.
35    /// Shared caller-owned theory/certificate/candidates, allocator overhead,
36    /// driver-private allocations and deferred retirement are excluded. Not RSS.
37    pub max_batch_bytes: u64,
38    /// Each candidate reserves `nodes + roots + producers + 2 * words`
39    /// operations, where `words = ceil(atoms / 32)`. One word scan initializes
40    /// or constructs packed support under the selected [`TightSupport`] policy;
41    /// the other finds unsupported candidate bits by words. This replaces the
42    /// former `nodes + roots + producers + atoms + words` accounting contract.
43    /// Every scan completes, including initialization and scans after an original
44    /// failure; this differs from scalar early-exit work. Insufficient work is
45    /// refused before dispatch, never converted into a logical result.
46    pub max_work_per_candidate: u64,
47    /// Bounded submission wait, with control polling at most 50 ms apart while
48    /// waiting. Not GPU preemption or a bound on device creation, allocation,
49    /// shader compilation or error-scope drainage.
50    /// Must be positive for a nonempty batch; empty batches perform no wait.
51    pub timeout: Duration,
52}
53
54impl Default for TightGpuLimits {
55    fn default() -> Self {
56        Self {
57            max_candidates: 1024,
58            max_batch_bytes: 128 * 1024 * 1024,
59            max_work_per_candidate: 100_000_000,
60            timeout: Duration::from_secs(30),
61        }
62    }
63}

65/// A complete device support check for one input occurrence.
66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub struct TightGpuCheck {
68    verdict: TightVerdict,
69    work: u64,
70}
71
72impl TightGpuCheck {
73    /// Exact scalar certificate verdict, including its ordered witness.
74    /// A residual names an unsupported present atom. A caller requiring a
75    /// concrete reduct witness must complete it; a proof-based protocol may
76    /// reject it using the complete certificate's support-necessity theorem.
77    #[must_use]
78    pub const fn verdict(&self) -> TightVerdict {
79        self.verdict
80    }
81
82    /// Completed full-scan operations under [`TightGpuLimits`].
83    #[must_use]
84    pub const fn work(&self) -> u64 {
85        self.work
86    }
87}

137/// An incomplete invocation or device failure; never a membership verdict.
138#[derive(Clone, Debug, PartialEq, Eq)]
139pub enum TightGpuError {
140    /// Shared cancellation or deadline was observed.
141    Stopped(Stop),
142    /// Adapter, identity, capacity, allocation, execution or readback failure.
143    Gpu(GpuError),
144}
145
146impl fmt::Display for TightGpuError {
147    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
148        match self {
149            Self::Stopped(stop) => stop.fmt(f),
150            Self::Gpu(error) => error.fmt(f),
151        }
152    }
153}
154impl std::error::Error for TightGpuError {
155    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
156        Some(match self {
157            Self::Stopped(stop) => stop,
158            Self::Gpu(error) => error,
159        })
160    }
161}
162impl From<GpuError> for TightGpuError {
163    fn from(error: GpuError) -> Self {
164        match error.interruption {
165            Some(stop) => Self::Stopped(stop),
166            None => Self::Gpu(error),
167        }
168    }
169}

171fn poll(cancellation: &Cancellation) -> Result<(), GpuError> {
172    cancellation.poll().map_err(|stop| GpuError {
173        kind: crate::GpuErrorKind::Device,
174        detail: "tight-support operation interrupted".to_owned(),
175        interruption: Some(stop),
176    })
177}
