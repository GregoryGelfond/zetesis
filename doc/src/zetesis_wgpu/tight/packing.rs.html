25fn capacity(detail: &str) -> GpuError {
26    GpuError::new(GpuErrorKind::Capacity, detail)
27}
28fn address(value: usize) -> Result<u32, GpuError> {
29    u32::try_from(value).map_err(|_| capacity("tight dimension exceeds u32"))
30}
31fn sum(values: &[u64]) -> Result<u64, GpuError> {
32    values
33        .iter()
34        .try_fold(0, |total, next| {
35            CheckedSum.combine(Some(total), Some(*next))
36        })
37        .ok_or_else(|| capacity("tight byte sum overflow"))
38}
39// This is a proposed device-work allowance, not a performed-work receipt.
40// Optional routes must fit both the authored ceiling and the unchanged wire word.
41fn extend_work(recorded: u32, extra: u32, ceiling: u64) -> Option<u32> {
42    Meter::check_receipt(
43        u64::from(recorded),
44        u128::from(extra),
45        ceiling.min(u64::from(u32::MAX)),
46    )
47    .ok()
48    .map(|work| u32::try_from(work).expect("optional work checked against u32 ceiling"))
49}

51fn buffer(bytes: u64, device: &wgpu::Limits) -> Result<(), GpuError> {
52    let ceiling = device
53        .max_buffer_size
54        .min(device.max_storage_buffer_binding_size);
55    MeterBytes::check(u128::from(bytes), u128::from(ceiling))
56        .map_err(|_| capacity("tight storage exceeds granted device limits"))
57}
58fn vector<T>(length: usize) -> Result<Vec<T>, GpuError> {
59    let mut output = Vec::new();
60    output
61        .try_reserve_exact(length)
62        .map_err(|_| GpuError::new(GpuErrorKind::Allocation, "tight host reservation failed"))?;
63    Ok(output)
64}
65fn words(bytes: u64) -> Result<Vec<u32>, GpuError> {
66    vector(usize::try_from(bytes / 4).map_err(|_| capacity("tight buffer exceeds host"))?)
67}

69fn producer_storage(producers: u32, words: u32, support: TightSupport) -> Result<u64, GpuError> {
70    let producer_words = u64::from(producers) * 4
71        + if support == TightSupport::Grouped {
72            u64::from(words) + 1
73        } else {
74            0
75        };
76    // The shader addresses records and appended offsets with u32 indices,
77    // independently of a device's hypothetical byte capacity.
78    if producer_words > u64::from(u32::MAX) {
79        return Err(capacity("tight producer indexing exceeds u32"));
80    }
81    Ok(producer_words.max(4) * 4)
82}

155    pub(super) fn with_control(
156        certificate: &TightPlan,
157        packing: Packing<'_>,
158        max_staging_bytes: u64,
159        cancellation: &Cancellation,
160    ) -> Result<Self, GpuError> {
161        let theory = certificate.theory();
162        let atoms = address(theory.atom_count())?;
163        let graph = crate::formula_graph::Shape::with_control(
164            theory,
165            packing.device,
166            max_staging_bytes,
167            cancellation,
168        )?;
169        let nodes = graph.nodes;
170        let roots = address(theory.roots().len())?;
171        let producers = address(certificate.producers().len())?;
172        // Reserve the maximum kernel stride after each loop's last valid address.
173        if [atoms, nodes, roots, producers]
174            .into_iter()
175            .any(|n| !crate::kernel::fits_strided_loop(n))
176        {
177            return Err(capacity("tight strided loop increment exceeds u32"));
178        }
179        let words = atoms.div_ceil(32);
180        let node_bytes = graph.bytes;
181        let root_bytes = u64::from(roots.max(1)) * 4;
182        let grouped = packing.support == TightSupport::Grouped;
183        let producer_bytes = producer_storage(producers, words, packing.support)?;
184        for bytes in [node_bytes, root_bytes, producer_bytes] {
185            buffer(bytes, packing.device)?;
186        }
187        let bytes = sum(&[node_bytes, root_bytes, producer_bytes])?;
188        let packing_bytes = sum(&[
189            bytes,
190            graph.rows.bytes()?,
191            if grouped { u64::from(words) * 4 } else { 0 },
192        ])?;
193        let mut shape = Self {
194            theory: theory.clone(),
195            atoms,
196            nodes,
197            roots,
198            edges: graph.edges,
199            wide_words: graph.wide_words,
200            levels: 0,
201            rows: graph.rows,
202            producers,
203            words,
204            node_bytes,
205            root_bytes,
206            producer_bytes,
207            bytes,
208            packing_bytes,
209            work: 0,
210            numeric_work: graph.numeric_work,
211            owners: Vec::new(),
212            owner_bytes: 0,
213            largest_producers: 0,
214            strided_depth: 0,
215            workgroup_size: crate::kernel::base_size(packing.workgroup_sizing),
216            dense_path: 0,
217            floor_path: 0,
218            support: packing.support,
219            workgroup_sizing: packing.workgroup_sizing,
220        };
221        shape.work = words
222            .checked_mul(2)
223            .and_then(|n| n.checked_add(nodes))
224            .and_then(|n| n.checked_add(shape.edges))
225            .and_then(|n| n.checked_add(roots))
226            .and_then(|n| n.checked_add(producers))
227            .and_then(|n| n.checked_add(graph.numeric_work))
228            .ok_or_else(|| capacity("tight complete-scan work exceeds u32"))?;
229        Ok(shape)
230    }

232    fn pack_nodes(
233        &self,
234        certificate: &TightPlan,
235        cancellation: &Cancellation,
236    ) -> Result<Vec<u32>, GpuError> {
237        if !self.theory.same_instance(certificate.theory()) {
238            return Err(GpuError::new(
239                GpuErrorKind::Seed,
240                "certificate belongs to another Theory",
241            ));
242        }
243        let mut nodes = words(self.node_bytes)?;
244        crate::formula_graph::pack(&self.theory, &self.rows, &mut nodes, cancellation)?;
245        Ok(nodes)
246    }

248    /// Complete an already-admitted serial shape. Optional level storage is
249    /// selected only when its complete cold layout and work also fit.
250    pub(super) fn prepare(
251        &mut self,
252        certificate: &TightPlan,
253        limits: TightGpuLimits,
254        device: &wgpu::Limits,
255        plan: &mut Plan,
256        cancellation: &Cancellation,
257    ) -> Result<Packed, GpuError> {
258        let mut nodes = self.pack_nodes(certificate, cancellation)?;
259        let depths =
260            dependency::label_depths(self.nodes, self.wide_words, &mut nodes, cancellation)?;
261        let levels = dependency::parallel_levels(self.nodes, depths);
262        if levels != 0 {
263            let serial = (self.root_bytes, self.bytes, self.packing_bytes);
264            let scheduled = (|| {
265                let root_bytes =
266                    u64::from(dependency::root_words(self.nodes, self.roots, levels)?) * 4;
267                buffer(root_bytes, device)?;
268                let added = root_bytes - self.root_bytes;
269                self.bytes = sum(&[self.bytes, added])?;
270                self.packing_bytes = sum(&[self.packing_bytes, added])?;
271                self.root_bytes = root_bytes;
272                self.levels = levels;
273                // Only a scheduled theory can update, so only it reserves the
274                // owner column of its live numeric rows and one transient word
275                // per atom for counting producers per head. Whether the route
276                // engages depends on the packed level widths; a narrow schedule
277                // releases both once packed.
278                self.owner_bytes = owner_bytes(self.rows.live_rows())?;
279                self.packing_bytes = sum(&[self.packing_bytes, u64::from(self.atoms) * 4])?;
280                Plan::new(self, plan.worlds as usize, limits, device, true, plan.epoch)
281            })();
282            match scheduled {
283                Ok(selected) if selected.levels != 0 => *plan = selected,
284                Ok(_) => {
285                    (self.root_bytes, self.bytes, self.packing_bytes) = serial;
286                    self.levels = 0;
287                    self.owner_bytes = 0;
288                }
289                Err(error) if error.kind() == GpuErrorKind::Capacity => {
290                    (self.root_bytes, self.bytes, self.packing_bytes) = serial;
291                    self.levels = 0;
292                    self.owner_bytes = 0;
293                }
294                Err(error) => return Err(error),
295            }
296        }
297        let packed = self.pack_columns(certificate, nodes, cancellation)?;
298        let widths = dependency::level_widths(&packed.roots, self.roots, self.levels)?;
299        self.workgroup_size =
300            crate::kernel::dispatch_size(self.workgroup_sizing, widths.clone(), device);
301        let size = self.workgroup_size.invocations();
302        self.strided_depth = widths.map(|width| u64::from(width.div_ceil(size))).sum();
303        // Only a schedule the update route engages needs its inputs, and the
304        // runtime guard's two paths at the size it dispatches. A narrow one
305        // releases what its admission reserved, and its plan, admitted with
306        // more bytes, accounts exactly what remains.
307        if update::engages(self.strided_depth, self.levels) {
308            self.owners = self.rows.owners(cancellation)?;
309            self.largest_producers = largest_producers(certificate, self.atoms, cancellation)?;
310            self.dense_path = update::dense_path(self, &packed, size, self.support, cancellation)?;
311            let offsets = &packed.roots[self.roots as usize..=(self.roots + self.levels) as usize];
312            self.floor_path = update::floor_path(self.words, offsets, size)
313                .ok_or_else(|| capacity("tight update floor path exceeds u32"))?;
314        } else if self.levels != 0 {
315            self.owner_bytes = 0;
316            self.packing_bytes -= u64::from(self.atoms) * 4;
317            *plan = Plan::new(self, plan.worlds as usize, limits, device, true, plan.epoch)?;
318        }
319        Ok(packed)
320    }

335    fn pack_columns(
336        &self,
337        certificate: &TightPlan,
338        mut nodes: Vec<u32>,
339        cancellation: &Cancellation,
340    ) -> Result<Packed, GpuError> {
341        let mut roots = words(self.root_bytes)?;
342        dependency::pack(
343            &self.theory,
344            self.nodes,
345            self.levels,
346            &nodes,
347            &mut roots,
348            cancellation,
349        )?;
350        // Depths are preparation scratch. A scheduled theory's header word 3
351        // holds each node's position in the level order instead, which the
352        // update route's queued set is indexed by; a serial one holds zero.
353        for header in nodes.chunks_exact_mut(4).take(self.nodes as usize) {
354            poll(cancellation)?;
355            header[3] = 0;
356        }
357        if self.levels != 0 {
358            let order = (self.roots + self.levels + 1) as usize;
359            for (position, &node) in roots[order..order + self.nodes as usize].iter().enumerate() {
360                poll(cancellation)?;
361                nodes[node as usize * 4 + 3] = address(position)?;
362            }
363        }
364        let producers = self.pack_producers(certificate, cancellation)?;
365        Ok(Packed {
366            nodes,
367            roots,
368            producers,
369        })
370    }

372    pub(super) fn pack_producers(
373        &self,
374        certificate: &TightPlan,
375        cancellation: &Cancellation,
376    ) -> Result<Vec<u32>, GpuError> {
377        let mut output = words(self.producer_bytes)?;
378        if self.support == TightSupport::Atomic {
379            for producer in certificate.producers() {
380                poll(cancellation)?;
381                output.extend(producer_record(producer)?);
382            }
383            if output.is_empty() {
384                output.extend([0; 4]);
385            }
386            return Ok(output);
387        }
388        let length = usize::try_from(self.producer_bytes / 4)
389            .map_err(|_| capacity("tight producers exceed host"))?;
390        output.resize(length, 0);
391        let offsets = self.producers as usize * 4;
392        // Counts occupy the next group's slot. Prefix summation produces W+1
393        // half-open offsets, including zero-length groups and the final bound P.
394        for producer in certificate.producers() {
395            poll(cancellation)?;
396            output[offsets + producer.head() / 32 + 1] += 1;
397        }
398        for word in 0..self.words as usize {
399            poll(cancellation)?;
400            let prefix = output[offsets + word];
401            output[offsets + word + 1] += prefix;
402        }
403        let mut cursors = vector(self.words as usize)?;
404        cursors.extend_from_slice(&output[offsets..offsets + self.words as usize]);
405        // Each canonical occurrence advances precisely its word's cursor once.
406        // Thus all occurrences are retained, in original order within a group.
407        for producer in certificate.producers() {
408            poll(cancellation)?;
409            let cursor = &mut cursors[producer.head() / 32];
410            let start = *cursor as usize * 4;
411            output[start..start + 4].copy_from_slice(&producer_record(producer)?);
412            *cursor += 1;
413        }
414        Ok(output)
415    }
416}

418/// The most producers of one head, with one transient counter per atom.
419fn largest_producers(
420    certificate: &TightPlan,
421    atoms: u32,
422    cancellation: &Cancellation,
423) -> Result<u32, GpuError> {
424    let mut heads = vector::<u32>(atoms as usize)?;
425    heads.resize(atoms as usize, 0);
426    let mut largest = 0;
427    for producer in certificate.producers() {
428        poll(cancellation)?;
429        let count = &mut heads[producer.head()];
430        *count = count
431            .checked_add(1)
432            .ok_or_else(|| capacity("tight producer count overflows"))?;
433        largest = largest.max(*count);
434    }
435    Ok(largest)
436}

438fn owner_bytes(rows: usize) -> Result<u64, GpuError> {
439    rows.checked_mul(size_of::<(u32, u32)>())

442}

444fn producer_record(producer: &zetesis_ferraris::TightProducer) -> Result<[u32; 4], GpuError> {
445    let (body, present) = match producer.body() {
446        Some(body) => (address(body)?, 1),
447        None => (0, 0),
448    };
449    Ok([address(producer.head())?, body, present, 0])
450}

516    /// Lay out a batch, with each slot's update state when `state` is given.
517    /// The state is part of the per-slot footprint the byte ceiling sees.
518    pub(super) fn with_state(
519        graph: &Graph,
520        worlds: usize,
521        limits: TightGpuLimits,
522        device: &wgpu::Limits,
523        fresh: bool,
524        epoch: u32,
525        state: Option<StateLayout>,
526    ) -> Result<Self, GpuError> {
527        let count = address(worlds)?;
528        crate::runtime::positive_timeout(limits.timeout)?;
529        if worlds > limits.max_candidates || count > device.max_compute_workgroups_per_dimension {
530            return Err(capacity("tight batch exceeds candidate/dispatch ceiling"));
531        }
532        if limits.max_work_per_candidate < u64::from(graph.work) || epoch == 0 {
533            return Err(capacity("tight full-scan work or epoch exceeds limits"));
534        }
535        // A resident schedule may be bypassed under a tighter work allowance.
536        // Its real retained storage is still charged below.
537        let (levels, work) = extend_work(graph.work, graph.levels, limits.max_work_per_candidate)
538            .map_or((0, graph.work), |work| (graph.levels, work));
539        let array = |length: u32| -> Result<u64, GpuError> {
540            let elements = u64::from(length) * u64::from(count);
541            if elements > u64::from(u32::MAX) {
542                return Err(capacity("tight world offset exceeds u32"));
543            }
544            Ok(elements.max(1) * 4)
545        };
546        let seeds = array(graph.words)?;
547        let truth = array(graph.nodes)?;
548        let support = array(graph.words)?;
549        let results = array(address(RESULT_WORDS)?)?;
550        let state_bytes = match state {
551            Some(layout) => array(layout.slot_words)?,
552            None => 0,
553        };
554        for bytes in [seeds, truth, support, results, state_bytes] {
555            buffer(bytes, device)?;
556        }
557        if device.max_uniform_buffer_binding_size < PARAM_BYTES
558            || device.max_buffer_size < PARAM_BYTES
559        {
560            return Err(capacity(
561                "tight uniform buffer exceeds granted device limits",
562            ));
563        }
564        let transport = sum(&[
565            PARAM_BYTES,
566            seeds,
567            truth,
568            support,
569            results,
570            results,
571            state_bytes,
572        ])?;
573        let host_results = u64::from(count) * std::mem::size_of::<TightGpuCheck>() as u64;
574        let (index_bytes, build_bytes) =
575            state.map_or((0, 0), |layout| (layout.index_bytes, layout.build_bytes));
576        let accounted = sum(&[
577            graph.bytes,
578            index_bytes,
579            graph.owner_bytes,
580            transport,
581            PARAM_BYTES,
582            seeds,
583            host_results,
584            if fresh { graph.packing_bytes } else { 0 },
585            build_bytes,
586        ])?;
587        if worlds != 0 && accounted > limits.max_batch_bytes {
588            return Err(capacity("tight batch exceeds authored byte ceiling"));
589        }
590        Ok(Self {
591            worlds: count,
592            seeds,
593            truth,
594            support,
595            results,
596            state: state_bytes,
597            transport,
598            accounted,
599            work,
600            allowance: work,
601            levels,
602            epoch,
603            original_truth_reused: false,
604            route: Route::default(),
605        })
606    }
607
608    /// Keep the already-admitted complete scan when optional change detection
609    /// cannot fit. The bound includes a header decision, old-value comparisons,
610    /// changed-bit publication, and every operand-change inspection; dense
611    /// changes can still require the entire original reduction afterwards.
612    pub(super) fn reuse_original_truth(&mut self, graph: &Graph, ceiling: u64) {
613        let work = graph
614            .nodes
615            .checked_mul(2)
616            .and_then(|n| n.checked_add(graph.edges))
617            .and_then(|n| n.checked_add(4))
618            .and_then(|extra| extend_work(self.work, extra, ceiling));
619        if let Some(work) = work {
620            self.work = work;
621            self.allowance = work;
622            self.original_truth_reused = true;
623        }
624    }
625
626    /// Decide whether this plan, which binds the update state of `index`,
627    /// allows the update route, and charge the routes accordingly. Returns why
628    /// the route is not allowed, if it is not.
629    ///
630    /// The route is allowed when the schedule is parallel, its reservation
631    /// fits `ceiling` (the larger of the update's worst case and dense reuse
632    /// after a change count, with the state's writes), and then, last, the
633    /// runtime guard `allows` it. Allowed, the complete and dense routes write
634    /// the slot's retained candidate and clear its summary, `words + 1`
635    /// operations; dense reuse after a change count adds the count's `2 *
636    /// words`, and with reuse authority every slot may take either route or
637    /// update, so each reserves the largest. Not allowed, for either reason, no
638    /// route writes the state and every charge is the stateless one, while the
639    /// layout stays state-inclusive and the state allocated: the batch runs the
640    /// dense twin, which leaves the state unbound.
641    pub(super) fn bind_state(
642        &mut self,
643        graph: &Graph,
644        index: &update::Index,
645        ceiling: u64,
646        allows: bool,
647    ) -> Option<TightGpuUpdateUnavailable> {
648        let worst = update::worst_case(
649            graph,
650            self.levels,
651            index.count_bits,
652            index.entries,
653            index.atom_entries,
654        );
655        let dense = update::dense_after_count(graph, self.levels);
656        let fits = self.levels != 0
657            && worst.is_some_and(|worst| worst.max(dense) <= ceiling.min(u64::from(u32::MAX)));
658        let allowed = fits && allows;
659        self.route = Route {
660            allowed,
661            threshold: update::threshold(graph.atoms),
662            readers: index.readers,
663            atom_nodes: index.atom_nodes,
664            count_bits: index.count_bits,
665            slot_words: index.slot_words,
666            floor: u32::try_from(update::floor(graph, self.levels)).unwrap_or(u32::MAX),
667            floor_path: graph.floor_path,
668        };
669        if allowed {
670            // The gate bounds every charge below by `ceiling` within u32.
671            self.work += graph.words + 1;
672            self.allowance = if self.original_truth_reused {
673                let counted = self.work + 2 * graph.words;
674                counted.max(
675                    worst
676                        .and_then(|worst| u32::try_from(worst).ok())
677                        .unwrap_or(counted),
678                )
679            } else {
680                self.work
681            };
682        }
683        if !fits {
684            Some(TightGpuUpdateUnavailable::Work)
685        } else if !allows {
686            Some(TightGpuUpdateUnavailable::GuardDeclined)
687        } else {
688            None
689        }
690    }
691
692    /// The kernel this plan runs: the full kernel exactly when the update
693    /// route is allowed, and otherwise its dense twin, which binds no update
694    /// state and carries none of the route's code. A theory whose route is
695    /// declined for its residency, a batch whose route a limit declines and a
696    /// batch the runtime guard declines therefore all run the twin; a plan
697    /// that holds the state keeps it allocated and accounted, unbound.
698    pub(super) const fn variant(&self) -> Variant {
699        if self.route.allowed {
700            Variant::Full
701        } else {
702            Variant::Dense
703        }
704    }
705
706    /// The parameter block. Words 9 and 15 are reserved: the kernel a batch
707    /// runs, not a parameter, says whether its update route is allowed.
708    pub(super) fn params(&self, graph: &Graph) -> [u32; 16] {
709        [
710            graph.atoms,
711            graph.nodes,
712            graph.roots,
713            graph.producers,
714            graph.words,
715            self.levels,
716            self.work,
717            self.epoch,
718            self.route.threshold,
719            0,
720            self.route.readers,
721            self.route.atom_nodes,
722            self.route.count_bits,
723            self.route.slot_words,
724            self.route.floor_path,
725            0,
726        ]
727    }
728
729    pub(super) fn pack(
730        &self,
731        graph: &Graph,
732        candidates: &[Interpretation],
733        staging: &mut crate::candidates::Staging,
734        ceiling: u64,
735        cancellation: &zetesis_algebra::Cancellation,
736    ) -> Result<u64, GpuError> {
737        if candidates.len() != self.worlds as usize {
738            staging.release();
739            return Err(capacity("tight candidate count differs from admitted plan"));
740        }
741        staging.pack(
742            &graph.theory,
743            candidates,
744            self.seeds,
745            self.accounted,
746            ceiling,
747            cancellation,
748        )
749    }
750}
751
752/// A batch's decoded checks with its route receipts.
753#[derive(Debug)]
754pub(super) struct Decoded {
755    pub(super) checks: Vec<TightGpuCheck>,
756    pub(super) routes: TightGpuRoutes,
757    pub(super) nodes_evaluated: u64,
758    /// The sum of the work each check reports performed.
759    pub(super) work: u64,
760    /// The largest path among the update slots; `None` without update slots.
761    pub(super) slowest_update_path: Option<u64>,
762}
763
764/// Validate every record against the batch and decode it in order.
765///
766/// A slot without reuse authority takes the complete route. With authority it
767/// takes dense reuse unless the update route is allowed; then a slot seeded by
768/// its previous record counts its change and updates, or takes dense reuse
769/// beyond the threshold. Complete and dense records charge the plan's fixed
770/// work, dense reuse beyond the threshold adds the count's `2 * words`, and none
771/// evaluates a node by update; an update charges at least its floor and at most
772/// the reservation, and evaluates at most every node. An update's path lies
773/// between the floor path and `path_ceiling`; every other route
774/// records a zero path. Every record is seeded exactly when the route was
775/// allowed, in the bit beside its route, and no other bit of that word is set.
776/// Any other record is an invalid readback.
777/// The largest path an update record that performed `performed` operations
778/// may carry: the floor path plus what the record counts beyond its floor.
779/// The path's phases are a subset of the work's: a summary counts in the work,
780/// and in the path only after the slot took dense reuse beyond the threshold;
781/// no phase's costliest invocation costs more than every invocation together;
782/// and no cost of the guard's table exceeds the work's charge for the same
783/// operation. Requires `performed` at least the floor.
784fn path_ceiling(route: &Route, performed: u32) -> u32 {
785    route.floor_path.saturating_add(performed - route.floor)
786}
787
788pub(super) fn decode(
789    words: &[u32],
790    graph: &Graph,
791    plan: &Plan,
792    seeds: &[u32],
793    cancellation: &Cancellation,
794) -> Result<Decoded, GpuError> {
795    let fail = || GpuError::new(GpuErrorKind::Readback, "invalid tight result record");
796    if words.len() != plan.worlds as usize * RESULT_WORDS
797        || u64::try_from(seeds.len())
798            .ok()
799            .and_then(|n| n.checked_mul(4))
800            != Some(plan.seeds)
801    {
802        return Err(fail());
803    }
804    let mut checks = vector(plan.worlds as usize)?;
805    let mut routes = TightGpuRoutes::default();
806    let mut nodes_evaluated = 0_u64;
807    let mut work = 0_u64;
808    let mut slowest_update_path: Option<u64> = None;
809    for (world, row) in words.chunks_exact(RESULT_WORDS).enumerate() {
810        poll(cancellation)?;
811        if row[0] != plan.epoch || row[1] as usize != world || row[5] != graph.result_marker() {
812            return Err(fail());
813        }
814        let (performed, evaluated, path) = (row[4], row[7], row[8]);
815        let (route, seeded) = (row[6] & !SEEDED, row[6] & SEEDED != 0);
816        let reused = plan.original_truth_reused;
817        let updating = reused && plan.route.allowed;
818        let receipt = match route {
819            ROUTE_COMPLETE => !reused && performed == plan.work && evaluated == 0 && path == 0,
820            ROUTE_DENSE => reused && performed == plan.work && evaluated == 0 && path == 0,
821            ROUTE_BEYOND_THRESHOLD => {
822                updating
823                    && plan.work.checked_add(2 * graph.words) == Some(performed)
824                    && evaluated == 0
825                    && path == 0
826            }
827            ROUTE_UPDATE => {
828                updating
829                    && evaluated <= graph.nodes
830                    && plan.route.floor <= performed
831                    && performed <= plan.allowance
832                    && plan.route.floor_path <= path
833                    && path <= path_ceiling(&plan.route, performed)
834            }
835            _ => false,
836        };
837        if !receipt || seeded != plan.route.allowed {
838            return Err(fail());
839        }
840        let verdict = match (row[2], row[3]) {
841            (STATUS_STABLE, 0) => TightVerdict::Stable,
842            (STATUS_NOT_MODEL, ordinal) if ordinal < graph.roots => TightVerdict::NotModel {
843                root: graph.theory.roots()[ordinal as usize],
844            },
845            (STATUS_RESIDUAL, atom)
846                if atom < graph.atoms
847                    && Words::<u32, _>::new(&seeds[world * graph.words as usize..])
848                        .contains(atom as usize) =>
849            {
850                TightVerdict::Residual {
851                    unsupported_atom: atom as usize,
852                }
853            }
854            _ => return Err(fail()),
855        };
856        match route {
857            ROUTE_COMPLETE => routes.complete += 1,
858            ROUTE_DENSE => routes.dense_reuse += 1,
859            ROUTE_BEYOND_THRESHOLD => {
860                routes.dense_reuse += 1;
861                routes.beyond_threshold += 1;
862            }
863            _ => routes.update += 1,
864        }
865        nodes_evaluated += u64::from(evaluated);
866        work += u64::from(performed);
867        if route == ROUTE_UPDATE {
868            slowest_update_path = slowest_update_path.max(Some(u64::from(path)));
869        }
870        checks.push(TightGpuCheck {
871            verdict,
872            work: u64::from(performed),
873            critical_path: u64::from(path),
874        });
875    }
876    Ok(Decoded {
877        checks,
878        routes,
879        nodes_evaluated,
880        work,
881        slowest_update_path,
882    })
883}
