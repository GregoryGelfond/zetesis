79    /// Replace the producer buffer with the records and the readers index, and
80    /// account the index as resident theory storage. The owner column is no
81    /// longer needed and is released. Returns the bytes uploaded.
82    pub(super) fn install_index(&mut self, device: &wgpu::Device, built: &Built) -> u64 {
83        let bytes = u64::try_from(built.words.len()).unwrap_or(u64::MAX) * 4;
84        self.producers = initialized(
85            device,
86            "tight producers and readers",
87            &built.words,
88            wgpu::BufferUsages::STORAGE,
89        );
90        self.producers_generation += 1;
91        let graph = &mut self.graph;
92        graph.bytes = graph.bytes - graph.producer_bytes + bytes;
93        graph.producer_bytes = bytes;
94        graph.owners = Vec::new();
95        graph.owner_bytes = 0;
96        self.updates = Stage::Built(built.index);
97        bytes
98    }

100    /// Write and submit the batch, and request its readback's mapping. The
101    /// submission is in flight when this returns; [`Self::finish`] waits for
102    /// it. An error leaves nothing in flight.
103    pub(super) fn submit(
104        &mut self,
105        execution: &Execution<'_>,
106        call: &Call<'_>,
107        activity: &mut TightGpuActivity,
108    ) -> Result<Submitted, GpuError> {
109        let Execution {
110            device,
111            queue,
112            pipeline,
113            variant,
114            cancellation,
115            timer,
116            ..
117        } = *execution;
118        let Call { seeds, plan } = *call;
119        let bindings = Bindings {
120            nodes: &self.nodes,
121            roots: &self.roots,
122            producers: &self.producers,
123            generation: self.producers_generation,
124        };
125        match self.transport.as_mut() {
126            Some(transport) => transport.conform(device, pipeline, variant, &bindings, plan),
127            None => {
128                self.transport = Some(Transport::new(device, pipeline, variant, &bindings, plan)?);
129            }
130        }
131        let transport = self
132            .transport
133            .as_mut()
134            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing tight transport"))?;
135        // Consume the host authority before the first queue write. Every error
136        // after this point leaves retained truth unavailable to the next call.
137        // Clearing the result hints also clears each slot's seeded record.
138        let retained = std::mem::take(&mut transport.original_truth_valid);
139        let clear_result = !retained || !plan.original_truth_reused;
140        poll(cancellation)?;
141        queue.write_buffer(
142            &transport.params,
143            0,
144            bytemuck::cast_slice(&plan.params(&self.graph)),
145        );
146        queue.write_buffer(&transport.seeds, 0, bytemuck::cast_slice(&seeds));
147        activity.uploaded_bytes += PARAM_BYTES + plan.seeds;
148        let submission = runtime::submit(
149            device,
150            queue,
151            &Dispatch {
152                command_label: "ranked support checking",
153                pass_label: "original truth and producer support",
154                pipeline,
155                group: &transport.group,
156                workgroups: [plan.worlds, 1, 1],
157                result: &transport.results,
158                readback: &transport.readback,
159                result_bytes: plan.results,
160                clear_result,
161                timing: timer,
162            },
163        );
164        activity.submissions = 1;
165        activity.submitted_candidates = u64::from(plan.worlds);
166        activity.scheduled_work = u64::from(plan.worlds) * u64::from(plan.allowance);
167        Ok(Submitted {
168            pending: runtime::request_read(&transport.readback, submission.index),
169            at: submission.at,
170        })
171    }

173    /// Wait for the submitted batch, observing the control as `observation`
174    /// says, then decode its readback. The transport is the one `submit` used:
175    /// nothing replaces it within one check.
176    pub(super) fn finish(
177        &self,
178        execution: &Execution<'_>,
179        submitted: Submitted,
180        observation: Observation,
181        call: &Call<'_>,
182        activity: &mut TightGpuActivity,
183        timing: &mut crate::timing::BatchTiming,
184    ) -> Result<runtime::Completion<Decoded>, GpuError> {
185        let Execution {
186            device,
187            timeout,
188            cancellation,
189            ..
190        } = *execution;
191        let Call { seeds, plan } = *call;
192        let transport = self
193            .transport
194            .as_ref()
195            .ok_or_else(|| GpuError::new(GpuErrorKind::Device, "missing tight transport"))?;
196        let mut performed = 0;
197        let result = runtime::finish_pending(
198            device,
199            &transport.readback,
200            submitted.pending,
201            timeout,
202            observation,
203            || poll(cancellation),
204            |words| {
205                let (words, tail) = crate::timing::split_readback(words, transport.timed)?;
206                let decoded = packing::decode(words, &self.graph, plan, &seeds, cancellation)?;
207                performed = decoded.work;
208                timing.tail = tail;
209                Ok(decoded)
210            },
211        );
212        timing.window = result.window_since(submitted.at);
213        if result.is_ok() {
214            activity.completed_candidates = u64::from(plan.worlds);
215            activity.completed_work = performed;
216            activity.downloaded_bytes = plan.results
217                + if transport.timed {
218                    crate::timing::READBACK_BYTES
219                } else {
220                    0
221                };
222        }
223        Ok(result)
224    }
225}

261pub(super) struct Transport {
262    worlds: u32,
263    params: wgpu::Buffer,
264    seeds: wgpu::Buffer,
265    truth: wgpu::Buffer,
266    support: wgpu::Buffer,
267    results: wgpu::Buffer,
268    readback: wgpu::Buffer,
269    /// The per-slot update state, when the plan holds one: bound by the full
270    /// kernel, and kept allocated but unbound by the dense twin when a limit or
271    /// the runtime guard declines the route of a batch that admitted it.
272    state: Option<wgpu::Buffer>,
273    /// The state's bytes; zero without state.
274    state_bytes: u64,
275    /// The kernel `group` binds for: the full kernel's layout has binding 8,
276    /// the dense twin's does not.
277    variant: Variant,
278    group: wgpu::BindGroup,
279    /// The producer buffer generation `group` binds.
280    generation: u64,
281    // The readback carries a timestamp tail after the logical results.
282    timed: bool,
283    original_truth_valid: bool,
284}
285impl Transport {
286    fn new(
287        device: &wgpu::Device,
288        pipeline: &wgpu::ComputePipeline,
289        variant: Variant,
290        bindings: &Bindings<'_>,
291        plan: &Plan,
292    ) -> Result<Self, GpuError> {
293        let timed = device.features().contains(wgpu::Features::TIMESTAMP_QUERY);
294        let readback_bytes =
295            crate::timing::readback_size(plan.results, timed, device.limits().max_buffer_size)?;
296        let params = buffer(
297            device,
298            "tight parameters",
299            PARAM_BYTES,
300            wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
301        );
302        let seeds = buffer(
303            device,
304            "tight candidates",
305            plan.seeds,
306            wgpu::BufferUsages::STORAGE | wgpu::BufferUsages::COPY_DST,
307        );
308        let truth = buffer(
309            device,
310            "original truth",
311            plan.truth,
312            wgpu::BufferUsages::STORAGE,
313        );
314        let support = buffer(
315            device,
316            "producer support",
317            plan.support,
318            wgpu::BufferUsages::STORAGE,
319        );
320        let results = buffer(
321            device,
322            "tight outcomes",
323            plan.results,
324            wgpu::BufferUsages::STORAGE
325                | wgpu::BufferUsages::COPY_SRC
326                | wgpu::BufferUsages::COPY_DST,
327        );
328        let readback = buffer(
329            device,
330            "tight readback",
331            readback_bytes,
332            wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
333        );
334        let state = state_buffer(device, plan.state);
335        let group = bind(
336            device,
337            pipeline,
338            bindings,
339            [&params, &seeds, &truth, &support, &results],
340            variant,
341            state.as_ref(),
342        );
343        Ok(Self {
344            worlds: plan.worlds,
345            params,
346            seeds,
347            truth,
348            support,
349            results,
350            readback,
351            state,
352            state_bytes: plan.state,
353            variant,
354            group,
355            generation: bindings.generation,
356            timed,
357            original_truth_valid: false,
358        })
359    }

361    /// Bring a retained transport to `plan`'s state, `variant`'s kernel and the
362    /// resident's current producer buffer, keeping its truth and result
363    /// authority. A state is added only where none was held, so the previous
364    /// records, written by the kernel without the route, say no slot is seeded;
365    /// a dropped state takes its retained evaluations with it, and this batch,
366    /// on that kernel, records every slot unseeded. A batch that keeps its
367    /// state while a limit or the runtime guard declines its route moves to the
368    /// kernel without the route and leaves the state allocated, unbound; every
369    /// record it writes says unseeded, so the state's contents bind no later
370    /// batch. Each change of state or kernel rebinds.
371    fn conform(
372        &mut self,
373        device: &wgpu::Device,
374        pipeline: &wgpu::ComputePipeline,
375        variant: Variant,
376        bindings: &Bindings<'_>,
377        plan: &Plan,
378    ) {
379        let kernel_changed = self.variant != variant;
380        let state_changed = self.state_bytes != plan.state;
381        if state_changed {
382            self.state = state_buffer(device, plan.state);
383            self.state_bytes = plan.state;
384        }
385        if state_changed || kernel_changed || self.generation != bindings.generation {
386            self.variant = variant;
387            self.group = bind(
388                device,
389                pipeline,
390                bindings,
391                [
392                    &self.params,
393                    &self.seeds,
394                    &self.truth,
395                    &self.support,
396                    &self.results,
397                ],
398                variant,
399                self.state.as_ref(),
400            );
401            self.generation = bindings.generation;
402        }
403    }
404}

406/// The update state of `bytes`, or none for a plan without state. Device
407/// buffers start zeroed, so every slot starts unseeded.
408fn state_buffer(device: &wgpu::Device, bytes: u64) -> Option<wgpu::Buffer> {
409    (bytes != 0).then(|| {
410        buffer(
411            device,
412            "tight update state",
413            bytes,
414            wgpu::BufferUsages::STORAGE,
415        )
416    })
417}

419/// The resident's and the transport's buffers as `pipeline`'s layout binds
420/// them: binding 8 holds `state` for the full kernel, which runs only with the
421/// route allowed and so with the state, and the dense twin's layout has no
422/// binding 8, whatever state the transport keeps.
423fn bind(
424    device: &wgpu::Device,
425    pipeline: &wgpu::ComputePipeline,
426    bindings: &Bindings<'_>,
427    [params, seeds, truth, support, results]: [&wgpu::Buffer; 5],
428    variant: Variant,
429    state: Option<&wgpu::Buffer>,
430) -> wgpu::BindGroup {
431    let mut entries = vec![
432        entry(0, params),
433        entry(1, bindings.nodes),
434        entry(2, bindings.roots),
435        entry(3, bindings.producers),
436        entry(4, seeds),
437        entry(5, truth),
438        entry(6, support),
439        entry(7, results),
440    ];
441    if let (Variant::Full, Some(state)) = (variant, state) {
442        entries.push(entry(8, state));
443    }
444    device.create_bind_group(&wgpu::BindGroupDescriptor {
445        label: Some("resident tight support"),
446        layout: &pipeline.get_bind_group_layout(0),
447        entries: &entries,
448    })
