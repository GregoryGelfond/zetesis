1//! The update route's host side: the readers index, its intrinsic bounds, the
2//! per-slot state's shape, the route's derived work, and the runtime guard
3//! that declines the route while its updates cost more than dense evaluation.
4//!
5//! A slot that holds reuse authority and whose state was seeded by an earlier
6//! dense or complete evaluation can bring that evaluation to a new candidate by
7//! re-evaluating, in dependency-level order, only the atom nodes of changed
8//! atoms and the readers of changed values. The readers index lists, for every
9//! node, the connectives and numeric-row owners reading it, the heads of the
10//! producers whose body it is, and the original root positions holding it; and
11//! for every atom, its atom nodes. It is built once per resident theory, at the
12//! first batch holding reuse authority, from the theory, the certificate's
13//! producers and roots, and the retained owner column of the cold numeric-row
14//! directory, and only for a schedule the route [`engages`].
15

22/// Result-record routes, shared with `check.wgsl`: the complete route, dense
23/// reuse, the update route, and dense reuse after a change count found the
24/// candidate's change beyond the threshold.
25pub(super) const ROUTE_COMPLETE: u32 = 0;
26pub(super) const ROUTE_DENSE: u32 = 1;
27pub(super) const ROUTE_UPDATE: u32 = 2;
28pub(super) const ROUTE_BEYOND_THRESHOLD: u32 = 3;
29/// A record's word 6 holds the route in its two low bits and, in this bit,
30/// whether the slot's update state was seeded.
31pub(super) const SEEDED: u32 = 4;
32/// Reader kinds in an entry's two high bits; the low 30 bits are its index.
33pub(super) const READER_NODE: u32 = 0;
34pub(super) const READER_PRODUCER: u32 = 1 << 30;
35pub(super) const READER_ROOT: u32 = 2 << 30;
36/// One more than the largest node ID, atom or root position an entry names.
37pub(super) const INDEX_LIMIT: u32 = 1 << 30;
38/// The per-slot summary block: summarized, and the false-root and unsupported
39/// counts. Whether a slot is seeded lives in its result record instead.
40pub(super) const SUMMARY_WORDS: u32 = 3;
41/// Words of the result record: epoch, slot, status, witness, work, marker,
42/// route with seeded, nodes evaluated and an update's path. Writing it is part
43/// of the update floor.
44pub(super) const RECORD_WORDS: u64 = 9;
45/// The most batches the runtime guard makes the route wait after a declining
46/// judgment: its backoff doubles from 1 up to this.
47pub(super) const GUARD_WAIT_LIMIT: u32 = 32;

49/// Workgroup strides per dependency level that dense evaluation must take, on
50/// average, before a theory builds the update route's readers index.
51///
52/// Dense evaluation of one candidate takes `I(W) = sum(ceil(width / W))`
53/// strided iterations over the level widths at the dispatched workgroup size
54/// `W`. An update keeps every level's synchronization, adds five per
55/// candidate (eight when it summarizes), and at each level evaluates each
56/// queued word's taken nodes, with their readers' queue bits, counts and root
57/// sets, in one invocation through storage atomics. It pays only where `I(W)`
58/// exceeds the level count several times. The factor covers the change
59/// patterns it was measured on, their per-level chains included: on an Apple
60/// M4 Pro GPU under Metal, scattered changes and clustered changes of up to
61/// 64 atoms with one reader per input. The runtime guard ([`Guard`]) bounds,
62/// per batch, the patterns beyond them, such as a change reaching many nodes
63/// of one queued word. The factor still decides whether a theory builds the
64/// index at all: it spares narrow theories the index, the state, the seeding
65/// batches and every losing probe, and covers the update's extra
66/// synchronization, which the guard cannot see. It is an execution choice and
67/// changes no verdict. Declining keeps the dense routes, so a factor set high
68/// forgoes gains, while one set low slows the schedules it engages.
69pub(super) const ENGAGEMENT_STRIDES: u64 = 32;
70
71/// Whether the update route engages for a resident schedule of `levels` levels
72/// whose dense strided depth at its workgroup size is `depth`.
73pub(super) const fn engages(depth: u64, levels: u32) -> bool {
74    levels != 0 && depth >= ENGAGEMENT_STRIDES * levels as u64
75}

77/// The device's large-change threshold: a slot whose candidate changed more
78/// atoms than this takes dense reuse. An execution choice, starting at the CPU
79/// update's rule; every route returns the same verdict.
80pub(super) const fn threshold(atoms: u32) -> u32 {
81    let eighth = atoms / 8;
82    if eighth > 64 { eighth } else { 64 }
83}

107fn vector<T>(length: usize) -> Result<Vec<T>, GpuError> {
108    let mut output = Vec::new();
109    output
110        .try_reserve_exact(length)
111        .map_err(|_| GpuError::new(GpuErrorKind::Allocation, "tight index reservation failed"))?;
112    Ok(output)
113}

131/// A theory's update shape, decided once from its intrinsic bounds: the
132/// count width, the per-slot state's words, and the readers index's bound.
133#[derive(Clone, Copy, Debug, PartialEq, Eq)]
134pub(super) struct Bounds {
135    pub(super) count_bits: u32,
136    pub(super) slot_words: u32,
137    /// Readers entries at most `edges + producers + roots`.
138    pub(super) entries: u32,
139    /// Atom-node entries at most `nodes`.
140    pub(super) atom_entries: u32,
141    /// Producer-buffer words before the index: records and grouped offsets.
142    pub(super) base: u32,
143}
144
145impl Bounds {
146    /// Decide the intrinsic bounds: every node ID, atom and root position fits
147    /// 30 bits; the producer buffer rebuilt with the index at its bound fits
148    /// `u32` with the kernel's stride headroom and both of the device's buffer
149    /// limits; and the per-slot state's word count fits `u32`, at the count
150    /// width the cold batch's producer count decided. `None` is a refusal that
151    /// holds for the resident theory, so a predictable overflow is refused
152    /// before any device effect. Constant time; allocates nothing.
153    pub(super) fn new(graph: &Graph, device: &wgpu::Limits) -> Option<Self> {
154        if ![graph.nodes, graph.atoms, graph.roots]
155            .into_iter()
156            .all(addressable)
157        {
158            return None;
159        }
160        let base = u32::try_from(graph.producer_bytes / 4).ok()?;
161        let entries = add(&[
162            u64::from(graph.edges),
163            u64::from(graph.producers),
164            u64::from(graph.roots),
165        ])?;
166        let words = add(&[
167            u64::from(base),
168            u64::from(graph.nodes) + 1,
169            entries,
170            u64::from(graph.atoms) + 1,
171            u64::from(graph.nodes),
172        ])?;
173        if !u32::try_from(words).is_ok_and(crate::kernel::fits_strided_loop)
174            || words * 4 > device.max_buffer_size
175            || words * 4 > device.max_storage_buffer_binding_size
176        {
177            return None;
178        }
179        let count_bits = count_bits(graph.largest_producers);
180        Some(Self {
181            count_bits,
182            slot_words: slot_words(graph, count_bits)?,
183            entries: u32::try_from(entries).ok()?,
184            atom_entries: graph.nodes,
185            base,
186        })
187    }
188
189    /// The index's resident bytes at its bound, admitted before it is built.
190    pub(super) fn index_bytes(&self, graph: &Graph) -> u64 {
191        u64::from(table_words(graph, self.entries, self.atom_entries)) * 4
192    }
193
194    /// Simultaneous host bytes of building the index: the repacked producer
195    /// records, the rebuilt producer buffer with its index at its bound, and
196    /// the per-node and per-atom counts and cursors.
197    pub(super) fn build_bytes(&self, graph: &Graph) -> u64 {
198        let counts = u64::from(graph.nodes) + 1 + u64::from(graph.atoms) + 1;
199        let cursors = u64::from(graph.nodes) + u64::from(graph.atoms);
200        let rebuilt =
201            u64::from(self.base) + u64::from(table_words(graph, self.entries, self.atom_entries));
202        (u64::from(self.base) + rebuilt + counts + cursors) * 4
203    }
204}

206/// Words of the readers index: node offsets and entries, then atom offsets and
207/// atom-node entries.
208fn table_words(graph: &Graph, entries: u32, atom_entries: u32) -> u32 {
209    // Bounds::new checked the bound's sum, which no actual table exceeds.
210    graph.nodes + 1 + entries + graph.atoms + 1 + atom_entries
211}

213/// Per-slot state words: the summary block, the retained candidate, the
214/// touched and unsupported atom sets, the false-root and queued sets, and the
215/// packed support counts. `None` when they exceed `u32`.
216pub(super) fn slot_words(graph: &Graph, count_bits: u32) -> Option<u32> {
217    let words = add(&[
218        u64::from(SUMMARY_WORDS),
219        3 * u64::from(graph.words),
220        u64::from(set_words(graph.roots)),
221        u64::from(set_words(graph.nodes)),
222        u64::from(count_words(graph.atoms, count_bits)),
223    ])?;
224    u32::try_from(words).ok()
225}

227/// Operations every update performs, whatever changed: the change count and
228/// the presence pass over both candidates' words, the settling scan, the
229/// queued-set scan of each level, and the record's words.
230pub(super) fn floor(graph: &Graph, levels: u32) -> u64 {
231    5 * u64::from(graph.words)
232        + u64::from(set_words(graph.nodes))
233        + u64::from(levels)
234        + RECORD_WORDS
235}

237/// Dense reuse after the update route's change count, with the state's writes:
238/// the dense charge, the count's `2 * words` and the retained candidate's
239/// `words + 1`. Both the index's build gate and every allowed plan reserve it.
240pub(super) fn dense_after_count(graph: &Graph, levels: u32) -> u64 {
241    u64::from(graph.work)
242        + u64::from(levels)
243        + 2 * u64::from(graph.nodes)
244        + u64::from(graph.edges)
245        + 4
246        + 3 * u64::from(graph.words)
247        + 1
248}

250/// The update route's derived worst case per candidate: the change count, a
251/// summary, presence of `min(threshold, atoms)` changed atoms with every
252/// atom-node entry, every node evaluated once with all of its reads and every
253/// readers entry visited, settling every atom, one verdict scan and the record.
254pub(super) fn worst_case(
255    graph: &Graph,
256    levels: u32,
257    count_bits: u32,
258    entries: u32,
259    atom_entries: u32,
260) -> Option<u64> {
261    let words = u64::from(graph.words);
262    let atoms = u64::from(graph.atoms);
263    let roots = set_words(graph.roots);
264    add(&[
265        2 * words,
266        summary(graph, count_bits)?,
267        atoms,
268        2 * words,
269        u64::from(threshold(graph.atoms)).min(atoms),
270        u64::from(atom_entries),
271        u64::from(set_words(graph.nodes)),
272        u64::from(levels),
273        u64::from(graph.nodes),
274        u64::from(graph.edges),
275        u64::from(graph.numeric_work),
276        u64::from(entries),
277        words,
278        atoms,
279        u64::from(roots).max(words),
280        RECORD_WORDS,
281    ])
282}

284/// A summary's fixed operations: clearing the counts, unsupported and
285/// false-root sets, then visiting every producer, root and retained word. The
286/// device adds one count read per present atom.
287pub(super) fn summary(graph: &Graph, count_bits: u32) -> Option<u64> {
288    add(&[
289        u64::from(count_words(graph.atoms, count_bits)),
290        2 * u64::from(graph.words),
291        u64::from(set_words(graph.roots)),
292        u64::from(graph.producers),
293        u64::from(graph.roots),
294    ])
295}

297/// The update path's floor at `size` invocations: what each invocation does of
298/// the update's uniform word scans, which no phase tallies, and the record.
299/// The change count's and presence's passes over both candidates' words and
300/// settling's scan are five strided loops over `words`; each level scans the
301/// queued words its positions span, `ceil(end / 32) - floor(start / 32)`, with
302/// one more operation; and invocation 0 writes the record's words. Linear in
303/// the levels. `None` when the offsets are malformed or a sum exceeds `u32`.
304pub(super) fn floor_path(words: u32, offsets: &[u32], size: u32) -> Option<u32> {
305    let share = |count: u64| count.div_ceil(u64::from(size));
306    let mut path = 5 * share(u64::from(words)) + RECORD_WORDS;
307    for level in offsets.windows(2) {
308        let (start, end) = (u64::from(level[0]), u64::from(level[1]));
309        let spanned = end.div_ceil(32).checked_sub(start / 32)?;
310        path = path.checked_add(share(spanned) + 1)?;
311    }
312    u32::try_from(path).ok()
313}

315/// Dense reuse's least cost of evaluating packed node `index`, by the guard's
316/// cost table: an atom, the falsum or a constant 1; a pair or implication 3,
317/// the node and both operands; a wide conjunction or disjunction 2, the node
318/// and the one operand that may end its reduction; a numeric row's owner
319/// `n + g`, the change-bit reads and guard clears of a row none of whose `n`
320/// conditions changed, since evaluating its `g` guards costs at least
321/// `2 + n + g`; and a row's follower 1, its ownership check. `None` for a
322/// malformed row.
323fn dense_cost(nodes: &[u32], node_count: u32, index: u32) -> Option<u64> {
324    let start = index as usize * 4;
325    let header = nodes.get(start..start + 3)?;
326    Some(match header[0] {
327        2..=4 => 3,
328        5 | 6 => 2,
329        7..=9 => {
330            let tail = node_count as usize * 4;
331            let row = tail + *nodes.get(tail + header[1] as usize)? as usize + 3;
332            if *nodes.get(row - 3)? == index {
333                u64::from(header[2]) + u64::from(*nodes.get(row - 2)?)
334            } else {
335                1
336            }
337        }
338        _ => 1,
339    })
340}

342/// The dense path at `size` invocations, by the guard's cost table: a lower
343/// bound on dense reuse's busiest invocation's cost, summed over its phases.
344/// Each level costs its bounds and its busiest invocation's nodes, where dense
345/// reuse gives invocation `lane` the positions `start + lane + j * size`; then
346/// support construction: atomic support clears the support words, then visits
347/// the roots and producers in one phase, while grouped support visits the
348/// roots and each support word with its group's producers in one phase; then
349/// the unsupported scan, two words per candidate word, and the record. Where
350/// dense reuse's cost depends on the candidate, each node takes its least
351/// ([`dense_cost`]). One pass over the schedule's positions, the producer
352/// groups and buffers of `size` sums: `O(nodes + producers + roots + words)`.
353///
354/// # Errors
355/// Reports cancellation and a failed reservation; a malformed schedule is a
356/// capacity failure, which packing never produces.
357pub(super) fn dense_path(
358    graph: &Graph,
359    packed: &Packed,
360    size: u32,
361    support: TightSupport,
362    cancellation: &Cancellation,
363) -> Result<u64, GpuError> {
364    let malformed = || capacity("tight schedule offsets exceed the packed schedule");
365    let cost = |node: u32| dense_cost(&packed.nodes, graph.nodes, node).ok_or_else(malformed);
366    let mut path = levels_path(graph, &packed.roots, size, cost, cancellation)?;
367    let share = |count: u32| u64::from(count.div_ceil(size));
368    path += u64::from(graph.levels);
369    path += match support {
370        TightSupport::Atomic => share(graph.words) + share(graph.roots) + share(graph.producers),
371        TightSupport::Grouped => {
372            // Invocation `lane` visits the roots `lane + j * size` and writes
373            // the support words `lane + j * size`, each after visiting the
374            // word's producers.
375            let start = graph.producers as usize * 4;
376            let groups = packed
377                .producers
378                .get(start..=start + graph.words as usize)
379                .ok_or_else(malformed)?;
380            let lanes = size as usize;
381            let mut sums = vector::<u64>(lanes)?;
382            sums.extend(
383                (0..size).map(|lane| u64::from(graph.roots.saturating_sub(lane).div_ceil(size))),
384            );
385            for (word, bounds) in groups.windows(2).enumerate() {
386                poll(cancellation)?;
387                sums[word % lanes] += 1 + u64::from(bounds[1] - bounds[0]);
388            }
389            sums.iter().copied().max().unwrap_or(0)
390        }
391    };
392    Ok(path + 2 * share(graph.words) + RECORD_WORDS)
393}

395/// Dense evaluation's busiest invocation's `cost`, summed over the dependency
396/// levels of the packed schedule `roots` (root IDs, then the levels' offsets,
397/// then the positions' node IDs) at `size` invocations: invocation `lane` takes
398/// a level's positions `start + lane + j * size`. With unit cost it is the
399/// strided depth. One pass over the positions with `size` sums.
400///
401/// # Errors
402/// Reports cancellation, a failed reservation, and a malformed schedule or a
403/// failing `cost`.
404pub(super) fn levels_path(
405    graph: &Graph,
406    roots: &[u32],
407    size: u32,
408    mut cost: impl FnMut(u32) -> Result<u64, GpuError>,
409    cancellation: &Cancellation,
410) -> Result<u64, GpuError> {
411    let malformed = || capacity("tight schedule offsets exceed the packed schedule");
412    let start = graph.roots as usize;
413    let order = start + graph.levels as usize + 1;
414    let offsets = roots.get(start..order).ok_or_else(malformed)?;
415    let lanes = size as usize;
416    let mut sums = vector::<u64>(lanes)?;
417    sums.resize(lanes, 0);
418    let mut path = 0_u64;
419    for level in offsets.windows(2) {
420        let positions = roots
421            .get(order + level[0] as usize..order + level[1] as usize)
422            .ok_or_else(malformed)?;
423        let used = positions.len().min(lanes);
424        sums[..used].fill(0);
425        for (offset, &node) in positions.iter().enumerate() {
426            poll(cancellation)?;
427            sums[offset % lanes] += cost(node)?;
428        }
429        path += sums[..used].iter().copied().max().unwrap_or(0);
430    }
431    Ok(path)
432}

434/// Whether the runtime guard declines the route after a batch whose slowest
435/// update slot's path is `slowest`, against dense reuse's path `dense`: that
436/// update costs more than dense reuse does at least. Equality keeps the route.
437pub(super) const fn declines(slowest: u64, dense: u64) -> bool {
438    slowest > dense
439}

441/// The runtime guard of a resident theory, which decides whether the batches
442/// planned next may update. It compares critical paths by one cost table: the
443/// slowest update slot's path, the costliest invocation of each phase after
444/// the summary summed, and of the summary's phases too when the slot took
445/// dense reuse beyond the threshold before it, against dense reuse's least
446/// path for one candidate. It is in one of two states. Engaged, a batch with
447/// update slots is judged, and one that loses declines the route for the next
448/// `backoff` successfully decoded batches, the backoff doubling up to
449/// [`GUARD_WAIT_LIMIT`]; it never shrinks while the theory stays resident.
450/// Declined, every decoded batch counts the wait down and is not judged; at
451/// zero the guard is engaged again. The batch planned next then seeds every
452/// slot, since every record written while the guard declined says unseeded,
453/// and the batch after it updates and is judged. A failed or cancelled batch
454/// is never settled, so it is not judged and consumes no wait. The guard
455/// changes which route evaluates a batch, never a verdict.
456#[derive(Clone, Copy, Debug, PartialEq, Eq)]
457pub(super) struct Guard {
458    /// Successfully decoded batches left to run with the route not allowed;
459    /// zero while engaged.
460    waiting: u32,
461    /// The wait the next declining judgment sets.
462    backoff: u32,
463}
464
465impl Guard {
466    /// An engaged guard with a backoff of one batch.
467    pub(super) const fn new() -> Self {
468        Self {
469            waiting: 0,
470            backoff: 1,
471        }
472    }
473
474    /// Whether the batch planned next may set the route allowed: whether the
475    /// guard is engaged.
476    pub(super) const fn allows(self) -> bool {
477        self.waiting == 0
478    }
479
480    /// Settle one successfully decoded batch, after its reuse authority is
481    /// re-established. Its slowest update slot's path is `slowest`, or `None`
482    /// without update slots, against dense reuse's path `dense`. Declined, the
483    /// batch counts the wait down; engaged, a batch with update slots is
484    /// judged.
485    pub(super) fn settle(&mut self, slowest: Option<u64>, dense: u64) {
486        if self.waiting != 0 {
487            self.waiting -= 1;
488        } else if slowest.is_some_and(|slowest| declines(slowest, dense)) {
489            self.waiting = self.backoff;
490            self.backoff = (self.backoff * 2).min(GUARD_WAIT_LIMIT);
491        }
492    }
493}

495/// A resident theory's update admission, decided at most once per step.
496#[derive(Clone, Copy, Debug, PartialEq, Eq)]
497pub(super) enum Stage {
498    /// The theory evaluates original truth serially, which has no levels.
499    Serial,
500    /// Dense evaluation takes fewer than [`ENGAGEMENT_STRIDES`] strides per
501    /// level at the resident's workgroup size, so the rule declines the route.
502    Narrow,
503    /// No batch holding reuse authority at its stateless count has decided the
504    /// bounds yet.
505    Pending,
506    /// The bounds hold; the index awaits a batch that admits it.
507    Bounded(Bounds),
508    /// The readers index is resident.
509    Built(Index),
510    /// An intrinsic bound refused the index, while the theory stays resident.
511    Refused,
512}
513
514impl Stage {
515    /// The stage a theory starts in once its schedule is packed: a serial
516    /// schedule has no levels to update by, and the engagement rule declines
517    /// a narrow one.
518    pub(super) const fn of(graph: &Graph) -> Self {
519        if graph.levels == 0 {
520            Self::Serial
521        } else if engages(graph.strided_depth, graph.levels) {
522            Self::Pending
523        } else {
524            Self::Narrow
525        }
526    }
527
528    /// The theory's bounds at a batch holding reuse authority: decided once by
529    /// `decide` and kept. A refusal is kept too, so no later batch decides the
530    /// bounds again or builds. A serial or built stage has no bounds to decide.
531    pub(super) fn bounds(&mut self, decide: impl FnOnce() -> Option<Bounds>) -> Option<Bounds> {
532        match *self {
533            Self::Bounded(bounds) => Some(bounds),
534            Self::Pending => {
535                let decided = decide();
536                *self = decided.map_or(Self::Refused, Self::Bounded);
537                decided
538            }
539            Self::Serial | Self::Narrow | Self::Built(_) | Self::Refused => None,
540        }
541    }
542}

563/// Build the readers index after `records`, the packed producer records and
564/// grouped offsets. Entries name, for every node, the connectives reading it,
565/// the owner of every live numeric row holding it as a condition (one entry
566/// per occurrence), the head of every producer whose body it is, and the root
567/// positions holding it; and, for every atom, its atom nodes. One counting
568/// pass and one filling pass, each `O(nodes + edges + producers + roots)`.
569///
570/// # Errors
571/// Reports cancellation, a failed reservation, or a table beyond its bounds.
572pub(super) fn build(
573    graph: &Graph,
574    certificate: &TightPlan,
575    owners: &[(u32, u32)],
576    bounds: Bounds,
577    records: &[u32],
578    cancellation: &Cancellation,
579) -> Result<Built, GpuError> {
580    let nodes = graph.nodes as usize;
581    let atoms = graph.atoms as usize;
582    let mut readers = vector::<u32>(nodes + 1)?;
583    readers.resize(nodes + 1, 0);
584    let mut atom_nodes = vector::<u32>(atoms + 1)?;
585    atom_nodes.resize(atoms + 1, 0);
586    // Counts occupy the next node's slot; prefix sums turn them into
587    // half-open offsets ending at the entry count.
588    visit(
589        graph,
590        certificate,
591        owners,
592        cancellation,
593        |slot, _| match slot {
594            Slot::Reader(node) => readers[node + 1] += 1,
595            Slot::Atom(atom) => atom_nodes[atom + 1] += 1,
596        },
597    )?;
598    prefix(&mut readers, cancellation)?;
599    prefix(&mut atom_nodes, cancellation)?;
600    let entries = readers[nodes];
601    let atom_entries = atom_nodes[atoms];
602    if entries > bounds.entries || atom_entries > bounds.atom_entries {
603        return Err(capacity("tight readers index exceeds its admitted bound"));
604    }
605    let base = u32::try_from(records.len()).map_err(|_| capacity("tight producers exceed u32"))?;
606    let reader_start = base + graph.nodes + 1;
607    let atom_offsets = reader_start + entries;
608    let total = atom_offsets as usize + atoms + 1 + atom_entries as usize;
609    let mut words = vector::<u32>(total)?;
610    words.extend_from_slice(records);
611    words.extend_from_slice(&readers);
612    words.resize(atom_offsets as usize, 0);
613    words.extend_from_slice(&atom_nodes);
614    words.resize(total, 0);
615    let mut reader_cursors = vector::<u32>(nodes)?;
616    reader_cursors.extend_from_slice(&readers[..nodes]);
617    let mut atom_cursors = vector::<u32>(atoms)?;
618    atom_cursors.extend_from_slice(&atom_nodes[..atoms]);
619    drop(readers);
620    drop(atom_nodes);
621    let atom_start = atom_offsets as usize + atoms + 1;
622    visit(graph, certificate, owners, cancellation, |slot, entry| {
623        let (start, cursor) = match slot {
624            Slot::Reader(node) => (reader_start as usize, &mut reader_cursors[node]),
625            Slot::Atom(atom) => (atom_start, &mut atom_cursors[atom]),
626        };
627        words[start + *cursor as usize] = entry;
628        *cursor += 1;
629    })?;
630    Ok(Built {
631        words,
632        index: Index {
633            readers: base,
634            atom_nodes: atom_offsets,
635            entries,
636            atom_entries,
637            count_bits: bounds.count_bits,
638            slot_words: bounds.slot_words,
639        },
640    })
641}

643/// Where an index entry belongs: the readers list of a node, or the atom-node
644/// list of an atom.
645#[derive(Clone, Copy)]
646enum Slot {
647    Reader(usize),
648    Atom(usize),
649}
650
651/// Pass every readers entry and every atom-node entry to `sink`, in one fixed
652/// order.
653fn visit(
654    graph: &Graph,
655    certificate: &TightPlan,
656    owners: &[(u32, u32)],
657    cancellation: &Cancellation,
658    mut sink: impl FnMut(Slot, u32),
659) -> Result<(), GpuError> {
660    // Bounds were checked before: every node ID, atom and root position fits
661    // an entry's 30 bits, so these conversions cannot fail.
662    let entry = |index: usize| {
663        u32::try_from(index)
664            .ok()
665            .filter(|index| *index < INDEX_LIMIT)
666            .ok_or_else(|| capacity("tight readers index exceeds 30 bits"))
667    };
668    let theory = &graph.theory;
669    let view = theory.view();
670    for index in 0..graph.nodes as usize {
671        poll(cancellation)?;
672        let entry = READER_NODE | entry(index)?;
673        match view.node(index).expect("admitted node") {
674            NodeView::And(children) | NodeView::Or(children) => {
675                for &child in children {
676                    sink(Slot::Reader(child), entry);
677                }
678            }
679            NodeView::Implies(left, right) => {
680                sink(Slot::Reader(left), entry);
681                sink(Slot::Reader(right), entry);
682            }
683            NodeView::Atom(atom) => sink(Slot::Atom(atom), entry),
684            NodeView::False | NodeView::Threshold(_) | NodeView::ExtremumCut(_) => {}
685        }
686    }
687    let parts = theory.parts();
688    for &(source, owner) in owners {
689        poll(cancellation)?;
690        let row = parts.rows()[source as usize];
691        for &condition in &parts.conditions()[row.start..row.start + row.length] {
692            sink(Slot::Reader(condition), READER_NODE | owner);
693        }
694    }
695    for producer in certificate.producers() {
696        poll(cancellation)?;
697        if let Some(body) = producer.body() {
698            sink(
699                Slot::Reader(body),
700                READER_PRODUCER | entry(producer.head())?,
701            );
702        }
703    }
704    for (position, &root) in theory.roots().iter().enumerate() {
705        poll(cancellation)?;
706        sink(Slot::Reader(root), READER_ROOT | entry(position)?);
707    }
708    Ok(())
709}
710
711fn prefix(counts: &mut [u32], cancellation: &Cancellation) -> Result<(), GpuError> {
712    let mut total = 0_u32;
713    for count in counts {
714        poll(cancellation)?;
715        total = total
716            .checked_add(*count)
717            .ok_or_else(|| capacity("tight readers index exceeds u32"))?;
718        *count = total;
719    }
720    Ok(())
721}
