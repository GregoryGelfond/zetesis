1//! Optional compute-pass timestamps, owned by one device oracle for its lifetime.
2//!
3//! A timer has four query slots used as two pairs, and one resolve buffer per
4//! pair. Each timed batch writes the pair that is not pending through its single
5//! compute pass. After that pass, the batch's wgpu command buffer records the
6//! resolve of the pending pair, written by an earlier queue submission whose
7//! completion the host has already observed, into that pair's resolve buffer,
8//! and a copy of it into the batch's readback tail. Because the pair was written
9//! outside this command buffer, wgpu-core defers the resolve to a hal command
10//! buffer of its own at submission and records the copy in another, so the
11//! batch's one queue submission carries two more hal command buffers (each an
12//! `MTLCommandBuffer` on Metal) than an untimed batch's one.
13//!
14//! The pair a batch writes is read by the oracle's next submitting call, or by
15//! a flush. One batch is in flight per oracle at a time, so two pairs suffice;
16//! an asynchronous schedule would need one more pair per batch in flight and
17//! resolution in completion order. A resolve that writes nothing leaves its
18//! pair's buffer holding that pair's last-read values, or zero when fresh, never
19//! the other pair's interval; validation refuses both.
20
21use crate::{GpuError, GpuErrorKind};
22use std::time::Duration;
23use zetesis_backend::BatchComputeTime;

25/// Two resolved timestamps, appended to a transport's mapped result readback or
26/// read by a flush.
27pub(crate) const READBACK_BYTES: u64 = 16;
28/// Logical timing payload of one timed batch: the oracle's four query slots,
29/// two resolve buffers and flush readback, plus the transport's readback tail.
30/// Query-slot allowance is not a measurement of driver-private allocation.
31pub(crate) const OVERHEAD_BYTES: u64 = 6 * READBACK_BYTES;
32
33pub(crate) fn readback_size(results: u64, timed: bool, maximum: u64) -> Result<u64, GpuError> {
34    results
35        .checked_add(if timed { READBACK_BYTES } else { 0 })
36        .filter(|bytes| *bytes <= maximum)
37        .ok_or_else(|| {
38            GpuError::new(
39                GpuErrorKind::Capacity,
40                "compute readback exceeds device buffer capacity",
41            )
42        })
43}

45/// Separate a timed readback's logical words from its 16-byte timestamp tail.
46/// The tail holds whatever pair the batch resolved; only the timer knows whether
47/// it resolved one.
48pub(crate) fn split_readback(
49    words: &[u32],
50    timed: bool,
51) -> Result<(&[u32], Option<[u64; 2]>), GpuError> {
52    if !timed {
53        return Ok((words, None));
54    }
55    let boundary = words.len().checked_sub(4).ok_or_else(|| {
56        GpuError::new(
57            GpuErrorKind::Readback,
58            "compute timestamp readback is truncated",
59        )
60    })?;
61    let (results, tail) = words.split_at(boundary);
62    Ok((results, Some(timestamps(tail))))
63}
64
65fn timestamps(words: &[u32]) -> [u64; 2] {
66    [
67        u64::from(words[0]) | (u64::from(words[1]) << 32),
68        u64::from(words[2]) | (u64::from(words[3]) << 32),
69    ]
70}

72/// What one timed batch's readback establishes for its oracle's timer: the
73/// resolved tail and the host window of the batch's own submission.
74#[derive(Debug, Default)]
75pub(crate) struct BatchTiming {
76    /// The pair the batch resolved after its pass, when its readback decoded.
77    pub(crate) tail: Option<[u64; 2]>,
78    /// From the instant before the submission to its observed completion.
79    pub(crate) window: Option<Duration>,
80}
81
82/// The report of a nonempty call that reached submission: under enabled timing
83/// the timer's (the previous batch, or nothing for a fresh timer); under
84/// unsupported timing the call's own completed batch, unavailable; nothing when
85/// timing was not requested. A call that submitted nothing reports nothing.
86///
87/// A call that fails after submission also retires the timer. Its own pair was
88/// written and never read, so a later batch writing that pair could otherwise
89/// resolve the abandoned interval if its pass wrote nothing; a fresh timer's
90/// unwritten slots resolve to zero instead, which rule 2 refuses.
91pub(crate) fn settle_call(
92    status: zetesis_backend::ComputeTimingStatus,
93    timer: &mut Option<ComputeTimer>,
94    completed: bool,
95    timing: &BatchTiming,
96    period_ns: f32,
97) -> Option<BatchComputeTime> {
98    if completed {
99        return match timer {
100            Some(active) => active.completed(timing, period_ns),
101            None => (status == zetesis_backend::ComputeTimingStatus::Unsupported)
102                .then_some(BatchComputeTime::Unavailable),
103        };
104    }
105    retire(timer)
106}

108/// Release a timer whose pending batch can no longer be read, reporting that
109/// batch unavailable: after a failure following submission, or when the
110/// oracle clears its residency.
111pub(crate) fn retire(timer: &mut Option<ComputeTimer>) -> Option<BatchComputeTime> {
112    timer
113        .take()
114        .and_then(|mut retired| retired.ledger.abandon())
115}

117/// One of a timer's two timestamp pairs: slots {0, 1} or {2, 3}.
118#[derive(Clone, Copy, Debug, PartialEq, Eq)]
119enum Pair {
120    First,
121    Second,
122}
123
124impl Pair {
125    const fn other(self) -> Self {
126        match self {
127            Self::First => Self::Second,
128            Self::Second => Self::First,
129        }
130    }
131
132    const fn begin(self) -> u32 {
133        match self {
134            Self::First => 0,
135            Self::Second => 2,
136        }
137    }
138
139    // Each pair resolves into its own buffer. A resolve destination's offset must
140    // be a multiple of `wgpu::QUERY_RESOLVE_BUFFER_ALIGNMENT` (256 bytes), so two
141    // 16-byte buffers replace the halves of one 32-byte buffer.
142    const fn resolve_buffer(self) -> usize {
143        match self {
144            Self::First => 0,
145            Self::Second => 1,
146        }
147    }
148}

150/// A completed batch whose pair has not been resolved yet.
151#[derive(Clone, Copy, Debug)]
152struct Pending {
153    pair: Pair,
154    window: Duration,
155}
156
157/// The timing protocol without device state: which pair a batch writes and
158/// resolves, and which batch each resolved interval reports.
159///
160/// Invariant: a pair is resolved only after the host observed completion of
161/// the submission that wrote it, and the pair a batch writes is never the
162/// pending one. Each completed batch becomes pending and is reported once:
163/// by the next completing call or flush from its own interval, or as
164/// unavailable when a failure or release abandons it.
165#[derive(Debug, Default)]
166pub(crate) struct PairLedger {
167    pending: Option<Pending>,
168    // The value last resolved from each slot, whatever its outcome.
169    history: [Option<u64>; 4],
170}
171
172impl PairLedger {
173    fn written(&self) -> Pair {
174        self.pending
175            .map_or(Pair::First, |pending| pending.pair.other())
176    }
177
178    fn resolving(&self) -> Option<Pair> {
179        self.pending.map(|pending| pending.pair)
180    }
181
182    pub(crate) const fn pending(&self) -> bool {
183        self.pending.is_some()
184    }
185
186    /// A submitting call completed. Report the pending batch from the tail its
187    /// command buffer resolved; this call's batch, whose host window is
188    /// `window`, becomes pending.
189    fn completed(&mut self, timing: &BatchTiming, period_ns: f32) -> Option<BatchComputeTime> {
190        let written = self.written();
191        let report = self
192            .pending
193            .take()
194            .map(|pending| self.report(pending, timing.tail, period_ns));
195        self.pending = Some(Pending {
196            pair: written,
197            // A completed call always observed its completion; without a
198            // window no later interval of this batch can be accepted.
199            window: timing.window.unwrap_or_default(),
200        });
201        report
202    }
203
204    /// A flush resolved the pending pair into `tail`.
205    fn flushed(&mut self, tail: [u64; 2], period_ns: f32) -> Option<BatchComputeTime> {
206        self.pending
207            .take()
208            .map(|pending| self.report(pending, Some(tail), period_ns))
209    }
210
211    /// The pending batch can no longer be read: a submitting call or a flush
212    /// failed after submission, or the timer is released. A failed call's own
213    /// batch did not complete; its pair is abandoned with it.
214    fn abandon(&mut self) -> Option<BatchComputeTime> {
215        self.pending.take().map(|_| BatchComputeTime::Unavailable)
216    }
217
218    fn report(
219        &mut self,
220        pending: Pending,
221        tail: Option<[u64; 2]>,
222        period_ns: f32,
223    ) -> BatchComputeTime {
224        let Some([begin, end]) = tail else {
225            return BatchComputeTime::Unavailable;
226        };
227        let slot = pending.pair.begin() as usize;
228        let previous = [self.history[slot], self.history[slot + 1]];
229        self.history[slot] = Some(begin);
230        self.history[slot + 1] = Some(end);
231        interval(begin, end, previous, period_ns, pending.window)
232            .map_or(BatchComputeTime::Unavailable, BatchComputeTime::Measured)
233    }
234}

236/// The duration a resolved pair establishes, when every rule holds:
237///
238/// 1. neither value is all ones, Metal's failed-sample value;
239/// 2. the begin is nonzero, since an unwritten slot of a fresh query set
240///    resolves to zero;
241/// 3. neither value repeats its slot's last resolved value, which would mean
242///    the slot was not rewritten (this holds whether or not the counter wraps);
243/// 4. the end follows the begin, and
244/// 5. the period and the converted duration are valid ([`duration`]);
245/// 6. the duration is no longer than the host window of the submission that
246///    wrote the pair. This rejects stale begins and an overstated period without
247///    a shared clock domain; durations trust the adapter's period from below.
248fn interval(
249    begin: u64,
250    end: u64,
251    previous: [Option<u64>; 2],
252    period_ns: f32,
253    window: Duration,
254) -> Option<Duration> {
255    if begin == u64::MAX || end == u64::MAX || begin == 0 {
256        return None;
257    }
258    if previous[0] == Some(begin) || previous[1] == Some(end) {
259        return None;
260    }
261    duration(begin, end, period_ns).filter(|elapsed| *elapsed <= window)
262}
263
264// wgpu timestamps can wrap or fail to resolve a positive interval. Equal
265// readings cannot distinguish a sub-resolution interval from an unwritten
266// counter, so they do not establish zero compute cost. A reversed interval is
267// also unavailable: guessing its device-specific counter width would manufacture
268// a duration. Invalid timing never changes an otherwise valid logical result.
269fn duration(begin: u64, end: u64, period_ns: f32) -> Option<Duration> {
270    if !period_ns.is_finite() || period_ns <= 0.0 {
271        return None;
272    }
273    let ticks = end.checked_sub(begin)?;
274    if ticks == 0 {
275        return None;
276    }
277    Duration::try_from_secs_f64(Duration::from_nanos(ticks).as_secs_f64() * f64::from(period_ns))
278        .ok()
279        .filter(|elapsed| !elapsed.is_zero())
280}

282/// The timestamp storage and pair ledger one device oracle owns, from its first
283/// timed batch until it clears its residency, fails after a submission or drops.
284/// Count, theory, cold preparation and workgroup-size changes leave it in place:
285/// the pending pair belongs to the oracle's sequence of batches, not to a
286/// transport.
287pub(crate) struct ComputeTimer {
288    queries: wgpu::QuerySet,
289    resolved: [wgpu::Buffer; 2],
290    flush: wgpu::Buffer,
291    ledger: PairLedger,
292}
293
294impl ComputeTimer {
295    /// Allocate four query slots, a resolve buffer per pair and a flush
296    /// readback on a device that enabled timestamp queries.
297    pub(crate) fn new(device: &wgpu::Device) -> Self {
298        let resolve = |label| {
299            crate::runtime::buffer(
300                device,
301                label,
302                READBACK_BYTES,
303                wgpu::BufferUsages::QUERY_RESOLVE | wgpu::BufferUsages::COPY_SRC,
304            )
305        };
306        Self {
307            queries: device.create_query_set(&wgpu::QuerySetDescriptor {
308                label: Some("compute pass timestamps"),
309                ty: wgpu::QueryType::Timestamp,
310                count: 4,
311            }),
312            resolved: [
313                resolve("first pair's resolved compute timestamps"),
314                resolve("second pair's resolved compute timestamps"),
315            ],
316            flush: crate::runtime::buffer(
317                device,
318                "flushed compute timestamps",
319                READBACK_BYTES,
320                wgpu::BufferUsages::COPY_DST | wgpu::BufferUsages::MAP_READ,
321            ),
322            ledger: PairLedger::default(),
323        }
324    }
325
326    /// The next batch's compute-pass writes: the pair that is not pending.
327    pub(crate) fn writes(&self) -> wgpu::ComputePassTimestampWrites<'_> {
328        let begin = self.ledger.written().begin();
329        wgpu::ComputePassTimestampWrites {
330            query_set: &self.queries,
331            beginning_of_pass_write_index: Some(begin),
332            end_of_pass_write_index: Some(begin + 1),
333        }
334    }
335
336    /// After the batch's pass, record the resolve of the pending pair into that
337    /// pair's own resolve buffer and its copy into `readback` at `offset`.
338    /// Without a pending pair the tail is left unwritten and ignored.
339    pub(crate) fn resolve(
340        &self,
341        encoder: &mut wgpu::CommandEncoder,
342        readback: &wgpu::Buffer,
343        offset: u64,
344    ) {
345        if let Some(pair) = self.ledger.resolving() {
346            let begin = pair.begin();
347            let resolved = &self.resolved[pair.resolve_buffer()];
348            encoder.resolve_query_set(&self.queries, begin..begin + 2, resolved, 0);
349            encoder.copy_buffer_to_buffer(resolved, 0, readback, offset, READBACK_BYTES);
350        }
351    }
352
353    /// Whether a completed batch is still unresolved: its pair is pending.
354    pub(crate) const fn unresolved(&self) -> bool {
355        self.ledger.pending()
356    }
357
358    /// A submitting call completed: see [`PairLedger`].
359    pub(crate) fn completed(
360        &mut self,
361        timing: &BatchTiming,
362        period_ns: f32,
363    ) -> Option<BatchComputeTime> {
364        self.ledger.completed(timing, period_ns)
365    }
366
367    /// Submit a resolve-only command for the pending pair and wait at most
368    /// `timeout` for it, polling no caller control: a control firing during
369    /// this bounded observation must not turn it into an interrupted submission.
370    /// The caller holds the context lease and the operation's error scopes.
371    pub(crate) fn flush(
372        &self,
373        device: &wgpu::Device,
374        queue: &wgpu::Queue,
375        timeout: Duration,
376    ) -> crate::runtime::Completion<[u64; 2]> {
377        let mut encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
378            label: Some("compute timestamp flush"),
379        });
380        self.resolve(&mut encoder, &self.flush, 0);
381        let submission = queue.submit([encoder.finish()]);
382        crate::runtime::read_polled(
383            device,
384            &self.flush,
385            submission,
386            timeout,
387            || Ok(()),
388            |words| {
389                if words.len() == 4 {
390                    Ok(timestamps(words))
391                } else {
392                    Err(GpuError::new(
393                        GpuErrorKind::Readback,
394                        "compute timestamp flush readback has the wrong size",
395                    ))
396                }
397            },
398        )
399    }
400
401    /// A flush read the pending pair.
402    pub(crate) fn flushed(&mut self, tail: [u64; 2], period_ns: f32) -> Option<BatchComputeTime> {
403        self.ledger.flushed(tail, period_ns)
404    }
405}
